    import win32com.client as win32
    
    
    #====================================================================================================================================
    # SETTINGS
    #====================================================================================================================================
    TEST_LENGTH = 1.0
    TEST_Y = 20.0
    
    CABLE_SHIELD_X = 20.0
    DEFAULT_SHIELD_X = 23.0
    TWIST_X = 26.0
    
    
    #====================================================================================================================================
    # CONNECT TO E3
    #====================================================================================================================================
    app = win32.gencache.EnsureDispatch("CT.Application")
    prj = app.CreateJobObject()
    sht = prj.CreateSheetObject()
    sym = prj.CreateSymbolObject()
    
    
    #====================================================================================================================================
    # FIND SHEET 1
    #====================================================================================================================================
    sheet_info = prj.GetAllSheetIds(None)
    sheet_ids = [value for value in sheet_info[1] if value is not None]
    sheet_1_id = None
    
    for sheet_id in sheet_ids:
        sht.SetId(sheet_id)
    
        if sht.GetName() == "1":
            sheet_1_id = sheet_id
            break
    
    if sheet_1_id is None:
        raise Exception("Could not find sheet 1")
    
    
    #====================================================================================================================================
    # TEST ONE REAL LIBRARY SYMBOL
    #====================================================================================================================================
    def test_symbol(symbol_name, x):
        print("")
        print("============================================================")
        print("TESTING:", symbol_name)
        print("============================================================")
    
        load_result = sym.Load(symbol_name, 1)
    
        print("LOAD RESULT:", load_result)
    
        if load_result == 0:
            print("LOAD FAILED")
            return
    
        placed_id = sym.Place(sheet_1_id, x, TEST_Y, 0, 1.0, 0)
    
        print("PLACE RESULT:", placed_id)
    
        if placed_id == 0:
            print("PLACE FAILED")
            return
    
        sym.SetId(placed_id)
    
        try:
            symbol_type = sym.GetSymbolType()
        except Exception:
            symbol_type = None
    
        try:
            is_shield = sym.IsShield()
        except Exception:
            is_shield = None
    
        try:
            is_twisted = sym.IsTwisted()
        except Exception:
            is_twisted = None
    
        try:
            is_bundle = sym.IsBundle()
        except Exception:
            is_bundle = None
    
        try:
            before = sym.GetDisplayLength()
        except Exception:
            before = None
    
        print("PLACED ID:", placed_id)
        print("SYMBOL TYPE:", symbol_type)
        print("IS SHIELD:", is_shield)
        print("IS TWISTED:", is_twisted)
        print("IS BUNDLE:", is_bundle)
        print("DISPLAY LENGTH BEFORE:", before)
    
        stretch_result = sym.SetDisplayLength(TEST_LENGTH)
    
        print("SET DISPLAY LENGTH RESULT:", stretch_result)
    
        sym.SetId(placed_id)
    
        try:
            after = sym.GetDisplayLength()
        except Exception:
            after = None
    
        print("REQUESTED LENGTH:", TEST_LENGTH)
        print("DISPLAY LENGTH AFTER:", after)
    
        if after is not None and abs(float(after) - TEST_LENGTH) < 0.001:
            print("STRETCH SUCCESS")
        else:
            print("STRETCH FAILED")
    
    
    #====================================================================================================================================
    # TEST CABLE SHIELD
    #====================================================================================================================================
    test_symbol("Cable Shield", CABLE_SHIELD_X)
    
    
    #====================================================================================================================================
    # TEST PROJECT DEFAULT SHIELD
    #====================================================================================================================================
    test_symbol("Shield_Demo", DEFAULT_SHIELD_X)
    
    
    #====================================================================================================================================
    # TEST KNOWN WORKING TWIST
    #====================================================================================================================================
    test_symbol("MIL_Twist_ArrowsDown", TWIST_X)
    
    
    #====================================================================================================================================
    # CLEANUP
    #====================================================================================================================================
    del sym
    del sht
    del prj
    del app
