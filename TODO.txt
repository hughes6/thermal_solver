import sys
import traceback
from tkinter import ttk

DB_TRANSACTION_OPEN = False
DB_TRANSACTION_DONE = False
E3_COMPONENT_CREATED = False


def describe_cable_error(error):
    raw = str(error).strip()
    msg = raw.lower()

    if "model check" in msg:
        return ("Excel Model Check Failed",
                "The workbook did not pass its model check in cell F20.",
                "Fix the workbook checks, recalculate and save Excel, then try again.")

    if "layout count" in msg or "part count" in msg:
        return ("Connector Count Mismatch", raw,
                "Check connector parts in B3:B14 against positions in W4:W11 and Z4:Z11.")

    if "duplicate connector" in msg:
        return ("Duplicate Connector", raw,
                "Each connector in the W/Z layout needs a unique designation.")

    if "pin " in msg and "not found on" in msg:
        return ("Connector Pin Not Found", raw,
                "Verify the Excel pin number and the actual physical pin in the E3 connector library.")

    if "connector not found" in msg:
        return ("Connector Part Not Found", raw,
                "Check the connector part number and confirm version 1 exists in E3.")

    if "could not create component" in msg:
        return ("Cable Assembly Could Not Be Created", raw,
                "Check whether the name/version is already used and whether the cable template exists.")

    if "componentdata" in msg or "active component id" in msg:
        return ("E3 Component Lookup Failed", raw,
                "Check whether E3 saved the new component and whether the database is accessible.")

    if "wire definitions require" in msg:
        return ("Wire Definitions Do Not Match", raw,
                "Check B15/B16, wire section quantities and the L/N/O/Q pinout table.")

    if isinstance(error, IndexError):
        return ("Missing Pinout Rows", "There are fewer usable pinout rows than the wire definitions require.",
                "Check wire section quantities and both sides of the Excel pinout table.")

    if isinstance(error, KeyError):
        return ("Unknown Connector", f"Connector reference not found: {raw}",
                "Check that every connector in L/Q is also listed in W/Z.")

    if isinstance(error, (ValueError, TypeError)):
        return ("Invalid Workbook Value", raw or "A required value is missing or invalid.",
                "Check B15/B16, wire quantities and pin assignments. See Technical Details for the line.")

    if isinstance(error, FileNotFoundError):
        return ("Excel File Not Found", raw,
                "Select an existing workbook and check that it has not been moved.")

    if isinstance(error, PermissionError):
        return ("File Access Denied", raw,
                "Check the file permissions or whether another program has locked the file.")

    if isinstance(error, pyodbc.Error):
        return ("Database Error", "The SQL Server operation failed.",
                "Check network/database access and your permissions. See Technical Details for SQL's message.")

    if "com_error" in type(error).__name__.lower():
        return ("Zuken E3 Error", "E3/DBE could not complete the requested operation.",
                "Check E3 and DBE status. See Technical Details for the COM error code.")

    return ("Cable Maker Error", raw or "An unexpected error interrupted Cable Maker.",
            "Review the workbook and technical details. Cable Maker has stopped.")


def show_cable_error(error, details, db_note=""):
    title, what, fix = describe_cable_error(error)
    root = tk.Tk()
    root.title("Cable Maker - Error")
    root.geometry("560x340")
    root.resizable(False, False)
    root.attributes("-topmost", True)

    frame = ttk.Frame(root, padding=20)
    frame.pack(fill="both", expand=True)

    ttk.Label(frame, text=title, font=("Segoe UI", 16, "bold"),
              foreground="#B42318").pack(anchor="w", pady=(0, 14))
    ttk.Label(frame, text="What happened:",
              font=("Segoe UI", 10, "bold")).pack(anchor="w")
    ttk.Label(frame, text=what, wraplength=510,
              justify="left").pack(anchor="w", pady=(3, 11))
    ttk.Label(frame, text="Suggested fix:",
              font=("Segoe UI", 10, "bold")).pack(anchor="w")
    ttk.Label(frame, text=fix, wraplength=510,
              justify="left").pack(anchor="w", pady=(3, 10))

    if db_note:
        ttk.Label(frame, text=db_note, wraplength=510,
                  foreground="#92600E").pack(anchor="w", pady=(0, 8))

    buttons = ttk.Frame(frame)
    buttons.pack(side="bottom", fill="x")

    details_frame = ttk.Frame(frame)
    error_text = tk.Text(details_frame, font=("Consolas", 9),
                         wrap="word", height=10)
    scrollbar = ttk.Scrollbar(details_frame, command=error_text.yview)
    error_text.config(yscrollcommand=scrollbar.set)
    scrollbar.pack(side="right", fill="y")
    error_text.pack(side="left", fill="both", expand=True)
    error_text.insert("1.0", details)
    error_text.config(state="disabled")

    expanded = False

    def toggle():
        nonlocal expanded
        expanded = not expanded
        if expanded:
            root.geometry("560x570")
            details_frame.pack(fill="both", expand=True,
                               pady=(8, 12), before=buttons)
            detail_button.config(text="Hide Technical Details")
        else:
            details_frame.pack_forget()
            root.geometry("560x340")
            detail_button.config(text="Show Technical Details")

    def copy_error():
        root.clipboard_clear()
        root.clipboard_append(details)
        root.update()

    detail_button = ttk.Button(buttons, text="Show Technical Details", command=toggle)
    detail_button.pack(side="left")
    ttk.Button(buttons, text="Copy Error", command=copy_error).pack(side="left", padx=8)
    ttk.Button(buttons, text="OK", command=root.destroy).pack(side="right")
    root.protocol("WM_DELETE_WINDOW", root.destroy)
    root.mainloop()


def cable_error_handler(error_type, error, tb):
    if issubclass(error_type, (KeyboardInterrupt, SystemExit)):
        sys.__excepthook__(error_type, error, tb)
        return

    details = "".join(traceback.format_exception(error_type, error, tb))
    print(details, file=sys.stderr)
    note = ""

    connection = globals().get("conn")
    if connection is not None and DB_TRANSACTION_OPEN and not DB_TRANSACTION_DONE:
        try:
            connection.rollback()
            note = "Uncommitted SQL changes were rolled back."
        except Exception as rollback_error:
            note = "WARNING: SQL rollback could not be confirmed."
            details += f"\nRollback error: {rollback_error!r}\n"
        try:
            connection.close()
        except Exception:
            pass

    if E3_COMPONENT_CREATED:
        note += " An E3 component created earlier may still exist."

    try:
        show_cable_error(error, details, note.strip())
    except Exception:
        sys.__excepthook__(error_type, error, tb)


sys.excepthook = cable_error_handler
