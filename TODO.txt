def log(message=""):

def normalize_pin(value):
    if value is None:
        return None
    if isinstance(value, float) and math.isfinite(value) and value.is_integer():
        return str(int(value))
    val = str(value).strip()
    return val or None


def parse_id_result(value):
    """Support E3's (count, ID-array) COM return and normal ID arrays."""
    if value is None:
        return []
    if isinstance(value, (tuple, list)):
        if (len(value) >= 2 and isinstance(value[1], (tuple, list))
                and isinstance(value[0], (int, float))):
            ids = value[1]
            try:
                expected_count = int(value[0])
                ids = list(ids)
                if expected_count < len(ids):
                    # COM wrappers can prepend output/status values.
                    ids = ids[-expected_count:] if expected_count > 0 else []
            except (ValueError, TypeError):
                pass
        else:
            ids = value
    else:
        ids = [value]
    result = []
    for v in ids:
        try:
            i = int(v)
            if i > 0 and i != COM_MISSING:
                result.append(i)
        except (TypeError, ValueError):
            pass
    return result


def get_ids(method, *arguments):
    # E3's GetSymbolIds(None, 0) uses an output array BEFORE a mode argument;
    # GetViewIds(None), GetAllSheetIds(None) and some wrapper versions differ.
    attempts = ((None, *arguments), arguments, (*arguments, None))
    for args in attempts:
        try:
            value = parse_id_result(method(*args))
            if value:
                return value
        except Exception:
            continue
    return []


def select_id(obj, object_id):
    """Do not mistake E3 versions returning None from SetId for failure."""
    try:
        result = obj.SetId(int(object_id))
        if isinstance(result, (int, float)) and result == 0:
            return False
        if hasattr(obj, "GetId"):
            current = obj.GetId()
            if current is not None and int(current) != int(object_id):
                return False
        return True
    except Exception:
        return False


def optional_call(obj, name):
    try:
        return getattr(obj, name)()
    except Exception:
        return None


def get_symbol_location(sym):
    try:
        raw = sym.GetSchemaLocation(None, None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 3:
            return None
        return {"sheet_id": int(raw[0]), "x": float(raw[1]), "y": float(raw[2])}
    except Exception:
        return None


def get_symbol_area(sym):
    try:
        raw = sym.GetPlacedArea(None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 5 or int(raw[0]) <= 0:
            return None
        l, b, r, t = map(float, raw[1:5])
        if not (r > l and t > b):
            return None
        return {"left": l, "bottom": b, "right": r, "top": t}
    except Exception:
        return None


def inside_block(loc, block):
    tol = BLOCK_LOCATION_TOLERANCE
    return (loc is not None and loc["sheet_id"] == block["sheet_id"]
            and block["left"] - tol <= loc["x"] <= block["right"] + tol
            and block["bottom"] - tol <= loc["y"] <= block["top"] + tol)


def scan_sheet_symbols(prj, sht, sheet_id):
    if not select_id(sht, sheet_id):
        raise RuntimeError(f"Could not select E3 sheet ID {sheet_id}.")
    ids = get_ids(sht.GetSymbolIds)
    if ids:
        return ids
    log(f"  Sheet {sheet_id} has no sheet-local symbol ID array; scanning project IDs instead.")
    return get_ids(prj.GetSymbolIds)

        import ctypes
        key=ctypes.windll.user32.GetAsyncKeyState
        previous=bool(key(0x0D)&0x8000)
        armed=not previous
        def poll():
            nonlocal previous, armed
            if done['value']: return
            try:
                down=bool(key(0x0D)&0x8000)
                if not down: armed=True

        pass
    try:
        root.mainloop()
        return result['action']
    finally:
        root.destroy()

def selected_symbol_id_candidates(prj, sht, app):
    """Read current E3 GUI selection, WITHOUT changing GUI selection.

    Different COM versions expose selection at job, sheet or application level;
    each getter may use an out parameter or return an array directly. We use
    only explicit *Selected* getter names and never auto-pick from sheet IDs.
    """
    active_sheet_id = None
    for method_name in ('GetActiveSheetId',):
        try:
            sheet_id = getattr(prj, method_name)()
            if isinstance(sheet_id, (int, float)) and int(sheet_id) > 0:
                active_sheet_id = int(sheet_id)
                break
        except Exception:
            continue
    if active_sheet_id is not None:
        select_id(sht, active_sheet_id)
    log(f'Active E3 sheet: {active_sheet_id if active_sheet_id else "not reported by COM"}')

    # Explicitly symbol-only queries first, generic selected object IDs last.
    # The fallback generic query is ALWAYS type-checked as a placed CABBLOCK.
    probes = (
        ('job', prj, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedSymbolsIds'),
        ('job', prj, 'GetSelectedSymbolId'),
        ('sheet', sht, 'GetSelectedSymbolIds'),
        ('sheet', sht, 'GetSelectedSymbolsIds'),
        ('sheet', sht, 'GetSelectedSymbolId'),
        ('app', app, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedObjectIds'),
        ('job', prj, 'GetSelectedIds'),
        ('sheet', sht, 'GetSelectedObjectIds'),
        ('sheet', sht, 'GetSelectedIds'),
        ('app', app, 'GetSelectedObjectIds'),
        ('app', app, 'GetSelectedIds'),
    )
    attempted = []
    found = set()
    for label, obj, method_name in probes:
        try:
            getter = getattr(obj, method_name)
        except Exception:
            continue
        # Avoid evaluating an integer method return as the selected ID unless
        # GetSelectedSymbolId explicitly returns one number. For array-style
        # getters, require an array (or a count + array) in the result.
        for args in ((None,), (), (None, 0), (0, None)):
            try:
                result = getter(*args)
            except Exception as exc:
                attempted.append(f'{label}.{method_name}{args}: {str(exc)[:75]}')
                continue
            if isinstance(result, (tuple, list)):
                ids = parse_id_result(result)
            elif method_name == 'GetSelectedSymbolId':
                ids = parse_id_result(result)
            else:
                ids = []
            attempted.append(f'{label}.{method_name}{args}: {ids}')
            if ids:
                found.update(ids)
                break
    log('GUI SELECTION READBACK:')
    for item in attempted[:45]:
        log(f'  {item}')
    if len(attempted) > 45:
        log(f'  ... {len(attempted)-45} additional getter attempts')
    if not found:
        raise RuntimeError(
            'No selected symbol IDs were returned by this E3 COM build. '
            'Select one or more CABBLOCK borders in E3 before pressing Enter. '
            'If it is already selected, send the GUI SELECTION READBACK log; '
            'your E3 version may use another selection API. NO CHANGES MADE.'
        )
    return sorted(found)


def get_selected_cabblocks(prj, sht, app):
    """Return ALL uniquely selected CABBLOCKs; ignore selected non-blocks.

    A drag/box selection can include pins, lines or text as well as the blocks.
    These are never used as targets. At least one placed CABBLOCK is required.
    """
    ids = selected_symbol_id_candidates(prj, sht, app)
    sym = prj.CreateSymbolObject()
    other_symbols = []
    blocks = []
    for sid in ids:
        if not select_id(sym, sid):
            log(f'  Selected ID {sid}: not a valid E3 symbol ID; ignored')
            continue
        kind = str(optional_call(sym, 'GetSymbolTypeName') or '')
        if 'CABBLOCK' not in kind.upper():
            other_symbols.append((sid, kind))
            continue
        location, area = get_symbol_location(sym), get_symbol_area(sym)
        if location is None or area is None:
            raise RuntimeError(
                f'Selected CABBLOCK {sid}: missing placed location/area. '
                'Batch cancelled: NO CHANGES MADE.'
            )
        block = {
            'id': sid, 'sheet_id': location['sheet_id'], 'kind': kind,
            'owner': '<selected block>', 'view_id': None, 'view_number': None,
            **area,
        }
        blocks.append(block)
    if not blocks:
        raise RuntimeError(
            f'No placed CABBLOCK found among {len(ids)} selected E3 ID(s). '
            f'Non-block symbols: {other_symbols[:12]}. '
            'Click one or more CABBLOCK borders and press Enter again. '
            'NO CHANGES MADE.'
        )
    blocks.sort(key=lambda b: (b['sheet_id'], -b['top'], b['left'], b['id']))
    log(f'SELECTION: {len(blocks)} selected CABBLOCK(s); '
        f'{len(other_symbols)} selected non-block symbol(s) ignored.')
    for b in blocks:
        log(f'  BLOCK {b["id"]}: sheet {b["sheet_id"]} | {b["kind"]!r} | '
            f'X=[{b["left"]:.3f},{b["right"]:.3f}] '
            f'Y=[{b["bottom"]:.3f},{b["top"]:.3f}]')
    if other_symbols:
        log(f'  IGNORED non-block selections (first 15): {other_symbols[:15]}')
    return blocks

def text_type_hint(txt):
    hints = []
    for getter in ('GetTypeName', 'GetTextTypeName', 'GetTypeDescription',
                   'GetAttributeName', 'GetName', 'GetSource'):
        val = optional_call(txt, getter)
        if isinstance(val, str) and val.strip():
            hints.append(val.strip())
    return ' | '.join(dict.fromkeys(hints))[:180]


def get_text_xy(text, block):
    """Read E3 SCHEMATIC coordinates; do not assume a symbol-relative value.

    The existing signal mover confirmed E3 text supports GetSchemaLocation.
    Only accept a decoded position reasonably close to the parent CABBLOCK.
    That protects from treating status codes/sheet IDs as coordinates.
    """
    probes = (
        (None, None, None, None, None),
        (None, None, None),
        (None, None),
        (),
    )
    sheet = int(block['sheet_id'])
    options = []
    raw_preview = []
    for args in probes:
        try:
            raw = text.GetSchemaLocation(*args)
        except Exception:
            continue
        raw_preview.append(repr(raw)[:140])
        if not isinstance(raw, (tuple, list)):
            continue
        nums = []
        for number in raw:
            try:
                nums.append(float(number))
            except (TypeError, ValueError):
                nums.append(None)
        candidates = []
        if len(nums) >= 3 and nums[0] is not None and int(nums[0]) == sheet:
            candidates.append((0, nums[1], nums[2]))
        if len(nums) >= 4 and nums[1] is not None and int(nums[1]) == sheet:
            candidates.append((0, nums[2], nums[3]))
        if len(nums) >= 3 and nums[0] in (-1, 0, 1):
            candidates.append((2, nums[1], nums[2]))
        if len(nums) >= 2:
            candidates.append((3, nums[0], nums[1]))
        for priority, x, y in candidates:
            if x is None or y is None or not (math.isfinite(x) and math.isfinite(y)):
                continue
            if (block['left'] - 2 <= x <= block['right'] + 2 and
                    block['bottom'] - 2 <= y <= block['top'] + 2):
                options.append((priority, x, y))
        if options and min(op[0] for op in options) == 0:
            break
    if not options:
        raise RuntimeError(f"Block {block['id']}: cannot read absolute schematic "
                           f"text position. GetSchemaLocation returned: {raw_preview[:3]}. "
                           'No E3 changes made.')
    options.sort(key=lambda item: item[0])
    best_priority = options[0][0]
    distinct = {(round(x, 4), round(y, 4)) for p, x, y in options if p == best_priority}
    if len(distinct) > 1:
        raise RuntimeError(f"Block {block['id']}: multiple possible coordinates "
                           f"from E3 text getter: {sorted(distinct)}. No changes made.")
    return {'x': options[0][1], 'y': options[0][2]}


def inspect_block_texts(prj, blocks):
    """Read only texts owned by the selected CABBLOCK symbols (not their pins)."""
    all_texts = {}
    sym = prj.CreateSymbolObject()
    txt = prj.CreateTextObject()
    for block in blocks:
        if not select_id(sym, block['id']):
            raise RuntimeError(f"Selected block {block['id']} disappeared.")
        ids = get_ids(sym.GetTextIds)
        entries = defaultdict(list)
        for tid in ids:
            if not select_id(txt, tid):
                continue
            try:
                typ = int(txt.GetType())
            except Exception:
                continue
            raw_val = optional_call(txt, 'GetText')
            value = '' if raw_val is None else str(raw_val)
            hint = text_type_hint(txt)
            entries[typ].append({'id': tid, 'type': typ, 'value': value, 'hint': hint})
        all_texts[block['id']] = entries
        log(f"BLOCK {block['id']} (sheet {block['sheet_id']}): "
            f"{len(ids)} attached text IDs; {len(entries)} text types")
        for typ in sorted(entries):
            for e in entries[typ]:
                log(f"  TYPE {typ:<5} ID {e['id']:<10} "
                    f"VALUE={e['value'][:120]!r} HINT={e['hint'][:90]!r}")
    return all_texts

def missing_header_nodes(blocks, all_texts, mapping):
    """Read-only list of missing/duplicate linked text nodes on original blocks."""
    issues = []
    for b in blocks:
        for field, typ in mapping.items():
            count = len(all_texts.get(b['id'], {}).get(typ, []))
            if count != 1:
                issues.append({
                    'id': b['id'], 'field': field, 'type': typ, 'count': count,
                    'left': b['left'], 'right': b['right'],
                    'bottom': b['bottom'], 'top': b['top'],
                })
    return issues


def guided_insert_missing_text(prj, blocks, mapping):
    """Assist with E3 GUI insertion, then recheck the EXACT originally selected blocks.

    Does not synthesize or insert an unrelated E3 graphic text and does not
    operate E3 UI controls. Users insert true property-linked text through E3.
    """
    import tkinter as tk
    from tkinter import ttk, messagebox

    root = tk.Tk()
    root.title('E3 - Attach missing linked header text')
    root.attributes('-topmost', True)
    root.geometry('720x510')
    root.minsize(620, 390)
    result = {'ready': False}
    frm = ttk.Frame(root, padding=14)
    frm.pack(fill='both', expand=True)
    ttk.Label(frm, text='Linked header text is missing on one or more blocks',
              font=('Segoe UI', 13, 'bold')).pack(anchor='w', pady=(0, 10))
    ttk.Label(frm, text=(
        'The value XXX exists, but SetVisibility cannot display an absent text node.\n'
        'For each missing item, select ONLY that CABBLOCK in E3, then choose:\n'
        'Insert > Attribute Text > the missing attribute > Bind to selected Cell.\n'
        'Place the label anywhere for now; this script positions it afterward.\n'
        'Use Check Again after each insertion. You do NOT need to reselect all\n'
        'original blocks; the script remembers their exact symbol IDs.'),
        justify='left').pack(anchor='w', pady=(0, 10))
    txt = tk.Text(frm, height=12, wrap='word', font=('Consolas', 10))
    txt.pack(fill='both', expand=True)
    status = ttk.Label(frm, text='', foreground='#a0522d')
    status.pack(anchor='w', pady=(8, 8))
    row = ttk.Frame(frm)
    row.pack(fill='x')
    finished = {'value': False}
    def close():
        if not finished['value']:
            finished['value'] = True
            root.quit()
    root.protocol('WM_DELETE_WINDOW', close)

    def show_issues(issues):
        txt.config(state='normal')
        txt.delete('1.0', 'end')
        txt.insert('end', f'Missing/duplicate linked text entries: {len(issues)}\n\n')
        for item in issues:
            title = FIELD_LABELS[item['field']]
            bnd = f"X {item['left']:.2f}..{item['right']:.2f}, Y {item['bottom']:.2f}..{item['top']:.2f}"
            if item['count'] == 0:
                txt.insert('end', f"Block {item['id']} ({bnd})\n"
                           f"  INSERT {title} (text type {item['type']})\n\n")
            else:
                txt.insert('end', f"Block {item['id']} ({bnd})\n"
                           f"  DUPLICATE {title}: {item['count']} nodes; resolve manually\n\n")
        txt.config(state='disabled')
        status.config(text=f'{len(issues)} unresolved entries — no header formatting yet')

    def check_again():
        try:
            refreshed = inspect_block_texts(prj, blocks)
            remaining = missing_header_nodes(blocks, refreshed, mapping)
            log(f'  GUIDED RECHECK: {len(remaining)} missing/duplicate required text entries')
            if remaining:
                show_issues(remaining)
            else:
                status.config(text='All linked text nodes found. Formatting the original selection...')
                result['ready'] = True
                finished['value'] = True
                root.quit()
        except Exception as exc:
            log(f'  GUIDED RECHECK FAILED: {type(exc).__name__}: {exc}')
            messagebox.showerror('E3 readback error', str(exc), parent=root)

    ttk.Button(row, text='Check Again / Finish Formatting',
               command=check_again).pack(side='left')
    ttk.Button(row, text='Cancel / Close', command=close).pack(side='right')
    root.bind('<Escape>', lambda e: close())
    root.bind('<Return>', lambda e: check_again())
    root.update_idletasks()
    x = max(0, root.winfo_screenwidth()-root.winfo_width()-35)
    root.geometry(f'+{x}+80')
    try:
        check_again()
        if not result['ready']:
            root.mainloop()
        return result['ready']
    finally:
        root.destroy()


def determine_mapping(prj, all_texts, override_saber=None,
                      override_equipment=None, override_designation=None):
    """Use confirmed type 5002. Never ask GetTextTypes or guess COM definitions."""
    saber_type = int(override_saber) if override_saber is not None else SABER_TYPE
    prohibited = {14, 15, 1000, 212, 1010}
    if saber_type <= 0 or saber_type in prohibited:
        raise RuntimeError(f'Invalid Saber type {saber_type}. No changes made.')
    mapping = {
        'location': LOCATION_TYPE,
        'saber': saber_type,
        'equipment': int(override_equipment) if override_equipment is not None else EQUIPMENT_TYPE,
        'designation': int(override_designation) if override_designation is not None else DEVICE_DESIGNATION_TYPE,
    }
    if len(set(mapping.values())) != 4 or (set(mapping.values()) & {14, 1010}):
        raise RuntimeError(f'Overlapping/protected text types: {mapping}. No changes made.')
    log('CONFIRMED HEADER TEXT TYPES: ' + ', '.join(
        f'{FIELD_LABELS[k]}={mapping[k]}' for k in HEADER_ORDER))
    # All visible linked nodes must be present. This E3 COM version exposes
    # neither Symbol.AddText nor a verified equivalent. No fake free-text labels.
    missing = []
    for block_id, entries in all_texts.items():
        for field, typ in mapping.items():
            if len(entries.get(typ, [])) != 1:
                missing.append(f'block {block_id}: {FIELD_LABELS[field]} type {typ} '
                               f'has {len(entries.get(typ, []))} attached text nodes')
    if missing:
        raise RuntimeError(
            'Missing required property-linked text on the selected block(s):\n'
            + '\n'.join(missing) + '\n\n'
            'To add Saber Parameter 1 text: In E3 select ONE block, use '
            'Insert > Attribute Text > Saber Parameter 1 and check '
            '"Bind to selected Cell". Insert the text. Repeat for each missing '
            'block, then rerun Format Headers.\n'
            'If Equipment Identification (1000) is missing, add its linked '
            'attribute text in E3 similarly.\n'
            '"Prepare XXX Attributes" can populate the symbol attribute '
            'before you insert its linked text. NO E3 CHANGES WERE MADE.'
        )
    return mapping

def available_method(obj, name):
    try:
        return callable(getattr(obj, name))
    except Exception:
        return False


def get_symbol_saber_value(sym, allow_missing=False):
    try:
        raw = sym.GetAttributeValue(SABER_ATTRIBUTE_NAME)
    except Exception as exc:
        if allow_missing:
            log(f'  Missing/inaccessible Saber attribute: {type(exc).__name__}: {str(exc)[:120]}')
            return None
        raise RuntimeError(f'Cannot read existing symbol saberParam1: {exc}') from exc
    if raw is None:
        return ''
    if isinstance(raw, (tuple, list)):
        if len(raw) != 1 or not isinstance(raw[0], str):
            raise RuntimeError(f'Ambiguous symbol GetAttributeValue response: {raw!r}')
        raw = raw[0]
    return str(raw).strip()


def preflight_saber_attributes(prj, plan):
    """Validate symbol-only Saber property access before changing any E3 data."""
    sym=prj.CreateSymbolObject()
    for item in plan:
        if item['field'] != 'saber': continue
        if not select_id(sym,item['block']['id']):
            raise RuntimeError(f"Block {item['block']['id']} disappeared.")
        attr=get_symbol_saber_value(sym,allow_missing=False)
        item['before_attribute_value']=attr
        item['set_xxx']=not attr.strip() and not item['before_value'].strip()
        item['new_attr_needed']=False
        if item['set_xxx'] and not (available_method(sym,'SetAttributeValue') or
                                   available_method(sym,'AddAttributeValue')):
            raise RuntimeError('No symbol-level Saber attribute setter in this E3 build.')
        log(f"  Block {item['block']['id']} saberParam1={attr!r}; "
            f"visible type {item['type']}={item['before_value']!r}; "
            f"set XXX={item['set_xxx']}")
    return plan

def build_plan(blocks, all_texts, mapping):
    """Four distinct linked text nodes required per CABBLOCK."""
    plan=[]
    seen=set()
    for b in blocks:
        texts=all_texts[b['id']]
        height=float(b['top'])-float(b['bottom'])
        required=TOP_INSET+3*TEXT_PITCH+BOTTOM_CLEARANCE
        if height<required:
            raise RuntimeError(f"Block {b['id']} too short ({height:.3f} < {required:.3f}).")
        midx=(float(b['left'])+float(b['right']))/2
        for i,field in enumerate(HEADER_ORDER):
            typ=mapping[field]
            matches=texts.get(typ,[])
            if len(matches)!=1:
                raise RuntimeError(f"Block {b['id']} missing/duplicate {field} type {typ}.")
            entry=matches[0]
            if entry['id'] in seen:
                raise RuntimeError(f"Text {entry['id']} is attached to multiple blocks.")
            seen.add(entry['id'])
            plan.append({'block':b,'field':field,'type':typ,'id':entry['id'],
                         'before_value':entry['value'],'x':midx,
                         'y':float(b['top'])-TOP_INSET-i*TEXT_PITCH})
    return plan

def preflight_positions(prj,plan):
    txt=prj.CreateTextObject()
    for item in plan:
        if not select_id(txt,item['id']):
            raise RuntimeError(f"Text {item['id']} disappeared before formatting.")
        item['before']=get_text_xy(txt,item['block'])
        item['move_needed']=(abs(item['before']['x']-item['x'])>POSITION_TOLERANCE or
                             abs(item['before']['y']-item['y'])>POSITION_TOLERANCE)
        log(f"  Block {item['block']['id']} {FIELD_LABELS[item['field']]} "
            f"text {item['id']}: {item['before']} -> "
            f"({item['x']:.3f},{item['y']:.3f}) move={item['move_needed']}")
    return plan

def old_function_nodes_to_hide(blocks, all_texts, plan):
    """Hide ONLY existing Device Function display nodes on selected blocks.

    The old formatter left these visible at the second header row. We do not
    modify their text content, property value, type, position or formatting.
    """
    target_ids = {item['id'] for item in plan}
    old_nodes = []
    seen = set()
    for block in blocks:
        for record in all_texts[block['id']].get(14, []):
            tid = record['id']
            if tid in target_ids or tid in seen:
                raise RuntimeError(
                    f'Text {tid} appears in multiple selected blocks or overlaps '
                    'a target header text; batch refused before changes.'
                )
            seen.add(tid)
            old_nodes.append({'id': tid, 'block': block, 'value': record['value']})
    log(f'  OLD DEVICE FUNCTION LABELS: {len(old_nodes)} will be hidden '
        '(their values and E3 attributes are not changed).')
    return old_nodes


def _same_xy(txt, block, x, y):
    try:
        actual = get_text_xy(txt, block)
        return (abs(actual['x'] - x) <= POSITION_TOLERANCE and
                abs(actual['y'] - y) <= POSITION_TOLERANCE)
    except Exception:
        return False


def move_text(txt, item):
    """Only try signatures already used by the working signal relocator."""
    method = getattr(txt, 'SetSchemaLocation', None)
    if method is None:
        raise RuntimeError('E3 Text.SetSchemaLocation is unavailable.')
    old = item['before']
    b = item['block']
    attempts = (
        ((item['x'], item['y']), (old['x'], old['y']), 'x,y'),
        ((b['sheet_id'], item['x'], item['y']),
         (b['sheet_id'], old['x'], old['y']), 'sheet,x,y'),
    )
    errors = []
    for new, rollback, signature in attempts:
        try:
            result = method(*new)
        except Exception as exc:
            errors.append(f'{signature}: {exc}')
            continue
        if _same_xy(txt, b, item['x'], item['y']):
            return
        try:
            method(*rollback)
        except Exception as exc:
            raise RuntimeError(f"Text {item['id']} moved but readback failed "
                               f"and rollback also failed: {exc}; CHECK E3") from exc
        raise RuntimeError(f"Text {item['id']}: SetSchemaLocation({signature}) "
                           'returned but new position was not verified. '
                           f'Rolled back (reported result={result!r}); batch stopped.')
    raise RuntimeError(f"Text {item['id']}: unable to set position: {errors}")


def checked_style_write(txt, method, value, field):
    try:
        result = getattr(txt, method)(value)
    except Exception as exc:
        raise RuntimeError(f"Text {field}: {method}({value}) failed: {exc}") from exc
    # E3 methods sometimes return None on success; negative status means error.
    if isinstance(result, (int, float)) and result < 0:
        raise RuntimeError(f"Text {field}: {method}({value}) returned {result}.")


def set_symbol_saber_xxx(sym,block_id):
    """Modify ONLY the selected SYMBOL attribute; verify by reading it back."""
    if not select_id(sym,block_id):
        raise RuntimeError(f'Block {block_id} no longer available')
    before=get_symbol_saber_value(sym)
    if before:
        return False
    errors=[]
    for name in ('SetAttributeValue','AddAttributeValue'):
        if not available_method(sym,name): continue
        try:
            rv=getattr(sym,name)(SABER_ATTRIBUTE_NAME,'XXX')
            if type(rv) is int and rv<0:
                errors.append(f'{name} returned {rv}')
                continue
            after=get_symbol_saber_value(sym)
            if after=='XXX':
                log(f'  BLOCK {block_id}: {name}(saberParam1, XXX) confirmed')
                return True
            errors.append(f'{name} readback={after!r}')
        except Exception as exc:
            errors.append(f'{name}: {type(exc).__name__}: {exc}')
    raise RuntimeError(f'Block {block_id}: cannot set symbol Saber Parameter 1 to XXX; '
                       + '; '.join(errors))


def apply_plan(prj,plan,old_function_labels=None):
    """Format verified existing linked texts only, not unbound graphics."""
    txt=prj.CreateTextObject()
    sym=prj.CreateSymbolObject()
    completed=0
    filled=0
    try:
        for item in plan:
            if item['field']=='saber' and item['set_xxx']:
                filled+=int(set_symbol_saber_xxx(sym,item['block']['id']))
            if not select_id(txt,item['id']):
                raise RuntimeError(f"Text {item['id']} disappeared.")
            field=FIELD_LABELS[item['field']]
            for method,val in (
                ('SetVisibility',1), ('SetStyle',BOLD_STYLE),
                ('SetColour',BLACK_COLOUR),('SetHeight',TEXT_HEIGHT),
                ('SetAlignment',CENTER_ALIGNMENT)):
                checked_style_write(txt,method,val,field)
            if item['move_needed']:
                move_text(txt,item)
            completed+=1
            log(f"  DONE block {item['block']['id']} | {field} text {item['id']}")
        for old in (old_function_labels or []):
            if not select_id(txt,old['id']):
                raise RuntimeError(f"Old function text {old['id']} disappeared")
            checked_style_write(txt,'SetVisibility',0,'Old Function display')
        return completed,filled
    except Exception:
        log(f'PARTIAL FAILURE: {completed}/{len(plan)} labels formatted; '
            'earlier E3 writes may remain. Inspect drawing.')
        raise

def prepare_selected_saber_attributes(prj,blocks,preview=False):
    """Add XXX as a SYMBOL attribute value, without touching unrelated text."""
    sym=prj.CreateSymbolObject()
    # Read/verify all target symbols before first write.
    staged=[]
    for b in blocks:
        if not select_id(sym,b['id']): raise RuntimeError(f"Block {b['id']} disappeared")
        before=get_symbol_saber_value(sym,allow_missing=False)
        if not before and not (available_method(sym,'SetAttributeValue') or
                               available_method(sym,'AddAttributeValue')):
            raise RuntimeError(f"Block {b['id']} has no symbol-level attribute setter")
        staged.append((b['id'],before))
        log(f"  PREPARE BLOCK {b['id']} saberParam1: {before!r} -> "
            f"{'XXX' if not before else '(preserve existing)'}")
    if preview:
        log('PREPARE DRY RUN: no E3 changes made')
        popup(f'Checked {len(staged)} block(s). No values changed.\n'
              'Start without --dry-run to set blank symbol Saber attributes to XXX.')
        return 0
    count=0
    for bid,before in staged:
        if not before: count+=int(set_symbol_saber_xxx(sym,bid))
    log(f'PREPARE COMPLETE: {count} blank symbol attributes set to XXX '
        f'across {len(staged)} selected block(s).')
    popup(f'Prepared {len(staged)} selected block(s).\n'
          f'{count} blank Saber Parameter 1 attributes set to XXX.\n\n'
          'Next: If type 5002 is missing, select ONE block in E3 and use\n'
          'Insert > Attribute Text > Saber Parameter 1, with\n'
          'Bind to selected Cell. Repeat for missing blocks.\n'
          'Then run Format Headers (Enter).')
    return count


def inspect_creation_api(prj, blocks, all_texts):
    """Read-only E3 COM inspection: no attribute writes, no added texts.

    This is intentionally conservative. Some pywin32 COM dispatch objects
    raise on str()/int(), so this never coerces live COM objects.
    """
    sym = prj.CreateSymbolObject()
    log('SABER CREATION API CAPABILITIES (READ ONLY):')
    member_names = ('GetAttributeValue', 'GetAttributeIds', 'GetAttributeNames',
                    'GetAttributeName', 'AddAttributeValue', 'SetAttributeValue',
                    'AddText', 'AddAttributeText', 'InsertText', 'CreateText',
                    'AddTextNode', 'GetTextIds')
    for block in blocks:
        if not select_id(sym, block['id']):
            log(f"  BLOCK {block['id']}: unable to bind symbol")
            continue
        log(f"  BLOCK {block['id']}:")
        for name in member_names:
            log(f'    {name}: {"AVAILABLE" if available_method(sym, name) else "not exposed"}')
        for attr in (SABER_ATTRIBUTE_NAME, 'Saber Parameter 1'):
            if not available_method(sym, 'GetAttributeValue'):
                break
            try:
                value = sym.GetAttributeValue(attr)
                if type(value) in (int, float, str, bool) or value is None:
                    shown = repr(value)
                elif type(value) in (tuple, list):
                    shown = repr([repr(v) if type(v) in (int,float,str,bool) or v is None else f'<{type(v).__name__}>' for v in value])
                else:
                    shown = f'<{type(value).__name__} COM object>'
                log(f'    GetAttributeValue({attr!r}) => {shown}')
            except Exception as exc:
                log(f'    GetAttributeValue({attr!r}) => {type(exc).__name__}: {str(exc)[:130]}')
        types = sorted(all_texts[block['id']])
        log(f'    Attached text types: {types}')
    log('The mere existence of AddText does NOT prove its signature or that it creates attribute-linked text.')


def run(action='format',preview=False,saber_type=None,
        equipment_type=None,designation_type=None,inspect_api=False):
    global LOG_STREAM
    path=choose_log_path()
    try:
        LOG_STREAM=open(path,'a',encoding='utf-8')
        log('\n'+'='*80)
        log('NEW SABER HEADER BATCH: '+datetime.now().astimezone().isoformat(timespec='seconds'))
        log('BUILD: '+BUILD_ID)
        log('SOURCE FILE: '+str(Path(__file__).resolve()))
        log(f'ACTION: {action}, dry-run={preview}, inspect-api={inspect_api}')
        log('HEADER TYPES: LOCATION=15, SABER PARAMETER 1=5002, EQUIPMENT=1000, DESIGNATION=212')
        log('NEVER CHANGE DEVICE FUNCTION VALUE (14) OR COMBINED TEXT (1010)')
        log('LINKED TEXT IS CREATED IN E3 GUI; THIS SCRIPT CHECKS AND FORMATS IT')
        log('='*80)
        import win32com.client as win32
        app=win32.gencache.EnsureDispatch('CT.Application')
        prj=app.CreateJobObject()
        sht=prj.CreateSheetObject()
        blocks=get_selected_cabblocks(prj,sht,app)
        all_texts=inspect_block_texts(prj,blocks)
        if inspect_api:
            inspect_creation_api(prj,blocks,all_texts)
            popup('Read-only API inspection complete.\nLog: '+str(path))
            return 0
        if action=='prepare':
            return prepare_selected_saber_attributes(prj,blocks,preview=preview)
        # Resolve confirmed text-type IDs, but allow temporarily missing nodes
        # so the guided workflow can insert them in E3 before preflight.
        mapping = {
            'location': LOCATION_TYPE,
            'saber': int(saber_type) if saber_type is not None else SABER_TYPE,
            'equipment': int(equipment_type) if equipment_type is not None else EQUIPMENT_TYPE,
            'designation': int(designation_type) if designation_type is not None else DEVICE_DESIGNATION_TYPE,
        }
        if (len(set(mapping.values())) != 4 or set(mapping.values()) & {14, 1010}
                or min(mapping.values()) <= 0):
            raise RuntimeError(f'Invalid/protected text-type mapping: {mapping}')
        issues = missing_header_nodes(blocks, all_texts, mapping)
        if issues:
            log('MISSING LINKED TEXT: ' + repr(issues))
            if preview:
                popup(f'{len(issues)} missing/duplicate required linked text entries.\n'
                      'Read-only preview: no changes made.\n'
                      f'Log: {path}')
                return 0
            if action != 'complete':
                # Preserve existing strict behavior for legacy format action.
                raise RuntimeError('Missing required linked header text. '
                                   'Run Complete Headers to use guided insertion.')
            # Prepare the per-SYMBOL property *before* manual placement.
            # Check the existing getter/setter for every original block first.
            sym = prj.CreateSymbolObject()
            staged = []
            for b in blocks:
                if not select_id(sym, b['id']):
                    raise RuntimeError(f"Block {b['id']} disappeared.")
                existing = get_symbol_saber_value(sym, allow_missing=False)
                staged.append((b['id'], existing))
            for bid, value in staged:
                if not value:
                    set_symbol_saber_xxx(sym, bid)
            log('Prepared empty per-symbol Saber Parameter 1 attributes; '
                'waiting for genuine bound E3 text nodes.')
            ready = guided_insert_missing_text(prj, blocks, mapping)
            if not ready:
                log('GUIDED INSERT CANCELLED: XXX attributes may have been prepared; '
                    'no header positions/styles changed.')
                return 0
            # The user may have changed GUI selection while inserting the text.
            # Intentionally reuse original block IDs, not current GUI selection.
            all_texts = inspect_block_texts(prj, blocks)
        # All four genuine linked text nodes must exist before formatting.
        mapping = determine_mapping(prj, all_texts, saber_type, equipment_type,
                                    designation_type)
        plan=build_plan(blocks,all_texts,mapping)
        preflight_positions(prj,plan)
        preflight_saber_attributes(prj,plan)
        old=old_function_nodes_to_hide(blocks,all_texts,plan)
        moves=sum(p['move_needed'] for p in plan)
        blanks=sum(p['set_xxx'] for p in plan)
        log(f'BATCH READY: {len(blocks)} blocks, {len(plan)} texts, '
            f'{moves} moves, {blanks} empty Saber attributes, {len(old)} old Function labels to hide')
        if preview:
            popup(f'Validated {len(blocks)} block(s), {len(plan)} texts.\n'
                  f'{moves} moves, {blanks} blank Saber attributes.\n'
                  'DRY RUN: NO E3 CHANGES.\nLog: '+str(path))
            return 0
        completed,filled=apply_plan(prj,plan,old)
        popup(f'Formatted {len(blocks)} selected block(s).\n'
              f'{completed} labels formatted; {filled} empty Saber values filled.\n'
              'Location > Saber Parameter 1 > Equipment ID > Device Designation.\n\n'
              'Save the E3 project.\nLog: '+str(path))
        return 0
    except Exception:
        log('ERROR: '+traceback.format_exc())
        raise
    finally:
        if LOG_STREAM is not None: LOG_STREAM.close()
        LOG_STREAM=None


def main(argv=None):
    parser=argparse.ArgumentParser(description='Format selected E3 CABBLOCK headers with Saber Parameter 1 text type 5002.')
    parser.add_argument('--dry-run',action='store_true',help='Preview actions only; do not modify E3')
    parser.add_argument('--inspect-api',action='store_true',help='Read-only E3 COM capability report')
    parser.add_argument('--saber-type',type=int,default=None,help='Override 5002 only if E3 database configuration differs')
    parser.add_argument('--equipment-type',type=int,default=None,help='Override Equipment ID type 1000')
    parser.add_argument('--designation-type',type=int,default=None,help='Override block designation type 212')
    args=parser.parse_args(argv)
    print('E3 HEADER FORMATTER BUILD:',BUILD_ID,flush=True)
    print('E3 HEADER FORMATTER FILE:',Path(__file__).resolve(),flush=True)
    while True:
        try:
            action=wait_for_block_selection(preview=(args.dry_run or args.inspect_api))
        except Exception as exc:
            print('Selection window failed:',exc)
            return 1
        if action=='close':
            print('CLOSED: no more E3 changes.')
            return 0
        try:
            run(action=action,preview=args.dry_run,saber_type=args.saber_type,
                equipment_type=args.equipment_type,
                designation_type=args.designation_type,
                inspect_api=args.inspect_api)
        except Exception as exc:
            print('FAILED:',exc)
            popup(str(exc)+'\n\nThe selection window will reopen.',error=True)


if __name__=='__main__':
    sys.exit(main())
