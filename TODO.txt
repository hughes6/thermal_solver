import time
import win32com.client as win32


SYMBOL_NAME = "Shield_Demo"
SYMBOL_VERSION = 1

TEST_X = 20.0
TEST_Y = 10.0

TEST_LENGTHS = [
    1.00,
    1.25,
    1.30,
    1.50,
    2.00,
    5.00,
    12.00
]


app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sht = prj.CreateSheetObject()
sym = prj.CreateSymbolObject()


sheet_info = prj.GetAllSheetIds(None)
sheet_ids = [value for value in sheet_info[1] if value is not None]
sheet_id = None


for candidate_id in sheet_ids:
    sht.SetId(candidate_id)

    if sht.GetName() == "1":
        sheet_id = candidate_id
        break


if sheet_id is None:
    raise Exception("Could not find sheet 1")


print("")
print("============================================================")
print("SHIELD DEMO GROWTH TEST")
print("============================================================")
print("SHEET ID:", sheet_id)


for index, requested_length in enumerate(TEST_LENGTHS):
    x = TEST_X + index * 0.5
    y = TEST_Y

    print("")
    print("------------------------------------------------------------")
    print("REQUESTED LENGTH:", requested_length)
    print("TEST X:", x)
    print("TEST Y:", y)

    load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)

    print("LOAD RESULT:", load_result)

    if load_result == 0:
        print("LOAD FAILED")
        continue

    placed_id = sym.Place(sheet_id, x, y, 0, 1.0, 0)

    print("PLACE RESULT:", placed_id)

    if placed_id == 0:
        print("PLACE FAILED")
        continue

    time.sleep(0.10)

    sym.SetId(placed_id)

    try:
        before = float(sym.GetDisplayLength())
    except Exception:
        before = None

    print("DISPLAY LENGTH BEFORE:", before)

    stretch_result = sym.SetDisplayLength(requested_length)

    print("SET DISPLAY LENGTH RESULT:", stretch_result)

    time.sleep(0.10)

    sym.SetId(placed_id)

    try:
        after = float(sym.GetDisplayLength())
    except Exception:
        after = None

    print("DISPLAY LENGTH AFTER:", after)

    if after is not None and abs(after - requested_length) <= 0.001:
        print("RESULT: SUCCESS")
    else:
        print("RESULT: FAILED")


print("")
print("============================================================")
print("TEST COMPLETE")
print("============================================================")


del sym
del sht
del prj
del app
