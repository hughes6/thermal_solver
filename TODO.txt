def extract_e3_ids(result):
    if result is None:
        return []

    if isinstance(result, int):
        if result > 0:
            return [result]

        return []

    if not isinstance(result, tuple):
        return []

    if len(result) < 2:
        return []

    values = result[1]

    if values is None:
        return []

    if isinstance(values, int):
        if values > 0:
            return [values]

        return []

    try:
        return [value for value in values if value is not None and int(value) > 0]
    except Exception:
        return []

The wire-location logic is now working correctly. The problem is specifically PlaceDynamic().

PlaceDynamic() creates E3's generic Dynamic Symbol. It does not use the library symbol previously loaded with sym.Load(). Zuken treats Dynamic Symbols and cable bundle symbols as separate categories. For the actual shield/twisted-pair library symbols, the correct sequence is Load → Place → SetDisplayLength. E3.series 2025 specifically added SetDisplayLength() support for shield, twisted-pair, and bundle symbols. 

Replace your current place_job() with this exact version. No multiline prints or function calls.

#====================================================================================================================================
# PLACE ONE REAL LIBRARY SHIELD / TWIST SYMBOL
#
# IMPORTANT:
#
# DO NOT USE PlaceDynamic HERE.
#
# PlaceDynamic creates E3's generic Dynamic Symbol.
#
# Instead:
#
#     1. Load actual database symbol
#     2. Place actual database symbol
#     3. Bind sym to newly placed symbol ID
#     4. Stretch actual shield/twist symbol vertically
#
# Existing cable objects are NEVER deleted, replaced, moved, or reconnected.
#====================================================================================================================================
def place_job(sym, job, placement):
    print("")
    print("------------------------------------------------------------")
    print("TYPE:", job["type"])
    print("SIDE:", job["side"])
    print("CONNECTOR:", job["connector"])
    print("PINS:", job["pins"])
    print("EXCEL ROWS:", job["excel_rows"])
    print("LIBRARY SYMBOL:", job["symbol_name"])
    print("SHEET:", placement["sheet_id"])
    print("CONNECTOR PIN X:", placement["anchor_x"])
    print("SYMBOL X:", placement["x"])
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("SYMBOL Y:", placement["y"])
    print("REQUESTED SYMBOL LENGTH:", placement["height"])

    if DRY_RUN:
        print("DRY RUN - NOT PLACED")
        return None

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("LOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not load library symbol {job['symbol_name']}")

    placed_id = sym.Place(placement["sheet_id"], placement["x"], placement["y"], 0, 1.0, 0)

    print("PLACE RESULT / SYMBOL ID:", placed_id)

    if placed_id == 0:
        raise Exception(f"Could not place library symbol {job['symbol_name']}")

    sym.SetId(placed_id)

    try:
        placed_name = sym.GetName()
    except Exception:
        placed_name = "UNKNOWN"

    try:
        symbol_type = sym.GetSymbolType()
    except Exception:
        symbol_type = "UNKNOWN"
