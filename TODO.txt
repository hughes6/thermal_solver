LEFT_INNER_SHIELD_COL = "I"

PIN_2_COL = "O"

RIGHT_CONNECTOR_COL = "Q"

RIGHT_TWIST_B_COL = "S"
RIGHT_INNER_SHIELD_COL = "T"
RIGHT_OUTER_SHIELD_COL = "U"


#====================================================================================================================================
# E3 CONNECTOR NAMING
#
# B1 is only a sheet/device reference.
#
# Connector cells contain only:
#     1
#     P1
#     2
#     P2
#
# Those become:
#     W1.P1
#     W1.P2
#
# The script prefers the schematic view:
#     W1.P1#
#     W1.P1#1
#     W1.P1#2
#
# It does NOT build A2.P1, B1.P1, etc.
#====================================================================================================================================
CONNECTOR_ROOT_NAME = "W1"


#====================================================================================================================================
# ZUKEN SYMBOLS
#====================================================================================================================================
TWIST_SYMBOL_NAME = "MIL_Twist_ArrowsDown"
SHIELD_SYMBOL_NAME = "Cable Shield"
SYMBOL_VERSION = 1

DYNAMIC_WIDTH = 0.10
MIN_DYNAMIC_HEIGHT = 0.10


#====================================================================================================================================
# VERTICAL PADDING
#====================================================================================================================================
TWIST_VERTICAL_PADDING = 0.00
INNER_SHIELD_VERTICAL_PADDING = 0.05
OUTER_SHIELD_VERTICAL_PADDING = 0.10


#====================================================================================================================================
# HORIZONTAL OFFSETS FROM EXISTING CONNECTOR PIN X
#====================================================================================================================================
TWIST_A_OFFSET = 1.00
TWIST_B_OFFSET = 1.25

INNER_SHIELD_TWIST_A_OFFSET = 1.15
INNER_SHIELD_TWIST_B_OFFSET = 1.40
INNER_SHIELD_ONLY_OFFSET = 1.15

OUTER_SHIELD_OFFSET = 1.65


#====================================================================================================================================
# OPTIONS
#====================================================================================================================================
DRY_RUN = False
STRICT_TWIST_VALIDATION = True


#====================================================================================================================================
# FILE PICKER
#====================================================================================================================================
def ask_excel_path():
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    selected_path = filedialog.askopenfilename(title="Select Cable Symbol Placement Excel File", filetypes=[("Excel Files", "*.xlsx"), ("Excel Macro Files", "*.xlsm"), ("All Files", "*.*")])
    root.destroy()

    if not selected_path:
        return None

    return os.path.abspath(selected_path)


#====================================================================================================================================
# VALUE HELPERS
#====================================================================================================================================
def clean_excel_value(value):
    if value is None:
        return ""

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def normalize_pin_name(value):
    return clean_excel_value(value).upper()


def normalize_connector_token(value):
    text = clean_excel_value(value).upper().replace(" ", "")

    if text == "":
        return ""

    root_prefix = CONNECTOR_ROOT_NAME.upper() + "."

    if text.startswith(root_prefix):
        text = text[len(root_prefix):]

    if "#" in text:
        text = text.split("#", 1)[0]

    match = re.fullmatch(r"P?0*(\d+)([A-Z]*)", text)

    if match:
        number = int(match.group(1))
        suffix = match.group(2)
        return f"P{number}{suffix}"

    return text


def connector_base_name(connector_token):
    connector = normalize_connector_token(connector_token)

    if connector == "":
        return ""

    return f"{CONNECTOR_ROOT_NAME}.{connector}"


def connector_view_prefix(connector_token):
    base_name = connector_base_name(connector_token)

    if base_name == "":
        return ""

    return base_name + "#"


def cell_has_bar(value):
    return "|" in clean_excel_value(value)


#====================================================================================================================================
# FIND DATA EXTENT
#====================================================================================================================================
def get_last_data_row(ws):
    columns = [LEFT_OUTER_SHIELD_COL, LEFT_INNER_SHIELD_COL, LEFT_TWIST_A_COL, LEFT_TWIST_B_COL, LEFT_CONNECTOR_COL, PIN_1_COL, PIN_2_COL, RIGHT_CONNECTOR_COL, RIGHT_TWIST_A_COL, RIGHT_TWIST_B_COL, RIGHT_INNER_SHIELD_COL, RIGHT_OUTER_SHIELD_COL]
    scan_end = min(max(ws.max_row, DATA_FIRST_ROW), DATA_FIRST_ROW + MAX_SCAN_ROWS)
    last_row = DATA_FIRST_ROW - 1

    for row in range(DATA_FIRST_ROW, scan_end + 1):
        for col in columns:
            if clean_excel_value(ws[f"{col}{row}"].value) != "":
                last_row = row
                break

    return last_row


#====================================================================================================================================
# READ LEFT OR RIGHT SIDE TABLE
#====================================================================================================================================
def read_side_rows(ws, side, last_row):
    if side == "LEFT":
        connector_col = LEFT_CONNECTOR_COL
        pin_col = PIN_1_COL
        twist_a_col = LEFT_TWIST_A_COL
        twist_b_col = LEFT_TWIST_B_COL
        inner_col = LEFT_INNER_SHIELD_COL
        outer_col = LEFT_OUTER_SHIELD_COL
    else:
        connector_col = RIGHT_CONNECTOR_COL
        pin_col = PIN_2_COL
        twist_a_col = RIGHT_TWIST_A_COL
        twist_b_col = RIGHT_TWIST_B_COL
        inner_col = RIGHT_INNER_SHIELD_COL
        outer_col = RIGHT_OUTER_SHIELD_COL

    rows = []
    current_connector = ""

    for excel_row in range(DATA_FIRST_ROW, last_row + 1):
        connector_value = clean_excel_value(ws[f"{connector_col}{excel_row}"].value)
        pin_name = normalize_pin_name(ws[f"{pin_col}{excel_row}"].value)

        if connector_value != "":
            current_connector = normalize_connector_token(connector_value)

        if pin_name == "":
            continue

        row_info = {
            "side": side,
            "excel_row": excel_row,
            "connector": current_connector,
            "pin": pin_name,
            "twist_a": cell_has_bar(ws[f"{twist_a_col}{excel_row}"].value),
            "twist_b": cell_has_bar(ws[f"{twist_b_col}{excel_row}"].value),
            "inner_shield": cell_has_bar(ws[f"{inner_col}{excel_row}"].value),
            "outer_shield": cell_has_bar(ws[f"{outer_col}{excel_row}"].value)
        }

        rows.append(row_info)

    return rows


#====================================================================================================================================
# CONNECTOR SECTIONS
#====================================================================================================================================
def split_into_sections(rows):
    sections = []
    current = []

    for row in rows:
        if not current:
            current = [row]
            continue

        previous = current[-1]
        connector_changed = row["connector"] != previous["connector"]
        row_gap = row["excel_row"] != previous["excel_row"] + 1

        if connector_changed or row_gap:
            sections.append(current)
            current = [row]
        else:
            current.append(row)

    if current:
        sections.append(current)

    return sections


#====================================================================================================================================
# CONTIGUOUS MARKED RANGES
#====================================================================================================================================
def true_ranges(rows, key):
    ranges = []
    start = None

    for index, row in enumerate(rows):
        marked = bool(row[key])

        if marked and start is None:
            start = index

        if not marked and start is not None:
            ranges.append((start, index - 1))
            start = None

    if start is not None:
        ranges.append((start, len(rows) - 1))

    return ranges


def range_pins(rows, start, end):
    pins = []

    for index in range(start, end + 1):
        pins.append(rows[index]["pin"])

    return pins


def range_excel_rows(rows, start, end):
    excel_rows = []

    for index in range(start, end + 1):
        excel_rows.append(rows[index]["excel_row"])

    return excel_rows


#====================================================================================================================================
# BUILD SYMBOL JOBS
#====================================================================================================================================
def build_section_jobs(rows):
    jobs = []
    warnings = []

    if not rows:
        return jobs, warnings

    side = rows[0]["side"]
    connector = rows[0]["connector"]

    if connector == "":
        warnings.append(f"{side}: rows {rows[0]['excel_row']}-{rows[-1]['excel_row']} have pins but no connector P#")
        return jobs, warnings

    conflicts = []

    for row in rows:
        if row["twist_a"] and row["twist_b"]:
            conflicts.append(row["excel_row"])

    if conflicts:
        message = f"{side} {connector}: same wire is marked in BOTH twist columns on Excel rows {conflicts}"

        if STRICT_TWIST_VALIDATION:
            raise Exception(message)

        warnings.append(message)

    twist_jobs = []

    for key, twist_column in [("twist_a", "A"), ("twist_b", "B")]:
        ranges = true_ranges(rows, key)

        for start, end in ranges:
            pins = range_pins(rows, start, end)
            excel_rows = range_excel_rows(rows, start, end)

            if len(pins) < 2:
                warnings.append(f"{side} {connector}: twist {twist_column} only contains one wire on Excel row {excel_rows[0]}")
                continue

            job = {
                "type": "twist",
                "side": side,
                "connector": connector,
                "twist_column": twist_column,
                "start": start,
                "end": end,
                "pins": pins,
                "excel_rows": excel_rows,
                "symbol_name": TWIST_SYMBOL_NAME
            }

            twist_jobs.append(job)
            jobs.append(job)

    inner_consumed = set()

    for twist_job in twist_jobs:
        indexes = list(range(twist_job["start"], twist_job["end"] + 1))
        fully_inner_shielded = all(rows[index]["inner_shield"] for index in indexes)

        if fully_inner_shielded:
            shield_job = {
                "type": "shield_inner",
                "side": side,
                "connector": connector,
                "twist_column": twist_job["twist_column"],
                "pins": list(twist_job["pins"]),
                "excel_rows": list(twist_job["excel_rows"]),
                "symbol_name": SHIELD_SYMBOL_NAME
            }

            jobs.append(shield_job)

            for index in indexes:
                inner_consumed.add(index)

    remaining = []

    for index, row in enumerate(rows):
        remaining.append(bool(row["inner_shield"] and index not in inner_consumed))

    start = None

    for index, marked in enumerate(remaining):
        if marked and start is None:
            start = index

        if not marked and start is not None:
            end = index - 1
            job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME}
            jobs.append(job)
            start = None

    if start is not None:
        end = len(rows) - 1
        job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME}
        jobs.append(job)

    outer_ranges = true_ranges(rows, "outer_shield")

    for start, end in outer_ranges:
        job = {"type": "shield_outer", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME}
        jobs.append(job)

    return jobs, warnings


#====================================================================================================================================
# GET ALL E3 DEVICES / VIEWS
#====================================================================================================================================
def get_all_devices(prj, dev):
    info = prj.GetAllDeviceIds(None)
    ids = [value for value in info[1] if value is not None]
    records = []

    for device_id in ids:
        dev.SetId(device_id)

        try:
            name = clean_excel_value(dev.GetName())
        except Exception:
            continue

        record = {"device_id": device_id, "name": name}
        records.append(record)

    return records


#====================================================================================================================================
# READ EXISTING PIN SCHEMATIC LOCATION
#====================================================================================================================================
def get_pin_location(pin, pin_id):
    pin.SetId(pin_id)
    xloc = [0.0]
    yloc = [0.0]
    grid = [""]
    column = [""]
    row = [""]

    try:
        result = pin.GetSchemaLocation(xloc, yloc, grid, column, row)
    except Exception:
        return None

    if not isinstance(result, tuple):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None

    if sheet_id <= 0:
        return None

    return {"sheet_id": sheet_id, "x": x, "y": y}


#====================================================================================================================================
# BUILD PIN MAP FOR ONE E3 DEVICE / VIEW
#====================================================================================================================================
def load_device_pin_map(dev, pin, device_id):
    dev.SetId(device_id)
    info = dev.GetAllPinIds(None)
    pin_ids = [value for value in info[1] if value is not None]
    pin_map = {}

    for pin_id in pin_ids:
        pin.SetId(pin_id)

        try:
            pin_name = normalize_pin_name(pin.GetName())
        except Exception:
            continue

        location = get_pin_location(pin, pin_id)

        if pin_name not in pin_map:
            pin_map[pin_name] = []

        if location is not None:
            pin_record = {"pin_id": pin_id, "sheet_id": location["sheet_id"], "x": location["x"], "y": location["y"]}
            pin_map[pin_name].append(pin_record)

    return pin_map


#====================================================================================================================================
# RESOLVE EXISTING CONNECTOR SCHEMATIC VIEW
#
# P1 -> W1.P1 -> prefer W1.P1#*
#
# B1 is NOT used here.
#====================================================================================================================================
def resolve_connector_view(connector_token, required_pins, all_devices, dev, pin, cache):
    base_name = connector_base_name(connector_token)
    view_prefix = connector_view_prefix(connector_token)
    base_upper = base_name.upper()
    view_prefix_upper = view_prefix.upper()
    view_matches = []
    base_matches = []

    for record in all_devices:
        name_upper = record["name"].upper()

        if name_upper.startswith(view_prefix_upper):
            view_matches.append(record)
        elif name_upper == base_upper:
            base_matches.append(record)

    matches = view_matches + base_matches

    if not matches:
        nearby = []

        for record in all_devices:
            name_upper = record["name"].upper()

            if name_upper.startswith(CONNECTOR_ROOT_NAME.upper() + "."):
                nearby.append(record["name"])

        nearby = nearby[:30]
        raise Exception(f"Could not find E3 connector/view for {connector_token}. Expected {base_name} or {view_prefix}*. Existing {CONNECTOR_ROOT_NAME} names: {nearby}")

    scored = []

    for record in matches:
        device_id = record["device_id"]

        if device_id not in cache:
            cache[device_id] = load_device_pin_map(dev, pin, device_id)

        pin_map = cache[device_id]
        pin_score = 0

        for pin_name in required_pins:
            if pin_name in pin_map and pin_map[pin_name]:
                pin_score += 1

        is_view = record["name"].upper().startswith(view_prefix_upper)
        view_bonus = 1 if is_view else 0
        scored.append((pin_score, view_bonus, record))

    scored.sort(key=lambda item: (item[0], item[1]), reverse=True)
    best_score = scored[0][0]
    best_record = scored[0][2]

    if best_score == 0:
        candidate_names = [record["name"] for record in matches]
        raise Exception(f"Found E3 names for {base_name}, but none contain placed schematic pins. Candidates: {candidate_names}")

    return best_record, cache[best_record["device_id"]]


#====================================================================================================================================
# SELECT THE SHEET CONTAINING THE MOST REQUIRED PINS
#====================================================================================================================================
def choose_section_sheet(required_pins, pin_map):
    coverage = {}

    for pin_name in required_pins:
        locations = pin_map.get(pin_name, [])

        for location in locations:
            sheet_id = location["sheet_id"]

            if sheet_id not in coverage:
                coverage[sheet_id] = set()

            coverage[sheet_id].add(pin_name)

    if not coverage:
        return None

    ranked = sorted(coverage.items(), key=lambda item: len(item[1]), reverse=True)
    best_sheet_id = ranked[0][0]
    best_count = len(ranked[0][1])
    tied = [sheet_id for sheet_id, pins_on_sheet in ranked if len(pins_on_sheet) == best_count]

    if len(tied) > 1:
        print("WARNING: multiple sheets have equal pin coverage:", tied)
        print("USING SHEET:", best_sheet_id)

    return best_sheet_id


#====================================================================================================================================
# GET ONE PIN LOCATION ON THE CHOSEN SHEET
#====================================================================================================================================
def pin_location_on_sheet(pin_map, pin_name, sheet_id):
    locations = pin_map.get(pin_name, [])

    for location in locations:
        if location["sheet_id"] == sheet_id:
            return location

    return None


#====================================================================================================================================
# HORIZONTAL OFFSET
#====================================================================================================================================
def job_offset(job):
    if job["type"] == "twist":
        if job["twist_column"] == "A":
            return TWIST_A_OFFSET

        return TWIST_B_OFFSET

    if job["type"] == "shield_inner":
        if job["twist_column"] == "A":
            return INNER_SHIELD_TWIST_A_OFFSET

        if job["twist_column"] == "B":
            return INNER_SHIELD_TWIST_B_OFFSET

        return INNER_SHIELD_ONLY_OFFSET

    return OUTER_SHIELD_OFFSET


#====================================================================================================================================
# VERTICAL PADDING
#====================================================================================================================================
def job_padding(job):
    if job["type"] == "twist":
        return TWIST_VERTICAL_PADDING

    if job["type"] == "shield_inner":
        return INNER_SHIELD_VERTICAL_PADDING

    return OUTER_SHIELD_VERTICAL_PADDING


#====================================================================================================================================
# DRAW ORDER
#====================================================================================================================================
def job_priority(job):
    if job["type"] == "shield_outer":
        return 0

    if job["type"] == "shield_inner":
        return 1

    return 2


#====================================================================================================================================
# CALCULATE PLACEMENT USING EXISTING CONNECTOR PIN / SIGNAL LINE Y LOCATIONS
#====================================================================================================================================
def calculate_placement(job, pin_map, sheet_id):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} is not placed on E3 sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]
    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)
    direction = 1.0 if job["side"] == "LEFT" else -1.0
    x = anchor_x + direction * job_offset(job)
    padding = job_padding(job)
    y = bottom_y - padding
    height = top_y - bottom_y + 2.0 * padding
    height = max(height, MIN_DYNAMIC_HEIGHT)

    return {"sheet_id": sheet_id, "x": x, "y": y, "height": height, "width": DYNAMIC_WIDTH, "anchor_x": anchor_x, "bottom_y": bottom_y, "top_y": top_y}


#====================================================================================================================================
# PLACE ONE DYNAMIC SYMBOL
#====================================================================================================================================
def place_job(sym, job, placement):
    print("")
    print("------------------------------------------------------------")
    print("TYPE:", job["type"])
    print("SIDE:", job["side"])
    print("CONNECTOR:", job["connector"])
    print("E3 CONNECTOR BASE:", connector_base_name(job["connector"]))
    print("PINS:", job["pins"])
    print("EXCEL ROWS:", job["excel_rows"])
    print("SYMBOL:", job["symbol_name"])
    print("SHEET:", placement["sheet_id"])
    print("ANCHOR PIN X:", placement["anchor_x"])
    print("X:", placement["x"])
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("PLACEMENT Y:", placement["y"])
    print("HEIGHT:", placement["height"])
    print("WIDTH:", placement["width"])

    if DRY_RUN:
        print("DRY RUN - NOT PLACED")
        return None

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    if load_result == 0:
        raise Exception(f"Could not load {job['symbol_name']}")

    placed_id = sym.PlaceDynamic(placement["sheet_id"], placement["x"], placement["y"], placement["height"], placement["width"], "", 0)

    if placed_id == 0:
        raise Exception(f"PlaceDynamic failed for {job['symbol_name']}")

    print("PLACED SYMBOL ID:", placed_id)

    return placed_id


#====================================================================================================================================
# MAIN
#====================================================================================================================================
def main():
    excel_path = ask_excel_path()

    if excel_path is None:
        raise Exception("No Excel file selected")

    workbook = load_workbook(excel_path, data_only=True, read_only=False)

    if TARGET_SHEET_NAME is None:
        ws = workbook.active
    else:
        if TARGET_SHEET_NAME not in workbook.sheetnames:
            raise Exception(f"Worksheet {TARGET_SHEET_NAME} was not found")

        ws = workbook[TARGET_SHEET_NAME]

    sheet_device_reference = clean_excel_value(ws[DEVICE_NAME_CELL].value)
    last_row = get_last_data_row(ws)

    if last_row < DATA_FIRST_ROW:
        raise Exception("No cable overlay data found")

    print("")
    print("============================================================")
    print("ZUKEN CABLE SHIELD / TWIST OVERLAY")
    print("============================================================")
    print("EXCEL:", excel_path)
    print("WORKSHEET:", ws.title)
    print("DEVICE REFERENCE FROM B1:", sheet_device_reference)
    print("B1 IS NOT USED FOR CONNECTOR LOOKUP")
    print("E3 CONNECTOR ROOT:", CONNECTOR_ROOT_NAME)
    print("P1 LOOKUP BASE:", connector_base_name("P1"))
    print("P1 VIEW PREFIX:", connector_view_prefix("P1") + "*")
    print("DATA ROWS:", DATA_FIRST_ROW, "TO", last_row)

    left_rows = read_side_rows(ws, "LEFT", last_row)
    right_rows = read_side_rows(ws, "RIGHT", last_row)
    left_sections = split_into_sections(left_rows)
    right_sections = split_into_sections(right_rows)
    sections = left_sections + right_sections
    packages = []
    warnings = []

    for section_rows in sections:
        jobs, section_warnings = build_section_jobs(section_rows)

        if jobs:
            package = {"rows": section_rows, "jobs": jobs}
            packages.append(package)

        warnings.extend(section_warnings)

    print("")
    print("CONNECTOR SECTIONS WITH SYMBOLS:", len(packages))

    if warnings:
        print("")
        print("============================================================")
        print("EXCEL WARNINGS")
        print("============================================================")

        for warning in warnings:
            print("WARNING:", warning)

    app = win32.gencache.EnsureDispatch("CT.Application")
    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()
    sym = prj.CreateSymbolObject()

    app.PutMessage("Starting Cable Shield / Twist Overlay Script")

    all_devices = get_all_devices(prj, dev)
    pin_map_cache = {}
    placed_ids = []

    print("")
    print("E3 DEVICE / VIEW COUNT:", len(all_devices))

    for package in packages:
        section_rows = package["rows"]
        jobs = package["jobs"]
        side = section_rows[0]["side"]
        connector = section_rows[0]["connector"]
        required_pins = list(dict.fromkeys(row["pin"] for row in section_rows))
        expected_base = connector_base_name(connector)
        expected_view_prefix = connector_view_prefix(connector)

        print("")
        print("============================================================")
        print("PROCESSING CONNECTOR SECTION")
        print("============================================================")
        print("SIDE:", side)
        print("EXCEL CONNECTOR:", connector)
        print("EXPECTED E3 CONNECTOR:", expected_base)
        print("EXPECTED E3 VIEW PREFIX:", expected_view_prefix + "*")
        print("EXCEL ROWS:", section_rows[0]["excel_row"], "TO", section_rows[-1]["excel_row"])
        print("REQUIRED PINS:", required_pins)

        device_record, pin_map = resolve_connector_view(connector, required_pins, all_devices, dev, pin, pin_map_cache)

        print("MATCHED E3 DEVICE / VIEW ID:", device_record["device_id"])
        print("MATCHED E3 DEVICE / VIEW NAME:", device_record["name"])

        sheet_id = choose_section_sheet(required_pins, pin_map)

        if sheet_id is None:
            raise Exception(f"Could not determine cable sheet for {device_record['name']}")

        print("SELECTED CABLE SHEET ID:", sheet_id)

        jobs = sorted(jobs, key=job_priority)

        for job in jobs:
            try:
                placement = calculate_placement(job, pin_map, sheet_id)
                placed_id = place_job(sym, job, placement)

                if placed_id is not None:
                    placed_ids.append(placed_id)

            except Exception as error:
                print("ERROR:", str(error))
                app.PutMessage(f"Cable overlay warning: {error}")

    print("")
    print("============================================================")
    print("OVERLAY COMPLETE")
    print("============================================================")
    print("PLACED SYMBOL COUNT:", len(placed_ids))
    print("PLACED SYMBOL IDS:", placed_ids)
    print("============================================================")

    app.PutMessage(f"Cable Shield / Twist Overlay complete. Placed {len(placed_ids)} symbols.")

    workbook.close()

    del app
