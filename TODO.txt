# -*- coding: utf-8 -*-
"""READ-ONLY E3 SQL library inspection for top-level plug symbol selection.

Run: py e3_plug_component_definition_probe.py

Purpose: locate the SQL tables/rows associating a component with its MIL
block plug (especially internal) symbol definitions, rather than repeating
PlacePins variations. This changes NO E3 settings, places NO objects, and
writes NOTHING to the Zuken database. It DOES write a JSON report locally.

No E3 application needs to be open. Uses the same SQL Server Windows login
and database as the working Block Maker.

The Zuken database schema may vary. The script inspects the actual schema
rather than assuming a connector-symbol association table name.
"""

from __future__ import annotations
import json
from datetime import datetime
from pathlib import Path

import pyodbc

COMPONENT_CODE = "M24308/4-1F_9P SD"
CONNECTION_STRING = (
    "DRIVER={SQL Server};"
    "SERVER=sql-zuken-prd;"
    "DATABASE=Zuken_E3Library;"
    "Trusted_Connection=yes;"
)
OUTPUT_FILE = Path(__file__).with_name("e3_plug_component_definition_results.json")
MAX_ROWS_PER_TABLE = 12
MAX_TABLES = 80


def sql_ident(v):
    """SQL identifier quoting; identifiers originate from sys.tables/columns."""
    return "[" + str(v).replace("]", "]]") + "]"


def safe_value(v):
    if isinstance(v, (str, bool, int, float)) or v is None:
        return v if not isinstance(v, str) else v[:1500]
    if isinstance(v, (bytes, bytearray, memoryview)):
        return "<binary %d bytes>" % len(v)
    return str(v)[:1500]


def query_rows(cursor, query, params=()):
    cursor.execute(query, *params)
    columns = [col[0] for col in cursor.description]
    rows = [[safe_value(v) for v in row] for row in cursor.fetchall()]
    return columns, rows


def rows_as_dicts(cols, rows):
    return [dict(zip(cols, row)) for row in rows]


def print_records(name, records, n=12):
    print("\n### %s (%d records shown)" % (name, min(len(records), n)), flush=True)
    for record in records[:n]:
        print("  " + json.dumps(record, ensure_ascii=False, default=str)[:1500], flush=True)


def main():
    report = {
        "time": datetime.now().isoformat(timespec="seconds"),
        "component_code": COMPONENT_CODE,
        "mode": "SQL SELECT ONLY; no E3 COM, no project settings, no placements",
        "component_data": [],
        "connector_type_attributes": [],
        "database_symbol_tables": [],
        "component_related_tables": [],
        "interesting_rows": [],
        "warnings": [],
    }
    print("E3 COMPONENT / MIL SYMBOL DEFINITION PROBE (READ ONLY)")
    print("Component:", COMPONENT_CODE)
    conn = pyodbc.connect(CONNECTION_STRING, autocommit=True, timeout=10)
    try:
        cursor = conn.cursor()
        cursor.timeout = 10

        # Two confirmed tables from the working SQL library. These queries
        # establish which component version/gender the database actually has.
        for key, title, sql in (
            ("component_data", "ComponentData",
             "SELECT TOP (10) * FROM dbo.ComponentData WHERE ENTRY = ?"),
            ("connector_type_attributes", "ComponentAttribute (Connector Type)",
             "SELECT TOP (20) * FROM dbo.ComponentAttribute "
             "WHERE ENTRY = ? AND AttributeName IN ('AV_ConnectorType', 'Connector Type')"),
        ):
            try:
                columns, data = query_rows(cursor, sql, (COMPONENT_CODE,))
                report[key] = rows_as_dicts(columns, data)
                print_records(title, report[key], 10)
            except Exception as exc:
                msg = "%s: %s" % (title, exc)
                report["warnings"].append(msg)
                print("WARNING:", msg)

        # Introspect *actual* database schema; symbols can be stored in
        # mapping tables with different names across E3 library versions.
        cursor.execute("""
            SELECT s.name AS SchemaName, t.name AS TableName,
                   c.name AS ColumnName, ty.name AS SqlType
            FROM sys.tables t
            JOIN sys.schemas s ON s.schema_id = t.schema_id
            JOIN sys.columns c ON c.object_id = t.object_id
            JOIN sys.types ty ON ty.user_type_id = c.user_type_id
            WHERE t.is_ms_shipped = 0
            ORDER BY s.name, t.name, c.column_id
        """)
        table_info = {}
        for schema, table, column, typ in cursor.fetchall():
            table_info.setdefault((schema, table), []).append((column, typ))

        symbol_tables = []
        component_link_tables = []
        for (schema, table), columns in table_info.items():
            col_names = [name for name, _ in columns]
            name_low = table.lower()
            column_lows = [col.lower() for col in col_names]
            entry_col = next((x for x in col_names if x.casefold() == "entry"), None)
            symbol_hint = ("sym" in name_low or "graphic" in name_low or "mil" in name_low or
                           any("sym" in x or "graphic" in x or "mil" in x for x in column_lows))
            if symbol_hint:
                symbol_tables.append({
                    "table": schema + "." + table,
                    "columns": col_names,
                })
            if entry_col and (symbol_hint or "comp" in name_low or "connect" in name_low or
                              "view" in name_low or "pin" in name_low or
                              any("ref" in x or "type" in x for x in column_lows)):
                component_link_tables.append((schema, table, entry_col, col_names, symbol_hint))

        report["database_symbol_tables"] = symbol_tables
        print("\nFound %d tables with symbol/graphic/MIL-related names or columns." % len(symbol_tables))
        print("Found %d potentially related tables containing an ENTRY column." % len(component_link_tables))
        print("Symbol-oriented tables:")
        for table in symbol_tables[:100]:
            print("  " + table["table"] + "  " + ", ".join(table["columns"][:24]))

        # Query records for THIS component. Do not guess foreign key names.
        # Each result carries its SQL table and actual column names.
        component_link_tables.sort(key=lambda item: (
            not item[4], "comp" not in item[1].lower(), item[1].lower()
        ))
        for schema, table, entry_col, column_names, symbol_hint in component_link_tables[:MAX_TABLES]:
            full_name = sql_ident(schema) + "." + sql_ident(table)
            try:
                columns, data = query_rows(
                    cursor,
                    "SELECT TOP (%d) * FROM %s WHERE %s = ?" % (
                        MAX_ROWS_PER_TABLE, full_name, sql_ident(entry_col)
                    ),
                    (COMPONENT_CODE,),
                )
                if not data:
                    continue
                record = {
                    "table": schema + "." + table,
                    "columns": columns,
                    "rows": rows_as_dicts(columns, data),
                    "sample_limited_to": MAX_ROWS_PER_TABLE,
                }
                report["component_related_tables"].append(record)
                # Prioritize tables with symbol-oriented columns or actual
                # symbol-library names in their returned values.
                row_repr = json.dumps(record["rows"], ensure_ascii=False, default=str).lower()
                is_interesting = symbol_hint or any(x in row_repr for x in (
                    "milblock", "plugint", "receptacle", "connectortype", "symbol"
                ))
                if is_interesting:
                    report["interesting_rows"].append(record)
                    print("\nMATCHING COMPONENT SYMBOL TABLE: " + record["table"])
                    for row in record["rows"][:5]:
                        print("  " + json.dumps(row, ensure_ascii=False, default=str)[:1800])
            except Exception as exc:
                msg = "Could not query %s: %s" % (schema + "." + table, exc)
                report["warnings"].append(msg)

        print("\n===== SUMMARY =====")
        print("SQL tables referencing this component:", len(report["component_related_tables"]))
        print("Likely MIL/symbol association tables:", len(report["interesting_rows"]))
        if not report["interesting_rows"]:
            print("No obvious component-to-symbol names were found with an ENTRY lookup.")
            print("The report still contains the discovered symbol-table column names")
            print("so a second targeted join can be constructed without guessing.")
        print("Warnings:", len(report["warnings"]))
    finally:
        try:
            cursor.close()
        except Exception:
            pass
        conn.close()
        OUTPUT_FILE.write_text(json.dumps(report, indent=2, ensure_ascii=False, default=str), encoding="utf-8")
        print("\nSaved report:", OUTPUT_FILE, flush=True)


if __name__ == "__main__":
    main()
