import win32com.client as win32


app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sym = prj.CreateSymbolObject()


print("")
print("============================================================")
print("E3 SYMBOL MOVE / POSITION METHOD PROBE")
print("============================================================")


search_words = [
    "move",
    "place",
    "position",
    "location",
    "schema",
    "coord"
]


method_names = []


for name in dir(sym):
    lower_name = name.lower()

    if any(word in lower_name for word in search_words):
        method_names.append(name)


method_names.sort()


for name in method_names:
    print("")
    print("METHOD:", name)

    try:
        method_object = getattr(sym, name)
        print("DOC:", method_object.__doc__)
    except Exception as error:
        print("DOC ERROR:", repr(error))


print("")
print("============================================================")
print("DONE")
print("============================================================")


del sym
del prj
del app
