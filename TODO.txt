
    def failed(error, details):
        status_label.config(text="Formatting failed.")
        run_button.config(state="normal")
        run_again_button.config(state="normal")
        root.config(cursor="")

        show_error_dialog(
            error,
            parent=root,
            details=details
        )

    def worker():
        error = None
        details = None
        com_initialized = False

        try:
            pythoncom.CoInitialize()
            com_initialized = True

            run_ct_code()

        except Exception as e:
            error = e
            details = traceback.format_exc()
            print(details)

        finally:
            if com_initialized:
                pythoncom.CoUninitialize()

        if error is None:
            root.after(0, finished)
        else:
            root.after(
                0,
                lambda err=error, trace=details: failed(err, trace)
            )
