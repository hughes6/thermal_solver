def get_manual_column_count():
    value = column_count_entry.get().strip()

    if value == "":
        return None

    try:
        column_count = int(value)
    except ValueError:
        return None

    if column_count <= 0:
        return None

def normalize_pdf_line(line):
    line = line.replace("\t", " ")
    line = re.sub(r"\s+", " ", line)

    return line.strip()


def is_integer_value(value):
    value = normalize_pdf_line(value)

    return re.fullmatch(r"\d+", value) is not None


# =========================================
# PDF CONNECTOR PINOUT TABLE DETECTION
#
# Handles copied PDF tables such as:
#
# Ref
# Des Pin Signal Name Direction Type Description
# J1R
# 1 NC NC NC No Connect
# 11 GND - GND Secondary ground
# 12 P5_1000BT_D4- Input BASE-T BUS ... pair 4/D, negative
#
# The important detail is that "Ref Des" is often a merged cell in the PDF,
# so the connector designator appears once and the following data rows begin
# with the pin number. Descriptions are allowed to contain commas.
# =========================================

PINOUT_HEADERS = [
    "REF DES",
    "PIN",
    "SIGNAL NAME",
    "DIRECTION",
    "TYPE",
    "DESCRIPTION"
]


def get_nonempty_normalized_lines(input_text):
    lines = []

    for raw_line in input_text.split("\n"):
        line = normalize_pdf_line(raw_line)

        if line != "":
            lines.append(line)

    return lines




# =========================================
# PDF 4-COLUMN PIN / DESCRIPTION / FIGURE / LOGIC TABLE
#
# Handles tables such as the J5 Clock Signals table:
#   Pin Number | Description     | Figure    | Logic
#   1          | Chassis GND     | -         | -
#   6          | 10 MHz+ Input   | Figure 10 | LVDS
#   2          | 10 MHz- Input   | Figure 10 | LVDS
#
# This parser intentionally runs BEFORE the older 3-column Figure-table parser.
# Otherwise a 4-column table can be mistaken for Pin/Description/Figure and the
# Logic column is silently lost.
#
# It supports both normal row-wise clipboard text and Acrobat's tagged/interleaved
# form where the next pin can be emitted alongside the previous row's Figure and
# Logic cells, for example:
#
#   Pin
#   Number
#   Description
#   1
#   Figure
#   Logic
#   6 - -
#   Chassis GND
#   2 Figure 10 LVDS
#   10 MHz+ Input
#   7 Figure 10 LVDS
#   10 MHz- Input
#
# Note that Description values are allowed to BEGIN WITH NUMBERS (10 MHz...), so
# this table family must never locate rows by blindly treating every numeric token
# as a new pin.
# =========================================

FIGURE_LOGIC_PINOUT_HEADERS = [
    "PIN NUMBER",
    "DESCRIPTION",
    "FIGURE",
    "LOGIC"
]


def normalize_dash_cell(value):
    text = normalize_pdf_line(value)
    if text in {"-", "–", "—"}:
        return "-"
    return text


# A description/signal is allowed to START with a number.  The J5 clock table
# contains values such as "10 MHz+ Input" and "10 MHz- Input".  Those leading
# 10s are part of Column 2; they are NOT pin numbers.
#
# Keep this deliberately unit-based instead of saying "every number followed by
# text is a description", because a normal row is also "6 10 MHz+ Input ...".
# In that normal row, the first number (6) is the pin and the *second* number is
# the start of the description.
ENGINEERING_DESCRIPTION_UNIT_RE = re.compile(
    r"(?:Hz|kHz|MHz|GHz|THz|bps|kbps|Mbps|Gbps|Tbps|"
    r"V|mV|uV|µV|A|mA|uA|µA|W|mW|kW|Ohm|ohms|Ω)"
    r"[+\-]?",
    re.IGNORECASE
)


def looks_like_numeric_leading_description(value):
    """
    True when the beginning of a cell is an engineering value such as:
        10 MHz+ Input
        10 MHz- Input
        3.3 V Logic

    This check happens BEFORE pin-row detection so the leading numeric value is
    preserved in Column 2 instead of being stolen and emitted as a pin.
    """
    text = normalize_pdf_line(value)
    if text == "":
        return False

    match = re.match(r"^(\d+(?:\.\d+)?)\s+([^\s]+)(?:\s|$)", text)
    if match is None:
        return False

    unit_token = match.group(2).strip().rstrip(",;:")
    return ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(unit_token) is not None


def next_line_continues_numeric_description(lines, index):
    """
    Handle the even uglier Acrobat form where one cell is broken as:
        10
        MHz+ Input

    An exact integer followed immediately by an engineering-unit fragment is
    treated as description text rather than a new pin.
    """
    if index < 0 or index >= len(lines):
        return False

    current = normalize_pdf_line(lines[index])
    if re.fullmatch(r"\d+(?:\.\d+)?", current) is None:
        return False

    lookahead = index + 1
    while lookahead < len(lines):
        nxt = normalize_pdf_line(lines[lookahead])
        if nxt == "":
            lookahead += 1
            continue
        first = nxt.split()[0].rstrip(",;:")
        return ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(first) is not None

    return False


def is_logic_cell_text(value):
    """Return True for the short electrical-interface values used in Logic columns."""
    text = normalize_dash_cell(value)
    if text == "-":
        return True

    upper = text.upper().replace(" ", "")

    known = {
        "LVDS", "LVTTL", "TTL", "CMOS", "LVCMOS", "ECL", "PECL", "LVPECL",
        "CML", "RS422", "RS-422", "RS485", "RS-485", "SINGLEENDED",
        "SINGLE-ENDED", "DIFFERENTIAL"
    }
    if upper in known:
        return True

    # Voltage-style logic labels such as 3.3V / 5V / 1.8 V.
    if re.fullmatch(r"\d+(?:\.\d+)?V", upper):
        return True

    return False


def is_figure_logic_table_header_fragment(value):
    upper = normalize_pdf_line(value).upper()
    if upper == "":
        return True

    if upper in {
        "PIN", "NUMBER", "PIN NUMBER", "DESCRIPTION", "FIGURE", "LOGIC",
        "LOGIC TYPE", "LOGIC LEVEL"
    }:
        return True

    if upper.startswith("TABLE ") and ("PIN" in upper or "CONNECTOR" in upper):
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True
    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True

    return False


def get_figure_logic_header_window(input_text):
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 4:
        return lines, None

    search_limit = min(len(lines), 40)
    pin_seen = False
    number_seen = False
    description_seen = False
    figure_seen = False
    logic_seen = False
    first_header_index = None
    last_header_index = None

    for index, line in enumerate(lines[:search_limit]):
        compact = re.sub(r"\s+", " ", line.upper()).strip()

        # These tests intentionally work for either individual header cells or
        # a whole header row copied as one line.
        if compact == "PIN" or "PIN NUMBER" in compact:
            pin_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if compact == "NUMBER" or "PIN NUMBER" in compact:
            number_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if re.search(r"\bDESCRIPTION\b", compact):
            description_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        # Require the standalone word Figure, not Figure 10 in a data cell.
        if compact == "FIGURE" or re.search(r"\bDESCRIPTION\b.*\bFIGURE\b", compact):
            figure_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if compact == "LOGIC" or compact.startswith("LOGIC ") or re.search(r"\bFIGURE\b.*\bLOGIC\b", compact):
            logic_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if pin_seen and number_seen and description_seen and figure_seen and logic_seen:
            break

    if not (pin_seen and number_seen and description_seen and figure_seen and logic_seen):
        return lines, None

    return lines, {
        "first": first_header_index,
        "last": last_header_index
    }


def extract_one_figure_cell_prefix(value):
    """
    Return (figure_cell, remainder) when value starts with one Figure-column cell.
    A comma-separated reference such as 'Figure 5, Figure 8' is one cell.
    """
    text = normalize_pdf_line(value)
    match = re.match(
        r"^(Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*)(?:\s+(.*))?$",
        text,
        re.IGNORECASE
    )
    if match is None:
        return "", text
    return normalize_pdf_line(match.group(1)), normalize_pdf_line(match.group(2) or "")


def split_pin_description_figure_logic_payload(payload):
    """
    Split one 4-column payload into Description / Figure / Logic.

    Important: Figure and Logic are independent cells. A dash is a REAL cell
    value, not punctuation to discard. Partial clipboard rows are allowed too,
    for example '10 MHz+ Input Figure 10' with Logic on the next line.
    """
    text = normalize_pdf_line(payload)
    if text == "":
        return "", "", ""

    figure_match = re.search(
        r"\bFigure\s+\d+(?:\s*,\s*Figure\s+\d+)*",
        text,
        re.IGNORECASE
    )
    if figure_match is not None:
        description = text[:figure_match.start()].strip()
        figure = normalize_pdf_line(figure_match.group(0))
        tail = normalize_dash_cell(text[figure_match.end():].strip())
        logic = tail
        return description, figure, logic

    dash_pair = re.match(r"^(.*?)\s+[-–—]\s+[-–—]$", text)
    if dash_pair is not None:
        return dash_pair.group(1).strip(), "-", "-"

    if re.fullmatch(r"[-–—]\s+[-–—]", text):
        return "", "-", "-"

    dash_logic = re.match(r"^(.*?)\s+[-–—]\s+(\S(?:.*\S)?)$", text)
    if dash_logic is not None and is_logic_cell_text(dash_logic.group(2)):
        return dash_logic.group(1).strip(), "-", normalize_dash_cell(dash_logic.group(2))

    single_dash = re.match(r"^(.*?)\s+[-–—]$", text)
    if single_dash is not None and single_dash.group(1).strip() != "":
        return single_dash.group(1).strip(), "-", ""

    return text, "", ""


def parse_normal_pin_description_figure_logic_rows(lines):
    rows = []

    for raw_line in lines:
        line = normalize_pdf_line(raw_line)
        if line == "" or is_figure_logic_table_header_fragment(line):
            continue

        # IMPORTANT: "10 MHz+ Input Figure 10 LVDS" is a Description/Figure/
        # Logic fragment, not pin 10.  Preserve the 10 in Column 2.
        if looks_like_numeric_leading_description(line):
            continue

        match = re.match(r"^(\d+)\s+(.+)$", line)
        if match is None:
            continue

        pin = match.group(1)
        description, figure, logic = split_pin_description_figure_logic_payload(match.group(2))

        # Require actual trailing column evidence.  This is what keeps a line
        # such as '10 MHz+ Input' from being interpreted as pin 10.
        if description == "" or figure == "" or logic == "":
            continue

        rows.append([pin, description, figure, logic])

    return rows


def split_figure_logic_only_payload(value):
    """Return (figure, logic) only when the entire value is those two cells."""
    text = normalize_pdf_line(value)
    if text == "":
        return "", ""

    if re.fullmatch(r"[-–—]\s+[-–—]", text):
        return "-", "-"

    figure, remainder = extract_one_figure_cell_prefix(text)
    if figure and remainder and is_logic_cell_text(remainder):
        return figure, normalize_dash_cell(remainder)

    return "", ""


def parse_stream_pin_description_figure_logic_rows(lines, header_info):
    """
    Reconstruct the 4-column table as independent ordered streams.

    Standalone '-' cells are preserved and assigned in physical column order:
    Figure first, then Logic. Description is always split away from Figure/Logic
    before rows are zipped back together.
    """
    pins = []
    descriptions = []
    figures = []
    logics = []

    start_index = max(0, header_info["first"])

    def append_figure(value):
        value = normalize_dash_cell(value)
        if value != "":
            figures.append(value)

    def append_logic(value):
        value = normalize_dash_cell(value)
        if value != "":
            logics.append(value)

    def append_ambiguous_dash():
        if len(figures) <= len(logics):
            figures.append("-")
        else:
            logics.append("-")

    line_index = start_index
    while line_index < len(lines):
        line = normalize_pdf_line(lines[line_index])
        if line == "" or is_figure_logic_table_header_fragment(line):
            line_index += 1
            continue

        # Protect engineering descriptions that start with a number.  This must
        # run before the generic ^\d+ pin matcher.
        if looks_like_numeric_leading_description(line):
            description, figure, logic = split_pin_description_figure_logic_payload(line)
            if description:
                descriptions.append(description)
            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)
            line_index += 1
            continue

        # Acrobat can split the same description into two physical lines:
        #     10
        #     MHz+ Input [Figure 10] [LVDS]
        # Rejoin those two pieces before pin detection.
        if next_line_continues_numeric_description(lines, line_index):
            next_text = normalize_pdf_line(lines[line_index + 1])
            combined = normalize_pdf_line(line + " " + next_text)
            description, figure, logic = split_pin_description_figure_logic_payload(combined)
            if description:
                descriptions.append(description)
            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)
            line_index += 2
            continue

        dash_tokens = re.findall(r"(?<!\S)[-–—](?!\S)", line)
        if dash_tokens:
            remainder_without_dashes = re.sub(r"(?<!\S)[-–—](?!\S)", " ", line)
            if normalize_pdf_line(remainder_without_dashes) == "":
                for _ in dash_tokens:
                    append_ambiguous_dash()
                line_index += 1
                continue

        pin_match = re.match(r"^(\d+)(?:\s+(.*))?$", line)
        if pin_match is not None:
            pin = pin_match.group(1)
            remainder = normalize_pdf_line(pin_match.group(2) or "")

            if remainder == "":
                pins.append(pin)
                line_index += 1
                continue

            description, figure, logic = split_pin_description_figure_logic_payload(remainder)

            # IMPORTANT FIX:
            # A PDF row can be split after column 2, for example:
            #     6 10 MHz+ Input
            #     Figure 10
            #     LVDS
            # or:
            #     8 1PPS+ Input
            #     Figure 10
            #     LVDS
            #
            # At this point the WHOLE line has already passed the
            # looks_like_numeric_leading_description(line) guard above, so the
            # first number is the pin.  Do not throw the pin away just because
            # Figure/Logic were copied onto later physical lines.
            pins.append(pin)

            if description:
                descriptions.append(description)

            # Do NOT append `remainder` when it contained only Figure/Logic
            # cells (for example the transition marker "2 Figure 10 LVDS").
            # Those trailing cells belong to the previous row in Acrobat's
            # interleaved clipboard order; treating them as Description is what
            # caused Column 3 to leak into Column 2.

            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)

            line_index += 1
            continue

        description, figure, logic = split_pin_description_figure_logic_payload(line)
        if figure or logic:
            if description:
                descriptions.append(description)
            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)
            line_index += 1
            continue

        figure_only, logic_only = split_figure_logic_only_payload(line)
        if figure_only or logic_only:
            if figure_only:
                append_figure(figure_only)
            if logic_only:
                append_logic(logic_only)
            line_index += 1
            continue

        if re.fullmatch(r"Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*", line, re.IGNORECASE):
            append_figure(line)
            line_index += 1
            continue

        if is_logic_cell_text(line) and normalize_dash_cell(line) != "-":
            append_logic(line)
            line_index += 1
            continue

        descriptions.append(line)
        line_index += 1

    if len(pins) < 2:
        return []

    if len(descriptions) < len(pins):
        expanded = []
        for description in descriptions:
            expanded.extend(split_merged_description_cell(description))
        if len(expanded) >= len(descriptions):
            descriptions = expanded

    descriptions = descriptions[:len(pins)]
    figures = figures[:len(pins)]
    logics = logics[:len(pins)]

    while len(descriptions) < len(pins):
        descriptions.append("")
    while len(figures) < len(pins):
        figures.append("")
    while len(logics) < len(pins):
        logics.append("")

    return [[pin, descriptions[i], figures[i], logics[i]] for i, pin in enumerate(pins)]


def expand_figure_logic_trailing_cells(value):
    """
    Expand a physical clipboard line into Figure/Logic cells.  This is used by
    the column-major fallback where Acrobat copies the table one column at a
    time.  Dashes are kept as real cells.
    """
    text = normalize_pdf_line(value)
    if text == "":
        return []

    cells = []
    position = 0
    token_re = re.compile(
        r"Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*|(?<!\S)[-–—](?!\S)|"
        r"LVDS|LVTTL|TTL|CMOS|LVCMOS|ECL|PECL|LVPECL|CML|RS-?422|RS-?485",
        re.IGNORECASE
    )

    for match in token_re.finditer(text):
        gap = normalize_pdf_line(text[position:match.start()])
        if gap:
            # Unknown text means this is probably not a pure trailing-cell line.
            return []
        cells.append(normalize_dash_cell(match.group(0)))
        position = match.end()

    if normalize_pdf_line(text[position:]):
        return []

    return cells


def parse_column_major_pin_description_figure_logic_rows(lines, header_info):
    """
    Fallback for PDFs that copy the J5-style table by columns rather than rows:

        1
        6
        2
        7
        ...
        Chassis GND
        10 MHz+ Input
        10 MHz- Input
        ...
        -
        Figure 10
        Figure 10
        ...
        -
        LVDS
        LVDS
        ...

    The key rule is that once the leading pin block has ended, numeric-leading
    descriptions such as "10 MHz+ Input" are collected as Description cells
    and can never re-enter the pin stream.
    """
    start_index = max(0, header_info["first"])
    data = []
    for line in lines[start_index:]:
        line = normalize_pdf_line(line)
        if line == "" or is_figure_logic_table_header_fragment(line):
            continue
        data.append(line)

    if len(data) < 4:
        return []

    # Column-major copies start with a contiguous run of exact pin cells.
    pins = []
    index = 0
    while index < len(data):
        line = data[index]
        if re.fullmatch(r"\d+", line) is None:
            break

        # Do not steal the first half of a split engineering description.
        if index + 1 < len(data):
            nxt = normalize_pdf_line(data[index + 1])
            first = nxt.split()[0].rstrip(",;:") if nxt else ""
            if ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(first):
                break

        pins.append(line)
        index += 1

    if len(pins) < 2:
        return []

    row_count = len(pins)

    # Collect exactly one Description cell per pin.  Numeric-leading engineering
    # descriptions are preserved verbatim, including the leading number.
    descriptions = []
    pending_trailing = []

    while index < len(data) and len(descriptions) < row_count:
        line = data[index]

        # Handle a description Acrobat split as "10" / "MHz+ Input".
        if re.fullmatch(r"\d+(?:\.\d+)?", line) and index + 1 < len(data):
            nxt = normalize_pdf_line(data[index + 1])
            first = nxt.split()[0].rstrip(",;:") if nxt else ""
            if ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(first):
                line = normalize_pdf_line(line + " " + nxt)
                index += 2
            else:
                index += 1
        else:
            index += 1

        description, figure, logic = split_pin_description_figure_logic_payload(line)

        # A pure Figure/Logic cell means Description collection ended too early;
        # keep it for the trailing columns and stop rather than contaminating col 2.
        if description == "" and (figure or logic):
            if figure:
                pending_trailing.append(figure)
            if logic:
                pending_trailing.append(logic)
            break

        # A pure trailing cell like '-' or 'LVDS' should not become Description.
        pure_cells = expand_figure_logic_trailing_cells(line)
        if pure_cells and (line == "-" or is_logic_cell_text(line) or line.upper().startswith("FIGURE ")):
            pending_trailing.extend(pure_cells)
            break

        if description:
            descriptions.append(description)
            if figure:
                pending_trailing.append(figure)
            if logic:
                pending_trailing.append(logic)
        else:
            descriptions.append(line)

    if len(descriptions) != row_count:
        return []

    trailing_cells = list(pending_trailing)
    while index < len(data):
        line = data[index]
        cells = expand_figure_logic_trailing_cells(line)
        if not cells:
            # Do not accept a candidate that would have to concatenate unknown
            # text into Description/Figure/Logic.
            return []
        trailing_cells.extend(cells)
        index += 1

    # We need exactly N Figure cells followed by N Logic cells.  Extra page text
    # invalidates this conservative fallback instead of shifting columns.
    if len(trailing_cells) < row_count * 2:
        return []

    trailing_cells = trailing_cells[:row_count * 2]
    figures = [normalize_dash_cell(v) for v in trailing_cells[:row_count]]
    logics = [normalize_dash_cell(v) for v in trailing_cells[row_count:row_count * 2]]

    if not all(is_figure_cell_text(v) for v in figures):
        return []
    if not all(is_logic_cell_text(v) for v in logics):
        return []

    return [[pins[i], descriptions[i], figures[i], logics[i]] for i in range(row_count)]


def figure_logic_candidate_score(rows):
    if not rows:
        return (0, 0, 0, 0, 0)

    complete = sum(
        1 for row in rows
        if len(row) >= 4
        and str(row[1]).strip() != ""
        and is_figure_cell_text(row[2])
        and is_logic_cell_text(row[3])
    )
    incomplete = len(rows) - complete
    descriptions = sum(1 for row in rows if len(row) >= 2 and str(row[1]).strip() != "")
    figures = sum(1 for row in rows if len(row) >= 3 and is_figure_cell_text(row[2]))
    logics = sum(1 for row in rows if len(row) >= 4 and is_logic_cell_text(row[3]))

    # Prefer candidates that reconstruct COMPLETE rows.  This prevents a parser
    # that invented an extra numeric pin from beating a shorter but correct
    # candidate merely because it returned more rows.
    return (complete, -incomplete, descriptions, min(figures, logics), len(rows))


def is_valid_figure_logic_rows(rows):
    if len(rows) < 2:
        return False

    complete = sum(
        1 for row in rows
        if len(row) >= 4
        and str(row[1]).strip() != ""
        and is_figure_cell_text(row[2])
        and is_logic_cell_text(row[3])
    )
    minimum_complete = max(2, (len(rows) + 1) // 2)
    return complete >= minimum_complete


def looks_like_pdf_pin_description_figure_logic_table(input_text):
    lines, header_info = get_figure_logic_header_window(input_text)
    if header_info is None:
        return False

    # Use line starts, not every numeric token, because Description values can
    # legitimately begin with numbers (10 MHz+ Input / 10 MHz- Input).
    candidates = 0
    for line in lines[header_info["first"] + 1:]:
        if re.match(r"^\d+(?:\s|$)", line):
            candidates += 1

    return candidates >= 2


def parse_pdf_pin_description_figure_logic_table(input_text):
    lines, header_info = get_figure_logic_header_window(input_text)
    rows = [list(FIGURE_LOGIC_PINOUT_HEADERS)]

    if header_info is None:
        return rows

    normal_rows = parse_normal_pin_description_figure_logic_rows(lines[header_info["first"]:])
    stream_rows = parse_stream_pin_description_figure_logic_rows(lines, header_info)
    column_major_rows = parse_column_major_pin_description_figure_logic_rows(lines, header_info)

    candidates = [
        candidate
        # Column-major reconstruction is deliberately ahead of the generic
        # stream candidate.  When both are equally complete, the column-major
        # parser has stronger structural evidence and keeps dash cells in their
        # real Figure/Logic columns.
        for candidate in (normal_rows, column_major_rows, stream_rows)
        if candidate
    ]
    if not candidates:
        return rows

    selected_rows = max(candidates, key=figure_logic_candidate_score)
    rows.extend(selected_rows)
    return rows


def parse_headerless_pin_description_figure_logic_rows(input_text):
    """Conservative body-only recovery for rows such as '1 Chassis GND - -'."""
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 2:
        return []

    # A real header belongs to the normal parser.
    _, header_info = get_figure_logic_header_window(input_text)
    if header_info is not None:
        return []

    normal_rows = parse_normal_pin_description_figure_logic_rows(lines)
    if is_valid_figure_logic_rows(normal_rows):
        return normal_rows

    # Headerless interleaved recovery starts at the first exact pin line.  It is
    # intentionally conservative so this path does not steal unrelated text.
    first_pin_index = None
    for index, line in enumerate(lines):
        if re.fullmatch(r"\d+", line):
            if next_line_continues_numeric_description(lines, index):
                continue
            first_pin_index = index
            break

    if first_pin_index is None:
        return []

    stream_rows = parse_stream_pin_description_figure_logic_rows(
        lines,
        {"first": first_pin_index, "last": first_pin_index}
    )
    if is_valid_figure_logic_rows(stream_rows):
        return stream_rows

    return []


def looks_like_headerless_pin_description_figure_logic_table(input_text):
    return len(parse_headerless_pin_description_figure_logic_rows(input_text)) >= 2


def parse_headerless_pin_description_figure_logic_table(input_text):
    rows = [list(FIGURE_LOGIC_PINOUT_HEADERS)]
    rows.extend(parse_headerless_pin_description_figure_logic_rows(input_text))
    return rows


# =========================================
# PDF 3-COLUMN PIN / DESCRIPTION / FIGURE TABLE
#
# Handles tables such as:
#   Pin Number | Description | Figure
#
# Some tagged PDFs copy this table in a staggered order.  A typical clipboard
# stream looks like:
#
#   Pin
#   Number
#   Description
#   1
#   Figure
#   19 -
#   Chassis GND
#   36 Figure 5, Figure 8
#   TEMP (Thermistor)
#
# In that form the NEXT pin is emitted on the same line as the PREVIOUS row's
# Figure cell, and the previous row's Description follows on the next line(s).
# The parser below reconstructs the rows instead of treating every newline as a
# new spreadsheet row.
# =========================================

FIGURE_PINOUT_HEADERS = [
    "PIN NUMBER",
    "DESCRIPTION",
    "FIGURE"
]


def is_figure_cell_text(value):
    text = normalize_pdf_line(value)
    if text == "":
        return False

    # The Figure column commonly uses a plain dash for "no figure".  Some PDFs
    # copy that glyph as an en/em dash, so accept those forms too.
    if text in {"-", "–", "—"}:
        return True

    return re.match(r"^FIGURE\b", text, re.IGNORECASE) is not None


def is_figure_table_header_fragment(value):
    upper = normalize_pdf_line(value).upper()
    if upper in {"PIN", "NUMBER", "PIN NUMBER", "DESCRIPTION", "FIGURE"}:
        return True
    if upper.startswith("TABLE ") and ("PIN" in upper or "CONNECTOR" in upper):
        return True
    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True
    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True
    return False


def get_figure_pinout_header_window(input_text):
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 5:
        return lines, None

    search_limit = min(len(lines), 30)
    upper_lines = [line.upper() for line in lines[:search_limit]]

    pin_seen = False
    number_seen = False
    description_seen = False
    figure_seen = False
    first_header_index = None
    last_header_index = None

    for index, upper in enumerate(upper_lines):
        compact = re.sub(r"\s+", " ", upper).strip()

        if compact == "PIN" or compact.startswith("PIN NUMBER") or "PIN NUMBER" in compact:
            pin_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if compact == "NUMBER" or "PIN NUMBER" in compact:
            number_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if re.search(r"\bDESCRIPTION\b", compact):
            description_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if re.search(r"\bFIGURE\b", compact):
            figure_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        # Do not scan forever into the data.  Once every required header has
        # appeared, the window is known even if Figure was emitted after pin 1.
        if pin_seen and number_seen and description_seen and figure_seen:
            break

    if not (pin_seen and number_seen and description_seen and figure_seen):
        return lines, None

    return lines, {
        "first": first_header_index,
        "last": last_header_index
    }


def split_pin_description_figure_payload(payload):
    payload = normalize_pdf_line(payload)
    if payload == "":
        return "", ""

    figure_match = re.search(r"\bFIGURE\b", payload, re.IGNORECASE)
    if figure_match is not None:
        description = payload[:figure_match.start()].strip()
        figure = payload[figure_match.start():].strip()
        return description, figure

    # A dash is a legitimate Figure-column value in this table family.  Accept
    # the common Unicode dash variants Acrobat may put on the clipboard.
    dash_match = re.search(r"(?:^|\s)[-–—](?:\s*)$", payload)
    if dash_match is not None:
        description = payload[:dash_match.start()].strip()
        return description, "-"

    return payload, ""


def parse_normal_pin_description_figure_rows(lines):
    rows = []

    for line in lines:
        if is_figure_table_header_fragment(line):
            continue

        # A Description such as "10 MHz+ Input Figure 10" must not create
        # phantom pin 10 or lose the leading 10 from Column 2.
        if looks_like_numeric_leading_description(line):
            continue

        match = re.match(r"^(\d+)\s+(.+)$", line)
        if match is None:
            continue

        pin = match.group(1)
        payload = match.group(2).strip()
        description, figure = split_pin_description_figure_payload(payload)

        # A line such as "19 -" or "36 Figure 5, Figure 8" is the staggered
        # form's transition marker, not a complete normal row.
        if description == "":
            continue

        rows.append([pin, description, figure])

    return rows


def get_staggered_figure_markers(lines, start_index):
    markers = []

    for index in range(start_index, len(lines)):
        line = lines[index]

        if is_figure_table_header_fragment(line):
            continue

        exact_pin = re.fullmatch(r"\d+", line)
        if exact_pin is not None:
            markers.append({"index": index, "pin": exact_pin.group(0), "figure": ""})
            continue

        match = re.match(r"^(\d+)\s+(.+)$", line)
        if match is None:
            continue

        remainder = normalize_pdf_line(match.group(2))
        if is_figure_cell_text(remainder):
            markers.append({
                "index": index,
                "pin": match.group(1),
                "figure": remainder
            })

    return markers


def clean_figure_description_lines(values):
    cleaned = []
    for value in values:
        text = normalize_pdf_line(value)
        if text == "" or is_figure_table_header_fragment(text):
            continue
        cleaned.append(text)
    return " ".join(cleaned).strip()


def extract_figure_cells_from_text(value):
    """
    Split one clipboard line into Figure-column cells without confusing a
    multi-reference cell such as "Figure 5, Figure 8" with two rows.

    Also handles PDFs that collapse many blank Figure cells onto one line:
        "- - - -" -> ["-", "-", "-", "-"]
    """
    text = normalize_pdf_line(value)
    if text == "":
        return []

    # One or more dash-only Figure cells.  This is important for continuation
    # tables like the Chassis GND block where every Figure cell is '-'.
    dash_tokens = re.findall(r"(?<!\S)[-–—](?!\S)", text)
    if dash_tokens:
        remainder = re.sub(r"(?<!\S)[-–—](?!\S)", " ", text)
        remainder = re.sub(r"\s+", "", remainder)
        if remainder == "":
            return ["-"] * len(dash_tokens)

    # A comma joins references that belong to ONE Figure cell.  A new Figure
    # token not introduced by a comma starts the next cell.
    pattern = re.compile(
        r"Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*",
        re.IGNORECASE
    )
    matches = [normalize_pdf_line(match.group(0)) for match in pattern.finditer(text)]
    return matches


def description_start_positions(text):
    """
    Return strong candidate starts of a second Description cell on a line.

    Some tagged PDFs collapse TWO description cells onto one clipboard line.
    We only split at signal-like all-caps identifiers, and only use this as a
    recovery step when there are fewer description cells than pin numbers.
    """
    positions = []

    # Signal-like identifiers seen in connector tables.  Require either an
    # underscore-containing identifier or TEMP/TEMP_RTN so ordinary prose is
    # not split merely because it contains a capitalized word.
    pattern = re.compile(
        r"(?<!\S)(?:[A-Z][A-Z0-9]*_[A-Z0-9_+\-./]+|TEMP(?:_RTN)?)(?=\s|$|\()"
    )

    for match in pattern.finditer(text):
        if match.start() > 0:
            positions.append(match.start())

    return positions


def split_repeated_description_cells(value):
    """
    Recover description cells when a PDF merges repeated neighboring cells.

    Example from the continuation table shown by the user:
        "Chassis GND Chassis GND Chassis GND"
    becomes:
        ["Chassis GND", "Chassis GND", "Chassis GND"]

    Only exact repeated token phrases are split, which keeps ordinary prose
    intact.
    """
    text = normalize_pdf_line(value)
    if text == "":
        return []

    tokens = text.split()
    if len(tokens) < 2:
        return [text]

    # Description cells in these pinout tables are normally short.  Limiting
    # the repeated unit to eight tokens keeps this recovery conservative.
    max_unit = min(8, len(tokens) // 2)

    for unit_length in range(1, max_unit + 1):
        if len(tokens) % unit_length != 0:
            continue

        repeat_count = len(tokens) // unit_length
        if repeat_count < 2:
            continue

        unit = tokens[:unit_length]
        if unit * repeat_count == tokens:
            phrase = " ".join(unit)
            return [phrase] * repeat_count

    return [text]


def split_merged_description_cell(value):
    text = normalize_pdf_line(value)
    if text == "":
        return []

    # First handle exact repeated phrases such as a whole run of "Chassis GND"
    # cells merged onto one clipboard line.
    repeated_parts = split_repeated_description_cells(text)
    if len(repeated_parts) > 1:
        return repeated_parts

    positions = description_start_positions(text)
    if not positions:
        return [text]

    # Split at every strong later signal start.  The first chunk remains intact
    # (e.g. "Chassis GND"), then each following signal begins a new cell.
    parts = []
    start = 0
    for position in positions:
        part = text[start:position].strip()
        if part:
            parts.append(part)
        start = position

    tail = text[start:].strip()
    if tail:
        parts.append(tail)

    return parts if parts else [text]


def parse_stream_pin_description_figure_rows(lines, header_info):
    """
    Reconstruct the 3-column table as independent ordered streams.

    Description is split from Figure before row reconstruction so Figure text
    and dash cells cannot leak into Column 2.
    """
    pins = []
    descriptions = []
    figures = []
    start_index = max(0, header_info["first"])

    line_index = start_index
    while line_index < len(lines):
        line = normalize_pdf_line(lines[line_index])
        if line == "" or is_figure_table_header_fragment(line):
            line_index += 1
            continue

        if looks_like_numeric_leading_description(line):
            description, figure = split_pin_description_figure_payload(line)
            if description:
                descriptions.append(description)
            if figure:
                figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
            line_index += 1
            continue

        if next_line_continues_numeric_description(lines, line_index):
            next_text = normalize_pdf_line(lines[line_index + 1])
            combined = normalize_pdf_line(line + " " + next_text)
            description, figure = split_pin_description_figure_payload(combined)
            if description:
                descriptions.append(description)
            if figure:
                figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
            line_index += 2
            continue

        pin_match = re.match(r"^(\d+)(?:\s+(.*))?$", line)
        if pin_match is not None:
            pin = pin_match.group(1)
            remainder = normalize_pdf_line(pin_match.group(2) or "")

            if remainder == "":
                pins.append(pin)
                line_index += 1
                continue

            description, figure = split_pin_description_figure_payload(remainder)
            if figure:
                pins.append(pin)
                if description:
                    descriptions.append(description)
                figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
                line_index += 1
                continue

            descriptions.append(line)
            line_index += 1
            continue

        description, figure = split_pin_description_figure_payload(line)
        if figure:
            if description:
                descriptions.append(description)
            figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
            line_index += 1
            continue

        figure_cells = extract_figure_cells_from_text(line)
        if figure_cells:
            figures.extend(figure_cells)
            line_index += 1
            continue

        if line in {"-", "–", "—"}:
            figures.append("-")
            line_index += 1
            continue

        descriptions.append(line)
        line_index += 1

    if not pins:
        return []

    if len(descriptions) < len(pins):
        expanded = []
        for description in descriptions:
            expanded.extend(split_merged_description_cell(description))
        if len(expanded) >= len(descriptions):
            descriptions = expanded

    descriptions = descriptions[:len(pins)]
    figures = figures[:len(pins)]
    while len(descriptions) < len(pins):
        descriptions.append("")
    while len(figures) < len(pins):
        figures.append("")

    return [[pin, descriptions[i], figures[i]] for i, pin in enumerate(pins)]

def parse_staggered_pin_description_figure_rows(lines, header_info):
    """Legacy adjacency parser retained only as a fallback."""
    start_index = max(0, header_info["first"])
    markers = get_staggered_figure_markers(lines, start_index)

    if len(markers) < 2:
        return []

    rows = []

    for marker_index in range(1, len(markers)):
        previous_marker = markers[marker_index - 1]
        current_marker = markers[marker_index]
        next_line_index = markers[marker_index + 1]["index"] if marker_index + 1 < len(markers) else len(lines)
        description_slice = lines[current_marker["index"] + 1:next_line_index]

        terminal_figure_offset = None
        if marker_index == len(markers) - 1:
            for offset, value in enumerate(description_slice):
                if is_figure_cell_text(value):
                    terminal_figure_offset = offset
                    break

        if terminal_figure_offset is None:
            previous_description_lines = description_slice
        else:
            previous_description_lines = description_slice[:terminal_figure_offset]

        description = clean_figure_description_lines(previous_description_lines)
        figure = current_marker["figure"]

        if description != "":
            rows.append([previous_marker["pin"], description, figure])

        if marker_index == len(markers) - 1 and terminal_figure_offset is not None:
            terminal_figure = normalize_pdf_line(description_slice[terminal_figure_offset])
            terminal_description = clean_figure_description_lines(description_slice[terminal_figure_offset + 1:])
            if terminal_description != "":
                rows.append([current_marker["pin"], terminal_description, terminal_figure])

    return rows


def looks_like_pdf_pin_description_figure_table(input_text):
    lines, header_info = get_figure_pinout_header_window(input_text)
    if header_info is None:
        return False

    # A visible Logic header means this is the 4-column family handled above.
    header_probe = " ".join(lines[:40]).upper()
    if re.search(r"\bLOGIC\b", header_probe):
        return False

    data_candidates = 0
    for line in lines[header_info["first"] + 1:]:
        if re.match(r"^\d+(?:\s|$)", line):
            data_candidates += 1

    return data_candidates >= 2


def parse_compact_pin_description_figure_columns(lines, start_index=0):
    """
    Recover a 3-column pin table when the PDF collapses an entire visual column
    onto one clipboard line, for example:

        47 13 31 48 14 32
        Chassis GND Chassis GND Chassis GND Chassis GND Chassis GND Chassis GND
        - - - - - -

    The visual pin order is preserved exactly; pin numbers do NOT need to be
    ascending or consecutive.
    """
    pins = []
    descriptions = []
    figures = []
    pin_block_found = False

    for raw_line in lines[start_index:]:
        line = normalize_pdf_line(raw_line)
        if line == "" or is_figure_table_header_fragment(line):
            continue

        tokens = line.split()

        # A compact pin-number column: at least two integer cells on one line.
        if not pin_block_found and len(tokens) >= 2 and all(is_integer_value(token) for token in tokens):
            pins.extend(tokens)
            pin_block_found = True
            continue

        if not pin_block_found:
            continue

        figure_cells = extract_figure_cells_from_text(line)
        if figure_cells:
            # Only accept the line as a Figure block when all of its visible
            # content is made from Figure cells/dashes.  This prevents a normal
            # description containing the word "Figure" from being stolen.
            dash_only = re.fullmatch(r"(?:\s*[-–—]\s*)+", line) is not None
            figure_only = re.fullmatch(
                r"\s*Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*(?:\s+Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*)*\s*",
                line,
                re.IGNORECASE
            ) is not None

            if dash_only or figure_only:
                figures.extend(figure_cells)
                continue

        descriptions.extend(split_merged_description_cell(line))

    if len(pins) < 2:
        return []

    descriptions = descriptions[:len(pins)]
    figures = figures[:len(pins)]

    while len(descriptions) < len(pins):
        descriptions.append("")
    while len(figures) < len(pins):
        figures.append("")

    return [
        [pin, descriptions[index], figures[index]]
        for index, pin in enumerate(pins)
    ]


def figure_table_candidate_score(rows):
    """Score a possible 3-column Pin / Description / Figure reconstruction."""
    if not rows:
        return (0, 0, 0)

    description_count = sum(
        1 for row in rows
        if len(row) >= 2 and str(row[1]).strip() != ""
    )
    figure_count = sum(
        1 for row in rows
        if len(row) >= 3 and is_figure_cell_text(row[2])
    )

    return (len(rows), description_count, figure_count)


def is_valid_headerless_figure_table_rows(rows):
    """
    Conservative validation for a headerless continuation table.

    Requiring real Figure-column markers is what keeps an arbitrary list of
    numbers and text from being misdetected as this special 3-column format.
    """
    if len(rows) < 3:
        return False

    description_count = sum(
        1 for row in rows
        if len(row) >= 2 and str(row[1]).strip() != ""
    )
    figure_count = sum(
        1 for row in rows
        if len(row) >= 3 and is_figure_cell_text(row[2])
    )

    minimum_descriptions = max(2, (len(rows) + 1) // 2)
    minimum_figures = max(2, (len(rows) + 1) // 2)

    return (
        description_count >= minimum_descriptions
        and figure_count >= minimum_figures
    )


def parse_headerless_pin_description_figure_rows(input_text):
    """
    Parse a continuation/body-only Pin Number | Description | Figure table.

    This is specifically for selections like the screenshot where the visible
    rows are only:
        47   Chassis GND   -
        13   Chassis GND   -
        31   Chassis GND   -
        ...
    and the Pin Number / Description / Figure header is not part of the copy.
    """
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 3:
        return []

    # If the real header is present, the normal parser should own this input.
    _, header_info = get_figure_pinout_header_window(input_text)
    if header_info is not None:
        return []

    # Do not steal the 4-column Figure/Logic body-only family.
    if looks_like_headerless_pin_description_figure_logic_table(input_text):
        return []

    # Do not steal the other known pinout families when their headers are clear.
    header_probe = " ".join(lines[:20]).upper()
    if "SIGNAL NAME" in header_probe and (
        "SIGNAL TYPE" in header_probe
        or "DIRECTION" in header_probe
    ):
        return []

    candidates = []

    # 1) Normal row-wise copy: 47 Chassis GND -
    normal_rows = parse_normal_pin_description_figure_rows(lines)
    if is_valid_headerless_figure_table_rows(normal_rows):
        candidates.append(normal_rows)

    # 2) Tagged/interleaved/column-major copy.  Start at the first numeric row so
    # page titles before the continuation block cannot shift the descriptions.
    first_pin_index = None
    for index, line in enumerate(lines):
        if re.match(r"^\d+(?:\s|$)", line):
            first_pin_index = index
            break

    if first_pin_index is not None:
        synthetic_header = {"first": first_pin_index, "last": first_pin_index}
        stream_rows = parse_stream_pin_description_figure_rows(lines, synthetic_header)
        if is_valid_headerless_figure_table_rows(stream_rows):
            candidates.append(stream_rows)

        # 3) Entire visual columns collapsed onto single lines.
        compact_rows = parse_compact_pin_description_figure_columns(lines, first_pin_index)
        if is_valid_headerless_figure_table_rows(compact_rows):
            candidates.append(compact_rows)

    if not candidates:
        return []

    # Prefer the candidate with the most complete Figure cells when row counts
    # are equal; those markers are the strongest evidence of correct alignment.
    return max(
        candidates,
        key=lambda rows: (
            figure_table_candidate_score(rows)[0],
            figure_table_candidate_score(rows)[2],
            figure_table_candidate_score(rows)[1]
        )
    )


def looks_like_headerless_pin_description_figure_table(input_text):
    return len(parse_headerless_pin_description_figure_rows(input_text)) >= 3


def parse_headerless_pin_description_figure_table(input_text):
    rows = [list(FIGURE_PINOUT_HEADERS)]
    rows.extend(parse_headerless_pin_description_figure_rows(input_text))
    return rows


def parse_pdf_pin_description_figure_table(input_text):
    lines, header_info = get_figure_pinout_header_window(input_text)
    rows = [list(FIGURE_PINOUT_HEADERS)]

    if header_info is None:
        return rows

    normal_rows = parse_normal_pin_description_figure_rows(lines[header_info["first"]:])
    stream_rows = parse_stream_pin_description_figure_rows(lines, header_info)
    legacy_rows = parse_staggered_pin_description_figure_rows(lines, header_info)
    compact_rows = parse_compact_pin_description_figure_columns(lines, header_info["first"])

    # Prefer stream reconstruction whenever it found the same or more rows than
    # the alternatives.  It is specifically designed for PDFs whose clipboard
    # tags interleave the three physical columns.
    candidates = [normal_rows, legacy_rows, stream_rows, compact_rows]
    selected_rows = max(
        candidates,
        key=lambda candidate: (
            len(candidate),
            sum(1 for row in candidate if len(row) >= 2 and str(row[1]).strip() != ""),
            1 if candidate is stream_rows else 0
        )
    )

    rows.extend(selected_rows)
    return rows


# =========================================
# PDF 4-COLUMN POWER / SIMPLE PINOUT TABLE DETECTION
#
# Handles copied PDF tables such as:
#
# TABLE 5 - POWER CONNECTOR PINOUT
# Pin #   Signal Name   Signal Type   Signal Description
# 1       VBUS          Power         22-36Vdc Power Input
# 2       VBUS          Power         22-36Vdc Power Input
# 3       INHIBIT       Power         Enable
# 4       BUS_RTN       Power         Board Ground
#
# This table family has NO Ref Des column and NO Direction column.  It is
# intentionally detected before the 6-column pinout parser so the words
# "Signal Type" / "Signal Description" cannot be mistaken for that format.
# =========================================

SIMPLE_PINOUT_HEADERS = [
    "PIN #",
    "SIGNAL NAME",
    "SIGNAL TYPE",
    "SIGNAL DESCRIPTION"
]


def get_flat_pdf_tokens(input_text):
    """Return a whitespace token stream regardless of how Acrobat inserted newlines."""
    tokens = []

    for raw_line in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        line = normalize_pdf_line(raw_line)

        if line == "":
            continue

        tokens.extend(line.split())

    return tokens


def normalized_header_token(value):
    value = str(value or "").strip().upper()
    value = value.replace("–", "-").replace("—", "-")
    return value.strip(":")


def find_token_pair(tokens, first, second, start=0, stop=None):
    upper_tokens = [normalized_header_token(token) for token in tokens]
    if stop is None:
        stop = len(tokens)

    for index in range(max(0, start), min(stop, len(tokens) - 1)):
        if upper_tokens[index] == first and upper_tokens[index + 1] == second:
            return index

    return None


def find_simple_pinout_header_end(tokens):
    """
    Find the end of:
        Pin # | Signal Name | Signal Type | Signal Description

    This works whether those headers arrive as four normal lines, individual
    cells on separate lines, or literally one word per line.
    """
    upper_tokens = [normalized_header_token(token) for token in tokens]
    search_stop = min(len(tokens), 120)

    for description_index in range(search_stop - 1):
        if not (
            upper_tokens[description_index] == "SIGNAL"
            and upper_tokens[description_index + 1] == "DESCRIPTION"
        ):
            continue

        before = upper_tokens[max(0, description_index - 20):description_index]
        before_text = " ".join(before)

        if "SIGNAL NAME" not in before_text:
            continue

        if "SIGNAL TYPE" not in before_text:
            continue

        if "PIN" not in before:
            continue

        return description_index + 2

    return None


def find_connector_pinout_header_end(tokens):
    """
    Find the end of:
        Ref Des | Pin | Signal Name | Direction | Type | Description

    The header can be split at arbitrary newline boundaries.
    """
    upper_tokens = [normalized_header_token(token) for token in tokens]
    search_stop = min(len(tokens), 140)

    for description_index in range(search_stop):
        if upper_tokens[description_index] != "DESCRIPTION":
            continue

        before = upper_tokens[max(0, description_index - 24):description_index]
        before_text = " ".join(before)

        if "SIGNAL NAME" not in before_text:
            continue

        if "DIRECTION" not in before:
            continue

        if "TYPE" not in before:
            continue

        if "PIN" not in before:
            continue

        if "REF" not in before or "DES" not in before:
            continue

        # A 4-column table has the phrase SIGNAL TYPE.  Do not steal it.
        if "SIGNAL TYPE" in before_text:
            continue

        return description_index + 1

    return None


def looks_highly_verticalized(input_text):
    """Heuristic used only for the status text, not for deciding correctness."""
    lines = get_nonempty_normalized_lines(input_text)

    if len(lines) < 6:
        return False

    short_lines = sum(1 for line in lines if len(line.split()) <= 1)
    return short_lines / len(lines) >= 0.55


def select_pin_token_indices(tokens, data_start, minimum_payload_tokens):
    """
    Pick the most plausible increasing pin-number sequence from a flattened PDF
    token stream.  This prevents an isolated numeric value in a description from
    normally being mistaken for a new pin row.
    """
    candidates = []

    for index in range(data_start, len(tokens)):
        value = tokens[index]

        if not is_integer_value(value):
            continue

        candidates.append((index, int(value)))

    if not candidates:
        return []

    # DP state: best path ending at each candidate.  Prefer more rows first,
    # then consecutive pin numbering, then a smaller first pin number.
    states = []

    for candidate_index, (token_index, pin_number) in enumerate(candidates):
        best_path = [candidate_index]
        best_count = 1
        best_consecutive = 0
        best_first_pin = pin_number

        for previous_candidate_index in range(candidate_index):
            previous_token_index, previous_pin_number = candidates[previous_candidate_index]

            if pin_number <= previous_pin_number:
                continue

            payload_end = token_index

            # If a connector designator sits immediately before this pin, it
            # belongs to this row, not the previous row's Description.
            if token_index - 1 > previous_token_index:
                possible_ref = tokens[token_index - 1]
                if is_pinout_ref_designator(possible_ref):
                    payload_end -= 1

            payload_count = payload_end - previous_token_index - 1

            if payload_count < minimum_payload_tokens:
                continue

            previous_state = states[previous_candidate_index]
            path = previous_state["path"] + [candidate_index]
            count = previous_state["count"] + 1
            consecutive = previous_state["consecutive"] + (1 if pin_number == previous_pin_number + 1 else 0)
            first_pin = previous_state["first_pin"]

            score = (count, consecutive, 1 if first_pin == 1 else 0, -first_pin)
            best_score = (best_count, best_consecutive, 1 if best_first_pin == 1 else 0, -best_first_pin)

            if score > best_score:
                best_path = path
                best_count = count
                best_consecutive = consecutive
                best_first_pin = first_pin

        states.append({
            "path": best_path,
            "count": best_count,
            "consecutive": best_consecutive,
            "first_pin": best_first_pin
        })

    valid_states = []

    for state in states:
        last_candidate_index = state["path"][-1]
        last_token_index = candidates[last_candidate_index][0]
        remaining_payload = len(tokens) - last_token_index - 1

        if remaining_payload >= minimum_payload_tokens:
            valid_states.append(state)

    if not valid_states:
        valid_states = states

    valid_states.sort(
        key=lambda state: (
            state["count"],
            state["consecutive"],
            1 if state["first_pin"] == 1 else 0,
            -state["first_pin"]
        ),
        reverse=True
    )

    selected = valid_states[0]["path"]
    return [candidates[index][0] for index in selected]


def looks_like_pdf_simple_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_simple_pinout_header_end(tokens)

    if header_end is None:
        return False

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=2)
    return len(pin_indices) >= 2


def is_simple_pinout_header_line(line):
    upper_text = normalize_pdf_line(line).upper()

    if upper_text == "":
        return True

    if upper_text in {
        "PIN",
        "PIN #",
        "SIGNAL NAME",
        "SIGNAL TYPE",
        "SIGNAL DESCRIPTION"
    }:
        return True

    if (
        "PIN" in upper_text
        and "SIGNAL NAME" in upper_text
        and "SIGNAL TYPE" in upper_text
        and "SIGNAL DESCRIPTION" in upper_text
    ):
        return True

    if upper_text.startswith("TABLE ") and "PINOUT" in upper_text:
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    return False


SIGNAL_START_JOIN_CHARS = "_-/#+.:"
SIGNAL_END_JOIN_CHARS = "_/#.:"
SIGNAL_STANDALONE_JOIN_CHARS = "_-/#+.:"


def token_starts_with_signal_joiner(value):
    text = str(value or "").strip()
    return text != "" and text[0] in SIGNAL_START_JOIN_CHARS


def token_ends_with_signal_joiner(value):
    text = str(value or "").strip()
    # Do not treat a trailing + or - as proof that the next word belongs to the
    # signal.  Differential signals commonly end in + / -, e.g. D4- Input.
    return text != "" and text[-1] in SIGNAL_END_JOIN_CHARS


def token_is_signal_joiner(value):
    text = str(value or "").strip()
    return text != "" and all(char in SIGNAL_STANDALONE_JOIN_CHARS for char in text)


def signal_tokens_are_continuous(previous_value, next_value):
    """
    Acrobat occasionally breaks an identifier at punctuation, for example:

        BUS_  RTN
        BUS   _RTN
        BUS   _   RTN
        P5_   1000BT_D4-

    Those are one Signal Name, not extra PDF columns.  Only fuse tokens when
    punctuation gives us strong evidence that the identifier continues.  This
    avoids swallowing an ordinary Signal Type such as "Power".
    """
    previous_text = str(previous_value or "").strip()
    next_text = str(next_value or "").strip()

    if previous_text == "" or next_text == "":
        return False

    if token_ends_with_signal_joiner(previous_text):
        return True

    if token_starts_with_signal_joiner(next_text):
        return True

    if token_is_signal_joiner(previous_text) or token_is_signal_joiner(next_text):
        return True

    return False


def normalize_signal_name_fragments(fragments):
    fragments = [str(value or "").strip() for value in fragments if str(value or "").strip() != ""]

    if not fragments:
        return ""

    # Preserve a real two-word mnemonic that occurs in the older table family.
    if len(fragments) == 2 and fragments[0].upper() == "CTU" and fragments[1].upper() == "RTN":
        return "CTU RTN"

    text = " ".join(fragments)

    # Rejoin identifiers that Acrobat split immediately before/after connector
    # punctuation.  Repeating the substitution handles forms such as
    # "BUS _ RTN" as well as "P5_ 1000BT_D4-".
    previous = None
    while previous != text:
        previous = text
        text = re.sub(r"\s*([_\-/#+.:])\s*", r"\1", text)

    return text.strip()


def consume_signal_name_tokens(payload):
    """Return (signal_name, number_of_payload_tokens_consumed)."""
    if not payload:
        return "", 0

    fragments = [payload[0]]
    index = 1

    while index < len(payload):
        current_signal = normalize_signal_name_fragments(fragments)
        next_value = payload[index]

        if signal_tokens_are_continuous(current_signal, next_value):
            fragments.append(next_value)
            index += 1
            continue

        # Existing PDFs also contain the legitimate two-word signal CTU RTN.
        if current_signal.upper() == "CTU" and str(next_value).strip().upper() == "RTN":
            fragments.append(next_value)
            index += 1
            continue

        break

    return normalize_signal_name_fragments(fragments), index


def parse_simple_pinout_data_line(line):
    line = normalize_pdf_line(line)

    if line == "":
        return None

    tokens = line.split()

    if len(tokens) < 3:
        return None

    if not is_integer_value(tokens[0]):
        return None

    pin = tokens.pop(0)
    signal_name, consumed = consume_signal_name_tokens(tokens)

    if consumed >= len(tokens):
        return None

    signal_type = tokens[consumed]
    description = " ".join(tokens[consumed + 1:])

    return [pin, signal_name, signal_type, description]


def parse_pdf_simple_pinout_table(input_text):
    # Flattening first is intentional.  The same parser therefore works when
    # Acrobat gives us normal rows, one cell per line, or one word per line.
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_simple_pinout_header_end(tokens)
    rows = [list(SIMPLE_PINOUT_HEADERS)]

    if header_end is None:
        return rows

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=2)

    for row_index, pin_index in enumerate(pin_indices):
        next_pin_index = pin_indices[row_index + 1] if row_index + 1 < len(pin_indices) else len(tokens)
        payload_end = next_pin_index

        payload = tokens[pin_index + 1:payload_end]

        if len(payload) < 2:
            continue

        pin = tokens[pin_index]
        signal_name, consumed = consume_signal_name_tokens(payload)

        if consumed >= len(payload):
            continue

        signal_type = payload[consumed]
        description = " ".join(payload[consumed + 1:])

        rows.append([
            pin,
            signal_name,
            signal_type,
            description
        ])

    return rows


def looks_like_pdf_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_connector_pinout_header_end(tokens)

    if header_end is None:
        return False

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=3)
    return len(pin_indices) >= 2


def is_pinout_header_line(line):
    upper_text = normalize_pdf_line(line).upper()

    if upper_text == "":
        return True

    if upper_text in {
        "REF",
        "DES",
        "REF DES",
        "PIN",
        "SIGNAL NAME",
        "DIRECTION",
        "TYPE",
        "DESCRIPTION"
    }:
        return True

    if (
        "PIN" in upper_text
        and "SIGNAL NAME" in upper_text
        and "DIRECTION" in upper_text
        and "TYPE" in upper_text
        and "DESCRIPTION" in upper_text
    ):
        return True

    if upper_text.startswith("TABLE ") and "PINOUT" in upper_text:
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    return False




def is_pinout_ref_designator(value):
    """Stricter Ref Des test so signal names like P5_1000BT_D4- are not connectors."""
    text = normalize_pdf_line(value).upper().replace(" ", "")

    if text == "":
        return False

    # Normal connector names used by these pinout tables, including a compact
    # shared label such as J1R/J2R.
    return re.fullmatch(r"J\d+[A-Z]*(?:[/,&+]J\d+[A-Z]*)*", text) is not None

def normalize_pinout_direction(value):
    upper = str(value or "").strip().upper().replace("\\", "/")

    mapping = {
        "INPUT": "Input",
        "IN": "In",
        "OUTPUT": "Output",
        "OUT": "Out",
        "INOUT": "Inout",
        "IN/OUT": "In/Out",
        "BIDIR": "Bidir",
        "BIDIRECTIONAL": "Bidirectional",
        "NC": "NC",
        "N/C": "N/C",
        "-": "-"
    }

    return mapping.get(upper, str(value or "").strip())


def find_direction_token_index(payload):
    direction_values = {
        "INPUT",
        "IN",
        "OUTPUT",
        "OUT",
        "INOUT",
        "IN/OUT",
        "BIDIR",
        "BIDIRECTIONAL",
        "NC",
        "N/C",
        "-"
    }

    # Start at 1 so a signal named NC is not mistaken for the direction when
    # the row is "NC NC NC No Connect".
    for index in range(1, len(payload)):
        upper = str(payload[index]).strip().upper().replace("\\", "/")

        if upper not in direction_values:
            continue

        previous_value = payload[index - 1] if index > 0 else ""
        next_value = payload[index + 1] if index + 1 < len(payload) else ""

        # Do not mistake IN / OUT for the Direction column when Acrobat split a
        # signal identifier around an underscore, e.g. DIG_ / IN / _1.
        if signal_tokens_are_continuous(previous_value, payload[index]):
            continue

        if signal_tokens_are_continuous(payload[index], next_value):
            continue

        # A standalone '-' can itself be part of a signal name.  If a stronger
        # direction word immediately follows, keep '-' with the signal and use
        # the later token as Direction.
        if upper == "-":
            for lookahead in range(index + 1, min(index + 3, len(payload))):
                later = str(payload[lookahead]).strip().upper().replace("\\", "/")
                if later in direction_values and later != "-":
                    break
            else:
                return index
            continue

        return index

    return None


def parse_pinout_data_line(line, current_ref_des):
    line = normalize_pdf_line(line)

    if line == "":
        return None

    tokens = line.split()

    if len(tokens) < 4:
        return None

    ref_des = current_ref_des

    if (
        len(tokens) >= 2
        and is_connector_designator(tokens[0])
        and is_integer_value(tokens[1])
    ):
        ref_des = tokens.pop(0)

    if len(tokens) == 0 or not is_integer_value(tokens[0]):
        return None

    pin = tokens.pop(0)
    direction_index = find_direction_token_index(tokens)

    if direction_index is not None and direction_index >= 1 and direction_index + 1 < len(tokens):
        signal_name = normalize_signal_name_fragments(tokens[:direction_index])
        direction = normalize_pinout_direction(tokens[direction_index])
        signal_type = tokens[direction_index + 1]
        description = " ".join(tokens[direction_index + 2:])
    else:
        if len(tokens) < 3:
            return None

        signal_name, consumed = consume_signal_name_tokens(tokens)

        if consumed + 1 >= len(tokens):
            return None

        direction = normalize_pinout_direction(tokens[consumed])
        signal_type = tokens[consumed + 1]
        description = " ".join(tokens[consumed + 2:])

    return [ref_des, pin, signal_name, direction, signal_type, description]


def parse_pdf_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_connector_pinout_header_end(tokens)
    rows = [list(PINOUT_HEADERS)]

    if header_end is None:
        return rows

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=3)
    current_ref_des = ""

    for row_index, pin_index in enumerate(pin_indices):
        next_pin_index = pin_indices[row_index + 1] if row_index + 1 < len(pin_indices) else len(tokens)

        # Ref Des can be a vertically merged value once before the first pin,
        # or it can be repeated immediately before each pin.
        if row_index == 0:
            # Only search the small pre-data area before the first pin.  Do not
            # scan the previous row payload for later pins, because signal names
            # such as P5_1000BT_D4- also contain letters and digits.
            for token in tokens[header_end:pin_index]:
                if is_pinout_ref_designator(token):
                    current_ref_des = token

        if pin_index > header_end and is_pinout_ref_designator(tokens[pin_index - 1]):
            current_ref_des = tokens[pin_index - 1]

        payload_end = next_pin_index

        if row_index + 1 < len(pin_indices):
            # A repeated connector name immediately before the next pin belongs
            # to the next row, not this row's Description.
            if payload_end - 1 > pin_index and is_pinout_ref_designator(tokens[payload_end - 1]):
                payload_end -= 1

        payload = tokens[pin_index + 1:payload_end]

        if len(payload) < 3:
            continue

        direction_index = find_direction_token_index(payload)

        if direction_index is not None and direction_index >= 1 and direction_index + 1 < len(payload):
            signal_name = normalize_signal_name_fragments(payload[:direction_index])
            direction = normalize_pinout_direction(payload[direction_index])
            signal_type = payload[direction_index + 1]
            description = " ".join(payload[direction_index + 2:])
        else:
            signal_name, consumed = consume_signal_name_tokens(payload)

            if consumed + 1 >= len(payload):
                continue

            direction = normalize_pinout_direction(payload[consumed])
            signal_type = payload[consumed + 1]
            description = " ".join(payload[consumed + 2:])

        rows.append([
            current_ref_des,
            tokens[pin_index],
            signal_name,
            direction,
            signal_type,
            description
        ])

    return rows


# =========================================
# SAFE CSV DETECTION
#
# Do NOT treat a block as CSV merely because a description contains a comma.
# Real CSV should show comma-separated structure on the first meaningful rows.
# =========================================

def looks_like_real_csv(input_text):
    raw_lines = [line for line in input_text.split("\n") if line.strip() != ""]

    if len(raw_lines) == 0:
        return False

    sample_lines = raw_lines[:6]
    comma_sample_count = sum(1 for line in sample_lines if "," in line)

    # A normal CSV header/data block will contain commas immediately. A PDF
    # table whose descriptions merely contain punctuation usually will not.
    if comma_sample_count < min(2, len(sample_lines)):
        return False

    widths = []

    for line in raw_lines:
        try:
            parsed = next(csv.reader([line]))
        except csv.Error:
            continue

        if len(parsed) > 1:
            widths.append(len(parsed))

    if len(widths) < 2:
        return False

    # Require a repeated multi-column width. This rejects isolated prose commas.
    width_counts = {}

    for width in widths:
        width_counts[width] = width_counts.get(width, 0) + 1

    most_common_width = max(width_counts, key=width_counts.get)
    most_common_count = width_counts[most_common_width]

    return most_common_width > 1 and most_common_count >= 2

# PDF WIRING TABLE DETECTION
# =========================================

def is_connector_designator(value):
    value = normalize_pdf_line(value)

    if value == "":
        return False

    if " " in value:
        return False

    if re.fullmatch(r"[A-Za-z][A-Za-z0-9_.#/-]*", value) is None:
        return False

    if re.search(r"[A-Za-z]", value) is None:
        return False

    if re.search(r"\d", value) is None:
        return False

    return True


def looks_like_inline_wiring_row(line):
    line = normalize_pdf_line(line)

    if line == "":
        return False

    match = re.match(
        r"^([A-Za-z][A-Za-z0-9_.#/-]*)\s+(\d+)\s+(\S+)(?:\s+.*)?$",
        line
    )

    if match is None:
        return False

    return is_connector_designator(match.group(1))


def is_pdf_junk_line(line):
    text = normalize_pdf_line(line)
    upper_text = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    exact_junk = {
        "FROM",
        "TO",
        "SIGNAL NAME",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE",
        "CODE",
        "WIRE CODE",
        "NOTE",
        "COMMENT",
        "WIRE CODES"
    }

    if upper_text in exact_junk:
        return True

    if upper_text.startswith("DES POSN DES POSN"):
        return True

    if upper_text.startswith("CODE NOTE COMMENT"):
        return True

    if "DES" in upper_text and "POSN" in upper_text and "MNEMONIC" in upper_text:
        return True

    if "MNEMONIC" in upper_text and "DESCRIPTION" in upper_text:
        return True

    if "WIRE CODE" in upper_text and "NOTE" in upper_text:
        return True

    return False


# =========================================
# FIND PDF RECORDS
#
# Supports both:
#
# J12 18 114 SPARE
#
# AND:
#
# J12
# 19
# 114
# MAG1-X-
# Magnetometer 1 X axis negative
# differential input
# IN
# =========================================

def find_pdf_records(input_text):
    raw_lines = input_text.split("\n")

    lines = []

    for raw_line in raw_lines:
        line = normalize_pdf_line(raw_line)

        if is_pdf_junk_line(line):
            continue

        lines.append(line)

    records = []
    current_record = None

    index = 0

    while index < len(lines):
        line = lines[index]

        # =========================================
        # INLINE RECORD
        #
        # J12 19 114 MAG1-X- ...
        # =========================================

        if looks_like_inline_wiring_row(line):

            if current_record is not None:
                records.append(current_record)

            match = re.match(
                r"^(\S+)\s+(\d+)\s+(\S+)(?:\s+(.*))?$",
                line
            )

            current_record = {
                "from_des": match.group(1),
                "from_posn": match.group(2),
                "to_des": match.group(3),
                "payload": []
            }

            remainder = match.group(4)

            if remainder is not None:
                remainder = remainder.strip()

                if remainder != "":
                    current_record["payload"].append(remainder)

            index += 1
            continue

        # =========================================
        # VERTICAL / BROKEN PDF RECORD
        #
        # J12
        # 19
        # 114
        # MAG1-X-
        # ...
        # =========================================

        if is_connector_designator(line):

            if index + 2 < len(lines):
                possible_pin = lines[index + 1]
                possible_destination = lines[index + 2]

                if is_integer_value(possible_pin):

                    if current_record is not None:
                        records.append(current_record)

                    current_record = {
                        "from_des": line,
                        "from_posn": possible_pin,
                        "to_des": possible_destination,
                        "payload": []
                    }

                    index += 3
                    continue

        # =========================================
        # CONTINUATION OF CURRENT ROW
        # =========================================

        if current_record is not None:
            current_record["payload"].append(line)

        index += 1

    if current_record is not None:
        records.append(current_record)

    return records


# =========================================
# PARSE CONTENT OF ONE PDF ROW
# =========================================

def parse_wiring_payload(payload_lines):
    to_posn = ""
    mnemonic = ""
    description = ""
    wire_code = ""
    note = ""
    comment = ""

    if len(payload_lines) == 0:
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    payload_text = " ".join(payload_lines)
    payload_text = normalize_pdf_line(payload_text)

    if payload_text == "":
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    tokens = payload_text.split()

    # =========================================
    # COMMENTS
    #
    # A SPARE row must produce a BLANK mnemonic.
    # =========================================

    comment_values = {
        "SPARE",
        "UNUSED",
        "NC",
        "N/C"
    }

    remaining = []

    for token in tokens:

        if token.upper() in comment_values:

            if comment == "":
                comment = token

        else:
            remaining.append(token)

    tokens = remaining

    if len(tokens) == 0:
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    # =========================================
    # NOTE / DIRECTION
    # =========================================

    note_values = {
        "IN",
        "OUT",
        "IN/OUT",
    }

    remaining = []

    for token in tokens:

        if token in note_values:

            if note == "":
                note = token

        else:
            remaining.append(token)

    tokens = remaining

    if note not in note_values:
        note = ""

    # =========================================
    # OPTIONAL TO POSITION
    # =========================================

    if len(tokens) >= 2:

        if is_integer_value(tokens[0]):
            to_posn = tokens.pop(0)

    # =========================================
    # MNEMONIC
    #
    # First remaining item is normally the
    # mnemonic.
    #
    # CTU RTN is a valid two-word mnemonic.
    # Preserve it instead of treating RTN as
    # the start of the description.
    # =========================================

    if len(tokens) >= 2 and tokens[0].upper() == "CTU" and tokens[1].upper() == "RTN":
        mnemonic = "CTU RTN"
        tokens = tokens[2:]
    elif len(tokens) >= 1 and tokens[0].upper() == "CTURTN":
        mnemonic = "CTU RTN"
        tokens.pop(0)
    elif len(tokens) >= 1:
        mnemonic = tokens.pop(0)

        if len(tokens) >= 1 and tokens[0].startswith("("):
            continuation_parts = []

            while len(tokens) >= 1:
                continuation_token = tokens.pop(0)
                continuation_parts.append(continuation_token)

                if continuation_token.endswith(")"):
                    break

            if len(continuation_parts) > 0:
                mnemonic = mnemonic + " " + " ".join(continuation_parts)

    # =========================================
    # DESCRIPTION
    #
    # Preserve all remaining text. This is
    # important because Acrobat frequently wraps
    # descriptions onto multiple clipboard lines.
    # =========================================

    if len(tokens) > 0:
        description = " ".join(tokens)

    return (
        to_posn,
        mnemonic,
        description,
        wire_code,
        note,
        comment
    )

def parse_flat_pdf_wiring_table(input_text):
    records = find_pdf_records(input_text)

    rows = []

    rows.append([
        "FROM DES",
        "FROM POSN",
        "TO DES",
        "TO POSN",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE CODE",
        "NOTE",
        "COMMENT"
    ])

    for record in records:

        (
            to_posn,
            mnemonic,

            comment
        ) = parse_wiring_payload(record["payload"])

        rows.append([

            record["to_des"],
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        ])

    return rows

# =========================================
# GENERIC PDF TABLE PARSER (3-10 COLUMNS)
#
# This is the fallback for PDF tables that do not match one of the specialized
# pinout/wiring formats above.  Set the "Columns" box to the physical number of
# columns in the PDF (3 through 10).  The parser deliberately works with CELLS
# and ROW ANCHORS instead of flattening the whole table into words.  That keeps
# literal '-' cells in their own columns and prevents the next column from being
# concatenated onto the selected output column.
# =========================================

GENERIC_PDF_MIN_COLUMNS = 3
GENERIC_PDF_MAX_COLUMNS = 10


def generic_pdf_columns_supported(column_count):
    return (
        column_count is not None
        and GENERIC_PDF_MIN_COLUMNS <= column_count <= GENERIC_PDF_MAX_COLUMNS
    )


def generic_pdf_normalized_cell(value):
    text = str(value or "")
    text = text.replace("\t", " ")
    text = re.sub(r"\s+", " ", text)
    return text.strip()


def generic_pdf_is_junk_line(value):
    text = generic_pdf_normalized_cell(value)
    upper = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True
    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True

    # Common PDF page/footer fragments.  Keep this intentionally conservative;
    # '-' and NA/N/A are REAL CELLS and must never be treated as junk.
    if upper in {"CONTINUED", "CONTINUED ON NEXT PAGE"}:
        return True

    return False


def generic_pdf_raw_lines(input_text):
    values = []
    for raw in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        if raw.strip() == "":
            continue
        if generic_pdf_is_junk_line(raw):
            continue
        values.append(raw.rstrip())
    return values


def generic_pdf_anchor_signature(value):
    """
    Return a conservative signature for a likely first-column row key.

    Requiring anchors to share a signature is what prevents a numeric value in
    another column (for example the 10 in "10 MHz+ Input") from being promoted
    to a new row when the real first column is 01, 02, 03, ... .
    """
    text = generic_pdf_normalized_cell(value)
    if text == "" or " " in text:
        return None

    # 01, 02, 03 ... -- preserve the leading zero and exact width.
    if re.fullmatch(r"\d+", text):
        if len(text) > 1 and text.startswith("0"):
            return ("DIGITS_ZERO_PADDED", len(text))
        return ("DIGITS",)

    # A01, J12, P003 etc.  Keep the alphabetic prefix stable.
    match = re.fullmatch(r"([A-Za-z]+)(\d+)([A-Za-z]*)", text)
    if match is not None:
        return (
            "ALPHA_DIGITS",
            match.group(1).upper(),
            match.group(3).upper(),
        )

    # Compact designators with punctuation, but only when a digit is present.
    if (
        len(text) <= 18
        and re.fullmatch(r"[A-Za-z0-9_.#/+\-]+", text) is not None
        and re.search(r"\d", text) is not None
    ):
        prefix = re.match(r"^[A-Za-z]+", text)
        return ("COMPACT_DESIGNATOR", prefix.group(0).upper() if prefix else "")

    return None


def generic_pdf_same_anchor_signature(value, signature):
    return generic_pdf_anchor_signature(value) == signature


def generic_pdf_choose_anchor_path(lines, column_count):
    """
    Find the longest plausible sequence of first-column cells in a vertical PDF
    clipboard stream.  A normal unwrapped row occupies exactly column_count
    physical cell lines.  Wrapped cells may make it a little longer.
    """
    candidates_by_signature = {}

    for index, raw in enumerate(lines):
        cell = generic_pdf_normalized_cell(raw)
        signature = generic_pdf_anchor_signature(cell)
        if signature is None:
            continue
        candidates_by_signature.setdefault(signature, []).append(index)

    best_path = []
    best_score = None

    # Allow wrapped text, but not an unlimited distance.  The limit is generous
    # enough for long Description cells while still rejecting unrelated numbers.
    max_gap = max(column_count + 24, column_count * 8)

    for signature, indices in candidates_by_signature.items():
        if len(indices) < 2:
            continue

        states = []

        for pos, line_index in enumerate(indices):
            cell_value = generic_pdf_normalized_cell(lines[line_index])
            best = {
                "path": [line_index],
                "rows": 1,
                "gap_error": 0,
                "wrapped": 0,
                "quality": 0.0,
                "values": [cell_value],
            }

            for prev_pos in range(pos):
                prev_index = indices[prev_pos]
                gap = line_index - prev_index

                # One row must have at least one physical cell per column.
                if gap < column_count:
                    continue
                if gap > max_gap:
                    continue

                prev_state = states[prev_pos]

                # Score how plausibly the physical lines between these two
                # anchors reconstruct into the remaining columns of ONE row.
                # This is what rejects a numeric token inside a signal/figure
                # (for example the 10 in "10 MHz- Input" or "Figure 10") from
                # becoming a phantom first-column value.
                transition_payload = lines[prev_index + 1:line_index]
                transition_cells = generic_pdf_partition_payload(
                    transition_payload,
                    column_count - 1,
                )
                transition_quality = 0.0
                if len(transition_cells) == column_count - 1:
                    for transition_col, transition_cell in enumerate(transition_cells):
                        transition_quality += generic_pdf_group_score(
                            [transition_cell],
                            transition_col,
                            column_count - 1,
                        )

                state_values = prev_state["values"] + [cell_value]
                state = {
                    "path": prev_state["path"] + [line_index],
                    "rows": prev_state["rows"] + 1,
                    "gap_error": prev_state["gap_error"] + abs(gap - column_count),
                    "wrapped": prev_state["wrapped"] + max(0, gap - column_count),
                    "quality": prev_state["quality"] + transition_quality,
                    "values": state_values,
                }

                unique_count = len(set(state_values))
                duplicate_count = len(state_values) - unique_count
                best_unique = len(set(best["values"]))
                best_duplicates = len(best["values"]) - best_unique

                state_score = (
                    state["rows"],
                    unique_count,
                    -duplicate_count,
                    round(state["quality"], 6),
                    -state["gap_error"],
                    -state["wrapped"],
                    -state["path"][0],
                )
                best_score_local = (
                    best["rows"],
                    best_unique,
                    -best_duplicates,
                    round(best["quality"], 6),
                    -best["gap_error"],
                    -best["wrapped"],
                    -best["path"][0],
                )

                if state_score > best_score_local:
                    best = state

            states.append(best)

        for state in states:
            unique_count = len(set(state["values"]))
            duplicate_count = len(state["values"]) - unique_count
            score = (
                state["rows"],
                unique_count,
                -duplicate_count,
                round(state["quality"], 6),
                -state["gap_error"],
                -state["wrapped"],
                # A zero-padded numeric contact/pin column is especially strong.
                1 if signature[0] == "DIGITS_ZERO_PADDED" else 0,
                -state["path"][0],
            )
            if best_score is None or score > best_score:
                best_score = score
                best_path = state["path"]

    return best_path


GENERIC_STRONG_SCALARS = {
    "-", "NA", "N/A", "NC", "N/C", "NONE", "NULL",
    "IN", "OUT", "INPUT", "OUTPUT", "INOUT", "IN/OUT", "I/O",
    "POWER", "GROUND", "GND", "ANALOG", "DIGITAL",
    "LVDS", "TTL", "CMOS", "RS422", "RS-422", "RS485", "RS-485",
    "SPARE", "RESERVED", "UNUSED", "OPEN",
    "YES", "NO", "TRUE", "FALSE",
}

GENERIC_ENGINEERING_UNITS = (
    "HZ", "KHZ", "MHZ", "GHZ",
    "V", "MV", "UV", "KV",
    "A", "MA", "UA",
    "OHM", "OHMS", "KOHM", "KOHMS", "MOHM", "MOHMS", "Ω",
    "W", "MW", "KW",
    "F", "UF", "NF", "PF",
)


def generic_pdf_join_cell_lines(values):
    return generic_pdf_normalized_cell(" ".join(values))


def generic_pdf_is_strong_scalar(text):
    upper = generic_pdf_normalized_cell(text).upper()
    if upper in GENERIC_STRONG_SCALARS:
        return True
    if re.fullmatch(r"(?:LVDS|TTL|CMOS|RS[- ]?4(?:22|85)|CAN|UART|SPI|I2C)", upper):
        return True
    return False


def generic_pdf_group_score(values, column_index, target_count):
    """Score one possible reconstructed cell made from consecutive PDF lines."""
    values = [generic_pdf_normalized_cell(v) for v in values if generic_pdf_normalized_cell(v) != ""]
    if not values:
        return -100000.0

    text = generic_pdf_join_cell_lines(values)
    upper = text.upper()
    line_count = len(values)
    word_count = len(text.split())
    is_last = column_index == target_count - 1
    is_first_payload = column_index == 0

    score = 0.0

    # If Acrobat already emitted a whole cell on one line, strongly prefer not
    # to merge it with a neighbor.  This is the normal "newline between every
    # input/cell" case.
    if line_count == 1:
        score += 3.0
    else:
        score -= 0.20 * (line_count - 1)

    # Literal dash / NA / direction / logic-type values are strong standalone
    # cells.  Merging them with neighboring text is almost always a column shift.
    if generic_pdf_is_strong_scalar(text):
        score += 7.0
        if line_count > 1:
            score -= 2.0
    else:
        scalar_parts = sum(1 for value in values if generic_pdf_is_strong_scalar(value))
        if scalar_parts:
            score -= 5.0 * scalar_parts

    # Figure references are commonly split by Acrobat as "Figure" then "10".
    if re.fullmatch(r"FIGURE\s+\d+(?:\s*,\s*FIGURE\s+\d+)*", upper):
        score += 8.0

    # Numeric-leading signal names such as "10 MHz+ Input" MUST stay together.
    unit_pattern = "|".join(re.escape(unit) for unit in GENERIC_ENGINEERING_UNITS)
    if re.match(
        rf"^\d+(?:\.\d+)?\s*(?:{unit_pattern})(?:[+\-])?(?:\s+.*)?$",
        upper,
    ):
        score += 8.0

    # Common two-word electrical cells that PDFs often split one word per line.
    if upper in {
        "CHASSIS GND", "CHASSIS GROUND", "SIGNAL GND", "SIGNAL GROUND",
        "POWER GND", "POWER GROUND", "FRAME GND", "FRAME GROUND",
        "NOT USED", "NO CONNECT", "DO NOT CONNECT",
    }:
        score += 6.0

    # Signal/designator-like text belongs comfortably in the first payload
    # column, even when Acrobat split it across several lines.
    if is_first_payload:
        if re.search(r"[A-Z0-9][A-Z0-9_./+\-]*", upper):
            score += 1.0
        score += min(2.5, 0.45 * max(0, line_count - 1))

    # The final column is often Description/Comment and can legitimately wrap.
    if is_last:
        score += min(4.0, 0.55 * max(0, line_count - 1))
        score += min(2.0, 0.20 * max(0, word_count - 1))
    elif not is_first_payload and line_count > 1:
        # Middle scalar columns are usually short.  Only a strong compound such
        # as Figure 10 should overcome this penalty.
        score -= 1.10 * (line_count - 1)

    return score


def generic_pdf_partition_payload(payload, target_count):
    """
    Partition a vertical PDF row payload into exactly target_count cells.

    This is a dynamic-programming recovery path for Acrobat copies where a
    newline is inserted between every cell OR even between words inside a cell.
    It keeps literal '-' values as their own cells, rejoins Figure 10, rejoins
    numeric-leading signals such as 10 MHz+ Input, and lets Description wrap.
    """
    values = [generic_pdf_normalized_cell(value) for value in payload]
    values = [value for value in values if value != ""]

    if target_count <= 0:
        return []
    if not values:
        return [""] * target_count
    if len(values) < target_count:
        return values + [""] * (target_count - len(values))
    if len(values) == target_count:
        return values
    if target_count == 1:
        return [generic_pdf_join_cell_lines(values)]

    n = len(values)
    neg = -10**18
    # dp[g][i] = best score for first i physical lines partitioned into g cells.
    dp = [[neg] * (n + 1) for _ in range(target_count + 1)]
    prev = [[None] * (n + 1) for _ in range(target_count + 1)]
    dp[0][0] = 0.0

    for groups in range(1, target_count + 1):
        min_i = groups
        max_i = n - (target_count - groups)
        for i in range(min_i, max_i + 1):
            # Keep search bounded.  Long text is normally Description, and the
            # last cell may consume everything that remains.
            max_group_len = n if groups == target_count else min(12, i - (groups - 1))
            j_min = max(groups - 1, i - max_group_len)
            for j in range(j_min, i):
                if dp[groups - 1][j] == neg:
                    continue
                group_values = values[j:i]
                group_score = generic_pdf_group_score(
                    group_values,
                    groups - 1,
                    target_count,
                )
                score = dp[groups - 1][j] + group_score
                if score > dp[groups][i]:
                    dp[groups][i] = score
                    prev[groups][i] = j

    if prev[target_count][n] is None:
        # Conservative fallback: preserve early scalar cells and merge overflow
        # only into the final column.
        head = values[:target_count - 1]
        tail = generic_pdf_join_cell_lines(values[target_count - 1:])
        return head + [tail]

    groups = []
    g = target_count
    i = n
    while g > 0:
        j = prev[g][i]
        if j is None:
            break
        groups.append(generic_pdf_join_cell_lines(values[j:i]))
        i = j
        g -= 1

    groups.reverse()
    if len(groups) != target_count:
        return values[:target_count - 1] + [generic_pdf_join_cell_lines(values[target_count - 1:])]
    return groups


def generic_pdf_merge_payload_to_width(payload, target_count):
    """
    Convert the physical lines after a row anchor into target_count cells.

    Unlike the old implementation, overflow is NOT blindly appended to the
    final column.  A DP partitioner reconstructs wrapped/interleaved cells while
    preserving explicit '-' / NA / Figure / Logic / Direction cells.
    """
    return generic_pdf_partition_payload(payload, target_count)


def parse_generic_pdf_anchor_stream(input_text, column_count):
    lines = generic_pdf_raw_lines(input_text)
    if len(lines) < column_count * 2:
        return []

    path = generic_pdf_choose_anchor_path(lines, column_count)
    if len(path) < 2:
        return []

    rows = []

    for row_index, start in enumerate(path):
        end = path[row_index + 1] if row_index + 1 < len(path) else len(lines)
        anchor = generic_pdf_normalized_cell(lines[start])
        payload = lines[start + 1:end]

        # The final row may be followed by a footer/title.  Once we have at
        # least enough cells, only let modest overflow be merged into the last
        # column.  Very large overflow is more likely outside the table.
        max_payload = (column_count - 1) + 8
        if len(payload) > max_payload:
            payload = payload[:max_payload]

        cells = [anchor] + generic_pdf_merge_payload_to_width(
            payload,
            column_count - 1,
        )

        if len(cells) == column_count:
            rows.append(cells)

    return rows


def generic_pdf_split_explicit_columns(raw_line, column_count):
    """
    Split one physical row only when the PDF preserved an actual column
    delimiter: tabs, pipes, or runs of 2+ whitespace.  Never split ordinary
    single spaces here because they belong inside values such as 10 MHz+ Input.
    """
    raw = raw_line.rstrip("\r\n")

    if "\t" in raw:
        parts = [generic_pdf_normalized_cell(value) for value in raw.split("\t")]
        if len(parts) == column_count:
            return parts

    if "|" in raw:
        parts = [generic_pdf_normalized_cell(value) for value in raw.split("|")]
        parts = [value for value in parts if value != "" or len(parts) == column_count]
        if len(parts) == column_count:
            return parts

    parts = [
        generic_pdf_normalized_cell(value)
        for value in re.split(r"\s{2,}", raw.strip())
    ]
    if len(parts) == column_count:
        return parts

    return None


def parse_generic_pdf_explicit_rows(input_text, column_count):
    rows = []
    for raw in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        if raw.strip() == "" or generic_pdf_is_junk_line(raw):
            continue
        row = generic_pdf_split_explicit_columns(raw, column_count)
        if row is not None:
            rows.append(row)

    if not rows:
        return []

    # A clean aligned header is also a column_count-wide row.  Do not return it
    # as DATA when the following rows share a clear first-column key pattern.
    signatures = {}
    anchored_rows = []
    for row in rows:
        signature = generic_pdf_anchor_signature(row[0])
        if signature is None:
            continue
        signatures[signature] = signatures.get(signature, 0) + 1
        anchored_rows.append((signature, row))

    if signatures:
        dominant = max(signatures, key=signatures.get)
        selected = [row for signature, row in anchored_rows if signature == dominant]
        if len(selected) >= 2:
            return selected

    return rows


def generic_pdf_partition_single_space_row(raw_line, column_count):
    """
    Last-resort row parser for PDFs that collapse column spacing to one space.

    It is conservative: the first column is required to look like a row key.
    Extra words are kept together in the LAST column instead of being truncated.
    Therefore `02 RESERVED NA NA do not use this contact` remains five cells.
    Specialized parsers still run before this fallback for layouts where a
    multi-word middle cell needs format-specific treatment.
    """
    text = generic_pdf_normalized_cell(raw_line)
    tokens = text.split()

    if len(tokens) < column_count:
        return None
    if generic_pdf_anchor_signature(tokens[0]) is None:
        return None

    if len(tokens) == column_count:
        return tokens

    if column_count == 1:
        return [text]

    return tokens[:column_count - 1] + [" ".join(tokens[column_count - 1:])]


def parse_generic_pdf_single_space_rows(input_text, column_count):
    rows = []
    signatures = {}

    for raw in generic_pdf_raw_lines(input_text):
        row = generic_pdf_partition_single_space_row(raw, column_count)
        if row is None:
            continue
        signature = generic_pdf_anchor_signature(row[0])
        signatures[signature] = signatures.get(signature, 0) + 1
        rows.append((signature, row))

    if not rows:
        return []

    # Keep the dominant first-column signature so numeric prose elsewhere in
    # the selection cannot become a phantom table row.
    dominant = max(signatures, key=signatures.get)
    selected = [row for signature, row in rows if signature == dominant]
    return selected if len(selected) >= 2 else []


def generic_pdf_cell_shape(value):
    text = generic_pdf_normalized_cell(value)
    upper = text.upper()
    if text == "":
        return "BLANK"
    if text == "-":
        return "DASH"
    if generic_pdf_is_strong_scalar(text):
        return "SCALAR"
    if re.fullmatch(r"\d+", text):
        return "DIGITS"
    if re.fullmatch(r"FIGURE\s+\d+(?:\s*,\s*FIGURE\s+\d+)*", upper):
        return "FIGURE"
    if re.fullmatch(r"[A-Z0-9_.#/+\-]+", upper):
        return "CODE"
    if len(text.split()) >= 2:
        return "PHRASE"
    return "TEXT"


GENERIC_HEADER_KEYWORDS = {
    "PIN", "PIN NUMBER", "NUMBER", "CONTACT", "CONTACT DESIGNATION",
    "DESIGNATION", "UNIT", "UNIT SIGNAL", "UNIT SIGNAL DESIGNATION",
    "SIGNAL", "SIGNAL NAME", "SIGNAL TYPE", "DESCRIPTION",
    "DIRECTION", "I/F", "I/F DIRECTION", "TYPE", "FIGURE", "LOGIC",
    "FUNCTION", "NAME", "REF", "REF DES", "DES", "POSN", "POSITION",
    "FROM", "TO", "MNEMONIC", "WIRE", "WIRE CODE", "NOTE", "COMMENT",
    "CODE", "VALUE", "PARAMETER", "CHANNEL", "PORT", "CONTACT NO",
}


def generic_pdf_row_looks_header_like(row):
    hits = 0
    for value in row:
        upper = generic_pdf_normalized_cell(value).upper()
        if upper in GENERIC_HEADER_KEYWORDS:
            hits += 1
            continue
        # Also recognize wrapped/composite labels such as "Signal Type" or
        # "Contact Designation" without matching arbitrary data prose.
        if any(
            keyword in upper
            for keyword in (
                "DESCRIPTION", "DIRECTION", "DESIGNATION", "SIGNAL TYPE",
                "SIGNAL NAME", "PIN NUMBER", "CONTACT", "FIGURE", "LOGIC",
            )
        ):
            hits += 1
    return hits >= max(2, (len(row) + 1) // 2)


def parse_generic_pdf_modulo_cell_stream(input_text, column_count):
    """
    Recover a pure row-major stream where Acrobat inserted a newline between
    every cell and the first column does not provide a usable numeric/designator
    anchor.  We try each possible starting offset and keep the alignment whose
    per-column value shapes are most consistent across rows.
    """
    lines = [generic_pdf_normalized_cell(v) for v in generic_pdf_raw_lines(input_text)]
    lines = [v for v in lines if v != ""]
    if len(lines) < column_count * 2:
        return []

    best_rows = []
    best_score = None
    max_start = min(len(lines) - column_count * 2, column_count * 3)
    max_start = max(0, max_start)

    for start in range(max_start + 1):
        remaining = lines[start:]
        full_row_count = len(remaining) // column_count
        if full_row_count < 2:
            continue

        # Ignore an incomplete trailing fragment/footer.
        rows = [
            remaining[i * column_count:(i + 1) * column_count]
            for i in range(full_row_count)
        ]

        # Score column-shape consistency.  Real cell-stream alignment tends to
        # keep each physical column semantically similar from row to row.
        consistency = 0.0
        for col in range(column_count):
            counts = {}
            for row in rows:
                shape = generic_pdf_cell_shape(row[col])
                counts[shape] = counts.get(shape, 0) + 1
            dominant_count = max(counts.values())
            consistency += dominant_count / len(rows)

        dash_cells = sum(1 for row in rows for v in row if v == "-")
        scalar_cells = sum(1 for row in rows for v in row if generic_pdf_is_strong_scalar(v))

        # Prefer more complete rows, consistent columns, and starts close to the
        # beginning without requiring that headers themselves be parseable.
        score = (
            full_row_count,
            round(consistency, 6),
            dash_cells + scalar_cells,
            -start,
        )
        if best_score is None or score > best_score:
            best_score = score
            best_rows = rows

    # If the first row looks header-like while the following rows have a much
    # stronger repeated shape, drop it from the data candidate.  A synthetic
    # header will be added later.
    if len(best_rows) >= 3:
        first_shapes = [generic_pdf_cell_shape(v) for v in best_rows[0]]
        next_shapes = [generic_pdf_cell_shape(v) for v in best_rows[1]]
        if generic_pdf_row_looks_header_like(best_rows[0]):
            best_rows = best_rows[1:]
        elif first_shapes.count("PHRASE") + first_shapes.count("TEXT") >= column_count - 1:
            if next_shapes != first_shapes:
                best_rows = best_rows[1:]

    return best_rows


def generic_pdf_candidate_score(rows, column_count, method_priority):
    if not rows:
        return (-1, -1, -1, -1, -1)

    valid = [row for row in rows if len(row) == column_count]
    if len(valid) < 2:
        return (-1, -1, -1, -1, -1)

    nonblank = sum(
        1
        for row in valid
        for value in row
        if generic_pdf_normalized_cell(value) != ""
    )
    dash_cells = sum(
        1
        for row in valid
        for value in row
        if generic_pdf_normalized_cell(value) == "-"
    )

    shape_consistency = 0.0
    for col in range(column_count):
        counts = {}
        for row in valid:
            shape = generic_pdf_cell_shape(row[col])
            counts[shape] = counts.get(shape, 0) + 1
        if counts:
            shape_consistency += max(counts.values()) / len(valid)

    return (len(valid), round(shape_consistency, 6), nonblank, dash_cells, method_priority)


def generic_pdf_extract_aligned_headers(input_text, column_count, first_data_row):
    """Use a clean aligned header when one exists; otherwise use safe labels."""
    if first_data_row is None:
        return [f"COLUMN {index + 1}" for index in range(column_count)]

    raw_lines = input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n")
    data_key = generic_pdf_normalized_cell(first_data_row[0])

    data_line_index = None
    for index, raw in enumerate(raw_lines):
        normalized = generic_pdf_normalized_cell(raw)
        if normalized == data_key or normalized.startswith(data_key + " "):
            data_line_index = index
            break

    if data_line_index is None:
        return [f"COLUMN {index + 1}" for index in range(column_count)]

    # Search only the nearby pre-data lines.  If one clean row already has the
    # correct width, it is a much better header than guessing from wrapped text.
    for raw in reversed(raw_lines[max(0, data_line_index - 6):data_line_index]):
        row = generic_pdf_split_explicit_columns(raw, column_count)
        if row is not None and any(row):
            return row

    return [f"COLUMN {index + 1}" for index in range(column_count)]


def parse_generic_pdf_table(input_text, column_count):
    """
    Generic 3-10 column PDF fallback.

    Candidate order:
      1. explicit/aligned rows (tabs, pipes, 2+ spaces)
      2. vertical/cell-stream rows reconstructed from first-column anchors
      3. pure newline-per-cell row-major stream
      4. conservative single-space rows

    The best reconstruction is returned with one header row.  Synthetic headers
    are intentional when the PDF header itself is wrapped/interleaved; they keep
    Extract Column aligned and let the existing Skip Header checkbox work.
    """
    if not generic_pdf_columns_supported(column_count):
        return []

    explicit_rows = parse_generic_pdf_explicit_rows(input_text, column_count)
    anchor_rows = parse_generic_pdf_anchor_stream(input_text, column_count)
    modulo_rows = parse_generic_pdf_modulo_cell_stream(input_text, column_count)
    single_space_rows = parse_generic_pdf_single_space_rows(input_text, column_count)

    candidates = [
        (explicit_rows, 4),
        (anchor_rows, 3),
    ]

    # A first-column anchor is much stronger evidence than modulo grouping.
    # Only use the pure modulo/cell-stream fallback when anchor reconstruction
    # is unavailable; otherwise extra newlines can make an incorrect modulo
    # alignment appear to contain more rows than the real table.
    if not anchor_rows and not explicit_rows:
        candidates.append((modulo_rows, 2))

    candidates.append((single_space_rows, 1))

    best_rows = []
    best_score = None

    for candidate_rows, priority in candidates:
        score = generic_pdf_candidate_score(candidate_rows, column_count, priority)
        if best_score is None or score > best_score:
            best_score = score
            best_rows = candidate_rows

    if not best_rows:
        return []

    headers = generic_pdf_extract_aligned_headers(
        input_text,
        column_count,
        best_rows[0] if best_rows else None,
    )

    return [headers] + best_rows


# =========================================
# GENERAL TABLE PARSER
# =========================================

def parse_table():
    input_text = input_box.get("1.0", "end-1c")

    if input_text == "":
        return [], 0, "NONE"

    input_text = input_text.replace("\r\n", "\n")
    input_text = input_text.replace("\r", "\n")

    manual_columns = get_manual_column_count()

    rows = []
    delimiter_name = ""

    # =========================================
    # MANUAL 3-10 COLUMN OVERRIDE
    #
    # If the user explicitly entered a column count, honor that before any
    # format-specific detector.  This prevents a 5-column table from being
    # accidentally stolen by a 3- or 4-column pinout parser just because some
    # header words overlap.  The generic parser is newline/cell-stream aware.
    # =========================================

    if generic_pdf_columns_supported(manual_columns):
        rows = parse_generic_pdf_table(input_text, manual_columns)
        if len(rows) > 2:
            label = f"GENERIC PDF TABLE / {manual_columns} COLUMNS"
            if looks_highly_verticalized(input_text):
                label += " / NEWLINE CELL STREAM"
            return rows, manual_columns, label

    # =========================================
    # PDF 4-COLUMN PIN / DESCRIPTION / FIGURE / LOGIC TABLE FIRST
    #
    # This must run before the 3-column Figure-table parser because both formats
    # contain Pin Number / Description / Figure headers.
    # =========================================

    if looks_like_pdf_pin_description_figure_logic_table(input_text):

        rows = parse_pdf_pin_description_figure_logic_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE / LOGIC TABLE"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 4, label

    # =========================================
    # HEADERLESS 4-COLUMN PIN / DESCRIPTION / FIGURE / LOGIC CONTINUATION
    # =========================================

    if looks_like_headerless_pin_description_figure_logic_table(input_text):

        rows = parse_headerless_pin_description_figure_logic_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE / LOGIC TABLE / HEADERLESS CONTINUATION"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 4, label

    # =========================================
    # PDF 3-COLUMN PIN / DESCRIPTION / FIGURE TABLE FIRST
    #
    # This format can be copied from the PDF in a staggered tagged-text order,
    # so it must be reconstructed before the generic spaced-table fallback.
    # =========================================

    if looks_like_pdf_pin_description_figure_table(input_text):

        rows = parse_pdf_pin_description_figure_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE TABLE"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 3, label

    # =========================================
    # HEADERLESS 3-COLUMN PIN / DESCRIPTION / FIGURE CONTINUATION
    #
    # Handles body-only selections such as:
    #   47   Chassis GND   -
    #   13   Chassis GND   -
    #   31   Chassis GND   -
    # where the table header is on a previous page/section and is not copied.
    # This also handles PDF clipboard streams that merge repeated descriptions
    # or many '-' Figure cells onto a single line.
    # =========================================

    if looks_like_headerless_pin_description_figure_table(input_text):

        rows = parse_headerless_pin_description_figure_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE TABLE / HEADERLESS CONTINUATION"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 3, label

    # =========================================
    # PDF 4-COLUMN SIMPLE / POWER PINOUT TABLE FIRST
    #
    # Example: Pin # | Signal Name | Signal Type | Signal Description.
    # This has no Ref Des and no Direction column.
    # =========================================

    if looks_like_pdf_simple_pinout_table(input_text):

        rows = parse_pdf_simple_pinout_table(input_text)

        if len(rows) > 1:
            label = "PDF SIMPLE PINOUT TABLE"
            if looks_highly_verticalized(input_text):
                label += " / VERTICAL PASTE"
            return rows, 4, label

    # =========================================
    # PDF 6-COLUMN CONNECTOR PINOUT TABLE
    #
    # This must run before the wiring-table and CSV checks. Pinout tables can
    # contain commas inside Description and can have a vertically merged Ref Des.
    # =========================================

    if looks_like_pdf_pinout_table(input_text):

        rows = parse_pdf_pinout_table(input_text)

        if len(rows) > 1:
            label = "PDF PINOUT TABLE"
            if looks_highly_verticalized(input_text):
                label += " / VERTICAL PASTE"
            return rows, 6, label

    # =========================================
    # PDF WIRING TABLE
    #
    # This MUST happen before generic TAB parsing because PDF clipboard data can
    # contain tabs without actually preserving table structure.
    # =========================================

    if looks_like_flat_pdf_wiring_table(input_text):

        rows = parse_flat_pdf_wiring_table(input_text)

        if len(rows) > 1:
            return rows, 9, "PDF WIRING TABLE"

    # =========================================
    # TRUE TAB-SEPARATED DATA
    # =========================================

    if "\t" in input_text:
        delimiter_name = "TAB"

        for line in input_text.split("\n"):
            rows.append(line.split("\t"))

    # =========================================
    # CSV
    #
    # Use structural CSV detection instead of `if "," in input_text` so a
    # normal Description such as "pair 4/D, negative" does not hijack parsing.
    # =========================================

    elif looks_like_real_csv(input_text):
        delimiter_name = "COMMA"

        reader = csv.reader(io.StringIO(input_text))

        for row in reader:
            rows.append(row)

    # =========================================
    # GENERIC SPACE-ALIGNED TABLE
    # =========================================

    elif manual_columns is not None and manual_columns > 1:
        delimiter_name = "SPACED TABLE"

        for line in input_text.split("\n"):

            if line.strip() == "":
                continue

            row = re.split(r" {2,}", line.strip())

            rows.append(row)

    # =========================================
    # SINGLE COLUMN
    # =========================================

    else:
        delimiter_name = "LINES"

        for line in input_text.split("\n"):
            rows.append([line])

    # =========================================
    # COLUMN COUNT
    # =========================================

    if manual_columns is not None:
        column_count = manual_columns

    else:
        column_count = 0

        for row in rows:

            if len(row) > column_count:
                column_count = len(row)

    # =========================================
    # NORMALIZE ALL ROW WIDTHS
    # =========================================

    normalized_rows = []

    for row in rows:

        normalized_row = list(row)

        while len(normalized_row) < column_count:
            normalized_row.append("")

        if len(normalized_row) > column_count:
            normalized_row = normalized_row[:column_count]

        normalized_rows.append(normalized_row)

    return normalized_rows, column_count, delimiter_name


# =========================================
# DETECT TABLE
# =========================================

def detect_table():
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        column_selector["values"] = []
        column_selector.set("")

        status_label.config(
            text="No table data found."
        )

        return

    column_choices = []

    first_row = rows[0]

    for index in range(column_count):
        header = ""

        if index < len(first_row):
            header = str(first_row[index]).strip()

        if header == "":
            column_choices.append(str(index + 1))

        else:
            column_choices.append(
                f"{index + 1}: {header}"
            )

    column_selector["values"] = column_choices

    if column_count > 0:
        column_selector.current(0)

    structured_pdf_table = (
        delimiter_name == "PDF WIRING TABLE"
        or delimiter_name.startswith("PDF PINOUT TABLE")
        or delimiter_name.startswith("PDF SIMPLE PINOUT TABLE")
        or delimiter_name.startswith("PDF PIN / DESCRIPTION / FIGURE TABLE")
        or delimiter_name.startswith("PDF PIN / DESCRIPTION / FIGURE / LOGIC TABLE")
        or delimiter_name.startswith("GENERIC PDF TABLE")
    )

    detected_row_count = len(rows) - 1 if structured_pdf_table else len(rows)

    status_label.config(
        text=f"Detected {detected_row_count} rows x {column_count} columns | {delimiter_name}"
    )


def get_selected_column():
    selected = column_selector.get().strip()

    if selected == "":
        return None

    selected_number = selected.split(":")[0].strip()

    try:
        return int(selected_number)

    except ValueError:
        return None


# =========================================
# EXTRACT SELECTED COLUMN
# =========================================

def extract_column():
    global current_count
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        status_label.config(
            text="No table data found."
        )

        return

    selected_column = get_selected_column()

    if selected_column is None:

        detect_table()

        selected_column = get_selected_column()

    if selected_column is None:
        status_label.config(
            text="Select a column."
        )

        return

    if selected_column < 1 or selected_column > column_count:
        status_label.config(
            text=f"Column must be between 1 and {column_count}."
        )

        return

    start_row = 0

    if skip_header_var.get():
        start_row = 1

    output_values = []

    for row_index in range(start_row, len(rows)):
        row = rows[row_index]

        if selected_column - 1 < len(row):
            value = str(row[selected_column - 1]).strip()

        else:
            value = ""

        output_values.append(value)

    output_text = "\n".join(output_values)

    output_box.delete("1.0", tk.END)
    output_box.insert(tk.END, output_text)

    current_count = len(output_values)

    blank_count = 0

    for value in output_values:

        if value.strip() == "":
            blank_count += 1

    count_label.config(
        text=f"Items: {current_count}"
    )

    status_label.config(
        text=f"Column {selected_column} extracted | Blank cells: {blank_count}"
    )

    column_selector["values"] = []

    detected_column_count = 0

    count_label.config(

    )

    status_label.config(
        text=""
    )

def copy_connector():
    connector_number = connector_entry.get().strip()
    copy_count_text = copy_count_entry.get().strip()

    if connector_number == "":
        connector_status.config(
            text="Enter a connector number."
        )

        return

    if copy_count_text == "":
        copy_count = current_count

    else:

        try:
            copy_count = int(copy_count_text)

        except ValueError:
            connector_status.config(
                text="Copy count must be a number."
            )

            return

    if copy_count <= 0:
        connector_status.config(
            text="No items to copy."
        )

        return

    connector_output = "\n".join(
        [connector_number] * copy_count
    )

    root.clipboard_clear()
    root.clipboard_append(connector_output)
    root.update()

    connector_status.config(
        text=f"Copied {copy_count} copies of connector {connector_number}"
    )
