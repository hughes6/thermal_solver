def _unpack(tool_id: str) -> bytes:
    item = TOOLS[tool_id]
    content = zlib.decompress(base64.b85decode(item["payload"].encode("ascii")))
    actual = hashlib.sha256(content).hexdigest()
    if actual != item["sha256"]:
        raise RuntimeError("Bundled tool failed integrity verification: " + tool_id)
    compile(content.decode("utf-8"), item["filename"], "exec")
    return content


def get_runtime_folder() -> Path:
    preferred = Path(__file__).resolve().parent / "E3_Tools_Runtime"
    try:
        preferred.mkdir(parents=True, exist_ok=True)
        with tempfile.NamedTemporaryFile(dir=str(preferred), delete=True):
            pass
        return preferred
    except OSError:
        fallback = Path(os.environ.get("LOCALAPPDATA") or Path.home()) / "E3_Tools_Runtime"
        fallback.mkdir(parents=True, exist_ok=True)
        return fallback


def install_tool(tool_id: str, base: Path) -> Path:
    item = TOOLS[tool_id]
    directory = base / item["folder"]
    directory.mkdir(parents=True, exist_ok=True)
    destination = directory / item["filename"]
    content = _unpack(tool_id)
    if destination.exists():
        existing = destination.read_bytes()
        if existing == content:
            return destination
        # Don't silently discard user edits or scripts from an older version.
        suffix = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup = directory / (destination.stem + ".backup_" + suffix + ".py")
        index = 2
        while backup.exists():
            backup = directory / (destination.stem + ".backup_" + suffix + f"_{index}.py")
            index += 1
        shutil.copy2(destination, backup)
    temporary = destination.with_suffix(".py.tmp")
    temporary.write_bytes(content)
    os.replace(temporary, destination)
    return destination


def all_sources_ok() -> bool:
    for key in TOOLS:
        _unpack(key)
    return True


def open_folder(path: Path):
    path.mkdir(parents=True, exist_ok=True)
    if sys.platform == "win32":
        os.startfile(str(path))
    elif sys.platform == "darwin":
        subprocess.Popen(["open", str(path)])
    else:
        subprocess.Popen(["xdg-open", str(path)])


class MegaApp:
    """A small, white/light-blue hub; opens tools only on user clicks."""
    def __init__(self):
        self.root=tk.Tk()
        self.root.title('E3 Tools | Main Menu')
        self.root.configure(bg=COLOR['page'])
        self.root.resizable(False,False)
        self.root.geometry('468x352+65+85')
        self.root.attributes('-topmost',True)
        self.runtime=get_runtime_folder()
        self.running=None
        self.running_id=None
        self.buttons={}
        self.status=tk.StringVar(value='Select a tool to open')
        self.detail=tk.StringVar(value='')
        self.build_ui()
        self.root.after(120,self.show_menu)
        self.root.after(650,self.watch_child)
        self.root.protocol('WM_DELETE_WINDOW',self.on_close)
        self.log_menu_open()

    def log_menu_open(self):
        try:
            path=self.runtime/'mega_launcher_startup.log'
            with path.open('a',encoding='utf-8') as f:
                f.write(f'{datetime.now().isoformat(timespec="seconds")} MENU START - {SUITE_VERSION}\n'
                        f'  file={Path(__file__).resolve()}\n  python={sys.executable}\n'
                        '  tools=header, signals, pin (none auto-started)\n')
        except Exception:pass

    def show_menu(self):
        if self.running is None:
            self.root.deiconify();self.root.lift()
            try:self.root.focus_force()
            except Exception:pass
            self.root.attributes('-topmost',True)

    def build_ui(self):
        bg=COLOR['page']; pale=COLOR['soft']
        head=tk.Frame(self.root,bg=pale,height=48)
        head.pack(fill='x');head.pack_propagate(False)
        tk.Label(head,text='E3 Tools',bg=pale,fg=COLOR['text'],
                 font=('Segoe UI',14,'bold')).pack(side='left',padx=13,pady=10)
        tk.Label(head,text='MAIN MENU',bg=pale,fg=COLOR['accent'],
                 font=('Segoe UI',8,'bold')).pack(side='right',padx=14,pady=14)
        tk.Label(self.root,text='Choose a tool:',bg=bg,fg=COLOR['muted'],
                 font=('Segoe UI',9)).pack(anchor='w',padx=13,pady=(8,6))
        for key,title,description in (
            ('header','Header Formatter','Format the text at the top of a block'),
            ('signals','Signal Mover','Move connection labels into place'),
            ('pin','Cable Pin Spacer','Match pin spacing and format labels'),
        ):
            self.tool_card(key,title,description)
        row=tk.Frame(self.root,bg=bg)
        row.pack(fill='x',padx=13,pady=(5,4))
        self.small_link(row,'Logs',lambda:open_folder(self.runtime)).pack(side='left')
        self.small_link(row,'Import pin data',self.import_pin_data).pack(side='left',padx=(15,0))
        self.small_link(row,'Close',self.on_close).pack(side='right')
        status=tk.Frame(self.root,bg=pale,height=28,highlightbackground=COLOR['border'],highlightthickness=1)
        status.pack(fill='x',padx=13,pady=(3,6));status.pack_propagate(False)
        tk.Label(status,textvariable=self.status,bg=pale,fg=COLOR['text'],
                 font=('Segoe UI',8),anchor='w').pack(fill='x',padx=8,pady=5)

    def tool_card(self,key,title,desc):
        panel=tk.Frame(self.root,bg=COLOR['panel'],height=61,
                       highlightthickness=1,highlightbackground=COLOR['border'])
        panel.pack(fill='x',padx=13,pady=(0,5));panel.pack_propagate(False)
        content=tk.Frame(panel,bg=COLOR['panel'])
        content.pack(side='left',fill='both',expand=True,padx=11,pady=(9,4))
        tk.Label(content,text=title,bg=COLOR['panel'],fg=COLOR['text'],
                 font=('Segoe UI',10,'bold'),anchor='w').pack(fill='x')
        tk.Label(content,text=desc,bg=COLOR['panel'],fg=COLOR['muted'],
                 font=('Segoe UI',8),anchor='w').pack(fill='x')
        button=tk.Button(panel,text='Open',command=lambda k=key:self.launch(k),
                         bg=COLOR['accent'],fg='#FFFFFF',
                         activebackground=COLOR['accent_hover'],activeforeground='#FFFFFF',
                         font=('Segoe UI',9,'bold'),width=7,borderwidth=0,
                         relief='flat',cursor='hand2',pady=5)
        button.pack(side='right',padx=10,pady=12)
        self.buttons[key]=button

    def small_link(self,frame,label,command):
        return tk.Button(frame,text=label,command=command,font=('Segoe UI',8),
                         fg=COLOR['accent'],bg=COLOR['page'],
                         activeforeground=COLOR['accent_hover'],
                         activebackground=COLOR['page'],borderwidth=0,
                         cursor='hand2',padx=0,pady=0)

    def set_buttons(self,enabled):
        for key,button in self.buttons.items():
            button.configure(state='normal' if enabled else 'disabled',
                             bg=COLOR['accent'] if enabled else '#B9CBDD',
                             cursor='hand2' if enabled else 'arrow')

    def launch(self,tool_id):
        # Called only from an explicitly clicked tool-specific Open button.
        if self.running is not None and self.running.poll() is None:
            messagebox.showinfo('Another E3 tool is open',
                'Close the running utility before opening another one. This prevents '
                'two scripts from modifying the E3 drawing simultaneously.',parent=self.root)
            return
        try:
            script=install_tool(tool_id,self.runtime)
            python=Path(sys.executable)
            if sys.platform=='win32' and python.name.lower()=='python.exe':
                candidate=python.with_name('pythonw.exe')
                if candidate.is_file():python=candidate
            cmd=[str(python),str(script)]
            if tool_id=='pin':cmd.append('--worker')
            output_path=script.parent/'launcher_output.log'
            flags=getattr(subprocess,'CREATE_NO_WINDOW',0) if sys.platform=='win32' else 0
            with output_path.open('a',encoding='utf-8') as fd:
                fd.write('\n--- User opened ' + tool_id + ' at ' +
                         datetime.now().isoformat(timespec='seconds') +
                         ' ['+SUITE_VERSION+'] ---\n')
            with output_path.open('a',encoding='utf-8') as fd:
                self.running=subprocess.Popen(cmd,cwd=str(script.parent),
                    stdin=subprocess.DEVNULL,stdout=fd,stderr=subprocess.STDOUT,
                    creationflags=flags,close_fds=True)
            self.running_id=tool_id
            self.set_buttons(False)
            self.status.set(TOOLS[tool_id]['title']+' is open')
            self.detail.set('Close its window to enable all three tools again')
            # Parent remains visible as a separate window. Do not launch another
            # tool, hide the menu, or steal E3's keyboard focus.
            self.root.attributes('-topmost',False)
        except Exception as exc:
            self.running=None
            self.running_id=None
            self.set_buttons(True)
            self.status.set('Launch failed — see error message')
            self.detail.set(str(exc)[:95])
            messagebox.showerror('E3 tool launch error',str(exc),parent=self.root)

    def watch_child(self):
        try:
            if self.running is not None:
                rc=self.running.poll()
                if rc is not None:
                    name=TOOLS[self.running_id]['title']
                    self.running=None
                    self.running_id=None
                    self.set_buttons(True)
                    self.status.set(name+(' closed' if rc==0 else ' exited with an error'))
                    self.detail.set('Choose another tool above' if rc==0 else
                                    'Check launcher_output.log in its tool folder')
                    self.show_menu()
        finally:
            self.root.after(650,self.watch_child)

    def import_pin_data(self):
        previous=filedialog.askdirectory(title='Select your previous Block Pin Spacer script folder',
                                          parent=self.root)
        if not previous:return
        destination=self.runtime/TOOLS['pin']['folder']
        destination.mkdir(parents=True,exist_ok=True)
        imported=[];skipped=[]
        for suffix in ('_recovery.json','_pin_identity.json'):
            candidate=Path(previous)/('main'+suffix)
            if not candidate.is_file():
                old=Path(previous)/('E3_Cable_To_Block_Pin_Spacing_V15_ZUKEN_LIGHT_GUI'+suffix)
                if old.is_file():candidate=old
            newfile=destination/('main'+suffix)
            if not candidate.is_file():
                skipped.append('not found: '+candidate.name);continue
            if newfile.exists():
                skipped.append('already exists: '+newfile.name);continue
            try:
                data=json.loads(candidate.read_text(encoding='utf-8-sig'))
                if not isinstance(data,dict):raise ValueError('Expected a JSON object')
                shutil.copy2(candidate,newfile)
                imported.append(newfile.name)
            except Exception as exc:
                skipped.append(f'{candidate.name}: {exc}')
        messagebox.showinfo('Pin spacer data',
            'Imported: '+(', '.join(imported) or 'none')+'\nSkipped: '+
            (', '.join(skipped) or 'none')+
            '\n\nImported pin identities are checked against live E3 before use.',
            parent=self.root)

    def on_close(self):
        if self.running is not None and self.running.poll() is None:
            if not messagebox.askyesno('Close launcher?',
                'An E3 utility is still open. Close only the launcher and let it finish?',
                parent=self.root):return
        self.root.destroy()

                assert hashlib.sha256(path.read_bytes()).hexdigest()==TOOLS[key]["sha256"]
                assert path.read_text(encoding="utf-8").strip()
                # A second extraction must be idempotent.
                assert install_tool(key,Path(temp))==path
        print("Extraction, checksums, Python syntax and idempotence: PASS")
        return

        folder=get_runtime_folder()
        for key in TOOLS:
            print(install_tool(key,folder))
        return

        print('Launcher version:', SUITE_VERSION)
        print('Three tools:', ', '.join(TOOLS))
        print('No E3 utility auto-launches: PASS')
        return
