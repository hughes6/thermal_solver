class EndtoEnd ():
    def __init__(self):
        pass

    def run(self):
        # set constants
        #====================================================================================================================================
        s_Report_Title = "End_to_End_Test"
        s_Default_Save_Path = "\\\\asdfiler2\\UserShares\\Zuken\\SimsLibrary\\Reports\\"
        s_assy_attr_Assy_Num = "Additional Part ID"
        s_Row_Drawing_Name = 1
        s_Col_Drawing_Name = 9
        s_Drawing_Name = "Ball Drawing Number"
        s_Drawing_Rev = "Ball Revision Number"
        s_Start_Row = 6
        s_Col_Line_No = 1
        s_Col_Ref = 2
        s_Col_Pin = 3
        s_Col_Part = 4
        s_Col_To_Ref = 5
        s_Col_To_pin = 6
        s_Col_To_Part = 7
        s_Col_Signal = 8
        s_Col_Status = 9
        #====================================================================================================================================

        # set objects
        #====================================================================================================================================
        app = win32.Dispatch('CT.Application')
        prj = app.CreateJobObject()
        dev = prj.CreateDeviceObject()
        s_Drawing_Name = "Ball Drawing Number"
        s_Drawing_Rev = "Ball Revision Number"
        s_Col_Header_Names = "WIRE NO.;FROM REF;FROM PIN;FROM PART;TO REF;TO PIN;TO PART;SIGNAL;STATUS"
        tar_dev = prj.CreateDeviceObject()
        connFROM = prj.CreateDeviceObject()
        connA = prj.CreateDeviceObject()
        blockA = prj.CreateDeviceObject()
        connB = prj.CreateDeviceObject()
        blockB = prj.CreateDeviceObject()
        mated_conn = prj.CreateDeviceObject()
        pinFROM = prj.CreatePinObject()
        pinA = prj.CreatePinObject()
        pinB = prj.CreatePinObject()
        mated_pin = prj.CreatePinObject()
        core = prj.CreatePinObject()
        sig = prj.CreateSignalObject()
        #====================================================================================================================================

        # more setup
        #====================================================================================================================================
        default_filename_full_path = s_Default_Save_Path + s_Report_Title + "_" + prj.GetName()
        output_file_name = default_filename_full_path + ".txt"
        app.PutMessage("Extracting End to End data")
        app.PutMessage("Working......  0%")
        current_row = s_Start_Row
        #====================================================================================================================================


        devinfo = prj.GetAllDeviceIds(None)
        devcount = devinfo[0]
        devids   = devinfo[1]
        devids   = [x for x in devids if x is not None]
        dev_total = len(devids)

        columns = s_Col_Header_Names.split(';')
        df = pd.DataFrame(columns=columns)


        reported_pins = []
        sigInfo = prj.GetSignalIds(None)
        sig_count = sigInfo[0]
        sigids = sigInfo[1]
        sigids = [x for x in sigids if x is not None]

        for sig_index, sigid in enumerate(sigids):
            sig.SetId(sigid)


            Number_of_statuses = 10
            for fraction in range(1, Number_of_statuses):
                threshold = round(sig_count * fraction / Number_of_statuses)
                if sig_index == threshold:
                    pct = round(100 * sig_index / sig_count)
                    app.PutMessage(f"Working...... {pct}%")
            
            pinInfo = sig.GetPinIds(None)
            pin_count = pinInfo[0]
            pinids = pinInfo[1]
            pinids = [x for x in pinids if x is not None]

            reported_pins = []
            if pin_count > 1 and sig.GetName() != "SPARE":
                for j in range(pin_count):
                    pinA.SetId(pinids[j])

                    if not (pinA.GetId() in reported_pins or pinA.GetOriginalId() in reported_pins):
                        connA.SetId(pinA.GetId())
                        blockA.SetId(connA.GetBlockId())
                        pinB.SetId(pinA.GetConnectedPinId())
                        connB.SetId(pinB.GetId())
                        blockB.SetId(connB.GetBlockId())
                        bool_IsConnector = connA.IsConnector() == 1
                        bool_IsMatedToBox = blockB.IsBlock() == 1
                        bool_MatedBoxNoSigEquiv = pinB.GetLogicalEquivalenceID() == 0
                        bool_NoSigEquiv = pinA.GetLogicalEquivalenceID() == 0
                        bool_IsStandardDevice = connA.IsDevice() == 1

                        if (bool_IsConnector and bool_IsMatedToBox and bool_MatedBoxNoSigEquiv) or (bool_IsStandardDevice and bool_NoSigEquiv):
                            if len(reported_pins) == 0:
                                pinFROM.SetId(pinA.GetId())
                                connFROM.SetId(connA.GetId())

                            else:
                                row = {
                                "WIRE NO." : current_row - s_Start_Row + 1,
                                "FROM REF" : f"{connFROM.GetLocation()}-{connFROM.GetName()}",
                                "FROM PIN" : pinFROM.GetName(),
                                "FROM PART" : connFROM.GetComponentName(),
                                "TO REF" : f"{connA.GetLocation()}-{connA.GetName()}",
                                "TO PIN" : pinA.GetName(),
                                "TO PART" : connA.GetComponentName(),
                                "SIGNAL" : sig.GetName(),
                                "STATUS" : ""
                                }
                                df.loc[len(df)] = row
                                current_row += 1
                            reported_pins.append(pinA.GetId())
            if len(reported_pins) == 1 and 1 == 0:
                row["FROM REF"] = f"{connFROM.GetLocation()}-{connFROM.GetName()}"
                row["FROM PIN"] = pinFROM.GetName()
                row["FROM PART"] = connFROM.GetComponentName()
                row["SIGNAL"] = sig.GetName()
                df.loc[len(df)] = row
                current_row += 1


        app.PutMessage(f"Working...... 100%")
        drawing_no = prj.GetAttributeValue(s_Drawing_Name)
        revision_no = prj.GetAttributeValue(s_Drawing_Rev)
        app.PutMessage("Converting data from csv to excel")

        # unset objects
        #====================================================================================================================================
        del dev
        del prj
        del app
        #====================================================================================================================================

        #====================================================================================================================================
        #================================================Running csv to excel code===========================================================
        #====================================================================================================================================

        csv_path = output_file_name
        drawing_no = drawing_no
        revision_no = revision_no
        save_path = str(csv_path[:-4]) + ".xlsx"
        output_path = utils.ask_save_path(save_path)

        if output_path != None:
            print("CSV file:", csv_path)
            print("Output file: ", output_path)
            print("Drawing number: ", drawing_no)
            print("Revision number: ", revision_no)

            df_csv = df

            df_template = pd.DataFrame([[""] * len(df_csv.columns)] * (5 + len(df_csv)))

            # Convert template to object dtype to avoid assignment errors
            df_template = df_template.astype(object)

            df_template.iloc[4, 0:len(df_csv.columns)] = df_csv.columns
            df_template.iloc[5 : 5 + len(df_csv), 0:len(df_csv.columns)] = df_csv.values

            # Save back to the copied template file
            df_template.to_excel(output_path, index=False, header=False)


            template_column_widths = {'A': 18.0, 'B': 15.0, 'C': 12.0, 'D': 35.0, 'E': 15.0, 'F': 10.0, 'G': 40.0, 'H': 12.0, 'I': 20.0}

            wb = load_workbook(output_path)
            ws = wb.active
            thick = Border(left=Side(style="thick"),right=Side(style="thick"),top=Side(style="thick"),bottom=Side(style="thick"))
            thin = Border(left=Side(style="thin"),right=Side(style="thin"),top=Side(style="thin"),bottom=Side(style="thin"))

            last_row = 5 + len(df_csv)       
            last_col = len(df_csv.columns)       
            last_col_letter = get_column_letter(last_col) # Convert last column number to Excel letters (A, B, C...)
            # Apply in-place filtering across all header columns
            ws.auto_filter.ref = f"A5:{last_col_letter}{last_row}"
            # 1. A1:A4 merged + bold
            ws.merge_cells("A1:A4")
            ws["A1"].value = "BAE Systems \n SMS \n P.O. Box 1062 \n Boulder, CO 80306"
            ws["A1"].alignment = Alignment(vertical="center", horizontal="left", wrap_text=True)
            ws["A1"].font = Font(bold=True, size=8, name="Arial")

            # 2. B1:M4 merged + bold + size 18
            ws.merge_cells("B1:H4")
            ws["B1"].value = "End-to-End Test"
            ws["B1"].alignment = Alignment(horizontal="center", vertical="center")
            ws["B1"].font = Font(bold=True, size=18, name="Arial")

            # 3. N1:N4 merged + bold + thick border
            ws.merge_cells("I1:I4")
            ws["I1"].value = f"Drawing No: {drawing_no} \n Revision: {revision_no}"
            ws["I1"].alignment = Alignment(horizontal="right", vertical="center", wrap_text=True)
            ws["I1"].font = Font(bold=True, size=8, name="Arial")

            # Left‑alignment object
            left_align = Alignment(horizontal="left", vertical="center")
            start_row = 5
            end_row = 5 + len(df_csv)   # df_csv is your pandas CSV DataFrame
            for row in ws.iter_rows(min_row=start_row, max_row=end_row):
                for cell in row:
                    cell.font = Font(name="Arial", size=8)
                    cell.alignment = left_align

            for col_letter, width in template_column_widths.items():
                if width is not None:      # some widths may be None
                    ws.column_dimensions[col_letter].width = width

            for row in ws.iter_rows(min_row=1, max_row=4, min_col=1, max_col=1):
                for cell in row:
                    cell.border = thick
            for row in ws.iter_rows(min_row=1, max_row=4, min_col=2, max_col=8):
                for cell in row:
                    cell.border = thick
            for row in ws.iter_rows(min_row=1, max_row=4, min_col=9, max_col=9):
                for cell in row:
                    cell.border = thick
            for row in ws.iter_rows(min_row=5, max_row=5, min_col=1, max_col=9):
                for cell in row:
                    cell.font = Font(bold=True)
                    cell.border = thin
            for row in ws.iter_rows(min_row=6, max_row=end_row, min_col=3, max_col=3):
                for cell in row:
                    cell.alignment = Alignment(horizontal="center")
            for row in ws.iter_rows(min_row=6, max_row=end_row, min_col=6, max_col=6):
                for cell in row:
                    cell.alignment = Alignment(horizontal="center")
            for row_idx in range(6, end_row + 1):
                ws.row_dimensions[row_idx].height = 11.25


            #===================================================================================
            #=====================BEGIN SORTING A-Z then ascending==============================
            #===================================================================================
            last_column = "I"

            match_column_i = (column_index_from_string("B")) - 1 # minus one because cell iteration start at 0 in loops

            match_relationships = {}
            for r in range(6, ws.max_row + 1):

                match_cell = ws[r][match_column_i].value # temporary key
                    
                key = str(match_cell)
                value = f"{str(r)}"
                if key not in match_relationships and key:
                    match_relationships[key] = [value]
                else:
                    match_relationships[key].append(value)

            # sort alphabetical but ascending
            sorted_dict = dict(sorted(match_relationships.items(), key=lambda x: utils.endtoend_sort_key(x[0])))


            swap_rows = []
            values = list(sorted_dict.values())
            r = 5
            
            for i in range(len(values)):
                v = values[i]
                for j in range(len(v)):
                    old_row = v[j]
                    r += 1

                    swap_rows.append([int(old_row), int(r)])


            ws_copy = wb.copy_worksheet(ws)

            i = 0
            for r in range(6, ws.max_row + 1):

                old = int(swap_rows[i][0])
                new = int(swap_rows[i][1])
                utils.write_row_from_copy(ws, ws_copy, old, new) # src dest
                i += 1
                
            for name in wb.sheetnames:
                if 'Copy' in name:
                    del wb[name]

            ws.freeze_panes = "A6"   # Everything above A6 is frozen

            utils.configurePrintArea(ws)
            
            wb.save(output_path)
            print("Finished writing data into:", output_path)
            print(output_path)
            output_path_normalized = os.path.normpath(output_path)
            utils.open_file_with_default_app(output_path_normalized)
            #===================================================================================
            #===================================END SORTING=====================================
            #===================================================================================

            app = win32.Dispatch('CT.Application')
            app.PutMessage("End to End Script Output Complete")
            del app
            return output_path
        else:
            app = win32.Dispatch('CT.Application')
            app.PutMessage('No output file selected, nothing was saved')
            del app
