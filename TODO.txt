def ask_output_excel_path(source_path):
    folder = os.path.dirname(source_path)
    base = os.path.basename(source_path)
    stem, extension = os.path.splitext(base)
    default_name = stem + "_pdf_filled" + extension
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    output_path = filedialog.asksaveasfilename(initialdir=folder, initialfile=default_name, defaultextension=extension, filetypes=[("Excel Macro Workbook", "*.xlsm"), ("Excel Workbook", "*.xlsx"), ("All Files", "*.*")])
    root.destroy()
    if not output_path:
        return None
    return os.path.abspath(output_path)


def ask_extraction_settings():
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    start_page = simpledialog.askinteger("PDF Start Page", "Enter the first PDF page to scrape:", minvalue=1, parent=root)
    if start_page is None:
        root.destroy()
        return None, None, None
    end_page_text = simpledialog.askstring("PDF Stop Page", "Enter the last PDF page to scrape. Leave blank to go to the end:", parent=root)
    if end_page_text is None:
        root.destroy()
        return None, None, None
    end_page_text = end_page_text.strip()
    if end_page_text == "":
        end_page = None
    else:
        try:
            end_page = int(end_page_text)
        except ValueError:
            root.destroy()
            raise ValueError("Stop page must be a whole number or blank")
        if end_page < start_page:
            root.destroy()
            raise ValueError("Stop page must be greater than or equal to start page")
    separator_keyword = simpledialog.askstring("Section Separator", "Enter the keyword that starts each connector section:", parent=root)
    root.destroy()
    if separator_keyword is None:
        return None, None, None
    separator_keyword = separator_keyword.strip()
    if separator_keyword == "":
        return None, None, None
    return start_page, end_page, separator_keyword


def parse_pasted_column_names(raw_text):
    raw_text = raw_text.replace("\r\n", "\n")
    raw_text = raw_text.replace("\r", "\n")
    raw_text = raw_text.strip()
    if raw_text == "":
        return []
    lines = [line.strip() for line in raw_text.split("\n") if line.strip() != ""]
    if len(lines) > 1:
        return lines
    one_line = lines[0] if lines else ""
    if "\t" in one_line:
        return [value.strip() for value in one_line.split("\t") if value.strip() != ""]
    if "," in one_line:
        return [value.strip() for value in one_line.split(",") if value.strip() != ""]
    spaced = [value.strip() for value in re.split(r"\s{2,}", one_line) if value.strip() != ""]
    if len(spaced) > 1:
        return spaced
    return [one_line] if one_line else []

def guess_role_column(column_names, possible_names):
    for column_name in column_names:
        upper_name = column_name.upper()
        for possible_name in possible_names:
            if possible_name.upper() == upper_name:
                return column_name
    for column_name in column_names:
        upper_name = column_name.upper()
        for possible_name in possible_names:
            if possible_name.upper() in upper_name:
                return column_name
    return column_names[0]

def should_skip_direction(pdf_roles):
    """Return True when the user chose not to scrape/write IN/OUT data."""
    return bool(pdf_roles.get("skip_direction", False)) or not pdf_roles.get("direction")


def extract_connector_names(value):
    """
    Return every connector designator found in a PDF/Excel label.

    Examples handled:
        J1R
        J1R / J2R
        J1R, J2R
        J1R & J2R
        SCP-J1R / SCP-J2R

    Keeping this as a list is important because some PDF tables use one
    shared pinout for two (or more) connectors.
    """
    text = str(value or "").strip().upper()
    connectors = []

    for match in re.finditer(r"J\s*0*(\d+)\s*([A-Z]*)\b", text):
        number = int(match.group(1))
        suffix = match.group(2)
        connector = f"J{number}{suffix}"

        if connector not in connectors:
            connectors.append(connector)

    return connectors


def canonical_connector_name(value):
    connectors = extract_connector_names(value)

    if connectors:
        return connectors[0]

    return str(value or "").strip().upper()


def connector_sort_key(value):
    canonical = canonical_connector_name(value)
    match = re.fullmatch(r"J(\d+)([A-Z]*)", canonical, re.IGNORECASE)
    if match:
        return int(match.group(1)), match.group(2).upper()
    return 999999, str(canonical)


def clean_extracted_value(value):
    value = str(value or "").strip()
    value = re.sub(r"\s+", " ", value)
    value = re.sub(r"\s+([,.;:])", r"\1", value)
    return value


def get_role_value(row, pdf_roles, role_name):
    column_name = pdf_roles[role_name]
    return str(row.get(column_name, "") or "").strip()


def normalize_local_signal_value(value):
    text = clean_extracted_value(value)
    text = re.sub(r"\bCTURTN\b", "CTU RTN", text, flags=re.IGNORECASE)
    return text


def normalize_direction_value(value):
    text = str(value or "").strip().upper()
    text = text.replace("\\", "/")
    text = re.sub(r"\s+", " ", text)
    compact = re.sub(r"[\s_-]+", "", text)

    # Common wiring-table values
    if compact in {"IN/OUT", "INOUT", "I/O", "IO", "BIDIR", "BIDIRECTIONAL"}:
        return "IN/OUT"
    if compact in {"IN", "INPUT"}:
        return "IN"
    if compact in {"OUT", "OUTPUT"}:
        return "OUT"

    if re.fullmatch(r"IN\s*/\s*OUT", text):
        return "IN/OUT"

    # NC, "-", blank, etc. intentionally remain blank.
    return ""


def is_signal_prefix_candidate(value):
    text = str(value or "").strip()

    if text == "":
        return False

    if re.fullmatch(r"[+-]?\d+(?:\.\d+)?", text):
        return False

    if re.fullmatch(r"J\d+", canonical_connector_name(text), re.IGNORECASE):
        return False

    if normalize_direction_value(text) != "":
        return False

    return True


def combine_local_signal_fragments(fragments):
    cleaned = []

    for fragment in fragments:
        value = normalize_local_signal_value(fragment)

        if value == "":
            continue

        if normalize_direction_value(value) != "":
            continue

        cleaned.append(value)

    if len(cleaned) == 0:
        return ""

    combined = cleaned[0]

    for fragment in cleaned[1:]:
        combined_upper = combined.upper()
        fragment_upper = fragment.upper()

        if combined_upper == "CTU" and fragment_upper == "RTN":
            combined = combined + " " + fragment
        elif fragment.startswith("("):
            combined = combined + " " + fragment
        elif combined.endswith("("):
            combined = combined + fragment
        elif combined.endswith(("_", "-", "/", "#")):
            combined = combined + fragment
        elif fragment.startswith(("_", "-", "/", "#")):
            combined = combined + fragment
        else:
            combined = combined + fragment

    return normalize_local_signal_value(combined)


def extract_local_signal_from_words(row_words, boundaries, signal_index):
    if signal_index < 0 or signal_index >= len(boundaries) - 1:
        return ""

    left = boundaries[signal_index]
    right = boundaries[signal_index + 1]
    sorted_words = sorted(row_words, key=lambda word: (float(word["top"]), float(word["x0"])))
    line_groups = []

    for word in sorted_words:
        top = float(word["top"])

        if not line_groups or abs(top - line_groups[-1]["top"]) > 3.0:
            line_groups.append({"top": top, "words": [word]})
        else:
            line_groups[-1]["words"].append(word)

    fragments = []

    for line in line_groups:
        words = sorted(line["words"], key=lambda word: float(word["x0"]))
        seed_indexes = []

        for index, word in enumerate(words):
            value = str(word.get("text", "")).strip()

            if value == "":
                continue

            if re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
                continue

            x0 = float(word["x0"])
            x1 = float(word["x1"])

            if x1 > left and x0 < right:
                seed_indexes.append(index)

        if not seed_indexes:
            continue

        start_index = min(seed_indexes)
        end_index = max(seed_indexes)

        while start_index > 0:
            previous_word = words[start_index - 1]
            current_word = words[start_index]
            previous_value = str(previous_word.get("text", "")).strip()

            if not is_signal_prefix_candidate(previous_value):
                break

            previous_x1 = float(previous_word["x1"])
            current_x0 = float(current_word["x0"])
            gap = current_x0 - previous_x1

            if gap > 8.0:
                break

            if previous_x1 < left - 35.0:
                break

            start_index -= 1

        selected_words = words[start_index:end_index + 1]
        line_text = ""
        previous_x1 = None

        for word in selected_words:
            value = str(word.get("text", "")).strip()

            if value == "":
                continue

            if re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
                continue

            x0 = float(word["x0"])
            x1 = float(word["x1"])

            if line_text != "" and previous_x1 is not None:
                gap = x0 - previous_x1

                if gap > 2.5:
                    line_text += " "

            line_text += value
            previous_x1 = x1

        line_text = normalize_local_signal_value(line_text)

        if line_text == "":
            continue

        if normalize_direction_value(line_text) != "":
            continue

        fragments.append({"top": float(line["top"]), "text": line_text})

    fragments.sort(key=lambda item: item["top"])
    fragment_values = [item["text"] for item in fragments]

    return combine_local_signal_fragments(fragment_values)


def find_direction_in_column_from_words(row_words, boundaries, column_index):
    if column_index < 0 or column_index >= len(boundaries) - 1:
        return ""

    left = boundaries[column_index]
    right = boundaries[column_index + 1]
    target_center = (left + right) / 2.0
    nearby = []

    for word in row_words:
        value = str(word.get("text", "")).strip()

        if value == "":
            continue

        x0 = float(word["x0"])
        x1 = float(word["x1"])
        center = (x0 + x1) / 2.0

        if center < left or center >= right:
            continue

        nearby.append({"text": value, "top": float(word["top"]), "x0": x0, "x1": x1, "center": center})

    nearby.sort(key=lambda item: (item["top"], item["x0"]))
    candidates = []

    for item in nearby:
        normalized = normalize_direction_value(item["text"])

        if normalized != "":
            candidates.append((abs(item["center"] - target_center), normalized))

    for index in range(len(nearby)):
        for length in (2, 3):
            end = index + length

            if end > len(nearby):
                continue

            group = nearby[index:end]

            if max(item["top"] for item in group) - min(item["top"] for item in group) > 3.0:
                continue

            combined = "".join(item["text"] for item in group)
            normalized = normalize_direction_value(combined)

            if normalized == "":
                continue

            group_center = sum(item["center"] for item in group) / len(group)
            candidates.append((abs(group_center - target_center), normalized))

    if not candidates:
        return ""

    candidates.sort(key=lambda item: item[0])

    return candidates[0][1]


def find_direction_from_row_words(row_words, boundaries, direction_index):
    normal_value = find_direction_in_column_from_words(row_words, boundaries, direction_index)

    if normal_value != "":
        return normal_value

    previous_column_index = direction_index - 1

    if previous_column_index < 0:
        return ""

    previous_value = find_direction_in_column_from_words(row_words, boundaries, previous_column_index)

    if previous_value != "":
        return previous_value

    return ""


def extract_role_text_from_words(row_words, boundaries, column_index, left_pad=0.0, right_pad=0.0, skip_numeric=False):
    if column_index < 0 or column_index >= len(boundaries) - 1:
        return ""
    left = boundaries[column_index] - left_pad
    right = boundaries[column_index + 1] + right_pad
    candidates = []
    for word in row_words:
        value = str(word.get("text", "")).strip()
        if value == "":
            continue
        x0 = float(word["x0"])
        if x0 < left or x0 >= right:
            continue
        if skip_numeric and re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
            continue
        candidates.append(word)
    if not candidates:
        return ""
    candidates.sort(key=lambda word: (float(word["top"]), float(word["x0"])))
    line_groups = []
    for word in candidates:
        top = float(word["top"])
        if not line_groups or abs(top - line_groups[-1]["top"]) > 3.0:
            line_groups.append({"top": top, "words": [word]})
        else:
            line_groups[-1]["words"].append(word)
    rebuilt_lines = []
    for line in line_groups:
        line["words"].sort(key=lambda word: float(word["x0"]))
        line_text = ""
        previous_x1 = None
        for word in line["words"]:
            value = str(word.get("text", "")).strip()
            if value == "":
                continue
            x0 = float(word["x0"])
            x1 = float(word["x1"])
            if line_text != "" and previous_x1 is not None:
                gap = x0 - previous_x1
                if gap > 2.5:
                    line_text += " "
            line_text += value
            previous_x1 = x1
        if line_text != "":
            rebuilt_lines.append(line_text)
    return clean_extracted_value(" ".join(value for value in rebuilt_lines if value != ""))

def word_center_x(word):
    return (float(word["x0"]) + float(word["x1"])) / 2.0


def char_center_x(char):
    return (float(char["x0"]) + float(char["x1"])) / 2.0


def group_words_into_lines(words, y_tolerance=3.0):
    lines = []
    sorted_words = sorted(words, key=lambda word: (float(word["top"]), float(word["x0"])))
    for word in sorted_words:
        word_top = float(word["top"])
        target_line = None
        for line in lines:
            if abs(line["top"] - word_top) <= y_tolerance:
                target_line = line
                break
        if target_line is None:
            target_line = {"top": word_top, "words": []}
            lines.append(target_line)
        target_line["words"].append(word)
    for line in lines:
        line["words"] = sorted(line["words"], key=lambda word: float(word["x0"]))
        line["text"] = " ".join(str(word["text"]) for word in line["words"]).strip()
    return lines


def group_chars_into_lines(chars, y_tolerance=2.5):
    lines = []
    sorted_chars = sorted(chars, key=lambda char: (float(char["top"]), float(char["x0"])))
    for char in sorted_chars:
        char_top = float(char["top"])
        target_line = None
        for line in lines:
            if abs(line["top"] - char_top) <= y_tolerance:
                target_line = line
                break
        if target_line is None:
            target_line = {"top": char_top, "chars": []}
            lines.append(target_line)
        target_line["chars"].append(char)
    for line in lines:
        line["chars"] = sorted(line["chars"], key=lambda char: float(char["x0"]))
    return lines


def rebuild_cell_from_chars(chars, y_tolerance=2.5):
    if not chars:
        return ""
    lines = group_chars_into_lines(chars, y_tolerance=y_tolerance)
    rebuilt_lines = []
    for line in lines:
        line_chars = line["chars"]
        widths = [float(char["x1"]) - float(char["x0"]) for char in line_chars if str(char.get("text", "")).strip() != ""]
        median_width = statistics.median(widths) if widths else 3.0
        space_gap = max(1.25, median_width * 0.55)
        line_text = ""
        previous_x1 = None
        for char in line_chars:
            value = str(char.get("text", ""))
            x0 = float(char["x0"])
            x1 = float(char["x1"])
            if value == "":
                continue
            if value.isspace():
                if line_text and not line_text.endswith(" "):
                    line_text += " "
                previous_x1 = x1
                continue
            if previous_x1 is not None:
                gap = x0 - previous_x1
                if gap > space_gap and line_text and not line_text.endswith(" "):
                    line_text += " "
            line_text += value
            previous_x1 = x1
        line_text = clean_extracted_value(line_text)
        if line_text:
            rebuilt_lines.append(line_text)
    return clean_extracted_value(" ".join(rebuilt_lines))


def find_exact_word(words, wanted_text, target_y=None, y_tolerance=60.0):
    candidates = []
    for word in words:
        text = str(word["text"]).strip().upper()
        if text != wanted_text.upper():
            continue
        if target_y is not None and abs(float(word["top"]) - target_y) > y_tolerance:
            continue
        candidates.append(word)
    if not candidates:
        return None
    if target_y is None:
        return candidates[0]
    candidates.sort(key=lambda word: abs(float(word["top"]) - target_y))
    return candidates[0]

def find_page_footer_top(lines):
    footer_tops = []
    for line in lines:
        line_text = str(line.get("text", "")).strip()
        if re.fullmatch(r"Page\s+\d+\s+of\s+\d+", line_text, re.IGNORECASE):
            footer_tops.append(float(line["top"]))
    return min(footer_tops) if footer_tops else None


def load_pdf_cache(pdf_path, start_page, end_page=None):
    start_time = time.perf_counter()
    pdf_cache = {}
    with pdfplumber.open(pdf_path) as pdf:
        total_pages = len(pdf.pages)
        if start_page > total_pages:
            raise ValueError(f"Start page {start_page} is greater than PDF page count {total_pages}")
        actual_end_page = total_pages if end_page is None else min(end_page, total_pages)
        for page_number in range(start_page, actual_end_page + 1):
            page = pdf.pages[page_number - 1]
            text = page.extract_text() or ""
            words = page.extract_words(x_tolerance=3, y_tolerance=3, keep_blank_chars=False, use_text_flow=False)
            chars = list(page.chars)
            lines = group_words_into_lines(words)
            edges = list(page.edges)
            pdf_cache[page_number] = {"text": text, "words": words, "chars": chars, "lines": lines, "edges": edges, "footer_top": find_page_footer_top(lines), "width": float(page.width), "height": float(page.height)}
    elapsed = time.perf_counter() - start_time
    print(f"PDF CACHE BUILT: {len(pdf_cache)} PAGES IN {elapsed:.2f} SECONDS")
    return pdf_cache

def detect_default_column_names(pdf_cache):
    # Prefer strong header geometry/text signatures over raw comma/space parsing.
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data.get("words", [])
        lines = page_data.get("lines", [])

        if not words:
            continue

        for line in lines:
            upper_values = {
                str(word.get("text", "")).strip().upper()
                for word in line.get("words", [])
            }

            # 6-column pinout:
            # REF DES | PIN | SIGNAL NAME | DIRECTION | TYPE | DESCRIPTION
            if {"DIRECTION", "TYPE", "DESCRIPTION"}.issubset(upper_values):
                header_y = float(line["top"])
                nearby = [
                    str(word.get("text", "")).strip().upper()
                    for word in words
                    if abs(float(word["top"]) - header_y) <= 40.0
                ]

                if (
                    "PIN" in nearby
                    and "SIGNAL" in nearby
                    and "REF" in nearby
                    and "DES" in nearby
                ):
                    print("AUTO-DETECTED TABLE FORMAT: 6-COLUMN CONNECTOR PINOUT")
                    return list(SIX_COLUMN_PINOUT_NAMES)

            # 9-column wiring:
            # FROM DES/POSN | TO DES/POSN | MNEMONIC | DESCRIPTION | WIRE | NOTE | COMMENT
            if "MNEMONIC" in upper_values and "DESCRIPTION" in upper_values:
                header_y = float(line["top"])
                nearby = [
                    str(word.get("text", "")).strip().upper()
                    for word in words
                    if abs(float(word["top"]) - header_y) <= 60.0
                ]

                if (
                    nearby.count("DES") >= 2
                    and nearby.count("POSN") >= 2
                    and "WIRE" in nearby
                    and "NOTE" in nearby
                    and "COMMENT" in nearby
                ):
                    print("AUTO-DETECTED TABLE FORMAT: 9-COLUMN WIRING TABLE")
                    return list(NINE_COLUMN_NAMES)

    # Text-level fallback for PDFs whose header words are oddly grouped.
    for page_number in sorted(pdf_cache):
        upper_text = str(pdf_cache[page_number].get("text", "") or "").upper()

        if (
            "SIGNAL NAME" in upper_text
            and "DIRECTION" in upper_text
            and "TYPE" in upper_text
            and "DESCRIPTION" in upper_text
            and "PIN" in upper_text
        ):
            print("AUTO-DETECTED TABLE FORMAT: 6-COLUMN CONNECTOR PINOUT (TEXT FALLBACK)")
            return list(SIX_COLUMN_PINOUT_NAMES)

        if (
            "MNEMONIC" in upper_text
            and "WIRE" in upper_text
            and "NOTE" in upper_text
            and "COMMENT" in upper_text
        ):
            print("AUTO-DETECTED TABLE FORMAT: 9-COLUMN WIRING TABLE (TEXT FALLBACK)")
            return list(NINE_COLUMN_NAMES)

    print("WARNING: TABLE FORMAT NOT AUTO-DETECTED; DEFAULTING TO 9-COLUMN WIRING")
    return list(DEFAULT_COLUMN_NAMES)

# LEARN COLUMN GEOMETRY FROM PDF HEADER
# ================================================
def edge_vertical_length(edge):
    if "top" in edge and "bottom" in edge:
        return abs(float(edge["bottom"]) - float(edge["top"]))
    if "y0" in edge and "y1" in edge:
        return abs(float(edge["y1"]) - float(edge["y0"]))
    return 0.0


def edge_x_position(edge):
    if "x0" not in edge or "x1" not in edge:
        return None
    x0 = float(edge["x0"])
    x1 = float(edge["x1"])
    if abs(x1 - x0) > 1.5:
        return None
    return (x0 + x1) / 2.0


def cluster_vertical_edges(edges, tolerance=1.5):
    raw = []
    for edge in edges:
        x = edge_x_position(edge)
        if x is None:
            continue
        length = edge_vertical_length(edge)
        if length < 8.0:
            continue
        raw.append((x, length))
    raw.sort(key=lambda item: item[0])
    clusters = []
    for x, length in raw:
        if not clusters or abs(x - clusters[-1]["x"]) > tolerance:
            clusters.append({"x": x, "count": 1, "length": length})
        else:
            cluster = clusters[-1]
            total = cluster["count"] + 1
            cluster["x"] = ((cluster["x"] * cluster["count"]) + x) / total
            cluster["count"] = total
            cluster["length"] += length
    return clusters


def choose_grid_boundary(clusters, left_center, right_center):
    candidates = [cluster for cluster in clusters if left_center < cluster["x"] < right_center]
    if not candidates:
        return None
    midpoint = (left_center + right_center) / 2.0
    candidates.sort(key=lambda cluster: (-cluster["length"], abs(cluster["x"] - midpoint)))
    return candidates[0]["x"]


def print_learned_columns(page_number, column_names, centers):
    print("")
    print("============================================================")
    print("TABLE COLUMNS LEARNED")
    print("============================================================")
    print("PAGE:", page_number)
    for name, center in zip(column_names, centers):
        print(name, "X =", round(center, 2))
    print("============================================================")


def find_nine_column_centers(pdf_cache, column_names):
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data["words"]
        lines = page_data["lines"]
        if not words:
            continue
        for line in lines:
            upper_values = [str(word["text"]).strip().upper() for word in line["words"]]
            if "MNEMONIC" not in upper_values or "DESCRIPTION" not in upper_values:
                continue
            header_y = line["top"]
            nearby_words = [word for word in words if abs(float(word["top"]) - header_y) <= 60.0]
            des_words = [word for word in nearby_words if str(word["text"]).strip().upper() == "DES"]
            posn_words = [word for word in nearby_words if str(word["text"]).strip().upper() == "POSN"]
            des_words.sort(key=lambda word: float(word["x0"]))
            posn_words.sort(key=lambda word: float(word["x0"]))
            if len(des_words) < 2 or len(posn_words) < 2:
                continue
            mnemonic_word = find_exact_word(nearby_words, "MNEMONIC", header_y)
            description_word = find_exact_word(nearby_words, "DESCRIPTION", header_y)
            wire_word = find_exact_word(nearby_words, "WIRE", header_y)
            note_word = find_exact_word(nearby_words, "NOTE", header_y)
            comment_word = find_exact_word(nearby_words, "COMMENT", header_y)
            if mnemonic_word is None or description_word is None or wire_word is None or note_word is None or comment_word is None:
                continue
            centers = [
                word_center_x(des_words[0]),
                word_center_x(posn_words[0]),
                word_center_x(des_words[1]),
                word_center_x(posn_words[1]),
                word_center_x(mnemonic_word),
                word_center_x(description_word),
                word_center_x(wire_word),
                word_center_x(note_word),
                word_center_x(comment_word),
            ]
            if any(centers[index] >= centers[index + 1] for index in range(len(centers) - 1)):
                continue
            print_learned_columns(page_number, column_names, centers)
            return centers, page_data["width"], page_number
    raise Exception("Could not automatically determine the 9-column wiring-table locations from PDF headers")


def find_six_column_centers(pdf_cache, column_names):
    """
    Learn the 6-column connector pinout layout:
        REF DES | PIN | SIGNAL NAME | DIRECTION | TYPE | DESCRIPTION

    REF/DES can be stacked on two lines, and SIGNAL/NAME can be split into
    separate PDF words. We therefore use a vertical header window instead of
    requiring every header word to live on one text line.
    """
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data["words"]
        lines = page_data["lines"]

        if not words:
            continue

        for line in lines:
            upper_values = {str(word["text"]).strip().upper() for word in line["words"]}

            # The right side of this header is the most stable identifying pattern.
            if not {"DIRECTION", "TYPE", "DESCRIPTION"}.issubset(upper_values):
                continue

            header_y = float(line["top"])
            nearby_words = [
                word for word in words
                if abs(float(word["top"]) - header_y) <= 40.0
            ]

            pin_word = find_exact_word(nearby_words, "PIN", header_y, y_tolerance=40.0)
            direction_word = find_exact_word(nearby_words, "DIRECTION", header_y, y_tolerance=40.0)
            type_word = find_exact_word(nearby_words, "TYPE", header_y, y_tolerance=40.0)
            description_word = find_exact_word(nearby_words, "DESCRIPTION", header_y, y_tolerance=40.0)

            signal_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "SIGNAL"
            ]
            name_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "NAME"
            ]
            ref_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "REF"
            ]
            des_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "DES"
            ]

            if pin_word is None or direction_word is None or type_word is None or description_word is None:
                continue
            if not signal_candidates or not ref_candidates or not des_candidates:
                continue

            # Pick the REF/DES pair that lives to the left of PIN.
            pin_center = word_center_x(pin_word)
            ref_candidates = [word for word in ref_candidates if word_center_x(word) < pin_center]
            des_candidates = [word for word in des_candidates if word_center_x(word) < pin_center]
            if not ref_candidates or not des_candidates:
                continue

            ref_word = min(ref_candidates, key=lambda word: abs(word_center_x(word) - pin_center))
            des_word = min(des_candidates, key=lambda word: abs(word_center_x(word) - word_center_x(ref_word)))
            ref_des_center = (word_center_x(ref_word) + word_center_x(des_word)) / 2.0

            # SIGNAL is enough to establish the column. If NAME is present nearby,
            # average the two centers for a better estimate of the visual header center.
            signal_word = min(signal_candidates, key=lambda word: abs(float(word["top"]) - header_y))
            signal_center = word_center_x(signal_word)
            if name_candidates:
                plausible_names = [
                    word for word in name_candidates
                    if abs(word_center_x(word) - signal_center) < 80.0
                ]
                if plausible_names:
                    name_word = min(plausible_names, key=lambda word: abs(word_center_x(word) - signal_center))
                    signal_center = (signal_center + word_center_x(name_word)) / 2.0

            centers = [
                ref_des_center,
                pin_center,
                signal_center,
                word_center_x(direction_word),
                word_center_x(type_word),
                word_center_x(description_word),
            ]

            if any(centers[index] >= centers[index + 1] for index in range(len(centers) - 1)):
                continue

            print_learned_columns(page_number, column_names, centers)
            return centers, page_data["width"], page_number

    raise Exception("Could not automatically determine the 6-column pinout locations from PDF headers")


def find_column_centers(pdf_cache, column_names):
    if len(column_names) == 9:
        return find_nine_column_centers(pdf_cache, column_names)

    if len(column_names) == 6:
        return find_six_column_centers(pdf_cache, column_names)

    raise ValueError(
        f"Unsupported physical column count: {len(column_names)}. "
        "This scraper supports 6-column pinout tables and 9-column wiring tables."
    )


def build_column_boundaries(pdf_cache, column_centers, page_width, header_page):
    page_data = pdf_cache[header_page]
    clusters = cluster_vertical_edges(page_data.get("edges", []))
    boundaries = []
    left_candidates = [cluster for cluster in clusters if cluster["x"] < column_centers[0]]
    right_candidates = [cluster for cluster in clusters if cluster["x"] > column_centers[-1]]
    left_boundary = max(left_candidates, key=lambda cluster: cluster["x"])["x"] if left_candidates else 0.0
    right_boundary = min(right_candidates, key=lambda cluster: cluster["x"])["x"] if right_candidates else page_width
    boundaries.append(left_boundary)
    grid_success = True
    for index in range(len(column_centers) - 1):
        boundary = choose_grid_boundary(clusters, column_centers[index], column_centers[index + 1])
        if boundary is None:
            grid_success = False
            boundary = (column_centers[index] + column_centers[index + 1]) / 2.0
        boundaries.append(boundary)
    boundaries.append(right_boundary)
    if any(boundaries[index] >= boundaries[index + 1] for index in range(len(boundaries) - 1)):
        boundaries = [0.0]
        for index in range(len(column_centers) - 1):
            boundaries.append((column_centers[index] + column_centers[index + 1]) / 2.0)
        boundaries.append(page_width)
        grid_success = False
    print("")
    print("============================================================")
    print("TABLE COLUMN BOUNDARIES")
    print("============================================================")
    print("METHOD:", "PDF GRID LINES" if grid_success else "GRID + MIDPOINT FALLBACK")
    for index, boundary in enumerate(boundaries):
        print("BOUNDARY", index, "X =", round(boundary, 2))
    print("============================================================")
    return boundaries


def get_column_index_for_x(x_value, boundaries):
    for index in range(len(boundaries) - 1):
        if boundaries[index] <= x_value < boundaries[index + 1]:
            return index
    return None

def is_connector_name(text):
    return re.fullmatch(r"J\d+[A-Z]*", canonical_connector_name(text), re.IGNORECASE) is not None


def find_connector_anchors(chars, boundaries):
    first_column_right = boundaries[1]
    first_column_chars = [char for char in chars if char_center_x(char) < first_column_right]
    lines = group_chars_into_lines(first_column_chars, y_tolerance=2.5)
    anchors = []

    for line in lines:
        value = rebuild_cell_from_chars(line["chars"])
        connectors = extract_connector_names(value)

        if not connectors:
            continue

        line_top = min(float(char["top"]) for char in line["chars"])
        line_bottom = max(float(char["bottom"]) for char in line["chars"])

        # One physical row can explicitly apply to multiple connectors, e.g.
        # "J1R / J2R".  Keep one row anchor and carry all connector aliases.
        anchors.append({
            "connector": connectors[0],
            "connectors": connectors,
            "top": line_top,
            "bottom": line_bottom,
        })

    anchors.sort(key=lambda anchor: anchor["top"])
    deduped = []

    for anchor in anchors:
        if deduped and abs(anchor["top"] - deduped[-1]["top"]) <= 2.0:
            # Merge aliases rather than dropping a second connector printed on
            # the same physical line.
            for connector in anchor.get("connectors", []):
                if connector not in deduped[-1]["connectors"]:
                    deduped[-1]["connectors"].append(connector)
            continue

        deduped.append(anchor)

    return deduped


# ================================================
# ROW GRID GEOMETRY
# ================================================
def edge_horizontal_y(edge):
    orientation = str(edge.get("orientation", "")).lower()
    if "top" in edge and "bottom" in edge:
        top = float(edge["top"])
        bottom = float(edge["bottom"])
        if orientation == "h" or abs(bottom - top) <= 1.5:
            return (top + bottom) / 2.0
    return None


def get_horizontal_table_grid_lines(edges, table_left, table_right, tolerance=1.25):
    table_width = max(1.0, table_right - table_left)
    raw = []
    for edge in edges:
        y = edge_horizontal_y(edge)
        if y is None:
            continue
        if "x0" not in edge or "x1" not in edge:
            continue
        x0 = min(float(edge["x0"]), float(edge["x1"]))
        x1 = max(float(edge["x0"]), float(edge["x1"]))
        overlap = max(0.0, min(x1, table_right) - max(x0, table_left))
        if overlap <= 0.0:
            continue
        raw.append((y, overlap))
    raw.sort(key=lambda item: item[0])
    clusters = []
    for y, overlap in raw:
        if not clusters or abs(y - clusters[-1]["y"]) > tolerance:
            clusters.append({"y": y, "overlap": overlap, "count": 1})
        else:
            cluster = clusters[-1]
            total = cluster["count"] + 1
            cluster["y"] = ((cluster["y"] * cluster["count"]) + y) / total
            cluster["overlap"] += overlap
            cluster["count"] = total
    minimum_coverage = table_width * 0.55
    return [cluster["y"] for cluster in clusters if cluster["overlap"] >= minimum_coverage]


def find_grid_row_bounds(anchor, horizontal_grid_lines):
    anchor_top = float(anchor["top"])
    anchor_bottom = float(anchor.get("bottom", anchor_top + 6.0))
    above = [y for y in horizontal_grid_lines if y < anchor_top - 0.25]
    below = [y for y in horizontal_grid_lines if y > anchor_bottom + 0.25]
    if not above or not below:
        return None
    row_top = max(above)
    row_bottom = min(below)
    row_height = row_bottom - row_top
    if row_height < 6.0 or row_height > 80.0:
        return None
    return row_top, row_bottom


# ================================================
# PARSE ONE ROW USING CHARACTER GEOMETRY
# ================================================
def parse_row_chars(row_chars, boundaries, column_names):
    column_chars = [[] for _ in column_names]
    for char in row_chars:
        column_index = get_column_index_for_x(char_center_x(char), boundaries)
        if column_index is None or column_index < 0 or column_index >= len(column_names):
            continue
        column_chars[column_index].append(char)
    output = {}
    for index, column_name in enumerate(column_names):
        output[column_name] = rebuild_cell_from_chars(column_chars[index])
    return output


# ================================================
# 6-COLUMN PINOUT ROW HELPERS
# ================================================
def find_pin_anchors(chars, boundaries, pin_index):
    if pin_index < 0 or pin_index >= len(boundaries) - 1:
        return []

    left = boundaries[pin_index]
    right = boundaries[pin_index + 1]

    pin_chars = [
        char for char in chars
        if left <= char_center_x(char) < right
    ]

    lines = group_chars_into_lines(pin_chars, y_tolerance=2.5)
    anchors = []

    for line in lines:
        value = rebuild_cell_from_chars(line["chars"])

        if not re.fullmatch(r"\d+", value):
            continue

        line_top = min(float(char["top"]) for char in line["chars"])
        line_bottom = max(float(char["bottom"]) for char in line["chars"])

        anchors.append({
            "pin": value,
            "top": line_top,
            "bottom": line_bottom,
        })

    anchors.sort(key=lambda anchor: anchor["top"])

    deduped = []

    for anchor in anchors:
        if deduped and abs(anchor["top"] - deduped[-1]["top"]) <= 2.0:
            # Keep the first numeric value found on this text line.
            continue
        deduped.append(anchor)

    return deduped


def find_connector_labels(chars, boundaries, connector_index):
    if connector_index < 0 or connector_index >= len(boundaries) - 1:
        return []

    left = boundaries[connector_index]
    right = boundaries[connector_index + 1]

    connector_chars = [
        char for char in chars
        if left <= char_center_x(char) < right
    ]

    lines = group_chars_into_lines(connector_chars, y_tolerance=2.5)
    labels = []

    for line in lines:
        raw_value = rebuild_cell_from_chars(line["chars"])
        connectors = extract_connector_names(raw_value)

        if not connectors:
            continue

        line_top = min(float(char["top"]) for char in line["chars"])
        line_bottom = max(float(char["bottom"]) for char in line["chars"])

        labels.append({
            "connector": connectors[0],
            "connectors": connectors,
            "top": line_top,
            "bottom": line_bottom,
            "center": (line_top + line_bottom) / 2.0,
        })

    labels.sort(key=lambda item: item["center"])

    deduped = []

    for label in labels:
        if deduped and abs(label["center"] - deduped[-1]["center"]) <= 2.0:
            # A combined label may be split into multiple text fragments by the
            # PDF. Merge all connector names at the same vertical location.
            for connector in label.get("connectors", []):
                if connector not in deduped[-1]["connectors"]:
                    deduped[-1]["connectors"].append(connector)
            continue

        deduped.append(label)

    return deduped


def split_pin_rows_into_groups(rows):
    if not rows:
        return []

    tops = [float(row["_anchor_top"]) for row in rows]
    gaps = [
        tops[index + 1] - tops[index]
        for index in range(len(tops) - 1)
        if tops[index + 1] - tops[index] > 0.0
    ]

    normal_gap = statistics.median(gaps) if gaps else 20.0
    large_gap_threshold = max(normal_gap * 2.5, 35.0)

    groups = [[rows[0]]]

    for row in rows[1:]:
        previous = groups[-1][-1]

        current_pin = int(row["_pin_number"])
        previous_pin = int(previous["_pin_number"])
        vertical_gap = float(row["_anchor_top"]) - float(previous["_anchor_top"])

        starts_new_group = (
            current_pin <= previous_pin
            or vertical_gap > large_gap_threshold
        )

        if starts_new_group:
            groups.append([row])
        else:
            groups[-1].append(row)

    return groups


def _unique_connectors(values):
    output = []

    for value in values:
        for connector in extract_connector_names(value):
            if connector not in output:
                output.append(connector)

    return output


def choose_connectors_for_pin_group(group, connector_labels, last_connectors=None):
    if last_connectors is None:
        last_connectors = []

    # First prefer any valid connector text that was physically inside a row.
    row_connectors = []

    for row in group:
        for connector in row.get("_parsed_connectors", []):
            if is_connector_name(connector) and connector not in row_connectors:
                row_connectors.append(connector)

    if row_connectors:
        return row_connectors

    group_top = min(float(row["_row_top"]) for row in group)
    group_bottom = max(float(row["_row_bottom"]) for row in group)
    group_center = (group_top + group_bottom) / 2.0

    inside = [
        label for label in connector_labels
        if group_top - 3.0 <= label["center"] <= group_bottom + 3.0
    ]

    if inside:
        connectors = []

        # All labels physically inside the same merged REF DES area apply to
        # this pin group. This is the key behavior for shared pinouts such as
        # J1R + J2R using one common table.
        for label in inside:
            for connector in label.get("connectors", [label.get("connector", "")]):
                if is_connector_name(connector) and connector not in connectors:
                    connectors.append(connector)

        if connectors:
            return connectors

    # If this page has one merged REF DES label, it belongs to the visible pin table.
    if len(connector_labels) == 1:
        return list(connector_labels[0].get("connectors", [connector_labels[0]["connector"]]))

    # Multiple tables can share a page. Choose the closest label cluster.
    if connector_labels:
        nearest = min(
            connector_labels,
            key=lambda label: abs(label["center"] - group_center)
        )

        # Avoid borrowing a connector from a very distant table.
        if abs(nearest["center"] - group_center) <= max(group_bottom - group_top, 80.0):
            cluster_tolerance = 12.0
            connectors = []

            for label in connector_labels:
                if abs(label["center"] - nearest["center"]) <= cluster_tolerance:
                    for connector in label.get("connectors", [label.get("connector", "")]):
                        if is_connector_name(connector) and connector not in connectors:
                            connectors.append(connector)

            if connectors:
                return connectors

    # Continuation-page fallback: if pins continue above 1 and the REF DES merged
    # cell was not repeated on the new page, keep every connector from the prior
    # shared table.
    first_pin = int(group[0]["_pin_number"])

    if last_connectors and first_pin > 1:
        return list(last_connectors)

    return []


# ================================================
# EXTRACT 6-COLUMN PINOUT TABLE
#
# REF DES is often one merged cell spanning many
# pin rows, so PIN is the row anchor instead of
# requiring the connector name on every row.
# ================================================
def extract_structured_rows_six_column(pdf_cache, separator_keyword, column_names, pdf_roles):
    start_time = time.perf_counter()

    column_centers, page_width, header_page = find_column_centers(pdf_cache, column_names)
    reference_boundaries = build_column_boundaries(
        pdf_cache,
        column_centers,
        page_width,
        header_page
    )

    section_markers = find_section_markers(pdf_cache, separator_keyword)

    connector_index = column_names.index(pdf_roles["connector"])
    pin_index = column_names.index(pdf_roles["pin"])
    signal_index = column_names.index(pdf_roles["signal"])
    skip_direction = should_skip_direction(pdf_roles)
    direction_index = None if skip_direction else column_names.index(pdf_roles["direction"])

    structured_rows = []
    grid_bounded_rows = 0
    fallback_bounded_rows = 0
    last_connectors = []

    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        chars = page_data["chars"]
        words = page_data["words"]

        if not chars:
            continue

        boundaries = reference_boundaries

        pin_anchors = find_pin_anchors(chars, boundaries, pin_index)

        if not pin_anchors:
            continue

        connector_labels = find_connector_labels(
            chars,
            boundaries,
            connector_index
        )

        horizontal_grid_lines = get_horizontal_table_grid_lines(
            page_data.get("edges", []),
            boundaries[0],
            boundaries[-1]
        )

        anchor_tops = [float(anchor["top"]) for anchor in pin_anchors]
        gaps = [
            anchor_tops[index + 1] - anchor_tops[index]
            for index in range(len(anchor_tops) - 1)
            if anchor_tops[index + 1] - anchor_tops[index] > 0
        ]

        normal_row_height = statistics.median(gaps) if gaps else 22.0
        maximum_last_row_height = max(normal_row_height * 2.75, 50.0)

        page_bottom = (
            page_data["footer_top"] - 2.0
            if page_data.get("footer_top") is not None
            else page_data["height"]
        )

        page_rows = []

        for anchor_index, anchor in enumerate(pin_anchors):
            grid_bounds = find_grid_row_bounds(anchor, horizontal_grid_lines)

            if grid_bounds is not None:
                row_top = grid_bounds[0] + 0.35
                row_bottom = grid_bounds[1] - 0.35
                grid_bounded_rows += 1
            else:
                row_top = float(anchor["top"]) - 2.0
                row_bottom = (
                    float(pin_anchors[anchor_index + 1]["top"]) - 1.0
                    if anchor_index + 1 < len(pin_anchors)
                    else float(anchor["top"]) + maximum_last_row_height
                )
                fallback_bounded_rows += 1

            row_bottom = min(row_bottom, page_bottom)

            if row_bottom <= row_top:
                continue

            row_chars = [
                char for char in chars
                if row_top <= float(char["top"]) < row_bottom
            ]

            row_words = [
                word for word in words
                if row_top <= float(word["top"]) < row_bottom
            ]

            parsed_row = parse_row_chars(
                row_chars,
                boundaries,
                column_names
            )

            # PIN is our authoritative row anchor.
            parsed_row[pdf_roles["pin"]] = str(anchor["pin"])

            signal_from_words = extract_local_signal_from_words(
                row_words,
                boundaries,
                signal_index
            )

            if signal_from_words != "":
                parsed_row[pdf_roles["signal"]] = normalize_local_signal_value(
                    signal_from_words
                )
            else:
                parsed_row[pdf_roles["signal"]] = normalize_local_signal_value(
                    parsed_row.get(pdf_roles["signal"], "")
                )

            if not skip_direction:
                direction_from_words = extract_role_text_from_words(
                    row_words,
                    boundaries,
                    direction_index,
                    left_pad=8.0,
                    right_pad=8.0,
                    skip_numeric=False
                )

                direction_value = normalize_direction_value(direction_from_words)

                if direction_value == "":
                    direction_value = normalize_direction_value(
                        parsed_row.get(pdf_roles["direction"], "")
                    )

                if direction_value == "":
                    direction_value = find_direction_from_row_words(
                        row_words,
                        boundaries,
                        direction_index
                    )

                parsed_row[pdf_roles["direction"]] = direction_value

            parsed_connectors = extract_connector_names(
                parsed_row.get(pdf_roles["connector"], "")
            )

            section_marker = find_section_for_row(
                page_number,
                float(anchor["top"]),
                section_markers
            )

            parsed_row["PAGE"] = page_number
            parsed_row["SECTION"] = (
                section_marker["section_number"]
                if section_marker
                else None
            )
            parsed_row["SECTION_TITLE"] = (
                section_marker["title"]
                if section_marker
                else ""
            )

            # Temporary geometry fields for merged REF DES assignment.
            parsed_row["_anchor_top"] = float(anchor["top"])
            parsed_row["_row_top"] = float(row_top)
            parsed_row["_row_bottom"] = float(row_bottom)
            parsed_row["_pin_number"] = int(anchor["pin"])
            parsed_row["_parsed_connectors"] = parsed_connectors

            page_rows.append(parsed_row)

        pin_groups = split_pin_rows_into_groups(page_rows)

        for group in pin_groups:
            connectors = choose_connectors_for_pin_group(
                group,
                connector_labels,
                last_connectors=last_connectors
            )

            if not connectors:
                print(
                    "WARNING: Could not assign REF DES to pin group on page",
                    page_number,
                    "pins",
                    group[0]["_pin_number"],
                    "through",
                    group[-1]["_pin_number"]
                )
                continue

            if len(connectors) > 1:
                print(
                    "SHARED CONNECTOR PINOUT:",
                    ", ".join(connectors),
                    "| PAGE",
                    page_number,
                    "| PINS",
                    group[0]["_pin_number"],
                    "THROUGH",
                    group[-1]["_pin_number"]
                )

            for parsed_row in group:
                base_row = dict(parsed_row)
                base_row.pop("_anchor_top", None)
                base_row.pop("_row_top", None)
                base_row.pop("_row_bottom", None)
                base_row.pop("_pin_number", None)
                base_row.pop("_parsed_connectors", None)

                # Duplicate the same physical PDF row under every connector
                # named by the shared REF DES cell. Excel matching is by name,
                # so J1R and J2R can appear in any spreadsheet order.
                for connector in connectors:
                    output_row = dict(base_row)
                    output_row[pdf_roles["connector"]] = connector
                    structured_rows.append(output_row)

            last_connectors = list(connectors)

    elapsed = time.perf_counter() - start_time

    print("PARSER MODE: 6-COLUMN PINOUT")
    print("GRID-BOUNDED ROWS:", grid_bounded_rows)
    print("FALLBACK-BOUNDED ROWS:", fallback_bounded_rows)
    print(f"STRUCTURED ROW EXTRACTION TIME: {elapsed:.2f} SECONDS")

    return structured_rows

def extract_structured_rows(pdf_cache, separator_keyword, column_names, pdf_roles):
    if len(column_names) == 6:
        return extract_structured_rows_six_column(
            pdf_cache,
            separator_keyword,
            column_names,
            pdf_roles
        )

    if len(column_names) == 9:
        return extract_structured_rows_nine_column(
            pdf_cache,
            separator_keyword,
            column_names,
            pdf_roles
        )

    raise ValueError(
        f"Unsupported physical column count: {len(column_names)}. "
        "Use 6 columns for pinout tables or 9 columns for wiring tables."
    )

def validate_row(row, pdf_roles):
    problems = []
    connectors = extract_connector_names(get_role_value(row, pdf_roles, "connector"))
    pin_text = get_role_value(row, pdf_roles, "pin")

    if not connectors:
        problems.append("invalid connector")

    if not re.fullmatch(r"\d+", pin_text):
        problems.append("invalid pin")

    return problems


def validate_structured_rows(structured_rows, pdf_roles):
    good_rows = []
    suspicious_rows = []
    for row in structured_rows:
        problems = validate_row(row, pdf_roles)
        if problems:
            suspicious_rows.append({"row": row, "problems": problems})
        else:
            good_rows.append(row)
    return good_rows, suspicious_rows


def build_pdf_connector_map(structured_rows, pdf_roles):
    pdf_connectors = {}

    for row in structured_rows:
        connectors = extract_connector_names(
            get_role_value(row, pdf_roles, "connector")
        )

        for connector in connectors:
            if not is_connector_name(connector):
                continue

            # Store a connector-specific copy so a shared PDF table can safely
            # feed multiple independent spreadsheet blocks.
            mapped_row = dict(row)
            mapped_row[pdf_roles["connector"]] = connector
            pdf_connectors.setdefault(connector, []).append(mapped_row)

    return pdf_connectors


def get_connector_pin_map(rows, pdf_roles):
    pin_map = {}
    for row in rows:
        pin_text = get_role_value(row, pdf_roles, "pin")
        if not re.fullmatch(r"\d+", pin_text):
            continue
        pin_map[int(pin_text)] = row
    return pin_map

    print("")

        else:
            print(connector, "| NO VALID PINS")
    print("============================================================")

def read_excel_connector_info(excel_path):
    import win32com.client as win32
    start_time = time.perf_counter()
    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    workbook = None
    try:
        workbook = excel.Workbooks.Open(os.path.abspath(excel_path), UpdateLinks=0, ReadOnly=True, AddToMru=False)
        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)
        raw_count = worksheet.Range(EXCEL_CONNECTOR_COUNT_CELL).Value2
        connector_count = int(raw_count) if raw_count is not None and str(raw_count).strip() != "" else 0
        if connector_count < 0:
            connector_count = 0
        if connector_count > MAX_BLOCKS:
            connector_count = MAX_BLOCKS
        if connector_count == 0:
            return [], {}
        data_range = worksheet.Range(worksheet.Cells(EXCEL_CONNECTOR_FIRST_ROW, EXCEL_CONNECTOR_NAME_COLUMN), worksheet.Cells(EXCEL_CONNECTOR_FIRST_ROW + connector_count - 1, EXCEL_EXPECTED_PIN_COUNT_COLUMN))
        data = data_range.Value2
        connector_info = []
        expected_counts = {}
        for index, row_values in enumerate(data):
            excel_name = str(row_values[0] or "").strip()
            canonical = canonical_connector_name(excel_name)
            raw_expected = row_values[EXCEL_EXPECTED_PIN_COUNT_COLUMN - EXCEL_CONNECTOR_NAME_COLUMN]
            try:
                expected = int(raw_expected)
            except Exception:
                expected = None
            connector_info.append({"block_index": index, "excel_row": EXCEL_CONNECTOR_FIRST_ROW + index, "excel_name": excel_name, "canonical": canonical, "expected_pins": expected})
            if canonical:
                expected_counts[canonical] = expected
        elapsed = time.perf_counter() - start_time
        print(f"EXCEL CONNECTOR READ TIME: {elapsed:.2f} SECONDS")
        return connector_info, expected_counts
    finally:
        if workbook is not None:
            workbook.Close(SaveChanges=False)
        excel.Quit()


def build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info):
    jobs = []
    skip_direction = should_skip_direction(pdf_roles)

    for connector in connector_info:
        block_index = connector["block_index"]
        excel_name = connector["excel_name"]
        canonical = connector["canonical"]
        expected_pins = connector["expected_pins"]

        if canonical not in pdf_connectors:
            jobs.append({
                "block_index": block_index,
                "excel_name": excel_name,
                "canonical": canonical,
                "expected_pins": expected_pins,
                "skip": True,
                "reason": "connector not found in PDF"
            })
            continue

        pin_map = get_connector_pin_map(pdf_connectors[canonical], pdf_roles)

        if expected_pins is not None and expected_pins > BLOCK_DATA_ROW_CAPACITY:
            raise ValueError(
                f"{excel_name} expects {expected_pins} pins but "
                f"BLOCK_DATA_ROW_CAPACITY is only {BLOCK_DATA_ROW_CAPACITY}"
            )

        # Normal mode writes PIN + SIGNAL + IN/OUT.
        # Skip-direction mode intentionally writes only PIN + SIGNAL so the
        # existing spreadsheet IN/OUT column is never cleared or overwritten.
        matrix_width = 2 if skip_direction else 3
        matrix = [[""] * matrix_width for _ in range(BLOCK_DATA_ROW_CAPACITY)]

        for pin_number, row in pin_map.items():
            if pin_number < 1 or pin_number > BLOCK_DATA_ROW_CAPACITY:
                continue

            matrix[pin_number - 1][0] = pin_number
            matrix[pin_number - 1][1] = normalize_local_signal_value(
                get_role_value(row, pdf_roles, "signal")
            )

            if not skip_direction:
                matrix[pin_number - 1][2] = normalize_direction_value(
                    get_role_value(row, pdf_roles, "direction")
                )

        missing = []
        if expected_pins is not None and expected_pins > 0:
            missing = [
                pin_number
                for pin_number in range(1, expected_pins + 1)
                if pin_number not in pin_map
            ]

        block_start_column = BLOCK_FIRST_COLUMN + block_index * BLOCK_WIDTH
        pin_column = block_start_column + BLOCK_PIN_COLUMN_OFFSET
        signal_column = block_start_column + BLOCK_SIGNAL_COLUMN_OFFSET
        direction_column = block_start_column + BLOCK_DIRECTION_COLUMN_OFFSET

        jobs.append({
            "block_index": block_index,
            "excel_name": excel_name,
            "canonical": canonical,
            "expected_pins": expected_pins,
            "skip": False,
            "pin_count": len(pin_map),
            "missing": missing,
            "first_row": BLOCK_DATA_FIRST_ROW,
            "last_row": BLOCK_DATA_FIRST_ROW + BLOCK_DATA_ROW_CAPACITY - 1,
            "first_column": pin_column,
            "last_column": signal_column if skip_direction else direction_column,
            "matrix": tuple(tuple(row) for row in matrix),
            "write_direction": not skip_direction,
        })

    return jobs


def write_pdf_data_to_excel(source_excel_path, output_excel_path, pdf_connectors, pdf_roles, connector_info):
    import win32com.client as win32
    total_start = time.perf_counter()
    source_excel_path = os.path.abspath(source_excel_path)
    output_excel_path = os.path.abspath(output_excel_path)
    if source_excel_path.lower() != output_excel_path.lower():
        shutil.copy2(source_excel_path, output_excel_path)
    jobs = build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info)
    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    excel.AskToUpdateLinks = False
    workbook = None
    previous_calculation = None
    calculation_changed = False
    try:
        open_start = time.perf_counter()
        workbook = excel.Workbooks.Open(output_excel_path, UpdateLinks=0, ReadOnly=False, AddToMru=False)
        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)
        open_elapsed = time.perf_counter() - open_start
        print(f"EXCEL OPEN TIME: {open_elapsed:.2f} SECONDS")
        try:
            previous_calculation = excel.Calculation
            excel.Calculation = XL_CALCULATION_MANUAL
            calculation_changed = True
            print("EXCEL CALCULATION MODE: MANUAL")
        except Exception as e:
            print("WARNING: COULD NOT SET MANUAL CALCULATION:", repr(e))
        print("")
        print("============================================================")
        print("EXCEL CONNECTOR CHECK")
        print("============================================================")
        print("EXCEL CONNECTORS:", len(connector_info))
        print("PDF CONNECTORS:", len(pdf_connectors))
        print("MATCH MODE: CONNECTOR NAME (PDF/EXCEL ORDER DOES NOT MATTER)")
        print(
            "IN/OUT WRITE MODE:",
            "SKIPPED - EXISTING EXCEL VALUES LEFT UNCHANGED"
            if should_skip_direction(pdf_roles)
            else "WRITE FROM PDF"
        )
        print("============================================================")
        write_start = time.perf_counter()
        written_blocks = 0
        written_pin_rows = 0
        for job in jobs:
            print("")
            print("BLOCK:", job["block_index"] + 1, "EXCEL CONNECTOR:", repr(job["excel_name"]), "PDF KEY:", repr(job["canonical"]))
            if job["skip"]:
                print("SKIPPED -", job["reason"], ":", job["excel_name"])
                continue
            destination_range = worksheet.Range(worksheet.Cells(job["first_row"], job["first_column"]), worksheet.Cells(job["last_row"], job["last_column"]))
            destination_range.Value2 = job["matrix"]
            written_blocks += 1
            written_pin_rows += job["pin_count"]
            print("WRITTEN PINS:", job["pin_count"], "EXPECTED:", job["expected_pins"], "MISSING:", job["missing"])
            if not job.get("write_direction", True):
                print("IN/OUT COLUMN: SKIPPED (LEFT UNCHANGED IN EXCEL)")
        write_elapsed = time.perf_counter() - write_start
        print(f"BULK EXCEL WRITE TIME: {write_elapsed:.2f} SECONDS")
        calc_start = time.perf_counter()
        try:
            worksheet.Calculate()
            print(f"SELECTION SHEET CALCULATION TIME: {time.perf_counter() - calc_start:.2f} SECONDS")
        except Exception as e:
            print("WARNING: SELECTION SHEET CALCULATION FAILED:", repr(e))
        if calculation_changed and previous_calculation is not None:
            try:
                excel.Calculation = previous_calculation
            except Exception as e:
                print("WARNING: COULD NOT RESTORE CALCULATION MODE:", repr(e))
        save_start = time.perf_counter()
        workbook.Save()
        save_elapsed = time.perf_counter() - save_start
        print(f"EXCEL SAVE TIME: {save_elapsed:.2f} SECONDS")
        print("")
        print("============================================================")
        print("EXCEL WRITE COMPLETE")
        print("============================================================")
        print("OUTPUT:", output_excel_path)
        print("BLOCKS WRITTEN:", written_blocks)
        print("PIN ROWS WRITTEN:", written_pin_rows)
        print("============================================================")
    finally:
        if workbook is not None:
            try:
                workbook.Close(SaveChanges=False)
            except Exception:
                pass
        try:
            excel.EnableEvents = True
            excel.ScreenUpdating = True
        except Exception:
            pass
        try:
            excel.Quit()
        except Exception:
            pass
    total_elapsed = time.perf_counter() - total_start
    print(f"TOTAL EXCEL WRITE PHASE: {total_elapsed:.2f} SECONDS")
