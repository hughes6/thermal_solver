"""Zuken E3.series: Excel J2 local-signal importer.

Version 2.6: Display missing pin-owned Local Signal Name attributes before making their placed text visible; retain v2.4 completion/error windows.
Looks up device designation text on connector-owned symbols in addition
 to the pin; prints before/after text style; no log file.
Run from E3 or Python, choose one Excel workbook, and immediately apply.
Diagnostics are captured in memory; on success only a small completion window appears.
On error, detailed diagnostics are displayed with a Copy button. No log file.

Excel worksheet: any name; uses the matching data tab (or active tab).
E3 drawing sheet 'J2' is used only to locate the block.

Worksheet cells:
    B2    = duplicate-pin status/check (if present)
    C4    = block device designation
    C5    = connector count N
    B7:B(6+N) = connector names
    C7:C(6+N) = connector types (additional identity information)
    I:J   = connector 1: pin, local signal
    M:N   = connector 2: pin, local signal
    Q:R   = connector 3, etc., all starting on row 7
    K:L, O:P, ... are intentionally unused spacer columns

Only rows with BOTH a pin value and a nonblank local signal are imported.
The header row (Pin Number / Local Signal Name) is ignored automatically.
Blank signal cells do not clear E3 data. Duplicate pin names within a
connector are the only spreadsheet/model integrity check. E3 object ownership
is additionally checked before writes to avoid modifying the wrong object.

IMPORTANT: This version NEVER calls Pin.SetSignalName() or writes type-7
reference signal text because those can change the electrical net and
propagate into other connectors. It writes only to an explicitly existing
pin-owned attribute named Local Signal Name (or listed direct alias).
When no such field exists, it performs READ-ONLY diagnostics and prints
what E3 exposes, without changing signal connectivity.
It calls Attribute.DisplayAttribute on hidden pin-owned Local Signal Name
attributes, then formats actual placed pin-view text type 617 after matching
its logical pin and connector. Type-12 designation is enabled and bolded on the unique
top placed pin identified by split-position marker type 213 (1/N).
Unlike E3 electrical net signal, these changes are only text styling.

Test in a copy of the E3 project. COM support cannot be run on non-Windows hosts.
Requires: py -m pip install pywin32 openpyxl
Optional: Zuken's version-matched 'e3series' Python package.
"""

SHEET_NAME = 'J2'  # E3 drawing-page identifier, NOT a required Excel tab name
VERSION = '2.6-show-pin-attributes'
FIRST_ROW = 7
FIRST_PIN_COL = 9  # I
CONNECTOR_COL_STEP = 4  # I:J, M:N, Q:R, U:V, ...
LOCAL_SIGNAL_TEXT_TYPE = 617  # Displayed Local Signal Name on placed pin view
DEVICE_DESIGNATION_TEXT_TYPE = 12
TEXT_HEIGHT_IN = 0.15
BOLD_STYLE = 1
NO_VALUE = -2147352572  # COM's DISP_E_PARAMNOTFOUND


class ConsoleLines(list):
    """Pass diagnostics through the host interpreter with plain print()."""
    def __init__(self, lines=()):
        super().__init__()
        self.extend(lines)

    def append(self, line):
        line = str(line)
        super().append(line)
        print(line, flush=True)

    def extend(self, lines):
        for line in lines:
            self.append(line)

    def __iadd__(self, lines):
        self.extend(lines)
        return self


@dataclass(frozen=True)
class PinEntry:
    connector_index: int
    connector_name: str
    connector_type: str
    pin_name: str
    signal_name: str
    cell: str


@dataclass
class ImportSheet:
    workbook_path: Path
    block_name: str
    connector_count: int
    entries: list[PinEntry]
    check_b2: object
    duplicate_problems: list[str] = field(default_factory=list)
    worksheet_name: str = ''


@dataclass
class ResolvedPin:
    entry: PinEntry
    connector_id: int
    pin_id: int


def as_text(value):
    if value is None:
        return ''
    if isinstance(value, float) and value.is_integer():
        return str(int(value))
    return str(value).strip()


def device_key(value):
    """Treat a leading E3 '-' designation marker as optional, not arbitrary substrings."""
    return as_text(value).lstrip('-').casefold()


def connector_keys(value):
    """Recognize a connector with or without its block prefix ('-A1.J1')."""
    v = device_key(value)
    return {v, v.rsplit('.', 1)[-1]}


def duplicate_flag_b2(value):
    """Check commonly used B2 duplicate-pin outcomes, not unrelated statuses."""
    if isinstance(value, bool):
        return bool(value)
    if isinstance(value, (int, float)):
        return value > 0
    s = as_text(value).lower().replace('_', ' ').strip()
    if not s or s in ('ok', 'pass', 'passed', 'false', 'none', 'clear', 'no', '0'):
        return False
    if any(x in s for x in ('no duplicate', '0 duplicate', 'without duplicate', 'duplicate-free')):
        return False
    return s in ('error', 'fail', 'failed', 'true') or 'duplicate' in s


def select_data_sheet(wb):
    """Choose the Excel data tab; J2 refers to the *E3 page*, not Excel.

    Prefer a populated J2 tab if present, then the active Excel tab,
    otherwise the first tab with a designation in C4 and connector count C5.
    """
    def has_inputs(sh):
        if not as_text(sh['C4'].value):
            return False
        try:
            n = float(sh['C5'].value)
            return n >= 1 and n.is_integer()
        except (TypeError, ValueError):
            return False

    matches = [sh for sh in wb.worksheets if has_inputs(sh)]
    if not matches:
        tabs = ', '.join(repr(sh.title) for sh in wb.worksheets)
        raise ValueError(
            'Could not find an Excel worksheet with the device designation '
            'in C4 and connector count in C5. Excel tab names found: '
            + (tabs or '(none)')
        )
    for sh in matches:
        if sh.title.casefold() == SHEET_NAME.casefold():
            return sh
    if wb.active in matches:
        return wb.active
    return matches[0]


def parse_workbook(filename: Path) -> ImportSheet:
    """Read Excel values without modifying the workbook or recalculating it."""
    from openpyxl import load_workbook
    wb = load_workbook(str(filename), read_only=False, data_only=True)
    try:
        sh = select_data_sheet(wb)
        block = as_text(sh['C4'].value)
        if not block:
            raise ValueError(f'{sh.title}!C4 (device designation) is empty.')
        raw_count = sh['C5'].value
        try:
            connector_count = int(raw_count)
            if connector_count < 1 or float(raw_count) != connector_count:
                raise ValueError()
        except (ValueError, TypeError):
            raise ValueError(f'{sh.title}!C5 must contain the number of connectors.') from None

        entries = []
        problems = []
        for ci in range(connector_count):
            connector_name = as_text(sh.cell(FIRST_ROW + ci, 2).value)
            connector_type = as_text(sh.cell(FIRST_ROW + ci, 3).value)
            # Pin/signal boxes are separated by two blank columns: I:J, M:N, Q:R, ...
            pin_col = FIRST_PIN_COL + CONNECTOR_COL_STEP * ci
            signal_col = pin_col + 1
            pin_col_letter = sh.cell(FIRST_ROW, pin_col).column_letter
            signal_col_letter = sh.cell(FIRST_ROW, signal_col).column_letter
            if not connector_name:
                # A missing connector name is a matching problem, not a model check.
                problems.append(f'Connector {ci + 1}: blank name at B{FIRST_ROW + ci}.')
            seen_pins = {}
            # Respect data even if some pin rows are intentionally left blank.
            # max_row is set by the actual workbook, not by the declared pin count.
            for row in range(FIRST_ROW, sh.max_row + 1):
                pin = as_text(sh.cell(row, pin_col).value)
                signal = as_text(sh.cell(row, signal_col).value)
                # Templates commonly place headings directly in I7/J7, M7/N7, ...
                # Neither the headings nor their formatting should become an E3 pin.
                if pin.casefold().replace(' ', '') in ('pinnumber', 'pinname', 'pin#') \
                        and signal.casefold().replace(' ', '') in ('localsignalname', 'localsignal', 'signalname', 'signal'):
                    continue
                if pin:
                    # Check duplicates even where a local signal is blank;
                    # this is the one requested model/spreadsheet check.
                    if pin.casefold() in seen_pins:
                        problems.append(f'DUPLICATE pin {pin!r} in connector {connector_name!r} '
                                        f'at {pin_col_letter}{seen_pins[pin.casefold()]} '
                                        f'and {pin_col_letter}{row}.')
                        continue
                    seen_pins[pin.casefold()] = row
                if not signal:
                    continue  # never clear or reformat an unspecified E3 pin
                if not pin:
                    problems.append(f'{signal_col_letter}{row}: signal without a pin number.')
                    continue
                entries.append(PinEntry(ci, connector_name, connector_type,
                                        pin, signal, f'{signal_col_letter}{row}'))

        sheet_data = ImportSheet(Path(filename), block, connector_count,
                                 entries, sh['B2'].value, problems, sh.title)
        # A B2 formula may have no cached value when Excel has not calculated.
        # The actual duplicate scan above remains authoritative in that case.
        if duplicate_flag_b2(sheet_data.check_b2):
            sheet_data.duplicate_problems.insert(
                0, f'{sh.title}!B2 reports duplicate pins: ' + repr(sheet_data.check_b2))
        return sheet_data
    finally:
        wb.close()


def require_no_duplicate_pins(data):
    dup = [p for p in data.duplicate_problems if 'DUPLICATE' in p.upper()]
    if dup:
        raise ValueError('Duplicate pin check failed:\n' + '\n'.join(dup))


def flatten_positive_ids(obj):
    if isinstance(obj, (list, tuple)):
        vals = []
        for x in obj:
            vals.extend(flatten_positive_ids(x))
        return vals
    if type(obj) is int and 0 < obj < 2147483647:
        return [obj]
    return []


def split_com_result(result):
    # E3 methods generally return (number_of_ids, ID-array).
    if isinstance(result, (tuple, list)) and result and type(result[0]) is int:
        return result[0], flatten_positive_ids(result[1:])
    return (result if type(result) is int else None), []


def com_ids(obj, method_name, text_type=None):
    """Read E3 arrays using both official-wrapper and direct pywin32 conventions."""
    method = getattr(obj, method_name, None)
    if method is None:
        return []
    variants = [(), (None,)] if text_type is None else [(text_type,), (None, text_type), ([], text_type)]
    for args in variants:
        try:
            raw = method(*args)
            count, ids = split_com_result(raw)
            if ids:
                return list(dict.fromkeys(ids[:count] if count and count > 0 else ids))
            if isinstance(raw, (tuple, list)) and len(raw) > 1 and all(type(i) is int and i > 0 for i in raw):
                return list(dict.fromkeys(raw))
        except Exception:
            pass
    # The Python COM interface sometimes requires an explicit output array.
    try:
        import pythoncom
        import win32com.client as wc
    except ImportError:
        return []
    for variant_type, seed in (
        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, []),
    ):
        try:
            arr = wc.VARIANT(variant_type, seed)
            result = method(arr) if text_type is None else method(arr, text_type)
            _, returned_ids = split_com_result(result)
            ids = returned_ids or flatten_positive_ids(arr.value)
            if ids:
                return list(dict.fromkeys(ids))
        except Exception:
            pass
    return []


def related_ids(obj, candidates, text_type=None):
    for method in candidates:
        ids = com_ids(obj, method, text_type=text_type)
        if ids:
            return ids
    return []


def get_scalar(obj, methods):
    for name in methods:
        method = getattr(obj, name, None)
        if method is None:
            continue
        try:
            result = method()
        except Exception:
            continue
        if isinstance(result, (str, float, int)) and result != NO_VALUE:
            return result
    return None


def object_api_names(obj, filter_words=('signal', 'text', 'pin', 'connector', 'block', 'device', 'sheet')):
    """Inspect the COM type library safely, also works with the official wrapper."""
    found = set()
    try:
        found.update(str(n) for n in dir(obj) if not str(n).startswith('_'))
    except Exception:
        pass
    try:
        ti = obj._oleobj_.GetTypeInfo()
        ta = ti.GetTypeAttr()
        for i in range(ta.cFuncs):
            names = ti.GetNames(ti.GetFuncDesc(i).memid)
            if names:
                found.add(str(names[0]))
    except Exception:
        pass
    return sorted(x for x in found if any(w in x.lower() for w in filter_words))


def diagnostic_ids(obj, method_names):
    """Read-only COM enumeration diagnostics, limited to short summaries."""
    results = []
    for method_name in method_names:
        try:
            m = getattr(obj, method_name, None)
            if not callable(m):
                results.append(f'  {method_name}: NOT EXPOSED')
                continue
            try:
                raw = m()
                typ = type(raw).__name__
                if isinstance(raw, (list, tuple)):
                    summary = f'length={len(raw)}, first={repr(raw[:3])[:180]}'
                else:
                    summary = repr(raw)[:280]
                results.append(f'  {method_name}(): {typ} {summary}')
            except Exception as ex:
                results.append(f'  {method_name}(): {type(ex).__name__}: {str(ex)[:180]}')
            ids = com_ids(obj, method_name)
            results.append(f'  {method_name} parsed IDs={len(ids)}, first={ids[:8]}')
        except Exception as ex:
            results.append(f'  {method_name} probe FAILED: {type(ex).__name__}: {ex}')
    return results


def pin_diagnostics(job, pin_id):
    """Show available pin signal APIs, and local-signal (#7) text ownership."""
    notes = []
    try:
        pin = make_object(job, 'Pin', pin_id)
        notes.append('PIN COM APIs: ' + str(object_api_names(pin, ('signal', 'text', 'name', 'symbol'))[:90]))
        for method in ('GetName','GetPinName','GetLocalSignalName','GetSignalName','GetSignal','GetSignalId'):
            fn = getattr(pin, method, None)
            if not callable(fn):
                notes.append(f'  {method}: not exposed')
            else:
                try:
                    notes.append(f'  {method}() = {repr(fn())[:180]}')
                except Exception as ex:
                    notes.append(f'  {method}(): {type(ex).__name__}: {str(ex)[:150]}')
        notes.extend(diagnostic_ids(pin, ('GetTextIds','GetSymbolIds')))
        ids = related_ids(pin, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
        notes.append(f'  GetTextIds(type=7) parsed IDs={ids[:30]}')
    except Exception as ex:
        notes.append(f'pin diagnostic error: {type(ex).__name__}: {ex}')
    return notes


def make_object(job, kind, object_id):
    creator = getattr(job, f'Create{kind}Object', None)
    if not callable(creator):
        raise RuntimeError(f'E3 does not expose Create{kind}Object().')
    obj = creator()
    result = obj.SetId(int(object_id))
    if result is False or (type(result) is int and result < 0):
        raise RuntimeError(f'Could not resolve {kind} ID {object_id}: SetId returned {result!r}.')
    return obj


# E3 block connectors may return their *parent block* designation in GetName().
# The individual connector designation can instead be exposed by other
# read-only device/pin methods or by the .CONNECTOR_NAME pin attribute.
CONNECTOR_NAME_GETTERS = (
    'GetDeviceDesignation', 'GetConnectorDesignation', 'GetInternalDeviceDesignation',
    'GetInternalDesignation', 'GetBlockConnectorDesignation',
    'GetConnectorName', 'GetBlockConnectorName', 'GetInternalName',
    'GetDeviceName', 'GetFullName', 'GetDesignation', 'GetName',
)
CONNECTOR_ATTRIBUTES = (
    '.CONNECTOR_NAME', 'CONNECTOR_NAME', 'Internal Device Designation',
)


def read_string_getters(obj, getters):
    """Read *all* available name-like getters, not just the first one."""
    result = {}
    for name in getters:
        method = getattr(obj, name, None)
        if not callable(method):
            continue
        try:
            value = method()
        except Exception:
            continue
        if isinstance(value, str) and value.strip():
            result[name] = value.strip()
    return result


def read_attribute_names(obj):
    """Read the E3 internal connector designation without modifying attributes."""
    values = {}
    for meth in ('GetAttributeValue', 'GetAttribute', 'GetPinAttributeValue'):
        method = getattr(obj, meth, None)
        if not callable(method):
            continue
        for attname in CONNECTOR_ATTRIBUTES:
            try:
                value = method(attname)
            except Exception:
                continue
            if isinstance(value, str) and value.strip():
                values[f'{meth}({attname})'] = value.strip()
    return values


def extended_connector_aliases(job, connector_id, connector_obj, log):
    """Collect name evidence from the connector device and its own pins.

    We intentionally do NOT use Excel connector order as an E3 connector index:
    GetConnectorIds order is not guaranteed to match the spreadsheet.
    """
    names = read_string_getters(connector_obj, CONNECTOR_NAME_GETTERS)
    names.update(read_attribute_names(connector_obj))
    log.append(f'  CONNECTOR ID {connector_id} DEVICE NAME GETTERS: {names!r}')
    avail = object_api_names(connector_obj, ('name', 'designat', 'connector', 'attribute'))
    log.append(f'  CONNECTOR ID {connector_id} AVAILABLE NAME/ATTRIBUTE APIS: {avail[:110]!r}')

    # Inspect only pins E3 explicitly associates with this connector object.
    pin_ids = related_ids(connector_obj, ('GetPinIds', 'GetAllPinIds'))
    log.append(f'  CONNECTOR ID {connector_id} PIN IDS ({len(pin_ids)}): {pin_ids[:35]!r}')
    pin_attr_hits = {}
    for pin_id in pin_ids[:12]:
        try:
            pin = make_object(job, 'Pin', pin_id)
            values = read_attribute_names(pin)
            pin_names = read_string_getters(pin, (
                'GetInternalDeviceDesignation', 'GetConnectorName',
                'GetBlockConnectorName', 'GetDeviceDesignation'))
            if values or pin_names:
                pin_attr_hits[pin_id] = {'attr': values, 'getters': pin_names}
            for k, v in {**values, **pin_names}.items():
                names[f'pin:{pin_id}:{k}'] = v
        except Exception as exc:
            log.append(f'    PIN ID {pin_id}: name probe failed: {type(exc).__name__}: {exc}')
    log.append(f'  CONNECTOR ID {connector_id} PIN-NAME EVIDENCE: {pin_attr_hits!r}')
    return names


class E3Project:
    def __init__(self):
        self.backend = ''
        try:
            import e3series
            application = e3series.Application()
            job = application.CreateJobObject()
            if job.GetId():
                self.application = application
                self.job = job
                self.backend = 'official e3series wrapper'
                return
        except Exception:
            pass
        import win32com.client as win32
        self.application = win32.Dispatch('CT.Application')
        self.job = self.application.CreateJobObject()
        self.backend = 'pywin32 COM'
        if not self.job.GetId():
            raise RuntimeError('E3.series is not running with an open project.')

    def sheet_name(self, sheet_id):
        try:
            sheet = make_object(self.job, 'Sheet', sheet_id)
            return as_text(get_scalar(sheet, ('GetName', 'GetSheetName')))
        except Exception:
            return ''

    def on_j2(self, device):
        """Optional E3 drawing-sheet scope: applies only if a page J2 exists."""
        sids = related_ids(device, ('GetSymbolIds', 'GetAllSymbolIds'))
        for sid in sids:
            try:
                sym = make_object(self.job, 'Symbol', sid)
                sheet_id = get_scalar(sym, ('GetSheetId',))
                if sheet_id and self.sheet_name(sheet_id).casefold() == SHEET_NAME.casefold():
                    return True
            except Exception:
                continue
        return False

    def j2_page_exists(self):
        for sid in related_ids(self.job, ('GetSheetIds',)):
            if self.sheet_name(sid).casefold() == SHEET_NAME.casefold():
                return True
        return False

    def block_device(self, designation, log):
        """Locate a block without accidentally stopping at placed-only device IDs.

        In large projects, GetDeviceIds() can return only placed devices (119),
        while GetBlockIds() returns block objects and GetAllDeviceIds()
        returns thousands of additional E3 devices. Those calls MUST NOT
        be chained with related_ids() because that helper stops at the first
        nonempty method result. Search blocks first, then all devices.

        Match exact designations (and optionally fully-qualified .PNL10).
        Never guess which one is intended when several objects match.
        """
        wanted = device_key(designation)
        if not wanted:
            raise ValueError('Block designation in Excel C4 is empty.')
        visited = set()
        hits = []
        samples = []
        stages = (
            ('GetBlockIds', 'blocks'),
            ('GetAllDeviceIds', 'all E3 devices'),
            ('GetDeviceIds', 'placed E3 devices'),
        )
        for method, label in stages:
            ids = com_ids(self.job, method)  # independently enumerate every source
            new_ids = [did for did in ids if did not in visited]
            log.append(f'BLOCK LOOKUP: {method} returned {len(ids)} IDs; '
                       f'{len(new_ids)} not previously checked.')
            for idx, did in enumerate(new_ids, 1):
                visited.add(did)
                try:
                    obj = make_object(self.job, 'Device', did)
                    names = read_string_getters(obj, (
                        'GetName', 'GetDeviceDesignation', 'GetDeviceName',
                        'GetCompleteName', 'GetFullName', 'GetNameEx',
                    ))
                    # A COM object can expose its parent designation in one
                    # getter and its own complete designation in another.
                    if len(samples) < 30:
                        samples.append((did, names))
                    for meth, name in names.items():
                        key = device_key(name)
                        if key == wanted or key.endswith('.' + wanted):
                            hits.append((did, obj))
                            log.append(f'  BLOCK MATCH: ID={did}, via {meth}={name!r}')
                            break
                except Exception as exc:
                    if len(samples) < 30:
                        samples.append((did, f'Could not read: {type(exc).__name__}: {exc}'))
                if idx % 500 == 0:
                    log.append(f'  Looked up {idx}/{len(new_ids)} additional {label}...')
            # Blocks are the authoritative class for this import. If the
            # designated block was found, don't spend time constructing 2,000+
            # unrelated connector/pin objects just to repeat the lookup.
            if hits:
                log.append(f'BLOCK LOOKUP: found {len(hits)} matching object(s) '
                           f'using {method}.')
                break

        if not hits:
            log.append(f'BLOCK LOOKUP FAILED: checked {len(visited)} unique E3 IDs.')
            log.append(f'SAMPLE DEVICE/BLOCK NAMES: {samples!r}')
            raise LookupError(
                f'No E3 block/device matches Excel C4 = {designation!r} after '
                f'checking {len(visited)} unique objects from GetBlockIds(), '
                'GetAllDeviceIds(), and GetDeviceIds(). Ensure the block is '
                'in the active E3 job (not just a different opened project).'
            )

        # If a sheet named J2 is present, continue requiring a unique match
        # on that page; when J2 isn't a literal sheet name, retain the
        # original fallback to unique block designation matching.
        if self.j2_page_exists():
            scoped = [(did, o) for did, o in hits if self.on_j2(o)]
            if not scoped:
                for did, o in hits[:10]:
                    log.append(f'BLOCK found ID={did}, but on_j2=False; '
                               f'symbols={related_ids(o, ("GetSymbolIds",))[:25]}')
                raise LookupError(
                    f'{designation!r} was found, but not on E3 drawing sheet J2.'
                )
            hits = scoped
            log.append('E3 drawing sheet J2 found; restricting block match to that page.')
        else:
            log.append('No E3 drawing sheet named J2 found; matching by C4 designation.')
        if len(hits) != 1:
            raise LookupError(
                f'C4 designation {designation!r} matches {len(hits)} E3 objects '
                f'({[id for id, _ in hits[:15]]}); refusing to select one arbitrarily.'
            )
        log.append(f'BLOCK {designation!r} => E3 device ID {hits[0][0]}')
        return hits[0]

    def connectors(self, block_device, log=None):
        ids = related_ids(block_device, ('GetConnectorIds', 'GetBlockConnectorIds', 'GetSubDeviceIds'))
        if not ids:
            for sym_id in related_ids(block_device, ('GetSymbolIds',)):
                try:
                    sym = make_object(self.job, 'Symbol', sym_id)
                except Exception:
                    continue
                ids.extend(related_ids(sym, ('GetConnectorIds', 'GetBlockConnectorIds')))
        found = []
        self.connector_evidence = {}
        for cid in dict.fromkeys(ids):
            try:
                obj = make_object(self.job, 'Device', cid)
                # Do not use GetName() as the sole source of connector identity.
                names = extended_connector_aliases(self.job, cid, obj, log) if log is not None \
                    else {**read_string_getters(obj, CONNECTOR_NAME_GETTERS),
                          **read_attribute_names(obj)}
                self.connector_evidence[cid] = names
                display = names.get('GetDeviceDesignation') or names.get('GetConnectorName') or \
                    names.get('GetName') or '(unnamed)'
                code = as_text(get_scalar(obj, ('GetComponentName', 'GetComponentCode',
                                                'GetPartNumber', 'GetTypeName')))
                found.append((cid, obj, display, code))
            except Exception as ex:
                if log is not None:
                    log.append(f'  CONNECTOR ID {cid}: could not inspect: {ex}')
        return found

    def connector(self, choices, connector_name, connector_type):
        """Match a uniquely identified connector by any verified internal designation.

        Never guess by GetConnectorIds() order or component type alone.
        """
        requested = device_key(connector_name)
        matches = []
        for item in choices:
            cid = item[0]
            methods = self.connector_evidence.get(cid, {})
            for method_name, value in methods.items():
                if requested and requested in connector_keys(value):
                    matches.append((item, method_name, value))
                    break
        if len(matches) == 1:
            self.last_connector_match = matches[0][1:]
            return matches[0][0]
        if not matches:
            raise LookupError(
                f'Connector {connector_name!r} not found using connector device designation '
                'or .CONNECTOR_NAME pin attributes. Refer to connector evidence in log.')
        raise LookupError(
            f'Connector {connector_name!r} matched {len(matches)} different E3 connectors '
            f'({[m[0][0] for m in matches]}); refusing to guess.')

    def pin(self, connector_obj, pin_name):
        pin_ids = related_ids(connector_obj, ('GetPinIds', 'GetAllPinIds'))
        hits = []
        for pid in pin_ids:
            try:
                p = make_object(self.job, 'Pin', pid)
                value = get_scalar(p, ('GetName', 'GetPinName'))
                if as_text(value).casefold() == pin_name.casefold():
                    hits.append(pid)
            except Exception:
                continue
        if not hits:
            raise LookupError(f'Pin {pin_name!r} not found on connector.')
        if len(hits) > 1:
            raise LookupError(f'Pin {pin_name!r} is ambiguous on connector.')
        return hits[0]


def resolve(data, e3, log):
    _, block = e3.block_device(data.block_name, log)
    candidates = e3.connectors(block, log)
    log.append(f'Associated connector devices found: {len(candidates)}')
    log.append('BLOCK CONNECTOR DISCOVERY:')
    log.extend(diagnostic_ids(block, ('GetConnectorIds','GetBlockConnectorIds','GetSubDeviceIds','GetPinIds','GetSymbolIds')))
    log.append('ASSOCIATED CONNECTOR SAMPLES (first 40):')
    for cid, _, cname, ctype in candidates[:40]:
        log.append(f'  id={cid}, name={cname!r}, type={ctype!r}')
    if not candidates:
        log.append('WARNING: No connector devices were discovered inside the block; '
                   'this is likely why no assignments will occur.')
    indexed = {}
    for i in range(data.connector_count):
        examples = [e for e in data.entries if e.connector_index == i]
        if not examples:
            continue
        sample = examples[0]
        try:
            choice = e3.connector(candidates, sample.connector_name, sample.connector_type)
            indexed[i] = choice
            log.append(f'CONNECTOR {sample.connector_name!r} (type {sample.connector_type!r}) '
                       f'=> {choice[2]!r}, ID {choice[0]}, component {choice[3]!r}; '
                       f'MATCH EVIDENCE={e3.last_connector_match!r}')
        except Exception as exc:
            log.append('SKIP CONNECTOR: ' + str(exc))
            log.append(f'  SEARCH KEY: {sample.connector_name!r}; available connector names: '
                       f'{[c[2] for c in candidates[:40]]!r}')
    # E3 COM GetPinIds/GetName calls are expensive, especially on 128+ pin
    # connectors. Index each matched connector ONCE, not once per Excel row.
    pin_indexes = {}
    for ci, choice in indexed.items():
        pin_index = {}
        for pid in related_ids(choice[1], ('GetPinIds','GetAllPinIds')):
            try:
                p = make_object(e3.job, 'Pin', pid)
                name = as_text(get_scalar(p, ('GetName', 'GetPinName'))).casefold()
                if name:
                    pin_index.setdefault(name, []).append(pid)
            except Exception:
                continue
        pin_indexes[ci] = pin_index
        log.append(f'PIN INDEX: connector {choice[0]}: '
                   f'{sum(map(len, pin_index.values()))} pins indexed once')
    plans = []
    for entry in data.entries:
        choice = indexed.get(entry.connector_index)
        if not choice:
            log.append(f'SKIP {entry.cell}: no unique connector match for {entry.connector_name!r}')
            continue
        try:
            hits = pin_indexes[entry.connector_index].get(entry.pin_name.casefold(), [])
            if not hits:
                raise LookupError(f'Pin {entry.pin_name!r} not found on connector.')
            if len(hits) != 1:
                raise LookupError(f'Pin {entry.pin_name!r} is ambiguous on connector: {hits!r}')
            pin_id = hits[0]
            plans.append(ResolvedPin(entry, choice[0], pin_id))
            log.append(f'PLAN {entry.cell}: {entry.connector_name}/{entry.pin_name} '
                       f'-> {entry.signal_name!r} [pin ID {pin_id}]')
        except Exception as exc:
            log.append(f'SKIP {entry.cell}: {exc}')
            try:
                conn = choice[1]
                log.append('  CONNECTOR PIN-ID DIAGNOSTICS:')
                log.extend(diagnostic_ids(conn, ('GetPinIds', 'GetAllPinIds')))
                pin_ids = related_ids(conn, ('GetPinIds','GetAllPinIds'))
                log.append('  FIRST 35 PINS ON THIS CONNECTOR: ' + repr([
                    (pid, get_scalar(make_object(e3.job, 'Pin', pid), ('GetName','GetPinName')))
                    for pid in pin_ids[:35]]))
            except Exception as ex:
                log.append(f'  PIN INVENTORY DIAGNOSTIC FAILED: {ex}')
    return plans


# Pin-local attribute names are distinct from E3's GetSignalName / SetSignalName
# electrical signal property.  Only EXISTING pin attributes can be written.
PIN_LOCAL_ATTRIBUTE_NAMES = (
    'Local Signal Name', 'Local Signal', '.LOCAL_SIGNAL_NAME', 'LOCAL_SIGNAL_NAME',
)


def get_signal(pin):
    """Read ONLY: electrical network signal, for before/after safety checks."""
    value = get_scalar(pin, ('GetSignalName',))
    return value if isinstance(value, str) else None


def local_attribute_field(pin):
    """Identify a pin-owned local label using affirmative E3 HasAttribute evidence."""
    has = getattr(pin, 'HasAttribute', None)
    getter = getattr(pin, 'GetAttributeValue', None)
    setter = getattr(pin, 'SetAttributeValue', None)
    if not (callable(has) and callable(getter) and callable(setter)):
        return None, 'No full HasAttribute/GetAttributeValue/SetAttributeValue pin API'
    probes = []
    for attr in PIN_LOCAL_ATTRIBUTE_NAMES:
        try:
            ret = has(attr)
            probes.append((attr, repr(ret)))
            # A missing / unsupported result must not authorize writing.
            if ret is True or (type(ret) is int and ret > 0):
                before = getter(attr)
                if isinstance(before, str):
                    return attr, f'attribute exists: {attr!r}, previous={before!r}'
                return None, f'{attr!r} exists but GetAttributeValue returned {before!r}'
        except Exception as exc:
            probes.append((attr, type(exc).__name__))
    return None, 'No verified pin-local attribute; HasAttribute probes=' + repr(probes)


def read_peer_attribute_state(pin, field):
    """Read ONLY the specified field; do not scan all alias names per read."""
    has = getattr(pin, 'HasAttribute', None)
    getter = getattr(pin, 'GetAttributeValue', None)
    if not (callable(has) and callable(getter)):
        return ('unreadable', None)
    try:
        exists = has(field)
        return (exists is True or (type(exists) is int and exists > 0), getter(field))
    except Exception:
        return ('unreadable', None)


def get_local_attribute(pin, attr):
    value = pin.GetAttributeValue(attr)
    return value if isinstance(value, str) else None


def set_pin_local_attribute(pin, attr, target_name):
    """Write only one verified pin attribute. NEVER use SetSignalName."""
    before_network = get_signal(pin)
    before = get_local_attribute(pin, attr)
    if before == target_name:
        return 'already assigned', before_network
    result = pin.SetAttributeValue(attr, target_name)
    if result is False or (type(result) is int and result <= 0):
        raise RuntimeError(f'SetAttributeValue({attr!r}) returned {result!r}')
    after = get_local_attribute(pin, attr)
    if after != target_name:
        raise RuntimeError(f'Pin attribute readback failed: expected {target_name!r}, got {after!r}')
    if get_signal(pin) != before_network:
        # Undo ONLY the local attribute; refuse further writes.
        try:
            pin.SetAttributeValue(attr, before)
        except Exception:
            pass
        raise RuntimeError('Electrical GetSignalName changed! Pin-local attribute rolled back where possible.')
    return f'SetAttributeValue({attr}) VERIFIED', before_network


def read_pin_local_evidence(job, pin_id):
    """Diagnostics for user-controlled pin/connector property naming."""
    p = make_object(job, 'Pin', pin_id)
    out = [f'PIN {pin_id} name={get_scalar(p, ("GetName",))!r}, electrical signal={get_signal(p)!r}']
    out.append('  Attribute API: ' + repr(object_api_names(p, ('attribute','signal'))[:80]))
    out.extend(diagnostic_ids(p, ('GetAttributeIds',)))
    field, reason = local_attribute_field(p)
    out.append(f'  PIN LOCAL FIELD: {field!r}: {reason}')
    text_ids = related_ids(p, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
    out.append(f'  TYPE-7 TEXT IDS: {text_ids!r}')
    for tid in text_ids[:8]:
        try:
            txt = make_object(job, 'Text', tid)
            out.append(f'  TEXT {tid} METHODS: ' + repr(object_api_names(txt, ('text','type','name','owner','pin','attr','visible'))[:110]))
            vals = {}
            for meth in ('GetText','GetTextString','GetString','GetValue','GetType','GetTextType',
                         'GetAttributeName','GetPinId','GetOwnerPinId','GetSymbolId','GetSheetId',
                         'GetVisible','GetHeight','GetStyle'):
                v = get_scalar(txt, (meth,))
                if v is not None:
                    vals[meth] = v
            out.append(f'  TEXT {tid} VALUES: {vals!r}')
        except Exception as ex:
            out.append(f'  TEXT {tid} diagnostic failed: {ex}')
    return out


def style_text_bold(job, text_id, *, height=None):
    """Bold an EXISTING, independently identified E3 text object.

    The actual displayed Text ID matters: an E3 pin may expose the text's
    visibility while the corresponding glyphs live on a multi-pin symbol.
    Do not claim success unless the text object's style can be read back.
    """
    t = make_object(job, 'Text', text_id)
    original_style = get_scalar(t, ('GetStyle',))
    setter = getattr(t, 'SetStyle', None)
    if not callable(setter):
        return False, f'text {text_id}: SetStyle unavailable; style before={original_style!r}'
    try:
        setter_return = setter(BOLD_STYLE)
    except Exception as exc:
        return False, f'text {text_id}: SetStyle({BOLD_STYLE}) failed: {exc}'
    if setter_return is False or (type(setter_return) is int and setter_return < 0):
        return False, f'text {text_id}: SetStyle rejected ({setter_return!r})'
    if height is not None:
        size_setter = getattr(t, 'SetHeight', None)
        if callable(size_setter):
            try:
                size_setter(height)
            except Exception as exc:
                return False, f'text {text_id}: SetHeight({height}) failed: {exc}'
    visible_via = set_visible(t)
    actual_style = get_scalar(t, ('GetStyle',))
    actual_h = get_scalar(t, ('GetHeight',))
    bold_ok = actual_style == BOLD_STYLE
    height_ok = (height is None or
                 (isinstance(actual_h, (int, float)) and
                  abs(float(actual_h) - height) <= 0.0001))
    info = (f'text {text_id}: SetStyle({BOLD_STYLE}) returned {setter_return!r}, '
            f'style {original_style!r} -> {actual_style!r}, bold readback={bold_ok}')
    if height is not None:
        info += f', 0.15-inch={height_ok} (height={actual_h!r})'
    info += f', visibility={visible_via or "owner-level"}'
    return bold_ok and height_ok, info


def format_verified_local_attribute_text(job, pin_id, attr, connector_id=None):
    """Bold local signal display text attached to ONLY this connector pin.

    Prefer explicit attribute-owned text nodes. E3 can instead represent the
    visual signal field as type-7 text associated with the PIN. In that case,
    formatting the existing text object is safe; we NEVER call SetText or
    SetSignalName (which could change shared signal data).
    """
    pin = make_object(job, 'Pin', pin_id)
    all_ids = related_ids(pin, ('GetTextIds',))
    attribute_ids=[]
    for tid in all_ids:
        try:
            t = make_object(job, 'Text', tid)
            text_attr = get_scalar(t, ('GetAttributeName','GetAssignedAttributeName'))
            owner = get_scalar(t, ('GetOwnerPinId','GetPinId'))
            if as_text(text_attr).casefold() == attr.casefold() and (owner is None or owner==pin_id):
                attribute_ids.append(tid)
        except Exception:
            pass

    if attribute_ids:
        text_ids = attribute_ids
        source = f'pin-local attribute {attr!r}'
    else:
        # Use type-7 only when the ID is returned from THIS selected pin.
        text_ids = related_ids(pin, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
        source = 'pin-owned type-7 Local Signal Name text'
        # Do not touch a text object if COM lists that same ID under another
        # pin of this connector. This is only a FORMAT setter, still conservative.
        if text_ids and connector_id is not None:
            conn = make_object(job,'Device',connector_id)
            shared=set()
            for peer_id in related_ids(conn, ('GetPinIds',)):
                if peer_id == pin_id:
                    continue
                try:
                    peer=make_object(job,'Pin',peer_id)
                    shared.update(related_ids(peer,('GetTextIds',),text_type=LOCAL_SIGNAL_TEXT_TYPE))
                except Exception:
                    pass
            text_ids=[tid for tid in text_ids if tid not in shared]

    if not text_ids:
        return 'NO VERIFIED PER-PIN LOCAL SIGNAL TEXT; local attribute written, formatting skipped'
    # A pin-level visibility setter can expose a hidden label without writing
    # any signal value, and is constrained to the selected pin ID.
    if source.startswith('pin-owned type-7'):
        show = getattr(pin,'SetTextVisibility',None)
        if callable(show):
            try:
                show(LOCAL_SIGNAL_TEXT_TYPE,1)
            except Exception:
                pass
    results=[style_text_bold(job,tid,height=TEXT_HEIGHT_IN) for tid in dict.fromkeys(text_ids)]
    ok=all(success for success,_ in results)
    return ('OK' if ok else 'PARTIAL') + ' LOCAL SIGNAL TEXT ('+source+'): '+ '; '.join(msg for _,msg in results)


def read_xy_local(obj):
    """Read schema coordinates from a placed pin, symbol or text object."""
    for method in ('GetSchemaLocation','GetPosition','GetLocation','GetCoordinates','GetXY'):
        fn=getattr(obj,method,None)
        if not callable(fn):
            continue
        try:
            result=fn()
        except Exception:
            continue
        if (isinstance(result,(tuple,list)) and len(result)>=2 and
                all(type(a) in (int,float) and math.isfinite(a) for a in result[:2])):
            return float(result[0]),float(result[1])
    x=get_scalar(obj,('GetSchemaX','GetX'))
    y=get_scalar(obj,('GetSchemaY','GetY'))
    if (type(x) in (int,float) and type(y) in (int,float) and
            math.isfinite(x) and math.isfinite(y)):
        return float(x),float(y)
    return None


def placed_pin_y(job,pin_id):
    """Determine actual pin Y, never infer top pin from ID/pin number."""
    pin=make_object(job,'Pin',pin_id)
    xy=read_xy_local(pin)
    if xy is not None:
        return xy[1],'Pin schema position'
    # Per-pin symbols, when exposed, are more reliable than text nodes.
    symbol_ids=related_ids(pin,('GetSymbolIds',))
    sid=get_scalar(pin,('GetSymbolId','GetOwnerSymbolId'))
    if type(sid) is int and sid>0:
        symbol_ids.append(sid)
    ys=[]
    for sid in dict.fromkeys(symbol_ids):
        try:
            xy=read_xy_local(make_object(job,'Symbol',sid))
            if xy is not None:
                ys.append(xy[1])
        except Exception:
            pass
    if len(ys)==1 or (ys and max(ys)-min(ys)<1e-7):
        return float(median(ys)),'placed pin symbol'
    # The pin's type-7 text and other text nodes commonly sit at its Y.
    # Prefer type 7 so consistent offsets between text types cannot bias order.
    for group, description in ((related_ids(pin,('GetTextIds',),text_type=LOCAL_SIGNAL_TEXT_TYPE),'type-7 pin text'),
                               (related_ids(pin,('GetTextIds',)),'pin-owned text')):
        ys=[]
        for tid in group:
            try:
                xy=read_xy_local(make_object(job,'Text',tid))
                if xy is not None:
                    ys.append(xy[1])
            except Exception:
                pass
        if ys:
            return float(median(ys)),description
    return None,'no exposed placed pin/text Y coordinate'


def designation_text_ids(job, connector_id, top_pin_id, connector_name,
                         sibling_connector_ids=()):
    """Locate the real, PLACED type-12 text, including multi-pin symbols.

    Device designations on connector groups frequently belong to the whole
    connector symbol rather than Pin.GetTextIds(12). Only accept symbols
    uniquely owned by the requested connector; never style a sibling's text.
    """
    conn = make_object(job, 'Device', connector_id)
    pin = make_object(job, 'Pin', top_pin_id)
    conn_pin_ids = set(related_ids(conn, ('GetPinIds',)))
    notes=[]

    # Highest confidence: a type-12 node directly returned by the top pin.
    direct = related_ids(pin, ('GetTextIds',), text_type=DEVICE_DESIGNATION_TEXT_TYPE)
    if direct:
        notes.append(f'top pin directly exposes type-12 text IDs {direct}; also checking placed symbols')
    else:
        notes.append('top pin GetTextIds(type=12) returned no text')

    # E3 can place all 9 pins of a connector in one symbol. Version 1.8 only
    # accepted a symbol owning EXACTLY one pin, which misses these labels.
    from_pin = related_ids(pin, ('GetSymbolIds',))
    pin_sym_id = get_scalar(pin, ('GetSymbolId', 'GetOwnerSymbolId'))
    if type(pin_sym_id) is int and pin_sym_id > 0:
        from_pin.append(pin_sym_id)
    from_conn = related_ids(conn, ('GetSymbolIds',))
    notes.append(f'pin symbol IDs={from_pin}; connector symbol IDs={from_conn}')
    candidates = list(dict.fromkeys(from_pin + from_conn))

    # A shared symbol returned by both connectors is not sufficient proof of
    # ownership; skip it unless its associated pin IDs belong only to J1B.
    shared_symbols=set()
    for sibling_id in sibling_connector_ids:
        if sibling_id == connector_id:
            continue
        try:
            other=make_object(job, 'Device', sibling_id)
            shared_symbols.update(related_ids(other, ('GetSymbolIds',)))
        except Exception:
            pass

    collected=[]
    for sid in candidates:
        try:
            sym = make_object(job, 'Symbol', sid)
            sym_pin_ids = set(related_ids(sym, ('GetPinIds',)))
            own_pin_id = get_scalar(sym, ('GetPinId','GetOwnerPinId'))
            if not sym_pin_ids and type(own_pin_id) is int and own_pin_id > 0:
                sym_pin_ids.add(own_pin_id)
            if sym_pin_ids:
                # A connector's text may be on a 9-pin symbol, and that is OK.
                if top_pin_id not in sym_pin_ids or not sym_pin_ids <= conn_pin_ids:
                    notes.append(f'symbol {sid}: rejected (pins {sorted(sym_pin_ids)[:15]} not exclusive to selected connector)')
                    continue
            elif sid in shared_symbols:
                notes.append(f'symbol {sid}: rejected (also returned by another connector; unknown pin ownership)')
                continue
            elif sid not in from_conn and sid not in from_pin:
                continue
            tids = related_ids(sym, ('GetTextIds',), text_type=DEVICE_DESIGNATION_TEXT_TYPE)
            notes.append(f'symbol {sid} ({len(sym_pin_ids)} associated pins): type-12 text IDs={tids}')
            for tid in tids:
                collected.append((tid,sid))
        except Exception as exc:
            notes.append(f'symbol {sid} lookup failed: {type(exc).__name__}: {exc}')

    unique=list(dict.fromkeys(tid for tid,_ in collected))
    if len(unique) == 1:
        return list(dict.fromkeys(direct + unique)), '; '.join(notes)
    if not unique:
        return list(dict.fromkeys(direct)), '; '.join(notes) + '; no connector-symbol type-12 text found'

    # Multiple designation nodes: first accept an exact displayed connector
    # name, if the text object's API exposes one and the match is unique.
    matching=[]
    for tid in unique:
        try:
            t=make_object(job,'Text',tid)
            value=get_scalar(t,('GetText','GetDisplayText','GetString','GetValue'))
            if isinstance(value,str) and value.strip().casefold()==connector_name.strip().casefold():
                matching.append(tid)
        except Exception:
            pass
    if len(matching)==1:
        notes.append(f'unique rendered designation {connector_name!r} identifies text {matching[0]}')
        return list(dict.fromkeys(direct + matching)), '; '.join(notes)

    # Next accept a unique highest-Y label on a symbol proven exclusive to
    # this connector. No extrapolation from pin number or memory addresses.
    positioned=[]
    for tid in unique:
        try:
            xy=read_xy_local(make_object(job,'Text',tid))
            if xy is not None:
                positioned.append((xy[1],tid))
        except Exception:
            pass
    if len(positioned)==len(unique):
        positioned.sort(reverse=True)
        if len(positioned)<2 or abs(positioned[0][0]-positioned[1][0])>1e-7:
            notes.append(f'chose unique highest-Y type-12 label, text ID {positioned[0][1]}')
            return list(dict.fromkeys(direct + [positioned[0][1]])), '; '.join(notes)
    notes.append('AMBIGUOUS multiple type-12 texts; no formatting attempted')
    return list(dict.fromkeys(direct)), '; '.join(notes)


def show_top_pin_designation(job, connector_id, connector_name, sibling_connector_ids=()):
    """Make the highest-Y connector pin's device designation visible AND bold.

    Crucial fix: retrieve the actual text from the *multi-pin symbol* when
    Pin.GetTextIds(type=12) is empty. Do not change other connector symbols.
    """
    conn=make_object(job,'Device',connector_id)
    pin_ids=related_ids(conn,('GetPinIds',))
    ranked=[]
    missing=[]
    for pid in pin_ids:
        try:
            y,how=placed_pin_y(job,pid)
            if y is None:
                missing.append(pid)
            else:
                ranked.append((y,pid,how))
        except Exception:
            missing.append(pid)
    if not pin_ids or missing or len(ranked)!=len(pin_ids):
        return False,(f'{connector_name}: cannot identify highest-Y pin safely; '
                      f'{len(ranked)}/{len(pin_ids)} locations available, missing={missing[:20]}')
    ranked.sort(key=lambda v:v[0],reverse=True)
    if len(ranked)>1 and abs(ranked[0][0]-ranked[1][0])<1e-7:
        return False,f'{connector_name}: multiple pins tie for highest Y; no designation visibility changed'
    y,top_id,location_method=ranked[0]
    pin=make_object(job,'Pin',top_id)

    # Find a safe, exact text object BEFORE changing visibility; no setter is
    # invoked if text ownership remains ambiguous.
    text_ids, discovery = designation_text_ids(job,connector_id,top_id,connector_name,
                                                sibling_connector_ids)
    if not text_ids:
        return False,(f'{connector_name}: TOP PIN {get_scalar(pin,("GetName",))!r} '
                      f'(ID {top_id}, Y={y:g}); {discovery}. '
                      f'No unrelated text was formatted.')

    vis_result='no pin visibility API'
    show=getattr(pin,'SetTextVisibility',None)
    if callable(show):
        try:
            res=show(DEVICE_DESIGNATION_TEXT_TYPE,1)
            vis_result=f'Pin.SetTextVisibility(12,1) returned {res!r}'
        except Exception as exc:
            vis_result=f'Pin.SetTextVisibility failed: {exc}'
    results=[style_text_bold(job,tid) for tid in text_ids]
    formatted=bool(results) and all(ok for ok,_ in results)
    detail='; '.join(msg for _,msg in results)
    return formatted,(f'{connector_name}: TOP PIN {get_scalar(pin,("GetName",))!r} '
                      f'(ID {top_id}, Y={y:g} via {location_method}); '
                      f'{discovery}; {vis_result}; {detail}')


def uniquely_owned_pin_symbols(job, pin_id, connector_id=None):
    """Find placed connector symbols which represent only the requested pin."""
    pin = make_object(job, 'Pin', pin_id)
    symbol_ids = related_ids(pin, ('GetSymbolIds',))
    sid = get_scalar(pin, ('GetSymbolId', 'GetOwnerSymbolId'))
    if type(sid) is int and sid > 0:
        symbol_ids.append(sid)
    if connector_id:
        # This path was needed in the previous Match Properties utility:
        # connector device -> placed pin symbol -> signal text type 7.
        conn = make_object(job, 'Device', connector_id)
        symbol_ids.extend(related_ids(conn, ('GetSymbolIds',)))
    found = []
    for symbol_id in dict.fromkeys(symbol_ids):
        try:
            sym = make_object(job, 'Symbol', symbol_id)
            owner_pins = related_ids(sym, ('GetPinIds',))
            single = get_scalar(sym, ('GetPinId', 'GetOwnerPinId'))
            if not owner_pins and type(single) is int and single > 0:
                owner_pins = [single]
            if owner_pins == [pin_id]:
                found.append(sym)
        except Exception:
            continue
    return found


def pin_label_text_ids(job, pin_id, connector_id=None):
    """Get type-7 text positively owned by this pin, never nearby unowned text."""
    pin = make_object(job, 'Pin', pin_id)
    ids = related_ids(pin, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
    if ids:
        return ids
    for sym in uniquely_owned_pin_symbols(job, pin_id, connector_id):
        ids.extend(related_ids(sym, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE))
    return list(dict.fromkeys(ids))


def make_pin_label_visible(job, pin_id, connector_id=None):
    """Try owner-level visibility, including placed one-pin connector symbols."""
    pin = make_object(job, 'Pin', pin_id)
    show = getattr(pin, 'SetTextVisibility', None)
    if callable(show):
        try:
            show(LOCAL_SIGNAL_TEXT_TYPE, 1)
        except Exception:
            pass
    for sym in uniquely_owned_pin_symbols(job, pin_id, connector_id):
        show = getattr(sym, 'SetTextVisibility', None)
        if callable(show):
            try:
                show(LOCAL_SIGNAL_TEXT_TYPE, 1)
            except Exception:
                pass


def set_visible(obj):
    """Visibility varies by E3 release; invoke only known text visibility names."""
    pairs = (('SetVisible', 'GetVisible'),
             ('SetVisibility', 'GetVisibility'),
             ('SetTextVisible', 'GetTextVisible'),
             ('SetDisplayed', 'GetDisplayed'))
    for setter, getter in pairs:
        method = getattr(obj, setter, None)
        if not callable(method):
            continue
        try:
            result = method(1)
            check = get_scalar(obj, (getter,))
            if result is not False and (check is None or check in (1, True)):
                return setter
        except Exception:
            continue
    return None


def format_local_signal(job, pin_id, connector_id=None):
    pin = make_object(job, 'Pin', pin_id)
    # Request visibility BEFORE collecting labels: hidden labels may not be
    # returned by all E3 builds until SetTextVisibility has been called.
    make_pin_label_visible(job, pin_id, connector_id)
    text_ids = pin_label_text_ids(job, pin_id, connector_id)
    if not text_ids:
        return 'NO LOCAL-SIGNAL TEXT OBJECT: signal set, text formatting/visibility not verified'
    good = 0
    issues = []
    for tid in text_ids:
        try:
            text = make_object(job, 'Text', tid)
            text.SetHeight(TEXT_HEIGHT_IN)
            text.SetStyle(BOLD_STYLE)
            height = get_scalar(text, ('GetHeight',))
            style = get_scalar(text, ('GetStyle',))
            formatted = (isinstance(height, (float, int)) and
                         abs(float(height) - TEXT_HEIGHT_IN) <= 0.0001 and style == BOLD_STYLE)
            visibility_api = set_visible(text)
            if not visibility_api:
                # Some releases put type visibility on the parent pin.
                show = getattr(pin, 'SetTextVisibility', None)
                if callable(show):
                    try:
                        show(LOCAL_SIGNAL_TEXT_TYPE, 1)
                        visibility_api = 'Pin.SetTextVisibility'
                    except Exception:
                        pass
            if formatted and visibility_api:
                good += 1
            else:
                issues.append(f'text {tid}: format={formatted}, '
                              f'visibility method={visibility_api or "NOT FOUND"}')
        except Exception as exc:
            issues.append(f'text {tid}: {exc}')
    if good == len(text_ids):
        return f'OK: {good} text label(s), bold 0.15 in, made visible'
    return f'PARTIAL ({good}/{len(text_ids)} labels): ' + '; '.join(issues)



# ==============================================================
# PIN ARRANGER'S PROVEN PLACED-VIEW FORMATTING PATH
# Logical Pin.GetTextIds(type=7) is NOT the displayed pin label.
# Pin.GetViewIds(None) -> Symbol.GetTextIds(None) -> Text.GetType()
# 617 = displayed Local Signal Name; 12 = Device Designation;
# 213 = split-view position (e.g. "1/9", identifying the TOP pin).
# ============================================================== 


def placed_pin_view_ids(job, pin_id):
    """Actual placed symbol/view IDs for one logical pin, not its net."""
    pin = make_object(job, 'Pin', pin_id)
    ids = com_ids(pin, 'GetViewIds')
    return list(dict.fromkeys(ids))


def view_text_items(job, view_id):
    """Return (type, text value, text ID) of a placed pin-view symbol."""
    sym = make_object(job, 'Symbol', view_id)
    out = []
    for tid in related_ids(sym, ('GetTextIds',)):
        try:
            t = make_object(job, 'Text', tid)
            typ = get_scalar(t, ('GetType',))
            value = get_scalar(t, ('GetText',))
            out.append((typ, as_text(value), tid))
        except Exception as ex:
            print(f'    WARNING: view {view_id} text {tid} unreadable: {ex}', flush=True)
    return out


def pin_split_position(items):
    """Pin Arranger: visible 213 text '1/N' means first/top split pin."""
    for typ, value, _ in items:
        if typ == 213 and '/' in value and not value.startswith('*'):
            part = value.split('/', 1)[0].strip()
            if part.isdigit() and int(part) > 0:
                return int(part)
    return None


def apply_pin_arranger_text_style(job, tid, *, local_signal):
    """Apply style ON THE ACTUAL DISPLAYED text, not a pin attribute/net.

    This intentionally follows the working Pin Arranger's direct text-object
    SetVisibility / SetStyle / SetHeight calls. Readback is printed.
    """
    t = make_object(job, 'Text', tid)
    before_style = get_scalar(t, ('GetStyle',))
    before_height = get_scalar(t, ('GetHeight',))
    operations=[]
    for method, value in (('SetVisibility', 1), ('SetColour', 0),
                          ('SetStyle', 1)):
        fn=getattr(t,method,None)
        if not callable(fn):
            operations.append(method + ': unavailable')
            continue
        try:
            operations.append(f'{method}({value})={fn(value)!r}')
        except Exception as exc:
            operations.append(f'{method}({value}) ERROR: {exc}')
    if local_signal:
        # 0.15 inches is the requested local-signal text size.
        fn=getattr(t,'SetHeight',None)
        if callable(fn):
            try:
                operations.append(f'SetHeight({TEXT_HEIGHT_IN})={fn(TEXT_HEIGHT_IN)!r}')
            except Exception as exc:
                operations.append(f'SetHeight ERROR: {exc}')
    after_style=get_scalar(t,('GetStyle',))
    after_height=get_scalar(t,('GetHeight',))
    visibility=get_scalar(t,('GetVisibility','GetVisible'))
    ok_style = after_style == 1
    ok_height = (not local_signal or (type(after_height) in (int,float)
                  and abs(float(after_height)-TEXT_HEIGHT_IN)<0.0001))
    ok_visible = (visibility is None or visibility in (1,True))
    label='LOCAL SIGNAL' if local_signal else 'DEVICE DESIGNATION'
    info=(f'{label} TEXT {tid}: style {before_style!r}->{after_style!r}, '
          f'height {before_height!r}->{after_height!r}, visible={visibility!r}; '
          + ', '.join(operations))
    return ok_style and ok_height and ok_visible, info


def format_placed_local_signal(job, pin_id, connector_name, pin_name):
    """Only touch type-617 labels on THIS pin's actual placed views."""
    view_ids=placed_pin_view_ids(job,pin_id)
    if not view_ids:
        return False,(f'{connector_name}/{pin_name}: no Pin.GetViewIds(None) found. '
                      'No text formatting performed.')
    messages=[]
    found=[]
    for vid in view_ids:
        items=view_text_items(job,vid)
        pos=pin_split_position(items)
        sigs=[tid for typ,_,tid in items if typ == 617]
        messages.append(f'view={vid} split={pos} types={[typ for typ,_,_ in items]} type617={sigs}')
        for tid in sigs:
            if tid in found:
                continue
            found.append(tid)
            ok, info=apply_pin_arranger_text_style(job,tid,local_signal=True)
            messages.append(('PASS ' if ok else 'CHECK ')+info)
    if not found:
        return False,(f'{connector_name}/{pin_name}: NO TYPE-617 PLACED LABEL; '
                      + ' | '.join(messages))
    verified=sum('PASS LOCAL SIGNAL TEXT' in m for m in messages)
    return verified == len(found), f'{connector_name}/{pin_name}: '+ ' | '.join(messages)


def _local_signal_attribute_on_pin(job, pin):
    """Return (attribute object, attribute ID), strictly owned by this pin.

    Pin.GetAttributeValue alone is not enough to show the corresponding text.
    The earlier working Block Maker used Pin.GetAttributeIds() followed by
    Attribute.DisplayAttribute() for Local Signal Name.  Do not create a new
    attribute or touch the electrical signal/network here.
    """
    for aid in related_ids(pin, ('GetAttributeIds',)):
        try:
            attr = make_object(job, 'Attribute', aid)
            name = as_text(get_scalar(attr, ('GetName',))).casefold()
            if name == 'local signal name':
                return attr, aid
        except Exception:
            continue
    return None, None


def _pin_local_display_texts(job, pin_id):
    """Collect actual placed type-617 texts for a single pin, no net objects."""
    texts = set()
    views = set()
    for vid in placed_pin_view_ids(job, pin_id):
        views.add(vid)
        for typ, _, tid in view_text_items(job, vid):
            if typ == LOCAL_SIGNAL_TEXT_TYPE:
                texts.add(tid)
    return views, texts


def _visibility_readback(job, text_id):
    text = make_object(job, 'Text', text_id)
    return get_scalar(text, ('GetVisibility', 'GetVisible'))


def ensure_connector_local_signals_visible(job, connector_id, connector_name):
    """Display *all* existing local signal names on one selected connector.

    Version 2.5 only SetVisibility(1) on type-617 text objects it could find.
    But some hidden Local Signal Name pin attributes have NO placed text object
    until Attribute.DisplayAttribute() has been called.  This function mirrors
    the previously working Block Maker, then re-enumerates pin views and the
    attribute's own text IDs and makes those text objects visible and bold.

    Only existing nonblank pin-local attributes are displayed; this code NEVER
    calls Pin.SetSignalName or changes a signal value, connector, or other pin.
    """
    connector = make_object(job, 'Device', connector_id)
    pin_ids = list(dict.fromkeys(related_ids(connector, ('GetPinIds',))))
    seen_texts = set()
    seen_views = set()
    visible_texts = set()
    expected_named_pins = 0
    displayed_named_pins = 0
    display_attribute_calls = 0
    failed = []
    uncertain = []
    samples = []

    for pin_id in pin_ids:
        pin = make_object(job, 'Pin', pin_id)
        pin_name = as_text(get_scalar(pin, ('GetName',))) or str(pin_id)
        # Read the pin-owned *attribute*, never the shared electrical net.
        value = None
        try:
            value = pin.GetAttributeValue('Local Signal Name')
        except Exception:
            pass
        existing_value = as_text(value) if isinstance(value, (str, int, float)) and value != NO_VALUE else ''
        views_before, ids_before = _pin_local_display_texts(job, pin_id)
        seen_views.update(views_before)

        # Make any existing 617 text visible even on pins with no attribute
        # value.  A blank value is not considered a missing rendered label.
        attr, aid = (None, None)
        if existing_value:
            expected_named_pins += 1
            attr, aid = _local_signal_attribute_on_pin(job, pin)
            if attr is None:
                if not ids_before:
                    failed.append(f'{connector_name}/{pin_name}: Local Signal Name value '
                                  f'{existing_value!r} exists, but no attribute ID or '
                                  'placed type-617 text was found')
                # Existing placed type-617 text can still be made visible.
            else:
                # Calling DisplayAttribute is what the original Block Maker
                # does.  Its absence in v2.5 is why text objects could remain
                # uncreated even after SetVisibility(1) was requested.
                need_display = not ids_before or any(
                    _visibility_readback(job, tid) not in (1, True)
                    for tid in ids_before
                )
                if need_display:
                    display = getattr(attr, 'DisplayAttribute', None)
                    if not callable(display):
                        failed.append(f'{connector_name}/{pin_name}: '
                                      'Attribute.DisplayAttribute not available')
                    else:
                        try:
                            result = display()
                            display_attribute_calls += 1
                            samples.append(f'{pin_name}: Attribute {aid} '
                                           f'DisplayAttribute() returned {result!r}')
                        except Exception as exc:
                            failed.append(f'{connector_name}/{pin_name}: '
                                          f'Attribute.DisplayAttribute error: {exc}')

        # Attribute.DisplayAttribute can create new text IDs, so we MUST
        # re-enumerate after it has executed.  Also inspect attribute-owned
        # text, whose type can differ from 617 on some E3 symbol libraries.
        views_after, type617_ids = _pin_local_display_texts(job, pin_id)
        seen_views.update(views_after)
        text_ids = set(type617_ids)
        if existing_value and attr is not None:
            text_ids.update(related_ids(attr, ('GetTextIds',)))
        if not text_ids and existing_value:
            failed.append(f'{connector_name}/{pin_name}: Local Signal Name '
                          f'{existing_value!r} still has NO DISPLAY TEXT after '
                          'DisplayAttribute; check placed pin symbol/attribute '
                          'visibility settings in E3')
            continue

        pin_visible = False
        for tid in sorted(text_ids):
            if tid not in seen_texts:
                seen_texts.add(tid)
                try:
                    txt = make_object(job, 'Text', tid)
                    set_vis = getattr(txt, 'SetVisibility', None)
                    if not callable(set_vis):
                        failed.append(f'{connector_name}/{pin_name}: '
                                      f'text {tid} has no SetVisibility')
                        continue
                    ret = set_vis(1)
                    # Preserve 0.15-inch bold formatting on all displayed
                    # local signal names, not just newly imported values.
                    if existing_value:
                        style = getattr(txt, 'SetStyle', None)
                        if callable(style):
                            style(BOLD_STYLE)
                        height = getattr(txt, 'SetHeight', None)
                        if callable(height):
                            height(TEXT_HEIGHT_IN)
                    state = _visibility_readback(job, tid)
                    if state in (1, True):
                        visible_texts.add(tid)
                    elif state is None:
                        uncertain.append(f'{connector_name}/{pin_name}: text {tid} '
                                         f'visibility cannot be read (setter={ret!r})')
                    else:
                        failed.append(f'{connector_name}/{pin_name}: text {tid} '
                                      f'visibility is {state!r} after SetVisibility(1)')
                except Exception as exc:
                    failed.append(f'{connector_name}/{pin_name}: text {tid}: '
                                  f'{type(exc).__name__}: {exc}')
            if tid in visible_texts:
                pin_visible = True
        if existing_value and pin_visible:
            displayed_named_pins += 1
        elif existing_value and text_ids and not pin_visible:
            # If a display text was found but GetVisibility is unsupported,
            # fail visibly rather than claiming success.
            uncertain.append(f'{connector_name}/{pin_name}: could not verify '
                             f'display of {existing_value!r}')

    stats = {
        'pins': len(pin_ids), 'views': len(seen_views),
        'labels': len(seen_texts), 'verified': len(visible_texts),
        'failed': len(failed), 'unverified': len(uncertain),
        'named_pins': expected_named_pins,
        'displayed_pins': displayed_named_pins,
        'display_calls': display_attribute_calls,
    }
    detail = (f'{connector_name}: {len(pin_ids)} pins, '
              f'{expected_named_pins} nonblank Local Signal Name attributes, '
              f'{display_attribute_calls} Attribute.DisplayAttribute calls; '
              f'{displayed_named_pins}/{expected_named_pins} named pins shown; '
              f'{len(visible_texts)}/{len(seen_texts)} text IDs visible')
    if samples:
        detail += '; display samples: ' + '; '.join(samples[:8])
    if failed:
        detail += '; ERRORS: ' + '; '.join(failed[:15])
    if uncertain:
        detail += '; UNVERIFIED: ' + '; '.join(uncertain[:10])
    return stats, detail


def show_top_designation_placed_view(job, connector_id, connector_name):
    """Use the Pin Arranger's type-213 '1/N' marker to find the TOP view.

    The logical pins returned the same Y in v1.9, which is why the old code
    always skipped. On split views the '1/N' marker gives definitive position.
    Modify only type-12 text ON THE MATCHED CONNECTOR'S top placed view.
    """
    conn=make_object(job,'Device',connector_id)
    pin_ids=related_ids(conn,('GetPinIds',))
    top=[]
    notes=[]
    for pid in pin_ids:
        pin=make_object(job,'Pin',pid)
        pin_name=as_text(get_scalar(pin,('GetName',)))
        for vid in placed_pin_view_ids(job,pid):
            items=view_text_items(job,vid)
            pos=pin_split_position(items)
            if pos == 1:
                top.append((pid,pin_name,vid,items))
            elif pos is not None:
                notes.append(f'{pin_name}: split {pos} on view {vid}')
    # Do not guess: top on this connector must be a unique placed split.
    unique_views={(pid,vid):(pid,name,vid,items) for pid,name,vid,items in top}
    if len(unique_views)!=1:
        return False,(f'{connector_name}: expected exactly 1 placed top split (type 213 = 1/N), '
                      f'found {[(x[1],x[2]) for x in unique_views.values()]}; '
                      'no designation was modified. '+ '; '.join(notes[:12]))
    pid,pname,vid,items=list(unique_views.values())[0]
    designation_ids=[tid for typ,_,tid in items if typ==12]
    if not designation_ids:
        return False,(f'{connector_name}: top pin {pname}, view {vid} has split 1/N, '
                      'but NO type-12 designation text. types='+repr([typ for typ,_,_ in items]))
    changed=[]
    for tid in designation_ids:
        ok, info=apply_pin_arranger_text_style(job,tid,local_signal=False)
        changed.append((ok,info))
    return all(ok for ok,_ in changed),(f'{connector_name}: TOP pin={pname}, pin ID={pid}, '
           f'view={vid}, split=1/N; '+' | '.join(info for _,info in changed))


def perform(path):
    """Apply immediately; print progress/timing to stdout, never write files."""
    started_at = time.perf_counter()
    lines = ConsoleLines([f'ZUKEN LOCAL SIGNAL IMPORTER v{VERSION}',
             f'TIME: {datetime.now().isoformat(timespec="seconds")}',
             f'EXCEL FILE: {path}', ''])

               'verified':0, 'formatted':0, 'designation_formatted':0, 'expected_designations':0, 'failed':0, 'skipped':0,
               'visibility_labels':0, 'visibility_verified':0, 'visibility_failed':0,
               'visibility_unverified':0, 'named_pins_expected':0, 'named_pins_displayed':0, 'fatal':None}
    com_initialized = False
    try:
        data = parse_workbook(path)
        lines.append(f'TIMING: Excel parsed in {time.perf_counter()-started_at:.1f}s')
        lines += [f'EXCEL WORKSHEET SELECTED: {data.worksheet_name}',
                  f'E3 DRAWING PAGE FOR IDENTIFICATION: {SHEET_NAME}',
                  f'DEVICE DESIGNATION (C4): {data.block_name!r}',
                  f'NUMBER OF CONNECTORS (C5): {data.connector_count}',
                  f'DUPLICATE-PIN STATUS (B2): {data.check_b2!r}',
                  f'SIGNAL ASSIGNMENTS READ: {len(data.entries)}']
        summary['defined'] = len(data.entries)
        for ci in range(data.connector_count):
            grouped = [e for e in data.entries if e.connector_index == ci]
            pc = FIRST_PIN_COL + CONNECTOR_COL_STEP * ci
            from openpyxl.utils import get_column_letter
            cols = f'{get_column_letter(pc)}:{get_column_letter(pc+1)}'
            lines.append(f'EXCEL CONNECTOR #{ci+1} {cols}: {len(grouped)} nonblank assignments; '
                         f'name (B{FIRST_ROW+ci})={grouped[0].connector_name if grouped else "(no nonblank signals)"!r}')
        for e in data.entries[:100]:
            lines.append(f'  INPUT {e.cell}: connector={e.connector_name!r}, type={e.connector_type!r}, '
                         f'pin={e.pin_name!r} -> signal={e.signal_name!r}')
        if len(data.entries) > 100:
            lines.append(f'  ... {len(data.entries)-100} additional assignments omitted from this preview')
        for prob in data.duplicate_problems:
            lines.append('EXCEL NOTICE: '+prob)
        require_no_duplicate_pins(data)
        if not data.entries:
            raise RuntimeError('No pin/signal pairs were read from the workbook. '
                               'Check starting row 7 and pairs I:J, M:N, Q:R. '
                               'Unspecified/blank signal cells are intentionally skipped.')

        import pythoncom
        pythoncom.CoInitialize()
        com_initialized = True
        e3 = E3Project()
        lines.append(f'E3 CONNECTION: {e3.backend}; job ID={e3.job.GetId()!r}')
        lines.append('E3 JOB AVAILABLE RELEVANT METHODS: '
                     + repr(object_api_names(e3.job, ('device','sheet','block','symbol','pin'))[:120]))
        lines.append('E3 JOB SHEET ENUMERATION:')
        lines.extend(diagnostic_ids(e3.job, ('GetSheetIds',)))
        sheet_ids = related_ids(e3.job, ('GetSheetIds',))
        lines.append('E3 SHEET NAMES (first 40): '
                     +repr([(sid,e3.sheet_name(sid)) for sid in sheet_ids[:40]]))
        lines.append('E3 JOB DEVICE ENUMERATION:')
        lines.extend(diagnostic_ids(e3.job, ('GetDeviceIds','GetAllDeviceIds','GetBlockIds')))

        plans = resolve(data, e3, lines)
        lines.append(f'TIMING: E3 block/connector/pin lookup done at +{time.perf_counter()-started_at:.1f}s')
        summary['resolved'] = len(plans)
        summary['expected_designations'] = len({p.connector_id for p in plans})
        summary['skipped'] = summary['defined'] - summary['resolved']
        lines.append(f'RESOLVED {len(plans)} / {len(data.entries)} PIN ASSIGNMENTS')
        if not plans:
            lines.append('NO PINS RESOLVED, so no E3 writes attempted. Check the SKIP and method reports above. '
                         'If connector names still all show PNL1, copy this output; it includes the '
                         'per-connector API list and .CONNECTOR_NAME attribute inspection.')
        else:
            lines.append('E3 COM PIN API SAMPLE FOR FIRST RESOLVED PIN:')
            lines.extend(pin_diagnostics(e3.job, plans[0].pin_id))

        # Preflight entire input before any write; never partly import a workbook
        # if E3 exposes only electrical net signals rather than pin-local attrs.
        lines.append('PIN-LOCAL ATTRIBUTE PREFLIGHT (NO WRITES YET):')
        missing=[]
        valid=[]
        for plan in plans:
            pin=make_object(e3.job, 'Pin', plan.pin_id)
            field, reason=local_attribute_field(pin)
            lines.append(f'  {plan.entry.connector_name}/{plan.entry.pin_name}, ID {plan.pin_id}: {reason}')
            if field:
                valid.append((plan,field))
            else:
                missing.append(plan)
                lines.extend(read_pin_local_evidence(e3.job,plan.pin_id))
        if missing:
            summary['skipped'] += len(missing)
            lines.append('SAFETY STOP: one or more requested pins have no VERIFIED PIN-LOCAL field.')
            lines.append('NO WRITES attempted to prevent shared-signal changes on J1A/J1B.')
            lines.append('Do not substitute SetSignalName or Text.SetText: E3 may propagate those to the whole net.')
            lines.append('Please copy this console output and share the exact pin property screenshot if available.')
        else:
            # Also report same-numbered pins on other connectors for context.
            _, block=e3.block_device(data.block_name, lines)
            others=e3.connectors(block)
            lines.append('COUNTERPART PIN READBACK BEFORE WRITES:')
            other_sigs={}
            other_attrs={}
            peers_by_name={}
            first_peer_per_connector={}
            monitored_fields={field for _,field in valid}
            selected={v.pin_id for v,_ in valid}
            for cid,conn,cn,ct in others:
                for pid in related_ids(conn, ('GetPinIds',)):
                    if pid in selected: continue
                    try:
                        op=make_object(e3.job,'Pin',pid)
                        key=(cid,pid)
                        other_sigs[key]=get_signal(op)
                        peer_name=as_text(get_scalar(op, ('GetName','GetPinName'))).casefold()
                        for f in monitored_fields:
                            pk=(cid,pid,f)
                            other_attrs[pk]=read_peer_attribute_state(op,f)
                            if peer_name:
                                peers_by_name.setdefault(peer_name,set()).add(pk)
                            first_peer_per_connector.setdefault(cid,pk)
                    except Exception: pass
            lines.append(f'  Non-target pin electrical signals snapshotted: {len(other_sigs)}')
            lines.append(f'  Non-target pin local attributes snapshotted: {len(other_attrs)}')
            lines.append('  SPEEDUP: first write checks all peers; later writes check '
                         'same-numbered pins + connector sentinels; all peers rechecked at end.')
            lines.append(f'TIMING: preflight and safety snapshots done at +{time.perf_counter()-started_at:.1f}s')
            lines.append('WRITE RESULTS (PIN ATTRIBUTES ONLY, NEVER NETWORK):')
            for write_idx,(plan,field) in enumerate(valid):
                e=plan.entry
                lines.append(f'ATTEMPT {e.cell}: connector={e.connector_name!r}, pin={e.pin_name!r}, ID {plan.pin_id}, attribute={field!r}, value={e.signal_name!r}')
                try:
                    pin=make_object(e3.job,'Pin',plan.pin_id)
                    original_local_value=get_local_attribute(pin,field)
                    result,net_before=set_pin_local_attribute(pin,field,e.signal_name)
                    # Detect cross-connector propagation as soon as practicable.
                    # First write: inspect ALL other pins to establish safety.
                    # Subsequent writes: same pin number on other connectors,
                    # plus one sentinel per connector. At the end ALL are checked.
                    watch = (other_attrs.keys() if write_idx == 0 else
                             (peers_by_name.get(e.pin_name.casefold(),set()) |
                              set(first_peer_per_connector.values())))
                    changed_now=[]
                    for peer_cid,peer_pid,peer_field in watch:
                        before_state=other_attrs[(peer_cid,peer_pid,peer_field)]
                        try:
                            peer=make_object(e3.job,'Pin',peer_pid)
                            now=read_peer_attribute_state(peer,peer_field)
                            if now != before_state:
                                changed_now.append((peer_cid,peer_pid,peer_field,before_state,now))
                        except Exception as exc:
                            changed_now.append((peer_cid,peer_pid,peer_field,'read failed',str(exc)))
                    if changed_now:
                        # Best-effort rollback of this assignment. Because E3 may
                        # share attribute objects, project backup remains essential.
                        try:
                            pin.SetAttributeValue(field,original_local_value)
                        except Exception:
                            pass
                        lines.append('  CRITICAL: other connector pin attributes changed: '+repr(changed_now[:8]))
                        raise RuntimeError('PIN-LOCAL WRITE PROPAGATED TO NON-TARGET PINS; '
                                           'no more assignments will be attempted. Restore your test copy.')
                    summary['signal_accepted']+=1
                    summary['verified']+=1
                    lines.append(f'  ATTRIBUTE SETTER: {result}; net unchanged: {get_signal(pin)==net_before}; peers unchanged.')
                    lines.append('  PLACED PIN-VIEW LABEL: scheduled formatting after all value writes')
                    if ((write_idx + 1) % 10 == 0 or write_idx + 1 == len(valid)):
                        lines.append(f'PROGRESS: {write_idx+1}/{len(valid)} pin attributes verified '
                                     f'at +{time.perf_counter()-started_at:.1f}s')
                except Exception as ex:
                    summary['failed']+=1
                    lines.append(f'  WRITE FAILED: {type(ex).__name__}: {ex}')
                    break  # avoid compounding a bad write
            changed_attrs=[]
            for (cid,pid,field),before in other_attrs.items():
                try:
                    op=make_object(e3.job,'Pin',pid)
                    now=read_peer_attribute_state(op,field)
                    if now != before:
                        changed_attrs.append((cid,pid,field,before,now))
                except Exception as exc:
                    changed_attrs.append((cid,pid,field,'read failed',str(exc)))
            lines.append(f'TIMING: attribute writes and full peer check done at +{time.perf_counter()-started_at:.1f}s')
            # Pin Arranger path: resolve *actual placed* pin view text type
            # 617 (Local Signal Name) after all local-attribute changes are done.
            # This is vital because E3 may regenerate label text on a write.
            if not summary['fatal'] and not summary['failed']:
                lines.append('PLACED PIN-VIEW LOCAL SIGNAL FORMATTING (TYPE 617):')
                for plan, field in valid:
                    e=plan.entry
                    try:
                        ok, detail=format_placed_local_signal(e3.job,plan.pin_id,
                                                                e.connector_name,e.pin_name)
                        lines.append('  '+('OK: ' if ok else 'NOTICE: ')+detail)
                        if ok: summary['formatted']+=1
                    except Exception as exc:
                        lines.append(f'  NOTICE: {e.connector_name}/{e.pin_name}: '
                                     f'view formatting failed: {type(exc).__name__}: {exc}')
                lines.append(f'TIMING: local-signal text formatting done at +{time.perf_counter()-started_at:.1f}s')
                # Use type-213 '1/N' marker from the placed pin-view symbols
                # to identify physical top split; logical Pin Y is unreliable.
                lines.append('TOP-PIN DEVICE DESIGNATION (TYPE 12 ON VIEW 1/N):')
                for cid, cname in dict.fromkeys((p.connector_id,p.entry.connector_name) for p,_ in valid):
                    try:
                        ok, detail=show_top_designation_placed_view(e3.job,cid,cname)
                        lines.append('  '+('OK: ' if ok else 'NOTICE: ')+detail)
                        if ok: summary['designation_formatted']+=1
                    except Exception as exc:
                        lines.append(f'  NOTICE: {cname}: type-12 formatting error: '
                                     f'{type(exc).__name__}: {exc}')
            lines.append(f'TIMING: top-pin designation formatting done at +{time.perf_counter()-started_at:.1f}s')
            # The normal type-617 bolding pass covers only newly assigned pins.
            # Ensure existing local signal labels are visible on ALL pins of
            # connectors actually selected by this workbook, not just edited pins.
            if not summary['fatal'] and not summary['failed']:
                lines.append('ALL LOCAL SIGNAL LABELS VISIBLE (TYPE 617):')
                for cid, cname in dict.fromkeys((p.connector_id, p.entry.connector_name)
                                              for p, _ in valid):
                    try:
                        stats, detail = ensure_connector_local_signals_visible(e3.job, cid, cname)
                        summary['named_pins_expected'] += stats['named_pins']
                        summary['named_pins_displayed'] += stats['displayed_pins']
                        summary['visibility_labels'] += stats['labels']
                        summary['visibility_verified'] += stats['verified']
                        summary['visibility_failed'] += stats['failed']
                        summary['visibility_unverified'] += stats['unverified']
                        lines.append('  ' + detail)
                    except Exception as exc:
                        summary['visibility_failed'] += 1
                        lines.append(f'  VISIBILITY ERROR {cname}: '
                                     f'{type(exc).__name__}: {exc}')
                lines.append(f'TIMING: all selected-connector local signal visibility done '
                             f'at +{time.perf_counter()-started_at:.1f}s')
            lines.append(f'COUNTERPART PIN ATTRIBUTES CHANGED: {changed_attrs[:20]!r}')
            if changed_attrs:
                summary['fatal']='A non-target PIN ATTRIBUTE changed. STOP and restore your project copy.'
            changed=[]
            for (cid,pid),before in other_sigs.items():
                try:
                    now=get_signal(make_object(e3.job,'Pin',pid))
                    if now!=before: changed.append((cid,pid,before,now))
                except Exception as exc:
                    changed.append((cid,pid,'read failed',str(exc)))
            lines.append(f'COUNTERPART ELECTRICAL SIGNALS CHANGED: {changed[:15]!r}')
            if changed:
                summary['fatal']='A non-target electrical signal changed. STOP: restore your test copy.'
        del e3
    except Exception as ex:
        summary['fatal'] = f'{type(ex).__name__}: {ex}'
        lines.append('FATAL ERROR: '+summary['fatal'])
        lines.append(traceback.format_exc())
    finally:
        if com_initialized:
            try:
                # This already follows normal Python reference lifetime. Avoid a
                # forced full garbage collection of thousands of E3 COM proxies.
                pythoncom.CoUninitialize()
            except Exception:
                pass
        lines += ['', '============= FINAL SUMMARY =============',
                  f'Excel defined assignments:       {summary["defined"]}',
                  f'Resolved pin IDs:                {summary["resolved"]}',
                  f'Pin-local attributes written:     {summary["signal_accepted"]}',
                  f'Pin-local value readback verified:{summary["verified"]}',
                  f'Pin-local text formatting verified:{summary["formatted"]}',
                  f'Top-pin designations bold/visible:{summary["designation_formatted"]}',
                  f'Named pins displayed (verified):      {summary["named_pins_displayed"]}/{summary["named_pins_expected"]}',
                  f'Local signal labels visible (verified): {summary["visibility_verified"]}/{summary["visibility_labels"]}',
                  f'Local label visibility errors:    {summary["visibility_failed"]}',
                  f'Local label visibility unverified:{summary["visibility_unverified"]}',
                  f'Skipped or not matched:         {summary["skipped"]}',
                  f'Write failures:                 {summary["failed"]}',
                  f'Fatal problem:                  {summary["fatal"] or "none"}',
                                    'IMPORTANT: GetSignalName/SetSignalName are ELECTRICAL NET signals; this version NEVER writes them.',
                  '=========================================',
                  f'E3 IMPORT FINISHED: all operations returned in {time.perf_counter()-started_at:.1f}s']
        summary['elapsed_seconds'] = time.perf_counter() - started_at
        # The error popup uses the same diagnostics as print(), without
        # creating a separate console or writing any log files.
        summary['output_lines'] = list(lines)
    return summary


def import_issues(summary):
    """Show an error window for actual import errors, not cosmetic text warnings.

    Placed type-617 labels may be perfectly visible/bold even when the API
    reports them as unverifiable. Keep that diagnostic for true error reports,
    rather than presenting a false failed-import popup on a successful run.
    """
    issues = []
    if summary.get('fatal'):
        issues.append('ERROR: ' + str(summary['fatal']))
    if summary.get('failed'):
        issues.append(f"{summary['failed']} write(s) failed.")
    if summary.get('named_pins_displayed', 0) < summary.get('named_pins_expected', 0):
        issues.append(f"Only {summary.get('named_pins_displayed', 0)}/{summary.get('named_pins_expected', 0)} pins with Local Signal Name values have verified visible text.")
    if summary.get('visibility_failed') or summary.get('visibility_unverified'):
        issues.append(f"Local signal visibility could not be verified for "
                      f"{summary.get('visibility_failed', 0) + summary.get('visibility_unverified', 0)} "
                      f"text object(s). See details.")
    if summary.get('skipped'):
        issues.append(f"{summary['skipped']} assignment(s) could not be resolved or applied.")
    defined = summary.get('defined', 0)
    done = summary.get('verified', 0)
    if defined and done < defined:
        issues.append(f"Only {done}/{defined} requested pin-local names were verified.")
    return issues


def show_complete_window(parent, result):
    """310x150 foreground completion dialog; Enter works without clicking first.

    Keep this as a Tk dialog, not input(): an E3 launcher may have no stdin.
    Reassert focus shortly after mapping because E3/Windows can take it back
    immediately after the Excel file-picker closes.
    """
    import tkinter as tk
    from tkinter import ttk

    window = tk.Toplevel(parent)
    window.title('Zuken - Script Complete')
    window.resizable(False, False)
    window.geometry('310x150')

    frame = ttk.Frame(window, padding=14)
    frame.pack(fill='both', expand=True)
    ttk.Label(frame, text='Script complete', font=('Segoe UI', 12, 'bold')).pack(pady=(0, 6))
    ttk.Label(frame, text=f"{result.get('verified', 0)} pin local signal(s) updated").pack()
    ttk.Label(frame, text='Press Enter to close', foreground='#555555').pack(pady=(6, 5))

    def finish(event=None):
        try:
            window.grab_release()
        except tk.TclError:
            pass
        if window.winfo_exists():
            window.destroy()

    window.protocol('WM_DELETE_WINDOW', finish)
    window.bind('<Return>', finish)
    window.bind('<KP_Enter>', finish)
    window.bind('<Escape>', finish)
    close_button = ttk.Button(frame, text='Close', command=finish)
    close_button.pack()

    def bring_to_front():
        if not window.winfo_exists():
            return
        try:
            window.deiconify()
            window.attributes('-topmost', True)
            window.lift()
            window.focus_force()
            close_button.focus_force()
        except tk.TclError:
            return

        # On Windows, visual topmost position does not always equal keyboard
        # focus; explicitly request foreground activation for the dialog.
        if sys.platform == 'win32':
            try:
                import ctypes
                from ctypes import wintypes
                user32 = ctypes.windll.user32
                user32.GetAncestor.argtypes = (wintypes.HWND, wintypes.UINT)
                user32.GetAncestor.restype = wintypes.HWND
                user32.ShowWindow.argtypes = (wintypes.HWND, ctypes.c_int)
                user32.ShowWindow.restype = wintypes.BOOL
                user32.BringWindowToTop.argtypes = (wintypes.HWND,)
                user32.BringWindowToTop.restype = wintypes.BOOL
                user32.SetForegroundWindow.argtypes = (wintypes.HWND,)
                user32.SetForegroundWindow.restype = wintypes.BOOL
                hwnd = user32.GetAncestor(int(window.winfo_id()), 2)  # GA_ROOT
                if not hwnd:
                    hwnd = int(window.winfo_id())
                user32.ShowWindow(hwnd, 5)  # SW_SHOW
                user32.BringWindowToTop(hwnd)
                user32.SetForegroundWindow(hwnd)
                close_button.focus_force()
            except Exception:
                # Tk's topmost/focus_force are the cross-platform fallback.
                pass

    # Make it modal with respect to the importer's hidden Tk root. This also
    # ensures keyboard events go to this window once foregrounded.
    window.update_idletasks()
    try:
        window.grab_set()
    except tk.TclError:
        pass
    bring_to_front()
    # A delayed focus request handles Windows/E3 briefly reclaiming focus.
    window.after(120, bring_to_front)
    window.after(350, bring_to_front)
    parent.wait_window(window)


def show_issue_window(parent, summary, issues, title='Zuken Import — Issues'):
    """Compact selectable scrollable error report, no spawned console or log."""
    import tkinter as tk
    from tkinter import ttk

    window = tk.Toplevel(parent)
    window.title(title + ' (import already finished)')
    window.geometry('700x385')
    window.minsize(460, 250)
    try:
        # Do NOT set transient(parent): the file-picker root is withdrawn,
        # and Windows can hide a transient child of a withdrawn parent.
        window.attributes('-topmost', True)
        window.deiconify()
        window.lift()
    except Exception:
        pass
    frame = ttk.Frame(window, padding=12)
    frame.pack(fill='both', expand=True)
    ttk.Label(frame, text='Import finished — review the items below',
              font=('Segoe UI', 11, 'bold')).pack(anchor='w', pady=(0, 7))
    ttk.Label(frame, text='\n'.join(issues), justify='left',
              wraplength=660).pack(anchor='w', pady=(0, 7))
    ttk.Label(frame, text='Details (select text or click Copy details):').pack(anchor='w')
    body = ttk.Frame(frame)
    body.pack(fill='both', expand=True, pady=(4, 10))
    scrollbar = ttk.Scrollbar(body)
    scrollbar.pack(side='right', fill='y')
    text_box = tk.Text(body, wrap='word', height=9, yscrollcommand=scrollbar.set,
                       font=('Consolas', 9))
    text_box.pack(side='left', fill='both', expand=True)
    scrollbar.config(command=text_box.yview)
    records = summary.get('output_lines') or []
    relevant = []
    tokens = ('BLOCK LOOKUP', 'BLOCK MATCH', 'BLOCK ', 'SKIP ', 'ERROR',
              'FATAL', 'STOP', 'NOTICE:', 'FAILED', 'WARNING', 'SUMMARY',
              'Pin-local attributes written:', 'Pin-local value readback verified:',
              'Pin-local text formatting verified:', 'Top-pin designations bold/visible:',
              'ALL LOCAL SIGNAL LABELS VISIBLE', 'VISIBILITY ERROR',
              'Local signal labels visible', 'Local label visibility')
    for line in records:
        if any(tag in line for tag in tokens):
            relevant.append(line)
    if not relevant:
        relevant = list(records[-50:])
    display = '\n'.join(relevant[-180:])
    full = '\n'.join(records)
    text_box.insert('1.0', display)
    text_box.config(state='disabled')
    row = ttk.Frame(frame)
    row.pack(fill='x')

    def copy_text():
        window.clipboard_clear()
        window.clipboard_append(full)
        window.update()

    ttk.Button(row, text='Copy full diagnostic output', command=copy_text).pack(side='left')
    ttk.Button(row, text='Close', command=window.destroy).pack(side='right')
    window.protocol('WM_DELETE_WINDOW', window.destroy)
    window.lift()
    window.focus_force()
    print('IMPORT FINISHED WITH ISSUES — visible issue window open; click Close to exit.', flush=True)
    parent.wait_window(window)
    print('Issue window closed; script exiting.', flush=True)


def run_once():
    """Pick workbook, import quietly, and show success or actionable errors."""
    import tkinter as tk
    from tkinter import filedialog

    root = tk.Tk()
    root.withdraw()
    try:
        root.attributes('-topmost', True)
    except Exception:
        pass
    try:
        filename = filedialog.askopenfilename(
            parent=root,
            title='Select Excel file for E3 local signal names',
            filetypes=[('Excel workbooks', '*.xlsx *.xlsm'), ('All files', '*.*')],
        )
        if not filename:
            return

        # In a normal E3 run, keep the output pane clean. We retain all existing
        # diagnostics in memory and display them if the import actually fails.
        capture = io.StringIO()
        with redirect_stdout(capture), redirect_stderr(capture), warnings.catch_warnings(record=True) as caught:
            warnings.simplefilter('always')
            result = perform(Path(filename))

        extra = capture.getvalue().splitlines()
        if extra:
            # perform() already saves ConsoleLines into output_lines, so keep
            # captured stdout only as extra diagnostics if an issue occurs.
            result.setdefault('output_lines', []).extend(
                ['--- Additional captured output ---'] + extra[-250:]
            )
        if caught:
            result.setdefault('output_lines', []).extend(
                ['--- Python warnings ---'] + [str(w.message) for w in caught]
            )

        issues = import_issues(result)
        if issues:
            show_issue_window(root, result, issues)
        else:
            show_complete_window(root, result)
    except Exception as exc:
        details = traceback.format_exc()
        try:
            show_issue_window(root,
                {'output_lines': details.splitlines()},
                [f'{type(exc).__name__}: {exc}'],
                title='Zuken Import - Unexpected Error')
        except Exception:
            try:
                from tkinter import messagebox
                messagebox.showerror('Zuken Import - Unexpected Error',
                                     f'{type(exc).__name__}: {exc}', parent=root)
            except Exception:
                # Keep the exception visible if tkinter's error dialog is unavailable.
                print('UNEXPECTED ERROR:\n' + details, flush=True)
    finally:
        root.destroy()


def main():
    run_once()


if __name__ == '__main__':
    main()
