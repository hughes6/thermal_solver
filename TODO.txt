"""Build the modular Zuken Mega Script as ONE Windows EXE.

Place the four original Excel templates in ./templates before building.
Run: py build_exe.py
"""
from __future__ import annotations

from pathlib import Path
import subprocess
import sys
import os
from importlib.util import find_spec

ROOT = Path(__file__).resolve().parent
EXPECTED_TEMPLATES = (
    "Cable_Template.xlsx",
    "Block_Maker_Template.xlsm",
    "Block_Pin_Arranger_Template.xlsm",
    "Block_Signal_Adder_Template.xlsm",
)


def main() -> int:
    if sys.platform != "win32":
        print("This project's E3/COM executable must be built on Windows.", file=sys.stderr)
        return 1
    if not (ROOT / "Zuken_Mega_Script.py").is_file():
        print("Missing launcher entry point.", file=sys.stderr)
        return 1
    # These imports are inside tool_scripts/ data files, not ordinary Python
    # imports that PyInstaller can discover from the entry point. Refuse to
    # silently build an EXE without the Runner's GUI/report dependencies.
    missing_modules = [package for module, package in
                       (("wx", "wxPython"), ("pandas", "pandas"))
                       if find_spec(module) is None]
    if missing_modules:
        print("Missing Zuken Script Runner build dependencies: " +
              ", ".join(missing_modules), file=sys.stderr)
        print("Install with: py -m pip install " +
              " ".join(missing_modules), file=sys.stderr)
        return 1
    missing = [p for p in EXPECTED_TEMPLATES if not (ROOT / "templates" / p).is_file()]
    if missing:
        print("WARNING: Missing original workbook templates:")
        for name in missing:
            print(" -", name)
        print("Build can proceed, but Downloads for these templates won't be bundled.")
        print("Copy your originals into templates/ before building to include them.\n")
    sep = os.pathsep  # ';' for PyInstaller --add-data on Windows
    args = [
        sys.executable, "-m", "PyInstaller",
        "--noconfirm", "--clean", "--onefile", "--windowed",
        "--name", "Zuken_Mega_Script",
        "--add-data", "templates" + sep + "templates",
        "--add-data", "tool_scripts" + sep + "tool_scripts",
        "--collect-all", "pdfplumber",
        "--collect-submodules", "pdfminer",
        # Zuken Script Runner source is bundled as data: PyInstaller cannot
        # statically see its wxPython GUI or pandas report generators.
        "--collect-all", "wx",
        "--collect-all", "pandas",
        "--hidden-import", "tkinter.ttk",
        "--hidden-import", "tkinter.scrolledtext",
        "--collect-submodules", "tkinter",
        "--hidden-import", "pythoncom",
        "--hidden-import", "pywintypes",
        "--hidden-import", "pyodbc",
        "--hidden-import", "win32timezone",
        "--hidden-import", "win32com.client",
        "--hidden-import", "e3series",
        "--collect-submodules", "openpyxl",
        "--collect-submodules", "win32com",
        "--collect-submodules", "e3series",
        "Zuken_Mega_Script.py",
    ]
    print("Building dist/Zuken_Mega_Script.exe from the separate tool .py files...")
    return subprocess.call(args, cwd=str(ROOT))

if __name__ == "__main__":
    raise SystemExit(main())
