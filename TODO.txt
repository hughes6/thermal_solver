    root.attributes("-topmost", True)
    root.update()

    selected_path = filedialog.askopenfilename(

#====================================================================================================================================
# VALUE HELPERS
#====================================================================================================================================
def clean_excel_value(value):
    if value is None:
        return ""

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def normalize_pin_name(value):
    return clean_excel_value(value).upper()


def normalize_connector_token(value):
    text = clean_excel_value(value).upper()

    text = text.replace(" ", "")

    if text == "":
        return ""

    match = re.fullmatch(
        r"P?(\d+[A-Z]*)",
        text
    )

    if match:
        return f"P{match.group(1)}"

    return text


def cell_has_bar(value):
    text = clean_excel_value(value)

    return "|" in text

# FIND DATA EXTENT
#====================================================================================================================================
def get_last_data_row(ws):
    columns = [
        LEFT_OUTER_SHIELD_COL,
        LEFT_INNER_SHIELD_COL,
        LEFT_TWIST_A_COL,
        LEFT_TWIST_B_COL,
        LEFT_CONNECTOR_COL,
        PIN_1_COL,
        PIN_2_COL,
        RIGHT_CONNECTOR_COL,
        RIGHT_TWIST_A_COL,
        RIGHT_TWIST_B_COL,
        RIGHT_INNER_SHIELD_COL,
        RIGHT_OUTER_SHIELD_COL
    ]

    scan_end = min(
        max(
            ws.max_row,
            DATA_FIRST_ROW
        ),
        DATA_FIRST_ROW + MAX_SCAN_ROWS
    )

    last_row = DATA_FIRST_ROW - 1

    for row in range(
        DATA_FIRST_ROW,
        scan_end + 1
    ):
        for col in columns:
            value = ws[
                f"{col}{row}"
            ].value

            if clean_excel_value(value) != "":
                last_row = row

                break

    return last_row


#====================================================================================================================================
# READ LEFT OR RIGHT SIDE TABLE
#
# LEFT:
#
# H = outer SHL
# I = inner SHL
# J = twist A
# K = twist B
# L = connector P#
# N = pin
#
#
# RIGHT:
#
# O = pin
# Q = connector P#
# R = twist A
# S = twist B
# T = inner SHL
# U = outer SHL
#====================================================================================================================================
def read_side_rows(
    ws,
    side,
    last_row
):
    if side == "LEFT":
        connector_col = LEFT_CONNECTOR_COL
        pin_col = PIN_1_COL
        twist_a_col = LEFT_TWIST_A_COL
        twist_b_col = LEFT_TWIST_B_COL
        inner_col = LEFT_INNER_SHIELD_COL
        outer_col = LEFT_OUTER_SHIELD_COL

    else:
        connector_col = RIGHT_CONNECTOR_COL
        pin_col = PIN_2_COL
        twist_a_col = RIGHT_TWIST_A_COL
        twist_b_col = RIGHT_TWIST_B_COL
        inner_col = RIGHT_INNER_SHIELD_COL
        outer_col = RIGHT_OUTER_SHIELD_COL

    rows = []

    current_connector = ""

    for excel_row in range(
        DATA_FIRST_ROW,
        last_row + 1
    ):
        connector_value = clean_excel_value(
            ws[
                f"{connector_col}{excel_row}"
            ].value
        )

        pin_name = normalize_pin_name(
            ws[
                f"{pin_col}{excel_row}"
            ].value
        )

        if connector_value != "":
            current_connector = normalize_connector_token(
                connector_value
            )

        if pin_name == "":
            continue

        row_info = {
            "side": side,
            "excel_row": excel_row,
            "connector": current_connector,
            "pin": pin_name,
            "twist_a": cell_has_bar(
                ws[
                    f"{twist_a_col}{excel_row}"
                ].value
            ),
            "twist_b": cell_has_bar(
                ws[
                    f"{twist_b_col}{excel_row}"
                ].value
            ),
            "inner_shield": cell_has_bar(
                ws[
                    f"{inner_col}{excel_row}"
                ].value
            ),
            "outer_shield": cell_has_bar(
                ws[
                    f"{outer_col}{excel_row}"
                ].value
            )
        }

        rows.append(
            row_info
        )

    return rows


#====================================================================================================================================
# CONNECTOR SECTIONS
#
# A symbol is NEVER allowed to cross:
#
# 1. A connector change
# 2. A blank row gap
#
# Therefore a twist or shield on P1 can never accidentally continue
# into P2.
#====================================================================================================================================
def split_into_sections(rows):
    sections = []

    current = []

    for row in rows:
        if not current:
            current = [
                row
            ]

            continue

        previous = current[-1]

        connector_changed = (
            row["connector"]
            != previous["connector"]
        )

        row_gap = (
            row["excel_row"]
            != previous["excel_row"] + 1
        )

        if connector_changed or row_gap:
            sections.append(
                current
            )

            current = [
                row
            ]

        else:
            current.append(
                row
            )

    if current:
        sections.append(
            current
        )

    return sections


#====================================================================================================================================
# CONTIGUOUS MARKED RANGES
#
# Example:
#
# |
# |
# |
# |
#
# becomes one range.
#
#
# |
# |
#
# |
# |
#
# becomes two ranges.
#====================================================================================================================================
def true_ranges(
    rows,
    key
):
    ranges = []

    start = None

    for index, row in enumerate(rows):
        marked = bool(
            row[key]
        )

        if marked and start is None:
            start = index

        if not marked and start is not None:
            ranges.append(
                (
                    start,
                    index - 1
                )
            )

            start = None

    if start is not None:
        ranges.append(
            (
                start,
                len(rows) - 1
            )
        )

    return ranges


def range_pins(
    rows,
    start,
    end
):
    pins = []

    for index in range(
        start,
        end + 1
    ):
        pins.append(
            rows[index]["pin"]
        )

    return pins


def range_excel_rows(
    rows,
    start,
    end
):
    excel_rows = []

    for index in range(
        start,
        end + 1
    ):
        excel_rows.append(
            rows[index]["excel_row"]
        )

    return excel_rows


#====================================================================================================================================
# BUILD SYMBOL JOBS
#
#
# TWIST A / TWIST B
# -----------------
# Each contiguous group of | marks becomes one twist symbol.
#
#
# INNER SHIELD
# ------------
# If an entire twist group is marked with inner SHL:
#
#     shield that individual twist.
#
# Example:
#
# inner    twist A    twist B
#   |         |
#   |         |
#   |                    |
#   |                    |
#
# becomes:
#
#     twist A + shield
#     twist B + shield
#
#
# INNER SHIELD WITHOUT TWIST
# --------------------------
# Remaining inner shield marks become normal shield-only groups.
#
#
# OUTER SHIELD
# ------------
# Each contiguous outer shield range becomes ONE shield over the
# complete section.
#====================================================================================================================================
def build_section_jobs(rows):
    jobs = []

    warnings = []

    if not rows:
        return jobs, warnings

    side = rows[0]["side"]

    connector = rows[0]["connector"]

    if connector == "":
        warnings.append(
            f"{side}: rows "
            f"{rows[0]['excel_row']}-"
            f"{rows[-1]['excel_row']} "
            f"have pins but no connector number"
        )

        return jobs, warnings

    conflicts = []

    for row in rows:
        if (
            row["twist_a"]
            and row["twist_b"]
        ):
            conflicts.append(
                row["excel_row"]
            )

    if conflicts:
        message = (
            f"{side} {connector}: "
            f"same wire is marked in BOTH twist columns "
            f"on Excel rows {conflicts}"
        )

        if STRICT_TWIST_VALIDATION:
            raise Exception(
                message
            )

        warnings.append(
            message
        )

    twist_jobs = []

    twist_definitions = [
        (
            "twist_a",
            "A"
        ),
        (
            "twist_b",
            "B"
        )
    ]

    for key, twist_column in twist_definitions:
        ranges = true_ranges(
            rows,
            key
        )

        for start, end in ranges:
            pins = range_pins(
                rows,
                start,
                end
            )

            excel_rows = range_excel_rows(
                rows,
                start,
                end
            )

            if len(pins) < 2:
                warnings.append(
                    f"{side} {connector}: "
                    f"twist {twist_column} only contains one wire "
                    f"on Excel row {excel_rows[0]}"
                )

                continue

            job = {
                "type": "twist",
                "side": side,
                "connector": connector,
                "twist_column": twist_column,
                "start": start,
                "end": end,
                "pins": pins,
                "excel_rows": excel_rows,
                "symbol_name": TWIST_SYMBOL_NAME
            }

            twist_jobs.append(
                job
            )

            jobs.append(
                job
            )

    #================================================================================================================================
    # INNER SHIELDS THAT BELONG TO TWISTS
    #================================================================================================================================
    inner_consumed = set()

    for twist_job in twist_jobs:
        indexes = list(
            range(
                twist_job["start"],
                twist_job["end"] + 1
            )
        )

        fully_inner_shielded = all(
            rows[index]["inner_shield"]
            for index in indexes
        )

        if fully_inner_shielded:
            shield_job = {
                "type": "shield_inner",
                "side": side,
                "connector": connector,
                "twist_column": twist_job["twist_column"],
                "pins": list(
                    twist_job["pins"]
                ),
                "excel_rows": list(
                    twist_job["excel_rows"]
                ),
                "symbol_name": SHIELD_SYMBOL_NAME
            }

            jobs.append(
                shield_job
            )

            for index in indexes:
                inner_consumed.add(
                    index
                )

    #================================================================================================================================
    # INNER SHIELD MARKS NOT USED BY A TWIST
    #
    # These become standalone shield spans.
    #================================================================================================================================
    remaining = []

    for index, row in enumerate(rows):
        marked = (
            row["inner_shield"]
            and index not in inner_consumed
        )

        remaining.append(
            bool(marked)
        )

    start = None

    for index, marked in enumerate(remaining):
        if marked and start is None:
            start = index

        if not marked and start is not None:
            end = index - 1

            jobs.append(
                {
                    "type": "shield_inner",
                    "side": side,
                    "connector": connector,
                    "twist_column": None,
                    "pins": range_pins(
                        rows,
                        start,
                        end
                    ),
                    "excel_rows": range_excel_rows(
                        rows,
                        start,
                        end
                    ),
                    "symbol_name": SHIELD_SYMBOL_NAME
                }
            )

            start = None

    if start is not None:
        end = len(rows) - 1

        jobs.append(
            {
                "type": "shield_inner",
                "side": side,
                "connector": connector,
                "twist_column": None,
                "pins": range_pins(
                    rows,
                    start,
                    end
                ),
                "excel_rows": range_excel_rows(
                    rows,
                    start,
                    end
                ),
                "symbol_name": SHIELD_SYMBOL_NAME
            }
        )

    #================================================================================================================================
    # OUTER SHIELDS
    #
    # Always one shield across the entire marked region.
    #================================================================================================================================
    outer_ranges = true_ranges(
        rows,
        "outer_shield"
    )

    for start, end in outer_ranges:
        outer_job = {
            "type": "shield_outer",
            "side": side,
            "connector": connector,
            "twist_column": None,
            "pins": range_pins(
                rows,
                start,
                end
            ),
            "excel_rows": range_excel_rows(
                rows,
                start,
                end
            ),
            "symbol_name": SHIELD_SYMBOL_NAME
        }

        jobs.append(
            outer_job
        )

    return jobs, warnings


#====================================================================================================================================
# BUILD EXPECTED E3 DEVICE NAME
#
# Example:
#
# B1:
#     W804
#
# Connector cell:
#     1
#
# E3 device:
#     W804.P1
#
#
# If connector cell already contains P1 it still becomes W804.P1.
#====================================================================================================================================
def device_name_candidates(
    base_device_name,
    connector_token
):
    base = clean_excel_value(
        base_device_name
    )

    connector = normalize_connector_token(
        connector_token
    )

    if connector == "":
        return []

    if "." in connector:
        return [
            connector
        ]

    return [
        f"{base}.{connector}"
    ]


#====================================================================================================================================
# GET ALL E3 DEVICES
#====================================================================================================================================
def get_all_devices(
    prj,
    dev
):
    info = prj.GetAllDeviceIds(
        None
    )

    ids = [
        value
        for value in info[1]
        if value is not None
    ]

    records = []

    for device_id in ids:
        dev.SetId(
            device_id
        )

        try:
            name = clean_excel_value(
                dev.GetName()
            )

        except Exception:
            continue

        records.append(
            {
                "device_id": device_id,
                "name": name
            }
        )

    return records


#====================================================================================================================================
# GET SCHEMATIC LOCATION OF AN ALREADY PLACED PIN
#
# This is how the overlay script knows where the existing cable
# signal lines are.
#
# We do NOT move the pin.
#
# We only read:
#
#     sheet
#     x
#     y
#====================================================================================================================================
def get_pin_location(
    pin,
    pin_id
):
    pin.SetId(
        pin_id
    )

    xloc = [
        0.0
    ]

    yloc = [
        0.0
    ]

    grid = [
        ""
    ]

    column = [
        ""
    ]

    row = [
        ""
    ]

    try:
        result = pin.GetSchemaLocation(
            xloc,
            yloc,
            grid,
            column,
            row
        )

    except Exception:
        return None

    if not isinstance(
        result,
        tuple
    ):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(
            result[0]
        )

        x = float(
            result[1]
        )

        y = float(
            result[2]
        )

    except Exception:
        return None

    if sheet_id <= 0:
        return None

    return {
        "sheet_id": sheet_id,
        "x": x,
        "y": y
    }


#====================================================================================================================================
# BUILD PIN NAME -> PLACED SCHEMATIC LOCATIONS FOR ONE E3 DEVICE
#
# More than one location is allowed because a device could have
# multiple views.
#====================================================================================================================================
def load_device_pin_map(
    dev,
    pin,
    device_id
):
    dev.SetId(
        device_id
    )

    info = dev.GetAllPinIds(
        None
    )

    pin_ids = [
        value
        for value in info[1]
        if value is not None
    ]

    pin_map = {}

    for pin_id in pin_ids:
        pin.SetId(
            pin_id
        )

        try:
            pin_name = normalize_pin_name(
                pin.GetName()
            )

        except Exception:
            continue

        location = get_pin_location(
            pin,
            pin_id
        )

        if pin_name not in pin_map:
            pin_map[
                pin_name
            ] = []

        if location is not None:
            pin_map[
                pin_name
            ].append(
                {
                    "pin_id": pin_id,
                    "sheet_id": location["sheet_id"],
                    "x": location["x"],
                    "y": location["y"]
                }
            )

    return pin_map


#====================================================================================================================================
# FIND BEST E3 DEVICE FOR ONE CONNECTOR SECTION
#
# Supports:
#
#     W804.P1
#
# and a view-like name such as:
#
#     W804.P1#1
#====================================================================================================================================
def resolve_device(
    base_device_name,
    connector_token,
    required_pins,
    all_devices,
    dev,
    pin,
    cache
):
    candidates = device_name_candidates(
        base_device_name,
        connector_token
    )

    candidate_upper = [
        value.upper()
        for value in candidates
    ]

    matches = []

    for record in all_devices:
        name_upper = record[
            "name"
        ].upper()

        if name_upper in candidate_upper:
            matches.append(
                record
            )

            continue

        for candidate in candidate_upper:
            if name_upper.startswith(
                candidate + "#"
            ):
                matches.append(
                    record
                )

                break

    if not matches:
        raise Exception(
            f"Could not find E3 device for connector "
            f"{connector_token}. "
            f"Tried {candidates}"
        )

    scored = []

    for record in matches:
        device_id = record[
            "device_id"
        ]

        if device_id not in cache:
            cache[
                device_id
            ] = load_device_pin_map(
                dev,
                pin,
                device_id
            )

        pin_map = cache[
            device_id
        ]

        score = 0

        for pin_name in required_pins:
            if (
                pin_name in pin_map
                and pin_map[pin_name]
            ):
                score += 1

        scored.append(
            (
                score,
                record
            )
        )

    scored.sort(
        key=lambda item: item[0],
        reverse=True
    )

    best_score = scored[0][0]

    best_record = scored[0][1]

    if best_score == 0:
        raise Exception(
            f"Found {best_record['name']} "
            f"but none of its requested pins "
            f"have schematic locations"
        )

    return (
        best_record,
        cache[
            best_record["device_id"]
        ]
    )


#====================================================================================================================================
# SELECT CABLE SHEET
#
# If the connector/device has multiple views, choose the sheet that
# contains the greatest number of the pins used by this connector
# section.
#====================================================================================================================================
def choose_section_sheet(
    required_pins,
    pin_map
):
    coverage = {}

    for pin_name in required_pins:
        locations = pin_map.get(
            pin_name,
            []
        )

        for location in locations:
            sheet_id = location[
                "sheet_id"
            ]

            if sheet_id not in coverage:
                coverage[
                    sheet_id
                ] = set()

            coverage[
                sheet_id
            ].add(
                pin_name
            )

    if not coverage:
        return None

    ranked = sorted(
        coverage.items(),
        key=lambda item: len(
            item[1]
        ),
        reverse=True
    )

    best_sheet_id = ranked[0][0]

    best_count = len(
        ranked[0][1]
    )

    tied = []

    for sheet_id, pins_on_sheet in ranked:
        if len(pins_on_sheet) == best_count:
            tied.append(
                sheet_id
            )

    if len(tied) > 1:
        print(
            "WARNING: multiple sheets have equal pin coverage:",
            tied
        )

        print(
            "USING SHEET:",
            best_sheet_id
        )

    return best_sheet_id


#====================================================================================================================================
# GET ONE SPECIFIC PIN LOCATION ON THE SELECTED CABLE SHEET
#====================================================================================================================================
def pin_location_on_sheet(
    pin_map,
    pin_name,
    sheet_id
):
    locations = pin_map.get(
        pin_name,
        []
    )

    for location in locations:
        if location["sheet_id"] == sheet_id:
            return location

    return None


#====================================================================================================================================
# HORIZONTAL OFFSET BY JOB TYPE
#====================================================================================================================================
def job_offset(job):
    if job["type"] == "twist":
        if job["twist_column"] == "A":
            return TWIST_A_OFFSET

        return TWIST_B_OFFSET

    if job["type"] == "shield_inner":
        if job["twist_column"] == "A":
            return INNER_SHIELD_TWIST_A_OFFSET

        if job["twist_column"] == "B":
            return INNER_SHIELD_TWIST_B_OFFSET

        return INNER_SHIELD_ONLY_OFFSET

    return OUTER_SHIELD_OFFSET


#====================================================================================================================================
# VERTICAL PADDING BY JOB TYPE
#====================================================================================================================================
def job_padding(job):
    if job["type"] == "twist":
        return TWIST_VERTICAL_PADDING

    if job["type"] == "shield_inner":
        return INNER_SHIELD_VERTICAL_PADDING

    return OUTER_SHIELD_VERTICAL_PADDING


#====================================================================================================================================
# DRAW ORDER
#
# Outer shield first.
# Inner shields second.
# Twists last.
#
# This should give the most sensible visual layering.
#====================================================================================================================================
def job_priority(job):
    if job["type"] == "shield_outer":
        return 0

    if job["type"] == "shield_inner":
        return 1

    return 2


#====================================================================================================================================
# CALCULATE SYMBOL LOCATION USING EXISTING CONNECTOR PIN LOCATIONS
#
# Existing pins give us:
#
#     connector X
#     wire Y
#
# Since the connected signal lines are horizontal, the pin Y values
# tell us exactly which wire rows the dynamic symbol needs to span.
#====================================================================================================================================
def calculate_placement(
    job,
    pin_map,
    sheet_id
):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(
            pin_map,
            pin_name,
            sheet_id
        )

        if location is None:
            raise Exception(
                f"{job['side']} "
                f"{job['connector']}: "
                f"pin {pin_name} "
                f"is not placed on E3 sheet {sheet_id}"
            )

        locations.append(
            location
        )

    x_values = [
        location["x"]
        for location in locations
    ]

    y_values = [
        location["y"]
        for location in locations
    ]

    anchor_x = statistics.median(
        x_values
    )

    bottom_y = min(
        y_values
    )

    top_y = max(
        y_values
    )

    # LEFT connector:
    #
    # wire travels to the right
    #
    # RIGHT connector:
    #
    # wire travels to the left
    direction = (
        1.0
        if job["side"] == "LEFT"
        else -1.0
    )

    x = (
        anchor_x
        + direction
        * job_offset(job)
    )

    padding = job_padding(
        job
    )

    y = (
        bottom_y
        - padding
    )

    height = (
        top_y
        - bottom_y
        + 2.0
        * padding
    )

    height = max(
        height,
        MIN_DYNAMIC_HEIGHT
    )

    placement = {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "height": height,
        "width": DYNAMIC_WIDTH,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y
    }

    return placement


#====================================================================================================================================
# PLACE ONE DYNAMIC SYMBOL
#
# IMPORTANT:
#
# This function ONLY adds the twist/shield graphic.
#
# It does NOT:
#
#     delete existing cable
#     replace existing cable
#     move connector
#     move connector pin
#     change signal line
#     re-place a pin view
#====================================================================================================================================
def place_job(
    sym,
    job,
    placement
):
    print("")
    print("------------------------------------------------------------")
    print("TYPE:", job["type"])
    print("SIDE:", job["side"])
    print("CONNECTOR:", job["connector"])
    print("PINS:", job["pins"])
    print("EXCEL ROWS:", job["excel_rows"])
    print("SYMBOL:", job["symbol_name"])
    print("SHEET:", placement["sheet_id"])
    print("ANCHOR PIN X:", placement["anchor_x"])
    print("X:", placement["x"])
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("PLACEMENT Y:", placement["y"])
    print("HEIGHT:", placement["height"])
    print("WIDTH:", placement["width"])

    if DRY_RUN:
        print(
            "DRY RUN - NOT PLACED"
        )

        return None

    load_result = sym.Load(
        job["symbol_name"],
        SYMBOL_VERSION
    )

    if load_result == 0:
        raise Exception(
            f"Could not load "
            f"{job['symbol_name']}"
        )

    placed_id = sym.PlaceDynamic(
        placement["sheet_id"],
        placement["x"],
        placement["y"],
        placement["height"],
        placement["width"],
        "",
        0
    )

    if placed_id == 0:
        raise Exception(
            f"PlaceDynamic failed for "
            f"{job['symbol_name']}"
        )

    print(
        "PLACED SYMBOL ID:",
        placed_id
    )

    return placed_id


#====================================================================================================================================
# MAIN
#====================================================================================================================================
def main():
    excel_path = ask_excel_path()

    if excel_path is None:
        raise Exception(
            "No Excel file selected"
        )

    workbook = load_workbook(
        excel_path,
        data_only=True,
        read_only=False
    )

    if TARGET_SHEET_NAME is None:
        ws = workbook.active

    else:
        if TARGET_SHEET_NAME not in workbook.sheetnames:
            raise Exception(
                f"Worksheet "
                f"{TARGET_SHEET_NAME} "
                f"was not found"
            )

        ws = workbook[
            TARGET_SHEET_NAME
        ]

    base_device_name = clean_excel_value(
        ws[
            DEVICE_NAME_CELL
        ].value
    )

    if base_device_name == "":
        raise Exception(
            f"Device name cell "
            f"{DEVICE_NAME_CELL} "
            f"is blank"
        )

    last_row = get_last_data_row(
        ws
    )

    if last_row < DATA_FIRST_ROW:
        raise Exception(
            "No cable overlay data found"
        )

    print("")
    print("============================================================")
    print("ZUKEN CABLE SHIELD / TWIST OVERLAY")
    print("============================================================")
    print("EXCEL:", excel_path)
    print("WORKSHEET:", ws.title)
    print("BASE DEVICE FROM B1:", base_device_name)
    print("DATA ROWS:", DATA_FIRST_ROW, "TO", last_row)

    #================================================================================================================================
    # READ EXCEL
    #================================================================================================================================
    left_rows = read_side_rows(
        ws,
        "LEFT",
        last_row
    )

    right_rows = read_side_rows(
        ws,
        "RIGHT",
        last_row
    )

    #================================================================================================================================
    # SPLIT BY CONNECTOR
    #================================================================================================================================
    left_sections = split_into_sections(
        left_rows
    )

    right_sections = split_into_sections(
        right_rows
    )

    sections = (
        left_sections
        + right_sections
    )

    packages = []

    warnings = []

    #================================================================================================================================
    # BUILD JOBS
    #================================================================================================================================
    for section_rows in sections:
        jobs, section_warnings = build_section_jobs(
            section_rows
        )

        if jobs:
            packages.append(
                {
                    "rows": section_rows,
                    "jobs": jobs
                }
            )

        warnings.extend(
            section_warnings
        )

    print("")
    print(
        "CONNECTOR SECTIONS WITH SYMBOLS:",
        len(packages)
    )

    if warnings:
        print("")
        print("============================================================")
        print("EXCEL WARNINGS")
        print("============================================================")

        for warning in warnings:
            print(
                "WARNING:",
                warning
            )

    #================================================================================================================================
    # CONNECT TO CURRENT E3 PROJECT
    #================================================================================================================================
    app = win32.gencache.EnsureDispatch(
        "CT.Application"
    )

    prj = app.CreateJobObject()

    dev = prj.CreateDeviceObject()

    pin = prj.CreatePinObject()

    sym = prj.CreateSymbolObject()

    app.PutMessage(
        "Starting Cable Shield / Twist Overlay Script"
    )

    #================================================================================================================================
    # GET EXISTING DEVICES
    #================================================================================================================================
    all_devices = get_all_devices(
        prj,
        dev
    )

    print("")
    print(
        "E3 DEVICE COUNT:",
        len(all_devices)
    )

    pin_map_cache = {}

    placed_ids = []

    #================================================================================================================================
    # PROCESS EACH CONNECTOR SECTION
    #================================================================================================================================
    for package in packages:
        section_rows = package[
            "rows"
        ]

        jobs = package[
            "jobs"
        ]

        side = section_rows[0][
            "side"
        ]

        connector = section_rows[0][
            "connector"
        ]

        required_pins = list(
            dict.fromkeys(
                row["pin"]
                for row in section_rows
            )
        )

        print("")
        print("============================================================")
        print("PROCESSING CONNECTOR SECTION")
        print("============================================================")
        print("SIDE:", side)
        print("CONNECTOR:", connector)
        print(
            "EXCEL ROWS:",
            section_rows[0]["excel_row"],
            "TO",
            section_rows[-1]["excel_row"]
        )
        print(
            "REQUIRED PINS:",
            required_pins
        )

        #============================================================================================================================
        # FIND EXISTING E3 CONNECTOR DEVICE
        #============================================================================================================================
        device_record, pin_map = resolve_device(
            base_device_name,
            connector,
            required_pins,
            all_devices,
            dev,
            pin,
            pin_map_cache
        )

        print(
            "E3 DEVICE ID:",
            device_record["device_id"]
        )

        print(
            "E3 DEVICE NAME:",
            device_record["name"]
        )

        #============================================================================================================================
        # FIND THE SCHEMATIC SHEET CONTAINING THIS EXISTING CABLE VIEW
        #============================================================================================================================
        sheet_id = choose_section_sheet(
            required_pins,
            pin_map
        )

        if sheet_id is None:
            raise Exception(
                f"Could not determine cable sheet "
                f"for {device_record['name']}"
            )

        print(
            "SELECTED CABLE SHEET ID:",
            sheet_id
        )

        #============================================================================================================================
        # PLACE OUTER SHIELD FIRST, INNER SHIELD SECOND, TWIST LAST
        #============================================================================================================================
        jobs = sorted(
            jobs,
            key=job_priority
        )

        for job in jobs:
            try:
                placement = calculate_placement(
                    job,
                    pin_map,
                    sheet_id
                )

                placed_id = place_job(
                    sym,
                    job,
                    placement
                )

                if placed_id is not None:
                    placed_ids.append(
                        placed_id
                    )

            except Exception as error:
                print(
                    "ERROR:",
                    str(error)
                )

                app.PutMessage(
                    f"Cable overlay warning: {error}"
                )

    #================================================================================================================================
    # COMPLETE
    #================================================================================================================================
    print("")
    print("============================================================")
    print("OVERLAY COMPLETE")
    print("============================================================")
    print(
        "PLACED SYMBOL COUNT:",
        len(placed_ids)
    )
    print(
        "PLACED SYMBOL IDS:",
        placed_ids
    )
    print("============================================================")

    app.PutMessage(
        f"Cable Shield / Twist Overlay complete. "
        f"Placed {len(placed_ids)} symbols."
    )

    workbook.close()

    del sym
    del pin
    del dev
    del prj
    del app


if __name__ == "__main__":
    main()
