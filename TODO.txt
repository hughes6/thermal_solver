def write_pdf_data_to_excel(source_excel_path, output_excel_path, pdf_connectors, pdf_roles, connector_info):
    import win32com.client as win32

    total_start = time.perf_counter()

    source_excel_path = os.path.abspath(source_excel_path)
    output_excel_path = os.path.abspath(output_excel_path)

    if source_excel_path.lower() != output_excel_path.lower():
        shutil.copy2(source_excel_path, output_excel_path)

    jobs = build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info)

    excel = win32.DispatchEx("Excel.Application")

    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    excel.AskToUpdateLinks = False

    workbook = None
    previous_calculation = None
    calculation_changed = False

    try:
        open_start = time.perf_counter()

        workbook = excel.Workbooks.Open(output_excel_path, UpdateLinks=0, ReadOnly=False, AddToMru=False)

        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)

        open_elapsed = time.perf_counter() - open_start

        print(f"EXCEL OPEN TIME: {open_elapsed:.2f} SECONDS")

        # ================================================
        # SET CALCULATION MODE AFTER WORKBOOK IS OPEN
        # ================================================
        try:
            previous_calculation = excel.Calculation
            excel.Calculation = XL_CALCULATION_MANUAL
            calculation_changed = True
            print("EXCEL CALCULATION MODE: MANUAL")
        except Exception as e:
            print("WARNING: COULD NOT SET EXCEL CALCULATION TO MANUAL:", repr(e))
            calculation_changed = False

        print("")
        print("============================================================")
        print("EXCEL CONNECTOR CHECK")
        print("============================================================")
        print("EXCEL CONNECTORS:", len(connector_info))
        print("PDF CONNECTORS:", len(pdf_connectors))
        print("============================================================")

        write_start = time.perf_counter()

        written_blocks = 0
        written_pin_rows = 0

        for job in jobs:
            print("")
            print("BLOCK:", job["block_index"] + 1, "EXCEL CONNECTOR:", repr(job["excel_name"]), "PDF KEY:", repr(job["canonical"]))

            if job["skip"]:
                print("SKIPPED -", job["reason"], ":", job["excel_name"])
                continue

            destination_range = worksheet.Range(worksheet.Cells(job["first_row"], job["first_column"]), worksheet.Cells(job["last_row"], job["last_column"]))

            destination_range.Value2 = job["matrix"]

            written_blocks += 1
            written_pin_rows += job["pin_count"]

            print("WRITTEN PINS:", job["pin_count"], "EXPECTED:", job["expected_pins"], "MISSING:", job["missing"])

        write_elapsed = time.perf_counter() - write_start

        print(f"BULK EXCEL WRITE TIME: {write_elapsed:.2f} SECONDS")

        # ================================================
        # RECALCULATION
        # ================================================
        if RECALCULATE_AFTER_WRITE:
            calc_start = time.perf_counter()

            try:
                if previous_calculation is not None:
                    excel.Calculation = previous_calculation

                excel.Calculate()

                calc_elapsed = time.perf_counter() - calc_start

                print(f"EXCEL RECALCULATION TIME: {calc_elapsed:.2f} SECONDS")

            except Exception as e:
                print("WARNING: EXCEL RECALCULATION FAILED:", repr(e))

        else:
            try:
                workbook.ForceFullCalculation = True
                workbook.FullCalculationOnLoad = True
            except Exception as e:
                print("WARNING: COULD NOT SET FULL CALCULATION ON LOAD:", repr(e))

        # ================================================
        # SAVE
        # ================================================
        save_start = time.perf_counter()

        workbook.Save()

        save_elapsed = time.perf_counter() - save_start

        print(f"EXCEL SAVE TIME: {save_elapsed:.2f} SECONDS")

        print("")
        print("============================================================")
        print("EXCEL WRITE COMPLETE")
        print("============================================================")
        print("OUTPUT:", output_excel_path)
        print("BLOCKS WRITTEN:", written_blocks)
        print("PIN ROWS WRITTEN:", written_pin_rows)
        print("============================================================")

    finally:
        if calculation_changed and previous_calculation is not None:
            try:
                excel.Calculation = previous_calculation
            except Exception:
                pass

        try:
            excel.EnableEvents = True
            excel.ScreenUpdating = True
        except Exception:
            pass

        if workbook is not None:
            try:
                workbook.Close(SaveChanges=False)
            except Exception:
                pass

        try:
            excel.Quit()
        except Exception:
            pass

    total_elapsed = time.perf_counter() - total_start

    print(f"TOTAL EXCEL WRITE PHASE: {total_elapsed:.2f} SECONDS")
