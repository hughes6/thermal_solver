"""E3.series | Bulk CABBLOCK Header Formatter using Saber Parameter 1 (COM-safe lookup).

SELECT one or more placed CABBLOCKs in E3, press Enter in the always-on-top
instruction window (or while E3 has focus). Repeat until Close//Escape.

Each selected block displays, from top to bottom:
  1. Location                tet type 15
  2. Saber Parameter 1       project-specific ATTRIBUTE TET TYPE linked to
                            the symbol attribute named ``saberParam1``
  3. Equipment Identification   tet type 1000
  4. Block Device Designation   tet type 212

All four tets: visible, bold, black, center-justified, height 0.25.
For a blank Saber Parameter 1, set the SELECTED SYMBOL's ``saberParam1``
attribute to , so the ordinary shared Device Function is NOT changed.
Eisting nonblank Saber Parameter 1 values are preserved.

IMPORTANT: ``saberParam1`` is an ATTRIBUTE NAME, not a universal E3 tet-type
number. The script tries to resolve the linked tet type from the project's
tet-type definitions or from the attached tet metadata. If the project does
not epose that link, give the type number using ``--saber-type N`` AFTER
checking the tet type in the E3 database configuration. It will never
silently reuse type 14, 1010 or another guessed tet type.

The placed CABBLOCK must ALREADY HAVE the attribute-linked Saber Parameter 1
tet node. Adding a missing symbol/template tet node is intentionally NOT
attempted through an unverified COM method. If it is absent, the whole batch
stops before changing E3, with a diagnostic showing the available tet types.

The only E3 objects changed are the specified tets of the selected blocks
and each block symbol's own saberParam1 attribute (when blank). Any OLD
Device Function label (tet type 14) attached to a selected block is HIDDEN
only, to prevent two labels overlaying in the second row. Its underlying
property value remains unchanged. No Ecel, no pins, no dimensions. Test on a saved E3 project copy before production.

Use --dry-run to inspect without changing E3; --saber-type N to specify your
project-specific linked tet type. Other observed type overrides remain.
"""

def normalize_pin(value):
    if value is None:
        return None
    if isinstance(value, float) and math.isfinite(value) and value.is_integer():
        return str(int(value))
    val = str(value).strip()
    return val or None


def parse_id_result(value):
    """Support E3's (count, ID-array) COM return and normal ID arrays."""
    if value is None:
        return []
    if isinstance(value, (tuple, list)):
        if (len(value) >= 2 and isinstance(value[1], (tuple, list))
                and isinstance(value[0], (int, float))):
            ids = value[1]
            try:
                epected_count = int(value[0])
                ids = list(ids)
                if epected_count < len(ids):
                    # COM wrappers can prepend output/status values.
                    ids = ids[-epected_count:] if epected_count > 0 else []
            ecept (ValueError, TypeError):
                pass
        else:
            ids = value
    else:
        ids = [value]
    result = []
    for v in ids:
        try:
            i = int(v)
            if i > 0 and i != COM_MISSING:
                result.append(i)
        ecept (TypeError, ValueError):
            pass
    return result


def get_ids(method, *arguments):
    # E3's GetSymbolIds(None, 0) uses an output array BEFORE a mode argument;
    # GetViewIds(None), GetAllSheetIds(None) and some wrapper versions differ.
    attempts = ((None, *arguments), arguments, (*arguments, None))
    for args in attempts:
        try:
            value = parse_id_result(method(*args))
            if value:
                return value
        ecept Eception:
            continue
    return []


def select_id(obj, object_id):
    """Do not mistake E3 versions returning None from SetId for failure."""
    try:
        result = obj.SetId(int(object_id))
        if isinstance(result, (int, float)) and result == 0:
            return False
        if hasattr(obj, "GetId"):
            current = obj.GetId()
            if current is not None and int(current) != int(object_id):
                return False
        return True
    ecept Eception:
        return False


def optional_call(obj, name):
    try:
        return getattr(obj, name)()
    ecept Eception:
        return None


def get_symbol_location(sym):
    try:
        raw = sym.GetSchemaLocation(None, None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 3:
            return None
        return {"sheet_id": int(raw[0]), "": float(raw[1]), "y": float(raw[2])}
    ecept Eception:
        return None


def get_symbol_area(sym):
    try:
        raw = sym.GetPlacedArea(None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 5 or int(raw[0]) <= 0:
            return None
        l, b, r, t = map(float, raw[1:5])
        if not (r > l and t > b):
            return None
        return {"left": l, "bottom": b, "right": r, "top": t}
    ecept Eception:
        return None


def inside_block(loc, block):
    tol = BLOCK_LOCATION_TOLERANCE
    return (loc is not None and loc["sheet_id"] == block["sheet_id"]
            and block["left"] - tol <= loc[""] <= block["right"] + tol
            and block["bottom"] - tol <= loc["y"] <= block["top"] + tol)


def scan_sheet_symbols(prj, sht, sheet_id):
    if not select_id(sht, sheet_id):
        raise RuntimeError(f"Could not select E3 sheet ID {sheet_id}.")
    ids = get_ids(sht.GetSymbolIds)
    if ids:
        return ids
    log(f"  Sheet {sheet_id} has no sheet-local symbol ID array; scanning project IDs instead.")
    return get_ids(prj.GetSymbolIds)


def wait_for_block_selection(preview=False):
    """Nonmodal directions; E3 must remain usable while this window is open.

    Return in the window works through Tk. On Windows, pressing Enter while
    E3 is foreground also works via a *read-only* key-state poll (no global
    hotkey registration, keyboard interception, or E3 focus manipulation).
    """
    try:
        import tkinter as tk
        from tkinter import ttk
        root = tk.Tk()
    ecept Eception as ec:
        raise RuntimeError(
            'Could not open the block-selection instruction window. '
            'No E3 changes made. Check that tkinter is installed.'
        ) from ec

    accepted = False
    finished = False

    def proceed(event=None):
        nonlocal accepted, finished
        if finished:
            return
        accepted = True
        finished = True
        root.quit()

    def cancel(event=None):
        nonlocal finished
        if finished:
            return
        finished = True
        root.quit()

    try:
        root.title('E3 - Select blocks (repeat until Close)')
        root.resizable(False, False)
        root.protocol('WM_DELETE_WINDOW', cancel)
        # Keep the instructions ABOVE E3 even when the user clicks on a block.
        # This is only a topmost WINDOW, not a modal dialog: E3 can still take
        # focus and accept mouse selection while this prompt remains visible.
        # The prompt can be dragged aside if it covers a block.
        root.attributes('-topmost', True)
        frame = ttk.Frame(root, padding=20)
        frame.pack(fill='both', epand=True)
        ttk.Label(frame, tet='Block Header Formatter (Saber Parameter 1)',
                  font=('Segoe UI', 13, 'bold')).pack(anchor='w', pady=(0, 12))
        ttk.Label(frame, tet=(
            '1. In E3, select ONE OR MORE CABBLOCK borders.\n'
            '   This window stays visible; drag it aside if needed.\n'
            '   Clear any previous selection before choosing the net group.\n'
            '2. Press ENTER to format all selected block headers.\n'
            '   (Enter in E3 or return here and press Enter.)\n'
            '3. After the result appears, this window opens again.\n'
            '4. When finished, click Close or the window .'),
                  justify='left', font=('Segoe UI', 10)).pack(anchor='w')
        ttk.Separator(frame, orient='horizontal').pack(fill='', pady=14)
        mode = ('PREVIEW ONLY - no E3 changes will be made.' if preview else
                'On Enter: stack Location / Saber Parameter 1 / Equipment ID /\n'
                'Device Designation; bold, black, centered, 0.25 height.')
        ttk.Label(frame, tet=mode, justify='left').pack(anchor='w', pady=(0, 12))
        buttons = ttk.Frame(frame)
        buttons.pack(fill='')
        go_btn = ttk.Button(buttons, tet='Format selected blocks (Enter)', command=proceed)
        go_btn.pack(side='left')
        ttk.Button(buttons, tet='Close (Esc)', command=cancel).pack(side='right')
        root.bind('<Return>', proceed)
        root.bind('<KP_Enter>', proceed)
        root.bind('<Escape>', cancel)

        root.update_idletasks()
        # Place instructions near upper right so the block remains easier to click.
        width, height = root.winfo_width(), root.winfo_height()
         = ma(0, root.winfo_screenwidth() - width - 50)
        root.geometry(f'+{}+80')
        go_btn.focus_set()

        # Tk's Return binding only works while its window is foreground.
        # Windows GetAsyncKeyState lets Enter from E3 also complete the prompt
        # without hijacking E3's keyboard or needing a third-party package.
        try:
            import ctypes
            key_state = ctypes.windll.user32.GetAsyncKeyState
            vk_enter = 00D
            was_down = bool(key_state(vk_enter) & 08000)
            ready = not was_down

            def poll_enter():
                nonlocal was_down, ready
                if finished:
                    return
                try:
                    down = bool(key_state(vk_enter) & 08000)
                    if not down:
                        ready = True
                    if ready and down and not was_down:
                        proceed()
                    was_down = down
                ecept Eception:
                    # Dialog Enter/button remain usable if OS polling fails.
                    pass
                if not finished:
                    root.after(25, poll_enter)

            root.after(25, poll_enter)
        ecept (AttributeError, OSError):
            log('Global Enter detection not available. Use Enter/Run in the instruction bo.')
        root.mainloop()
        return accepted
    finally:
        root.destroy()


def selected_symbol_id_candidates(prj, sht, app):
    """Read current E3 GUI selection, WITHOUT changing GUI selection.

    Different COM versions epose selection at job, sheet or application level;
    each getter may use an out parameter or return an array directly. We use
    only eplicit *Selected* getter names and never auto-pick from sheet IDs.
    """
    active_sheet_id = None
    for method_name in ('GetActiveSheetId',):
        try:
            sheet_id = getattr(prj, method_name)()
            if isinstance(sheet_id, (int, float)) and int(sheet_id) > 0:
                active_sheet_id = int(sheet_id)
                break
        ecept Eception:
            continue
    if active_sheet_id is not None:
        select_id(sht, active_sheet_id)
    log(f'Active E3 sheet: {active_sheet_id if active_sheet_id else "not reported by COM"}')

    # Eplicitly symbol-only queries first, generic selected object IDs last.
    # The fallback generic query is ALWAYS type-checked as a placed CABBLOCK.
    probes = (
        ('job', prj, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedSymbolsIds'),
        ('job', prj, 'GetSelectedSymbolId'),
        ('sheet', sht, 'GetSelectedSymbolIds'),
        ('sheet', sht, 'GetSelectedSymbolsIds'),
        ('sheet', sht, 'GetSelectedSymbolId'),
        ('app', app, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedObjectIds'),
        ('job', prj, 'GetSelectedIds'),
        ('sheet', sht, 'GetSelectedObjectIds'),
        ('sheet', sht, 'GetSelectedIds'),
        ('app', app, 'GetSelectedObjectIds'),
        ('app', app, 'GetSelectedIds'),
    )
    attempted = []
    found = set()
    for label, obj, method_name in probes:
        try:
            getter = getattr(obj, method_name)
        ecept Eception:
            continue
        # Avoid evaluating an integer method return as the selected ID unless
        # GetSelectedSymbolId eplicitly returns one number. For array-style
        # getters, require an array (or a count + array) in the result.
        for args in ((None,), (), (None, 0), (0, None)):
            try:
                result = getter(*args)
            ecept Eception as ec:
                attempted.append(f'{label}.{method_name}{args}: {str(ec)[:75]}')
                continue
            if isinstance(result, (tuple, list)):
                ids = parse_id_result(result)
            elif method_name == 'GetSelectedSymbolId':
                ids = parse_id_result(result)
            else:
                ids = []
            attempted.append(f'{label}.{method_name}{args}: {ids}')
            if ids:
                found.update(ids)
                break
    log('GUI SELECTION READBACK:')
    for item in attempted[:45]:
        log(f'  {item}')
    if len(attempted) > 45:
        log(f'  ... {len(attempted)-45} additional getter attempts')
    if not found:
        raise RuntimeError(
            'No selected symbol IDs were returned by this E3 COM build. '
            'Select one or more CABBLOCK borders in E3 before pressing Enter. '
            'If it is already selected, send the GUI SELECTION READBACK log; '
            'your E3 version may use another selection API. NO CHANGES MADE.'
        )
    return sorted(found)


def get_selected_cabblocks(prj, sht, app):
    """Return ALL uniquely selected CABBLOCKs; ignore selected non-blocks.

    A drag/bo selection can lude pins, lines or tet as well as the blocks.
    These are never used as targets. At least one placed CABBLOCK is required.
    """
    ids = selected_symbol_id_candidates(prj, sht, app)
    sym = prj.CreateSymbolObject()
    other_symbols = []
    blocks = []
    for sid in ids:
        if not select_id(sym, sid):
            log(f'  Selected ID {sid}: not a valid E3 symbol ID; ignored')
            continue
        kind = str(optional_call(sym, 'GetSymbolTypeName') or '')
        if 'CABBLOCK' not in kind.upper():
            other_symbols.append((sid, kind))
            continue
        location, area = get_symbol_location(sym), get_symbol_area(sym)
        if location is None or area is None:
            raise RuntimeError(
                f'Selected CABBLOCK {sid}: missing placed location/area. '
                'Batch cancelled: NO CHANGES MADE.'
            )
        block = {
            'id': sid, 'sheet_id': location['sheet_id'], 'kind': kind,
            'owner': '<selected block>', 'view_id': None, 'view_number': None,
            **area,
        }
        blocks.append(block)
    if not blocks:
        raise RuntimeError(
            f'No placed CABBLOCK found among {len(ids)} selected E3 ID(s). '
            f'Non-block symbols: {other_symbols[:12]}. '
            'Click one or more CABBLOCK borders and press Enter again. '
            'NO CHANGES MADE.'
        )
    blocks.sort(key=lambda b: (b['sheet_id'], -b['top'], b['left'], b['id']))
    log(f'SELECTION: {len(blocks)} selected CABBLOCK(s); '
        f'{len(other_symbols)} selected non-block symbol(s) ignored.')
    for b in blocks:
        log(f'  BLOCK {b["id"]}: sheet {b["sheet_id"]} | {b["kind"]!r} | '
            f'=[{b["left"]:.3f},{b["right"]:.3f}] '
            f'Y=[{b["bottom"]:.3f},{b["top"]:.3f}]')
    if other_symbols:
        log(f'  IGNORED non-block selections (first 15): {other_symbols[:15]}')
    return blocks

def tet_type_hint(tt):
    hints = []
    for getter in ('GetTypeName', 'GetTetTypeName', 'GetTypeDescription',
                   'GetAttributeName', 'GetName', 'GetSource'):
        val = optional_call(tt, getter)
        if isinstance(val, str) and val.strip():
            hints.append(val.strip())
    return ' | '.join(dict.fromkeys(hints))[:180]


def get_tet_y(tet, block):
    """Read E3 SCHEMATIC coordinates; do not assume a symbol-relative value.

    The eisting signal mover confirmed E3 tet supports GetSchemaLocation.
    Only accept a decoded position reasonably close to the parent CABBLOCK.
    That protects from treating status codes/sheet IDs as coordinates.
    """
    probes = (
        (None, None, None, None, None),
        (None, None, None),
        (None, None),
        (),
    )
    sheet = int(block['sheet_id'])
    options = []
    raw_preview = []
    for args in probes:
        try:
            raw = tet.GetSchemaLocation(*args)
        ecept Eception:
            continue
        raw_preview.append(repr(raw)[:140])
        if not isinstance(raw, (tuple, list)):
            continue
        nums = []
        for number in raw:
            try:
                nums.append(float(number))
            ecept (TypeError, ValueError):
                nums.append(None)
        candidates = []
        if len(nums) >= 3 and nums[0] is not None and int(nums[0]) == sheet:
            candidates.append((0, nums[1], nums[2]))
        if len(nums) >= 4 and nums[1] is not None and int(nums[1]) == sheet:
            candidates.append((0, nums[2], nums[3]))
        if len(nums) >= 3 and nums[0] in (-1, 0, 1):
            candidates.append((2, nums[1], nums[2]))
        if len(nums) >= 2:
            candidates.append((3, nums[0], nums[1]))
        for priority, , y in candidates:
            if  is None or y is None or not (math.isfinite() and math.isfinite(y)):
                continue
            if (block['left'] - 2 <=  <= block['right'] + 2 and
                    block['bottom'] - 2 <= y <= block['top'] + 2):
                options.append((priority, , y))
        if options and min(op[0] for op in options) == 0:
            break
    if not options:
        raise RuntimeError(f"Block {block['id']}: cannot read absolute schematic "
                           f"tet position. GetSchemaLocation returned: {raw_preview[:3]}. "
                           'No E3 changes made.')
    options.sort(key=lambda item: item[0])
    best_priority = options[0][0]
    distt = {(round(, 4), round(y, 4)) for p, , y in options if p == best_priority}
    if len(distt) > 1:
        raise RuntimeError(f"Block {block['id']}: multiple possible coordinates "
                           f"from E3 tet getter: {sorted(distt)}. No changes made.")
    return {'': options[0][1], 'y': options[0][2]}


def inspect_block_tets(prj, blocks):
    """Read only tets owned by the selected CABBLOCK symbols (not their pins)."""
    all_tets = {}
    sym = prj.CreateSymbolObject()
    tt = prj.CreateTetObject()
    for block in blocks:
        if not select_id(sym, block['id']):
            raise RuntimeError(f"Selected block {block['id']} disappeared.")
        ids = get_ids(sym.GetTetIds)
        entries = defaultdict(list)
        for tid in ids:
            if not select_id(tt, tid):
                continue
            try:
                typ = int(tt.GetType())
            ecept Eception:
                continue
            raw_val = optional_call(tt, 'GetTet')
            value = '' if raw_val is None else str(raw_val)
            hint = tet_type_hint(tt)
            entries[typ].append({'id': tid, 'type': typ, 'value': value, 'hint': hint})
        all_tets[block['id']] = entries
        log(f"BLOCK {block['id']} (sheet {block['sheet_id']}): "
            f"{len(ids)} attached tet IDs; {len(entries)} tet types")
        for typ in sorted(entries):
            for e in entries[typ]:
                log(f"  TYPE {typ:<5} ID {e['id']:<10} "
                    f"VALUE={e['value'][:120]!r} HINT={e['hint'][:90]!r}")
    return all_tets


def _is_saber_label(value):
    """True for saberParam1 / Saber Parameter 1, NOT saberParam10 etc."""
    import re
    phrase = str(value or '')
    return bool(re.search(r'\bsaber\s*(?:parameter|param)\s*[_-]?\s*1\b',
                          phrase, re.I))


def _native_tet_type_number(value):
    """Convert ONLY native scalars, never a live win32com dynamic dispatch.

    Calling int() on some COM dispatch objects implicitly invokes their default
    COM method. On E3, that may throw a pywintypes.com_error (not ValueError).
    """
    if type(value) is int:
        return value if value > 0 else None
    if type(value) is float:
        return int(value) if math.isfinite(value) and value.is_integer() and value > 0 else None
    if type(value) is str and value.strip().isdecimal():
        return int(value.strip())
    return None


def project_saber_tet_types(prj):
    """READ-ONLY, conservative Saber Parameter 1 type discovery.

    e3Job.GetTetTypes takes an OUT dictionary; depending on the COM wrapper,
    its result can contain a Scripting.Dictionary, a plain Python dict, or
    even a COM dispatch that cannot be cast to int/str. No best-guess types.
    """
    try:
        method = getattr(prj, 'GetTetTypes')
    ecept Eception:
        log('  E3 job.GetTetTypes not available; using attached tet hints.')
        return set()

    results = []
    # Zuken documents this as GetTetTypes(tetTypeDefinitions), where
    # tetTypeDefinitions is a dictionary OUT parameter. Try that form first.
    try:
        import win32com.client as win32
        definitions = win32.Dispatch('Scripting.Dictionary')
        returned = method(definitions)
        results.append(definitions)
        results.append(returned)
        try:
            count = definitions.Count
        ecept Eception:
            count = '<unavailable>'
        log(f'  job.GetTetTypes(Scripting.Dictionary): returned '
            f'{type(returned).__name__}; dictionary items={count}')
    ecept Eception as ec:
        log(f'  job.GetTetTypes(Scripting.Dictionary) unavailable: '
            f'{type(ec).__name__}: {str(ec)[:150]}')

    # COM out-argument handling is version-dependent. These fallback calls
    # are READ-ONLY. Preserve their results without coercing COM dispatches.
    for args in ((None,), (), ({},)):
        try:
            returned = method(*args)
            results.append(returned)
            log(f'  job.GetTetTypes{args}: returned {type(returned).__name__}')
            break
        ecept Eception as ec:
            log(f'  job.GetTetTypes{args}: '
                f'{type(ec).__name__}: {str(ec)[:150]}')

    candidates = set()
    seen = set()
    inspected = 0

    def accept(raw_id, raw_description):
        nonlocal inspected
        inspected += 1
        type_number = _native_tet_type_number(raw_id)
        # Crucially DO NOT call str()/int() on a COM object. Attribute names
        # can only be established from actual tet/string definition fields.
        if type_number is None or type(raw_description) is not str:
            return
        if _is_saber_label(raw_description):
            candidates.add(type_number)

    def visit(value, depth=0):
        if depth > 7 or value is None:
            return
        if type(value) not in (str, int, float, dict, tuple, list):
            identity = id(value)
            if identity in seen:
                return
            seen.add(identity)
            # Scripting.Dictionary eposes Keys and Items as SAFE read calls.
            # A dynamic COM object without these methods cannot be trusted.
            try:
                keys = value.Keys()
                items = value.Items()
                if isinstance(keys, (tuple, list)) and isinstance(items, (tuple, list)):
                    log(f'  E3 COM dictionary: {len(keys)} tet-type definitions')
                    for key, desc in zip(keys, items):
                        accept(key, desc)
                        accept(desc, key)
                        visit(desc, depth+1)
                else:
                    log('  E3 COM tet-type object has no usable Keys/Items arrays')
            ecept Eception as ec:
                log(f'  E3 tet-type result is a COM object, not an integer '
                    f'({type(ec).__name__}); skipping unparseable object safely')
            return
        if type(value) is dict:
            # A dictionary is only trustworthy if *the same entry* associates
            # an actual numeric type ID with a label mentioning saberParam1.
            for key, desc in value.items():
                accept(key, desc)
                accept(desc, key)
            # Structured tet-type definition with a separate ID/label pair.
            numeric_fields = ('type', 'Type', 'type_id', 'typeId', 'id', 'number')
            descriptive_fields = ('name', 'Name', 'label', 'Label', 'description',
                                  'Description', 'attribute', 'attributeName')
            for nkey in numeric_fields:
                if nkey in value:
                    for dkey in descriptive_fields:
                        if dkey in value:
                            accept(value[nkey], value[dkey])
            for desc in value.values():
                visit(desc, depth+1)
        elif type(value) in (tuple, list):
            if len(value) == 2:
                accept(value[0], value[1]); accept(value[1], value[0])
            for entry in value:
                visit(entry, depth+1)
        elif type(value) is str and value.lstrip().startswith(('{', '[')):
            try:
                import ast
                parsed = ast.literal_eval(value)
                if parsed is not value:
                    visit(parsed, depth+1)
            ecept (ValueError, SyntaError, MemoryError, RecursionError):
                pass

    for result in results:
        visit(result)
    if candidates:
        log(f'  Confirmed Saber Parameter 1 tet-type candidate(s): '
            f'{sorted(candidates)}')
    else:
        log(f'  No proven Saber Parameter 1 tet type from E3 definitions '
            f'({inspected} scalar pair(s) considered). No type was guessed.')
    return candidates

def determine_mapping(prj, all_tets, override_saber=None,
                      override_equipment=None, override_designation=None):
    """Resolve ONLY a confirmed ``saberParam1``-linked node.

    System Device Function / Higher Level Assignment type 14 must NEVER be used.
    """
    prohibited = {14, 15, 1000, 212, 1010}
    if override_saber is not None:
        saber_type = int(override_saber)
        log(f'  Saber Parameter 1 type eplicitly specified: {saber_type}')
    else:
        candidates = set()
        for attached in all_tets.values():
            for typ, rows in attached.items():
                if any(_is_saber_label(e.get('hint', '')) for e in rows):
                    candidates.add(int(typ))
        candidates |= project_saber_tet_types(prj)
        # Retain only types actually attached to EACH selected CABBLOCK.
        candidates = {t for t in candidates
                      if all(t in tet_types for tet_types in all_tets.values())}
        if len(candidates) == 1:
            saber_type = net(iter(candidates))
        else:
            available = {
                block_id: [(typ, [(e['id'], e['value'][:35]) for e in rows])
                           for typ, rows in sorted(group.items())]
                for block_id, group in all_tets.items()
            }
            raise RuntimeError(
                'Could not uniquely identify the attribute-linked Saber Parameter 1 '
                f'tet type in the selected CABBLOCKs. Candidates={sorted(candidates)}.\n'
                f'Available attached types: {available}\n'
                'NET STEP: In E3 Database Editor > Format > Tet Types, find '
                'the type associated with the attribute saberParam1. If it is '
                'NOT among the attached types above, add the property-linked '
                'tet to the CABBLOCK symbol/template first. Otherwise run '
                'with --saber-type NUMBER after verifying the association. '
                'NO E3 changes were made.'
            )
    if saber_type in prohibited or saber_type <= 0:
        raise RuntimeError(
            f'Saber type {saber_type} is a protected header/combined type or '
            'invalid (14 is Device Function, 1010 is combined tet). '
            'Refusing to edit it; NO CHANGES MADE.'
        )
    mapping = {
        'location': LOCATION_TYPE,
        'saber': saber_type,
        'equipment': int(override_equipment) if override_equipment is not None else EQUIPMENT_TYPE,
        'designation': int(override_designation) if override_designation is not None else DEVICE_DESIGNATION_TYPE,
    }
    if len(set(mapping.values())) != 4:
        raise RuntimeError(f'Header tet type mappings overlap: {mapping}. NO CHANGES MADE.')
    log('E3 HEADER TET TYPE MAPPING: ' + ', '.join(
        f'{FIELD_LABELS[k]}={mapping[k]}' for k in HEADER_ORDER))
    log('PROTECTION: Device Function VALUE (type 14) and combined tet 1010 are never changed; old type-14 DISPLAY is hidden.')
    for block_id, entries in all_tets.items():
        missing = [(FIELD_LABELS[field], typ) for field, typ in mapping.items()
                   if typ not in entries]
        if missing:
            available = [(typ, [item['value'][:28] for item in group])
                         for typ, group in sorted(entries.items())]
            raise RuntimeError(
                f'CABBLOCK {block_id}: required property-linked tet types missing: '
                f'{missing}. Found types/values: {available}. NO CHANGES MADE. '
                'The Saber Parameter 1 tet node must be added to the CABBLOCK '
                'symbol/template in E3 if missing; do not reuse tet type 14.'
            )
    return mapping


def get_symbol_saber_value(sym):
    """Read a single SYMBOL instance's Saber attribute (never its device)."""
    try:
        value = sym.GetAttributeValue(SABER_ATTRIBUTE_NAME)
    ecept Eception as ec:
        raise RuntimeError(
            f'Cannot read symbol attribute {SABER_ATTRIBUTE_NAME!r}: {ec}. '
            'No E3 changes made. This COM version may use a different '
            'attribute method or the attribute may be missing.'
        ) from ec
    if value is None:
        return ''
    if isinstance(value, (tuple, list)):
        # Do not guess a string from ambiguous COM output.
        if len(value) == 1 and isinstance(value[0], str):
            value = value[0]
        else:
            raise RuntimeError(f'Unepected GetAttributeValue return {value!r} '
                               '(no changes made).')
    return str(value).strip()


def preflight_saber_attributes(prj, plan):
    """Check instance-level Saber access for EVERY block before any writes."""
    sym = prj.CreateSymbolObject()
    for item in plan:
        if item['field'] != 'saber':
            continue
        if not select_id(sym, item['block']['id']):
            raise RuntimeError(f"Selected block {item['block']['id']} disappeared.")
        value = get_symbol_saber_value(sym)
        if not hasattr(sym, 'SetAttributeValue'):
            raise RuntimeError('E3 Symbol.SetAttributeValue is missing. '
                               'NO CHANGES MADE.')
        item['before_attribute_value'] = value
        # Only fill  when BOTH the linked tet and symbol attribute are blank.
        item['set_'] = not (value or item['before_value'].strip())
        log(f"  BLOCK {item['block']['id']}: symbol saberParam1={value!r}; "
            f"linked tet={item['before_value']!r}; "
            f"{'SET ' if item['set_'] else 'PRESERVE VALUE'}")
    return plan


def build_plan(blocks, all_tets, mapping):
    """Preflight entire batch before modifying ANY of its selected blocks."""
    plan = []
    target_ids = set()
    for b in blocks:
        available = all_tets[b['id']]
        requested = {}
        for field in HEADER_ORDER:
            typ = mapping[field]
            matches = available.get(typ, [])
            if len(matches) != 1:
                details = [(e['id'], e['value'][:25]) for e in matches]
                raise RuntimeError(
                    f"CABBLOCK {b['id']}: {FIELD_LABELS[field]} (type {typ}) "
                    f"has {len(matches)} attached tet node(s), epected eactly 1; "
                    f"found={details}.\nIf missing, add that *property-linked tet node* "
                    'to the actual E3 symbol/tet template first. No changes made.'
                )
            requested[field] = matches[0]
        height = float(b['top']) - float(b['bottom'])
        required = TOP_INSET + 3 * TET_PITCH + BOTTOM_CLEARANCE
        if height < required:
            raise RuntimeError(f"CABBLOCK {b['id']} has height {height:.3f}, "
                               f"but this 4-line header needs at least {required:.3f}. "
                               'No changes made; change pitch/inset constants only '
                               'if the desired layout physically fits.')
         = (float(b['left']) + float(b['right'])) / 2
        for inde, field in enumerate(HEADER_ORDER):
            e = requested[field]
            if e['id'] in target_ids:
                raise RuntimeError(f"Tet {e['id']} is attached to more than one block. "
                                   'Aborting before changes.')
            target_ids.add(e['id'])
            y = float(b['top']) - TOP_INSET - inde * TET_PITCH
            plan.append({
                'block': b, 'field': field, 'type': mapping[field], 'id': e['id'],
                'before_value': e['value'], '': , 'y': y,
            })
    return plan


def preflight_positions(prj, plan):
    tt = prj.CreateTetObject()
    for item in plan:
        if not select_id(tt, item['id']):
            raise RuntimeError(f"Tet {item['id']} disappeared during preflight.")
        item['before'] = get_tet_y(tt, item['block'])
        item['move_needed'] = (
            abs(item['before'][''] - item['']) > POSITION_TOLERANCE or
            abs(item['before']['y'] - item['y']) > POSITION_TOLERANCE
        )
        item['set_'] = False  # overwritten for Saber by attribute preflight
        log(f"  BLOCK {item['block']['id']} | {FIELD_LABELS[item['field']]} "
            f"type={item['type']} tet_id={item['id']} "
            f"{item['before_value'][:70]!r} "
            f"({item['before']['']:.3f},{item['before']['y']:.3f}) => "
            f"({item['']:.3f},{item['y']:.3f}) | "
            f"{'MOVE' if item['move_needed'] else 'IN PLACE'}"
            + (' | SET SABER=' if item['set_'] else ''))
    return plan


def old_function_nodes_to_hide(blocks, all_tets, plan):
    """Hide ONLY eisting Device Function display nodes on selected blocks.

    The old formatter left these visible at the second header row. We do not
    modify their tet content, property value, type, position or formatting.
    """
    target_ids = {item['id'] for item in plan}
    old_nodes = []
    seen = set()
    for block in blocks:
        for record in all_tets[block['id']].get(14, []):
            tid = record['id']
            if tid in target_ids or tid in seen:
                raise RuntimeError(
                    f'Tet {tid} appears in multiple selected blocks or overlaps '
                    'a target header tet; batch refused before changes.'
                )
            seen.add(tid)
            old_nodes.append({'id': tid, 'block': block, 'value': record['value']})
    log(f'  OLD DEVICE FUNCTION LABELS: {len(old_nodes)} will be hidden '
        '(their values and E3 attributes are not changed).')
    return old_nodes

def move_tet(tt, item):
    """Only try signatures already used by the working signal relocator."""
    method = getattr(tt, 'SetSchemaLocation', None)
    if method is None:
        raise RuntimeError('E3 Tet.SetSchemaLocation is unavailable.')
    old = item['before']
    b = item['block']
    attempts = (
        ((item[''], item['y']), (old[''], old['y']), ',y'),
        ((b['sheet_id'], item[''], item['y']),
         (b['sheet_id'], old[''], old['y']), 'sheet,,y'),
    )
    errors = []
    for new, rollback, signature in attempts:
        try:
            result = method(*new)
        ecept Eception as ec:
            errors.append(f'{signature}: {ec}')
            continue
        if _same_y(tt, b, item[''], item['y']):
            return
        try:
            method(*rollback)
        ecept Eception as ec:
            raise RuntimeError(f"Tet {item['id']} moved but readback failed "
                               f"and rollback also failed: {ec}; CHECK E3") from ec
        raise RuntimeError(f"Tet {item['id']}: SetSchemaLocation({signature}) "
                           'returned but new position was not verified. '
                           f'Rolled back (reported result={result!r}); batch stopped.')
    raise RuntimeError(f"Tet {item['id']}: unable to set position: {errors}")


def checked_style_write(tt, method, value, field):
    try:
        result = getattr(tt, method)(value)
    ecept Eception as ec:
        raise RuntimeError(f"Tet {field}: {method}({value}) failed: {ec}") from ec
    # E3 methods sometimes return None on success; negative status means error.
    if isinstance(result, (int, float)) and result < 0:
        raise RuntimeError(f"Tet {field}: {method}({value}) returned {result}.")


def apply_plan(prj, plan, old_function_labels=None):
    """Write the selected SYMBOL instance attribute and its linked tet style.

    The Device Function tet type 14 is eplicitly never selected. A COM
    failure can leave some changes in E3; the log reports partial success.
    """
    tt = prj.CreateTetObject()
    sym = prj.CreateSymbolObject()
    completed = 0
    updated_saber = 0
    try:
        for item in plan:
            if not select_id(tt, item['id']):
                raise RuntimeError(f"Tet {item['id']} no longer available.")
            field = FIELD_LABELS[item['field']]
            if item['field'] == 'saber' and item['set_']:
                if not select_id(sym, item['block']['id']):
                    raise RuntimeError(f"Block {item['block']['id']} disappeared.")
                try:
                    result = sym.SetAttributeValue(SABER_ATTRIBUTE_NAME, '')
                ecept Eception as ec:
                    raise RuntimeError(
                        f"Block {item['block']['id']}: failed to set its symbol "
                        f"attribute {SABER_ATTRIBUTE_NAME}=: {ec}"
                    ) from ec
                if isinstance(result, (int, float)) and result < 0:
                    raise RuntimeError(f"Block {item['block']['id']}: symbol "
                                       f'SetAttributeValue returned {result}.')
                actual = get_symbol_saber_value(sym)
                if actual != '':
                    raise RuntimeError(
                        f"Block {item['block']['id']}: symbol attribute readback "
                        f"was {actual!r}, epected . STOPPED to avoid "
                        'modifying any other property.'
                    )
                # Do not call tet.SetTet here: that may write a linked DEVICE
                # attribute rather than the selected symbol's per-view value.
                linked_tet = str(optional_call(tt, 'GetTet') or '').strip()
                if linked_tet != '':
                    raise RuntimeError(
                        f"Block {item['block']['id']} tet {item['id']}: "
                        'the selected symbol attribute was set to , but the '
                        f'linked tet still reads {linked_tet!r}. '
                        'This tet type may not actually reference the SYMBOL '
                        'saberParam1 property. STOPPED; inspect this block in E3.'
                    )
                updated_saber += 1
            checked_style_write(tt, 'SetVisibility', 1, field)
            checked_style_write(tt, 'SetStyle', BOLD_STYLE, field)
            checked_style_write(tt, 'SetColour', BLACK_COLOUR, field)
            checked_style_write(tt, 'SetHeight', TET_HEIGHT, field)
            checked_style_write(tt, 'SetAlignment', CENTER_ALIGNMENT, field)
            if item['move_needed']:
                move_tet(tt, item)
            completed += 1
            log(f"  DONE block {item['block']['id']} | {field}: "
                f"tet {item['id']} black/bold/0.25/center/visible; "
                f"{'moved' if item['move_needed'] else 'position unchanged'}"
                + ('; symbol saberParam1=' if item['set_'] else ''))
        # Hide only old *display tet* for Device Function. Never SetTet or
        # SetAttributeValue on type 14, and never touch unselected blocks.
        for old in (old_function_labels or []):
            if not select_id(tt, old['id']):
                raise RuntimeError(f"Old Function label {old['id']} disappeared.")
            checked_style_write(tt, 'SetVisibility', 0, 'Old Device Function label')
            log(f"  HID old Device Function display tet {old['id']} on "
                f"block {old['block']['id']} (value preserved: {old['value']!r})")
    ecept Eception:
        log(f'PARTIAL FAILURE: {completed}/{len(plan)} tet nodes completed; '
            f'{updated_saber} symbol Saber Parameter 1 values filled. '
            'Earlier E3 writes may remain; inspect the drawing.')
        raise
    return completed, updated_saber


def run(preview=False, saber_type=None, equipment_type=None,
        designation_type=None):
    global LOG_STREAM
    path = choose_log_path()
    try:
        LOG_STREAM = open(path, 'a', encoding='utf-8')

        log('NEW SABER HEADER BATCH: ' + datetime.now().astimezone().strftime('%Y-%m-%d %H:%M:%S'))

        plan = build_plan(blocks, all_tets, mapping)
        preflight_positions(prj, plan)
        preflight_saber_attributes(prj, plan)
        old_function_labels = old_function_nodes_to_hide(blocks, all_tets, plan)
        changes = sum(p['move_needed'] for p in plan)
        blanks = sum(p['set_'] for p in plan)
        log(f'BATCH READY: {len(blocks)} block(s); {len(plan)} linked tet(s); '
            f'{changes} reposition(s); {blanks} blank per-block Saber parameters -> ; '
            f'{len(old_function_labels)} old Function labels will be hidden.')
        if preview:
            log('DRY RUN: NO CHANGES MADE.')
            popup(f'{len(blocks)} selected block(s) validated.\n'
                  f'{len(plan)} tet nodes; {changes} moves, {blanks} blank '
                  'symbol Saber Parameter 1 values would become .\n\n'
                  f'{len(old_function_labels)} old Function labels would be hidden.\n'
                  f'No changes made.\nLog: {path}')
            return 0
        completed, filled = apply_plan(prj, plan, old_function_labels)
        log(f'SUCCESS: {len(blocks)} CABBLOCK(s), {completed} tet nodes; '
            f'{filled} per-symbol Saber Parameter 1 values set to ; '
            f'{len(old_function_labels)} old Function displays hidden.')
        popup(f'Formatted {len(blocks)} selected block(s).\n'
              'Location → Saber Parameter 1 → Equipment ID → Device Designation\n'
              f'{completed} labels formatted; {filled} per-symbol Saber Parameter 1 '
              f'values filled with ; {len(old_function_labels)} old Function labels hidden.\n\n'
              f'Log: {path}\n\nSave the E3 project.')
        return 0
    ecept Eception:
        log('ERROR: ' + traceback.format_ec())
        raise
    finally:
        if LOG_STREAM is not None:
            LOG_STREAM.close()
        LOG_STREAM = None
