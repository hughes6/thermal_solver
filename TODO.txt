"""Zuken E3.series: Excel J2 local-signal importer.

Version 1.7 CONSOLE OUTPUT / PIN-LOCAL SAFETY -- Windows / E3.series.
Open one Excel picker, then perform the import and PRINT results in a console window.
No text log is written and no other configuration window is opened.

Excel worksheet: any name; uses the matching data tab (or active tab).
E3 drawing sheet 'J2' is used only to locate the block.

Worksheet cells:
    B2    = duplicate-pin status/check (if present)
    C4    = block device designation
    C5    = connector count N
    B7:B(6+N) = connector names
    C7:C(6+N) = connector types (additional identity information)
    I:J   = connector 1: pin, local signal
    M:N   = connector 2: pin, local signal
    Q:R   = connector 3, etc., all starting on row 7
    K:L, O:P, ... are intentionally unused spacer columns

Only rows with BOTH a pin value and a nonblank local signal are imported.
The header row (Pin Number / Local Signal Name) is ignored automatically.
Blank signal cells do not clear E3 data. Duplicate pin names within a
connector are the only spreadsheet/model integrity check. E3 object ownership
is additionally checked before writes to avoid modifying the wrong object.

IMPORTANT: This version NEVER calls Pin.SetSignalName() or writes type-7
reference signal text because those can change the electrical net and
propagate into other connectors. It writes only to an explicitly existing
pin-owned attribute named Local Signal Name (or listed direct alias).
When no such field exists, it performs READ-ONLY diagnostics and prints
what E3 exposes, without changing signal connectivity.
It formats only text labels whose attribute identity and owner can be
positively verified. Unverified labels are not modified.

Test in a copy of the E3 project. COM support cannot be run on non-Windows hosts.
Requires: py -m pip install pywin32 openpyxl
Optional: Zuken's version-matched 'e3series' Python package.
"""

SHEET_NAME = 'J2'  # E3 drawing-page identifier, NOT a required Excel tab name
VERSION = '1.7-console-pin-local-safe'
FIRST_ROW = 7
FIRST_PIN_COL = 9  # I
CONNECTOR_COL_STEP = 4  # I:J, M:N, Q:R, U:V, ...
LOCAL_SIGNAL_TEXT_TYPE = 7
TEXT_HEIGHT_IN = 0.15
BOLD_STYLE = 1
NO_VALUE = -2147352572  # COM's DISP_E_PARAMNOTFOUND


class ConsoleLines(list):
    """Capture diagnostics for programmatic tests and print them immediately."""
    def __init__(self, lines=()):
        super().__init__()
        self.extend(lines)

    def append(self, line):
        line = str(line)
        super().append(line)
        print(line, flush=True)

    def extend(self, lines):
        for line in lines:
            self.append(line)

    def __iadd__(self, lines):
        self.extend(lines)
        return self


@dataclass(frozen=True)
class PinEntry:
    connector_index: int
    connector_name: str
    connector_type: str
    pin_name: str
    signal_name: str
    cell: str


@dataclass
class ImportSheet:
    workbook_path: Path
    block_name: str
    connector_count: int
    entries: list[PinEntry]
    check_b2: object
    duplicate_problems: list[str] = field(default_factory=list)
    worksheet_name: str = ''


@dataclass
class ResolvedPin:
    entry: PinEntry
    connector_id: int
    pin_id: int


def as_text(value):
    if value is None:
        return ''
    if isinstance(value, float) and value.is_integer():
        return str(int(value))
    return str(value).strip()


def device_key(value):
    """Treat a leading E3 '-' designation marker as optional, not arbitrary substrings."""
    return as_text(value).lstrip('-').casefold()


def connector_keys(value):
    """Recognize a connector with or without its block prefix ('-A1.J1')."""
    v = device_key(value)
    return {v, v.rsplit('.', 1)[-1]}


def duplicate_flag_b2(value):
    """Check commonly used B2 duplicate-pin outcomes, not unrelated statuses."""
    if isinstance(value, bool):
        return bool(value)
    if isinstance(value, (int, float)):
        return value > 0
    s = as_text(value).lower().replace('_', ' ').strip()
    if not s or s in ('ok', 'pass', 'passed', 'false', 'none', 'clear', 'no', '0'):
        return False
    if any(x in s for x in ('no duplicate', '0 duplicate', 'without duplicate', 'duplicate-free')):
        return False
    return s in ('error', 'fail', 'failed', 'true') or 'duplicate' in s


def select_data_sheet(wb):
    """Choose the Excel data tab; J2 refers to the *E3 page*, not Excel.

    Prefer a populated J2 tab if present, then the active Excel tab,
    otherwise the first tab with a designation in C4 and connector count C5.
    """
    def has_inputs(sh):
        if not as_text(sh['C4'].value):
            return False
        try:
            n = float(sh['C5'].value)
            return n >= 1 and n.is_integer()
        except (TypeError, ValueError):
            return False

    matches = [sh for sh in wb.worksheets if has_inputs(sh)]
    if not matches:
        tabs = ', '.join(repr(sh.title) for sh in wb.worksheets)
        raise ValueError(
            'Could not find an Excel worksheet with the device designation '
            'in C4 and connector count in C5. Excel tab names found: '
            + (tabs or '(none)')
        )
    for sh in matches:
        if sh.title.casefold() == SHEET_NAME.casefold():
            return sh
    if wb.active in matches:
        return wb.active
    return matches[0]


def parse_workbook(filename: Path) -> ImportSheet:
    """Read Excel values without modifying the workbook or recalculating it."""
    from openpyxl import load_workbook
    wb = load_workbook(str(filename), read_only=False, data_only=True)
    try:
        sh = select_data_sheet(wb)
        block = as_text(sh['C4'].value)
        if not block:
            raise ValueError(f'{sh.title}!C4 (device designation) is empty.')
        raw_count = sh['C5'].value
        try:
            connector_count = int(raw_count)
            if connector_count < 1 or float(raw_count) != connector_count:
                raise ValueError()
        except (ValueError, TypeError):
            raise ValueError(f'{sh.title}!C5 must contain the number of connectors.') from None

        entries = []
        problems = []
        for ci in range(connector_count):
            connector_name = as_text(sh.cell(FIRST_ROW + ci, 2).value)
            connector_type = as_text(sh.cell(FIRST_ROW + ci, 3).value)
            # Pin/signal boxes are separated by two blank columns: I:J, M:N, Q:R, ...
            pin_col = FIRST_PIN_COL + CONNECTOR_COL_STEP * ci
            signal_col = pin_col + 1
            pin_col_letter = sh.cell(FIRST_ROW, pin_col).column_letter
            signal_col_letter = sh.cell(FIRST_ROW, signal_col).column_letter
            if not connector_name:
                # A missing connector name is a matching problem, not a model check.
                problems.append(f'Connector {ci + 1}: blank name at B{FIRST_ROW + ci}.')
            seen_pins = {}
            # Respect data even if some pin rows are intentionally left blank.
            # max_row is set by the actual workbook, not by the declared pin count.
            for row in range(FIRST_ROW, sh.max_row + 1):
                pin = as_text(sh.cell(row, pin_col).value)
                signal = as_text(sh.cell(row, signal_col).value)
                # Templates commonly place headings directly in I7/J7, M7/N7, ...
                # Neither the headings nor their formatting should become an E3 pin.
                if pin.casefold().replace(' ', '') in ('pinnumber', 'pinname', 'pin#') \
                        and signal.casefold().replace(' ', '') in ('localsignalname', 'localsignal', 'signalname', 'signal'):
                    continue
                if pin:
                    # Check duplicates even where a local signal is blank;
                    # this is the one requested model/spreadsheet check.
                    if pin.casefold() in seen_pins:
                        problems.append(f'DUPLICATE pin {pin!r} in connector {connector_name!r} '
                                        f'at {pin_col_letter}{seen_pins[pin.casefold()]} '
                                        f'and {pin_col_letter}{row}.')
                        continue
                    seen_pins[pin.casefold()] = row
                if not signal:
                    continue  # never clear or reformat an unspecified E3 pin
                if not pin:
                    problems.append(f'{signal_col_letter}{row}: signal without a pin number.')
                    continue
                entries.append(PinEntry(ci, connector_name, connector_type,
                                        pin, signal, f'{signal_col_letter}{row}'))

        sheet_data = ImportSheet(Path(filename), block, connector_count,
                                 entries, sh['B2'].value, problems, sh.title)
        # A B2 formula may have no cached value when Excel has not calculated.
        # The actual duplicate scan above remains authoritative in that case.
        if duplicate_flag_b2(sheet_data.check_b2):
            sheet_data.duplicate_problems.insert(
                0, f'{sh.title}!B2 reports duplicate pins: ' + repr(sheet_data.check_b2))
        return sheet_data
    finally:
        wb.close()


def require_no_duplicate_pins(data):
    dup = [p for p in data.duplicate_problems if 'DUPLICATE' in p.upper()]
    if dup:
        raise ValueError('Duplicate pin check failed:\n' + '\n'.join(dup))


def flatten_positive_ids(obj):
    if isinstance(obj, (list, tuple)):
        vals = []
        for x in obj:
            vals.extend(flatten_positive_ids(x))
        return vals
    if type(obj) is int and 0 < obj < 2147483647:
        return [obj]
    return []


def split_com_result(result):
    # E3 methods generally return (number_of_ids, ID-array).
    if isinstance(result, (tuple, list)) and result and type(result[0]) is int:
        return result[0], flatten_positive_ids(result[1:])
    return (result if type(result) is int else None), []


def com_ids(obj, method_name, text_type=None):
    """Read E3 arrays using both official-wrapper and direct pywin32 conventions."""
    method = getattr(obj, method_name, None)
    if method is None:
        return []
    variants = [()] if text_type is None else [(text_type,), (None, text_type), ([], text_type)]
    for args in variants:
        try:
            raw = method(*args)
            count, ids = split_com_result(raw)
            if ids:
                return list(dict.fromkeys(ids[:count] if count and count > 0 else ids))
            if isinstance(raw, (tuple, list)) and len(raw) > 1 and all(type(i) is int and i > 0 for i in raw):
                return list(dict.fromkeys(raw))
        except Exception:
            pass
    # The Python COM interface sometimes requires an explicit output array.
    try:
        import pythoncom
        import win32com.client as wc
    except ImportError:
        return []
    for variant_type, seed in (
        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, []),
    ):
        try:
            arr = wc.VARIANT(variant_type, seed)
            result = method(arr) if text_type is None else method(arr, text_type)
            _, returned_ids = split_com_result(result)
            ids = returned_ids or flatten_positive_ids(arr.value)
            if ids:
                return list(dict.fromkeys(ids))
        except Exception:
            pass
    return []


def related_ids(obj, candidates, text_type=None):
    for method in candidates:
        ids = com_ids(obj, method, text_type=text_type)
        if ids:
            return ids
    return []


def get_scalar(obj, methods):
    for name in methods:
        method = getattr(obj, name, None)
        if method is None:
            continue
        try:
            result = method()
        except Exception:
            continue
        if isinstance(result, (str, float, int)) and result != NO_VALUE:
            return result
    return None


def object_api_names(obj, filter_words=('signal', 'text', 'pin', 'connector', 'block', 'device', 'sheet')):
    """Inspect the COM type library safely, also works with the official wrapper."""
    found = set()
    try:
        found.update(str(n) for n in dir(obj) if not str(n).startswith('_'))
    except Exception:
        pass
    try:
        ti = obj._oleobj_.GetTypeInfo()
        ta = ti.GetTypeAttr()
        for i in range(ta.cFuncs):
            names = ti.GetNames(ti.GetFuncDesc(i).memid)
            if names:
                found.add(str(names[0]))
    except Exception:
        pass
    return sorted(x for x in found if any(w in x.lower() for w in filter_words))


def diagnostic_ids(obj, method_names):
    """Read-only COM enumeration diagnostics, limited to short summaries."""
    results = []
    for method_name in method_names:
        try:
            m = getattr(obj, method_name, None)
            if not callable(m):
                results.append(f'  {method_name}: NOT EXPOSED')
                continue
            try:
                raw = m()
                typ = type(raw).__name__
                if isinstance(raw, (list, tuple)):
                    summary = f'length={len(raw)}, first={repr(raw[:3])[:180]}'
                else:
                    summary = repr(raw)[:280]
                results.append(f'  {method_name}(): {typ} {summary}')
            except Exception as ex:
                results.append(f'  {method_name}(): {type(ex).__name__}: {str(ex)[:180]}')
            ids = com_ids(obj, method_name)
            results.append(f'  {method_name} parsed IDs={len(ids)}, first={ids[:8]}')
        except Exception as ex:
            results.append(f'  {method_name} probe FAILED: {type(ex).__name__}: {ex}')
    return results


def pin_diagnostics(job, pin_id):
    """Show available pin signal APIs, and local-signal (#7) text ownership."""
    notes = []
    try:
        pin = make_object(job, 'Pin', pin_id)
        notes.append('PIN COM APIs: ' + str(object_api_names(pin, ('signal', 'text', 'name', 'symbol'))[:90]))
        for method in ('GetName','GetPinName','GetLocalSignalName','GetSignalName','GetSignal','GetSignalId'):
            fn = getattr(pin, method, None)
            if not callable(fn):
                notes.append(f'  {method}: not exposed')
            else:
                try:
                    notes.append(f'  {method}() = {repr(fn())[:180]}')
                except Exception as ex:
                    notes.append(f'  {method}(): {type(ex).__name__}: {str(ex)[:150]}')
        notes.extend(diagnostic_ids(pin, ('GetTextIds','GetSymbolIds')))
        ids = related_ids(pin, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
        notes.append(f'  GetTextIds(type=7) parsed IDs={ids[:30]}')
    except Exception as ex:
        notes.append(f'pin diagnostic error: {type(ex).__name__}: {ex}')
    return notes


def make_object(job, kind, object_id):
    creator = getattr(job, f'Create{kind}Object', None)
    if not callable(creator):
        raise RuntimeError(f'E3 does not expose Create{kind}Object().')
    obj = creator()
    result = obj.SetId(int(object_id))
    if result is False or (type(result) is int and result < 0):
        raise RuntimeError(f'Could not resolve {kind} ID {object_id}: SetId returned {result!r}.')
    return obj


# E3 block connectors may return their *parent block* designation in GetName().
# The individual connector designation can instead be exposed by other
# read-only device/pin methods or by the .CONNECTOR_NAME pin attribute.
CONNECTOR_NAME_GETTERS = (
    'GetDeviceDesignation', 'GetConnectorDesignation', 'GetInternalDeviceDesignation',
    'GetInternalDesignation', 'GetBlockConnectorDesignation',
    'GetConnectorName', 'GetBlockConnectorName', 'GetInternalName',
    'GetDeviceName', 'GetFullName', 'GetDesignation', 'GetName',
)
CONNECTOR_ATTRIBUTES = (
    '.CONNECTOR_NAME', 'CONNECTOR_NAME', 'Internal Device Designation',
)


def read_string_getters(obj, getters):
    """Read *all* available name-like getters, not just the first one."""
    result = {}
    for name in getters:
        method = getattr(obj, name, None)
        if not callable(method):
            continue
        try:
            value = method()
        except Exception:
            continue
        if isinstance(value, str) and value.strip():
            result[name] = value.strip()
    return result


def read_attribute_names(obj):
    """Read the E3 internal connector designation without modifying attributes."""
    values = {}
    for meth in ('GetAttributeValue', 'GetAttribute', 'GetPinAttributeValue'):
        method = getattr(obj, meth, None)
        if not callable(method):
            continue
        for attname in CONNECTOR_ATTRIBUTES:
            try:
                value = method(attname)
            except Exception:
                continue
            if isinstance(value, str) and value.strip():
                values[f'{meth}({attname})'] = value.strip()
    return values


def extended_connector_aliases(job, connector_id, connector_obj, log):
    """Collect name evidence from the connector device and its own pins.

    We intentionally do NOT use Excel connector order as an E3 connector index:
    GetConnectorIds order is not guaranteed to match the spreadsheet.
    """
    names = read_string_getters(connector_obj, CONNECTOR_NAME_GETTERS)
    names.update(read_attribute_names(connector_obj))
    log.append(f'  CONNECTOR ID {connector_id} DEVICE NAME GETTERS: {names!r}')
    avail = object_api_names(connector_obj, ('name', 'designat', 'connector', 'attribute'))
    log.append(f'  CONNECTOR ID {connector_id} AVAILABLE NAME/ATTRIBUTE APIS: {avail[:110]!r}')

    # Inspect only pins E3 explicitly associates with this connector object.
    pin_ids = related_ids(connector_obj, ('GetPinIds', 'GetAllPinIds'))
    log.append(f'  CONNECTOR ID {connector_id} PIN IDS ({len(pin_ids)}): {pin_ids[:35]!r}')
    pin_attr_hits = {}
    for pin_id in pin_ids[:12]:
        try:
            pin = make_object(job, 'Pin', pin_id)
            values = read_attribute_names(pin)
            pin_names = read_string_getters(pin, (
                'GetInternalDeviceDesignation', 'GetConnectorName',
                'GetBlockConnectorName', 'GetDeviceDesignation'))
            if values or pin_names:
                pin_attr_hits[pin_id] = {'attr': values, 'getters': pin_names}
            for k, v in {**values, **pin_names}.items():
                names[f'pin:{pin_id}:{k}'] = v
        except Exception as exc:
            log.append(f'    PIN ID {pin_id}: name probe failed: {type(exc).__name__}: {exc}')
    log.append(f'  CONNECTOR ID {connector_id} PIN-NAME EVIDENCE: {pin_attr_hits!r}')
    return names


class E3Project:
    def __init__(self):
        self.backend = ''
        try:
            import e3series
            application = e3series.Application()
            job = application.CreateJobObject()
            if job.GetId():
                self.application = application
                self.job = job
                self.backend = 'official e3series wrapper'
                return
        except Exception:
            pass
        import win32com.client as win32
        self.application = win32.Dispatch('CT.Application')
        self.job = self.application.CreateJobObject()
        self.backend = 'pywin32 COM'
        if not self.job.GetId():
            raise RuntimeError('E3.series is not running with an open project.')

    def sheet_name(self, sheet_id):
        try:
            sheet = make_object(self.job, 'Sheet', sheet_id)
            return as_text(get_scalar(sheet, ('GetName', 'GetSheetName')))
        except Exception:
            return ''

    def on_j2(self, device):
        """Optional E3 drawing-sheet scope: applies only if a page J2 exists."""
        sids = related_ids(device, ('GetSymbolIds', 'GetAllSymbolIds'))
        for sid in sids:
            try:
                sym = make_object(self.job, 'Symbol', sid)
                sheet_id = get_scalar(sym, ('GetSheetId',))
                if sheet_id and self.sheet_name(sheet_id).casefold() == SHEET_NAME.casefold():
                    return True
            except Exception:
                continue
        return False

    def j2_page_exists(self):
        for sid in related_ids(self.job, ('GetSheetIds',)):
            if self.sheet_name(sid).casefold() == SHEET_NAME.casefold():
                return True
        return False

    def devices(self):
        ids = related_ids(self.job, ('GetDeviceIds', 'GetAllDeviceIds', 'GetBlockIds'))
        return [(i, make_object(self.job, 'Device', i)) for i in ids]

    def block_device(self, designation, log):
        hits = []
        inventory = self.devices()
        for did, obj in inventory:
            name = as_text(get_scalar(obj, ('GetName', 'GetDeviceDesignation', 'GetDeviceName')))
            if device_key(name) == device_key(designation):
                hits.append((did, obj))
        log.append(f'ALL E3 DEVICES ENUMERATED: {len(inventory)}; matching C4 {designation!r}')
        if not hits:
            log.append('DEVICE EXAMPLES (first 40):')
            for did, ob in inventory[:40]:
                log.append(f'  id {did} name={get_scalar(ob, ("GetName", "GetDeviceDesignation", "GetDeviceName"))!r}')
            log.append('JOB DEVICE-ID METHOD DIAGNOSTICS:')
            log.extend(diagnostic_ids(self.job, ('GetDeviceIds','GetAllDeviceIds','GetBlockIds')))
            raise LookupError(f'No E3 block/device matches C4 = {designation!r}.')
        if self.j2_page_exists():
            scoped = [(did, o) for did, o in hits if self.on_j2(o)]
            if not scoped:
                for did, o in hits[:10]:
                    log.append(f'BLOCK found ID={did}, but on_j2=False; symbols={related_ids(o, ("GetSymbolIds",))[:25]}')
                log.append('E3 J2 PAGE EXISTS BUT MATCHED BLOCK NOT ON IT. Sheet ids/names:')
                for sid in related_ids(self.job, ('GetSheetIds',))[:40]:
                    log.append(f'  sheet ID={sid}: {self.sheet_name(sid)!r}')
                raise LookupError(f'{designation!r} was found, but not on E3 drawing sheet J2.')
            hits = scoped
            log.append('E3 drawing sheet J2 found; restricting block match to that page.')
        else:
            log.append('No E3 drawing sheet named J2 found; matching the block by C4 designation.')
        if len(hits) != 1:
            raise LookupError(f'C4 matches {len(hits)} E3 devices; use a unique device designation.')
        log.append(f'BLOCK {designation!r} => E3 device ID {hits[0][0]}')
        return hits[0]

    def connectors(self, block_device, log=None):
        ids = related_ids(block_device, ('GetConnectorIds', 'GetBlockConnectorIds', 'GetSubDeviceIds'))
        if not ids:
            for sym_id in related_ids(block_device, ('GetSymbolIds',)):
                try:
                    sym = make_object(self.job, 'Symbol', sym_id)
                except Exception:
                    continue
                ids.extend(related_ids(sym, ('GetConnectorIds', 'GetBlockConnectorIds')))
        found = []
        self.connector_evidence = {}
        for cid in dict.fromkeys(ids):
            try:
                obj = make_object(self.job, 'Device', cid)
                # Do not use GetName() as the sole source of connector identity.
                names = extended_connector_aliases(self.job, cid, obj, log) if log is not None \
                    else {**read_string_getters(obj, CONNECTOR_NAME_GETTERS),
                          **read_attribute_names(obj)}
                self.connector_evidence[cid] = names
                display = names.get('GetDeviceDesignation') or names.get('GetConnectorName') or \
                    names.get('GetName') or '(unnamed)'
                code = as_text(get_scalar(obj, ('GetComponentName', 'GetComponentCode',
                                                'GetPartNumber', 'GetTypeName')))
                found.append((cid, obj, display, code))
            except Exception as ex:
                if log is not None:
                    log.append(f'  CONNECTOR ID {cid}: could not inspect: {ex}')
        return found

    def connector(self, choices, connector_name, connector_type):
        """Match a uniquely identified connector by any verified internal designation.

        Never guess by GetConnectorIds() order or component type alone.
        """
        requested = device_key(connector_name)
        matches = []
        for item in choices:
            cid = item[0]
            methods = self.connector_evidence.get(cid, {})
            for method_name, value in methods.items():
                if requested and requested in connector_keys(value):
                    matches.append((item, method_name, value))
                    break
        if len(matches) == 1:
            self.last_connector_match = matches[0][1:]
            return matches[0][0]
        if not matches:
            raise LookupError(
                f'Connector {connector_name!r} not found using connector device designation '
                'or .CONNECTOR_NAME pin attributes. Refer to connector evidence in log.')
        raise LookupError(
            f'Connector {connector_name!r} matched {len(matches)} different E3 connectors '
            f'({[m[0][0] for m in matches]}); refusing to guess.')

    def pin(self, connector_obj, pin_name):
        pin_ids = related_ids(connector_obj, ('GetPinIds', 'GetAllPinIds'))
        hits = []
        for pid in pin_ids:
            try:
                p = make_object(self.job, 'Pin', pid)
                value = get_scalar(p, ('GetName', 'GetPinName'))
                if as_text(value).casefold() == pin_name.casefold():
                    hits.append(pid)
            except Exception:
                continue
        if not hits:
            raise LookupError(f'Pin {pin_name!r} not found on connector.')
        if len(hits) > 1:
            raise LookupError(f'Pin {pin_name!r} is ambiguous on connector.')
        return hits[0]


def resolve(data, e3, log):
    _, block = e3.block_device(data.block_name, log)
    candidates = e3.connectors(block, log)
    log.append(f'Associated connector devices found: {len(candidates)}')
    log.append('BLOCK CONNECTOR DISCOVERY:')
    log.extend(diagnostic_ids(block, ('GetConnectorIds','GetBlockConnectorIds','GetSubDeviceIds','GetPinIds','GetSymbolIds')))
    log.append('ASSOCIATED CONNECTOR SAMPLES (first 40):')
    for cid, _, cname, ctype in candidates[:40]:
        log.append(f'  id={cid}, name={cname!r}, type={ctype!r}')
    if not candidates:
        log.append('WARNING: No connector devices were discovered inside the block; '
                   'this is likely why no assignments will occur.')
    indexed = {}
    for i in range(data.connector_count):
        examples = [e for e in data.entries if e.connector_index == i]
        if not examples:
            continue
        sample = examples[0]
        try:
            choice = e3.connector(candidates, sample.connector_name, sample.connector_type)
            indexed[i] = choice
            log.append(f'CONNECTOR {sample.connector_name!r} (type {sample.connector_type!r}) '
                       f'=> {choice[2]!r}, ID {choice[0]}, component {choice[3]!r}; '
                       f'MATCH EVIDENCE={e3.last_connector_match!r}')
        except Exception as exc:
            log.append('SKIP CONNECTOR: ' + str(exc))
            log.append(f'  SEARCH KEY: {sample.connector_name!r}; available connector names: '
                       f'{[c[2] for c in candidates[:40]]!r}')
    plans = []
    for entry in data.entries:
        choice = indexed.get(entry.connector_index)
        if not choice:
            log.append(f'SKIP {entry.cell}: no unique connector match for {entry.connector_name!r}')
            continue
        try:
            pin_id = e3.pin(choice[1], entry.pin_name)
            plans.append(ResolvedPin(entry, choice[0], pin_id))
            log.append(f'PLAN {entry.cell}: {entry.connector_name}/{entry.pin_name} '
                       f'-> {entry.signal_name!r} [pin ID {pin_id}]')
        except Exception as exc:
            log.append(f'SKIP {entry.cell}: {exc}')
            try:
                conn = choice[1]
                log.append('  CONNECTOR PIN-ID DIAGNOSTICS:')
                log.extend(diagnostic_ids(conn, ('GetPinIds', 'GetAllPinIds')))
                pin_ids = related_ids(conn, ('GetPinIds','GetAllPinIds'))
                log.append('  FIRST 35 PINS ON THIS CONNECTOR: ' + repr([
                    (pid, get_scalar(make_object(e3.job, 'Pin', pid), ('GetName','GetPinName')))
                    for pid in pin_ids[:35]]))
            except Exception as ex:
                log.append(f'  PIN INVENTORY DIAGNOSTIC FAILED: {ex}')
    return plans


# Pin-local attribute names are distinct from E3's GetSignalName / SetSignalName
# electrical signal property.  Only EXISTING pin attributes can be written.
PIN_LOCAL_ATTRIBUTE_NAMES = (
    'Local Signal Name', 'Local Signal', '.LOCAL_SIGNAL_NAME', 'LOCAL_SIGNAL_NAME',
)


def get_signal(pin):
    """Read ONLY: electrical network signal, for before/after safety checks."""
    value = get_scalar(pin, ('GetSignalName',))
    return value if isinstance(value, str) else None


def local_attribute_field(pin):
    """Identify a pin-owned local label using affirmative E3 HasAttribute evidence."""
    has = getattr(pin, 'HasAttribute', None)
    getter = getattr(pin, 'GetAttributeValue', None)
    setter = getattr(pin, 'SetAttributeValue', None)
    if not (callable(has) and callable(getter) and callable(setter)):
        return None, 'No full HasAttribute/GetAttributeValue/SetAttributeValue pin API'
    probes = []
    for attr in PIN_LOCAL_ATTRIBUTE_NAMES:
        try:
            ret = has(attr)
            probes.append((attr, repr(ret)))
            # A missing / unsupported result must not authorize writing.
            if ret is True or (type(ret) is int and ret > 0):
                before = getter(attr)
                if isinstance(before, str):
                    return attr, f'attribute exists: {attr!r}, previous={before!r}'
                return None, f'{attr!r} exists but GetAttributeValue returned {before!r}'
        except Exception as exc:
            probes.append((attr, type(exc).__name__))
    return None, 'No verified pin-local attribute; HasAttribute probes=' + repr(probes)


def get_local_attribute(pin, attr):
    value = pin.GetAttributeValue(attr)
    return value if isinstance(value, str) else None


def set_pin_local_attribute(pin, attr, target_name):
    """Write only one verified pin attribute. NEVER use SetSignalName."""
    before_network = get_signal(pin)
    before = get_local_attribute(pin, attr)
    if before == target_name:
        return 'already assigned', before_network
    result = pin.SetAttributeValue(attr, target_name)
    if result is False or (type(result) is int and result <= 0):
        raise RuntimeError(f'SetAttributeValue({attr!r}) returned {result!r}')
    after = get_local_attribute(pin, attr)
    if after != target_name:
        raise RuntimeError(f'Pin attribute readback failed: expected {target_name!r}, got {after!r}')
    if get_signal(pin) != before_network:
        # Undo ONLY the local attribute; refuse further writes.
        try:
            pin.SetAttributeValue(attr, before)
        except Exception:
            pass
        raise RuntimeError('Electrical GetSignalName changed! Pin-local attribute rolled back where possible.')
    return f'SetAttributeValue({attr}) VERIFIED', before_network


def read_pin_local_evidence(job, pin_id):
    """Diagnostics for user-controlled pin/connector property naming."""
    p = make_object(job, 'Pin', pin_id)
    out = [f'PIN {pin_id} name={get_scalar(p, ("GetName",))!r}, electrical signal={get_signal(p)!r}']
    out.append('  Attribute API: ' + repr(object_api_names(p, ('attribute','signal'))[:80]))
    out.extend(diagnostic_ids(p, ('GetAttributeIds',)))
    field, reason = local_attribute_field(p)
    out.append(f'  PIN LOCAL FIELD: {field!r}: {reason}')
    text_ids = related_ids(p, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
    out.append(f'  TYPE-7 TEXT IDS: {text_ids!r}')
    for tid in text_ids[:8]:
        try:
            txt = make_object(job, 'Text', tid)
            out.append(f'  TEXT {tid} METHODS: ' + repr(object_api_names(txt, ('text','type','name','owner','pin','attr','visible'))[:110]))
            vals = {}
            for meth in ('GetText','GetTextString','GetString','GetValue','GetType','GetTextType',
                         'GetAttributeName','GetPinId','GetOwnerPinId','GetSymbolId','GetSheetId',
                         'GetVisible','GetHeight','GetStyle'):
                v = get_scalar(txt, (meth,))
                if v is not None:
                    vals[meth] = v
            out.append(f'  TEXT {tid} VALUES: {vals!r}')
        except Exception as ex:
            out.append(f'  TEXT {tid} diagnostic failed: {ex}')
    return out


def format_verified_local_attribute_text(job, pin_id, attr):
    """Do not style type-7 NET text as if it were a local pin attribute.

    A separate pin-owned attribute text may exist, identified by its GetAttributeName.
    It is safe to format only a positively identified attribute-associated text node.
    """
    pin = make_object(job, 'Pin', pin_id)
    all_ids = related_ids(pin, ('GetTextIds',))
    # Filter by exact pin-attribute binding; if not exposed, do not guess.
    trusted=[]
    for tid in all_ids:
        try:
            t = make_object(job, 'Text', tid)
            name = get_scalar(t, ('GetAttributeName',))
            if as_text(name).casefold() != attr.casefold():
                continue
            owner = get_scalar(t, ('GetOwnerPinId','GetPinId'))
            if owner is not None and owner != pin_id:
                continue
            trusted.append((tid,t))
        except Exception:
            continue
    if not trusted:
        return 'NO VERIFIED PIN-LOCAL ATTRIBUTE TEXT. Attribute was set; visual text NOT verified/formatted. Type-7 net text was deliberately untouched.'
    result=[]
    for tid,t in trusted:
        try:
            t.SetHeight(TEXT_HEIGHT_IN)
            t.SetStyle(BOLD_STYLE)
            visible = set_visible(t)
            h=get_scalar(t, ('GetHeight',))
            sty=get_scalar(t, ('GetStyle',))
            correct=(isinstance(h,(int,float)) and abs(h-TEXT_HEIGHT_IN)<0.0001 and sty==BOLD_STYLE)
            result.append((tid,correct,visible))
        except Exception as ex:
            result.append((tid,'FAILED',str(ex)))
    return 'PIN ATTRIBUTE TEXT RESULTS: '+repr(result)


def uniquely_owned_pin_symbols(job, pin_id, connector_id=None):
    """Find placed connector symbols which represent only the requested pin."""
    pin = make_object(job, 'Pin', pin_id)
    symbol_ids = related_ids(pin, ('GetSymbolIds',))
    sid = get_scalar(pin, ('GetSymbolId', 'GetOwnerSymbolId'))
    if type(sid) is int and sid > 0:
        symbol_ids.append(sid)
    if connector_id:
        # This path was needed in the previous Match Properties utility:
        # connector device -> placed pin symbol -> signal text type 7.
        conn = make_object(job, 'Device', connector_id)
        symbol_ids.extend(related_ids(conn, ('GetSymbolIds',)))
    found = []
    for symbol_id in dict.fromkeys(symbol_ids):
        try:
            sym = make_object(job, 'Symbol', symbol_id)
            owner_pins = related_ids(sym, ('GetPinIds',))
            single = get_scalar(sym, ('GetPinId', 'GetOwnerPinId'))
            if not owner_pins and type(single) is int and single > 0:
                owner_pins = [single]
            if owner_pins == [pin_id]:
                found.append(sym)
        except Exception:
            continue
    return found


def pin_label_text_ids(job, pin_id, connector_id=None):
    """Get type-7 text positively owned by this pin, never nearby unowned text."""
    pin = make_object(job, 'Pin', pin_id)
    ids = related_ids(pin, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
    if ids:
        return ids
    for sym in uniquely_owned_pin_symbols(job, pin_id, connector_id):
        ids.extend(related_ids(sym, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE))
    return list(dict.fromkeys(ids))


def make_pin_label_visible(job, pin_id, connector_id=None):
    """Try owner-level visibility, including placed one-pin connector symbols."""
    pin = make_object(job, 'Pin', pin_id)
    show = getattr(pin, 'SetTextVisibility', None)
    if callable(show):
        try:
            show(LOCAL_SIGNAL_TEXT_TYPE, 1)
        except Exception:
            pass
    for sym in uniquely_owned_pin_symbols(job, pin_id, connector_id):
        show = getattr(sym, 'SetTextVisibility', None)
        if callable(show):
            try:
                show(LOCAL_SIGNAL_TEXT_TYPE, 1)
            except Exception:
                pass


def set_visible(obj):
    """Visibility varies by E3 release; invoke only known text visibility names."""
    pairs = (('SetVisible', 'GetVisible'),
             ('SetVisibility', 'GetVisibility'),
             ('SetTextVisible', 'GetTextVisible'),
             ('SetDisplayed', 'GetDisplayed'))
    for setter, getter in pairs:
        method = getattr(obj, setter, None)
        if not callable(method):
            continue
        try:
            result = method(1)
            check = get_scalar(obj, (getter,))
            if result is not False and (check is None or check in (1, True)):
                return setter
        except Exception:
            continue
    return None


def format_local_signal(job, pin_id, connector_id=None):
    pin = make_object(job, 'Pin', pin_id)
    # Request visibility BEFORE collecting labels: hidden labels may not be
    # returned by all E3 builds until SetTextVisibility has been called.
    make_pin_label_visible(job, pin_id, connector_id)
    text_ids = pin_label_text_ids(job, pin_id, connector_id)
    if not text_ids:
        return 'NO LOCAL-SIGNAL TEXT OBJECT: signal set, text formatting/visibility not verified'
    good = 0
    issues = []
    for tid in text_ids:
        try:
            text = make_object(job, 'Text', tid)
            text.SetHeight(TEXT_HEIGHT_IN)
            text.SetStyle(BOLD_STYLE)
            height = get_scalar(text, ('GetHeight',))
            style = get_scalar(text, ('GetStyle',))
            formatted = (isinstance(height, (float, int)) and
                         abs(float(height) - TEXT_HEIGHT_IN) <= 0.0001 and style == BOLD_STYLE)
            visibility_api = set_visible(text)
            if not visibility_api:
                # Some releases put type visibility on the parent pin.
                show = getattr(pin, 'SetTextVisibility', None)
                if callable(show):
                    try:
                        show(LOCAL_SIGNAL_TEXT_TYPE, 1)
                        visibility_api = 'Pin.SetTextVisibility'
                    except Exception:
                        pass
            if formatted and visibility_api:
                good += 1
            else:
                issues.append(f'text {tid}: format={formatted}, '
                              f'visibility method={visibility_api or "NOT FOUND"}')
        except Exception as exc:
            issues.append(f'text {tid}: {exc}')
    if good == len(text_ids):
        return f'OK: {good} text label(s), bold 0.15 in, made visible'
    return f'PARTIAL ({good}/{len(text_ids)} labels): ' + '; '.join(issues)


def perform(path):
    """Apply immediately; print all decisions to stdout, never write files."""
    lines = ConsoleLines([f'ZUKEN LOCAL SIGNAL IMPORTER v{VERSION}',
             f'TIME: {datetime.now().isoformat(timespec="seconds")}',
             f'EXCEL FILE: {path}', ''])
    summary = {'defined':0, 'resolved':0, 'signal_accepted':0,
               'verified':0, 'formatted':0, 'failed':0, 'skipped':0,
               'fatal':None}
    com_initialized = False
    try:
        data = parse_workbook(path)
        lines += [f'EXCEL WORKSHEET SELECTED: {data.worksheet_name}',
                  f'E3 DRAWING PAGE FOR IDENTIFICATION: {SHEET_NAME}',
                  f'DEVICE DESIGNATION (C4): {data.block_name!r}',
                  f'NUMBER OF CONNECTORS (C5): {data.connector_count}',
                  f'DUPLICATE-PIN STATUS (B2): {data.check_b2!r}',
                  f'SIGNAL ASSIGNMENTS READ: {len(data.entries)}']
        summary['defined'] = len(data.entries)
        for ci in range(data.connector_count):
            grouped = [e for e in data.entries if e.connector_index == ci]
            pc = FIRST_PIN_COL + CONNECTOR_COL_STEP * ci
            from openpyxl.utils import get_column_letter
            cols = f'{get_column_letter(pc)}:{get_column_letter(pc+1)}'
            lines.append(f'EXCEL CONNECTOR #{ci+1} {cols}: {len(grouped)} nonblank assignments; '
                         f'name (B{FIRST_ROW+ci})={grouped[0].connector_name if grouped else "(no nonblank signals)"!r}')
        for e in data.entries[:100]:
            lines.append(f'  INPUT {e.cell}: connector={e.connector_name!r}, type={e.connector_type!r}, '
                         f'pin={e.pin_name!r} -> signal={e.signal_name!r}')
        if len(data.entries) > 100:
            lines.append(f'  ... {len(data.entries)-100} additional assignments omitted from this preview')
        for prob in data.duplicate_problems:
            lines.append('EXCEL NOTICE: '+prob)
        require_no_duplicate_pins(data)
        if not data.entries:
            raise RuntimeError('No pin/signal pairs were read from the workbook. '
                               'Check starting row 7 and pairs I:J, M:N, Q:R. '
                               'Unspecified/blank signal cells are intentionally skipped.')

        import pythoncom
        pythoncom.CoInitialize()
        com_initialized = True
        e3 = E3Project()
        lines.append(f'E3 CONNECTION: {e3.backend}; job ID={e3.job.GetId()!r}')
        lines.append('E3 JOB AVAILABLE RELEVANT METHODS: '
                     + repr(object_api_names(e3.job, ('device','sheet','block','symbol','pin'))[:120]))
        lines.append('E3 JOB SHEET ENUMERATION:')
        lines.extend(diagnostic_ids(e3.job, ('GetSheetIds',)))
        sheet_ids = related_ids(e3.job, ('GetSheetIds',))
        lines.append('E3 SHEET NAMES (first 40): '
                     +repr([(sid,e3.sheet_name(sid)) for sid in sheet_ids[:40]]))
        lines.append('E3 JOB DEVICE ENUMERATION:')
        lines.extend(diagnostic_ids(e3.job, ('GetDeviceIds','GetAllDeviceIds','GetBlockIds')))

        plans = resolve(data, e3, lines)
        summary['resolved'] = len(plans)
        summary['skipped'] = summary['defined'] - summary['resolved']
        lines.append(f'RESOLVED {len(plans)} / {len(data.entries)} PIN ASSIGNMENTS')
        if not plans:
            lines.append('NO PINS RESOLVED, so no E3 writes attempted. Check the SKIP and method reports above. '
                         'If connector names still all show PNL1, copy this output; it includes the '
                         'per-connector API list and .CONNECTOR_NAME attribute inspection.')
        else:
            lines.append('E3 COM PIN API SAMPLE FOR FIRST RESOLVED PIN:')
            lines.extend(pin_diagnostics(e3.job, plans[0].pin_id))

        # Preflight entire input before any write; never partly import a workbook
        # if E3 exposes only electrical net signals rather than pin-local attrs.
        lines.append('PIN-LOCAL ATTRIBUTE PREFLIGHT (NO WRITES YET):')
        missing=[]
        valid=[]
        for plan in plans:
            pin=make_object(e3.job, 'Pin', plan.pin_id)
            field, reason=local_attribute_field(pin)
            lines.append(f'  {plan.entry.connector_name}/{plan.entry.pin_name}, ID {plan.pin_id}: {reason}')
            if field:
                valid.append((plan,field))
            else:
                missing.append(plan)
                lines.extend(read_pin_local_evidence(e3.job,plan.pin_id))
        if missing:
            summary['skipped'] += len(missing)
            lines.append('SAFETY STOP: one or more requested pins have no VERIFIED PIN-LOCAL field.')
            lines.append('NO WRITES attempted to prevent shared-signal changes on J1A/J1B.')
            lines.append('Do not substitute SetSignalName or Text.SetText: E3 may propagate those to the whole net.')
            lines.append('Please copy this console output and share the exact pin property screenshot if available.')
        else:
            # Also report same-numbered pins on other connectors for context.
            _, block=e3.block_device(data.block_name, lines)
            others=e3.connectors(block)
            lines.append('COUNTERPART PIN READBACK BEFORE WRITES:')
            other_sigs={}
            other_attrs={}
            monitored_fields={field for _,field in valid}
            selected={v.pin_id for v,_ in valid}
            for cid,conn,cn,ct in others:
                for pid in related_ids(conn, ('GetPinIds',)):
                    if pid in selected: continue
                    try:
                        op=make_object(e3.job,'Pin',pid)
                        key=(cid,pid)
                        other_sigs[key]=get_signal(op)
                        for f in monitored_fields:
                            try:
                                existing, why=local_attribute_field(op)
                                # Capture both the attribute's existence and value:
                                # even creation of an attribute on an unselected pin is a side effect.
                                getter=getattr(op,'GetAttributeValue',None)
                                raw=getter(f) if callable(getter) else None
                                other_attrs[(cid,pid,f)]=(existing==f,raw)
                            except Exception:
                                other_attrs[(cid,pid,f)]=('unreadable',None)
                    except Exception: pass
            lines.append(f'  Non-target pin electrical signals snapshotted: {len(other_sigs)}')
            lines.append(f'  Non-target pin local attributes snapshotted: {len(other_attrs)}')
            lines.append('WRITE RESULTS (PIN ATTRIBUTES ONLY, NEVER NETWORK):')
            for plan,field in valid:
                e=plan.entry
                lines.append(f'ATTEMPT {e.cell}: connector={e.connector_name!r}, pin={e.pin_name!r}, ID {plan.pin_id}, attribute={field!r}, value={e.signal_name!r}')
                try:
                    pin=make_object(e3.job,'Pin',plan.pin_id)
                    original_local_value=get_local_attribute(pin,field)
                    result,net_before=set_pin_local_attribute(pin,field,e.signal_name)
                    # Check cross-connector propagation immediately, BEFORE another write.
                    changed_now=[]
                    for (peer_cid,peer_pid,peer_field),before_state in other_attrs.items():
                        try:
                            peer=make_object(e3.job,'Pin',peer_pid)
                            peer_existing,_=local_attribute_field(peer)
                            getter=getattr(peer,'GetAttributeValue',None)
                            peer_raw=getter(peer_field) if callable(getter) else None
                            now=(peer_existing==peer_field,peer_raw)
                            if now != before_state:
                                changed_now.append((peer_cid,peer_pid,peer_field,before_state,now))
                        except Exception as exc:
                            changed_now.append((peer_cid,peer_pid,peer_field,'read failed',str(exc)))
                    if changed_now:
                        # Best-effort rollback of this assignment. Because E3 may
                        # share attribute objects, project backup remains essential.
                        try:
                            pin.SetAttributeValue(field,original_local_value)
                        except Exception:
                            pass
                        lines.append('  CRITICAL: other connector pin attributes changed: '+repr(changed_now[:8]))
                        raise RuntimeError('PIN-LOCAL WRITE PROPAGATED TO NON-TARGET PINS; '
                                           'no more assignments will be attempted. Restore your test copy.')
                    summary['signal_accepted']+=1
                    summary['verified']+=1
                    lines.append(f'  ATTRIBUTE SETTER: {result}; net unchanged: {get_signal(pin)==net_before}; peers unchanged.')
                    fmt=format_verified_local_attribute_text(e3.job,plan.pin_id,field)
                    lines.append(f'  LOCAL ATTRIBUTE TEXT FORMAT: {fmt}')
                    if fmt.startswith('PIN ATTRIBUTE TEXT RESULTS:') and 'False' not in fmt and 'FAILED' not in fmt and 'None' not in fmt:
                        summary['formatted']+=1
                except Exception as ex:
                    summary['failed']+=1
                    lines.append(f'  WRITE FAILED: {type(ex).__name__}: {ex}')
                    break  # avoid compounding a bad write
            changed_attrs=[]
            for (cid,pid,field),before in other_attrs.items():
                try:
                    op=make_object(e3.job,'Pin',pid)
                    existing,_=local_attribute_field(op)
                    getter=getattr(op,'GetAttributeValue',None)
                    raw=getter(field) if callable(getter) else None
                    now=(existing==field,raw)
                    if now != before:
                        changed_attrs.append((cid,pid,field,before,now))
                except Exception as exc:
                    changed_attrs.append((cid,pid,field,'read failed',str(exc)))
            lines.append(f'COUNTERPART PIN ATTRIBUTES CHANGED: {changed_attrs[:20]!r}')
            if changed_attrs:
                summary['fatal']='A non-target PIN ATTRIBUTE changed. STOP and restore your project copy.'
            changed=[]
            for (cid,pid),before in other_sigs.items():
                try:
                    now=get_signal(make_object(e3.job,'Pin',pid))
                    if now!=before: changed.append((cid,pid,before,now))
                except Exception as exc:
                    changed.append((cid,pid,'read failed',str(exc)))
            lines.append(f'COUNTERPART ELECTRICAL SIGNALS CHANGED: {changed[:15]!r}')
            if changed:
                summary['fatal']='A non-target electrical signal changed. STOP: restore your test copy.'
        del e3
    except Exception as ex:
        summary['fatal'] = f'{type(ex).__name__}: {ex}'
        lines.append('FATAL ERROR: '+summary['fatal'])
        lines.append(traceback.format_exc())
    finally:
        if com_initialized:
            try:
                gc.collect()
                pythoncom.CoUninitialize()
            except Exception:
                pass
        lines += ['', '============= FINAL SUMMARY =============',
                  f'Excel defined assignments:       {summary["defined"]}',
                  f'Resolved pin IDs:                {summary["resolved"]}',
                  f'Pin-local attributes written:     {summary["signal_accepted"]}',
                  f'Pin-local value readback verified:{summary["verified"]}',
                  f'Pin-local text formatting verified:{summary["formatted"]}',
                  f'Skipped or not matched:         {summary["skipped"]}',
                  f'Write failures:                 {summary["failed"]}',
                  f'Fatal problem:                  {summary["fatal"] or "none"}',
                                    'IMPORTANT: GetSignalName/SetSignalName are ELECTRICAL NET signals; this version NEVER writes them.',
                  '=========================================']
    return summary


def run_once():
    """Ask for Excel once, then print all outcomes directly. No log file."""
    import tkinter as tk
    from tkinter import filedialog
    root = tk.Tk()
    root.withdraw()
    try:
        root.attributes('-topmost', True)
    except Exception:
        pass
    try:
        print('ZUKEN E3 LOCAL SIGNAL IMPORTER v1.7', flush=True)
        print('Choose the Excel workbook; results will print in this window.', flush=True)
        filename = filedialog.askopenfilename(
            parent=root,
            title='Select Excel file for E3 local signal names',
            filetypes=[('Excel workbooks', '*.xlsx *.xlsm'), ('All files', '*.*')],
        )
        if filename:
            result = perform(Path(filename))
            if result['fatal'] or result['failed'] or result['signal_accepted']==0:
                print('\nRESULT: NO VERIFIED COMPLETED IMPORT / REVIEW ERRORS ABOVE.', flush=True)
            elif result['formatted'] < result['signal_accepted']:
                print('\nRESULT: Pin-local fields updated, but not all display text could be verified/formatted.', flush=True)
            else:
                print('\nRESULT: Pin-local fields updated and formatting verified.', flush=True)
        else:
            print('Cancelled: no workbook selected.', flush=True)
    except Exception:
        print('UNEXPECTED ERROR:\n'+traceback.format_exc(), flush=True)
    finally:
        root.destroy()


def main():
    # The E3 Tools host may have no visible stdout. Launch a separate Python
    # process with its own console and release E3's Tools-menu selection host.
    if sys.platform == 'win32' and '--child' not in sys.argv:
        exe = Path(sys.executable)
        if exe.name.lower() == 'pythonw.exe':
            normal_exe = exe.with_name('python.exe')
            if normal_exe.exists():
                exe = normal_exe
        try:
            subprocess.Popen(
                [str(exe), str(Path(__file__).resolve()), '--child'],
                creationflags=subprocess.CREATE_NEW_CONSOLE | subprocess.CREATE_NEW_PROCESS_GROUP,
                cwd=str(Path(__file__).resolve().parent),
                close_fds=True,
            )
        except Exception as exc:
            # E3 may embed its own interpreter; expose launch failure visibly.
            try:
                from tkinter import Tk, messagebox
                r=Tk(); r.withdraw()
                messagebox.showerror('E3 importer launch failed', str(exc), parent=r)
                r.destroy()
            except Exception:
                print('Could not open importer console:',exc,flush=True)
        return
    try:
        run_once()
    finally:
        if '--child' in sys.argv:
            print('\nPress ENTER to close this console.', flush=True)
            try:
                input()
            except (EOFError, OSError, KeyboardInterrupt):
                pass


if __name__ == '__main__':
    main()
