# VALUE HELPERS

def clean_excel_value(value):
    if value is None:
        return ""

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def normalize_pin_name(value):
    return clean_excel_value(value).upper()


def normalize_connector_token(value):
    text = clean_excel_value(value).upper().replace(" ", "")

    if text == "":
        return ""

    root_prefix = CONNECTOR_ROOT_NAME.upper() + "."

    if text.startswith(root_prefix):
        text = text[len(root_prefix):]

    if "#" in text:
        text = text.split("#", 1)[0]

    match = re.fullmatch(r"P?0*(\d+)([A-Z]*)", text)

    if match:
        number = int(match.group(1))
        suffix = match.group(2)
        return f"P{number}{suffix}"

    return text


def connector_base_name(connector_token):
    connector = normalize_connector_token(connector_token)

    if connector == "":
        return ""

    return f"{CONNECTOR_ROOT_NAME}.{connector}"


def cell_has_bar(value):
    return "|" in clean_excel_value(value)


#====================================================================================================================================
# YES / NO SETTING
#====================================================================================================================================
def parse_yes_no(value, setting_name):
    text = clean_excel_value(value).upper()

    if text == "":
        print("WARNING:", setting_name, "IS BLANK - DEFAULTING TO YES")
        return True

    if text in ["YES", "Y", "TRUE", "1"]:
        return True

    if text in ["NO", "N", "FALSE", "0"]:
        return False

    raise Exception(f"{setting_name} must contain YES or NO. Current value: {value}")

# SPLIT INTO CONNECTOR SECTIONS
#====================================================================================================================================
def split_into_sections(rows):
    sections = []
    current = []

    for row in rows:
        if not current:
            current = [row]
            continue

        previous = current[-1]
        connector_changed = row["connector"] != previous["connector"]
        row_gap = row["excel_row"] != previous["excel_row"] + 1

        if connector_changed or row_gap:
            sections.append(current)
            current = [row]
        else:
            current.append(row)

    if current:
        sections.append(current)

    return sections


#====================================================================================================================================
# CONTIGUOUS MARKED RANGES
#====================================================================================================================================
def true_ranges(rows, key):
    ranges = []
    start = None

    for index, row in enumerate(rows):
        marked = bool(row[key])

        if marked and start is None:
            start = index

        if not marked and start is not None:
            ranges.append((start, index - 1))
            start = None

    if start is not None:
        ranges.append((start, len(rows) - 1))

    return ranges


def range_pins(rows, start, end):
    pins = []

    for index in range(start, end + 1):
        pins.append(rows[index]["pin"])

    return pins


def range_excel_rows(rows, start, end):
    excel_rows = []

    for index in range(start, end + 1):
        excel_rows.append(rows[index]["excel_row"])

    return excel_rows



#====================================================================================================================================
# BUILD TWIST / INNER-SHIELD JOBS FOR ONE CONNECTOR SECTION
#
# INNER SHIELD RULE:
#   Inner SHLD does NOT create a standalone shield.
#   It only places Shield_Demo around a twist run when every wire in
#   that twist run is marked in the inner-SHLD column.
#
# OUTER SHIELDS ARE BUILT SEPARATELY ACROSS THE WHOLE SIDE SO THEY MAY
# CROSS CONNECTOR CHANGES (EXAMPLE: LEFT P1 -> P3 -> P4).
#====================================================================================================================================
def build_section_jobs(rows):
    jobs = []
    warnings = []

    if not rows:
        return jobs, warnings

    side = rows[0]["side"]
    connector = rows[0]["connector"]

    if connector == "":
        warnings.append(f"{side}: rows {rows[0]['excel_row']}-{rows[-1]['excel_row']} have pins but no connector")
        return jobs, warnings

    conflicts = []

    for row in rows:
        if row["twist_a"] and row["twist_b"]:
            conflicts.append(row["excel_row"])

    if conflicts:
        message = f"{side} {connector}: same wire marked in both twist columns on Excel rows {conflicts}"

        if STRICT_TWIST_VALIDATION:
            raise Exception(message)

        warnings.append(message)

    twist_jobs = []

    for key, twist_column in [("twist_a", "A"), ("twist_b", "B")]:
        ranges = true_ranges(rows, key)

        for start, end in ranges:
            pins = range_pins(rows, start, end)
            excel_rows = range_excel_rows(rows, start, end)

            if len(pins) < 2:
                warnings.append(
                    f"{side} {connector}: twist {twist_column} only contains one wire "
                    f"on Excel row {excel_rows[0]}"
                )
                continue

            job = {
                "type": "twist",
                "side": side,
                "connector": connector,
                "twist_column": twist_column,
                "start": start,
                "end": end,
                "pins": pins,
                "excel_rows": excel_rows,
                "symbol_name": TWIST_SYMBOL_NAME
            }

            twist_jobs.append(job)
            jobs.append(job)

    # Inner SHLD only wraps an existing twist. It never makes its own span.
    for twist_job in twist_jobs:
        indexes = list(range(twist_job["start"], twist_job["end"] + 1))
        inner_marks = [bool(rows[index]["inner_shield"]) for index in indexes]

        if all(inner_marks):
            jobs.append({
                "type": "shield_inner",
                "side": side,
                "connector": connector,
                "twist_column": twist_job["twist_column"],
                "pins": list(twist_job["pins"]),
                "excel_rows": list(twist_job["excel_rows"]),
                "symbol_name": SHIELD_SYMBOL_NAME
            })
        elif any(inner_marks):
            warnings.append(
                f"{side} {connector}: inner SHLD partially covers twist "
                f"{twist_job['twist_column']} on Excel rows {twist_job['excel_rows']}; "
                f"inner shield ignored because the full twist must be marked"
            )

    return jobs, warnings


#====================================================================================================================================
# BUILD OUTER-SHIELD JOBS ACROSS AN ENTIRE SIDE
#
# Connector changes DO NOT break an outer shield.
# Blank Excel row gaps or an unmarked outer-SHLD row DO break it.
#====================================================================================================================================
def build_outer_shield_jobs(rows):
    jobs = []

    if not rows:
        return jobs

    side = rows[0]["side"]
    start = None

    def append_job(start_index, end_index):
        covered_rows = rows[start_index:end_index + 1]

        connectors = []
        for row in covered_rows:
            connector = row["connector"]
            if connector and connector not in connectors:
                connectors.append(connector)

        jobs.append({
            "type": "shield_outer",
            "side": side,
            "connector": "/".join(connectors),
            "connectors": connectors,
            "rows": covered_rows,
            "pins": [row["pin"] for row in covered_rows],
            "excel_rows": [row["excel_row"] for row in covered_rows],
            "twist_column": None,
            "symbol_name": SHIELD_SYMBOL_NAME
        })

    for index, row in enumerate(rows):
        marked = bool(row["outer_shield"])

        if start is not None:
            previous = rows[index - 1]
            row_gap = row["excel_row"] != previous["excel_row"] + 1

            if row_gap or not marked:
                append_job(start, index - 1)
                start = None

        if marked and start is None:
            start = index

    if start is not None:
        append_job(start, len(rows) - 1)

    return jobs

# E3 ID EXTRACTOR
#====================================================================================================================================
def extract_e3_ids(result):
    if result is None:
        return []

    if isinstance(result, int):
        if result > 0:
            return [result]

        return []

    if isinstance(result, list):
        output = []

        for value in result:
            if value is not None:
                try:
                    if int(value) > 0:
                        output.append(int(value))
                except Exception:
                    pass

        return output

    if not isinstance(result, tuple):
        return []

    if len(result) == 0:
        return []

    if len(result) >= 2:
        values = result[1]

        if values is None:
            return []

        if isinstance(values, int):
            if values > 0:
                return [values]

            return []

        try:
            return [int(value) for value in values if value is not None and int(value) > 0]
        except Exception:
            pass

    output = []

    for value in result:
        try:
            if int(value) > 0:
                output.append(int(value))
        except Exception:
            pass

    return output

# READ PIN SCHEMATIC LOCATION
#====================================================================================================================================
def get_pin_location(pin, pin_id):
    pin.SetId(pin_id)

    xloc = [0.0]
    yloc = [0.0]
    grid = [""]
    column = [""]
    row = [""]

    try:
        result = pin.GetSchemaLocation(xloc, yloc, grid, column, row)
    except Exception:
        return None

    if not isinstance(result, tuple):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None

    if sheet_id <= 0:
        return None

    return {"sheet_id": sheet_id, "x": x, "y": y}

# LOAD DEVICE PIN MAP
#====================================================================================================================================
def load_device_pin_map(dev, pin, device_id):
    dev.SetId(device_id)

    try:
        info = dev.GetAllPinIds(None)
    except Exception:
        return {}

    pin_ids = extract_e3_ids(info)
    pin_map = {}

    for pin_id in pin_ids:
        pin.SetId(pin_id)

        try:
            pin_name = normalize_pin_name(pin.GetName())
        except Exception:
            continue

        location = get_pin_location(pin, pin_id)

        if pin_name not in pin_map:
            pin_map[pin_name] = []

        if location is not None:
            record = {"pin_id": pin_id, "sheet_id": location["sheet_id"], "x": location["x"], "y": location["y"]}
            pin_map[pin_name].append(record)

    return pin_map

def get_same_name_assembly_devices(dev, base_device_id, expected_name):
    found_ids = []

    dev.SetId(base_device_id)

    try:
        assembly_id = dev.GetAssemblyId()
    except Exception:
        return found_ids

    if assembly_id is None:
        return found_ids

    try:
        assembly_id = int(assembly_id)
    except Exception:
        return found_ids

    if assembly_id <= 0:
        return found_ids

    dev.SetId(assembly_id)

    try:
        device_info = dev.GetDeviceIds(None)
    except Exception:
        return found_ids

    device_ids = extract_e3_ids(device_info)

    for device_id in device_ids:
        dev.SetId(device_id)

        try:
            name = clean_excel_value(dev.GetName())
        except Exception:
            continue

        if name.upper() == expected_name.upper():
            found_ids.append(device_id)

    return found_ids


#====================================================================================================================================
# COLLECT CONNECTOR CANDIDATES
#====================================================================================================================================
def collect_connector_candidate_ids(connector_token, all_devices, dev):
    expected_name = connector_base_name(connector_token)
    base_records = find_base_connector_devices(connector_token, all_devices)

    if not base_records:
        raise Exception(f"Could not find base E3 connector {expected_name}")

    candidate_sources = {}

    for record in base_records:
        base_id = record["device_id"]

        candidate_sources[base_id] = "BASE"

        view_ids = get_device_view_ids(dev, base_id)

        for view_id in view_ids:
            candidate_sources[view_id] = "VIEW"

        child_ids = get_child_device_ids(dev, base_id)

        for child_id in child_ids:
            if child_id not in candidate_sources:
                candidate_sources[child_id] = "CHILD"

        assembly_ids = get_same_name_assembly_devices(dev, base_id, expected_name)

        for assembly_device_id in assembly_ids:
            if assembly_device_id not in candidate_sources:
                candidate_sources[assembly_device_id] = "ASSEMBLY"

    return candidate_sources


#====================================================================================================================================
# ANALYZE CANDIDATE VIEW
#====================================================================================================================================
def analyze_candidate(required_pins, pin_map):
    sheets = {}

    for pin_name in required_pins:
        locations = pin_map.get(pin_name, [])

        for location in locations:
            sheet_id = location["sheet_id"]

            if sheet_id not in sheets:
                sheets[sheet_id] = {"pins": set(), "xs": [], "ys": []}

            sheets[sheet_id]["pins"].add(pin_name)
            sheets[sheet_id]["xs"].append(location["x"])
            sheets[sheet_id]["ys"].append(location["y"])

    if not sheets:
        return None

    results = []

    for sheet_id, data in sheets.items():
        placed_count = len(data["pins"])
        unique_x_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["xs"])
        unique_y_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["ys"])
        unique_x_count = len(unique_x_values)
        unique_y_count = len(unique_y_values)
        x_span = max(data["xs"]) - min(data["xs"]) if data["xs"] else 0.0
        y_span = max(data["ys"]) - min(data["ys"]) if data["ys"] else 0.0
        has_real_y_spread = y_span >= SCHEMATIC_MIN_Y_SPAN and unique_y_count > 1
        result = {"sheet_id": sheet_id, "placed_count": placed_count, "unique_x_count": unique_x_count, "unique_y_count": unique_y_count, "x_span": x_span, "y_span": y_span, "has_real_y_spread": has_real_y_spread}

        results.append(result)

    results.sort(key=lambda item: (item["has_real_y_spread"], item["placed_count"], item["unique_y_count"], item["y_span"]), reverse=True)

    return results[0]

        if candidate_id not in cache:
            try:
                cache[candidate_id] = load_device_pin_map(dev, pin, candidate_id)
            except Exception as error:
                print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "PIN MAP FAILED:", repr(error))
                continue

        pin_map = cache[candidate_id]
        analysis = analyze_candidate(required_pins, pin_map)

        if analysis is None:
            print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "NO PLACED REQUIRED PINS")
            continue

        print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "SHEET:", analysis["sheet_id"], "PLACED:", analysis["placed_count"], "UNIQUE X:", analysis["unique_x_count"], "UNIQUE Y:", analysis["unique_y_count"], "X SPAN:", analysis["x_span"], "Y SPAN:", analysis["y_span"])

        candidate = {"device_id": candidate_id, "name": candidate_name, "source": source, "pin_map": pin_map, "analysis": analysis}

        candidates.append(candidate)

    if not candidates:
        raise Exception(f"No usable E3 device/view candidates found for {expected_name}")

    candidates.sort(key=lambda item: (item["analysis"]["has_real_y_spread"], item["analysis"]["placed_count"], item["analysis"]["unique_y_count"], item["analysis"]["y_span"]), reverse=True)

    best = candidates[0]

    if not best["analysis"]["has_real_y_spread"]:
        raise Exception(f"Could not locate the split schematic view for {expected_name}")

    sheet_id = best["analysis"]["sheet_id"]
    missing_pins = []

    for pin_name in required_pins:
        locations = best["pin_map"].get(pin_name, [])
        found_on_sheet = False

        for location in locations:
            if location["sheet_id"] == sheet_id:
                found_on_sheet = True
                break

        if not found_on_sheet:
            missing_pins.append(pin_name)

    if missing_pins:
        raise Exception(f"Selected schematic view for {expected_name} is missing pins on sheet {sheet_id}: {missing_pins}")

    print("")
    print("SELECTED SCHEMATIC VIEW ID:", best["device_id"])
    print("SELECTED SOURCE:", best["source"])
    print("SELECTED NAME:", best["name"])
    print("SELECTED SHEET:", sheet_id)
    print("SELECTED PLACED PIN COUNT:", best["analysis"]["placed_count"])
    print("SELECTED UNIQUE Y COUNT:", best["analysis"]["unique_y_count"])
    print("SELECTED Y SPAN:", best["analysis"]["y_span"])
    print("============================================================")

    record = {"device_id": best["device_id"], "name": best["name"]}

    return record, best["pin_map"], sheet_id

def pin_location_on_sheet(pin_map, pin_name, sheet_id):

#====================================================================================================================================
# JOB X OFFSET
#====================================================================================================================================
def job_offset(job, twist_offset_settings):
    offset_enabled = twist_offset_settings[job["side"]]

    if job["type"] == "twist":
        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if job["type"] == "shield_inner":
        if job["twist_column"] is None:
            return INNER_SHIELD_ONLY_OFFSET

        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if not offset_enabled:
        return OFFSET_NO_OUTER_SHIELD

    return OFFSET_YES_OUTER_SHIELD


#====================================================================================================================================
# DRAW ORDER
#
# TWISTS FIRST.
#
# Pair shields second.
# Outer shields last.
#====================================================================================================================================
def job_priority(job):
    if job["type"] == "twist":
        return 0

    if job["type"] == "shield_inner":
        return 1

    return 2


#====================================================================================================================================
# CALCULATE FINAL SYMBOL PLACEMENT
#====================================================================================================================================
def calculate_placement(job, pin_map, sheet_id, twist_offset_settings):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} has no schematic location on sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    if top_y - bottom_y < SCHEMATIC_MIN_Y_SPAN and len(job["pins"]) > 1:
        raise Exception(f"{job['side']} {job['connector']} {job['type']}: pins {job['pins']} all have effectively the same Y coordinate")

    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job, twist_offset_settings)
    x = anchor_x + direction * offset

    if job["type"] == "twist":
        y = bottom_y - TWIST_BOTTOM_OVERLAP
        length = None
    else:
        y = bottom_y - SHIELD_BOTTOM_OVERLAP
        length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

        if length < MIN_SYMBOL_LENGTH:
            length = MIN_SYMBOL_LENGTH

    placement = {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y,
        "direction": direction,
        "offset_enabled": twist_offset_settings[job["side"]]
    }

    return placement



#====================================================================================================================================
# CALCULATE OUTER SHIELD PLACEMENT ACROSS ONE OR MORE CONNECTORS
#
# Each Excel row is resolved through the connector named on that row.
# This allows one outer shield to span, for example, LEFT P1 + P3 + P4.
# All covered connector views must resolve to the same E3 sheet.
#====================================================================================================================================
def calculate_outer_shield_placement(job, all_devices, dev, pin, cache, twist_offset_settings):
    resolved_by_connector = {}
    sheet_ids = set()
    locations = []

    rows_by_connector = {}

    for row in job["rows"]:
        connector = row["connector"]

        if connector == "":
            raise Exception(
                f"{job['side']} outer shield: Excel row {row['excel_row']} has no connector"
            )

        rows_by_connector.setdefault(connector, []).append(row)

    for connector, connector_rows in rows_by_connector.items():
        required_pins = list(dict.fromkeys(row["pin"] for row in connector_rows))

        device_record, pin_map, sheet_id = resolve_connector_schematic_view(
            connector,
            required_pins,
            all_devices,
            dev,
            pin,
            cache
        )

        resolved_by_connector[connector] = {
            "device_record": device_record,
            "pin_map": pin_map,
            "sheet_id": sheet_id
        }

        sheet_ids.add(sheet_id)

    if len(sheet_ids) != 1:
        details = ", ".join(
            f"{connector}=sheet {data['sheet_id']}"
            for connector, data in resolved_by_connector.items()
        )
        raise Exception(
            f"{job['side']} outer shield spans connectors on different E3 sheets: {details}"
        )

    sheet_id = next(iter(sheet_ids))

    for row in job["rows"]:
        connector = row["connector"]
        pin_name = row["pin"]
        pin_map = resolved_by_connector[connector]["pin_map"]

        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(
                f"{job['side']} outer shield: {connector} pin {pin_name} "
                f"has no schematic location on sheet {sheet_id}"
            )

        locations.append(location)

    if not locations:
        raise Exception(f"{job['side']} outer shield has no resolved schematic locations")

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    direction = 1.0 if job["side"] == "LEFT" else -1.0

    if twist_offset_settings[job["side"]]:
        offset = OFFSET_YES_OUTER_SHIELD
    else:
        offset = OFFSET_NO_OUTER_SHIELD

    x = anchor_x + direction * offset
    y = bottom_y - SHIELD_BOTTOM_OVERLAP
    length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

    if length < MIN_SYMBOL_LENGTH:
        length = MIN_SYMBOL_LENGTH

    return {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y,
        "direction": direction,
        "offset_enabled": twist_offset_settings[job["side"]]
    }


#====================================================================================================================================
# SAFE DISPLAY LENGTH READ
#====================================================================================================================================
def get_display_length_safe(sym, symbol_id):
    sym.SetId(symbol_id)

    try:
        return float(sym.GetDisplayLength())
    except Exception:
        return None


#====================================================================================================================================
# TRY TO SIZE SHIELD AT CURRENT LOCATION
#====================================================================================================================================
def try_size_shield(sym, symbol_id, requested_length):
    # EXACT KNOWN-GOOD E3 SEQUENCE:
    #   Place -> SetId(placed id) -> GetDisplayLength ->
    #   SetDisplayLength -> SetId(placed id) -> GetDisplayLength
    sym.SetId(symbol_id)

    try:
        before = float(sym.GetDisplayLength())
    except Exception:
        before = None

    print("SHIELD LENGTH BEFORE:", before)

    try:
        stretch_result = sym.SetDisplayLength(float(requested_length))
    except Exception as error:
        stretch_result = None
        print("SHIELD STRETCH EXCEPTION:", repr(error))

    print("SHIELD STRETCH RESULT:", stretch_result)

    # Re-bind exactly like the standalone test that successfully resized.
    sym.SetId(symbol_id)
    time.sleep(STRETCH_DELAY_SECONDS)

    try:
        after = float(sym.GetDisplayLength())
    except Exception:
        after = None

    print("SHIELD REQUESTED LENGTH:", requested_length)
    print("SHIELD LENGTH AFTER:", after)

    return after is not None and abs(after - float(requested_length)) <= 0.001


def shield_stage_y_values(requested_length):
    # Shield_Demo grows upward from its insertion point.
    # 10.0 at Y=22.0 was proven to resize successfully in the earlier test.
    # For longer shields, lower the staging Y enough to keep the full symbol on-sheet.
    safe_y = min(
        SHIELD_STAGE_DEFAULT_Y,
        SHIELD_STAGE_TOP_LIMIT - float(requested_length) - SHIELD_STAGE_MARGIN
    )
    safe_y = max(0.75, safe_y)

    candidates = [
        safe_y,
        max(0.75, safe_y - 0.50),
        min(SHIELD_STAGE_DEFAULT_Y, safe_y + 0.50),
        max(0.75, safe_y - 1.00),
        min(SHIELD_STAGE_DEFAULT_Y, safe_y + 1.00),
    ]

    output = []
    for value in candidates:
        value = round(float(value), 3)
        if value not in output:
            output.append(value)

    return output


#====================================================================================================================================
# PLACE TWIST
#
# NO STRETCH.
#
# E3/database decides display length automatically.
#====================================================================================================================================
def place_twist(sym, job, placement):
    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("TWIST LOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not load library twist {job['symbol_name']}")

    print("TWIST TARGET X:", placement["x"])
    print("TWIST TARGET Y:", placement["y"])

    placed_id = sym.Place(placement["sheet_id"], placement["x"], placement["y"], 0, 1.0, 0)

    print("TWIST PLACE RESULT:", placed_id)

    if placed_id == 0:
        raise Exception(f"Could not place library twist {job['symbol_name']} at X={placement['x']} Y={placement['y']}")

    time.sleep(PLACEMENT_DELAY_SECONDS)

    sym.SetId(placed_id)

    try:
        placed_name = sym.GetName()
    except Exception:
        placed_name = "UNKNOWN"

    try:
        symbol_type = sym.GetSymbolType()
    except Exception:
        symbol_type = "UNKNOWN"

    try:
        automatic_length = float(sym.GetDisplayLength())
    except Exception:
        automatic_length = None

    print("TWIST PLACED ID:", placed_id)
    print("TWIST PLACED NAME:", placed_name)
    print("TWIST SYMBOL TYPE:", symbol_type)
    print("TWIST AUTOMATIC DISPLAY LENGTH:", automatic_length)
    print("TWIST LEFT AT E3 DATABASE AUTO LENGTH")

    return placed_id


#====================================================================================================================================
# PLACE SHIELD
#
# PROCESS:
#
# 1. Load Shield_Demo.
# 2. Search staging coordinates until the symbol can be placed.
# 3. Try SetDisplayLength at that staging position.
# 4. If resize fails, move SAME shield to another staging location.
# 5. Once correct size is reached, move SAME shield to final cable X/Y.
#
# No duplicate shields are intentionally created while searching.
#====================================================================================================================================
def place_shield(prj, job, placement):
    requested_length = float(placement["length"])
    final_x = float(placement["x"])
    final_y = float(placement["y"])

    print("SHIELD FINAL X:", final_x)
    print("SHIELD FINAL Y:", final_y)
    print("SHIELD REQUESTED LENGTH:", requested_length)

    stage_y_values = shield_stage_y_values(requested_length)
    print("SHIELD STAGE X SEARCH:", SHIELD_STAGE_X_VALUES)
    print("SHIELD STAGE Y SEARCH:", stage_y_values)

    successful_sym = None
    successful_id = None
    successful_stage_x = None
    successful_stage_y = None

    # IMPORTANT: create a FRESH E3 symbol object for every shield attempt.
    # This reproduces the earlier standalone test rather than reusing the
    # symbol object that just loaded/placed twist symbols.
    for stage_y in stage_y_values:
        if successful_id is not None:
            break

        for stage_x in SHIELD_STAGE_X_VALUES:
            print("")
            print("TRYING SHIELD STAGE X:", stage_x, "Y:", stage_y)

            shield_sym = prj.CreateSymbolObject()

            load_result = shield_sym.Load(job["symbol_name"], SYMBOL_VERSION)
            print("SHIELD LOAD RESULT:", load_result)

            if load_result == 0:
                try:
                    del shield_sym
                except Exception:
                    pass
                continue

            # Exact working sequence: LOAD -> PLACE -> SETID -> STRETCH.
            placed_id = shield_sym.Place(
                placement["sheet_id"],
                float(stage_x),
                float(stage_y),
                0,
                1.0,
                0
            )

            print("SHIELD STAGE PLACE RESULT:", placed_id)

            if placed_id == 0:
                try:
                    del shield_sym
                except Exception:
                    pass
                time.sleep(PLACEMENT_DELAY_SECONDS)
                continue

            time.sleep(PLACEMENT_DELAY_SECONDS)

            set_id_result = shield_sym.SetId(placed_id)
            print("SHIELD SET ID RESULT:", set_id_result)
            print("SHIELD STAGED ID:", placed_id)

            if try_size_shield(shield_sym, placed_id, requested_length):
                successful_sym = shield_sym
                successful_id = placed_id
                successful_stage_x = stage_x
                successful_stage_y = stage_y
                print("SHIELD RESIZE SUCCESS")
                break

            print("SHIELD RESIZE FAILED AT THIS STAGE")

            # Remove the failed test symbol before trying again.
            try:
                shield_sym.SetId(placed_id)
                delete_result = shield_sym.Delete()
                print("FAILED STAGE SHIELD DELETE RESULT:", delete_result)
            except Exception as error:
                print("FAILED STAGE SHIELD DELETE EXCEPTION:", repr(error))

            try:
                del shield_sym
            except Exception:
                pass

            time.sleep(PLACEMENT_DELAY_SECONDS)

    if successful_id is None or successful_sym is None:
        raise Exception(
            f"{job['symbol_name']} could not resize to {requested_length}. "
            f"The script used the same Load -> Place -> SetId -> SetDisplayLength "
            f"sequence as the earlier working E3 test."
        )

    print("")
    print("SHIELD STAGING COMPLETE")
    print("SHIELD STAGED ID:", successful_id)
    print("SHIELD STAGED X:", successful_stage_x)
    print("SHIELD STAGED Y:", successful_stage_y)
    print("SHIELD STAGED LENGTH:", successful_sym.GetDisplayLength())

    # Move the already-sized placed symbol to the final cable location.
    successful_sym.SetId(successful_id)
    final_move_result = successful_sym.Place(
        placement["sheet_id"],
        final_x,
        final_y,
        0,
        1.0,
        0
    )

    print("SHIELD FINAL MOVE RESULT:", final_move_result)

    if final_move_result == 0:
        raise Exception(f"Could not move sized shield to final X={final_x} Y={final_y}")

    final_id = final_move_result
    time.sleep(MOVE_DELAY_SECONDS)

    successful_sym.SetId(final_id)
    final_length = float(successful_sym.GetDisplayLength())
    final_location = get_symbol_location(successful_sym, final_id)

    print("SHIELD FINAL ID:", final_id)
    print("SHIELD FINAL LENGTH:", final_length)

    if final_location is not None:
        print("SHIELD FINAL SHEET:", final_location["sheet_id"])
        print("SHIELD FINAL ACTUAL X:", final_location["x"])
        print("SHIELD FINAL ACTUAL Y:", final_location["y"])

    # If E3 changed the length during the move, immediately re-apply it
    # while bound to the final placed ID using the same proven sequence.
    if abs(final_length - requested_length) > 0.001:
        print("SHIELD LENGTH CHANGED DURING MOVE - REAPPLYING AT FINAL LOCATION")
        if not try_size_shield(successful_sym, final_id, requested_length):
            raise Exception(
                f"Shield changed length after final move. "
                f"Expected {requested_length}, actual {successful_sym.GetDisplayLength()}"
            )

    print("SHIELD FINAL LENGTH VERIFIED")
    print("SHIELD PLACED AND RESIZED SUCCESSFULLY")

    return final_id

# MAIN

def main():
    excel_path = ask_excel_path()

    if excel_path is None:
        raise Exception("No Excel file selected")

    workbook = load_workbook(excel_path, data_only=True, read_only=False)

    if TARGET_SHEET_NAME is None:
        ws = workbook.active
    else:
        if TARGET_SHEET_NAME not in workbook.sheetnames:
            raise Exception(f"Worksheet {TARGET_SHEET_NAME} was not found")

        ws = workbook[TARGET_SHEET_NAME]

    sheet_device_reference = clean_excel_value(ws[DEVICE_NAME_CELL].value)
    twist_offset_settings = read_twist_offset_settings(ws)
    last_row = get_last_data_row(ws)

    if last_row < DATA_FIRST_ROW:
        raise Exception("No cable overlay data found")

    print("")
    print("============================================================")
    print("ZUKEN CABLE SHIELD / TWIST OVERLAY")
    print("============================================================")
    print("EXCEL:", excel_path)
    print("WORKSHEET:", ws.title)
    print("DEVICE REFERENCE FROM B1:", sheet_device_reference)
    print("CONNECTOR ROOT:", CONNECTOR_ROOT_NAME)
    print("SHIELD SYMBOL:", SHIELD_SYMBOL_NAME)
    print("TWIST SYMBOL:", TWIST_SYMBOL_NAME)
    print("LEFT TWIST OFFSET J2:K2:", "YES" if twist_offset_settings["LEFT"] else "NO")
    print("RIGHT TWIST OFFSET R2:S2:", "YES" if twist_offset_settings["RIGHT"] else "NO")
    print("PLACEMENT DELAY:", PLACEMENT_DELAY_SECONDS)
    print("TWIST STRETCHING: DISABLED")
    print("SHIELD MODE: SEARCH STAGE -> SIZE -> MOVE")
    print("DATA ROWS:", DATA_FIRST_ROW, "TO", last_row)

    left_rows = read_side_rows(ws, "LEFT", last_row)
    right_rows = read_side_rows(ws, "RIGHT", last_row)

    left_sections = split_into_sections(left_rows)
    right_sections = split_into_sections(right_rows)

    sections = left_sections + right_sections

    packages = []
    warnings = []

    # Twists and inner shields are connector-specific.
    for section_rows in sections:
        jobs, section_warnings = build_section_jobs(section_rows)

        if jobs:
            packages.append({"rows": section_rows, "jobs": jobs})

        warnings.extend(section_warnings)

    # Outer shields are side-wide and may cross connector changes.
    outer_jobs = (
        build_outer_shield_jobs(left_rows) +
        build_outer_shield_jobs(right_rows)
    )

    print("")
    print("CONNECTOR SECTIONS WITH TWIST / INNER-SHIELD SYMBOLS:", len(packages))
    print("SIDE-WIDE OUTER SHIELD JOBS:", len(outer_jobs))

    if warnings:
        print("")
        print("============================================================")
        print("EXCEL WARNINGS")
        print("============================================================")

        for warning in warnings:
            print("WARNING:", warning)

    app = win32.gencache.EnsureDispatch("CT.Application")
    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()
    sym = prj.CreateSymbolObject()

    app.PutMessage("Starting Cable Shield / Twist Overlay Script")

    all_devices = get_all_devices(prj, dev)

    print("")
    print("E3 BASE DEVICE ID COUNT:", len(all_devices))

    pin_map_cache = {}
    placed_ids = []

    for package in packages:
        section_rows = package["rows"]
        jobs = package["jobs"]

        side = section_rows[0]["side"]
        connector = section_rows[0]["connector"]

        required_pins = list(dict.fromkeys(row["pin"] for row in section_rows))

        print("")
        print("============================================================")
        print("PROCESSING CONNECTOR SECTION")
        print("============================================================")
        print("SIDE:", side)
        print("TWIST OFFSET ENABLED:", "YES" if twist_offset_settings[side] else "NO")
        print("EXCEL CONNECTOR:", connector)
        print("EXPECTED E3 CONNECTOR:", connector_base_name(connector))
        print("EXCEL ROWS:", section_rows[0]["excel_row"], "TO", section_rows[-1]["excel_row"])
        print("REQUIRED PINS:", required_pins)

        device_record, pin_map, sheet_id = resolve_connector_schematic_view(connector, required_pins, all_devices, dev, pin, pin_map_cache)

        print_section_pin_locations(required_pins, pin_map, sheet_id)

        jobs = sorted(jobs, key=job_priority)

        for job in jobs:
            try:
                placement = calculate_placement(job, pin_map, sheet_id, twist_offset_settings)
                placed_id = place_job(prj, sym, job, placement)

                if placed_id is not None:
                    placed_ids.append(placed_id)

            except Exception as error:
                print("ERROR:", str(error))
                app.PutMessage(f"Cable overlay warning: {error}")

            time.sleep(PLACEMENT_DELAY_SECONDS)

    # Outer shields are placed last so they wrap all connector sections on that side.
    for job in outer_jobs:
        try:
            print("")
            print("============================================================")
            print("PROCESSING SIDE-WIDE OUTER SHIELD")
            print("============================================================")
            print("SIDE:", job["side"])
            print("CONNECTORS:", job["connectors"])
            print("EXCEL ROWS:", job["excel_rows"][0], "TO", job["excel_rows"][-1])

            placement = calculate_outer_shield_placement(
                job,
                all_devices,
                dev,
                pin,
                pin_map_cache,
                twist_offset_settings
            )

            placed_id = place_job(prj, sym, job, placement)

            if placed_id is not None:
                placed_ids.append(placed_id)

        except Exception as error:
            print("ERROR:", str(error))
            app.PutMessage(f"Cable overlay warning: {error}")

        time.sleep(PLACEMENT_DELAY_SECONDS)

    print("")
    print("============================================================")
    print("OVERLAY COMPLETE")
    print("============================================================")
    print("PLACED SYMBOL COUNT:", len(placed_ids))
    print("PLACED SYMBOL IDS:", placed_ids)
    print("============================================================")

    app.PutMessage(f"Cable Shield / Twist Overlay complete. Placed {len(placed_ids)} symbols.")

    workbook.close()

    del sym
    del pin
    del dev
    del prj
    del app


if __name__ == "__main__":
    main()
