import time
import win32com.client as win32


SYMBOL_NAME = "Shield_Demo"
SYMBOL_VERSION = 1
SHEET_ID = 204828


app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sym = prj.CreateSymbolObject()


print("")
print("============================================================")
print("SHIELD RESIZE CONFIRMATION")
print("============================================================")


load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)

print("LOAD RESULT:", load_result)


placed_id = sym.Place(SHEET_ID, 10.0, 2.0, 0, 1.0, 0)

print("PLACE RESULT:", placed_id)


time.sleep(0.10)


sym.SetId(placed_id)


before = sym.GetDisplayLength()

print("LENGTH BEFORE:", before)


result = sym.SetDisplayLength(2.0)

print("SET LENGTH RESULT:", result)


time.sleep(0.10)


sym.SetId(placed_id)


after = sym.GetDisplayLength()

print("LENGTH AFTER:", after)


print("")
print("============================================================")
print("DONE")
print("============================================================")


del sym
del prj
del app
