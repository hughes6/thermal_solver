"""E3 Block Maker launcher with descriptive, copyable error popups.

Run `python launcher.py` (or the packaged launcher executable).
Runs main.py and test.py in separate processes, as in the original design.
"""

import os
import sys
import tempfile
import subprocess
import traceback
from collections import deque
from datetime import datetime
from pathlib import Path


ROLE_ENV = "BLOCK_MAKER_ROLE"
METADATA_ENV = "E3_SECONDARY_BLOCK_FILE"


def get_application_directory():
    # Block Maker support modules (main.py, test.py, utils.py, error_popup.py)
    # are installed next to launcher.py. EXE location is NOT their directory.
    return Path(__file__).resolve().parent


def get_child_command():
    if (getattr(sys, "frozen", False) or os.environ.get("E3_MEGA_PACKAGED_EXE") == "1"):
        # Both child stages run this EXE in Block Maker worker mode; the
        # BLOCK_MAKER_ROLE environment variable selects main.py vs test.py.
        return [sys.executable, '--run-tool', 'block_maker']
    return [sys.executable, str(Path(__file__).resolve())]


def run_builder_child():
    import main  # Exceptions must propagate so subprocess returns a failure code.


def run_formatter_child():
    """Actually RUN the formatter, rather than only importing its module.

    test.py deliberately guards its entry point with ``if __name__ ==
    '__main__'``.  A plain ``import test`` therefore succeeds immediately
    without formatting the connector/pin text or showing the Saber prompt.
    Load by absolute file path to avoid collisions with another ``test``
    module, and invoke its main() explicitly.
    """
    import importlib.util

    formatter_path = get_application_directory() / 'test.py'
    spec = importlib.util.spec_from_file_location(
        'e3_block_maker_formatter_stage', str(formatter_path)
    )
    if spec is None or spec.loader is None:
        raise RuntimeError(f'Could not load Block Maker formatter: {formatter_path}')
    formatter = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(formatter)
    return formatter.main()


class StageFailed(Exception):
    def __init__(self, stage, return_code, output):
        self.stage = stage
        self.return_code = return_code
        self.output = output
        super().__init__(f"{stage} exited with code {return_code}")


def emit(line, log_handle):
    """Stream to both console and persistent log, without waiting for the child."""
    log_handle.write(line)
    log_handle.flush()
    if sys.stdout is not None:
        try:
            sys.stdout.write(line)
            sys.stdout.flush()
        except (OSError, ValueError):
            pass


def run_stage(stage, command, environment, cwd, log_handle):
    # A Block Maker stage must never fall back to the Mega Tools main menu.
    # Both markers identify this child independently of its command line.
    if (getattr(sys, 'frozen', False) or os.environ.get("E3_MEGA_PACKAGED_EXE") == "1"):
        environment['E3_MEGA_TOOLS_CHILD_ID'] = 'block_maker'
        environment['E3_MEGA_TOOLS_CHILD_ARGS_JSON'] = '[]'
    heading = f"\n{'=' * 60}\nSTARTING {stage.upper()}\n{'=' * 60}\n"
    emit(heading, log_handle)

    # Keep a tail in memory, but persist the entire output to the log file.
    recent_lines = deque(maxlen=400)
    process = subprocess.Popen(
        command,
        cwd=str(cwd),
        env=environment,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        text=True,
        encoding="utf-8",
        errors="replace",
        bufsize=1,
    )

    assert process.stdout is not None
    with process.stdout:
        for line in process.stdout:
            emit(line, log_handle)
            recent_lines.append(line)

    exit_code = process.wait()
    if exit_code != 0:
        raise StageFailed(stage, exit_code, "".join(recent_lines))

    emit(f"\n{stage.upper()} COMPLETE\n", log_handle)


def choose_workbook_before_workers():
    """Ask once, before spawning DB-writing stages; Cancel/X is success."""
    chosen = os.environ.get('E3_MEGA_BLOCK_MAKER_WORKBOOK', '').strip()
    if chosen:
        if not Path(chosen).is_file():
            raise FileNotFoundError('Selected Block Maker workbook not found: ' + chosen)
        return str(Path(chosen).resolve())
    import tkinter as tk
    from tkinter import filedialog
    chooser = tk.Tk()
    chooser.withdraw()
    try:
        chooser.attributes('-topmost', True)
        chooser.update()
        chosen = filedialog.askopenfilename(
            parent=chooser,
            title='Select Block Maker Excel workbook',
            filetypes=[('Excel workbooks', '*.xlsm *.xlsx'), ('All files', '*.*')])
        return str(Path(chosen).resolve()) if chosen else None
    finally:
        chooser.destroy()


def main():
    # Children should never create their own popups. One parent-level popup
    # handles exceptions from both stages, including those in utils.py.
    role = os.environ.get(ROLE_ENV, "").strip().lower()
    if role == "main":
        run_builder_child()
        return 0
    if role == "formatter":
        # Preserve the formatter's exit code instead of always reporting success.
        return run_formatter_child()

    from error_popup import show_error_dialog

    # The user's Cancel/X must not be interpreted as a failed builder stage.
    # No subprocess or E3/SQL connection is opened until a workbook is chosen.
    try:
        workbook = choose_workbook_before_workers()
    except Exception as exc:
        show_error_dialog('Workbook Selection', str(exc), '', 1)
        return 1
    if not workbook:
        return 0
    os.environ['E3_MEGA_BLOCK_MAKER_WORKBOOK'] = workbook

    app_dir = get_application_directory()
    command = get_child_command()
    log_dir = Path(tempfile.gettempdir()) / "E3_Block_Maker_Logs"
    log_dir.mkdir(parents=True, exist_ok=True)
    log_path = log_dir / f"block_maker_{datetime.now():%Y%m%d_%H%M%S_%f}.log"

    stage = "Launcher"
    try:
        with log_path.open("w", encoding="utf-8", errors="replace") as log:
            emit(f"Block Maker log: {log_path}\n", log)
            emit(f"Application directory: {app_dir}\n", log)
            with tempfile.TemporaryDirectory(prefix="e3_block_maker_") as temp_dir:
                metadata_path = str(Path(temp_dir) / "secondary_blocks.json")
                emit(f"Metadata path: {metadata_path}\n", log)

                # Each child receives its own environment. The original
                # builder->formatter sequencing is preserved.
                builder_env = os.environ.copy()
                builder_env[ROLE_ENV] = "main"
                builder_env[METADATA_ENV] = metadata_path
                builder_env["PYTHONUNBUFFERED"] = "1"
                stage = "Main Builder"
                run_stage(stage, command, builder_env, app_dir, log)

                if not Path(metadata_path).is_file():
                    raise RuntimeError(
                        "Main builder exited successfully but did not write "
                        "the E3_SECONDARY_BLOCK_FILE metadata JSON."
                    )

                formatter_env = os.environ.copy()
                formatter_env[ROLE_ENV] = "formatter"
                formatter_env[METADATA_ENV] = metadata_path
                formatter_env["PYTHONUNBUFFERED"] = "1"
                stage = "Formatter"
                run_stage(stage, command, formatter_env, app_dir, log)

            emit("\nBLOCK MAKER COMPLETE — BOTH STAGES SUCCEEDED\n", log)
        return 0

    except StageFailed as exc:
        # No hidden success state: a failed child stops the next stage.
        show_error_dialog(exc.stage, exc.output, str(log_path), exc.return_code)
        return exc.return_code if exc.return_code > 0 else 1

    except Exception:
        details = traceback.format_exc()
        try:
            with log_path.open("a", encoding="utf-8") as log:
                emit(f"\nLAUNCHER/STAGE FAILURE\n{details}", log)
        except OSError:
            sys.stderr.write(details)
        show_error_dialog(stage, details, str(log_path), 1)
        return 1


if __name__ == "__main__":
    sys.exit(main())
