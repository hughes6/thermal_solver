def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def generate_master_schematic_layout(PIN_SIGNAL_INFO,x_block,main_block_width):
    SHEET_DICT = {1:sheet_1_id}

    # PAGE BOUNDS
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    # PAGE MARGINS
    TOP_MARGIN = 1.0
    BOTTOM_MARGIN = 1.0
    LEFT_MARGIN = 1.0
    RIGHT_MARGIN = 1.0

    # BLOCK SPACING
    BLOCK_X_SPACING = 1.0
    BLOCK_Y_SPACING = 1.0
    SCHEMATIC_BLOCK_WIDTH = 3.0
    PIN_SPACING = 0.25
    PIN_START_OFFSET = 2.25

    # USABLE PAGE LIMITS
    PAGE_TOP = TOP_OF_PAGE - TOP_MARGIN
    PAGE_BOTTOM = BOTTOM_OF_PAGE + BOTTOM_MARGIN
    PAGE_LEFT = LEFT_OF_PAGE + LEFT_MARGIN
    PAGE_RIGHT = RIGHT_OF_PAGE - RIGHT_MARGIN
    USABLE_PAGE_HEIGHT = PAGE_TOP - PAGE_BOTTOM

    # OUTPUT
    SPACING_DICT = []
    sheet = 1
    # On sheet 1, start to the RIGHT of the main block
    current_x = (x_block + main_block_width + BLOCK_X_SPACING)

    # Current Y is the TOP edge available for placement
    current_top_y = PAGE_TOP
    # Tracks widest block in current column
    current_column_width = 0.0

    # ================================================
    # BUILD LAYOUT
    # ================================================
    for i in range(len(PIN_SIGNAL_INFO)):
        connector_name = (PIN_SIGNAL_INFO[i][0]["connector name"])
        number_of_pins = (PIN_SIGNAL_INFO[i][0]["number of pins"])
        width = SCHEMATIC_BLOCK_WIDTH
        height = (PIN_START_OFFSET + (PIN_SPACING * number_of_pins))

        # --------------------------------------------
        # SAFETY CHECK:
        # Block cannot physically fit on ANY page
        # --------------------------------------------
        if height > USABLE_PAGE_HEIGHT:
            raise Exception(f"Schematic block for {connector_name} is too tall for a page. Height = {height}, usable page height = {USABLE_PAGE_HEIGHT}")

        # CALCULATE BOTTOM Y IF PLACED HERE
        candidate_y = current_top_y - height

        # DOES IT FIT VERTICALLY?
        if candidate_y < PAGE_BOTTOM:
            next_x = (current_x + current_column_width + BLOCK_X_SPACING)

            # DOES NEXT COLUMN FIT ON THIS PAGE?
            if next_x + width <= PAGE_RIGHT:
                current_x = next_x
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            else:
                # PAGE IS FULL -> NEW PAGE
                sheet += 1
                sheet_symbol = "ANSI Size E - BAE"
                res = sht.Create(0, str(sheet), sheet_symbol, 0, 0)
                if res == 0:
                    app.PutMessage("failed to create new sheet")
                    raise Exception("failed to create new sheet")
                SHEET_DICT[sheet] = res
                current_x = PAGE_LEFT
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            # Recalculate Y at new column/page
            candidate_y = current_top_y - height

        # CHECK HORIZONTAL FIT
        if current_x + width > PAGE_RIGHT:
            sheet += 1
            res = sht.Create(0, sheet, "DINA3", 0)
            if res == 0:
                app.PutMessage("failed to create new sheet")
                raise Exception("failed to create new sheet")
            SHEET_DICT[sheet] = res
            current_x = PAGE_LEFT
            current_top_y = PAGE_TOP
            current_column_width = 0.0
            candidate_y = current_top_y - height

        # SAVE PLACEMENT
        x_start = current_x
        y_start = candidate_y
        SPACING_DICT.append({"connector name": connector_name, "connector index": i, "sheet": sheet, "x start": x_start, "y start": y_start, "x end": x_start + width, "y end": y_start + height, "height": height, "width": width, "number of pins": number_of_pins})
        # UPDATE CURSOR FOR NEXT BLOCK
        current_top_y = (y_start - BLOCK_Y_SPACING)
        current_column_width = max(current_column_width, width)

    return SPACING_DICT, SHEET_DICT



def generate_schematic_pin_map(connector_info,newpindict,layout,pin_spacing=0.25,pin_start_offset=2.0,edge_offset=0.3):
    x=layout["x start"]+layout["width"]-0.3
    y_start=layout["y start"]+layout["height"]-pin_start_offset
    pin_map=[]
    for pos,pin_data in enumerate(connector_info[1:]):
        pin_name=str(pin_data[0]).strip()
        resolved_pin_name = pin_name

        # Normal case: the Excel pin name exactly matches the E3 view pin name.
        if resolved_pin_name not in newpindict:
            # Minimal fallback for library pins such as "1-C" / "2-S" when
            # Excel identifies those same pins as "1" / "2".  Only use the
            # fallback when there is exactly one unambiguous match.
            prefix_matches = [
                e3_pin_name
                for e3_pin_name in newpindict
                if str(e3_pin_name).strip().split("-", 1)[0] == pin_name
            ]

            if len(prefix_matches) == 1:
                resolved_pin_name = prefix_matches[0]
                print(
                    "PIN NAME FALLBACK:",
                    connector_info[0]['connector name'],
                    "EXCEL:", pin_name,
                    "E3:", resolved_pin_name
                )
            else:
                raise Exception(
                    f"Pin {pin_name} not found in new view for {connector_info[0]['connector name']}. "
                    f"Available E3 pins: {list(newpindict.keys())}"
                )

        pin_map.append({"pin name":pin_name,"pin id":newpindict[resolved_pin_name],"x":x,"y":y_start-(pos*pin_spacing),"flip":"/1"})
    return pin_map

def format_main_block_view(device_view_id, x, y, width, height, block_symbol_id=None):
    print("FORMATTING DEVICE VIEW:", device_view_id)

    x_center = x + (width / 2.0)

    # Split blocks share one logical device view.  Do NOT reformat every
    # symbol from that view for each connector's independent block layout.
    # The saved exact symbol ID ensures each split block keeps its own text.
    if block_symbol_id is not None:
        symids = [block_symbol_id]
    else:
        dev.SetId(device_view_id)
        syminfo = dev.GetSymbolIds(None)
        symids = [s for s in syminfo[1] if s is not None]

    for sym_id in symids:
        sym.SetId(sym_id)

        sym.SetBlockHatchColour(58)
        sym.SetBlockOutlineColour(0)

        #=========================================
        # NORMAL SYMBOL TEXT
        #=========================================
        textinfo = sym.GetTextIds(None)
        textids = [t for t in textinfo[1] if t is not None]

        for text_id in textids:
            text.SetId(text_id)
            txt = str(text.GetText()).strip()

            print("SYMBOL TEXT:", "VIEW:", device_view_id, "SYM:", sym_id, "TEXT ID:", text_id, "VALUE:", repr(txt))

            if txt == "BxRxUx":
                text.SetColour(0)
                text.SetMode(1)
                text.SetAlignment(2)
                text.SetStyle(1)
                text.SetHeight(0.2)
                text.SetSchemaLocation(x_center, y + height - 0.3)

            elif txt == str(DEVICE_DESIGNATION).strip() or str(DEVICE_DESIGNATION).strip() in txt:
                text.SetColour(0)
                text.SetMode(1)
                text.SetAlignment(2)
                text.SetStyle(1)
                text.SetHeight(0.2)
                text.SetSchemaLocation(x_center, y + height - 0.7)

                print("MOVED DEVICE DESIGNATION:", repr(txt), "TO:", x_center, y + height - 0.7)

            elif DEVICE_FUNCTION is not None and txt == str(DEVICE_FUNCTION).strip():
                text.SetVisibility(0)

        #=========================================
        # ATTRIBUTE TEXT
        #=========================================
        attrinfo = sym.GetAttributeIds(None)
        attrids = [a for a in attrinfo[1] if a is not None]

        for attr_id in attrids:
            attr.SetId(attr_id)
            attr_name = str(attr.GetName()).strip()

            print("ATTRIBUTE:", "VIEW:", device_view_id, "SYM:", sym_id, "ATTR ID:", attr_id, "NAME:", repr(attr_name))

            if attr_name == "Equipment Identifacation":
                attrtextinfo = attr.GetTextIds(None)
                attrtextids = [t for t in attrtextinfo[1] if t is not None]

                print("EQUIPMENT IDENTIFICATION TEXT IDS:", attrtextids)

                for text_id in attrtextids:
                    text.SetId(text_id)
                    txt = str(text.GetText()).strip()

                    print("ATTRIBUTE TEXT:", "ID:", text_id, "VALUE:", repr(txt))

                    text.SetVisibility(1)
                    text.SetColour(0)
                    text.SetMode(1)
                    text.SetAlignment(2)
                    text.SetStyle(1)
                    text.SetHeight(0.2)
                    text.SetSchemaLocation(x_center, y + height - 1.1)

                    print("MOVED EQUIPMENT IDENTIFICATION:", repr(txt), "TO:", x_center, y + height - 1.1)

    connector_map_info = CONNECTOR_MAP[i]
    connector_position = connector_map_info['connector position']
    flip = connector_map_info['connector flip']

    if flip == 0:
        x = x_block + 0.3
        y = y_start - (connector_position * PIN_VIEW_SPACING)
    else:
        x = x_block + block_width - 0.3
        y = y_start - (connector_position * PIN_VIEW_SPACING)
        flip = "/1"

    # print(f'placing {PIN_SIGNAL_INFO[i][0]['connector name']} at x = {x}, y = {y}, flip = {flip} when block_x = {x_block} block_y: {y_block} start: {y_start}')
    app.PutMessage(f'placing {PIN_SIGNAL_INFO[i][0]['connector name']} at x = {x}, y = {y}, flip = {flip}')
    res = sym.PlacePins(PINDICT, symname, ver, sheet_1_id, x, y, flip, 1.0)
    sym.SetId(res[0])
    textinfo = sym.GetTextIds(None)
    textids = [x for x in textinfo[1] if x is not None]
    for text_id in textids:
        text.SetId(text_id)
        if PIN_SIGNAL_INFO[i][0]['connector name'] == text.GetText():
            text.SetColour(0)
            text.SetMode(1)
            text.SetStyle(1)
            text.SetHeight(0.2)
    time.sleep(0.5)

#=========================================
# LOAD AND PLACE SCHEMATIC VIEWS

MAIN_DEVICE_ID = None
devinfo = prj.GetAllDeviceIds(None)
devids = [x for x in devinfo[1] if x is not None]
for dev_id in devids:
    dev.SetId(dev_id)
    name = dev.GetName()
    if name == DEVICE_DESIGNATION:
        MAIN_DEVICE_ID = dev_id
        break

if MAIN_DEVICE_ID is None:
    raise Exception(f"Could not find main device {DEVICE_DESIGNATION}")

MASTER_SCHEMATIC_LAYOUT, SHEET_DICT = generate_master_schematic_layout(PIN_SIGNAL_INFO, x_block, block_width)

SCHEMATIC_DATA = []

#=========================================
# PHASE 1 - CREATE ONE SECONDARY VIEW,
# THEN PLACE ITS BLOCK AS SPLIT SYMBOLS.
#
# All connector blocks belong to this SAME logical main-device view.
# Each call to PlaceBlock reuses that main-device view ID. Whether E3
# treats the resulting symbols as true split blocks must be verified in E3.
# Each connector gets a new device view from its original parent connector.
#=========================================
SHARED_MAIN_VIEW_NUMBER = 1

dev.SetId(MAIN_DEVICE_ID)
SHARED_MAIN_VIEW_ID = dev.CreateView(MAIN_DEVICE_ID, SHARED_MAIN_VIEW_NUMBER, 0, "0")
if SHARED_MAIN_VIEW_ID == 0:
    raise Exception(f"Failed creating shared main schematic view {SHARED_MAIN_VIEW_NUMBER}")

print("SHARED MAIN DEVICE VIEW:", "SOURCE:", MAIN_DEVICE_ID,
      "VIEW:", SHARED_MAIN_VIEW_NUMBER, "ID:", SHARED_MAIN_VIEW_ID)

placed_split_block_ids = set()

for i in range(len(PIN_SIGNAL_INFO)):
    conn_name = PIN_SIGNAL_INFO[i][0]["connector name"]
    conn_layout_dict = MASTER_SCHEMATIC_LAYOUT[i]
    current_sheet_id = SHEET_DICT[conn_layout_dict["sheet"]]

    xp = conn_layout_dict["x start"]
    yp = conn_layout_dict["y start"]
    xwidth = conn_layout_dict["width"]
    yheight = conn_layout_dict["height"]

    # All split block symbols share main-device view 1.  Connector view
    # numbers should be aligned with that main view.  Each connector is a
    # distinct logical device, so each may independently have view 1.
    view_number = SHARED_MAIN_VIEW_NUMBER
    parent_block = SHARED_MAIN_VIEW_ID
    print("SPLIT BLOCK:", conn_name, "SHARED MAIN VIEW:", parent_block,
          "SPLIT INDEX:", i + 1)

    # Fresh symbol object ONLY for this block
    block_sym = prj.CreateSymbolObject()
    is_loaded = block_sym.Load("CABBLOCK", 1)

    if not is_loaded:
        raise Exception(f"Failed loading CABBLOCK for {conn_name}")

    actual_yp = yp
    block_symbol_id = block_sym.PlaceBlock(current_sheet_id, xp, actual_yp, xwidth, yheight, parent_block)

    if block_symbol_id == 0:
        for retry_offset in range(1, 5):
            actual_yp = yp - retry_offset

            block_symbol_id = block_sym.PlaceBlock(
                current_sheet_id,
                xp,
                actual_yp,
                xwidth,
                yheight,
                parent_block
            )

            if block_symbol_id != 0:
                break

        if block_symbol_id == 0:
            raise Exception(
                f"Failed placing split block for {conn_name} in shared view "
                f"{SHARED_MAIN_VIEW_ID}. E3 did not accept another PlaceBlock "
                "placement for the same main-device view."
            )

    # Guard against a COM result that returns an existing symbol instead of
    # generating a new split representation. Never format the wrong block.
    if block_symbol_id in placed_split_block_ids:
        raise Exception(
            f"E3 reused symbol ID {block_symbol_id} for {conn_name} "
            "instead of creating a distinct split block symbol."
        )
    placed_split_block_ids.add(block_symbol_id)

    # If this block had to move down, move every still-unplaced block below it
    # in the SAME sheet/column by the same amount.  This keeps the original
    # spacing intact instead of letting a retry move on top of the next block.
    placement_shift = yp - actual_yp
    if placement_shift > 0:
        for future_layout in MASTER_SCHEMATIC_LAYOUT[i + 1:]:
            if (
                future_layout["sheet"] == conn_layout_dict["sheet"]
                and abs(future_layout["x start"] - conn_layout_dict["x start"]) < 0.000001
            ):
                future_layout["y start"] -= placement_shift
                future_layout["y end"] -= placement_shift

        print(
            "SHIFTED REMAINING BLOCKS IN COLUMN:",
            conn_name,
            "SHIFT:", placement_shift
        )

    # Keep the layout synchronized with where the block was ACTUALLY placed.
    # The later pin map and text formatter both use this layout Y.
    conn_layout_dict["y start"] = actual_yp
    conn_layout_dict["y end"] = actual_yp + yheight

    # =========================================
    # SET COLOR
    # =========================================
    block_sym.SetId(block_symbol_id)
    block_sym.SetBlockHatchPattern(1, 45.0, 135.0)
    block_sym.SetBlockHatchStyle(0)
    block_sym.SetBlockHatchWidth(0.25)
    block_sym.SetBlockHatchDistance(1.0)
    block_sym.SetBlockHatchColour(58)

    print("BLOCK:", conn_name, "DEVICE VIEW:", parent_block, "SYMBOL:", block_symbol_id)

    # =========================================
    # ADD EQUIPMENT IDENTIFACATION DIRECTLY TO
    # THE EXACT SECONDARY BLOCK SYMBOL WE JUST PLACED
    # =========================================
    block_sym.SetId(block_symbol_id)

    equipment_attr_id = block_sym.AddAttributeValue("Equipment Identifacation", ADDITIONAL_PART_ID)

    if equipment_attr_id == -1:
        print("ADD EQUIPMENT IDENTIFACATION RETURNED -1:", "CONNECTOR:", conn_name, "VIEW:", parent_block, "SYMBOL:", block_symbol_id)
    else:
        print("ADDED EQUIPMENT IDENTIFACATION:", "CONNECTOR:", conn_name, "VIEW:", parent_block, "SYMBOL:", block_symbol_id, "ATTR ID:", equipment_attr_id, "VALUE:", repr(ADDITIONAL_PART_ID))

    # =========================================
    # VERIFY IT IS ACTUALLY ON THIS SYMBOL
    # DO NOT DISPLAY OR FORMAT IT
    # =========================================
    verify_attr_info = block_sym.GetAttributeIds(None)
    verify_attr_ids = [x for x in verify_attr_info[1] if x is not None]

    equipment_attr_found = False

    for verify_attr_id in verify_attr_ids:
        attr.SetId(verify_attr_id)

        try:
            verify_attr_name = str(attr.GetName()).strip()
        except Exception:
            continue

        if verify_attr_name == "Equipment Identifacation":
            equipment_attr_found = True
            print("VERIFIED EQUIPMENT IDENTIFACATION ON BLOCK SYMBOL:", "CONNECTOR:", conn_name, "VIEW:", parent_block, "SYMBOL:", block_symbol_id, "ATTR ID:", verify_attr_id)
            break

    if not equipment_attr_found:
        print("WARNING - EQUIPMENT IDENTIFACATION NOT FOUND AFTER ADD:", "CONNECTOR:", conn_name, "VIEW:", parent_block, "SYMBOL:", block_symbol_id)

    # IMPORTANT - release this symbol object NOW
    del block_sym
    time.sleep(0.25)

    connector_device_id = CONNECTOR_DEVICE_IDS[conn_name]

    # Create a NEW DEVICE VIEW OF THE ORIGINAL/PARENT CONNECTOR.
    # The shared CABBLOCK main device still has only ONE view; each
    # connector has its own separate view number 1 underneath it.
    # Zuken requires that block connector views have the same view number
    # as their block, so do not increment this view number per pin.
    dev.SetId(connector_device_id)
    connector_view_number = SHARED_MAIN_VIEW_NUMBER
    connector_view_id = dev.CreateView(
        connector_device_id,
        connector_view_number,
        parent_block
    )
    if not connector_view_id:
        raise Exception(
            f"Failed creating new view from parent connector {conn_name} "
            f"(source ID {connector_device_id}, view number "
            f"{connector_view_number}, parent block {parent_block})"
        )

    print("CREATED VIEW OF PARENT CONNECTOR:",
          "CONNECTOR:", conn_name,
          "ORIGINAL ID:", connector_device_id,
          "NEW VIEW ID:", connector_view_id,
          "VIEW NUMBER:", connector_view_number,
          "SHARED BLOCK:", parent_block, flush=True)

    SCHEMATIC_DATA.append({
        "index": i,
        "name": conn_name,
        "parent_block": parent_block,
        "block_symbol_id": block_symbol_id,
        "connector_device_id": connector_device_id,
        "connector_view": connector_view_id,
        "sheet_id": current_sheet_id,
        "layout": conn_layout_dict
    })

    time.sleep(0.5)

        placed_sym_id = res[0] if isinstance(res, (tuple, list)) else res
        if not placed_sym_id:
            raise Exception(

                f"from connector view {connector_view_id}, "

                f"block {data['block_symbol_id']}, sheet {current_sheet_id}; "
