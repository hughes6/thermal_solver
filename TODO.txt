def get_manual_column_count():
    value = column_count_entry.get().strip()

    if value == "":
        return None

    try:
        column_count = int(value)
    except ValueError:
        return None

    if column_count <= 0:
        return None

    return column_count


def normalize_pdf_line(line):
    line = line.replace("\t", " ")
    line = re.sub(r"\s+", " ", line)

    return line.strip()


def is_integer_value(value):
    value = normalize_pdf_line(value)

    return re.fullmatch(r"\d+", value) is not None


# =========================================
# PDF CONNECTOR PINOUT TABLE DETECTION
#
# Handles copied PDF tables such as:
#
# Ref
# Des Pin Signal Name Direction Type Description
# J1R
# 1 NC NC NC No Connect
# 11 GND - GND Secondary ground
# 12 P5_1000BT_D4- Input BASE-T BUS ... pair 4/D, negative
#
# The important detail is that "Ref Des" is often a merged cell in the PDF,
# so the connector designator appears once and the following data rows begin
# with the pin number. Descriptions are allowed to contain commas.
# =========================================

PINOUT_HEADERS = [
    "REF DES",
    "PIN",
    "SIGNAL NAME",
    "DIRECTION",
    "TYPE",
    "DESCRIPTION"
]


def get_nonempty_normalized_lines(input_text):
    lines = []

    for raw_line in input_text.split("\n"):
        line = normalize_pdf_line(raw_line)

        if line != "":
            lines.append(line)

    return lines


# =========================================
# PDF 4-COLUMN POWER / SIMPLE PINOUT TABLE DETECTION
#
# Handles copied PDF tables such as:
#
# TABLE 5 - POWER CONNECTOR PINOUT
# Pin #   Signal Name   Signal Type   Signal Description
# 1       VBUS          Power         22-36Vdc Power Input
# 2       VBUS          Power         22-36Vdc Power Input
# 3       INHIBIT       Power         Enable
# 4       BUS_RTN       Power         Board Ground
#
# This table family has NO Ref Des column and NO Direction column.  It is
# intentionally detected before the 6-column pinout parser so the words
# "Signal Type" / "Signal Description" cannot be mistaken for that format.
# =========================================

SIMPLE_PINOUT_HEADERS = [
    "PIN #",
    "SIGNAL NAME",
    "SIGNAL TYPE",
    "SIGNAL DESCRIPTION"
]


def get_flat_pdf_tokens(input_text):
    """Return a whitespace token stream regardless of how Acrobat inserted newlines."""
    tokens = []

    for raw_line in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        line = normalize_pdf_line(raw_line)

        if line == "":
            continue

        tokens.extend(line.split())

    return tokens


def normalized_header_token(value):
    value = str(value or "").strip().upper()
    value = value.replace("–", "-").replace("—", "-")
    return value.strip(":")


def find_token_pair(tokens, first, second, start=0, stop=None):
    upper_tokens = [normalized_header_token(token) for token in tokens]
    if stop is None:
        stop = len(tokens)

    for index in range(max(0, start), min(stop, len(tokens) - 1)):
        if upper_tokens[index] == first and upper_tokens[index + 1] == second:
            return index

    return None


def find_simple_pinout_header_end(tokens):
    """
    Find the end of:
        Pin # | Signal Name | Signal Type | Signal Description

    This works whether those headers arrive as four normal lines, individual
    cells on separate lines, or literally one word per line.
    """
    upper_tokens = [normalized_header_token(token) for token in tokens]
    search_stop = min(len(tokens), 120)

    for description_index in range(search_stop - 1):
        if not (
            upper_tokens[description_index] == "SIGNAL"
            and upper_tokens[description_index + 1] == "DESCRIPTION"
        ):
            continue

        before = upper_tokens[max(0, description_index - 20):description_index]
        before_text = " ".join(before)

        if "SIGNAL NAME" not in before_text:
            continue

        if "SIGNAL TYPE" not in before_text:
            continue

        if "PIN" not in before:
            continue

        return description_index + 2

    return None


def find_connector_pinout_header_end(tokens):
    """
    Find the end of:
        Ref Des | Pin | Signal Name | Direction | Type | Description

    The header can be split at arbitrary newline boundaries.
    """
    upper_tokens = [normalized_header_token(token) for token in tokens]
    search_stop = min(len(tokens), 140)

    for description_index in range(search_stop):
        if upper_tokens[description_index] != "DESCRIPTION":
            continue

        before = upper_tokens[max(0, description_index - 24):description_index]
        before_text = " ".join(before)

        if "SIGNAL NAME" not in before_text:
            continue

        if "DIRECTION" not in before:
            continue

        if "TYPE" not in before:
            continue

        if "PIN" not in before:
            continue

        if "REF" not in before or "DES" not in before:
            continue

        # A 4-column table has the phrase SIGNAL TYPE.  Do not steal it.
        if "SIGNAL TYPE" in before_text:
            continue

        return description_index + 1

    return None


def looks_highly_verticalized(input_text):
    """Heuristic used only for the status text, not for deciding correctness."""
    lines = get_nonempty_normalized_lines(input_text)

    if len(lines) < 6:
        return False

    short_lines = sum(1 for line in lines if len(line.split()) <= 1)
    return short_lines / len(lines) >= 0.55


def select_pin_token_indices(tokens, data_start, minimum_payload_tokens):
    """
    Pick the most plausible increasing pin-number sequence from a flattened PDF
    token stream.  This prevents an isolated numeric value in a description from
    normally being mistaken for a new pin row.
    """
    candidates = []

    for index in range(data_start, len(tokens)):
        value = tokens[index]

        if not is_integer_value(value):
            continue

        candidates.append((index, int(value)))

    if not candidates:
        return []

    # DP state: best path ending at each candidate.  Prefer more rows first,
    # then consecutive pin numbering, then a smaller first pin number.
    states = []

    for candidate_index, (token_index, pin_number) in enumerate(candidates):
        best_path = [candidate_index]
        best_count = 1
        best_consecutive = 0
        best_first_pin = pin_number

        for previous_candidate_index in range(candidate_index):
            previous_token_index, previous_pin_number = candidates[previous_candidate_index]

            if pin_number <= previous_pin_number:
                continue

            payload_end = token_index

            # If a connector designator sits immediately before this pin, it
            # belongs to this row, not the previous row's Description.
            if token_index - 1 > previous_token_index:
                possible_ref = tokens[token_index - 1]
                if is_pinout_ref_designator(possible_ref):
                    payload_end -= 1

            payload_count = payload_end - previous_token_index - 1

            if payload_count < minimum_payload_tokens:
                continue

            previous_state = states[previous_candidate_index]
            path = previous_state["path"] + [candidate_index]
            count = previous_state["count"] + 1
            consecutive = previous_state["consecutive"] + (1 if pin_number == previous_pin_number + 1 else 0)
            first_pin = previous_state["first_pin"]

            score = (count, consecutive, 1 if first_pin == 1 else 0, -first_pin)
            best_score = (best_count, best_consecutive, 1 if best_first_pin == 1 else 0, -best_first_pin)

            if score > best_score:
                best_path = path
                best_count = count
                best_consecutive = consecutive
                best_first_pin = first_pin

        states.append({
            "path": best_path,
            "count": best_count,
            "consecutive": best_consecutive,
            "first_pin": best_first_pin
        })

    valid_states = []

    for state in states:
        last_candidate_index = state["path"][-1]
        last_token_index = candidates[last_candidate_index][0]
        remaining_payload = len(tokens) - last_token_index - 1

        if remaining_payload >= minimum_payload_tokens:
            valid_states.append(state)

    if not valid_states:
        valid_states = states

    valid_states.sort(
        key=lambda state: (
            state["count"],
            state["consecutive"],
            1 if state["first_pin"] == 1 else 0,
            -state["first_pin"]
        ),
        reverse=True
    )

    selected = valid_states[0]["path"]
    return [candidates[index][0] for index in selected]


def looks_like_pdf_simple_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_simple_pinout_header_end(tokens)

    if header_end is None:
        return False

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=2)
    return len(pin_indices) >= 2


def is_simple_pinout_header_line(line):
    upper_text = normalize_pdf_line(line).upper()

    if upper_text == "":
        return True

    if upper_text in {
        "PIN",
        "PIN #",
        "SIGNAL NAME",
        "SIGNAL TYPE",
        "SIGNAL DESCRIPTION"
    }:
        return True

    if (
        "PIN" in upper_text
        and "SIGNAL NAME" in upper_text
        and "SIGNAL TYPE" in upper_text
        and "SIGNAL DESCRIPTION" in upper_text
    ):
        return True

    if upper_text.startswith("TABLE ") and "PINOUT" in upper_text:
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    return False


def parse_simple_pinout_data_line(line):
    line = normalize_pdf_line(line)

    if line == "":
        return None

    tokens = line.split()

    if len(tokens) < 3:
        return None

    if not is_integer_value(tokens[0]):
        return None

    pin = tokens.pop(0)
    signal_name = tokens.pop(0)
    signal_type = tokens.pop(0)
    description = " ".join(tokens)

    return [pin, signal_name, signal_type, description]


def parse_pdf_simple_pinout_table(input_text):
    # Flattening first is intentional.  The same parser therefore works when
    # Acrobat gives us normal rows, one cell per line, or one word per line.
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_simple_pinout_header_end(tokens)
    rows = [list(SIMPLE_PINOUT_HEADERS)]

    if header_end is None:
        return rows

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=2)

    for row_index, pin_index in enumerate(pin_indices):
        next_pin_index = pin_indices[row_index + 1] if row_index + 1 < len(pin_indices) else len(tokens)
        payload_end = next_pin_index

        payload = tokens[pin_index + 1:payload_end]

        if len(payload) < 2:
            continue

        pin = tokens[pin_index]
        signal_name = payload[0]
        signal_type = payload[1]
        description = " ".join(payload[2:])

        rows.append([
            pin,
            signal_name,
            signal_type,
            description
        ])

    return rows

def is_pinout_ref_designator(value):
    """Stricter Ref Des test so signal names like P5_1000BT_D4- are not connectors."""
    text = normalize_pdf_line(value).upper().replace(" ", "")

    if text == "":
        return False

    # Normal connector names used by these pinout tables, including a compact
    # shared label such as J1R/J2R.
    return re.fullmatch(r"J\d+[A-Z]*(?:[/,&+]J\d+[A-Z]*)*", text) is not None

def normalize_pinout_direction(value):
    upper = str(value or "").strip().upper().replace("\\", "/")

    mapping = {
        "INPUT": "Input",
        "IN": "In",
        "OUTPUT": "Output",
        "OUT": "Out",
        "INOUT": "Inout",
        "IN/OUT": "In/Out",
        "BIDIR": "Bidir",
        "BIDIRECTIONAL": "Bidirectional",
        "NC": "NC",
        "N/C": "N/C",
        "-": "-"
    }

    return mapping.get(upper, str(value or "").strip())


def find_direction_token_index(payload):
    direction_values = {
        "INPUT",
        "IN",
        "OUTPUT",
        "OUT",
        "INOUT",
        "IN/OUT",
        "BIDIR",
        "BIDIRECTIONAL",
        "NC",
        "N/C",
        "-"
    }

    # Start at 1 so a signal named NC is not mistaken for the direction when
    # the row is "NC NC NC No Connect".
    for index in range(1, len(payload)):
        upper = str(payload[index]).strip().upper().replace("\\", "/")

        if upper in direction_values:
            return index

    return None


def parse_pinout_data_line(line, current_ref_des):
    line = normalize_pdf_line(line)

    if line == "":
        return None

    tokens = line.split()

    if len(tokens) < 4:
        return None

    ref_des = current_ref_des

    if (
        len(tokens) >= 2
        and is_connector_designator(tokens[0])
        and is_integer_value(tokens[1])
    ):
        ref_des = tokens.pop(0)

    if len(tokens) == 0 or not is_integer_value(tokens[0]):
        return None

    pin = tokens.pop(0)
    direction_index = find_direction_token_index(tokens)

    if direction_index is not None and direction_index >= 1 and direction_index + 1 < len(tokens):
        signal_name = " ".join(tokens[:direction_index])
        direction = normalize_pinout_direction(tokens[direction_index])
        signal_type = tokens[direction_index + 1]
        description = " ".join(tokens[direction_index + 2:])
    else:
        if len(tokens) < 3:
            return None

        signal_name = tokens[0]
        direction = normalize_pinout_direction(tokens[1])
        signal_type = tokens[2]
        description = " ".join(tokens[3:])

    return [ref_des, pin, signal_name, direction, signal_type, description]


def parse_pdf_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_connector_pinout_header_end(tokens)
    rows = [list(PINOUT_HEADERS)]

    if header_end is None:
        return rows

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=3)
    current_ref_des = ""

    for row_index, pin_index in enumerate(pin_indices):
        next_pin_index = pin_indices[row_index + 1] if row_index + 1 < len(pin_indices) else len(tokens)

        # Ref Des can be a vertically merged value once before the first pin,
        # or it can be repeated immediately before each pin.
        if row_index == 0:
            # Only search the small pre-data area before the first pin.  Do not
            # scan the previous row payload for later pins, because signal names
            # such as P5_1000BT_D4- also contain letters and digits.
            for token in tokens[header_end:pin_index]:
                if is_pinout_ref_designator(token):
                    current_ref_des = token

        if pin_index > header_end and is_pinout_ref_designator(tokens[pin_index - 1]):
            current_ref_des = tokens[pin_index - 1]

        payload_end = next_pin_index

        if row_index + 1 < len(pin_indices):
            # A repeated connector name immediately before the next pin belongs
            # to the next row, not this row's Description.
            if payload_end - 1 > pin_index and is_pinout_ref_designator(tokens[payload_end - 1]):
                payload_end -= 1

        payload = tokens[pin_index + 1:payload_end]

        if len(payload) < 3:
            continue

        direction_index = find_direction_token_index(payload)

        if direction_index is not None and direction_index >= 1 and direction_index + 1 < len(payload):
            signal_name = " ".join(payload[:direction_index])
            direction = normalize_pinout_direction(payload[direction_index])
            signal_type = payload[direction_index + 1]
            description = " ".join(payload[direction_index + 2:])
        else:
            signal_name = payload[0]
            direction = normalize_pinout_direction(payload[1])
            signal_type = payload[2]
            description = " ".join(payload[3:])

        rows.append([
            current_ref_des,
            tokens[pin_index],
            signal_name,
            direction,
            signal_type,
            description
        ])

    return rows


# =========================================
# SAFE CSV DETECTION
#
# Do NOT treat a block as CSV merely because a description contains a comma.
# Real CSV should show comma-separated structure on the first meaningful rows.
# =========================================

def looks_like_real_csv(input_text):
    raw_lines = [line for line in input_text.split("\n") if line.strip() != ""]

    if len(raw_lines) == 0:
        return False

    sample_lines = raw_lines[:6]
    comma_sample_count = sum(1 for line in sample_lines if "," in line)

    # A normal CSV header/data block will contain commas immediately. A PDF
    # table whose descriptions merely contain punctuation usually will not.
    if comma_sample_count < min(2, len(sample_lines)):
        return False

    widths = []

    for line in raw_lines:
        try:
            parsed = next(csv.reader([line]))
        except csv.Error:
            continue

        if len(parsed) > 1:
            widths.append(len(parsed))

    if len(widths) < 2:
        return False

    # Require a repeated multi-column width. This rejects isolated prose commas.
    width_counts = {}

    for width in widths:
        width_counts[width] = width_counts.get(width, 0) + 1

    most_common_width = max(width_counts, key=width_counts.get)
    most_common_count = width_counts[most_common_width]

    return most_common_width > 1 and most_common_count >= 2

def is_connector_designator(value):
    value = normalize_pdf_line(value)

    if value == "":
        return False

    if " " in value:
        return False

    if re.fullmatch(r"[A-Za-z][A-Za-z0-9_.#/-]*", value) is None:
        return False

    if re.search(r"[A-Za-z]", value) is None:
        return False

    if re.search(r"\d", value) is None:
        return False

    return True


def looks_like_inline_wiring_row(line):
    line = normalize_pdf_line(line)

    if line == "":
        return False

    match = re.match(
        r"^([A-Za-z][A-Za-z0-9_.#/-]*)\s+(\d+)\s+(\S+)(?:\s+.*)?$",
        line
    )

    if match is None:
        return False

    return is_connector_designator(match.group(1))


def is_pdf_junk_line(line):
    text = normalize_pdf_line(line)
    upper_text = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    exact_junk = {
        "FROM",
        "TO",
        "SIGNAL NAME",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE",
        "CODE",
        "WIRE CODE",
        "NOTE",
        "COMMENT",
        "WIRE CODES"
    }

    if upper_text in exact_junk:
        return True

    if upper_text.startswith("DES POSN DES POSN"):
        return True

    if upper_text.startswith("CODE NOTE COMMENT"):
        return True

    if "DES" in upper_text and "POSN" in upper_text and "MNEMONIC" in upper_text:
        return True

    if "MNEMONIC" in upper_text and "DESCRIPTION" in upper_text:
        return True

    if "WIRE CODE" in upper_text and "NOTE" in upper_text:
        return True

    return False


# =========================================
# FIND PDF RECORDS
#
# Supports both:
#
# J12 18 114 SPARE
#
# AND:
#
# J12
# 19
# 114
# MAG1-X-
# Magnetometer 1 X axis negative
# differential input
# IN
# =========================================

def find_pdf_records(input_text):
    raw_lines = input_text.split("\n")

    lines = []

    for raw_line in raw_lines:
        line = normalize_pdf_line(raw_line)

        if is_pdf_junk_line(line):
            continue

        lines.append(line)

    records = []
    current_record = None

    index = 0

    while index < len(lines):
        line = lines[index]

        # =========================================
        # INLINE RECORD
        #
        # J12 19 114 MAG1-X- ...
        # =========================================

        if looks_like_inline_wiring_row(line):

            if current_record is not None:
                records.append(current_record)

            match = re.match(
                r"^(\S+)\s+(\d+)\s+(\S+)(?:\s+(.*))?$",
                line
            )

            current_record = {
                "from_des": match.group(1),
                "from_posn": match.group(2),
                "to_des": match.group(3),
                "payload": []
            }

            remainder = match.group(4)

            if remainder is not None:
                remainder = remainder.strip()

                if remainder != "":
                    current_record["payload"].append(remainder)

            index += 1
            continue

        # =========================================
        # VERTICAL / BROKEN PDF RECORD
        #
        # J12
        # 19
        # 114
        # MAG1-X-
        # ...
        # =========================================

        if is_connector_designator(line):

            if index + 2 < len(lines):
                possible_pin = lines[index + 1]
                possible_destination = lines[index + 2]

                if is_integer_value(possible_pin):

                    if current_record is not None:
                        records.append(current_record)

                    current_record = {
                        "from_des": line,
                        "from_posn": possible_pin,
                        "to_des": possible_destination,
                        "payload": []
                    }

                    index += 3
                    continue

        # =========================================
        # CONTINUATION OF CURRENT ROW
        # =========================================

        if current_record is not None:
            current_record["payload"].append(line)

        index += 1

    if current_record is not None:
        records.append(current_record)

    return records


# =========================================
# PARSE CONTENT OF ONE PDF ROW
# =========================================

def parse_wiring_payload(payload_lines):
    to_posn = ""
    mnemonic = ""
    description = ""
    wire_code = ""
    note = ""
    comment = ""

    if len(payload_lines) == 0:
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    payload_text = " ".join(payload_lines)
    payload_text = normalize_pdf_line(payload_text)

    if payload_text == "":
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    tokens = payload_text.split()

    # =========================================
    # COMMENTS
    #
    # A SPARE row must produce a BLANK mnemonic.
    # =========================================

    comment_values = {
        "SPARE",
        "UNUSED",
        "NC",
        "N/C"
    }

    remaining = []

    for token in tokens:

        if token.upper() in comment_values:

            if comment == "":
                comment = token

        else:
            remaining.append(token)

    tokens = remaining

    if len(tokens) == 0:
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    # =========================================
    # NOTE / DIRECTION
    # =========================================

    note_values = {
        "IN",
        "OUT",
        "IN/OUT",
    }

    remaining = []

    for token in tokens:

        if token in note_values:

            if note == "":
                note = token

        else:
            remaining.append(token)

    tokens = remaining

    if note not in note_values:
        note = ""

    # =========================================
    # OPTIONAL TO POSITION
    # =========================================

    if len(tokens) >= 2:

        if is_integer_value(tokens[0]):
            to_posn = tokens.pop(0)

    # =========================================
    # MNEMONIC
    #
    # First remaining item is normally the
    # mnemonic.
    #
    # CTU RTN is a valid two-word mnemonic.
    # Preserve it instead of treating RTN as
    # the start of the description.
    # =========================================

    if len(tokens) >= 2 and tokens[0].upper() == "CTU" and tokens[1].upper() == "RTN":
        mnemonic = "CTU RTN"
        tokens = tokens[2:]
    elif len(tokens) >= 1 and tokens[0].upper() == "CTURTN":
        mnemonic = "CTU RTN"
        tokens.pop(0)
    elif len(tokens) >= 1:
        mnemonic = tokens.pop(0)

        if len(tokens) >= 1 and tokens[0].startswith("("):
            continuation_parts = []

            while len(tokens) >= 1:
                continuation_token = tokens.pop(0)
                continuation_parts.append(continuation_token)

                if continuation_token.endswith(")"):
                    break

            if len(continuation_parts) > 0:
                mnemonic = mnemonic + " " + " ".join(continuation_parts)

    # =========================================
    # DESCRIPTION
    #
    # Preserve all remaining text. This is
    # important because Acrobat frequently wraps
    # descriptions onto multiple clipboard lines.
    # =========================================

    if len(tokens) > 0:
        description = " ".join(tokens)

    return (
        to_posn,
        mnemonic,
        description,
        wire_code,
        note,
        comment
    )


# =========================================
# BUILD 9-COLUMN PDF TABLE
# =========================================

def parse_flat_pdf_wiring_table(input_text):
    records = find_pdf_records(input_text)

    rows = []

    rows.append([
        "FROM DES",
        "FROM POSN",
        "TO DES",
        "TO POSN",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE CODE",
        "NOTE",
        "COMMENT"
    ])

    for record in records:

        (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        ) = parse_wiring_payload(record["payload"])

        rows.append([
            record["from_des"],
            record["from_posn"],
            record["to_des"],
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        ])

    return rows


def looks_like_flat_pdf_wiring_table(input_text):
    records = find_pdf_records(input_text)

    manual_columns = get_manual_column_count()

    if manual_columns == 9 and len(records) >= 1:
        return True

    if len(records) >= 2:
        return True

    return False


# =========================================
# GENERAL TABLE PARSER
# =========================================

def parse_table():
    input_text = input_box.get("1.0", "end-1c")

    if input_text == "":
        return [], 0, "NONE"

    input_text = input_text.replace("\r\n", "\n")
    input_text = input_text.replace("\r", "\n")

    manual_columns = get_manual_column_count()

    rows = []
    delimiter_name = ""

    # =========================================
    # PDF 4-COLUMN SIMPLE / POWER PINOUT TABLE FIRST
    #
    # Example: Pin # | Signal Name | Signal Type | Signal Description.
    # This has no Ref Des and no Direction column.
    # =========================================

    if looks_like_pdf_simple_pinout_table(input_text):

        rows = parse_pdf_simple_pinout_table(input_text)

        if len(rows) > 1:
            label = "PDF SIMPLE PINOUT TABLE"
            if looks_highly_verticalized(input_text):
                label += " / VERTICAL PASTE"
            return rows, 4, label

    # =========================================
    # PDF 6-COLUMN CONNECTOR PINOUT TABLE
    #
    # This must run before the wiring-table and CSV checks. Pinout tables can
    # contain commas inside Description and can have a vertically merged Ref Des.
    # =========================================

    if looks_like_pdf_pinout_table(input_text):

        rows = parse_pdf_pinout_table(input_text)

        if len(rows) > 1:
            label = "PDF PINOUT TABLE"
            if looks_highly_verticalized(input_text):
                label += " / VERTICAL PASTE"
            return rows, 6, label

    # =========================================
    # PDF WIRING TABLE
    #
    # This MUST happen before generic TAB parsing because PDF clipboard data can
    # contain tabs without actually preserving table structure.
    # =========================================

    if looks_like_flat_pdf_wiring_table(input_text):

        rows = parse_flat_pdf_wiring_table(input_text)

        if len(rows) > 1:
            return rows, 9, "PDF WIRING TABLE"

    # =========================================
    # TRUE TAB-SEPARATED DATA
    # =========================================

    if "\t" in input_text:
        delimiter_name = "TAB"

        for line in input_text.split("\n"):
            rows.append(line.split("\t"))

    # =========================================
    # CSV
    #
    # Use structural CSV detection instead of `if "," in input_text` so a
    # normal Description such as "pair 4/D, negative" does not hijack parsing.
    # =========================================

    elif looks_like_real_csv(input_text):
        delimiter_name = "COMMA"

        reader = csv.reader(io.StringIO(input_text))

        for row in reader:
            rows.append(row)

    # =========================================
    # GENERIC SPACE-ALIGNED TABLE
    # =========================================

    elif manual_columns is not None and manual_columns > 1:
        delimiter_name = "SPACED TABLE"

        for line in input_text.split("\n"):

            if line.strip() == "":
                continue

            row = re.split(r" {2,}", line.strip())

            rows.append(row)

    # =========================================
    # SINGLE COLUMN
    # =========================================

    else:
        delimiter_name = "LINES"

        for line in input_text.split("\n"):
            rows.append([line])

    # =========================================
    # COLUMN COUNT
    # =========================================

    if manual_columns is not None:
        column_count = manual_columns

    else:
        column_count = 0

        for row in rows:

            if len(row) > column_count:
                column_count = len(row)

    # =========================================
    # NORMALIZE ALL ROW WIDTHS
    # =========================================

    normalized_rows = []

    for row in rows:

        normalized_row = list(row)

        while len(normalized_row) < column_count:
            normalized_row.append("")

        if len(normalized_row) > column_count:
            normalized_row = normalized_row[:column_count]

        normalized_rows.append(normalized_row)

    return normalized_rows, column_count, delimiter_name


# =========================================
# DETECT TABLE
# =========================================

def detect_table():
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        column_selector["values"] = []
        column_selector.set("")

        status_label.config(
            text="No table data found."
        )

        return

    column_choices = []

    first_row = rows[0]

    for index in range(column_count):
        header = ""

        if index < len(first_row):
            header = str(first_row[index]).strip()

        if header == "":
            column_choices.append(str(index + 1))

        else:
            column_choices.append(
                f"{index + 1}: {header}"
            )

    column_selector["values"] = column_choices

    if column_count > 0:
        column_selector.current(0)

    structured_pdf_table = (
        delimiter_name == "PDF WIRING TABLE"
        or delimiter_name.startswith("PDF PINOUT TABLE")
        or delimiter_name.startswith("PDF SIMPLE PINOUT TABLE")
    )

    detected_row_count = len(rows) - 1 if structured_pdf_table else len(rows)

    status_label.config(
        text=f"Detected {detected_row_count} rows x {column_count} columns | {delimiter_name}"
    )


def get_selected_column():
    selected = column_selector.get().strip()

    if selected == "":
        return None

    selected_number = selected.split(":")[0].strip()

    try:
        return int(selected_number)

    except ValueError:
        return None


# =========================================
# EXTRACT SELECTED COLUMN
# =========================================

def extract_column():
    global current_count
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        status_label.config(
            text="No table data found."
        )

        return

    selected_column = get_selected_column()

    if selected_column is None:

        detect_table()

        selected_column = get_selected_column()

    if selected_column is None:
        status_label.config(
            text="Select a column."
        )

        return

    if selected_column < 1 or selected_column > column_count:
        status_label.config(
            text=f"Column must be between 1 and {column_count}."
        )

        return

    start_row = 0

    if skip_header_var.get():
        start_row = 1

    output_values = []

    for row_index in range(start_row, len(rows)):
        row = rows[row_index]

        if selected_column - 1 < len(row):
            value = str(row[selected_column - 1]).strip()

        else:
            value = ""

        output_values.append(value)

    output_text = "\n".join(output_values)

    output_box.delete("1.0", tk.END)
    output_box.insert(tk.END, output_text)

    current_count = len(output_values)

    blank_count = 0

    for value in output_values:

        if value.strip() == "":
            blank_count += 1

    count_label.config(
        text=f"Items: {current_count}"
    )

    status_label.config(
        text=f"Column {selected_column} extracted | Blank cells: {blank_count}"
    )

    root.clipboard_clear()

    root.update()

    status_label.config(

def clear_all():
    global current_count
    global detected_table
    global detected_column_count

    input_box.delete("1.0", tk.END)
    output_box.delete("1.0", tk.END)

    column_count_entry.delete(0, tk.END)

    column_selector.set("")
    column_selector["values"] = []

    detected_table = []
    detected_column_count = 0

    current_count = 0

    count_label.config(
        text="Items: 0"
    )

    status_label.config(
        text=""
    )

# CONNECTOR NUMBER GENERATOR
# =========================================

def copy_connector():
    connector_number = connector_entry.get().strip()
    copy_count_text = copy_count_entry.get().strip()

    if connector_number == "":
        connector_status.config(
            text="Enter a connector number."
        )

        return

    if copy_count_text == "":
        copy_count = current_count

    else:

        try:
            copy_count = int(copy_count_text)

        except ValueError:
            connector_status.config(
                text="Copy count must be a number."
            )

            return

    if copy_count <= 0:
        connector_status.config(
            text="No items to copy."
        )

        return

    connector_output = "\n".join(
        [connector_number] * copy_count
    )

    root.clipboard_clear()
    root.clipboard_append(connector_output)
    root.update()

    connector_status.config(
        text=f"Copied {copy_count} copies of connector {connector_number}"
    )
