def make_gui(body):
    root = tk.Tk()
    root.title("Commit changes to database")

    choice = {"value": None}

    def pressed(value):
        choice["value"] = value
        root.quit()

    # Frame with scrollbar
    frame = tk.Frame(root)
    frame.pack(padx=10, pady=10)

    scrollbar = tk.Scrollbar(frame)
    scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

    text = tk.Text(frame, height=20, width=80, wrap="word",
                   yscrollcommand=scrollbar.set)
    text.pack(side=tk.LEFT, fill=tk.BOTH)

    scrollbar.config(command=text.yview)

    text.insert("1.0", body)
    text.config(state="disabled")

    tk.Button(root, text="COMMIT", width=50,
              command=lambda: pressed("YES")).pack(pady=5)
    tk.Button(root, text="ROLLBACK", width=50,
              command=lambda: pressed("NO")).pack(pady=5)

    root.mainloop()
    return choice["value"]



def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsx"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

# New sheet pinout mapping:
#   LEFT connector  = column L
#   LEFT pin        = column N
#   RIGHT pin       = column O
#   RIGHT connector = column Q
#
# Connector cells may only be populated at the beginning of a connector
# block, so forward-fill the most recent connector exactly as the new sheet
# layout requires.

def connector_number(value):
    if value is None:
        return None

    text = str(value).strip().upper().replace(" ", "")

    if text == "":
        return None

    # Accept P1, W1.P1, P1#..., or the old numeric form.
    if "." in text:
        text = text.split(".")[-1]

    if "#" in text:
        text = text.split("#", 1)[0]

    if text.startswith("P"):
        text = text[1:]

    return int(text)

def add_member(copy_no, part, version):
    cur.execute("""
    INSERT INTO dbo.Assembly
    (LPNTR, COPY, FLAGS, NAME, SUBENTRY, SUBVERSION)
    VALUES (?, ?, 2, '%.#', ?, ?)
    """, LPNTR, copy_no, part, version)

    cur.execute("""
    INSERT INTO dbo.GatePinRelation
    (ENTRY, GateNumber, PinNumber, Representation, NodeNumber)
    VALUES (?, ?, 0, 0, 0)
    """, LPNTR, copy_no)


def connect_wire(wire_copy, end1_copy, pin1, end2_copy, pin2):
    print(wire_copy, end1_copy, pin1, end2_copy, pin2)
    cur.execute("""
    INSERT INTO dbo.WIRTYP_CONNECT_INFOS
    (LPNTR, FLAGS, COPY, NODENO, COPYEND1, PINCOPYEND1, COPYEND2, PINCOPYEND2)
    VALUES (?, 0, ?, 4, ?, ?, ?, ?)
    """, LPNTR, wire_copy, end1_copy, pin1, end2_copy, pin2)

def connect_wire(wire_copy, node_no, end1_copy, pin1, end2_copy, pin2):
    print("Wire:", wire_copy, "Node:", node_no, "End1:", end1_copy,  "Pin1:", pin1,   "End2:", end2_copy, "Pin2:", pin2)

    INSERT INTO dbo.WIRTYP_CONNECT_INFOS
    (LPNTR, FLAGS, COPY, NODENO, COPYEND1, PINCOPYEND1, COPYEND2, PINCOPYEND2)
    VALUES (?, 0, ?, ?, ?, ?, ?, ?)
    """,
    LPNTR,
    wire_copy,  node_no, end1_copy, pin1, end2_copy, pin2)

def mark_component_updated():

def remove_template_member(part_name):
    cur.execute("""
    SELECT COPY
    FROM dbo.Assembly
    WHERE LPNTR = ?
    AND SUBENTRY = ?
    """, LPNTR, part_name)

    rows = cur.fetchall()

    if not rows:
        print(f"Template member not found: {part_name}")
        return

    for row in rows:
        copy_no = row[0]

        print(f"Removing template member COPY {copy_no}: {part_name}")

        # Remove any wire connections involving this member
        cur.execute("""
        DELETE FROM dbo.WIRTYP_CONNECT_INFOS
        WHERE LPNTR = ?
        AND (COPY = ? OR COPYEND1 = ? OR COPYEND2 = ?)
        """, LPNTR, copy_no, copy_no, copy_no)

        # Remove corresponding gate
        cur.execute("""
        DELETE FROM dbo.GatePinRelation
        WHERE ENTRY = ?
        AND GateNumber = ?
        """, LPNTR, copy_no)

        # Remove member from assembly
        cur.execute("""
        DELETE FROM dbo.Assembly
        WHERE LPNTR = ?
        AND COPY = ?
        """, LPNTR, copy_no)

    app.PutMessage("Template member removed.")

def get_pin_copy(connector_name, connector_version, physical_pin):
    cur.execute("""
    SELECT LPNTR
    FROM dbo.ComponentData
    WHERE ENTRY = ?
    AND VERSION = ?
    """, connector_name, connector_version)

    row = cur.fetchone()

    if not row:
        raise RuntimeError(f"Connector not found: {connector_name}")

    connector_lpntr = row[0]

    cur.execute("""
    SELECT COPY
    FROM dbo.Pin
    WHERE ENTRY = ?
    AND PHYSPIN = ?
    """, connector_lpntr, str(physical_pin))

    row = cur.fetchone()

    if not row:
        raise RuntimeError(f"Pin {physical_pin} not found on {connector_name}")

    return row[0]
