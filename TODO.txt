"""Read the separately maintained tool sources in Python and bundled EXE modes.

The canonical source of each tool is its .py file under tool_scripts/. No
code generation, eval, or hidden executable is used to restore its contents.
The main launcher uses SOURCE_FILES as a compatibility interface so the
existing runtime installation, hash, backup, and worker logic stay unchanged.
"""
from __future__ import annotations

from pathlib import Path
import sys

TOOL_SOURCE_PATHS = (
    'header_formatter/header_formatter.py',
    'signal_mover/signal_mover.py',
    'cable_pin_spacer/main.py',
    'cable_maker/cable_maker.py',
    'cable_placer/cable_placer.py',
    'block_maker/main.py',
    'block_maker/launcher.py',
    'block_maker/test.py',
    'block_maker/utils.py',
    'block_maker/error_popup.py',
    'block_pin_arranger/block_pin_arranger.py',
    'block_pin_formatter/block_pin_formatter.py',
    'block_signal_adder/block_signal_adder.py',
    'misc_tools/excel_formatter.py',
    'misc_tools/excel_diff.py',
    'misc_tools/missing_number_finder.py',
    'misc_tools/eicd_to_excel.py',
    'misc_tools/excel_to_pdf.py',
    'zuken_script_runner/appmain.py',
    'zuken_script_runner/AppGuiCode.py',
    'zuken_script_runner/utils.py',
    'zuken_script_runner/cabletable.py',
    'zuken_script_runner/wirelist.py',
    'zuken_script_runner/endtoend.py',
)

def tool_source_directory() -> Path:
    """Use PyInstaller's extracted data root, or the development project."""
    frozen_root = getattr(sys, "_MEIPASS", None)
    if frozen_root:
        return Path(frozen_root) / "tool_scripts"
    return Path(__file__).resolve().parent / "tool_scripts"

def load_sources() -> dict[str, str]:
    base = tool_source_directory()
    files = {}
    missing = []
    for relative in TOOL_SOURCE_PATHS:
        filename = base / relative
        if not filename.is_file():
            missing.append(relative)
            continue
        files[relative] = filename.read_text(encoding="utf-8")
    if missing:
        raise FileNotFoundError(
            "Missing tool source files under " + str(base) + ": " + ", ".join(missing)
            + ". Keep the entire project folder together during development; "
            + "when building the EXE, include --add-data tool_scripts;tool_scripts."
        )
    return files

SOURCE_FILES = load_sources()
