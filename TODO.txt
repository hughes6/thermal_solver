import time
import win32com.client as win32


SYMBOL_NAME = "Shield_Demo"
SYMBOL_VERSION = 1
SHEET_ID = 204828


app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sym = prj.CreateSymbolObject()


print("")
print("============================================================")
print("SHIELD SetId COMPARISON TEST")
print("============================================================")


print("")
print("TEST 1 - NO SetId BEFORE SetDisplayLength")
print("------------------------------------------------------------")

load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)
print("LOAD RESULT:", load_result)

placed_id_1 = sym.Place(SHEET_ID, 10.0, 2.0, 0, 1.0, 0)
print("PLACE RESULT:", placed_id_1)

time.sleep(0.10)

try:
    before_1 = sym.GetDisplayLength()
except Exception as error:
    before_1 = None
    print("GET LENGTH BEFORE ERROR:", repr(error))

print("LENGTH BEFORE:", before_1)

try:
    result_1 = sym.SetDisplayLength(2.0)
except Exception as error:
    result_1 = None
    print("SET LENGTH ERROR:", repr(error))

print("SET LENGTH RESULT:", result_1)

time.sleep(0.10)

try:
    after_1 = sym.GetDisplayLength()
except Exception as error:
    after_1 = None
    print("GET LENGTH AFTER ERROR:", repr(error))

print("LENGTH AFTER:", after_1)


print("")
print("TEST 2 - SetId BEFORE SetDisplayLength")
print("------------------------------------------------------------")

load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)
print("LOAD RESULT:", load_result)

placed_id_2 = sym.Place(SHEET_ID, 11.0, 2.0, 0, 1.0, 0)
print("PLACE RESULT:", placed_id_2)

time.sleep(0.10)

sym.SetId(placed_id_2)

try:
    before_2 = sym.GetDisplayLength()
except Exception as error:
    before_2 = None
    print("GET LENGTH BEFORE ERROR:", repr(error))

print("LENGTH BEFORE:", before_2)

try:
    result_2 = sym.SetDisplayLength(2.0)
except Exception as error:
    result_2 = None
    print("SET LENGTH ERROR:", repr(error))

print("SET LENGTH RESULT:", result_2)

time.sleep(0.10)

sym.SetId(placed_id_2)

try:
    after_2 = sym.GetDisplayLength()
except Exception as error:
    after_2 = None
    print("GET LENGTH AFTER ERROR:", repr(error))

print("LENGTH AFTER:", after_2)


print("")
print("============================================================")
print("DONE")
print("============================================================")


del sym
del prj
del app
