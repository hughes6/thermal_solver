def run_ct_code():
    try:
        app = win32.gencache.EnsureDispatch('CT.Application')
        prj = app.CreateJobObject()
        text = prj.CreateTextObject()
        dev = prj.CreateDeviceObject()
        sym = prj.CreateSymbolObject()

        # =====================================================
        # SCHEMATIC-VIEW ADDITIONAL PART ID
        #
        # Keep exactly ONE Additional Part ID across ALL split
        # schematic pin views.
        #
        # The text is attached to the overall top-left main pin
        # symbol, but moved to the horizontal midpoint between the
        # LEFT and RIGHT main pin columns and slightly above the
        # highest main pin.
        #
        # This prevents one Additional Part ID from appearing on
        # both the left and right sides.
        # =====================================================
        all_schematic_symbol_locations = {}

        for scan_dev_id in new_dev_ids:
            dev.SetId(scan_dev_id)

            scan_symbol_info = dev.GetSymbolIds(None)
            scan_symbol_ids = [
                x for x in scan_symbol_info[1]
                if x is not None
            ]

            for scan_symbol_id in scan_symbol_ids:
                sym.SetId(scan_symbol_id)

                xloc = [0.0]
                yloc = [0.0]
                grid = [""]
                column = [""]
                row = [""]

                try:
                    location = sym.GetSchemaLocation(
                        xloc,
                        yloc,
                        grid,
                        column,
                        row
                    )

                    if isinstance(location, tuple) and len(location) >= 3:
                        all_schematic_symbol_locations[scan_symbol_id] = (
                            float(location[1]),
                            float(location[2])
                        )
                except Exception:
                    pass

        schematic_additional_part_symbol_id = None
        schematic_additional_part_x = None
        schematic_additional_part_y = None

        # PINN_LOCS contains the actual main pin-column X locations
        # generated by the cable placer.  Spare pins use SPARE_X and
        # therefore are not included here.
        main_pin_x_values = sorted(set(
            round(float(location[1]), 6)
            for location in PINN_LOCS
        ))

        if all_schematic_symbol_locations and main_pin_x_values:
            left_main_x = min(main_pin_x_values)
            right_main_x = max(main_pin_x_values)

            # Center the single label between the two cable sides.
            schematic_additional_part_x = (
                left_main_x + right_main_x
            ) / 2.0

            main_symbol_ids = [
                symbol_id
                for symbol_id, location
                in all_schematic_symbol_locations.items()
                if (
                    abs(location[0] - left_main_x) <= 0.001
                    or abs(location[0] - right_main_x) <= 0.001
                )
            ]

            if main_symbol_ids:
                top_main_y = max(
                    all_schematic_symbol_locations[symbol_id][1]
                    for symbol_id in main_symbol_ids
                )

                # Keep the text just above the top pin.
                schematic_additional_part_y = top_main_y + 0.25

                # Use the top-most symbol on the LEFT side as the single
                # owner of the type-1000 text object.
                left_symbol_ids = [
                    symbol_id
                    for symbol_id in main_symbol_ids
                    if abs(
                        all_schematic_symbol_locations[symbol_id][0]
                        - left_main_x
                    ) <= 0.001
                ]

                if left_symbol_ids:
                    schematic_additional_part_symbol_id = max(
                        left_symbol_ids,
                        key=lambda symbol_id:
                            all_schematic_symbol_locations[symbol_id][1]
                    )
                else:
                    schematic_additional_part_symbol_id = max(
                        main_symbol_ids,
                        key=lambda symbol_id:
                            all_schematic_symbol_locations[symbol_id][1]
                    )

                print(
                    "SINGLE SCHEMATIC ADDITIONAL PART ID OWNER:",
                    schematic_additional_part_symbol_id
                )
                print(
                    "SCHEMATIC ADDITIONAL PART ID CENTER:",
                    schematic_additional_part_x,
                    schematic_additional_part_y
                )

        # =====================================================
        # LOOP THROUGH EACH NEW DEVICE VIEW
        # =====================================================

        for new_dev_id in new_dev_ids:
            connector = NEW_DEV_CONNECTORS.get(new_dev_id)

            print("\n======================================")
            print("NEW VIEW ID:", new_dev_id)
            print("CONNECTOR:", connector)
            print("======================================")

            if connector is None:
                print("WARNING: no connector mapping for view", new_dev_id)
                continue

            dev.SetId(new_dev_id)
            symbol_info = dev.GetSymbolIds(None)
            symbol_ids = [x for x in symbol_info[1] if x is not None]
            print("NEW VIEW SYMBOL IDS:", symbol_ids)

            for symbol_id in symbol_ids:
                sym.SetId(symbol_id)
                print("new view symbol:",  symbol_id, sym.GetName())
                texinfo = sym.GetTextIds(None)
                text_ids = [x for x in texinfo[1] if x is not None]

                for t_id in text_ids:
                    text.SetId(t_id)
                    text_type = text.GetType()

                    print(
                        "PIN SYMBOL:",
                        sym.GetName(),
                        "| text:",
                        text.GetText(),
                        "| type:",
                        text_type
                    )

                    if text_type == 5:
                        text.SetVisibility(0)

                    elif text_type == 1000:
                        if (
                            symbol_id
                            == schematic_additional_part_symbol_id
                            and schematic_additional_part_x is not None
                            and schematic_additional_part_y is not None
                        ):
                            app.PutMessage(
                                "Configuring single centered schematic "

                            )

                            text.SetVisibility(1)

                            text.SetHeight(0.2)

                            text.SetAlignment(2)
                            text.SetSchemaLocation(
                                schematic_additional_part_x,
                                schematic_additional_part_y
                            )
                        else:
                            # Every other split-view Additional Part ID,
                            # including the previous top-right one, is hidden.
                            text.SetVisibility(0)

                    if text_type == 12:
                        text.SetStyle(1)

            connector_id = CONNECTORS.get(connector)
            print("CONNECTOR:",  connector, "ID:",  connector_id)

                print("WARNING: connector ID not found for", connector)

            dev.SetId(connector_id)

def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsx"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def get_connector_pin_ids(dev, device_id, connector_name="", prefer_all=False):
    """
    Return the actual project pin IDs for a connector/device view.

    Base connector devices are queried with GetPinIds first because those are
    the physical connector pins.  Created schematic views are queried with
    GetAllPinIds first because their view pins are what PlacePins needs.

    The alternate method is used only as a fallback.  This keeps the old
    behavior available while fixing connectors for which GetAllPinIds returns
    an empty list on the base device.
    """
    dev.SetId(device_id)

    if prefer_all:
        method_names = ("GetAllPinIds", "GetPinIds")
    else:
        method_names = ("GetPinIds", "GetAllPinIds")

    for method_name in method_names:
        method = getattr(dev, method_name, None)
        if method is None:
            continue

        # E3 COM methods can differ between generated wrappers, so support
        # both the explicit output-array form and the no-argument form.
        for use_none in (True, False):
            try:
                result = method(None) if use_none else method()
            except Exception as error:
                print(
                    "PIN ENUMERATION FAILED:",
                    connector_name,
                    "DEVICE:", device_id,
                    "METHOD:", method_name,
                    "ERROR:", repr(error)
                )
                continue

            pin_ids = extract_e3_ids(result)

            print(
                "PIN ENUMERATION:",
                connector_name,
                "DEVICE:", device_id,
                "METHOD:", method_name,
                "RAW:", repr(result),
                "PIN IDS:", pin_ids
            )

            if pin_ids:
                return pin_ids

    return []


def connector_designation_from_device_name(value):
    """
    Convert an E3 connector name/reference to the workbook layout designation.

    Current cable-maker output uses W1.<designation>, e.g. W1.9B.  The older
    W1.9B / P9B form is still accepted for numeric designations.
    """
    return normalize_connector_token(value)


def generate_placement_locations(Pins, layout):
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    PIN_OFFSET_FROM_CONN = 3.0

    X_START = 1.0
    Y_START = 32.5
    main_conn_locs = [] # conn, x, y, flip

    # Center the shorter connector column vertically against the longer one.
    # Example:
    #   LEFT  = [P1, P3, P4]
    #   RIGHT = [P2]
    # gives P2 the same Y as P3.
    active_rows = [
        [conn for conn in row if conn is not None]
        for row in layout
    ]

    max_connector_count = max(
        (len(row) for row in active_rows),
        default=0
    )

    for i, row in enumerate(active_rows):
        vertical_offset = (max_connector_count - len(row)) / 2.0

        for j, conn in enumerate(row):
            flip = i > 0
            y_index = j + vertical_offset

            main_conn_locs.append([
                conn,
                X_START + (i * 3.0),
                Y_START - (y_index * 1.0),
                flip
            ])

    y_pin_start = 32.9
    for r in main_conn_locs:
        if y_pin_start > r[2]:
            y_pin_start = r[2]
    y_pin_start -= PIN_OFFSET_FROM_CONN

    print(Pins)
    print(layout)

    pin_locs = [] # conn, xstart, ystart, flip

    for dictionary in Pins:
        # Actual connector designation from the E3 device name.
        # Examples: W1.1 -> "1", W1.9B -> "9B".
        conn_key = list(dictionary.keys())[0]
        conn = connector_designation_from_device_name(conn_key)

        # number of pins = all keys except the "W1.<designation>"
        count = len(dictionary) - 1

        top = True
        flip = 0
        xstart = 0
        ystart = 0

        # find this connector's placement data
        for r in main_conn_locs:
            if str(r[0]) == conn:
                xstart = r[1]
                ystart = r[2]
                flip = r[3]

        # determine if connector is below another connector
        for r in main_conn_locs:
            if r[3] == flip:       # same flip side (top or bottom)
                if ystart < r[2]:  # this connector is below another
                    top = False

        if top:
            # normal top placement
            ystart = y_pin_start
        else:
            # below another connector:
            # new rule:
            #  p_start = top_p_start - pins_above*0.25 - conns_above*1

            pins_above = 0
            conns_above = 0

            # iterate over all pin dictionaries ABOVE our current connector
            for r in Pins:
                r_conn_key = list(r.keys())[0]
                r_conn = connector_designation_from_device_name(r_conn_key)

                if r_conn == conn:
                    break  # stop when reaching our connector

                # check if the connector is above AND same flip
                for c in main_conn_locs:
                    if str(c[0]) == r_conn and c[3] == flip:
                        if c[2] > ystart:
                            pins_above += (len(r) - 1)
                            conns_above += 1

            ystart = y_pin_start - (pins_above * 0.25) - (conns_above * 0.5)

        pin_locs.append([conn, xstart, ystart, flip])

    return main_conn_locs, pin_locs, y_pin_start + 2

def generate_aligned_pin_y(pinouts, y_start, pin_spacing=0.25, connector_gap=0.5):
    pin_y = {}

    current_y = y_start

    previous_c1 = None
    previous_c2 = None

    for row_index, (c1, p1, c2, p2) in enumerate(pinouts):
        if c1 is None or c2 is None:
            continue

        # Add a gap whenever we move into a new connector group
        # on EITHER side of the cable.
        if previous_c1 is not None:
            connector_changed = (
                c1 != previous_c1
                or c2 != previous_c2
            )

            if connector_changed:
                current_y -= connector_gap

        # Both ends of this wire get EXACTLY the same Y.
        pin_y[(str(c1), str(p1))] = current_y
        pin_y[(str(c2), str(p2))] = current_y

        current_y -= pin_spacing

        previous_c1 = c1
        previous_c2 = c2

    return pin_y

def get_spare_pins(pins, pinouts):
    spares = {}

    # Build quick connector -> pin dictionary lookup
    pin_lookup = {}

    for pindict in pins:
        connector = list(pindict.keys())[0]
        pin_lookup[connector] = pindict

    for c1, p1, c2, p2 in pinouts:

        # ==========================================
        # SPARE ON SIDE 1
        # ==========================================
        if c1 is not None and p1 is not None and c2 is None:
            connector = f"W1.{c1}"

            if connector not in spares:
                spares[connector] = {}

            if connector in pin_lookup:
                pin_id = pin_lookup[connector].get(str(p1))

                if pin_id is not None:

        # ==========================================

def clean_excel_value(value):
    if value is None:
        return ""

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def normalize_pin_name(value):
    return clean_excel_value(value).upper()


def normalize_connector_token(value):
    """
    Normalize an Excel/E3 connector reference to the layout designation only.

    Examples:
        3       -> "3"
        P3      -> "3"
        W1.3    -> "3"
        W1.P3   -> "3"
        9B      -> "9B"
        P9B     -> "9B"
        W1.9B   -> "9B"

    A leading P is removed only for the old P<number> form.  Names such as
    PMOD are left alone.
    """
    text = clean_excel_value(value).upper().replace(" ", "")

    if text == "":
        return ""

    root_prefix = CONNECTOR_ROOT_NAME.upper() + "."

    if text.startswith(root_prefix):
        text = text[len(root_prefix):]

    if "#" in text:
        text = text.split("#", 1)[0]

    match = re.fullmatch(r"P?0*(\d+)([A-Z]*)", text)

    if match:
        number = int(match.group(1))
        suffix = match.group(2)
        return f"{number}{suffix}"

    return text


def connector_base_name(connector_token):
    connector = normalize_connector_token(connector_token)

    if connector == "":
        return ""

    # Cable-maker assembly members are named %.<designation>, so E3 exposes
    # them as W1.<designation> (for example W1.3 and W1.9B), NOT W1.P3.
    return f"{CONNECTOR_ROOT_NAME}.{connector}"


def cell_has_bar(value):

# SPLIT INTO CONNECTOR SECTIONS
#====================================================================================================================================
def split_into_sections(rows):
    sections = []
    current = []

    for row in rows:
        if not current:
            current = [row]
            continue

        previous = current[-1]
        connector_changed = row["connector"] != previous["connector"]
        row_gap = row["excel_row"] != previous["excel_row"] + 1

        if connector_changed or row_gap:
            sections.append(current)
            current = [row]
        else:
            current.append(row)

    if current:
        sections.append(current)

    return sections

# CONTIGUOUS MARKED RANGES
#====================================================================================================================================
def true_ranges(rows, key):
    ranges = []
    start = None

    for index, row in enumerate(rows):
        marked = bool(row[key])

        if marked and start is None:
            start = index

        if not marked and start is not None:
            ranges.append((start, index - 1))
            start = None

    if start is not None:
        ranges.append((start, len(rows) - 1))

    return ranges


def range_pins(rows, start, end):
    pins = []

    for index in range(start, end + 1):
        pins.append(rows[index]["pin"])

    return pins


def range_excel_rows(rows, start, end):
    excel_rows = []

    for index in range(start, end + 1):
        excel_rows.append(rows[index]["excel_row"])

    return excel_rows



#====================================================================================================================================
# BUILD TWIST / INNER-SHIELD JOBS FOR ONE CONNECTOR SECTION
#
# INNER SHIELD RULE:
#   Inner SHLD does NOT create a standalone shield.
#   It only places Shield_Demo around a twist run when every wire in
#   that twist run is marked in the inner-SHLD column.
#
# OUTER SHIELDS ARE BUILT SEPARATELY ACROSS THE WHOLE SIDE SO THEY MAY
# CROSS CONNECTOR CHANGES (EXAMPLE: LEFT P1 -> P3 -> P4).
#====================================================================================================================================
def build_section_jobs(rows):
    jobs = []
    warnings = []

    if not rows:
        return jobs, warnings

    side = rows[0]["side"]
    connector = rows[0]["connector"]

    if connector == "":
        warnings.append(f"{side}: rows {rows[0]['excel_row']}-{rows[-1]['excel_row']} have pins but no connector")
        return jobs, warnings

    conflicts = []

    for row in rows:
        if row["twist_a"] and row["twist_b"]:
            conflicts.append(row["excel_row"])

    if conflicts:
        message = f"{side} {connector}: same wire marked in both twist columns on Excel rows {conflicts}"

        if STRICT_TWIST_VALIDATION:
            raise Exception(message)

        warnings.append(message)

    twist_jobs = []

    for key, twist_column in [("twist_a", "A"), ("twist_b", "B")]:
        ranges = true_ranges(rows, key)

        for start, end in ranges:
            pins = range_pins(rows, start, end)
            excel_rows = range_excel_rows(rows, start, end)

            if len(pins) < 2:
                warnings.append(
                    f"{side} {connector}: twist {twist_column} only contains one wire "
                    f"on Excel row {excel_rows[0]}"
                )
                continue

            job = {
                "type": "twist",
                "side": side,
                "connector": connector,
                "twist_column": twist_column,
                "start": start,
                "end": end,
                "pins": pins,
                "excel_rows": excel_rows,
                "symbol_name": TWIST_SYMBOL_NAME
            }

            twist_jobs.append(job)
            jobs.append(job)

    # Inner SHLD only wraps an existing twist. It never makes its own span.
    for twist_job in twist_jobs:
        indexes = list(range(twist_job["start"], twist_job["end"] + 1))
        inner_marks = [bool(rows[index]["inner_shield"]) for index in indexes]

        if all(inner_marks):
            jobs.append({
                "type": "shield_inner",
                "side": side,
                "connector": connector,
                "twist_column": twist_job["twist_column"],
                "pins": list(twist_job["pins"]),
                "excel_rows": list(twist_job["excel_rows"]),
                "symbol_name": SHIELD_SYMBOL_NAME
            })
        elif any(inner_marks):
            warnings.append(
                f"{side} {connector}: inner SHLD partially covers twist "
                f"{twist_job['twist_column']} on Excel rows {twist_job['excel_rows']}; "
                f"inner shield ignored because the full twist must be marked"
            )

    return jobs, warnings


#====================================================================================================================================
# BUILD OUTER-SHIELD JOBS ACROSS AN ENTIRE SIDE
#
# Connector changes DO NOT break an outer shield.
# Blank Excel row gaps or an unmarked outer-SHLD row DO break it.
#====================================================================================================================================
def build_outer_shield_jobs(rows):
    jobs = []

    if not rows:
        return jobs

    side = rows[0]["side"]
    start = None

    def append_job(start_index, end_index):
        covered_rows = rows[start_index:end_index + 1]

        connectors = []
        for row in covered_rows:
            connector = row["connector"]
            if connector and connector not in connectors:
                connectors.append(connector)

        jobs.append({
            "type": "shield_outer",
            "side": side,
            "connector": "/".join(connectors),
            "connectors": connectors,
            "rows": covered_rows,
            "pins": [row["pin"] for row in covered_rows],
            "excel_rows": [row["excel_row"] for row in covered_rows],
            "twist_column": None,
            "symbol_name": SHIELD_SYMBOL_NAME
        })

    for index, row in enumerate(rows):
        marked = bool(row["outer_shield"])

        if start is not None:
            previous = rows[index - 1]
            row_gap = row["excel_row"] != previous["excel_row"] + 1

            if row_gap or not marked:
                append_job(start, index - 1)
                start = None

        if marked and start is None:
            start = index

    if start is not None:
        append_job(start, len(rows) - 1)

    return jobs

# E3 ID EXTRACTOR
#====================================================================================================================================
def extract_e3_ids(result):
    if result is None:
        return []

    if isinstance(result, int):
        if result > 0:
            return [result]

        return []

    if isinstance(result, list):
        output = []

        for value in result:
            if value is not None:
                try:
                    if int(value) > 0:
                        output.append(int(value))
                except Exception:
                    pass

        return output

    if not isinstance(result, tuple):
        return []

    if len(result) == 0:
        return []

    if len(result) >= 2:
        values = result[1]

        if values is None:
            return []

        if isinstance(values, int):
            if values > 0:
                return [values]

            return []

        try:
            return [int(value) for value in values if value is not None and int(value) > 0]
        except Exception:
            pass

    output = []

    for value in result:
        try:
            if int(value) > 0:
                output.append(int(value))
        except Exception:
            pass

    return output

# COLLECT CONNECTOR CANDIDATES
#====================================================================================================================================
def collect_connector_candidate_ids(connector_token, all_devices, dev):
    expected_name = connector_base_name(connector_token)
    base_records = find_base_connector_devices(connector_token, all_devices)

    if not base_records:
        raise Exception(f"Could not find base E3 connector {expected_name}")

    candidate_sources = {}

    for record in base_records:
        base_id = record["device_id"]

        candidate_sources[base_id] = "BASE"

        view_ids = get_device_view_ids(dev, base_id)

        for view_id in view_ids:
            candidate_sources[view_id] = "VIEW"

        child_ids = get_child_device_ids(dev, base_id)

        for child_id in child_ids:
            if child_id not in candidate_sources:
                candidate_sources[child_id] = "CHILD"

        assembly_ids = get_same_name_assembly_devices(dev, base_id, expected_name)

        for assembly_device_id in assembly_ids:
            if assembly_device_id not in candidate_sources:
                candidate_sources[assembly_device_id] = "ASSEMBLY"

    return candidate_sources


#====================================================================================================================================
# ANALYZE CANDIDATE VIEW
#====================================================================================================================================
def analyze_candidate(required_pins, pin_map):
    sheets = {}

    for pin_name in required_pins:
        locations = pin_map.get(pin_name, [])

        for location in locations:
            sheet_id = location["sheet_id"]

            if sheet_id not in sheets:
                sheets[sheet_id] = {"pins": set(), "xs": [], "ys": []}

            sheets[sheet_id]["pins"].add(pin_name)
            sheets[sheet_id]["xs"].append(location["x"])
            sheets[sheet_id]["ys"].append(location["y"])

    if not sheets:
        return None

    results = []

    for sheet_id, data in sheets.items():
        placed_count = len(data["pins"])
        unique_x_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["xs"])
        unique_y_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["ys"])
        unique_x_count = len(unique_x_values)
        unique_y_count = len(unique_y_values)
        x_span = max(data["xs"]) - min(data["xs"]) if data["xs"] else 0.0
        y_span = max(data["ys"]) - min(data["ys"]) if data["ys"] else 0.0
        has_real_y_spread = y_span >= SCHEMATIC_MIN_Y_SPAN and unique_y_count > 1
        result = {"sheet_id": sheet_id, "placed_count": placed_count, "unique_x_count": unique_x_count, "unique_y_count": unique_y_count, "x_span": x_span, "y_span": y_span, "has_real_y_spread": has_real_y_spread}

        results.append(result)

    results.sort(key=lambda item: (item["has_real_y_spread"], item["placed_count"], item["unique_y_count"], item["y_span"]), reverse=True)

    return results[0]


#====================================================================================================================================
# RESOLVE REAL SCHEMATIC VIEW
#====================================================================================================================================
def resolve_connector_schematic_view(connector_token, required_pins, all_devices, dev, pin, cache):
    expected_name = connector_base_name(connector_token)
    candidate_sources = collect_connector_candidate_ids(connector_token, all_devices, dev)
    candidates = []

    print("")
    print("============================================================")
    print("E3 CANDIDATES FOR:", expected_name)
    print("============================================================")

    for candidate_id, source in candidate_sources.items():
        dev.SetId(candidate_id)

        try:
            candidate_name = clean_excel_value(dev.GetName())
        except Exception:
            candidate_name = "UNKNOWN"

        if candidate_id not in cache:
            try:
                cache[candidate_id] = load_device_pin_map(dev, pin, candidate_id)
            except Exception as error:
                print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "PIN MAP FAILED:", repr(error))
                continue

        pin_map = cache[candidate_id]
        analysis = analyze_candidate(required_pins, pin_map)

        if analysis is None:
            print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "NO PLACED REQUIRED PINS")
            continue

        print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "SHEET:", analysis["sheet_id"], "PLACED:", analysis["placed_count"], "UNIQUE X:", analysis["unique_x_count"], "UNIQUE Y:", analysis["unique_y_count"], "X SPAN:", analysis["x_span"], "Y SPAN:", analysis["y_span"])

        candidate = {"device_id": candidate_id, "name": candidate_name, "source": source, "pin_map": pin_map, "analysis": analysis}

        candidates.append(candidate)

    if not candidates:
        raise Exception(f"No usable E3 device/view candidates found for {expected_name}")

    candidates.sort(key=lambda item: (item["analysis"]["has_real_y_spread"], item["analysis"]["placed_count"], item["analysis"]["unique_y_count"], item["analysis"]["y_span"]), reverse=True)

    best = candidates[0]

    if not best["analysis"]["has_real_y_spread"]:
        raise Exception(f"Could not locate the split schematic view for {expected_name}")

    sheet_id = best["analysis"]["sheet_id"]
    missing_pins = []

    for pin_name in required_pins:
        locations = best["pin_map"].get(pin_name, [])
        found_on_sheet = False

        for location in locations:
            if location["sheet_id"] == sheet_id:
                found_on_sheet = True
                break

        if not found_on_sheet:
            missing_pins.append(pin_name)

    if missing_pins:
        raise Exception(f"Selected schematic view for {expected_name} is missing pins on sheet {sheet_id}: {missing_pins}")

    print("")
    print("SELECTED SCHEMATIC VIEW ID:", best["device_id"])
    print("SELECTED SOURCE:", best["source"])
    print("SELECTED NAME:", best["name"])
    print("SELECTED SHEET:", sheet_id)
    print("SELECTED PLACED PIN COUNT:", best["analysis"]["placed_count"])
    print("SELECTED UNIQUE Y COUNT:", best["analysis"]["unique_y_count"])
    print("SELECTED Y SPAN:", best["analysis"]["y_span"])
    print("============================================================")

    record = {"device_id": best["device_id"], "name": best["name"]}

    return record, best["pin_map"], sheet_id

# JOB X OFFSET
#====================================================================================================================================
def job_offset(job, twist_offset_settings):
    offset_enabled = twist_offset_settings[job["side"]]

    if job["type"] == "twist":
        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if job["type"] == "shield_inner":
        if job["twist_column"] is None:
            return INNER_SHIELD_ONLY_OFFSET

        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if not offset_enabled:
        return OFFSET_NO_OUTER_SHIELD

    return OFFSET_YES_OUTER_SHIELD


#====================================================================================================================================
# DRAW ORDER
#
# ACTUAL WORKING ORDER INSIDE EACH CONNECTOR SECTION:
# Inner shields first, then twists.
# Side-wide outer shields are processed separately before connector sections.
#====================================================================================================================================
def job_priority(job):
    # This matches the earlier working dynamic-placement implementation:
    # shields are created before twists.
    if job["type"] == "shield_outer":
        return 0

    if job["type"] == "shield_inner":
        return 1

    return 2


#====================================================================================================================================
# CALCULATE FINAL SYMBOL PLACEMENT
#====================================================================================================================================
def calculate_placement(job, pin_map, sheet_id, twist_offset_settings):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} has no schematic location on sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    if top_y - bottom_y < SCHEMATIC_MIN_Y_SPAN and len(job["pins"]) > 1:
        raise Exception(f"{job['side']} {job['connector']} {job['type']}: pins {job['pins']} all have effectively the same Y coordinate")

    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job, twist_offset_settings)
    x = anchor_x + direction * offset

    if job["type"] == "twist":
        y = bottom_y - TWIST_BOTTOM_OVERLAP
        length = None
    else:
        y = bottom_y - SHIELD_BOTTOM_OVERLAP
        length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

        if length < MIN_SYMBOL_LENGTH:
            length = MIN_SYMBOL_LENGTH

    placement = {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y,
        "direction": direction,
        "offset_enabled": twist_offset_settings[job["side"]]
    }

    return placement



#====================================================================================================================================
# CALCULATE OUTER SHIELD PLACEMENT ACROSS ONE OR MORE CONNECTORS
#
# Each Excel row is resolved through the connector named on that row.
# This allows one outer shield to span, for example, LEFT P1 + P3 + P4.
# All covered connector views must resolve to the same E3 sheet.
#====================================================================================================================================
def calculate_outer_shield_placement(job, all_devices, dev, pin, cache, twist_offset_settings):
    resolved_by_connector = {}
    sheet_ids = set()
    locations = []

    rows_by_connector = {}

    for row in job["rows"]:
        connector = row["connector"]

        if connector == "":
            raise Exception(
                f"{job['side']} outer shield: Excel row {row['excel_row']} has no connector"
            )

        rows_by_connector.setdefault(connector, []).append(row)

    for connector, connector_rows in rows_by_connector.items():
        required_pins = list(dict.fromkeys(row["pin"] for row in connector_rows))

        device_record, pin_map, sheet_id = resolve_connector_schematic_view(
            connector,
            required_pins,
            all_devices,
            dev,
            pin,
            cache
        )

        resolved_by_connector[connector] = {
            "device_record": device_record,
            "pin_map": pin_map,
            "sheet_id": sheet_id
        }

        sheet_ids.add(sheet_id)

    if len(sheet_ids) != 1:
        details = ", ".join(
            f"{connector}=sheet {data['sheet_id']}"
            for connector, data in resolved_by_connector.items()
        )
        raise Exception(
            f"{job['side']} outer shield spans connectors on different E3 sheets: {details}"
        )

    sheet_id = next(iter(sheet_ids))

    for row in job["rows"]:
        connector = row["connector"]
        pin_name = row["pin"]
        pin_map = resolved_by_connector[connector]["pin_map"]

        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(
                f"{job['side']} outer shield: {connector} pin {pin_name} "
                f"has no schematic location on sheet {sheet_id}"
            )

        locations.append(location)

    if not locations:
        raise Exception(f"{job['side']} outer shield has no resolved schematic locations")

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    direction = 1.0 if job["side"] == "LEFT" else -1.0

    if twist_offset_settings[job["side"]]:
        offset = OFFSET_YES_OUTER_SHIELD
    else:
        offset = OFFSET_NO_OUTER_SHIELD

    x = anchor_x + direction * offset
    y = bottom_y - SHIELD_BOTTOM_OVERLAP
    length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

    if length < MIN_SYMBOL_LENGTH:
        length = MIN_SYMBOL_LENGTH

    return {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y,
        "direction": direction,
        "offset_enabled": twist_offset_settings[job["side"]]
    }

# TRY TO SIZE SHIELD AT CURRENT LOCATION
#====================================================================================================================================
def try_size_shield(sym, symbol_id, requested_length):
    # EXACT KNOWN-GOOD E3 SEQUENCE:
    #   Place -> SetId(placed id) -> GetDisplayLength ->
    #   SetDisplayLength -> SetId(placed id) -> GetDisplayLength
    sym.SetId(symbol_id)

    try:
        before = float(sym.GetDisplayLength())
    except Exception:
        before = None

    print("SHIELD LENGTH BEFORE:", before)

    try:
        stretch_result = sym.SetDisplayLength(float(requested_length))
    except Exception as error:
        stretch_result = None
        print("SHIELD STRETCH EXCEPTION:", repr(error))

    print("SHIELD STRETCH RESULT:", stretch_result)

    # Re-bind exactly like the standalone test that successfully resized.
    sym.SetId(symbol_id)
    time.sleep(STRETCH_DELAY_SECONDS)

    try:
        after = float(sym.GetDisplayLength())
    except Exception:
        after = None

    print("SHIELD REQUESTED LENGTH:", requested_length)
    print("SHIELD LENGTH AFTER:", after)

    return after is not None and abs(after - float(requested_length)) <= 0.001


def shield_stage_y_values(requested_length):
    # Shield_Demo grows upward from its insertion point.
    # 10.0 at Y=22.0 was proven to resize successfully in the earlier test.
    # For longer shields, lower the staging Y enough to keep the full symbol on-sheet.
    safe_y = min(
        SHIELD_STAGE_DEFAULT_Y,
        SHIELD_STAGE_TOP_LIMIT - float(requested_length) - SHIELD_STAGE_MARGIN
    )
    safe_y = max(0.75, safe_y)

    candidates = [
        safe_y,
        max(0.75, safe_y - 0.50),
        min(SHIELD_STAGE_DEFAULT_Y, safe_y + 0.50),
        max(0.75, safe_y - 1.00),
        min(SHIELD_STAGE_DEFAULT_Y, safe_y + 1.00),
    ]

    output = []
    for value in candidates:
        value = round(float(value), 3)
        if value not in output:
            output.append(value)

    return output

#====================================================================================================================================
# PLACE SHIELD
#
# PROCESS:
#
# 1. Load Shield_Demo.
# 2. Search staging coordinates until the symbol can be placed.
# 3. Try SetDisplayLength at that staging position.
# 4. If resize fails, move SAME shield to another staging location.
# 5. Once correct size is reached, move SAME shield to final cable X/Y.
#
# No duplicate shields are intentionally created while searching.
#====================================================================================================================================
def place_shield(prj, job, placement):
    """
    Place a REAL Shield_Demo symbol and resize it with SetDisplayLength.

    CRITICAL E3 BEHAVIOR:
    The earlier working test called SetDisplayLength IMMEDIATELY after Place()
    on the SAME symbol object returned from Load/Place.

    Do NOT call SetId() first.
    Do NOT call GetDisplayLength() first.
    Do NOT use PlaceDynamic().

    Calling SetId before SetDisplayLength was the important difference in the
    later broken versions.
    """
    requested_length = float(placement["length"])
    final_x = float(placement["x"])
    final_y = float(placement["y"])
    sheet_id = int(placement["sheet_id"])

    print("")
    print("SHIELD FINAL X:", final_x)
    print("SHIELD FINAL Y:", final_y)
    print("SHIELD REQUESTED LENGTH:", requested_length)
    print("SHIELD LIBRARY SYMBOL:", job["symbol_name"])
    print("SHIELD MODE: Load -> Place -> IMMEDIATE SetDisplayLength")
    print("SHIELD NOTE: NO SetId / GetDisplayLength BEFORE STRETCH")

    # Try the final location first.  If E3 rejects the resize there, try the
    # same exact sequence in a clear staging area and move the resized shield.
    attempts = [("FINAL", final_x, final_y)]

    # Pick stage Y values that leave enough room for a long shield.
    max_stage_y = SHIELD_STAGE_TOP_LIMIT - requested_length - SHIELD_STAGE_MARGIN
    if max_stage_y < 0.5:
        max_stage_y = 0.5

    stage_base_y = min(SHIELD_STAGE_DEFAULT_Y, max_stage_y)
    stage_y_values = [
        stage_base_y,
        max(0.5, stage_base_y - 0.5),
        max(0.5, stage_base_y + 0.5),
        max(0.5, stage_base_y - 1.0),
        max(0.5, stage_base_y + 1.0),
    ]

    seen_stage_y = []
    for y in stage_y_values:
        y = round(float(y), 6)
        if y not in seen_stage_y:
            seen_stage_y.append(y)

    for stage_y in seen_stage_y:
        for stage_x in SHIELD_STAGE_X_VALUES:
            attempts.append(("STAGE", float(stage_x), float(stage_y)))

    for attempt_number, (mode, x, y) in enumerate(attempts, start=1):
        print("")
        print("SHIELD ATTEMPT:", attempt_number, "MODE:", mode, "X:", x, "Y:", y)

        # Fresh object for each attempt.
        shield_sym = prj.CreateSymbolObject()

        load_result = shield_sym.Load(job["symbol_name"], SYMBOL_VERSION)
        print("SHIELD LOAD RESULT:", load_result)

        if load_result == 0:
            try:
                del shield_sym
            except Exception:
                pass
            continue

        placed_id = shield_sym.Place(sheet_id, x, y, 0, 1.0, 0)
        print("SHIELD PLACE RESULT:", placed_id)

        if placed_id == 0:
            try:
                del shield_sym
            except Exception:
                pass
            time.sleep(PLACEMENT_DELAY_SECONDS)
            continue

        # ================================================================
        # THIS IS THE KEY FIX.
        #
        # The old working test did:
        #     Load(...)
        #     Place(...)
        #     SetDisplayLength(...)
        #
        # It did NOT SetId() between Place and SetDisplayLength.
        # It did NOT read GetDisplayLength() before SetDisplayLength.
        # ================================================================
        try:
            stretch_result = shield_sym.SetDisplayLength(requested_length)
        except Exception as error:
            stretch_result = None
            print("SHIELD IMMEDIATE STRETCH EXCEPTION:", repr(error))

        print("SHIELD IMMEDIATE STRETCH RESULT:", stretch_result)

        time.sleep(STRETCH_DELAY_SECONDS)

        try:
            immediate_length = float(shield_sym.GetDisplayLength())
        except Exception:
            immediate_length = None

        print("SHIELD IMMEDIATE LENGTH AFTER:", immediate_length)

        immediate_success = (
            immediate_length is not None
            and abs(immediate_length - requested_length) <= 0.001
        )

        # Only NOW bind by ID, after the resize attempt is already complete.
        try:
            set_id_result = shield_sym.SetId(placed_id)
            print("SHIELD POST-STRETCH SetId RESULT:", set_id_result)
        except Exception as error:
            print("SHIELD POST-STRETCH SetId EXCEPTION:", repr(error))

        try:
            verified_length = float(shield_sym.GetDisplayLength())
        except Exception:
            verified_length = immediate_length

        print("SHIELD VERIFIED LENGTH:", verified_length)

        verified_success = (
            verified_length is not None
            and abs(verified_length - requested_length) <= 0.001
        )

        if immediate_success or verified_success:
            print("SHIELD RESIZE SUCCESS")

            final_id = placed_id

            # If resized in staging, move that SAME real shield to final X/Y.
            if mode == "STAGE":
                time.sleep(MOVE_DELAY_SECONDS)

                try:
                    shield_sym.SetId(placed_id)
                except Exception:
                    pass

                move_result = shield_sym.Place(
                    sheet_id,
                    final_x,
                    final_y,
                    0,
                    1.0,
                    0
                )

                print("SHIELD FINAL MOVE RESULT:", move_result)

                if move_result == 0:
                    raise Exception(
                        f"{job['symbol_name']} resized correctly but could not "
                        f"move to final X={final_x} Y={final_y}"
                    )

                final_id = move_result
                time.sleep(PLACEMENT_DELAY_SECONDS)

            # Final verification.
            try:
                shield_sym.SetId(final_id)
            except Exception:
                pass

            try:
                final_length = float(shield_sym.GetDisplayLength())
            except Exception:
                final_length = None

            try:
                final_name = shield_sym.GetName()
            except Exception:
                final_name = "UNKNOWN"

            try:
                final_type = shield_sym.GetSymbolType()
            except Exception:
                final_type = "UNKNOWN"

            try:
                final_is_shield = shield_sym.IsShield()
            except Exception:
                final_is_shield = "UNKNOWN"

            final_location = get_symbol_location(shield_sym, final_id)

            print("SHIELD FINAL ID:", final_id)
            print("SHIELD FINAL NAME:", final_name)
            print("SHIELD FINAL SYMBOL TYPE:", final_type)
            print("SHIELD FINAL IS SHIELD:", final_is_shield)
            print("SHIELD FINAL DISPLAY LENGTH:", final_length)

            if final_location is not None:
                print("SHIELD FINAL SHEET:", final_location["sheet_id"])
                print("SHIELD FINAL ACTUAL X:", final_location["x"])
                print("SHIELD FINAL ACTUAL Y:", final_location["y"])

            if final_length is None or abs(final_length - requested_length) > 0.001:
                raise Exception(
                    f"{job['symbol_name']} resized successfully but final length "
                    f"changed. Expected {requested_length}, actual {final_length}"
                )

            print("SHIELD REAL SYMBOL + LENGTH VERIFIED")
            return final_id

        print("SHIELD RESIZE FAILED FOR THIS ATTEMPT")

        # Remove the failed trial before trying another position.
        try:
            shield_sym.SetId(placed_id)
            delete_result = shield_sym.Delete()
            print("FAILED SHIELD DELETE RESULT:", delete_result)
        except Exception as error:
            print("FAILED SHIELD DELETE EXCEPTION:", repr(error))

        try:
            del shield_sym
        except Exception:
            pass

        time.sleep(PLACEMENT_DELAY_SECONDS)

    # ================================================================
    # MANUAL-STRETCH FALLBACK
    #
    # If E3 refuses every automatic SetDisplayLength attempt, do not
    # abandon the shield.  Place one normal, unstretched Shield_Demo at
    # the correct FINAL bottom insertion point.  The user can stretch it
    # manually in E3 after the script finishes.
    #
    # All failed trial shields above were already deleted, so this leaves
    # exactly one fallback shield at the intended cable location.
    # ================================================================
    print("")
    print("============================================================")
    print("SHIELD AUTO-RESIZE FAILED - USING MANUAL-STRETCH FALLBACK")
    print("============================================================")
    print("FALLBACK SYMBOL:", job["symbol_name"])
    print("FALLBACK X:", final_x)
    print("FALLBACK BOTTOM Y:", final_y)
    print("TARGET LENGTH FOR MANUAL STRETCH:", requested_length)

    fallback_sym = prj.CreateSymbolObject()

    fallback_load_result = fallback_sym.Load(
        job["symbol_name"],
        SYMBOL_VERSION
    )

    print("FALLBACK LOAD RESULT:", fallback_load_result)

    if fallback_load_result == 0:
        raise Exception(
            f"{job['symbol_name']} could not resize and the fallback symbol "
            f"could not be loaded."
        )

    fallback_id = fallback_sym.Place(
        sheet_id,
        final_x,
        final_y,
        0,
        1.0,
        0
    )

    print("FALLBACK PLACE RESULT:", fallback_id)

    if fallback_id == 0:
        raise Exception(
            f"{job['symbol_name']} could not resize and the unstretched "
            f"fallback could not be placed at X={final_x} Y={final_y}."
        )

    time.sleep(PLACEMENT_DELAY_SECONDS)

    try:
        fallback_sym.SetId(fallback_id)
    except Exception:
        pass

    try:
        fallback_length = float(fallback_sym.GetDisplayLength())
    except Exception:
        fallback_length = None

    print("FALLBACK NATIVE LENGTH:", fallback_length)
    print("MANUAL ACTION REQUIRED: stretch this shield upward to:", requested_length)
    print("SHIELD FALLBACK PLACED AT CORRECT BOTTOM LOCATION")
    print("============================================================")

    return fallback_id


#====================================================================================================================================
# PLACE ONE JOB
#====================================================================================================================================
def place_job(prj, sym, job, placement):
    print("")
    print("------------------------------------------------------------")

def placer_connector_designation(value):
    """
    Return the actual connector layout designation as a string.

    Accepts 9, 9B, P9B, W1.9B, or W1.9B.  The canonical result is 9B.

            "DEVICE ID:", device_id,
            "VIEW NUMBER:", candidate_view_number,
            "RESULT:", res
        )

        if res != 0:
            used_view_number = candidate_view_number
            break

    if res == 0:
        app.PutMessage(f"Step 4 not working. Result: {res}\n")
        raise Exception(
            f"Step 4 failed. No new device view created for {connector} "
            f"(device ID {device_id}) using view numbers 1-10.\n"
        )
    else:
        app.PutMessage(
            f"Step 4 success. New device view placed. "
            f"Connector: {connector}, View number: {used_view_number}, Result: {res}"
        )

    new_device_view_id = res
    new_dev_ids.append(new_device_view_id)
    NEW_DEV_CONNECTORS[new_device_view_id] = connector
    print("new_device_view_id ", res)
    # The created schematic view normally exposes its view pins through
    # GetAllPinIds.  Fall back to GetPinIds if this E3 object exposes them
    # directly instead.
    newpinids = get_connector_pin_ids(
        dev,
        new_device_view_id,
        connector_name=f"{connector} VIEW",
        prefer_all=True
    )

    if not newpinids:
        raise Exception(
            f"No pins found in created schematic view for {connector} "
            f"(view ID {new_device_view_id})."
        )

    newpindict = {}
    for pin_id in newpinids:
        pin.SetId(pin_id)
        pin_name = str(pin.GetName()).strip()
        newpindict[pin_name] = pin_id

    print("SCHEMATIC VIEW PINS:", connector, newpindict)

    # PINS ARE GETTING ORDERED HERE
    p = connector_designation_from_device_name(connector)
    ordered_pin_ids = generate_ordered_pin_ids(newpindict, PINORDER[p])

    for c in PINN_LOCS:
        if str(c[0]) == p:
            xp = c[1]
            yp = c[2]
            pflip = c[3]
    if pflip == 1:
        pflip = "/1"
    else:
        pflip = 0

    symname = "MilSymbolReceptacle¶END_SPLIT"
    for key, value in ordered_pin_ids.items():
        pin_key = (str(p), str(key))
        if pin_key not in PIN_Y:
            app.PutMessage(f"WARNING: No aligned Y position found forConnector {p} Pin {key}")
            continue
        pin_y = PIN_Y[pin_key]
        res = sym.PlacePins(value, symname, ver,  sheet_1_id, xp, pin_y, pflip, 1.0)
        app.PutMessage(f"placing pin: {key} id: {value} symbol: {symname} y: {pin_y} x: {xp}, rot: {pflip} conn: {p}")
        symname = "MilSymbolReceptacle¶SPLIT_BOTTOM"
        time.sleep(0.5)

    connector_spares = SPARES.get(connector, {})
    if connector_spares:
        symname = "MilSymbolReceptacle¶END_SPLIT"
        app.PutMessage(f"placing {len(connector_spares)} spare pins for connector {connector}")
        for spare_pin_name in connector_spares.keys():
            # IMPORTANT: Use the pin ID from the NEW DEVICE VIEW, not the original connector pin ID.
            spare_pin_id = newpindict.get(str(spare_pin_name))
            if spare_pin_id is None:
                app.PutMessage(f"WARNING: Could not find spare pin {spare_pin_name} in new view for {connector}")
                continue
            res = sym.PlacePins(spare_pin_id, symname, ver, sheet_1_id, SPARE_X, SPARE_Y, 0, 1.0)
            app.PutMessage(f"placing spare pin: {spare_pin_name} id: {spare_pin_id} symbol: {symname} y: {SPARE_Y} x: {SPARE_X}, rot: {0} conn: {connector}")
            if res == 0:
                app.PutMessage(f"WARNING: Spare pin placement failed for {connector} pin {spare_pin_name} id {spare_pin_id}")
            SPARE_Y -= SPARE_PIN_SPACING
            symname = "MilSymbolReceptacle¶SPLIT_BOTTOM"
            time.sleep(0.55)
            app.PutMessage(f"Result: {res}")
        SPARE_Y -= SPARE_CONN_GAP



    dev.SetId(new_device_view_id)
    symbol_info = dev.GetSymbolIds(None)
    symbol_ids = [x for x in symbol_info[1] if x is not None]
    for id in symbol_ids:
        sym.SetId(id)
        texinfo = sym.GetTextIds(None)
        text_ids = [x for x in texinfo[1] if x is not None]
        for t_id in text_ids:
            text.SetId(t_id)
            if text.GetType() == 5:
                text.SetVisibility(0)
            if text.GetType() == 12:
                text.SetStyle(1) # bold
            

    dev.SetId(dev.GetAssemblyId())
    assydevinfo = dev.GetDeviceIds(None)
    assydevids = [x for x in assydevinfo[1] if x is not None]
    for assydev_id in assydevids:
        dev.SetId(assydev_id)
        if(dev.GetName() == connector): # if it is W1.<designation>
            assydevsymbolinfo = dev.GetSymbolIds(None)
            assydevsymbolids = [x for x in assydevsymbolinfo[1] if x is not None]
            for assydevsymbol_id in assydevsymbolids:
                sym.SetId(assydevsymbol_id)
                assydevsymboltextinfo = sym.GetTextIds(None)
                assydevsymboltextids = [x for x in assydevsymboltextinfo[1] if x is not None]
                for assydevsymboltext_id in assydevsymboltextids:
                    text.SetId(assydevsymboltext_id)
                    if text.GetType() == 5:
                        text.SetVisibility(0)
                    if text.GetType() == 12:
                        text.SetStyle(1)


    j += 1
