        if current_pin_info is not None:
            signal = current_pin_info[1] if len(current_pin_info) > 1 else None
            in_out = current_pin_info[2] if len(current_pin_info) > 2 else None
        else:
            signal = None
            in_out = None

        if signal is not None:
            pin.SetAttributeValue("Local Signal Name", signal)

        if in_out is not None:
            pin.SetAttributeValue("Signal Types", in_out)
