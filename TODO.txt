"""Run under xvfb-run in Linux for safe GUI routing tests. No real E3 calls."""
from pathlib import Path
import tempfile
import sys
import tkinter as tk
ROOT=Path(__file__).resolve().parents[1]
sys.path.insert(0,str(ROOT))
import mega_launcher as launcher

if __name__=='__main__':
    with tempfile.TemporaryDirectory() as temp:
        launcher.runtime_directory=lambda:Path(temp)
        # Exercise the packaged EXE direct-launch route, not Python subprocesses.
        launcher.running_as_packaged_exe=lambda:True
        called=[]
        def simulated_tool(key, args):
            called.append((key,args))
            window=tk.Tk()
            window.title('Simulated '+key)
            window.after(30,window.destroy)
            window.mainloop()
        launcher.run_embedded_tool=simulated_tool
        app=launcher.MegaApp()
        keys=[k for k in launcher.TOOLS if k in launcher.FROZEN_INPROCESS_TOOLS]
        for key in keys:
            def begin(k=key):
                assert app.running_local is None, ('not restored',k)
                app.launch(k)
            # stagger to allow each simulated window to open and close
            app.root.after(300 + keys.index(key)*240,begin)
        def finish():
            assert app.running_local is None
            assert [k for k,args in called]==keys,(keys,called)
            assert app.root.state()=='normal'
            app.root.destroy()
        app.root.after(300 + len(keys)*240,finish)
        app.root.mainloop()
        print('GUI PASS: all 14 Tkinter tool handoffs; wxPython Runner stays isolated')
