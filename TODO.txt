import re
import tkinter as tk
from tkinter import scrolledtext
from tkinter import messagebox


# ================================================
# CONFIG
# ================================================
WATTS_PATTERN = re.compile(r'(?m)^([ \t]*watts[ \t]*=[ \t]*)([-+]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][-+]?\d+)?)([ \t]*(?:#.*)?)$')


# ================================================
# NUMBER HELPERS
# ================================================
def parse_percent(value, field_name):
    text = value.strip().replace("%", "")

    if text == "":
        raise ValueError(f"{field_name} is required.")

    try:
        number = float(text)
    except ValueError:
        raise ValueError(f"{field_name} must be a number.")

    if number < 0:
        raise ValueError(f"{field_name} cannot be negative.")

    return number


def format_scaled_number(value):
    if abs(value) < 1e-12:
        value = 0.0

    if abs(value - round(value)) < 1e-10:
        return str(int(round(value)))

    return f"{value:.12g}"


# ================================================
# TOML SCALING
# ================================================
def scale_toml_watts(toml_text, current_percent, desired_percent):
    if current_percent == 0:
        raise ValueError("Initial/current load percent cannot be 0.")

    scale_factor = desired_percent / current_percent
    replacement_count = 0

    def replace_watts(match):
        nonlocal replacement_count

        prefix = match.group(1)
        watts_text = match.group(2)
        suffix = match.group(3)

        current_watts = float(watts_text)
        desired_watts = current_watts * scale_factor
        new_watts_text = format_scaled_number(desired_watts)

        replacement_count += 1

        return prefix + new_watts_text + suffix

    output_text = WATTS_PATTERN.sub(replace_watts, toml_text)

    return output_text, replacement_count, scale_factor


# ================================================
# GUI ACTIONS
# ================================================
def convert_toml():
    toml_text = input_box.get("1.0", "end-1c")

    if toml_text.strip() == "":
        status_label.config(text="Paste a TOML file first.")
        return

    try:
        current_percent = parse_percent(current_percent_entry.get(), "Initial/current percent")
        desired_percent = parse_percent(desired_percent_entry.get(), "Desired percent")
        output_text, replacement_count, scale_factor = scale_toml_watts(toml_text, current_percent, desired_percent)
    except ValueError as error:
        status_label.config(text=str(error))
        messagebox.showerror("Invalid Input", str(error))
        return

    output_box.delete("1.0", tk.END)
    output_box.insert("1.0", output_text)

    status_label.config(text=f"Updated {replacement_count} watts entries | Scale factor: {scale_factor:.6g}")

    if replacement_count == 0:
        messagebox.showwarning("No watts entries found", "No numeric lines matching watts = # were found. The TOML text was otherwise left unchanged.")


def copy_output():
    output_text = output_box.get("1.0", "end-1c")

    if output_text == "":
        status_label.config(text="Nothing to copy.")
        return

    root.clipboard_clear()
    root.clipboard_append(output_text)
    root.update()

    status_label.config(text="Scaled TOML copied to clipboard.")


def clear_all():
    input_box.delete("1.0", tk.END)
    output_box.delete("1.0", tk.END)
    current_percent_entry.delete(0, tk.END)
    desired_percent_entry.delete(0, tk.END)
    status_label.config(text="")


def use_example():
    example = """[rack.server_1]
name = "Server 1"
watts = 8
enabled = true

[rack.server_2]
name = "Server 2"
watts = 16 # original load
fans = 4

[other]
watts = 2.5
note = "Everything except watts stays exactly the same."
"""

    input_box.delete("1.0", tk.END)
    input_box.insert("1.0", example)

    current_percent_entry.delete(0, tk.END)
    current_percent_entry.insert(0, "80")

    desired_percent_entry.delete(0, tk.END)
    desired_percent_entry.insert(0, "20")

    status_label.config(text="Example loaded. Click Scale Watts.")


# ================================================
# GUI
# ================================================
root = tk.Tk()
root.title("TOML Watts Load Scaler")
root.geometry("900x760")
root.minsize(760, 620)

title_label = tk.Label(root, text="TOML Watts Load Scaler", font=("Arial", 15, "bold"))
title_label.pack(pady=(12, 3))

description_label = tk.Label(root, text="Preserves the TOML text and only rescales numeric watts = values.")
description_label.pack(pady=(0, 10))

percent_frame = tk.Frame(root)
percent_frame.pack(pady=4)

current_percent_label = tk.Label(percent_frame, text="Initial / Current Load %:")
current_percent_label.grid(row=0, column=0, padx=5, pady=4, sticky="e")

current_percent_entry = tk.Entry(percent_frame, width=12, font=("Consolas", 11))
current_percent_entry.grid(row=0, column=1, padx=5, pady=4)

desired_percent_label = tk.Label(percent_frame, text="Desired Load %:")
desired_percent_label.grid(row=0, column=2, padx=(20, 5), pady=4, sticky="e")

desired_percent_entry = tk.Entry(percent_frame, width=12, font=("Consolas", 11))
desired_percent_entry.grid(row=0, column=3, padx=5, pady=4)

formula_label = tk.Label(root, text="new watts = existing watts × desired % ÷ initial/current %", font=("Arial", 9))
formula_label.pack(pady=(0, 8))

input_label = tk.Label(root, text="Paste Original TOML:", font=("Arial", 11, "bold"))
input_label.pack(anchor="w", padx=18)

input_box = scrolledtext.ScrolledText(root, width=105, height=17, font=("Consolas", 10), wrap=tk.NONE)
input_box.pack(fill=tk.BOTH, expand=True, padx=18, pady=(4, 8))

button_frame = tk.Frame(root)
button_frame.pack(pady=3)

scale_button = tk.Button(button_frame, text="Scale Watts", command=convert_toml, width=16)
scale_button.pack(side=tk.LEFT, padx=5)

copy_button = tk.Button(button_frame, text="Copy Output", command=copy_output, width=16)
copy_button.pack(side=tk.LEFT, padx=5)

example_button = tk.Button(button_frame, text="Load Example", command=use_example, width=16)
example_button.pack(side=tk.LEFT, padx=5)

clear_button = tk.Button(button_frame, text="Clear", command=clear_all, width=12)
clear_button.pack(side=tk.LEFT, padx=5)

output_label = tk.Label(root, text="Scaled TOML:", font=("Arial", 11, "bold"))
output_label.pack(anchor="w", padx=18, pady=(7, 0))

output_box = scrolledtext.ScrolledText(root, width=105, height=17, font=("Consolas", 10), wrap=tk.NONE)
output_box.pack(fill=tk.BOTH, expand=True, padx=18, pady=(4, 8))

status_label = tk.Label(root, text="", font=("Arial", 9, "bold"))
status_label.pack(pady=(0, 10))

root.mainloop()
