def _unpack(tool_id: str) -> bytes:
    item = TOOLS[tool_id]
    content = zlib.decompress(base64.b85decode(item["payload"].encode("ascii")))
    actual = hashlib.sha256(content).hexdigest()
    if actual != item["sha256"]:
        raise RuntimeError("Bundled tool failed integrity verification: " + tool_id)
    compile(content.decode("utf-8"), item["filename"], "exec")
    return content


def get_runtime_folder() -> Path:
    preferred = Path(__file__).resolve().parent / "E3_Tools_Runtime"
    try:
        preferred.mkdir(parents=True, exist_ok=True)
        with tempfile.NamedTemporaryFile(dir=str(preferred), delete=True):
            pass
        return preferred
    except OSError:
        fallback = Path(os.environ.get("LOCALAPPDATA") or Path.home()) / "E3_Tools_Runtime"
        fallback.mkdir(parents=True, exist_ok=True)
        return fallback


def install_tool(tool_id: str, base: Path) -> Path:
    item = TOOLS[tool_id]
    directory = base / item["folder"]
    directory.mkdir(parents=True, exist_ok=True)
    destination = directory / item["filename"]
    content = _unpack(tool_id)
    if destination.exists():
        existing = destination.read_bytes()
        if existing == content:
            return destination
        # Don't silently discard user edits or scripts from an older version.
        suffix = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup = directory / (destination.stem + ".backup_" + suffix + ".py")
        index = 2
        while backup.exists():
            backup = directory / (destination.stem + ".backup_" + suffix + f"_{index}.py")
            index += 1
        shutil.copy2(destination, backup)
    temporary = destination.with_suffix(".py.tmp")
    temporary.write_bytes(content)
    os.replace(temporary, destination)
    return destination


def all_sources_ok() -> bool:
    for key in TOOLS:
        _unpack(key)
    return True


def open_folder(path: Path):
    path.mkdir(parents=True, exist_ok=True)
    if sys.platform == "win32":
        os.startfile(str(path))
    elif sys.platform == "darwin":
        subprocess.Popen(["open", str(path)])
    else:
        subprocess.Popen(["xdg-open", str(path)])

    def launch(self, tool_id):
        if self.running is not None and self.running.poll() is None:
            messagebox.showinfo("Tool already running",
                "Close the open E3 utility before launching another. This prevents "
                "two scripts changing E3 objects at the same time.",parent=self.root)
            return
        try:
            script=install_tool(tool_id,self.runtime)
            python=Path(sys.executable)
            if sys.platform == "win32" and python.name.lower()=="python.exe":
                candidate=python.with_name("pythonw.exe")
                if candidate.is_file(): python=candidate
            cmd=[str(python),str(script)]
            # V15 normally exits after spawning a detached worker. Launch its
            # worker directly so the hub can track the actual live E3 window.
            if tool_id=="pin": cmd.append("--worker")
            output_path=script.parent / "launcher_output.log"
            flags=getattr(subprocess,"CREATE_NO_WINDOW",0) if sys.platform=="win32" else 0
            with output_path.open("a",encoding="utf-8") as fd:
                fd.write("\n--- Launch " + datetime.now().isoformat(timespec="seconds")
                         + " [" + SUITE_VERSION + "] ---\n")
            with output_path.open("a",encoding="utf-8") as fd:
                self.running=subprocess.Popen(cmd,cwd=str(script.parent),
                       stdin=subprocess.DEVNULL,stdout=fd,stderr=subprocess.STDOUT,
                       creationflags=flags,close_fds=True)
            self.running_id=tool_id
            self.set_buttons(False)
            self.status.set(TOOLS[tool_id]["title"]+" running")
            self.detail.set("Its E3 window is open · close it to return here")
            # Let the child window receive focus; keep hub behind it.
            self.root.attributes("-topmost",False)
        except Exception as exc:
            self.running=None
            self.running_id=None
            self.set_buttons(True)
            self.status.set("Could not launch tool")
            self.detail.set(str(exc)[:90])
            messagebox.showerror("Launch error",str(exc),parent=self.root)

    def watch_child(self):
        try:
            if self.running is not None:
                rc=self.running.poll()
                if rc is not None:
                    name=TOOLS[self.running_id]["title"]
                    self.running=None
                    self.running_id=None
                    self.set_buttons(True)
                    self.status.set(name + (" closed" if rc==0 else " exited with an error"))
                    self.detail.set("Ready for another tool" if rc==0 else
                         "See Tool files / logs → launcher_output.log for details")
        finally:
            self.root.after(650,self.watch_child)

    def import_pin_data(self):
        """Optional migration only; never overwrite current run's recovery files."""
        previous=filedialog.askdirectory(title="Select your previous Block Pin Spacer script folder",
                                          parent=self.root)
        if not previous:return
        destination=self.runtime / TOOLS["pin"]["folder"]
        destination.mkdir(parents=True,exist_ok=True)
        imported=[]
        skipped=[]
        for suffix in ("_recovery.json","_pin_identity.json"):
            candidate=Path(previous) / ("main"+suffix)
            # Earlier versions may have used the named V15 script rather than main.py.
            if not candidate.is_file():
                old=Path(previous) / ("E3_Cable_To_Block_Pin_Spacing_V15_ZUKEN_LIGHT_GUI"+suffix)
                if old.is_file():candidate=old
            newfile=destination / ("main"+suffix)
            if not candidate.is_file():
                skipped.append("not found: "+candidate.name)
                continue
            if newfile.exists():
                skipped.append("already exists: "+newfile.name)
                continue
            try:
                data=json.loads(candidate.read_text(encoding="utf-8-sig"))
                if not isinstance(data,dict):
                    raise ValueError("Expected a JSON object")
                shutil.copy2(candidate,newfile)
                imported.append(newfile.name)
            except Exception as exc:
                skipped.append(f"{candidate.name}: {exc}")
        messagebox.showinfo("Pin spacer data", "Imported: " + (", ".join(imported) or "none")
                    + "\nSkipped: " + (", ".join(skipped) or "none")
                    + "\n\nImported identities are still checked against live E3 before use.",
                    parent=self.root)

    def on_close(self):
        if self.running is not None and self.running.poll() is None:
            if not messagebox.askyesno("Close launcher?",
                "An E3 utility is still open. Close only the launcher and let that utility finish?",
                parent=self.root):return
        self.root.destroy()


def main():
    if "--self-test" in sys.argv:
        print("Bundled sources valid:",all_sources_ok())
        with tempfile.TemporaryDirectory() as temp:
            for key in TOOLS:
                path=install_tool(key,Path(temp))
                assert hashlib.sha256(path.read_bytes()).hexdigest()==TOOLS[key]["sha256"]
                assert path.read_text(encoding="utf-8").strip()
                # A second extraction must be idempotent.
                assert install_tool(key,Path(temp))==path
        print("Extraction, checksums, Python syntax and idempotence: PASS")
        return
    if "--extract" in sys.argv:
        folder=get_runtime_folder()
        for key in TOOLS:
            print(install_tool(key,folder))
        return
    MegaApp().root.mainloop()

if __name__=="__main__":main()
