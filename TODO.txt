
# -------------------------------------------------------
# GUI SETUP
# -------------------------------------------------------
def create_gui():
    from tkinter import ttk
    import threading

    root = tk.Tk()
    root.title("Zuken Symbol Formatter")
    root.geometry("560x500")
    root.resizable(False, False)

    style = ttk.Style()
    try:
        style.theme_use("vista")
    except:
        pass

    style.configure("Title.TLabel", font=("Segoe UI", 18, "bold"))
    style.configure("Subtitle.TLabel", font=("Segoe UI", 10))
    style.configure("Step.TLabel", font=("Segoe UI", 10))
    style.configure("Status.TLabel", font=("Segoe UI", 10, "bold"))
    style.configure("Run.TButton", font=("Segoe UI", 11, "bold"), padding=(20, 10))
    style.configure("Secondary.TButton", font=("Segoe UI", 10), padding=(15, 8))

    main_frame = ttk.Frame(root, padding=25)
    main_frame.pack(fill="both", expand=True)

    ttk.Label(
        main_frame,
        text="Zuken Symbol Formatter",
        style="Title.TLabel"
    ).pack(anchor="w")

    ttk.Label(
        main_frame,
        text="Configure and format generated connector symbols.",
        style="Subtitle.TLabel"
    ).pack(anchor="w", pady=(2, 20))

    instruction_frame = ttk.LabelFrame(
        main_frame,
        text="Before Running",
        padding=15
    )
    instruction_frame.pack(fill="x", pady=(0, 20))

    instructions = (
        "1. Open Device Properties in Zuken.\n\n"
        "2. Enable:\n"
        "   Assembly  →  Additional Part ID\n\n"
        "3. Return here and click Run Formatter."
    )

    ttk.Label(
        instruction_frame,
        text=instructions,
        justify="left",
        style="Step.TLabel"
    ).pack(anchor="w")

    status_frame = ttk.LabelFrame(
        main_frame,
        text="Status",
        padding=12
    )
    status_frame.pack(fill="x", pady=(0, 20))

    status_label = ttk.Label(
        status_frame,
        text="Ready",
        style="Status.TLabel"
    )
    status_label.pack(anchor="w")

    # -------------------------------------------------------
    # FORMATTER FINISHED SUCCESSFULLY
    # -------------------------------------------------------
    def finished():
        root.config(cursor="")

        # Hide the formatter window immediately.
        root.withdraw()

        # Show completion popup.
        try:
            messagebox.showinfo(
                "Complete",
                "Formatting complete.",
                parent=root
            )
        finally:
            # Fully close formatter GUI after OK.
            root.destroy()

    # -------------------------------------------------------
    # FORMATTER ERROR
    # -------------------------------------------------------
    def failed(error):
        status_label.config(text=f"Error: {error}")
        run_button.config(state="normal")
        run_again_button.config(state="normal")
        root.config(cursor="")

        messagebox.showerror(
            "Formatter Error",
            str(error),
            parent=root
        )

    # -------------------------------------------------------
    # BACKGROUND FORMATTER WORKER
    # -------------------------------------------------------
    def worker():
        error = None
        com_initialized = False

        try:
            pythoncom.CoInitialize()
            com_initialized = True

            # Run existing E3 formatting code.
            run_ct_code()

        except Exception as e:
            error = str(e)
            print(f"Formatter error: {e}")

        finally:
            if com_initialized:
                pythoncom.CoUninitialize()

            # Update the GUI only after formatting ends.
            if error is None:
                root.after(0, finished)
            else:
                root.after(
                    0,
                    lambda msg=error: failed(msg)
                )

    # -------------------------------------------------------
    # START FORMATTER
    # -------------------------------------------------------
    def run_formatter():
        status_label.config(text="Running...")
        run_button.config(state="disabled")
        run_again_button.config(state="disabled")
        root.config(cursor="watch")

        threading.Thread(
            target=worker,
            daemon=True
        ).start()

    # -------------------------------------------------------
    # CLOSE
    # -------------------------------------------------------
    def close_app():
        root.destroy()

    # -------------------------------------------------------
    # BUTTONS
    # -------------------------------------------------------
    button_frame = ttk.Frame(main_frame)
    button_frame.pack(fill="x")

    run_button = ttk.Button(
        button_frame,
        text="Run Formatter",
        style="Run.TButton",
        command=run_formatter
    )
    run_button.pack(fill="x")

    secondary_frame = ttk.Frame(main_frame)
    secondary_frame.pack(pady=(12, 0))

    run_again_button = ttk.Button(
        secondary_frame,
        text="Run Again",
        style="Secondary.TButton",
        command=run_formatter
    )
    run_again_button.pack(side="left", padx=(0, 8))

    ttk.Button(
        secondary_frame,
        text="Close",
        style="Secondary.TButton",
        command=close_app
    ).pack(side="left")

    ttk.Label(
        main_frame,
        text="E3 Symbol Automation",
        font=("Segoe UI", 8)
    ).pack(side="bottom", anchor="e", pady=(15, 0))

    root.protocol("WM_DELETE_WINDOW", close_app)

    root.mainloop()
