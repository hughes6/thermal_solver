"""E3.series | Bulk CABBLOCK Header Formatter (select blocks, press Enter).

Select one or more CABBLOCK borders in E3, press Enter in the always-on-top
instruction window (or E3), and format their attached header texts:
  Location                  bold, black, center, height 0.25
  Device Function           bold, black, center, height 0.25
  Equipment Identification  bold, black, center, height 0.25
  Device Designation        bold, black, center, height 0.25

The labels are stacked from the top INSIDE each selected block. Existing
content is preserved. Empty Device Function texts are populated with XXX.
No pins, wires, other blocks, or dimensions are altered.

E3 text types 15 (Location) and 12 (Device Designation) are documented.
Device Function and Equipment Identification may be custom: on the first
run the script will ask which E3 text types represent them, using a list of
actual text types/values found on the selected blocks. The choices are saved
next to this script for subsequent batches. If the E3 text node itself is
missing, the batch will stop; this code will NOT substitute an unrelated
free-floating text label for a real property-bound E3 text node.

Optional safe preview: python E3_Bulk_Block_Header_Formatter.py --dry-run
CLI type override: --function-type N --equipment-type M
Close, X or Escape exits; the window returns after every batch.
Requires Windows, pywin32, tkinter, and open E3.series schematic.
"""

LOCATION_TYPE = 15
DEVICE_DESIGNATION_TYPE = 12
HEADER_ORDER = ('location', 'function', 'equipment', 'designation')

def log(message=""):
    line = str(message)
    print(line, flush=True)
    if LOG_STREAM is not None:
        LOG_STREAM.write(line + "\n")
        LOG_STREAM.flush()


def normalize_pin(value):
    if value is None:
        return None
    if isinstance(value, float) and math.isfinite(value) and value.is_integer():
        return str(int(value))
    val = str(value).strip()
    return val or None


def parse_id_result(value):
    """Support E3's (count, ID-array) COM return and normal ID arrays."""
    if value is None:
        return []
    if isinstance(value, (tuple, list)):
        if (len(value) >= 2 and isinstance(value[1], (tuple, list))
                and isinstance(value[0], (int, float))):
            ids = value[1]
            try:
                expected_count = int(value[0])
                ids = list(ids)
                if expected_count < len(ids):
                    # COM wrappers can prepend output/status values.
                    ids = ids[-expected_count:] if expected_count > 0 else []
            except (ValueError, TypeError):
                pass
        else:
            ids = value
    else:
        ids = [value]
    result = []
    for v in ids:
        try:
            i = int(v)
            if i > 0 and i != COM_MISSING:
                result.append(i)
        except (TypeError, ValueError):
            pass
    return result


def get_ids(method, *arguments):
    # E3's GetSymbolIds(None, 0) uses an output array BEFORE a mode argument;
    # GetViewIds(None), GetAllSheetIds(None) and some wrapper versions differ.
    attempts = ((None, *arguments), arguments, (*arguments, None))
    for args in attempts:
        try:
            value = parse_id_result(method(*args))
            if value:
                return value
        except Exception:
            continue
    return []


def select_id(obj, object_id):
    """Do not mistake E3 versions returning None from SetId for failure."""
    try:
        result = obj.SetId(int(object_id))
        if isinstance(result, (int, float)) and result == 0:
            return False
        if hasattr(obj, "GetId"):
            current = obj.GetId()
            if current is not None and int(current) != int(object_id):
                return False
        return True
    except Exception:
        return False


def optional_call(obj, name):
    try:
        return getattr(obj, name)()
    except Exception:
        return None


def get_symbol_location(sym):
    try:
        raw = sym.GetSchemaLocation(None, None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 3:
            return None
        return {"sheet_id": int(raw[0]), "x": float(raw[1]), "y": float(raw[2])}
    except Exception:
        return None


def get_symbol_area(sym):
    try:
        raw = sym.GetPlacedArea(None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 5 or int(raw[0]) <= 0:
            return None
        l, b, r, t = map(float, raw[1:5])
        if not (r > l and t > b):
            return None
        return {"left": l, "bottom": b, "right": r, "top": t}
    except Exception:
        return None


def inside_block(loc, block):
    tol = BLOCK_LOCATION_TOLERANCE
    return (loc is not None and loc["sheet_id"] == block["sheet_id"]
            and block["left"] - tol <= loc["x"] <= block["right"] + tol
            and block["bottom"] - tol <= loc["y"] <= block["top"] + tol)


def scan_sheet_symbols(prj, sht, sheet_id):
    if not select_id(sht, sheet_id):
        raise RuntimeError(f"Could not select E3 sheet ID {sheet_id}.")
    ids = get_ids(sht.GetSymbolIds)
    if ids:
        return ids
    log(f"  Sheet {sheet_id} has no sheet-local symbol ID array; scanning project IDs instead.")
    return get_ids(prj.GetSymbolIds)


def wait_for_block_selection(preview=False):
    """Nonmodal directions; E3 must remain usable while this window is open.

    Return in the window works through Tk. On Windows, pressing Enter while
    E3 is foreground also works via a *read-only* key-state poll (no global
    hotkey registration, keyboard interception, or E3 focus manipulation).
    """
    try:
        import tkinter as tk
        from tkinter import ttk
        root = tk.Tk()
    except Exception as exc:
        raise RuntimeError(
            'Could not open the block-selection instruction window. '
            'No E3 changes made. Check that tkinter is installed.'
        ) from exc

    accepted = False
    finished = False

    def proceed(event=None):
        nonlocal accepted, finished
        if finished:
            return
        accepted = True
        finished = True
        root.quit()

    def cancel(event=None):
        nonlocal finished
        if finished:
            return
        finished = True
        root.quit()

    try:
        root.title('E3 - Select blocks (repeat until Close)')
        root.resizable(False, False)
        root.protocol('WM_DELETE_WINDOW', cancel)
        # Keep the instructions ABOVE E3 even when the user clicks on a block.
        # This is only a topmost WINDOW, not a modal dialog: E3 can still take
        # focus and accept mouse selection while this prompt remains visible.
        # The prompt can be dragged aside if it covers a block.
        root.attributes('-topmost', True)
        frame = ttk.Frame(root, padding=20)
        frame.pack(fill='both', expand=True)
        ttk.Label(frame, text='Block Header Formatter',
                  font=('Segoe UI', 13, 'bold')).pack(anchor='w', pady=(0, 12))
        ttk.Label(frame, text=(
            '1. In E3, select ONE OR MORE CABBLOCK borders.\n'
            '   This window stays visible; drag it aside if needed.\n'
            '   Clear any previous selection before choosing the next group.\n'
            '2. Press ENTER to format all selected block headers.\n'
            '   (Enter in E3 or return here and press Enter.)\n'
            '3. After the result appears, this window opens again.\n'
            '4. When finished, click Close or the window X.'),
                  justify='left', font=('Segoe UI', 10)).pack(anchor='w')
        ttk.Separator(frame, orient='horizontal').pack(fill='x', pady=14)
        mode = ('PREVIEW ONLY - no E3 changes will be made.' if preview else
                'On Enter: stack Location / Device Function / Equipment ID /\n'
                'Device Designation; bold, black, centered, 0.25 height.')
        ttk.Label(frame, text=mode, justify='left').pack(anchor='w', pady=(0, 12))
        buttons = ttk.Frame(frame)
        buttons.pack(fill='x')
        go_btn = ttk.Button(buttons, text='Format selected blocks (Enter)', command=proceed)
        go_btn.pack(side='left')
        ttk.Button(buttons, text='Close (Esc)', command=cancel).pack(side='right')
        root.bind('<Return>', proceed)
        root.bind('<KP_Enter>', proceed)
        root.bind('<Escape>', cancel)

        root.update_idletasks()
        # Place instructions near upper right so the block remains easier to click.
        width, height = root.winfo_width(), root.winfo_height()
        x = max(0, root.winfo_screenwidth() - width - 50)
        root.geometry(f'+{x}+80')
        go_btn.focus_set()

        # Tk's Return binding only works while its window is foreground.
        # Windows GetAsyncKeyState lets Enter from E3 also complete the prompt
        # without hijacking E3's keyboard or needing a third-party package.
        try:
            import ctypes
            key_state = ctypes.windll.user32.GetAsyncKeyState
            vk_enter = 0x0D
            was_down = bool(key_state(vk_enter) & 0x8000)
            ready = not was_down

            def poll_enter():
                nonlocal was_down, ready
                if finished:
                    return
                try:
                    down = bool(key_state(vk_enter) & 0x8000)
                    if not down:
                        ready = True
                    if ready and down and not was_down:
                        proceed()
                    was_down = down
                except Exception:
                    # Dialog Enter/button remain usable if OS polling fails.
                    pass
                if not finished:
                    root.after(25, poll_enter)

            root.after(25, poll_enter)
        except (AttributeError, OSError):
            log('Global Enter detection not available. Use Enter/Run in the instruction box.')
        root.mainloop()
        return accepted
    finally:
        root.destroy()


def selected_symbol_id_candidates(prj, sht, app):
    """Read current E3 GUI selection, WITHOUT changing GUI selection.

    Different COM versions expose selection at job, sheet or application level;
    each getter may use an out parameter or return an array directly. We use
    only explicit *Selected* getter names and never auto-pick from sheet IDs.
    """
    active_sheet_id = None
    for method_name in ('GetActiveSheetId',):
        try:
            sheet_id = getattr(prj, method_name)()
            if isinstance(sheet_id, (int, float)) and int(sheet_id) > 0:
                active_sheet_id = int(sheet_id)
                break
        except Exception:
            continue
    if active_sheet_id is not None:
        select_id(sht, active_sheet_id)
    log(f'Active E3 sheet: {active_sheet_id if active_sheet_id else "not reported by COM"}')

    # Explicitly symbol-only queries first, generic selected object IDs last.
    # The fallback generic query is ALWAYS type-checked as a placed CABBLOCK.
    probes = (
        ('job', prj, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedSymbolsIds'),
        ('job', prj, 'GetSelectedSymbolId'),
        ('sheet', sht, 'GetSelectedSymbolIds'),
        ('sheet', sht, 'GetSelectedSymbolsIds'),
        ('sheet', sht, 'GetSelectedSymbolId'),
        ('app', app, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedObjectIds'),
        ('job', prj, 'GetSelectedIds'),
        ('sheet', sht, 'GetSelectedObjectIds'),
        ('sheet', sht, 'GetSelectedIds'),
        ('app', app, 'GetSelectedObjectIds'),
        ('app', app, 'GetSelectedIds'),
    )
    attempted = []
    found = set()
    for label, obj, method_name in probes:
        try:
            getter = getattr(obj, method_name)
        except Exception:
            continue
        # Avoid evaluating an integer method return as the selected ID unless
        # GetSelectedSymbolId explicitly returns one number. For array-style
        # getters, require an array (or a count + array) in the result.
        for args in ((None,), (), (None, 0), (0, None)):
            try:
                result = getter(*args)
            except Exception as exc:
                attempted.append(f'{label}.{method_name}{args}: {str(exc)[:75]}')
                continue
            if isinstance(result, (tuple, list)):
                ids = parse_id_result(result)
            elif method_name == 'GetSelectedSymbolId':
                ids = parse_id_result(result)
            else:
                ids = []
            attempted.append(f'{label}.{method_name}{args}: {ids}')
            if ids:
                found.update(ids)
                break
    log('GUI SELECTION READBACK:')
    for item in attempted[:45]:
        log(f'  {item}')
    if len(attempted) > 45:
        log(f'  ... {len(attempted)-45} additional getter attempts')
    if not found:
        raise RuntimeError(
            'No selected symbol IDs were returned by this E3 COM build. '
            'Select one or more CABBLOCK borders in E3 before pressing Enter. '
            'If it is already selected, send the GUI SELECTION READBACK log; '
            'your E3 version may use another selection API. NO CHANGES MADE.'
        )
    return sorted(found)


def get_selected_cabblocks(prj, sht, app):
    """Return ALL uniquely selected CABBLOCKs; ignore selected non-blocks.

    A drag/box selection can include pins, lines or text as well as the blocks.
    These are never used as targets. At least one placed CABBLOCK is required.
    """
    ids = selected_symbol_id_candidates(prj, sht, app)
    sym = prj.CreateSymbolObject()
    other_symbols = []
    blocks = []
    for sid in ids:
        if not select_id(sym, sid):
            log(f'  Selected ID {sid}: not a valid E3 symbol ID; ignored')
            continue
        kind = str(optional_call(sym, 'GetSymbolTypeName') or '')
        if 'CABBLOCK' not in kind.upper():
            other_symbols.append((sid, kind))
            continue
        location, area = get_symbol_location(sym), get_symbol_area(sym)
        if location is None or area is None:
            raise RuntimeError(
                f'Selected CABBLOCK {sid}: missing placed location/area. '
                'Batch cancelled: NO CHANGES MADE.'
            )
        block = {
            'id': sid, 'sheet_id': location['sheet_id'], 'kind': kind,
            'owner': '<selected block>', 'view_id': None, 'view_number': None,
            **area,
        }
        blocks.append(block)
    if not blocks:
        raise RuntimeError(
            f'No placed CABBLOCK found among {len(ids)} selected E3 ID(s). '
            f'Non-block symbols: {other_symbols[:12]}. '
            'Click one or more CABBLOCK borders and press Enter again. '
            'NO CHANGES MADE.'
        )
    blocks.sort(key=lambda b: (b['sheet_id'], -b['top'], b['left'], b['id']))
    log(f'SELECTION: {len(blocks)} selected CABBLOCK(s); '
        f'{len(other_symbols)} selected non-block symbol(s) ignored.')
    for b in blocks:
        log(f'  BLOCK {b["id"]}: sheet {b["sheet_id"]} | {b["kind"]!r} | '
            f'X=[{b["left"]:.3f},{b["right"]:.3f}] '
            f'Y=[{b["bottom"]:.3f},{b["top"]:.3f}]')
    if other_symbols:
        log(f'  IGNORED non-block selections (first 15): {other_symbols[:15]}')
    return blocks

    try:
        import tkinter as tk

def type_config_path():
    """The file is saved next to the Python script, not inside the E3 project."""
    return Path(__file__).resolve().parent / CONFIG_NAME


def read_config():
    path = type_config_path()
    try:
        obj = json.loads(path.read_text(encoding='utf-8'))
        result = {}
        for field in ('function', 'equipment'):
            val = obj.get(field)
            if val is not None and int(val) > 0:
                result[field] = int(val)
        return result
    except (OSError, ValueError, TypeError, AttributeError):
        return {}


def save_config(mapping):
    path = type_config_path()
    try:
        path.write_text(json.dumps({
            'function': int(mapping['function']),
            'equipment': int(mapping['equipment']),
        }, indent=2) + '\n', encoding='utf-8')
        log('Saved E3 text type mappings to ' + str(path))
    except OSError as exc:
        log(f'WARNING: could not save text type mapping {path}: {exc}')


def text_type_hint(txt):
    hints = []
    for getter in ('GetTypeName', 'GetTextTypeName', 'GetTypeDescription',
                   'GetAttributeName', 'GetName', 'GetSource'):
        val = optional_call(txt, getter)
        if isinstance(val, str) and val.strip():
            hints.append(val.strip())
    return ' | '.join(dict.fromkeys(hints))[:180]


def get_text_xy(text, block):
    """Read E3 SCHEMATIC coordinates; do not assume a symbol-relative value.

    The existing signal mover confirmed E3 text supports GetSchemaLocation.
    Only accept a decoded position reasonably close to the parent CABBLOCK.
    That protects from treating status codes/sheet IDs as coordinates.
    """
    probes = (
        (None, None, None, None, None),
        (None, None, None),
        (None, None),
        (),
    )
    sheet = int(block['sheet_id'])
    options = []
    raw_preview = []
    for args in probes:
        try:
            raw = text.GetSchemaLocation(*args)
        except Exception:
            continue
        raw_preview.append(repr(raw)[:140])
        if not isinstance(raw, (tuple, list)):
            continue
        nums = []
        for number in raw:
            try:
                nums.append(float(number))
            except (TypeError, ValueError):
                nums.append(None)
        candidates = []
        if len(nums) >= 3 and nums[0] is not None and int(nums[0]) == sheet:
            candidates.append((0, nums[1], nums[2]))
        if len(nums) >= 4 and nums[1] is not None and int(nums[1]) == sheet:
            candidates.append((0, nums[2], nums[3]))
        if len(nums) >= 3 and nums[0] in (-1, 0, 1):
            candidates.append((2, nums[1], nums[2]))
        if len(nums) >= 2:
            candidates.append((3, nums[0], nums[1]))
        for priority, x, y in candidates:
            if x is None or y is None or not (math.isfinite(x) and math.isfinite(y)):
                continue
            if (block['left'] - 2 <= x <= block['right'] + 2 and
                    block['bottom'] - 2 <= y <= block['top'] + 2):
                options.append((priority, x, y))
        if options and min(op[0] for op in options) == 0:
            break
    if not options:
        raise RuntimeError(f"Block {block['id']}: cannot read absolute schematic "
                           f"text position. GetSchemaLocation returned: {raw_preview[:3]}. "
                           'No E3 changes made.')
    options.sort(key=lambda item: item[0])
    best_priority = options[0][0]
    distinct = {(round(x, 4), round(y, 4)) for p, x, y in options if p == best_priority}
    if len(distinct) > 1:
        raise RuntimeError(f"Block {block['id']}: multiple possible coordinates "
                           f"from E3 text getter: {sorted(distinct)}. No changes made.")
    return {'x': options[0][1], 'y': options[0][2]}


def inspect_block_texts(prj, blocks):
    """Read only texts owned by the selected CABBLOCK symbols (not their pins)."""
    all_texts = {}
    sym = prj.CreateSymbolObject()
    txt = prj.CreateTextObject()
    for block in blocks:
        if not select_id(sym, block['id']):
            raise RuntimeError(f"Selected block {block['id']} disappeared.")
        ids = get_ids(sym.GetTextIds)
        entries = defaultdict(list)
        for tid in ids:
            if not select_id(txt, tid):
                continue
            try:
                typ = int(txt.GetType())
            except Exception:
                continue
            raw_val = optional_call(txt, 'GetText')
            value = '' if raw_val is None else str(raw_val)
            hint = text_type_hint(txt)
            entries[typ].append({'id': tid, 'type': typ, 'value': value, 'hint': hint})
        all_texts[block['id']] = entries
        log(f"BLOCK {block['id']} (sheet {block['sheet_id']}): "
            f"{len(ids)} attached text IDs; {len(entries)} text types")
        for typ in sorted(entries):
            for e in entries[typ]:
                log(f"  TYPE {typ:<5} ID {e['id']:<10} "
                    f"VALUE={e['value'][:120]!r} HINT={e['hint'][:90]!r}")
    return all_texts


def infer_field_type_from_hints(all_texts, field):
    needles = {
        'function': ('device function', 'devicefunction'),
        'equipment': ('equipment identification', 'equipment id',
                      'equipmentidentification'),
    }[field]
    types = set()
    for items in all_texts.values():
        for typ, entries in items.items():
            for entry in entries:
                hint = entry['hint'].casefold()
                if any(n in hint for n in needles):
                    types.add(typ)
    return next(iter(types)) if len(types) == 1 else None


def prompt_for_text_types(all_texts, known):
    """First-run mapping; only shows real types actually found in selected blocks."""
    possible = defaultdict(list)
    for entries in all_texts.values():
        for typ, texts in entries.items():
            if typ not in (LOCATION_TYPE, DEVICE_DESIGNATION_TYPE):
                for e in texts:
                    tag = (e['hint'] + ' ' + e['value']).strip()
                    if tag and tag[:50] not in possible[typ]:
                        possible[typ].append(tag[:50])
                possible.setdefault(typ, [])
    if not possible:
        raise RuntimeError('No additional text types attached to these CABBLOCKs. '
                           'Create/enable the Device Function and Equipment '
                           'Identification text nodes in the E3 symbol template, '
                           'then rerun. No changes made.')
    try:
        import tkinter as tk
        from tkinter import ttk, messagebox
        root = tk.Tk()
    except Exception as exc:
        raise RuntimeError('Text type selection GUI could not open. Use '
                           '--function-type and --equipment-type instead.') from exc
    result = None
    try:
        root.title('E3 - Identify header text types (one time)')
        root.attributes('-topmost', True)
        root.resizable(False, False)
        frame = ttk.Frame(root, padding=18)
        frame.pack(fill='both', expand=True)
        ttk.Label(frame, text='Identify two E3 text types',
                  font=('Segoe UI', 12, 'bold')).pack(anchor='w', pady=(0, 7))
        ttk.Label(frame, text=('Location = type 15 and Device Designation = type 12 are known.\n'
                  'Choose the types your project uses for the two remaining fields.\n'
                  'Examples below come directly from the selected block texts.'),
                  justify='left').pack(anchor='w', pady=(0, 12))
        shown = [f"{typ}   |   {', '.join(possible[typ][:2])[:85]}" for typ in sorted(possible)]
        from_label = {row: int(row.split('|')[0].strip()) for row in shown}
        from_type = {val: key for key, val in from_label.items()}
        choices = {}
        for field in ('function', 'equipment'):
            ttk.Label(frame, text=FIELD_LABELS[field], font=('Segoe UI', 10, 'bold')).pack(anchor='w')
            widget = ttk.Combobox(frame, values=shown, width=80, state='readonly')
            widget.pack(fill='x', pady=(4, 10))
            if known.get(field) in from_type:
                widget.set(from_type[known[field]])
            choices[field] = widget
        status = ttk.Label(frame, text='', foreground='#b00020')
        status.pack(anchor='w')

        def accept(event=None):
            nonlocal result
            try:
                pair = {f: from_label[choices[f].get()] for f in choices}
            except KeyError:
                status.config(text='Please select a type for both fields.')
                return
            if pair['function'] == pair['equipment']:
                status.config(text='Two fields cannot use the same text type.')
                return
            result = pair
            root.quit()

        def cancel(event=None):
            root.quit()

        buttons = ttk.Frame(frame)
        buttons.pack(fill='x', pady=(10, 0))
        ttk.Button(buttons, text='Use these text types', command=accept).pack(side='left')
        ttk.Button(buttons, text='Cancel batch', command=cancel).pack(side='right')
        root.bind('<Escape>', cancel)
        root.bind('<Return>', accept)
        root.protocol('WM_DELETE_WINDOW', cancel)
        root.mainloop()
        return result
    finally:
        root.destroy()


def determine_mapping(all_texts, override_function=None, override_equipment=None,
                      allow_prompt=True):
    saved = read_config()
    mapping = {'location': LOCATION_TYPE, 'designation': DEVICE_DESIGNATION_TYPE}
    for field, override in (('function', override_function), ('equipment', override_equipment)):
        value = override or saved.get(field) or infer_field_type_from_hints(all_texts, field)
        if value:
            value = int(value)
            if override is None and not any(value in texts for texts in all_texts.values()):
                # The saved text-type mapping can be for a different E3
                # project; reopen the one-time chooser rather than fail with
                # a stale value that does not occur on any selected block.
                log(f'Previously saved {FIELD_LABELS[field]} type {value} was '
                    'not found on this selection; asking for a new mapping.')
            else:
                mapping[field] = value
    types_used = set(mapping.values())
    if (len(mapping) < 4 or len(types_used) != len(mapping)) and allow_prompt:
        chosen = prompt_for_text_types(all_texts, mapping)
        if chosen is None:
            raise RuntimeError('Header text type mapping cancelled. No E3 changes made.')
        mapping.update(chosen)
    if len(mapping) < 4 or len(set(mapping.values())) != 4:
        raise RuntimeError('Need unique text type IDs for Location, Device Function, '
                           'Equipment Identification, and Device Designation. '
                           'Use the mapping popup or explicit --function-type and '
                           '--equipment-type. No changes made.')
    save_config(mapping)
    log('E3 HEADER TEXT TYPE MAPPING: ' + ', '.join(
        f"{FIELD_LABELS[k]}={mapping[k]}" for k in HEADER_ORDER))
    return mapping


def build_plan(blocks, all_texts, mapping):
    """Preflight entire batch before modifying ANY of its selected blocks."""
    plan = []
    target_ids = set()
    for b in blocks:
        available = all_texts[b['id']]
        requested = {}
        for field in HEADER_ORDER:
            typ = mapping[field]
            matches = available.get(typ, [])
            if len(matches) != 1:
                details = [(e['id'], e['value'][:25]) for e in matches]
                raise RuntimeError(
                    f"CABBLOCK {b['id']}: {FIELD_LABELS[field]} (type {typ}) "
                    f"has {len(matches)} attached text node(s), expected exactly 1; "
                    f"found={details}.\nIf missing, add that *property-linked text node* "
                    'to the actual E3 symbol/text template first. No changes made.'
                )
            requested[field] = matches[0]
        height = float(b['top']) - float(b['bottom'])
        required = TOP_INSET + 3 * TEXT_PITCH + BOTTOM_CLEARANCE
        if height < required:
            raise RuntimeError(f"CABBLOCK {b['id']} has height {height:.3f}, "
                               f"but this 4-line header needs at least {required:.3f}. "
                               'No changes made; change pitch/inset constants only '
                               'if the desired layout physically fits.')
        x = (float(b['left']) + float(b['right'])) / 2
        for index, field in enumerate(HEADER_ORDER):
            e = requested[field]
            if e['id'] in target_ids:
                raise RuntimeError(f"Text {e['id']} is attached to more than one block. "
                                   'Aborting before changes.')
            target_ids.add(e['id'])
            y = float(b['top']) - TOP_INSET - index * TEXT_PITCH
            plan.append({
                'block': b, 'field': field, 'type': mapping[field], 'id': e['id'],
                'before_value': e['value'], 'x': x, 'y': y,
            })
    return plan


def preflight_positions(prj, plan):
    txt = prj.CreateTextObject()
    for item in plan:
        if not select_id(txt, item['id']):
            raise RuntimeError(f"Text {item['id']} disappeared during preflight.")
        item['before'] = get_text_xy(txt, item['block'])
        item['move_needed'] = (
            abs(item['before']['x'] - item['x']) > POSITION_TOLERANCE or
            abs(item['before']['y'] - item['y']) > POSITION_TOLERANCE
        )
        if item['field'] == 'function' and not item['before_value'].strip():
            item['set_xxx'] = True
        else:
            item['set_xxx'] = False
        log(f"  BLOCK {item['block']['id']} | {FIELD_LABELS[item['field']]} "
            f"type={item['type']} text_id={item['id']} "
            f"{item['before_value'][:70]!r} "
            f"({item['before']['x']:.3f},{item['before']['y']:.3f}) => "
            f"({item['x']:.3f},{item['y']:.3f}) | "
            f"{'MOVE' if item['move_needed'] else 'IN PLACE'}"
            + (' | SET FUNCTION=XXX' if item['set_xxx'] else ''))
    return plan


def _same_xy(txt, block, x, y):
    try:
        actual = get_text_xy(txt, block)
        return (abs(actual['x'] - x) <= POSITION_TOLERANCE and
                abs(actual['y'] - y) <= POSITION_TOLERANCE)
    except Exception:
        return False


def move_text(txt, item):
    """Only try signatures already used by the working signal relocator."""
    method = getattr(txt, 'SetSchemaLocation', None)
    if method is None:
        raise RuntimeError('E3 Text.SetSchemaLocation is unavailable.')
    old = item['before']
    b = item['block']
    attempts = (
        ((item['x'], item['y']), (old['x'], old['y']), 'x,y'),
        ((b['sheet_id'], item['x'], item['y']),
         (b['sheet_id'], old['x'], old['y']), 'sheet,x,y'),
    )
    errors = []
    for new, rollback, signature in attempts:
        try:
            result = method(*new)
        except Exception as exc:
            errors.append(f'{signature}: {exc}')
            continue
        if _same_xy(txt, b, item['x'], item['y']):
            return
        try:
            method(*rollback)
        except Exception as exc:
            raise RuntimeError(f"Text {item['id']} moved but readback failed "
                               f"and rollback also failed: {exc}; CHECK E3") from exc
        raise RuntimeError(f"Text {item['id']}: SetSchemaLocation({signature}) "
                           'returned but new position was not verified. '
                           f'Rolled back (reported result={result!r}); batch stopped.')
    raise RuntimeError(f"Text {item['id']}: unable to set position: {errors}")


def checked_style_write(txt, method, value, field):
    try:
        result = getattr(txt, method)(value)
    except Exception as exc:
        raise RuntimeError(f"Text {field}: {method}({value}) failed: {exc}") from exc
    # E3 methods sometimes return None on success; negative status means error.
    if isinstance(result, (int, float)) and result < 0:
        raise RuntimeError(f"Text {field}: {method}({value}) returned {result}.")


def apply_plan(prj, plan):
    """Execute after all selection, text nodes and positions pass preflight.

    A COM failure is NOT atomic; previous successful writes remain, and the
    log says so. Users should test on a saved copy before bulk production use.
    """
    txt = prj.CreateTextObject()
    completed = 0
    updated_functions = 0
    try:
        for item in plan:
            if not select_id(txt, item['id']):
                raise RuntimeError(f"Text {item['id']} no longer available.")
            field = FIELD_LABELS[item['field']]
            if item['set_xxx']:
                checked_style_write(txt, 'SetText', 'XXX', field)
                actual = str(optional_call(txt, 'GetText') or '').strip()
                if actual != 'XXX':
                    raise RuntimeError(f"Text {item['id']}: SetText('XXX') returned "
                                       f"but readback was {actual!r}. "
                                       'The text may be controlled by an E3 attribute. '
                                       'Check this block before continuing.')
                updated_functions += 1
            checked_style_write(txt, 'SetVisibility', 1, field)
            checked_style_write(txt, 'SetStyle', BOLD_STYLE, field)
            checked_style_write(txt, 'SetColour', BLACK_COLOUR, field)
            checked_style_write(txt, 'SetHeight', TEXT_HEIGHT, field)
            checked_style_write(txt, 'SetAlignment', CENTER_ALIGNMENT, field)
            if item['move_needed']:
                move_text(txt, item)
            completed += 1
            log(f"  DONE block {item['block']['id']} | {field}: "
                f"text {item['id']} black/bold/0.25/center/visible; "
                f"{'moved' if item['move_needed'] else 'position unchanged'}"
                + ('; XXX added' if item['set_xxx'] else ''))
    except Exception:
        log(f"PARTIAL FAILURE: {completed}/{len(plan)} text nodes completed; "
            f"{updated_functions} Device Function values filled. "
            'Earlier E3 writes may remain; inspect the drawing.')
        raise
    return completed, updated_functions


def run(preview=False, function_type=None, equipment_type=None):
    global LOG_STREAM
    path = choose_log_path()
    try:
        LOG_STREAM = open(path, 'a', encoding='utf-8')
        log('\n' + '=' * 80)
        log('NEW HEADER BATCH: ' + datetime.now().astimezone().strftime('%Y-%m-%d %H:%M:%S'))
        log('FORMAT: LOCATION > DEVICE FUNCTION > EQUIPMENT ID > DEVICE DESIGNATION')
        log('BOLD, BLACK (0), CENTER (2), HEIGHT 0.25, VISIBLE')
        log('Using current E3 GUI selection. No spreadsheet required.')
        log('=' * 80)
        import win32com.client as win32
        app = win32.gencache.EnsureDispatch('CT.Application')
        prj = app.CreateJobObject()
        sht = prj.CreateSheetObject()
        blocks = get_selected_cabblocks(prj, sht, app)
        all_texts = inspect_block_texts(prj, blocks)
        mapping = determine_mapping(all_texts, function_type, equipment_type)
        plan = build_plan(blocks, all_texts, mapping)
        preflight_positions(prj, plan)
        changes = sum(p['move_needed'] for p in plan)
        blanks = sum(p['set_xxx'] for p in plan)
        log(f'BATCH READY: {len(blocks)} block(s); {len(plan)} linked text(s); '
            f'{changes} reposition(s); {blanks} blank function(s) -> XXX.')
        if preview:
            log('DRY RUN: NO CHANGES MADE.')
            popup(f'{len(blocks)} selected block(s) validated.\n'
                  f'{len(plan)} text nodes; {changes} moves, {blanks} blank '
                  f'Device Functions would become XXX.\n\n'
                  f'No changes made.\nLog: {path}')
            return 0
        completed, filled = apply_plan(prj, plan)
        log(f'SUCCESS: {len(blocks)} CABBLOCK(s), {completed} text nodes; '
            f'{filled} Device Function placeholders set to XXX.')
        popup(f'Formatted {len(blocks)} selected block(s).\n'
              'Location → Device Function → Equipment ID → Device Designation\n'
              f'{completed} labels formatted; {filled} Device Function values '
              f'filled with XXX.\n\nLog: {path}\n\nSave the E3 project.')
        return 0
    except Exception:
        log('ERROR: ' + traceback.format_exc())
        raise
    finally:
        if LOG_STREAM is not None:
            LOG_STREAM.close()
        LOG_STREAM = None


def main(argv=None):
    parser = argparse.ArgumentParser(
        description='Select any number of E3 CABBLOCKs and press Enter to '
                    'format a 4-line header on each block.')
    parser.add_argument('--dry-run', action='store_true', help='Preview only.')
    parser.add_argument('--function-type', type=int, default=None,
                        help='Project-specific E3 text type for Device Function.')
    parser.add_argument('--equipment-type', type=int, default=None,
                        help='Project-specific E3 text type for Equipment Identification.')
    args = parser.parse_args(argv)
    while True:
        try:
            accepted = wait_for_block_selection(preview=args.dry_run)
        except Exception as exc:
            print('Could not open selection instructions:', exc)
            return 1
        if not accepted:
            print('CLOSED: no additional E3 changes.')
            return 0
        try:
            run(preview=args.dry_run,
                function_type=args.function_type,
                equipment_type=args.equipment_type)
        except Exception as exc:
            print('FAILED:', exc)
            popup(str(exc) + '\n\nThe instruction window will reopen.\n'
                  'Log: E3_Block_Header_Formatter_Log.txt', error=True)


if __name__ == '__main__':
    sys.exit(main())
