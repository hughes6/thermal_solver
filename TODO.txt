"""Zuken E3.series: Excel J2 local-signal importer.

Version 1.1 -- Windows / E3.series. Run by double-clicking or from an E3 Tools menu.

Worksheet 'J2':
    B2    = duplicate-pin status/check (if present)
    C4    = block device designation
    C5    = connector count N
    B7:B(6+N) = connector names
    C7:C(6+N) = connector types (additional identity information)
    I:J   = connector 1: pin, local signal
    M:N   = connector 2: pin, local signal
    Q:R   = connector 3, etc., all starting on row 7
    K:L, O:P, ... are intentionally unused spacer columns

Only rows with BOTH a pin value and a nonblank local signal are imported.
Blank signal cells do not clear E3 data. Duplicate pin names within a
connector are the only spreadsheet/model integrity check. E3 object ownership
is additionally checked before writes to avoid modifying the wrong object.

For each resolved pin the importer sets the pin's local signal via the E3 pin
COM interface, and then styles its EXISTING type-7 text: height 0.15 in,
bold style 1, visible. It never modifies pin names, connector types or blocks.
Text objects that E3 does not expose are logged, not silently invented.

Test in a copy of the E3 project. COM support cannot be run on non-Windows hosts.
Requires: py -m pip install pywin32 openpyxl
Optional: Zuken's version-matched 'e3series' Python package.
"""

from __future__ import annotations

import argparse
from collections import Counter
from dataclasses import dataclass, field
import gc
from pathlib import Path
import re
import subprocess
import sys
import traceback

SHEET_NAME = 'J2'
FIRST_ROW = 7
FIRST_PIN_COL = 9  # I
CONNECTOR_COL_STEP = 4  # I:J, M:N, Q:R, U:V, ...
LOCAL_SIGNAL_TEXT_TYPE = 7
TEXT_HEIGHT_IN = 0.15
BOLD_STYLE = 1
NO_VALUE = -2147352572  # COM's DISP_E_PARAMNOTFOUND


@dataclass(frozen=True)
class PinEntry:
    connector_index: int
    connector_name: str
    connector_type: str
    pin_name: str
    signal_name: str
    cell: str


@dataclass
class ImportSheet:
    workbook_path: Path
    block_name: str
    connector_count: int
    entries: list[PinEntry]
    check_b2: object
    duplicate_problems: list[str] = field(default_factory=list)


@dataclass
class ResolvedPin:
    entry: PinEntry
    connector_id: int
    pin_id: int


def as_text(value):
    if value is None:
        return ''
    if isinstance(value, float) and value.is_integer():
        return str(int(value))
    return str(value).strip()


def device_key(value):
    """Treat a leading E3 '-' designation marker as optional, not arbitrary substrings."""
    return as_text(value).lstrip('-').casefold()


def connector_keys(value):
    """Recognize a connector with or without its block prefix ('-A1.J1')."""
    v = device_key(value)
    return {v, v.rsplit('.', 1)[-1]}


def duplicate_flag_b2(value):
    """Check commonly used B2 duplicate-pin outcomes, not unrelated statuses."""
    if isinstance(value, bool):
        return bool(value)
    if isinstance(value, (int, float)):
        return value > 0
    s = as_text(value).lower().replace('_', ' ').strip()
    if not s or s in ('ok', 'pass', 'passed', 'false', 'none', 'clear', 'no', '0'):
        return False
    if any(x in s for x in ('no duplicate', '0 duplicate', 'without duplicate', 'duplicate-free')):
        return False
    return s in ('error', 'fail', 'failed', 'true') or 'duplicate' in s


def parse_workbook(filename: Path) -> ImportSheet:
    """Read Excel values without modifying the workbook or recalculating it."""
    from openpyxl import load_workbook
    wb = load_workbook(str(filename), read_only=False, data_only=True)
    try:
        if SHEET_NAME not in wb.sheetnames:
            raise ValueError("Excel worksheet 'J2' was not found.")
        sh = wb[SHEET_NAME]
        block = as_text(sh['C4'].value)
        if not block:
            raise ValueError('J2!C4 (device designation) is empty.')
        raw_count = sh['C5'].value
        try:
            connector_count = int(raw_count)
            if connector_count < 1 or float(raw_count) != connector_count:
                raise ValueError()
        except (ValueError, TypeError):
            raise ValueError('J2!C5 must contain the number of connectors.') from None

        entries = []
        problems = []
        for ci in range(connector_count):
            connector_name = as_text(sh.cell(FIRST_ROW + ci, 2).value)
            connector_type = as_text(sh.cell(FIRST_ROW + ci, 3).value)
            # Pin/signal boxes are separated by two blank columns: I:J, M:N, Q:R, ...
            pin_col = FIRST_PIN_COL + CONNECTOR_COL_STEP * ci
            signal_col = pin_col + 1
            pin_col_letter = sh.cell(FIRST_ROW, pin_col).column_letter
            signal_col_letter = sh.cell(FIRST_ROW, signal_col).column_letter
            if not connector_name:
                # A missing connector name is a matching problem, not a model check.
                problems.append(f'Connector {ci + 1}: blank name at B{FIRST_ROW + ci}.')
            seen_pins = {}
            # Respect data even if some pin rows are intentionally left blank.
            # max_row is set by the actual workbook, not by the declared pin count.
            for row in range(FIRST_ROW, sh.max_row + 1):
                pin = as_text(sh.cell(row, pin_col).value)
                signal = as_text(sh.cell(row, signal_col).value)
                if pin:
                    # Check duplicates even where a local signal is blank;
                    # this is the one requested model/spreadsheet check.
                    if pin.casefold() in seen_pins:
                        problems.append(f'DUPLICATE pin {pin!r} in connector {connector_name!r} '
                                        f'at {pin_col_letter}{seen_pins[pin.casefold()]} '
                                        f'and {pin_col_letter}{row}.')
                        continue
                    seen_pins[pin.casefold()] = row
                if not signal:
                    continue  # never clear or reformat an unspecified E3 pin
                if not pin:
                    problems.append(f'{signal_col_letter}{row}: signal without a pin number.')
                    continue
                entries.append(PinEntry(ci, connector_name, connector_type,
                                        pin, signal, f'{signal_col_letter}{row}'))

        sheet_data = ImportSheet(Path(filename), block, connector_count,
                                 entries, sh['B2'].value, problems)
        # A B2 formula may have no cached value when Excel has not calculated.
        # The actual duplicate scan above remains authoritative in that case.
        if duplicate_flag_b2(sheet_data.check_b2):
            sheet_data.duplicate_problems.insert(
                0, 'J2!B2 reports duplicate pins: ' + repr(sheet_data.check_b2))
        return sheet_data
    finally:
        wb.close()


def require_no_duplicate_pins(data):
    dup = [p for p in data.duplicate_problems if 'DUPLICATE' in p.upper()]
    if dup:
        raise ValueError('Duplicate pin check failed:\n' + '\n'.join(dup))


def flatten_positive_ids(obj):
    if isinstance(obj, (list, tuple)):
        vals = []
        for x in obj:
            vals.extend(flatten_positive_ids(x))
        return vals
    if type(obj) is int and 0 < obj < 2147483647:
        return [obj]
    return []


def split_com_result(result):
    # E3 methods generally return (number_of_ids, ID-array).
    if isinstance(result, (tuple, list)) and result and type(result[0]) is int:
        return result[0], flatten_positive_ids(result[1:])
    return (result if type(result) is int else None), []


def com_ids(obj, method_name, text_type=None):
    """Read E3 arrays using both official-wrapper and direct pywin32 conventions."""
    method = getattr(obj, method_name, None)
    if method is None:
        return []
    variants = [()] if text_type is None else [(text_type,), (None, text_type), ([], text_type)]
    for args in variants:
        try:
            raw = method(*args)
            count, ids = split_com_result(raw)
            if ids:
                return list(dict.fromkeys(ids[:count] if count and count > 0 else ids))
            if isinstance(raw, (tuple, list)) and len(raw) > 1 and all(type(i) is int and i > 0 for i in raw):
                return list(dict.fromkeys(raw))
        except Exception:
            pass
    # The Python COM interface sometimes requires an explicit output array.
    try:
        import pythoncom
        import win32com.client as wc
    except ImportError:
        return []
    for variant_type, seed in (
        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, []),
    ):
        try:
            arr = wc.VARIANT(variant_type, seed)
            result = method(arr) if text_type is None else method(arr, text_type)
            _, returned_ids = split_com_result(result)
            ids = returned_ids or flatten_positive_ids(arr.value)
            if ids:
                return list(dict.fromkeys(ids))
        except Exception:
            pass
    return []


def related_ids(obj, candidates, text_type=None):
    for method in candidates:
        ids = com_ids(obj, method, text_type=text_type)
        if ids:
            return ids
    return []


def get_scalar(obj, methods):
    for name in methods:
        method = getattr(obj, name, None)
        if method is None:
            continue
        try:
            result = method()
        except Exception:
            continue
        if isinstance(result, (str, float, int)) and result != NO_VALUE:
            return result
    return None


def make_object(job, kind, object_id):
    creator = getattr(job, f'Create{kind}Object', None)
    if not callable(creator):
        raise RuntimeError(f'E3 does not expose Create{kind}Object().')
    obj = creator()
    result = obj.SetId(int(object_id))
    if result is False:
        raise RuntimeError(f'Could not resolve {kind} ID {object_id}.')
    return obj


class E3Project:
    def __init__(self):
        self.backend = ''
        try:
            import e3series
            application = e3series.Application()
            job = application.CreateJobObject()
            if job.GetId():
                self.application = application
                self.job = job
                self.backend = 'official e3series wrapper'
                return
        except Exception:
            pass
        import win32com.client as win32
        self.application = win32.Dispatch('CT.Application')
        self.job = self.application.CreateJobObject()
        self.backend = 'pywin32 COM'
        if not self.job.GetId():
            raise RuntimeError('E3.series is not running with an open project.')

    def sheet_name(self, sheet_id):
        try:
            sheet = make_object(self.job, 'Sheet', sheet_id)
            return as_text(get_scalar(sheet, ('GetName', 'GetSheetName')))
        except Exception:
            return ''

    def on_j2(self, device):
        """Optional E3 drawing-sheet scope: applies only if a page J2 exists."""
        sids = related_ids(device, ('GetSymbolIds', 'GetAllSymbolIds'))
        for sid in sids:
            try:
                sym = make_object(self.job, 'Symbol', sid)
                sheet_id = get_scalar(sym, ('GetSheetId',))
                if sheet_id and self.sheet_name(sheet_id).casefold() == SHEET_NAME.casefold():
                    return True
            except Exception:
                continue
        return False

    def j2_page_exists(self):
        for sid in related_ids(self.job, ('GetSheetIds',)):
            if self.sheet_name(sid).casefold() == SHEET_NAME.casefold():
                return True
        return False

    def devices(self):
        ids = related_ids(self.job, ('GetDeviceIds', 'GetAllDeviceIds', 'GetBlockIds'))
        return [(i, make_object(self.job, 'Device', i)) for i in ids]

    def block_device(self, designation, log):
        hits = []
        for did, obj in self.devices():
            name = as_text(get_scalar(obj, ('GetName', 'GetDeviceDesignation', 'GetDeviceName')))
            if device_key(name) == device_key(designation):
                hits.append((did, obj))
        if not hits:
            raise LookupError(f'No E3 block/device matches C4 = {designation!r}.')
        if self.j2_page_exists():
            scoped = [(did, o) for did, o in hits if self.on_j2(o)]
            if not scoped:
                raise LookupError(f'{designation!r} was found, but not on E3 drawing sheet J2.')
            hits = scoped
            log.append('E3 drawing sheet J2 found; restricting block match to that page.')
        else:
            log.append('No E3 drawing sheet named J2 found; matching the block by C4 designation.')
        if len(hits) != 1:
            raise LookupError(f'C4 matches {len(hits)} E3 devices; use a unique device designation.')
        log.append(f'BLOCK {designation!r} => E3 device ID {hits[0][0]}')
        return hits[0]

    def connectors(self, block_device):
        ids = related_ids(block_device, ('GetConnectorIds', 'GetBlockConnectorIds', 'GetSubDeviceIds'))
        # Some E3 versions attach connector IDs to the placed block symbol.
        if not ids:
            for sym_id in related_ids(block_device, ('GetSymbolIds',)):
                try:
                    sym = make_object(self.job, 'Symbol', sym_id)
                except Exception:
                    continue
                ids.extend(related_ids(sym, ('GetConnectorIds', 'GetBlockConnectorIds')))
        found = []
        for cid in dict.fromkeys(ids):
            try:
                obj = make_object(self.job, 'Device', cid)
                designation = as_text(get_scalar(obj, ('GetName', 'GetDeviceDesignation', 'GetDeviceName')))
                code = as_text(get_scalar(obj, ('GetComponentName', 'GetComponentCode',
                                                'GetPartNumber', 'GetTypeName')))
                found.append((cid, obj, designation, code))
            except Exception:
                continue
        return found

    def connector(self, choices, connector_name, connector_type):
        key = device_key(connector_name)
        hits = [x for x in choices if key in connector_keys(x[2])]
        if len(hits) > 1 and connector_type:
            narrowed = [x for x in hits if as_text(x[3]).casefold() == connector_type.casefold()]
            if narrowed:
                hits = narrowed
        if not hits:
            raise LookupError(f'Connector {connector_name!r} not found within the matched block.')
        if len(hits) != 1:
            raise LookupError(f'Connector {connector_name!r} matches {len(hits)} E3 connectors; skipped.')
        return hits[0]

    def pin(self, connector_obj, pin_name):
        pin_ids = related_ids(connector_obj, ('GetPinIds', 'GetAllPinIds'))
        hits = []
        for pid in pin_ids:
            try:
                p = make_object(self.job, 'Pin', pid)
                value = get_scalar(p, ('GetName', 'GetPinName'))
                if as_text(value).casefold() == pin_name.casefold():
                    hits.append(pid)
            except Exception:
                continue
        if not hits:
            raise LookupError(f'Pin {pin_name!r} not found on connector.')
        if len(hits) > 1:
            raise LookupError(f'Pin {pin_name!r} is ambiguous on connector.')
        return hits[0]


def resolve(data, e3, log):
    _, block = e3.block_device(data.block_name, log)
    candidates = e3.connectors(block)
    log.append(f'Associated connector devices found: {len(candidates)}')
    indexed = {}
    for i in range(data.connector_count):
        examples = [e for e in data.entries if e.connector_index == i]
        if not examples:
            continue
        sample = examples[0]
        try:
            choice = e3.connector(candidates, sample.connector_name, sample.connector_type)
            indexed[i] = choice
            log.append(f'CONNECTOR {sample.connector_name!r} (type {sample.connector_type!r}) '
                       f'=> {choice[2]!r}, ID {choice[0]}, component {choice[3]!r}')
        except Exception as exc:
            log.append('SKIP CONNECTOR: ' + str(exc))
    plans = []
    for entry in data.entries:
        choice = indexed.get(entry.connector_index)
        if not choice:
            log.append(f'SKIP {entry.cell}: no unique connector match for {entry.connector_name!r}')
            continue
        try:
            pin_id = e3.pin(choice[1], entry.pin_name)
            plans.append(ResolvedPin(entry, choice[0], pin_id))
            log.append(f'PLAN {entry.cell}: {entry.connector_name}/{entry.pin_name} '
                       f'-> {entry.signal_name!r} [pin ID {pin_id}]')
        except Exception as exc:
            log.append(f'SKIP {entry.cell}: {exc}')
    return plans


def get_signal(pin):
    """Return the actual signal name, never confuse an E3 signal ID with a name."""
    for getter in ('GetLocalSignalName', 'GetSignalName', 'GetSignal'):
        value = get_scalar(pin, (getter,))
        if isinstance(value, str):
            return value
    return None


def set_pin_signal(pin, target_name):
    """Write pin-level signal, and read back where E3 exposes its name."""
    existing = get_signal(pin)
    if existing is not None and as_text(existing) == target_name:
        return 'already assigned'
    failures = []
    for method_name in ('SetLocalSignalName', 'SetSignalName', 'SetSignal'):
        method = getattr(pin, method_name, None)
        if not callable(method):
            continue
        try:
            ret = method(target_name)
            if ret is False or (type(ret) is int and ret < 0):
                failures.append(f'{method_name} returned {ret!r}')
                continue
            current = get_signal(pin)
            if current is not None and as_text(current) == target_name:
                return method_name + ' (verified)'
            if current is None:
                return method_name + ' (setter accepted; readback unavailable)'
            failures.append(f'{method_name} did not verify (readback={current!r})')
        except Exception as exc:
            failures.append(f'{method_name}: {type(exc).__name__}: {exc}')
    raise RuntimeError('No pin-level signal setter succeeded; ' + ('; '.join(failures) or
                       'this E3 build exposes no supported signal setter.'))


def uniquely_owned_pin_symbols(job, pin_id, connector_id=None):
    """Find placed connector symbols which represent only the requested pin."""
    pin = make_object(job, 'Pin', pin_id)
    symbol_ids = related_ids(pin, ('GetSymbolIds',))
    sid = get_scalar(pin, ('GetSymbolId', 'GetOwnerSymbolId'))
    if type(sid) is int and sid > 0:
        symbol_ids.append(sid)
    if connector_id:
        # This path was needed in the previous Match Properties utility:
        # connector device -> placed pin symbol -> signal text type 7.
        conn = make_object(job, 'Device', connector_id)
        symbol_ids.extend(related_ids(conn, ('GetSymbolIds',)))
    found = []
    for symbol_id in dict.fromkeys(symbol_ids):
        try:
            sym = make_object(job, 'Symbol', symbol_id)
            owner_pins = related_ids(sym, ('GetPinIds',))
            single = get_scalar(sym, ('GetPinId', 'GetOwnerPinId'))
            if not owner_pins and type(single) is int and single > 0:
                owner_pins = [single]
            if owner_pins == [pin_id]:
                found.append(sym)
        except Exception:
            continue
    return found


def pin_label_text_ids(job, pin_id, connector_id=None):
    """Get type-7 text positively owned by this pin, never nearby unowned text."""
    pin = make_object(job, 'Pin', pin_id)
    ids = related_ids(pin, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE)
    if ids:
        return ids
    for sym in uniquely_owned_pin_symbols(job, pin_id, connector_id):
        ids.extend(related_ids(sym, ('GetTextIds',), text_type=LOCAL_SIGNAL_TEXT_TYPE))
    return list(dict.fromkeys(ids))


def make_pin_label_visible(job, pin_id, connector_id=None):
    """Try owner-level visibility, including placed one-pin connector symbols."""
    pin = make_object(job, 'Pin', pin_id)
    show = getattr(pin, 'SetTextVisibility', None)
    if callable(show):
        try:
            show(LOCAL_SIGNAL_TEXT_TYPE, 1)
        except Exception:
            pass
    for sym in uniquely_owned_pin_symbols(job, pin_id, connector_id):
        show = getattr(sym, 'SetTextVisibility', None)
        if callable(show):
            try:
                show(LOCAL_SIGNAL_TEXT_TYPE, 1)
            except Exception:
                pass


def set_visible(obj):
    """Visibility varies by E3 release; invoke only known text visibility names."""
    pairs = (('SetVisible', 'GetVisible'),
             ('SetVisibility', 'GetVisibility'),
             ('SetTextVisible', 'GetTextVisible'),
             ('SetDisplayed', 'GetDisplayed'))
    for setter, getter in pairs:
        method = getattr(obj, setter, None)
        if not callable(method):
            continue
        try:
            result = method(1)
            check = get_scalar(obj, (getter,))
            if result is not False and (check is None or check in (1, True)):
                return setter
        except Exception:
            continue
    return None


def format_local_signal(job, pin_id, connector_id=None):
    pin = make_object(job, 'Pin', pin_id)
    # Request visibility BEFORE collecting labels: hidden labels may not be
    # returned by all E3 builds until SetTextVisibility has been called.
    make_pin_label_visible(job, pin_id, connector_id)
    text_ids = pin_label_text_ids(job, pin_id, connector_id)
    if not text_ids:
        return 'NO LOCAL-SIGNAL TEXT OBJECT: signal set, text formatting/visibility not verified'
    good = 0
    issues = []
    for tid in text_ids:
        try:
            text = make_object(job, 'Text', tid)
            text.SetHeight(TEXT_HEIGHT_IN)
            text.SetStyle(BOLD_STYLE)
            height = get_scalar(text, ('GetHeight',))
            style = get_scalar(text, ('GetStyle',))
            formatted = (isinstance(height, (float, int)) and
                         abs(float(height) - TEXT_HEIGHT_IN) <= 0.0001 and style == BOLD_STYLE)
            visibility_api = set_visible(text)
            if not visibility_api:
                # Some releases put type visibility on the parent pin.
                show = getattr(pin, 'SetTextVisibility', None)
                if callable(show):
                    try:
                        show(LOCAL_SIGNAL_TEXT_TYPE, 1)
                        visibility_api = 'Pin.SetTextVisibility'
                    except Exception:
                        pass
            if formatted and visibility_api:
                good += 1
            else:
                issues.append(f'text {tid}: format={formatted}, '
                              f'visibility method={visibility_api or "NOT FOUND"}')
        except Exception as exc:
            issues.append(f'text {tid}: {exc}')
    if good == len(text_ids):
        return f'OK: {good} text label(s), bold 0.15 in, made visible'
    return f'PARTIAL ({good}/{len(text_ids)} labels): ' + '; '.join(issues)


def log_path(path):
    return path.with_name(path.stem + '_J2_E3_Import_Log.txt')


def write_log(path, content):
    log_path(path).write_text(content + '\n', encoding='utf-8')


def perform(path, do_apply=False):
    lines = [f'EXCEL: {path}', f'WORKSHEET: {SHEET_NAME}']
    data = parse_workbook(path)
    require_no_duplicate_pins(data)
    lines += [f'BLOCK (C4): {data.block_name}',
              f'CONNECTORS (C5): {data.connector_count}',
              f'DEFINED PINS: {len(data.entries)}',
              f'B2: {data.check_b2!r}', '']
    # Worksheet issues that are NOT duplicate errors are reported and skipped.
    for problem in data.duplicate_problems:
        lines.append('WORKSHEET NOTICE: ' + problem)
    try:
        import pythoncom
    except ImportError:
        raise RuntimeError('pywin32 is required on Windows: py -m pip install pywin32')
    pythoncom.CoInitialize()
    try:
        e3 = E3Project()
        lines.append(f'E3 connection: {e3.backend}')
        plans = resolve(data, e3, lines)
        lines.append(f'RESOLVED PINS: {len(plans)} / {len(data.entries)}')
        if not do_apply:
            lines.append('PREVIEW ONLY: E3 has not been modified.')
        else:
            lines.append('APPLY RESULTS:')
            for plan in plans:
                entry = plan.entry
                try:
                    pin = make_object(e3.job, 'Pin', plan.pin_id)
                    signal_result = set_pin_signal(pin, entry.signal_name)
                    format_result = format_local_signal(e3.job, plan.pin_id, plan.connector_id)
                    lines.append(f'{entry.connector_name}/{entry.pin_name} -> '
                                 f'{entry.signal_name!r}: signal={signal_result}; {format_result}')
                except Exception as exc:
                    lines.append(f'FAILED {entry.connector_name}/{entry.pin_name}: {exc}')
        del e3
    finally:
        gc.collect()
        pythoncom.CoUninitialize()
    output = '\n'.join(lines)
    write_log(path, output)
    return output


def run_gui(first_file=None):
    import tkinter as tk
    from tkinter import ttk, filedialog, messagebox, scrolledtext

    win = tk.Tk()
    win.title('E3.series | Import Local Signals from Excel J2')
    win.geometry('1040x690')
    file_var = tk.StringVar(value=str(first_file or ''))
    banner = ttk.Frame(win, padding=12)
    banner.pack(fill='x')
    ttk.Label(banner, text='Excel workbook (.xlsx)').pack(side='left')
    ttk.Entry(banner, textvariable=file_var, width=85).pack(side='left', padx=8, fill='x', expand=True)
    def browse():
        name = filedialog.askopenfilename(title='Choose Excel pin assignment workbook',
                                          filetypes=[('Excel workbook', '*.xlsx *.xlsm')])
        if name:
            file_var.set(name)
    ttk.Button(banner, text='Browse…', command=browse).pack(side='left')
    instructions = ('J2: C4=block, C5=connector count, B7/C7 down=connector name/type, '
                    'I:J / M:N / Q:R ... = pin/signal pairs (2 spacer columns). Blank signals do nothing. '
                    'Preview first; then Apply.')
    ttk.Label(win, text=instructions, padding=(12, 0, 12, 8), wraplength=990).pack(fill='x')
    output = scrolledtext.ScrolledText(win, wrap='none', font=('Consolas', 10))
    output.pack(fill='both', expand=True, padx=12, pady=(0, 12))
    controls = ttk.Frame(win, padding=(12, 0, 12, 12))
    controls.pack(fill='x')
    btn_apply = ttk.Button(controls, text='Apply to E3', state='disabled')
    btn_apply.pack(side='right')
    def output_text(s):
        output.delete('1.0', tk.END)
        output.insert('1.0', s)
        output.see('1.0')
    def do_preview():
        btn_apply.config(state='disabled')
        try:
            fname = Path(file_var.get()).expanduser()
            if not fname.is_file():
                raise FileNotFoundError('Select an existing Excel workbook first.')
            result = perform(fname, False)
            output_text(result)
            if 'RESOLVED PINS: 0 /' not in result:
                btn_apply.config(state='normal')
        except Exception:
            output_text(traceback.format_exc())
    def do_apply():
        if not messagebox.askyesno('Apply local signals',
                   'Assign signals and format their labels in the CURRENT E3 project?\n'
                   'Save a project copy before applying.'):
            return
        try:
            result = perform(Path(file_var.get()).expanduser(), True)
            output_text(result)
            messagebox.showinfo('Completed', 'Finished. Review the per-pin results and log.\n'
                                + str(log_path(Path(file_var.get()))))
        except Exception:
            output_text(traceback.format_exc())
            messagebox.showerror('Import failed', 'See the error details in the window.')
        btn_apply.config(state='disabled')
    ttk.Button(controls, text='Preview / Resolve', command=do_preview).pack(side='right', padx=10)
    btn_apply.config(command=do_apply)
    win.mainloop()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--excel', type=Path, help='Optional Excel file to preselect')
    parser.add_argument('--child', action='store_true', help=argparse.SUPPRESS)
    parser.add_argument('--preview', action='store_true', help='Print preview and exit')
    parser.add_argument('--apply', action='store_true', help='Apply without GUI after selecting Excel')
    args = parser.parse_args()
    if args.preview or args.apply:
        if not args.excel:
            parser.error('--preview/--apply requires --excel FILE')
        print(perform(args.excel, args.apply))
        return
    if sys.platform == 'win32' and not args.child:
        # E3 add-on runner must exit immediately; otherwise E3 selection may lock.
        command = [sys.executable, str(Path(__file__).resolve()), '--child']
        if args.excel:
            command += ['--excel', str(args.excel)]
        subprocess.Popen(command, creationflags=subprocess.CREATE_NEW_PROCESS_GROUP,
                         cwd=str(Path(__file__).resolve().parent),
                         stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL,
                         stderr=subprocess.DEVNULL, close_fds=True)
        return
    run_gui(args.excel)


if __name__ == '__main__':
    main()
