def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        selected_path = filedialog.askopenfilename(initialdir=default_folder, filetypes=[("Excel Files", "*.xlsx"), ("All Files", "*.*")])
    except Exception:
        return None

    if not selected_path:
        root.destroy()
        return None

    root.destroy()

    return os.path.abspath(selected_path)


def generate_ordered_pin_ids(dict_to_order, dest):
    return {str(pin): dict_to_order[str(pin)] for pin in dest}


def generate_placement_locations(Pins, layout):
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    PIN_OFFSET_FROM_CONN = 3.0

    X_START = 1.0
    Y_START = 32.5

    main_conn_locs = []

    for i, row in enumerate(layout):
        for j, conn in enumerate(row):
            if conn is not None:
                flip = i > 0
                main_conn_locs.append([conn, X_START + (i * 3.0), Y_START - (j * 1.0), flip])

    y_pin_start = 32.9

    for r in main_conn_locs:
        if y_pin_start > r[2]:
            y_pin_start = r[2]

    y_pin_start -= PIN_OFFSET_FROM_CONN

    print(Pins)
    print(layout)

    pin_locs = []

    for dictionary in Pins:
        conn_key = list(dictionary.keys())[0]
        conn = int(conn_key.split(".")[1][1:])

        count = len(dictionary) - 1

        top = True
        flip = 0
        xstart = 0
        ystart = 0

        for r in main_conn_locs:
            if int(r[0]) == conn:
                xstart = r[1]
                ystart = r[2]
                flip = r[3]

        for r in main_conn_locs:
            if r[3] == flip:
                if ystart < r[2]:
                    top = False

        if top:
            ystart = y_pin_start

        else:
            pins_above = 0
            conns_above = 0

            for r in Pins:
                r_conn_key = list(r.keys())[0]
                r_conn = int(r_conn_key.split(".")[1][1:])

                if r_conn == conn:
                    break

                for c in main_conn_locs:
                    if int(c[0]) == r_conn and c[3] == flip:
                        if c[2] > ystart:
                            pins_above += len(r) - 1
                            conns_above += 1

            ystart = y_pin_start - (pins_above * 0.25) - (conns_above * 0.5)

        pin_locs.append([conn, xstart, ystart, flip])

    return main_conn_locs, pin_locs, y_pin_start + 2


def generate_aligned_pin_y(pinouts, y_start, pin_spacing=0.25, connector_gap=0.5):
    pin_y = {}

    current_y = y_start

    previous_c1 = None
    previous_c2 = None

    for row_index, (c1, p1, c2, p2) in enumerate(pinouts):
        if c1 is None or c2 is None:
            continue

        if previous_c1 is not None:
            connector_changed = c1 != previous_c1 or c2 != previous_c2

            if connector_changed:
                current_y -= connector_gap

        pin_y[(int(c1), str(p1))] = current_y
        pin_y[(int(c2), str(p2))] = current_y

        current_y -= pin_spacing

        previous_c1 = c1
        previous_c2 = c2

    return pin_y


def get_spare_pins(pins, pinouts):
    spares = {}

    pin_lookup = {}

    for pindict in pins:
        connector = list(pindict.keys())[0]
        pin_lookup[connector] = pindict

    for c1, p1, c2, p2 in pinouts:
        if c1 is not None and p1 is not None and c2 is None:
            connector = f"W1.P{c1}"

            if connector not in spares:
                spares[connector] = {}

            if connector in pin_lookup:
                pin_id = pin_lookup[connector].get(str(p1))

                if pin_id is not None:
                    spares[connector][str(p1)] = pin_id

        elif c2 is not None and p2 is not None and c1 is None:
            connector = f"W1.P{c2}"

            if connector not in spares:
                spares[connector] = {}

            if connector in pin_lookup:
                pin_id = pin_lookup[connector].get(str(p2))

                if pin_id is not None:
                    spares[connector][str(p2)] = pin_id

    return spares
