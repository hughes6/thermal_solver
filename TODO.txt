def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog
    default_folder = os.path.dirname(default_full_path)

    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    try:
        selected_path = filedialog.askopenfilename(initialdir=default_folder, filetypes=[("PDF Files", "*.pdf"), ("All Files", "*.*")])
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def extract_pdf_text(file_path):
    try:
        all_text = []
        with pdfplumber.open(file_path) as pdf:
            for page_num, page in enumerate(pdf.pages, start=1):
                text = page.extract_text()
                if text:
                    all_text.append(f"--- Page {page_num} ---\n{text}")
        return "\n".join(all_text) if all_text else "No text found."
    except FileNotFoundError:
        return "Error: File not found."
    except Exception as e:
        return f"Error reading PDF: {e}"


def extract_data_by_keywords(pdf_path, keywords):
    """
    Extract text from PDF and organize data by given keywords.
    
    Args:
        pdf_path (str): Path to the PDF file.
        keywords (list): List of keywords to search in the PDF text.
        
    Returns:
        dict: A dictionary where each keyword maps to a list of extracted values.
    """
    data = {key: [] for key in keywords}  # Initialize dictionary
    
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if not text:
                continue
            
            # Iterate through each keyword
            for key in keywords:
                # Simple approach: find lines containing the keyword
                pattern = rf"{key}\s*[:\-]?\s*(.*)"
                matches = re.findall(pattern, text, re.IGNORECASE)
                
                for match in matches:
                    # Clean up leading/trailing spaces
                    data[key].append(match.strip())
                    
    return data
