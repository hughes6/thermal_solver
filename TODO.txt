#!/usr/bin/env python3
"""Compare two Excel/CSV spreadsheets and generate HTML and CSV diff reports.

Requires Python 3.9+; no third-party packages.

Usage:
    python spreadsheet_diff.py                   # Pick files with dialog boxes
    python spreadsheet_diff.py old.xlsx new.xlsx
    python spreadsheet_diff.py old.xlsx new.xlsx -o my_report.html
    python spreadsheet_diff.py old.xlsx new.xlsx --ignore-cached

Supported inputs: .xlsx, .xlsm, .csv, .tsv (not legacy .xls or .xlsb).
Compares nonempty cells at their exact coordinates, including formulas and
cached formula results by default. Cell formatting is not compared.
"""

from __future__ import annotations

import argparse
import csv
from dataclasses import dataclass
from datetime import datetime, timedelta
from decimal import Decimal, InvalidOperation
from collections import Counter
from html import escape
from pathlib import Path
import posixpath
import re
import sys
import zipfile
from xml.etree import ElementTree as ET

MAIN = "http://schemas.openxmlformats.org/spreadsheetml/2006/main"
DOCREL = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"
PKGREL = "http://schemas.openxmlformats.org/package/2006/relationships"
MAX_HTML_ROWS = 5000


def tag(local):
    return f"{{{MAIN}}}{local}"


@dataclass(frozen=True)
class Cell:
    kind: str             # text / number / boolean / error / formula / date
    raw: str              # normalized comparable value
    shown: str            # readable representation
    cached: str = ""      # formula result (may be absent / stale)
    formula_kind: str = ""  # shared formula information when applicable


def number_key(s):
    try:
        return str(Decimal(s).normalize())
    except (InvalidOperation, ValueError):
        return s


def col_number(cell_ref):
    m = re.match(r"([A-Za-z]+)", cell_ref)
    if not m:
        return 0
    n = 0
    for ch in m.group(1).upper():
        n = 26 * n + ord(ch) - ord("A") + 1
    return n


def address_key(ref):
    m = re.fullmatch(r"([A-Za-z]+)([0-9]+)", ref)
    return (int(m.group(2)), col_number(ref)) if m else (float("inf"), ref)


def date_style_formats(z):
    """Recognize conventional Excel date formats for legible output."""
    try:
        root = ET.fromstring(z.read("xl/styles.xml"))
    except KeyError:
        return {}
    builtins = set(range(14, 23)) | set(range(27, 37)) | set(range(45, 48)) | set(range(50, 59))
    custom = {}
    numfmts = root.find(tag("numFmts"))
    if numfmts is not None:
        for f in numfmts.findall(tag("numFmt")):
            custom[int(f.get("numFmtId", "0"))] = f.get("formatCode", "")
    result = {}
    xfs = root.find(tag("cellXfs"))
    if xfs is None:
        return result
    for index, xf in enumerate(xfs.findall(tag("xf"))):
        n = int(xf.get("numFmtId", "0"))
        fmt = custom.get(n, "")
        # Strip format literals and color/condition brackets before matching.
        fmt_clean = re.sub(r'"[^"]*"|\\.|\[[^\]]*\]', "", fmt)
        is_date = n in builtins or bool(re.search(r"[ymdhHsS]", fmt_clean))
        if is_date:
            result[index] = "time" if (n in {18, 19, 20, 21, 45, 46, 47} or
                                          ("h" in fmt_clean.lower() and "d" not in fmt_clean.lower() and "y" not in fmt_clean.lower())) else "date"
    return result


def excel_date(raw, date_kind, use_1904):
    try:
        d = Decimal(raw)
        serial = float(d)
        if not (-100000 < serial < 2000000):
            return None
        if use_1904:
            origin = datetime(1904, 1, 1)
        else:
            origin = datetime(1899, 12, 30) if serial >= 60 else datetime(1899, 12, 31)
        dt = origin + timedelta(days=serial)
        if date_kind == "time":
            return dt.strftime("%H:%M:%S")
        if d % 1:
            return dt.strftime("%Y-%m-%d %H:%M:%S")
        return dt.strftime("%Y-%m-%d")
    except (ValueError, OverflowError, InvalidOperation):
        return None


def shared_strings(z):
    try:
        with z.open("xl/sharedStrings.xml") as stream:
            root = ET.parse(stream).getroot()
    except KeyError:
        return []
    strings = []
    for item in root.findall(tag("si")):
        strings.append("".join(t.text or "" for t in item.iter(tag("t"))))
    return strings


def column_label(index):
    letters = ""
    while index:
        index, digit = divmod(index - 1, 26)
        letters = chr(65 + digit) + letters
    return letters


def expand_shared_formula(formula, base_ref, target_ref):
    """Translate the usual A1/$A$1 references used in Excel shared formulas."""
    old_row, old_col = address_key(base_ref)
    new_row, new_col = address_key(target_ref)
    dr, dc = new_row - old_row, new_col - old_col
    if dr == 0 and dc == 0:
        return formula
    pattern = re.compile(r"(?<![A-Za-z0-9_.])(\$?)([A-Za-z]{1,3})(\$?)([0-9]{1,7})(?![A-Za-z0-9_.]|\s*\()")

    def replace_ref(match):
        dollar_col, letters, dollar_row, row = match.groups()
        col = col_number(letters)
        nrow = int(row)
        if not 1 <= col <= 16384 or not 1 <= nrow <= 1048576:
            return match.group(0)
        ncol = col if dollar_col else col + dc
        nrow = nrow if dollar_row else nrow + dr
        if not (1 <= ncol <= 16384 and 1 <= nrow <= 1048576):
            return "#REF!"
        return f"{dollar_col}{column_label(ncol)}{dollar_row}{nrow}"

    # Preserve quoted text and quoted worksheet names.
    parts = re.split(r'''("(?:[^"]|"")*"|'(?:[^']|'')*')''', formula)
    return "".join(p if p.startswith(('"', "'")) else pattern.sub(replace_ref, p) for p in parts)


def xml_cell(element, strings, date_styles, use_1904, shared_masters):
    typ = element.get("t", "n")
    value_el = element.find(tag("v"))
    raw = value_el.text if value_el is not None and value_el.text is not None else ""
    formula_el = element.find(tag("f"))
    if formula_el is not None:
        formula = formula_el.text or ""
        ftype = formula_el.get("t", "")
        si = formula_el.get("si", "")
        if ftype == "shared":
            if formula:
                shared_masters[si] = (element.get("r", ""), formula)
            elif si in shared_masters:
                base_ref, base_formula = shared_masters[si]
                formula = expand_shared_formula(base_formula, base_ref, element.get("r", ""))
            else:
                formula = f"[shared formula group {si}; expression not stored here]"
        shown = "=" + formula
        if raw:
            shown += "  [cached: " + raw + "]"
        return Cell("formula", formula, shown, raw, ftype)
    if typ == "inlineStr":
        inline = element.find(tag("is"))
        raw = "" if inline is None else "".join(t.text or "" for t in inline.iter(tag("t")))
        return Cell("text", raw, raw) if raw else None
    if typ == "s":
        try:
            text = strings[int(raw)]
        except (IndexError, ValueError):
            text = f"[invalid shared-string index: {raw}]"
        return Cell("text", text, text) if text else None
    if typ in ("str", "e", "d"):
        if not raw:
            return None
        return Cell({"str": "text", "e": "error", "d": "date"}[typ], raw, raw)
    if typ == "b":
        return Cell("boolean", "1" if raw in ("1", "true", "TRUE") else "0",
                    "TRUE" if raw in ("1", "true", "TRUE") else "FALSE")
    if raw:
        style_num = int(element.get("s", "0"))
        rendered_date = excel_date(raw, date_styles[style_num], use_1904) if style_num in date_styles else None
        display = f"{rendered_date} (serial {raw})" if rendered_date else raw
        return Cell("number", number_key(raw), display)
    return None  # A formatted but empty cell isn't a content change.


def load_excel(path):
    sheets = {}
    with zipfile.ZipFile(path) as z:
        strings = shared_strings(z)
        dates = date_style_formats(z)
        workbook = ET.fromstring(z.read("xl/workbook.xml"))
        wbpr = workbook.find(tag("workbookPr"))
        use_1904 = wbpr is not None and wbpr.get("date1904", "0").lower() in ("1", "true")
        relationships = ET.fromstring(z.read("xl/_rels/workbook.xml.rels"))
        targets = {}
        for r in relationships.findall(f"{{{PKGREL}}}Relationship"):
            dest = r.get("Target", "")
            if r.get("TargetMode") == "External":
                continue
            if dest.startswith("/"):
                part = dest.lstrip("/")
            else:
                part = posixpath.normpath(posixpath.join("xl", dest))
            targets[r.get("Id")] = part
        ss = workbook.find(tag("sheets"))
        if ss is None:
            return sheets
        for sheet_el in ss.findall(tag("sheet")):
            name = sheet_el.get("name", "Unnamed")
            rel = sheet_el.get(f"{{{DOCREL}}}id")
            part = targets.get(rel)
            if not part or not part.startswith("xl/worksheets/"):
                continue  # Ignore non-worksheet entries such as chartsheets.
            cells = {}
            shared_masters = {}
            with z.open(part) as stream:
                # Parsing only cell nodes keeps memory reasonable for large sheets.
                for event, elem in ET.iterparse(stream, events=("end",)):
                    if elem.tag == tag("c"):
                        ref = elem.get("r", "")
                        cell = xml_cell(elem, strings, dates, use_1904, shared_masters)
                        if ref and cell is not None:
                            cells[ref.upper()] = cell
                        elem.clear()
            sheets[name] = cells
    return sheets


def load_delimited(path, delimiter):
    cells = {}
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        for r, row in enumerate(csv.reader(stream, delimiter=delimiter), start=1):
            for c, value in enumerate(row, start=1):
                if not value:
                    continue
                n = c
                letters = ""
                while n:
                    n, digit = divmod(n - 1, 26)
                    letters = chr(65 + digit) + letters
                cells[f"{letters}{r}"] = Cell("text", value, value)
    return {"Data": cells}


def load_spreadsheet(path):
    suffix = path.suffix.lower()
    if suffix in (".xlsx", ".xlsm"):
        return load_excel(path)
    if suffix in (".csv", ".tsv"):
        return load_delimited(path, "," if suffix == ".csv" else "\t")
    raise ValueError("Unsupported file type. Use .xlsx, .xlsm, .csv, or .tsv (not .xls/.xlsb).")


def classify(old, new, ignore_cached):
    if old is None:
        return "ADDED"
    if new is None:
        return "REMOVED"
    if old.kind == new.kind and old.raw == new.raw:
        if old.kind == "formula" and not ignore_cached and old.cached != new.cached:
            return "CACHED RESULT CHANGED"
        return None
    if old.kind == "formula" or new.kind == "formula":
        return "FORMULA CHANGED"
    return "CHANGED"


def excel_safe(value):
    """Prevent CSV cells from becoming executable spreadsheet formulas."""
    value = str(value)
    if value and value.lstrip(" \t\r\n").startswith(("=", "+", "-", "@")):
        return "'" + value
    return value


def choose_paths(args):
    if args.old and args.new:
        old = Path(args.old).expanduser()
        new = Path(args.new).expanduser()
    elif not args.old and not args.new:
        try:
            from tkinter import Tk, filedialog
            root = Tk()
            root.withdraw()
            root.attributes("-topmost", True)
            allowed = [("Excel and CSV files", "*.xlsx *.xlsm *.csv *.tsv"), ("All files", "*.*")]
            old_name = filedialog.askopenfilename(title="Select the OLD spreadsheet", filetypes=allowed, parent=root)
            if not old_name:
                root.destroy()
                raise SystemExit("No old spreadsheet selected.")
            new_name = filedialog.askopenfilename(title="Select the NEW spreadsheet", filetypes=allowed, parent=root)
            if not new_name:
                root.destroy()
                raise SystemExit("No new spreadsheet selected.")
            old, new = Path(old_name), Path(new_name)
            if args.output is None:
                out_name = filedialog.asksaveasfilename(title="Save the diff report", parent=root,
                    defaultextension=".html", filetypes=[("HTML report", "*.html")],
                    initialfile="spreadsheet_diff_report.html")
                if not out_name:
                    root.destroy()
                    raise SystemExit("Report save canceled.")
                args.output = out_name
            root.destroy()
        except (ImportError, RuntimeError, OSError):
            print("File dialogs unavailable; enter file paths instead.")
            old = Path(input("Path to OLD spreadsheet: ").strip().strip('"')).expanduser()
            new = Path(input("Path to NEW spreadsheet: ").strip().strip('"')).expanduser()
    else:
        raise SystemExit("Provide both old and new paths, or neither to use file dialogs.")
    for file in (old, new):
        if not file.is_file():
            raise FileNotFoundError(f"File not found: {file}")
    out = Path(args.output).expanduser() if args.output else new.parent / "spreadsheet_diff_report.html"
    if out.suffix.lower() != ".html":
        out = out.with_suffix(".html")
    return old, new, out


def html_top(old, new, stats, sheet_stats, total):
    rows = "\n".join(
        f"<tr><td>{escape(name)}</td><td>{escape(state)}</td>"
        f"<td>{o:,}</td><td>{n:,}</td><td>{count:,}</td></tr>"
        for name, state, o, n, count in sheet_stats
    )
    stat_cards = "".join(
        f'<div class="stat"><span>{escape(k.title())}</span><strong>{v:,}</strong></div>'
        for k, v in [("Total changes", total), ("Added", stats["ADDED"]),
                     ("Removed", stats["REMOVED"]), ("Changed", stats["CHANGED"]),
                     ("Formula changed", stats["FORMULA CHANGED"]),
                     ("Cached result changed", stats["CACHED RESULT CHANGED"])]
    )
    return f'''<!DOCTYPE html><html lang="en"><head><meta charset="utf-8">

.card {{ background: white; border: 1px solid #dfe6ee; border-radius: 12px; padding: 22px; margin: 18px 0; }}
.stats {{ display: grid; grid-template-columns: repeat(auto-fit, minmax(150px, 1fr)); gap: 12px; }}
.stat {{ padding: 15px; border: 1px solid #dfe6ee; background: white; border-radius: 10px; }}

<div class="table-wrap"><table><thead><tr><th>Sheet</th><th>Cell</th><th>Change</th><th>Old value</th><th>New value</th></tr></thead><tbody id="changes">
'''


def html_end(preview, total):
    missing = max(0, total - preview)
    note = (f"Only the first {preview:,} changes appear in this HTML preview; "
            "the CSV contains all changes." if missing else f"Showing all {preview:,} changes.")
    return f'''</tbody></table></div><p class="muted">{escape(note)}</p></section>
<script>
function filterRows() {{
 const q=document.getElementById('filter').value.toLowerCase();
 let matched=0;
 for (const row of document.querySelectorAll('#changes tr')) {{
   const ok=row.textContent.toLowerCase().includes(q);
   row.hidden=!ok; if(ok) matched++;
 }}
 document.getElementById('shown').textContent=matched.toLocaleString()+' preview rows match';
}}
filterRows();
</script></body></html>'''


def make_report(old_path, new_path, html_path, ignore_cached=False):
    old_sheets = load_spreadsheet(old_path)
    new_sheets = load_spreadsheet(new_path)
    all_sheets = sorted(set(old_sheets) | set(new_sheets), key=str.casefold)
    stats = Counter()
    summary = []
    changes = []
    for sheet in all_sheets:
        old = old_sheets.get(sheet, {})
        new = new_sheets.get(sheet, {})
        sheet_count = 0
        for addr in sorted(set(old) | set(new), key=address_key):
            oc = old.get(addr)
            nc = new.get(addr)
            change = classify(oc, nc, ignore_cached)
            if change is None:
                continue
            stats[change] += 1
            sheet_count += 1
            changes.append((sheet, addr, change, oc.shown if oc else "", nc.shown if nc else ""))
        state = ("ADDED SHEET" if sheet not in old_sheets else
                 "REMOVED SHEET" if sheet not in new_sheets else "EXISTING")
        summary.append((sheet, state, len(old), len(new), sheet_count))
    total = len(changes)
    html_path.parent.mkdir(parents=True, exist_ok=True)
    csv_path = html_path.with_suffix(".csv")
    with csv_path.open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.writer(stream)
        writer.writerow(["Worksheet", "Cell", "Change type", "Old value", "New value"])
        for row in changes:
            writer.writerow([excel_safe(v) for v in row])
    with html_path.open("w", encoding="utf-8") as stream:
        stream.write(html_top(old_path, new_path, stats, summary, total))
        for sheet, addr, change, ov, nv in changes[:MAX_HTML_ROWS]:
            klass = change.replace(" ", "-")
            stream.write("<tr>" + f"<td>{escape(sheet)}</td><td>{escape(addr)}</td>"
                         + f'<td><span class="badge {klass}">{escape(change)}</span></td>'
                         + f'<td class="value">{escape(ov)}</td><td class="value">{escape(nv)}</td></tr>\n')
        stream.write(html_end(min(MAX_HTML_ROWS, total), total))
    return csv_path, total, stats, summary


def main():
    parser = argparse.ArgumentParser(description="Compare every cell in two Excel or CSV workbooks.")
    parser.add_argument("old", nargs="?", help="Old workbook (.xlsx/.xlsm/.csv/.tsv)")
    parser.add_argument("new", nargs="?", help="New workbook (.xlsx/.xlsm/.csv/.tsv)")
    parser.add_argument("-o", "--output", help="HTML report filename (CSV generated alongside it)")
    parser.add_argument("--ignore-cached", action="store_true",
                        help="Don't treat changed cached formula results as differences")
    args = parser.parse_args()
    try:
        old, new, report = choose_paths(args)
        csv_report, total, stats, summary = make_report(old, new, report, args.ignore_cached)
    except (FileNotFoundError, ValueError, KeyError, zipfile.BadZipFile, ET.ParseError, OSError) as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 1
    print(f"Compared {len(summary)} worksheet(s). Found {total:,} changed cell(s).")
    for name, state, nold, nnew, nchanges in summary:
        print(f"  {name}: {nchanges:,} changes ({state.lower()})")
    print(f"HTML report: {report.resolve()}")
    print(f"Full CSV log: {csv_report.resolve()}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
