"""E3.series - spreadsheet-driven, formatting-only connector pin repair.

Reads the SAME Excel layout as the existing Pin Arranger:
    B2 = duplicate-pin check (PASS)
    C4 = optional owning device name, e.g. PNL2
    C5 = connector count
    J2 = optional schematic page/sheet
    Connector #n: left column I+4*n, right column J+4*n (n starts at 0)
                  pin row count at right-column row 5
                  connector designation at right-column row 6
                  pin names in row 8 onward
    J1 (pin pitch) is intentionally ignored: NO pins are moved.

For each Excel connector, the script finds matching E3 pin symbols across ALL
its physical CABBLOCKs, including a split-off lower block. The TOP pin of
EACH side of EACH physical block shows device designation text type 12,
bold, height 0.20; every other pin on that same side hides type 12.
Local signal names (type 617) are made visible, black, bold, height 0.15,
and aligned by the ACTUAL E3 block side. Pin numbers (type 203) are made
visible and bold, matching the earlier arranger's formatter.

No symbol placement, resizing, removal, rebuilding, or pin-order changes.

Requires: Windows, Python, pywin32, openpyxl, running E3.series project.
For visible equipment identification, the E3 display property
"Equipment Identification" must be enabled in E3.

Usage:
    python E3_Split_Block_Pin_Formatter.py              # choose Excel with dialog
    python E3_Split_Block_Pin_Formatter.py pins.xlsm    # format immediately
    python E3_Split_Block_Pin_Formatter.py --dry-run pins.xlsm
"""

from __future__ import annotations

import argparse
import math
import os
import sys
import traceback
from collections import defaultdict
from pathlib import Path

try:
    from openpyxl import load_workbook
except ImportError:
    raise SystemExit("Missing openpyxl. Install with: pip install openpyxl pywin32")

# Import win32com only when connecting so --help / offline tests work.

# ------------------------ Formatting settings ----------------------------
SIGNAL_NAME_TYPE = 617
PIN_NUMBER_TYPE = 203
DEVICE_DESIGNATION_TYPE = 12
BOLD_STYLE = 1                 # Same Zuken SetStyle(1) used in Pin Arranger
SIGNAL_NAME_HEIGHT = 0.15       # Same as the original formatter
TOP_DESIGNATION_HEIGHT = 0.20
BOLD_PIN_NUMBERS = True         # Preserve the old formatter's pin-number bolding
SIGNAL_BLACK = True             # Preserve black signal labels
SIGNAL_ALIGNMENT_BY_SIDE = True # LEFT -> 1; RIGHT -> 3, as in Pin Arranger
MAX_PIN_EDGE_OFFSET = 0.65     # Max allowed pin origin distance from block edge
BLOCK_LOCATION_TOLERANCE = 0.10
TOP_Y_TOLERANCE = 1e-4
COM_MISSING = -2147352572

LOG_STREAM = None


def log(message=""):
    line = str(message)
    print(line, flush=True)
    if LOG_STREAM is not None:
        LOG_STREAM.write(line + "\n")
        LOG_STREAM.flush()


def normalize_pin(value):
    if value is None:
        return None
    if isinstance(value, float) and math.isfinite(value) and value.is_integer():
        return str(int(value))
    val = str(value).strip()
    return val or None


def parse_id_result(value):
    """Support E3's (count, ID-array) COM return and normal ID arrays."""
    if value is None:
        return []
    if isinstance(value, (tuple, list)):
        if (len(value) >= 2 and isinstance(value[1], (tuple, list))
                and isinstance(value[0], (int, float))):
            ids = value[1]
            try:
                expected_count = int(value[0])
                ids = list(ids)
                if expected_count < len(ids):
                    # COM wrappers can prepend output/status values.
                    ids = ids[-expected_count:] if expected_count > 0 else []
            except (ValueError, TypeError):
                pass
        else:
            ids = value
    else:
        ids = [value]
    result = []
    for v in ids:
        try:
            i = int(v)
            if i > 0 and i != COM_MISSING:
                result.append(i)
        except (TypeError, ValueError):
            pass
    return result


def get_ids(method, *arguments):
    # E3's GetSymbolIds(None, 0) uses an output array BEFORE a mode argument;
    # GetViewIds(None), GetAllSheetIds(None) and some wrapper versions differ.
    attempts = ((None, *arguments), arguments, (*arguments, None))
    for args in attempts:
        try:
            value = parse_id_result(method(*args))
            if value:
                return value
        except Exception:
            continue
    return []


def select_id(obj, object_id):
    """Do not mistake E3 versions returning None from SetId for failure."""
    try:
        result = obj.SetId(int(object_id))
        if isinstance(result, (int, float)) and result == 0:
            return False
        if hasattr(obj, "GetId"):
            current = obj.GetId()
            if current is not None and int(current) != int(object_id):
                return False
        return True
    except Exception:
        return False


def optional_call(obj, name):
    try:
        return getattr(obj, name)()
    except Exception:
        return None


def read_sheet(prj, sht):
    rows = []
    ids = get_ids(prj.GetAllSheetIds)
    for sid in ids:
        if not select_id(sht, sid):
            continue
        name = str(sht.GetName()).strip()
        aliases = {name.casefold()}
        for method in ("GetSheetNumber", "GetNumber"):
            val = optional_call(sht, method)
            if val is not None:
                aliases.add(str(normalize_pin(val)).casefold())
        rows.append({"id": sid, "name": name, "aliases": aliases})
    if not rows:
        raise RuntimeError("E3 returned no sheets in the open project.")
    return rows


def same_sheet_label(label, expected):
    a, b = str(label).strip().casefold(), str(expected).strip().casefold()
    if a == b:
        return True
    return a.isdecimal() and b.isdecimal() and int(a) == int(b)


def read_excel(excel_path):
    wb = load_workbook(str(excel_path), read_only=True, data_only=True)
    try:
        ws = wb.active
        dup = ws["B2"].value
        if str(dup).strip().casefold() != "pass":
            raise ValueError(
                "Excel B2 duplicate-pin check is not PASS. As in Pin Arranger, "
                "recalculate/save the workbook in Excel and resolve any duplicate-pin errors."
            )
        num_raw = ws["C5"].value
        if num_raw is None or isinstance(num_raw, bool):
            raise ValueError("Excel C5 connector count is blank or invalid.")
        count = int(num_raw)
        if count < 0 or count > 2000:
            raise ValueError(f"Excel C5 connector count {count} is invalid.")
        owner = normalize_pin(ws["C4"].value)
        requested_page = normalize_pin(ws["J2"].value)
        records = []
        for idx in range(count):
            left_col = 9 + 4 * idx
            right_col = left_col + 1
            pin_rows_raw = ws.cell(row=5, column=right_col).value
            count_rows = int(pin_rows_raw) if pin_rows_raw is not None else 0
            if count_rows < 0 or count_rows > 10000:
                raise ValueError(f"Connector {idx + 1}: invalid pin row count {count_rows}.")
            conn_name = normalize_pin(ws.cell(row=6, column=right_col).value)
            pins_left, pins_right = [], []
            for r in range(8, 8 + count_rows):
                lp = normalize_pin(ws.cell(row=r, column=left_col).value)
                rp = normalize_pin(ws.cell(row=r, column=right_col).value)
                if lp is not None:
                    pins_left.append(lp)
                if rp is not None:
                    pins_right.append(rp)
            all_pins = pins_left + pins_right
            if not all_pins:
                continue
            if not conn_name:
                raise ValueError(f"Excel connector {idx + 1}: designation in row 6 is blank.")
            if len(all_pins) != len(set(all_pins)):
                raise ValueError(f"Excel connector {conn_name}: duplicate pin names in its columns.")
            records.append({
                "name": conn_name, "expected_pins": set(all_pins),
                "excel_count": len(all_pins), "left": pins_left, "right": pins_right
            })
        if not records:
            raise ValueError("No connector pins were specified in the spreadsheet.")
        names = [r["name"].casefold() for r in records]
        if len(set(names)) != len(names):
            raise ValueError("A connector designation appears more than once in Excel; "
                             "combine its listed pins into one connector group.")
        return owner, requested_page, records
    finally:
        wb.close()


def get_symbol_location(sym):
    try:
        raw = sym.GetSchemaLocation(None, None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 3:
            return None
        return {"sheet_id": int(raw[0]), "x": float(raw[1]), "y": float(raw[2])}
    except Exception:
        return None


def get_symbol_area(sym):
    try:
        raw = sym.GetPlacedArea(None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 5 or int(raw[0]) <= 0:
            return None
        l, b, r, t = map(float, raw[1:5])
        if not (r > l and t > b):
            return None
        return {"left": l, "bottom": b, "right": r, "top": t}
    except Exception:
        return None


def read_device_views(dev, device_id):
    if not select_id(dev, device_id):
        return []
    return get_ids(dev.GetViewIds)


def read_device_symbols(dev, view_id):
    if not select_id(dev, view_id):
        return []
    return get_ids(dev.GetSymbolIds, 0)


def find_cabblocks(prj, dev, owner_name, requested_page, sheets):
    allowed_pages = None
    if requested_page:
        eligible = [s for s in sheets if any(same_sheet_label(a, requested_page)
                                            for a in s["aliases"])]
        if len(eligible) != 1:
            choices = ", ".join(s["name"] for s in sheets)
            raise RuntimeError(f"J2 page {requested_page!r} matched {len(eligible)} E3 "
                               f"sheets. Available sheets: {choices}")
        allowed_pages = {eligible[0]["id"]}
    device_ids = get_ids(prj.GetAllDeviceIds)
    if owner_name:
        filtered = []
        for did in device_ids:
            if select_id(dev, did) and str(dev.GetName()).strip() == owner_name:
                filtered.append(did)
        if not filtered:
            raise RuntimeError(f"C4 device {owner_name!r} not found by E3 GetName().")
        device_ids = filtered
    blocks = {}
    sym = prj.CreateSymbolObject()
    for did in device_ids:
        for view_id in read_device_views(dev, did):
            if not select_id(dev, view_id):
                continue
            try:
                view_number = int(dev.GetViewNumber())
            except Exception:
                view_number = None
            for sid in read_device_symbols(dev, view_id):
                if sid in blocks or not select_id(sym, sid):
                    continue
                try:
                    kind = str(sym.GetSymbolTypeName() or "")
                except Exception:
                    continue
                if "CABBLOCK" not in kind.upper():
                    continue
                loc, area = get_symbol_location(sym), get_symbol_area(sym)
                if not loc or not area:
                    continue
                if allowed_pages is not None and loc["sheet_id"] not in allowed_pages:
                    continue
                blocks[sid] = {
                    "id": sid, "owner": did, "view_id": view_id,
                    "view_number": view_number,
                    "sheet_id": loc["sheet_id"], "kind": kind,
                    **area,
                }
    if not blocks:
        raise RuntimeError("No placed CABBLOCKs matched Excel's C4/J2 filters.")
    return list(blocks.values())


def inside_block(loc, block):
    tol = BLOCK_LOCATION_TOLERANCE
    return (loc is not None and loc["sheet_id"] == block["sheet_id"]
            and block["left"] - tol <= loc["x"] <= block["right"] + tol
            and block["bottom"] - tol <= loc["y"] <= block["top"] + tol)


def match_designation(actual, requested):
    """Actual J1A or J1A#1 may match Excel J1A, not J1A2."""
    if not actual:
        return False
    a, b = str(actual).strip().casefold(), str(requested).strip().casefold()
    return bool(b) and (a == b or
                        (a.startswith(b + "#") and a[len(b) + 1:].isdecimal()))


def scan_sheet_symbols(prj, sht, sheet_id):
    if not select_id(sht, sheet_id):
        raise RuntimeError(f"Could not select E3 sheet ID {sheet_id}.")
    ids = get_ids(sht.GetSymbolIds)
    if ids:
        return ids
    log(f"  Sheet {sheet_id} has no sheet-local symbol ID array; scanning project IDs instead.")
    return get_ids(prj.GetSymbolIds)


def read_pin_identity(prj, symbol_id, expected_connectors):
    sym = prj.CreateSymbolObject()
    if not select_id(sym, symbol_id):
        return None
    kind = str(optional_call(sym, "GetSymbolTypeName") or "")
    # The pin arranger uses MilSymbolReceptacle and MilBlockSymbolReceptacle.
    if "RECEPTACLE" not in kind.upper() or "¶" not in kind:
        return None
    loc = get_symbol_location(sym)
    if loc is None:
        return None
    txt = prj.CreateTextObject()
    label, pin_number, type_ids = None, None, defaultdict(list)
    for tid in get_ids(sym.GetTextIds):
        if not select_id(txt, tid):
            continue
        try:
            typ = int(txt.GetType())
        except Exception:
            continue
        type_ids[typ].append(tid)
        if typ in (DEVICE_DESIGNATION_TYPE, PIN_NUMBER_TYPE):
            raw_text = optional_call(txt, "GetText")
            if typ == DEVICE_DESIGNATION_TYPE and raw_text is not None:
                label = str(raw_text).strip()
            elif typ == PIN_NUMBER_TYPE:
                pin_number = normalize_pin(raw_text)
    if not pin_number or not label:
        return None
    if not any(match_designation(label, name) for name in expected_connectors):
        return None
    return {"symbol_id": symbol_id, "kind": kind, "location": loc,
            "pin_name": pin_number, "designation": label,
            "texts": dict(type_ids)}


def discover_pins(prj, sht, blocks, connectors):
    """Read ALL pins on ALL candidate blocks; no Excel pin completeness per block."""
    blocks_by_sheet = defaultdict(list)
    for b in blocks:
        blocks_by_sheet[b["sheet_id"]].append(b)
    expected_names = [x["name"] for x in connectors]
    pins = []
    for sheet_id, sheet_blocks in blocks_by_sheet.items():
        ids = scan_sheet_symbols(prj, sht, sheet_id)
        log(f"Scanning E3 sheet {sheet_id}: {len(ids)} symbols, "
            f"{len(sheet_blocks)} candidate CABBLOCKs")
        block_ids = {b["id"] for b in sheet_blocks}
        for symbol_id in set(ids):
            if symbol_id in block_ids:
                continue
            ident = read_pin_identity(prj, symbol_id, expected_names)
            if ident is None:
                continue
            matched_blocks = [b for b in sheet_blocks
                              if inside_block(ident["location"], b)]
            if not matched_blocks:
                continue
            # An E3 symbol in two CABBLOCK rectangles is ambiguous. Do not
            # arbitrarily assign its top designation to the wrong block.
            if len(matched_blocks) > 1:
                detail = [(b["id"], b["owner"]) for b in matched_blocks]
                raise RuntimeError(
                    f"Pin symbol {symbol_id} {ident['designation']} "
                    f"pin {ident['pin_name']} intersects multiple CABBLOCKs "
                    f"{detail}. No formatting applied."
                )
            pins.append({**ident, "block": matched_blocks[0]})
    return pins


def determine_side(pin):
    x = pin["location"]["x"]
    block = pin["block"]
    ld, rd = abs(x - block["left"]), abs(x - block["right"])
    if abs(ld - rd) < 0.05 or min(ld, rd) > MAX_PIN_EDGE_OFFSET:
        raise RuntimeError(
            f"Cannot determine LEFT/RIGHT for {pin['designation']} "
            f"pin {pin['pin_name']} (symbol {pin['symbol_id']}) in block {block['id']}; "
            f"X={x:.4f}, left={block['left']:.4f}, right={block['right']:.4f}. "
            "No formatting applied."
        )
    return "LEFT" if ld < rd else "RIGHT"


def make_plan(pins, connectors):
    """Require one unambiguous device owner for EACH Excel connector."""
    planned = []
    used_symbols = set()
    for conn in connectors:
        matching = [p for p in pins if match_designation(p["designation"], conn["name"])]
        by_owner = defaultdict(list)
        for p in matching:
            by_owner[p["block"]["owner"]].append(p)
        matching_owners = []
        for owner, owner_pins in by_owner.items():
            names = {p["pin_name"] for p in owner_pins}
            if conn["expected_pins"].issubset(names):
                matching_owners.append(owner)
        if len(matching_owners) != 1:
            diagnostics = [f"owner {owner}: {len(ps)} pins; "
                           f"Excel missing {sorted(conn['expected_pins'] - {p['pin_name'] for p in ps})[:12]}"
                           for owner, ps in by_owner.items()]
            raise RuntimeError(
                f"Connector {conn['name']}: expected one E3 device owner containing all "
                f"{conn['excel_count']} Excel pin(s), found {len(matching_owners)}. "
                f"Candidates: {' | '.join(diagnostics) or 'none'}. "
                "Set C4 (device) and/or J2 (page) in Excel if needed. "
                "NO formatting changes applied."
            )
        owner = matching_owners[0]
        selected = by_owner[owner]
        by_block_and_name = defaultdict(list)
        for p in selected:
            by_block_and_name[(p["block"]["id"], p["pin_name"])].append(p)
            if DEVICE_DESIGNATION_TYPE not in p["texts"]:
                raise RuntimeError(
                    f"Connector {conn['name']} pin {p['pin_name']} symbol {p['symbol_id']} "
                    "has no device-designation text (type 12). "
                    "Enable E3 'Equipment Identification' display property "
                    "and re-run. NO formatting changes applied."
                )
        duplicated = [k for k, group in by_block_and_name.items() if len(group) > 1]
        if duplicated:
            raise RuntimeError(f"Connector {conn['name']}: duplicated pin numbers "
                               f"inside physical block(s): {duplicated}. No changes applied.")
        by_side = defaultdict(list)
        for p in selected:
            p["side"] = determine_side(p)
            key = (p["block"]["id"], p["side"])
            by_side[key].append(p)
        for (block_id, side), group in by_side.items():
            group.sort(key=lambda p: (-p["location"]["y"], p["pin_name"]))
            if (len(group) > 1 and
                abs(group[0]["location"]["y"] - group[1]["location"]["y"]) <= TOP_Y_TOLERANCE):
                raise RuntimeError(f"Connector {conn['name']} block {block_id} {side}: "
                                   "more than one pin shares the highest Y position; "
                                   "cannot choose one top pin safely. No changes applied.")
            for row, p in enumerate(group):
                if p["symbol_id"] in used_symbols:
                    raise RuntimeError(f"Symbol {p['symbol_id']} was selected twice. No changes applied.")
                used_symbols.add(p["symbol_id"])
                planned.append({**p, "connector": conn["name"],
                                "top": row == 0, "row": row})
        log(f"  {conn['name']}: owner {owner}, {len(selected)} E3 pins "
            f"({conn['excel_count']} named in Excel), "
            f"{len({p['block']['id'] for p in selected})} physical block(s)")
    return sorted(planned, key=lambda p: (p["connector"], p["block"]["sheet_id"],
                                           p["block"]["id"], p["side"], p["row"]))


def apply_plan(prj, plan):
    """Set only font/visibility/alignment/colour; never move symbols or pins."""
    text = prj.CreateTextObject()
    completed = 0
    for p in plan:
        label = f"{p['connector']} pin {p['pin_name']} (symbol {p['symbol_id']})"
        type_ids = p["texts"]
        # GetTextIds identified these exact text objects during preflight.
        # Rebind each before use, so COM's current selection cannot drift.
        for text_id in type_ids.get(DEVICE_DESIGNATION_TYPE, []):
            if not select_id(text, text_id):
                raise RuntimeError(f"{label}: cannot select type-12 text ID {text_id}")
            if p["top"]:
                text.SetVisibility(1)
                text.SetStyle(BOLD_STYLE)
                text.SetHeight(TOP_DESIGNATION_HEIGHT)
            else:
                text.SetVisibility(0)
        for text_id in type_ids.get(SIGNAL_NAME_TYPE, []):
            if not select_id(text, text_id):
                raise RuntimeError(f"{label}: cannot select type-617 text ID {text_id}")
            text.SetVisibility(1)
            text.SetStyle(BOLD_STYLE)
            text.SetHeight(SIGNAL_NAME_HEIGHT)
            if SIGNAL_BLACK:
                text.SetColour(0)
            if SIGNAL_ALIGNMENT_BY_SIDE:
                text.SetAlignment(1 if p["side"] == "LEFT" else 3)
        if BOLD_PIN_NUMBERS:
            for text_id in type_ids.get(PIN_NUMBER_TYPE, []):
                if not select_id(text, text_id):
                    raise RuntimeError(f"{label}: cannot select type-203 text ID {text_id}")
                text.SetVisibility(1)
                text.SetStyle(BOLD_STYLE)
        completed += 1
        log(f"  FORMATTED {label} | block {p['block']['id']} "
            f"| {p['side']} row {p['row'] + 1} | "
            f"designation {'ON, 0.20 BOLD' if p['top'] else 'OFF'} "
            f"| signal labels {len(type_ids.get(SIGNAL_NAME_TYPE, []))}")
    return completed


def file_dialog():
    import tkinter as tk
    from tkinter import filedialog
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    path = filedialog.askopenfilename(
        title="Select the SAME Pin Arranger Excel workbook",
        filetypes=[("Excel macro workbook", "*.xlsm"),
                   ("Excel workbook", "*.xlsx"), ("All files", "*.*")],
    )
    root.destroy()
    return path


def message_box(title, message, error=False):
    try:
        import tkinter as tk
        from tkinter import messagebox
        root = tk.Tk()
        root.withdraw()
        root.attributes("-topmost", True)
        if error:
            messagebox.showerror(title, message, parent=root)
        else:
            messagebox.showinfo(title, message, parent=root)
        root.destroy()
    except Exception:
        pass


def run(excel_path, dry_run=False):
    global LOG_STREAM
    excel_path = Path(excel_path).resolve()
    if not excel_path.is_file():
        raise FileNotFoundError(f"Excel file does not exist: {excel_path}")
    log_path = excel_path.with_name("E3_Pin_Formatter_Log.txt")
    try:
        LOG_STREAM = open(log_path, "w", encoding="utf-8")
    except OSError:
        log_path = None
        LOG_STREAM = None
    try:
        log("=" * 72)
        log("E3 SPLIT-BLOCK PIN FORMATTER - FORMATTING ONLY, NO MOVEMENT")
        log(f"Workbook: {excel_path}")
        log("=" * 72)
        owner, page, connectors = read_excel(excel_path)
        log(f"Excel C4 device: {owner or '<any>'}")
        log(f"Excel J2 page: {page or '<any>'}")
        log(f"Excel connectors: {[c['name'] for c in connectors]}")
        log("Enable E3 View Properties > Equipment Identification before running if "
            "designation labels are not available/visible.")
        log("Excel side/row order is NOT used to move pins or select 'top'.")
        log("'Top' is determined by the highest placed pin on each physical block side.")
        try:
            import win32com.client as win32
        except ImportError as exc:
            raise RuntimeError("pywin32 is required. Install: pip install pywin32") from exc
        app = prj = dev = sht = None
        try:
            app = win32.gencache.EnsureDispatch("CT.Application")
            prj = app.CreateJobObject()
            dev = prj.CreateDeviceObject()
            sht = prj.CreateSheetObject()
            log("Connected to E3.series; all targeting is read-only until preflight passes.")
            sheets = read_sheet(prj, sht)
            blocks = find_cabblocks(prj, dev, owner, page, sheets)
            log(f"Found {len(blocks)} placed CABBLOCK(s) within C4/J2 restrictions.")
            pins = discover_pins(prj, sht, blocks, connectors)
            plan = make_plan(pins, connectors)
            log("\nPREFLIGHT PASSED. Target per physical CABBLOCK side:")
            by_side = defaultdict(list)
            for p in plan:
                by_side[(p["connector"], p["block"]["id"], p["side"])].append(p)
            for (connector, block_id, side), group in sorted(by_side.items()):
                top_pin = next(p for p in group if p["top"])
                log(f"  {connector} | block {block_id} | {side}: "
                    f"{len(group)} pins, top = {top_pin['pin_name']} "
                    f"at Y={top_pin['location']['y']:.3f}")
            log(f"Total symbols to format: {len(plan)}")
            if dry_run:
                log("DRY RUN: no changes were made to E3.")
            else:
                log("\nAPPLYING FORMATTING (no delete/move/resize operations)")
                try:
                    app.PutMessage("\nStarting split-block pin text formatter")
                except Exception:
                    pass
                completed = apply_plan(prj, plan)
                log(f"SUCCESS: formatted {completed} pin symbols. Save the project in E3.")
            summary = (f"{'Preview complete (no changes).' if dry_run else 'Pin formatting complete.'}\n\n"
                       f"{len(plan)} pins, {len(by_side)} physical block sides, "
                       f"{len(connectors)} connector(s).\n"
                       f"No pins were moved or recreated.\n\n"
                       "Ensure E3's 'Equipment Identification' display property "
                       "is enabled so device designations appear.\n"
                       "Save your E3 drawing.\n\n"
                       f"Log: {log_path or 'console only'}")
            message_box("E3 Pin Formatter", summary)
            return 0
        finally:
            sht = dev = prj = app = None
    except Exception:
        log("\nERROR: " + traceback.format_exc())
        raise
    finally:
        if LOG_STREAM is not None:
            LOG_STREAM.close()
        LOG_STREAM = None


def main(argv=None):
    parser = argparse.ArgumentParser(description="Format pins across split E3 CABBLOCKs, without moving pins")
    parser.add_argument("excel", nargs="?", help="Pin Arranger .xlsm / .xlsx (file picker if omitted)")
    parser.add_argument("--dry-run", action="store_true", help="Resolve and preview all pin targets; do not write to E3")
    args = parser.parse_args(argv)
    path = args.excel or file_dialog()
    if not path:
        log("No Excel workbook selected.")
        return 1
    try:
        return run(path, dry_run=args.dry_run)
    except Exception as exc:
        log(f"FAILED: {exc}")
        message_box("E3 Pin Formatter - Error", str(exc) + "\n\nSee E3_Pin_Formatter_Log.txt next to the workbook.", error=True)
        return 1


if __name__ == "__main__":
    sys.exit(main())
