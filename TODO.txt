cur.execute("""
SELECT
    c.ENTRY,
    c.VERSION,
    c.Description,
    c.Class,
    COUNT(DISTINCT p.PinNumber) AS NumberOfPins
FROM dbo.ComponentData c
LEFT JOIN dbo.PhysPin p
    ON p.Entry = c.ENTRY
WHERE c.ENTRYTYP = 24
GROUP BY
    c.ENTRY,
    c.VERSION,
    c.Description,
    c.Class
ORDER BY
    c.ENTRY,
    c.VERSION
""")
