def ask_save_path(default_full_path=""):
    # Optional E3 Mega Tools workbook selection. Fall back to this tool's own picker.
    import os as _mega_os
    _mega_excel = _mega_os.environ.get("E3_MEGA_BLOCK_MAKER_WORKBOOK", "").strip()
    if _mega_excel:
        if not _mega_os.path.isfile(_mega_excel):
            raise FileNotFoundError(f"Selected Block Maker workbook no longer exists: {_mega_excel}")
        return _mega_os.path.abspath(_mega_excel)

    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)
