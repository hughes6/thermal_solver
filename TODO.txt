import inspect
import win32com.client as win32


app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sym = prj.CreateSymbolObject()


print("")
print("============================================================")
print("E3 SYMBOL METHOD SIGNATURE PROBE")
print("============================================================")


method_names = [
    "SetDeviceLocation",
    "GetSchemaLocation",
    "Place",
    "SetDisplayLength",
    "GetDisplayLength"
]


for method_name in method_names:
    print("")
    print("============================================================")
    print("METHOD:", method_name)
    print("============================================================")

    if not hasattr(sym, method_name):
        print("METHOD NOT AVAILABLE")
        continue

    method = getattr(sym, method_name)

    print("BOUND METHOD:", method)

    try:
        signature = inspect.signature(method)
        print("BOUND SIGNATURE:", signature)
    except Exception as error:
        print("BOUND SIGNATURE ERROR:", repr(error))

    try:
        class_method = getattr(type(sym), method_name)
        print("CLASS METHOD:", class_method)
    except Exception as error:
        print("CLASS METHOD ERROR:", repr(error))
        class_method = None

    if class_method is not None:
        try:
            class_signature = inspect.signature(class_method)
            print("CLASS SIGNATURE:", class_signature)
        except Exception as error:
            print("CLASS SIGNATURE ERROR:", repr(error))

        try:
            source = inspect.getsource(class_method)
            print("CLASS SOURCE:")
            print(source)
        except Exception as error:
            print("CLASS SOURCE ERROR:", repr(error))


print("")
print("============================================================")
print("OBJECT INFORMATION")
print("============================================================")
print("OBJECT TYPE:", type(sym))
print("OBJECT MODULE:", type(sym).__module__)


try:
    print("CLSID:", sym.CLSID)
except Exception as error:
    print("CLSID ERROR:", repr(error))


print("")
print("============================================================")
print("DONE")
print("============================================================")


del sym
del prj
del app
