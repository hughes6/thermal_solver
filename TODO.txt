import inspect
import win32com.client as win32


#====================================================================================================================================
# CONNECT TO E3
#====================================================================================================================================
app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
dev = prj.CreateDeviceObject()
sym = prj.CreateSymbolObject()
pin = prj.CreatePinObject()


#====================================================================================================================================
# OBJECTS TO INSPECT
#====================================================================================================================================
objects = {
    "APPLICATION": app,
    "PROJECT": prj,
    "DEVICE": dev,
    "SYMBOL": sym,
    "PIN": pin
}


#====================================================================================================================================
# TRY TO CREATE CABLE OBJECT
#====================================================================================================================================
try:
    cable = prj.CreateCableObject()
    objects["CABLE"] = cable
    print("CreateCableObject SUCCESS")
except Exception as error:
    cable = None
    print("CreateCableObject FAILED:", repr(error))


#====================================================================================================================================
# SEARCH TERMS
#====================================================================================================================================
keywords = [
    "shield",
    "twist",
    "bundle",
    "auto",
    "place",
    "length",
    "display",
    "core",
    "wire",
    "cable"
]


#====================================================================================================================================
# INSPECT METHODS
#====================================================================================================================================
for object_name, obj in objects.items():
    print("")
    print("============================================================")
    print(object_name)
    print("============================================================")

    names = []

    for name in dir(obj):
        lower_name = name.lower()

        if any(keyword in lower_name for keyword in keywords):
            names.append(name)

    names.sort()

    for name in names:
        try:
            member = getattr(obj, name)
        except Exception as error:
            print(name, "| GET FAILED:", repr(error))
            continue

        if callable(member):
            try:
                signature = inspect.signature(member)
                print(name, signature)
            except Exception:
                print(name, "| CALLABLE")
        else:
            print(name, "| PROPERTY:", repr(member))


#====================================================================================================================================
# PROJECT SETTINGS RELATED TO BUNDLE SYMBOLS
#====================================================================================================================================
print("")
print("============================================================")
print("CURRENT BUNDLE / SHIELD SETTINGS")
print("============================================================")

setting_names = [
    "DEFAULT_SHIELDSYM",
    "DEFAULT_TWISTED_PAIRSYM",
    "DEFAULT_BUNDLESYM",
    "BUNDLE_DISTANCE_PIN",
    "BUNDLE_DISTANCE_MIDDLE",
    "SHIELD_BOTTOM_OVERLAPPING",
    "SHIELD_TOP_OVERLAPPING",
    "TWISTED_PAIR_BOTTOM_OVERLAPPING",
    "TWISTED_PAIR_TOP_OVERLAPPING",
    "ADJUST_BUNDLE_SYMBOL_SIZE_AFTER_MODIFYING_CONNECTION_LINE"
]

for setting_name in setting_names:
    try:
        value = prj.GetSettingValue(setting_name)
        print(setting_name, "=", value)
    except Exception as error:
        print(setting_name, "FAILED:", repr(error))


#====================================================================================================================================
# CLEANUP
#====================================================================================================================================
if cable is not None:
    del cable

del pin
del sym
del dev
del prj
del app
