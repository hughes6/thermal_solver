"""E3.series | Bulk CABBLOCK Header Formatter using Saber Parameter 1 (COM-safe lookup).

SELECT one or more placed CABBLOCKs in E3, press Enter in the always-on-top
instruction window (or while E3 has focus). Repeat until Close/X/Escape.

Each selected block displays, from top to bottom:
  1. Location                text type 15
  2. Saber Parameter 1       project-specific ATTRIBUTE TEXT TYPE linked to
                            the symbol attribute named ``saberParam1``
  3. Equipment Identification   text type 1000
  4. Block Device Designation   text type 212

All four texts: visible, bold, black, center-justified, height 0.25.
For a blank Saber Parameter 1, set the SELECTED SYMBOL's ``saberParam1``
attribute to XXX, so the ordinary shared Device Function is NOT changed.
Existing nonblank Saber Parameter 1 values are preserved.

IMPORTANT: ``saberParam1`` is an ATTRIBUTE NAME, not a universal E3 text-type
number. The script tries to resolve the linked text type from the project's
text-type definitions or from the attached text metadata. If the project does
not expose that link, give the type number using ``--saber-type N`` AFTER
checking the text type in the E3 database configuration. It will never
silently reuse type 14, 1010 or another guessed text type.

If the selected CABBLOCK is missing Saber Parameter 1, an optional creation
path checks for symbol-level AddAttributeValue and AddText COM members, and
verifies that the new text ID is actually attached to that exact block.
Creation is attempted only with a VERIFIED project-specific --saber-type N.
If the COM interface cannot do this, it stops without creating an unrelated
free text or editing the shared Device Function.

The only E3 objects changed are the specified texts of the selected blocks
and each block symbol's own saberParam1 attribute (when blank). Any OLD
Device Function label (text type 14) attached to a selected block is HIDDEN
only, to prevent two labels overlaying in the second row. Its underlying
property value remains unchanged. No Excel, no pins, no dimensions. Test on a saved E3 project copy before production.

Use --dry-run to inspect without changing E3; --saber-type N to specify your
project-specific linked text type. Other observed type overrides remain.
"""

def log(message=""):
    line = str(message)
    print(line, flush=True)
    if LOG_STREAM is not None:
        LOG_STREAM.write(line + "\n")
        LOG_STREAM.flush()


def normalize_pin(value):
    if value is None:
        return None
    if isinstance(value, float) and math.isfinite(value) and value.is_integer():
        return str(int(value))
    val = str(value).strip()
    return val or None


def parse_id_result(value):
    """Support E3's (count, ID-array) COM return and normal ID arrays."""
    if value is None:
        return []
    if isinstance(value, (tuple, list)):
        if (len(value) >= 2 and isinstance(value[1], (tuple, list))
                and isinstance(value[0], (int, float))):
            ids = value[1]
            try:
                expected_count = int(value[0])
                ids = list(ids)
                if expected_count < len(ids):
                    # COM wrappers can prepend output/status values.
                    ids = ids[-expected_count:] if expected_count > 0 else []
            except (ValueError, TypeError):
                pass
        else:
            ids = value
    else:
        ids = [value]
    result = []
    for v in ids:
        try:
            i = int(v)
            if i > 0 and i != COM_MISSING:
                result.append(i)
        except (TypeError, ValueError):
            pass
    return result


def get_ids(method, *arguments):
    # E3's GetSymbolIds(None, 0) uses an output array BEFORE a mode argument;
    # GetViewIds(None), GetAllSheetIds(None) and some wrapper versions differ.
    attempts = ((None, *arguments), arguments, (*arguments, None))
    for args in attempts:
        try:
            value = parse_id_result(method(*args))
            if value:
                return value
        except Exception:
            continue
    return []


def select_id(obj, object_id):
    """Do not mistake E3 versions returning None from SetId for failure."""
    try:
        result = obj.SetId(int(object_id))
        if isinstance(result, (int, float)) and result == 0:
            return False
        if hasattr(obj, "GetId"):
            current = obj.GetId()
            if current is not None and int(current) != int(object_id):
                return False
        return True
    except Exception:
        return False


def optional_call(obj, name):
    try:
        return getattr(obj, name)()
    except Exception:
        return None


def get_symbol_location(sym):
    try:
        raw = sym.GetSchemaLocation(None, None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 3:
            return None
        return {"sheet_id": int(raw[0]), "x": float(raw[1]), "y": float(raw[2])}
    except Exception:
        return None


def get_symbol_area(sym):
    try:
        raw = sym.GetPlacedArea(None, None, None, None)
        if not isinstance(raw, (tuple, list)) or len(raw) < 5 or int(raw[0]) <= 0:
            return None
        l, b, r, t = map(float, raw[1:5])
        if not (r > l and t > b):
            return None
        return {"left": l, "bottom": b, "right": r, "top": t}
    except Exception:
        return None


def inside_block(loc, block):
    tol = BLOCK_LOCATION_TOLERANCE
    return (loc is not None and loc["sheet_id"] == block["sheet_id"]
            and block["left"] - tol <= loc["x"] <= block["right"] + tol
            and block["bottom"] - tol <= loc["y"] <= block["top"] + tol)


def scan_sheet_symbols(prj, sht, sheet_id):
    if not select_id(sht, sheet_id):
        raise RuntimeError(f"Could not select E3 sheet ID {sheet_id}.")
    ids = get_ids(sht.GetSymbolIds)
    if ids:
        return ids
    log(f"  Sheet {sheet_id} has no sheet-local symbol ID array; scanning project IDs instead.")
    return get_ids(prj.GetSymbolIds)


def wait_for_block_selection(preview=False):
    """Nonmodal directions; E3 must remain usable while this window is open.

    Return in the window works through Tk. On Windows, pressing Enter while
    E3 is foreground also works via a *read-only* key-state poll (no global
    hotkey registration, keyboard interception, or E3 focus manipulation).
    """
    try:
        import tkinter as tk
        from tkinter import ttk
        root = tk.Tk()
    except Exception as exc:
        raise RuntimeError(
            'Could not open the block-selection instruction window. '
            'No E3 changes made. Check that tkinter is installed.'
        ) from exc

    accepted = False
    finished = False

    def proceed(event=None):
        nonlocal accepted, finished
        if finished:
            return
        accepted = True
        finished = True
        root.quit()

    def cancel(event=None):
        nonlocal finished
        if finished:
            return
        finished = True
        root.quit()

    try:
        root.title('E3 - Select blocks (repeat until Close)')
        root.resizable(False, False)
        root.protocol('WM_DELETE_WINDOW', cancel)
        # Keep the instructions ABOVE E3 even when the user clicks on a block.
        # This is only a topmost WINDOW, not a modal dialog: E3 can still take
        # focus and accept mouse selection while this prompt remains visible.
        # The prompt can be dragged aside if it covers a block.
        root.attributes('-topmost', True)
        frame = ttk.Frame(root, padding=20)
        frame.pack(fill='both', expand=True)
        ttk.Label(frame, text='Block Header Formatter (Saber Parameter 1)',
                  font=('Segoe UI', 13, 'bold')).pack(anchor='w', pady=(0, 12))
        ttk.Label(frame, text=(
            '1. In E3, select ONE OR MORE CABBLOCK borders.\n'
            '   This window stays visible; drag it aside if needed.\n'
            '   Clear any previous selection before choosing the next group.\n'
            '2. Press ENTER to format all selected block headers.\n'
            '   (Enter in E3 or return here and press Enter.)\n'
            '3. After the result appears, this window opens again.\n'
            '4. When finished, click Close or the window X.'),
                  justify='left', font=('Segoe UI', 10)).pack(anchor='w')
        ttk.Separator(frame, orient='horizontal').pack(fill='x', pady=14)
        mode = ('PREVIEW ONLY - no E3 changes will be made.' if preview else
                'On Enter: stack Location / Saber Parameter 1 / Equipment ID /\n'
                'Device Designation; bold, black, centered, 0.25 height.')
        ttk.Label(frame, text=mode, justify='left').pack(anchor='w', pady=(0, 12))
        buttons = ttk.Frame(frame)
        buttons.pack(fill='x')
        go_btn = ttk.Button(buttons, text='Format selected blocks (Enter)', command=proceed)
        go_btn.pack(side='left')
        ttk.Button(buttons, text='Close (Esc)', command=cancel).pack(side='right')
        root.bind('<Return>', proceed)
        root.bind('<KP_Enter>', proceed)
        root.bind('<Escape>', cancel)

        root.update_idletasks()
        # Place instructions near upper right so the block remains easier to click.
        width, height = root.winfo_width(), root.winfo_height()
        x = max(0, root.winfo_screenwidth() - width - 50)
        root.geometry(f'+{x}+80')
        go_btn.focus_set()

        # Tk's Return binding only works while its window is foreground.
        # Windows GetAsyncKeyState lets Enter from E3 also complete the prompt
        # without hijacking E3's keyboard or needing a third-party package.
        try:
            import ctypes
            key_state = ctypes.windll.user32.GetAsyncKeyState
            vk_enter = 0x0D
            was_down = bool(key_state(vk_enter) & 0x8000)
            ready = not was_down

            def poll_enter():
                nonlocal was_down, ready
                if finished:
                    return
                try:
                    down = bool(key_state(vk_enter) & 0x8000)
                    if not down:
                        ready = True
                    if ready and down and not was_down:
                        proceed()
                    was_down = down
                except Exception:
                    # Dialog Enter/button remain usable if OS polling fails.
                    pass
                if not finished:
                    root.after(25, poll_enter)

            root.after(25, poll_enter)
        except (AttributeError, OSError):
            log('Global Enter detection not available. Use Enter/Run in the instruction box.')
        root.mainloop()
        return accepted
    finally:
        root.destroy()


def selected_symbol_id_candidates(prj, sht, app):
    """Read current E3 GUI selection, WITHOUT changing GUI selection.

    Different COM versions expose selection at job, sheet or application level;
    each getter may use an out parameter or return an array directly. We use
    only explicit *Selected* getter names and never auto-pick from sheet IDs.
    """
    active_sheet_id = None
    for method_name in ('GetActiveSheetId',):
        try:
            sheet_id = getattr(prj, method_name)()
            if isinstance(sheet_id, (int, float)) and int(sheet_id) > 0:
                active_sheet_id = int(sheet_id)
                break
        except Exception:
            continue
    if active_sheet_id is not None:
        select_id(sht, active_sheet_id)
    log(f'Active E3 sheet: {active_sheet_id if active_sheet_id else "not reported by COM"}')

    # Explicitly symbol-only queries first, generic selected object IDs last.
    # The fallback generic query is ALWAYS type-checked as a placed CABBLOCK.
    probes = (
        ('job', prj, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedSymbolsIds'),
        ('job', prj, 'GetSelectedSymbolId'),
        ('sheet', sht, 'GetSelectedSymbolIds'),
        ('sheet', sht, 'GetSelectedSymbolsIds'),
        ('sheet', sht, 'GetSelectedSymbolId'),
        ('app', app, 'GetSelectedSymbolIds'),
        ('job', prj, 'GetSelectedObjectIds'),
        ('job', prj, 'GetSelectedIds'),
        ('sheet', sht, 'GetSelectedObjectIds'),
        ('sheet', sht, 'GetSelectedIds'),
        ('app', app, 'GetSelectedObjectIds'),
        ('app', app, 'GetSelectedIds'),
    )
    attempted = []
    found = set()
    for label, obj, method_name in probes:
        try:
            getter = getattr(obj, method_name)
        except Exception:
            continue
        # Avoid evaluating an integer method return as the selected ID unless
        # GetSelectedSymbolId explicitly returns one number. For array-style
        # getters, require an array (or a count + array) in the result.
        for args in ((None,), (), (None, 0), (0, None)):
            try:
                result = getter(*args)
            except Exception as exc:
                attempted.append(f'{label}.{method_name}{args}: {str(exc)[:75]}')
                continue
            if isinstance(result, (tuple, list)):
                ids = parse_id_result(result)
            elif method_name == 'GetSelectedSymbolId':
                ids = parse_id_result(result)
            else:
                ids = []
            attempted.append(f'{label}.{method_name}{args}: {ids}')
            if ids:
                found.update(ids)
                break
    log('GUI SELECTION READBACK:')
    for item in attempted[:45]:
        log(f'  {item}')
    if len(attempted) > 45:
        log(f'  ... {len(attempted)-45} additional getter attempts')
    if not found:
        raise RuntimeError(
            'No selected symbol IDs were returned by this E3 COM build. '
            'Select one or more CABBLOCK borders in E3 before pressing Enter. '
            'If it is already selected, send the GUI SELECTION READBACK log; '
            'your E3 version may use another selection API. NO CHANGES MADE.'
        )
    return sorted(found)


def get_selected_cabblocks(prj, sht, app):
    """Return ALL uniquely selected CABBLOCKs; ignore selected non-blocks.

    A drag/box selection can include pins, lines or text as well as the blocks.
    These are never used as targets. At least one placed CABBLOCK is required.
    """
    ids = selected_symbol_id_candidates(prj, sht, app)
    sym = prj.CreateSymbolObject()
    other_symbols = []
    blocks = []
    for sid in ids:
        if not select_id(sym, sid):
            log(f'  Selected ID {sid}: not a valid E3 symbol ID; ignored')
            continue
        kind = str(optional_call(sym, 'GetSymbolTypeName') or '')
        if 'CABBLOCK' not in kind.upper():
            other_symbols.append((sid, kind))
            continue
        location, area = get_symbol_location(sym), get_symbol_area(sym)
        if location is None or area is None:
            raise RuntimeError(
                f'Selected CABBLOCK {sid}: missing placed location/area. '
                'Batch cancelled: NO CHANGES MADE.'
            )
        block = {
            'id': sid, 'sheet_id': location['sheet_id'], 'kind': kind,
            'owner': '<selected block>', 'view_id': None, 'view_number': None,
            **area,
        }
        blocks.append(block)
    if not blocks:
        raise RuntimeError(
            f'No placed CABBLOCK found among {len(ids)} selected E3 ID(s). '
            f'Non-block symbols: {other_symbols[:12]}. '
            'Click one or more CABBLOCK borders and press Enter again. '
            'NO CHANGES MADE.'
        )
    blocks.sort(key=lambda b: (b['sheet_id'], -b['top'], b['left'], b['id']))
    log(f'SELECTION: {len(blocks)} selected CABBLOCK(s); '
        f'{len(other_symbols)} selected non-block symbol(s) ignored.')
    for b in blocks:
        log(f'  BLOCK {b["id"]}: sheet {b["sheet_id"]} | {b["kind"]!r} | '
            f'X=[{b["left"]:.3f},{b["right"]:.3f}] '
            f'Y=[{b["bottom"]:.3f},{b["top"]:.3f}]')
    if other_symbols:
        log(f'  IGNORED non-block selections (first 15): {other_symbols[:15]}')
    return blocks

def text_type_hint(txt):
    hints = []
    for getter in ('GetTypeName', 'GetTextTypeName', 'GetTypeDescription',
                   'GetAttributeName', 'GetName', 'GetSource'):
        val = optional_call(txt, getter)
        if isinstance(val, str) and val.strip():
            hints.append(val.strip())
    return ' | '.join(dict.fromkeys(hints))[:180]


def get_text_xy(text, block):
    """Read E3 SCHEMATIC coordinates; do not assume a symbol-relative value.

    The existing signal mover confirmed E3 text supports GetSchemaLocation.
    Only accept a decoded position reasonably close to the parent CABBLOCK.
    That protects from treating status codes/sheet IDs as coordinates.
    """
    probes = (
        (None, None, None, None, None),
        (None, None, None),
        (None, None),
        (),
    )
    sheet = int(block['sheet_id'])
    options = []
    raw_preview = []
    for args in probes:
        try:
            raw = text.GetSchemaLocation(*args)
        except Exception:
            continue
        raw_preview.append(repr(raw)[:140])
        if not isinstance(raw, (tuple, list)):
            continue
        nums = []
        for number in raw:
            try:
                nums.append(float(number))
            except (TypeError, ValueError):
                nums.append(None)
        candidates = []
        if len(nums) >= 3 and nums[0] is not None and int(nums[0]) == sheet:
            candidates.append((0, nums[1], nums[2]))
        if len(nums) >= 4 and nums[1] is not None and int(nums[1]) == sheet:
            candidates.append((0, nums[2], nums[3]))
        if len(nums) >= 3 and nums[0] in (-1, 0, 1):
            candidates.append((2, nums[1], nums[2]))
        if len(nums) >= 2:
            candidates.append((3, nums[0], nums[1]))
        for priority, x, y in candidates:
            if x is None or y is None or not (math.isfinite(x) and math.isfinite(y)):
                continue
            if (block['left'] - 2 <= x <= block['right'] + 2 and
                    block['bottom'] - 2 <= y <= block['top'] + 2):
                options.append((priority, x, y))
        if options and min(op[0] for op in options) == 0:
            break
    if not options:
        raise RuntimeError(f"Block {block['id']}: cannot read absolute schematic "
                           f"text position. GetSchemaLocation returned: {raw_preview[:3]}. "
                           'No E3 changes made.')
    options.sort(key=lambda item: item[0])
    best_priority = options[0][0]
    distinct = {(round(x, 4), round(y, 4)) for p, x, y in options if p == best_priority}
    if len(distinct) > 1:
        raise RuntimeError(f"Block {block['id']}: multiple possible coordinates "
                           f"from E3 text getter: {sorted(distinct)}. No changes made.")
    return {'x': options[0][1], 'y': options[0][2]}


def inspect_block_texts(prj, blocks):
    """Read only texts owned by the selected CABBLOCK symbols (not their pins)."""
    all_texts = {}
    sym = prj.CreateSymbolObject()
    txt = prj.CreateTextObject()
    for block in blocks:
        if not select_id(sym, block['id']):
            raise RuntimeError(f"Selected block {block['id']} disappeared.")
        ids = get_ids(sym.GetTextIds)
        entries = defaultdict(list)
        for tid in ids:
            if not select_id(txt, tid):
                continue
            try:
                typ = int(txt.GetType())
            except Exception:
                continue
            raw_val = optional_call(txt, 'GetText')
            value = '' if raw_val is None else str(raw_val)
            hint = text_type_hint(txt)
            entries[typ].append({'id': tid, 'type': typ, 'value': value, 'hint': hint})
        all_texts[block['id']] = entries
        log(f"BLOCK {block['id']} (sheet {block['sheet_id']}): "
            f"{len(ids)} attached text IDs; {len(entries)} text types")
        for typ in sorted(entries):
            for e in entries[typ]:
                log(f"  TYPE {typ:<5} ID {e['id']:<10} "
                    f"VALUE={e['value'][:120]!r} HINT={e['hint'][:90]!r}")
    return all_texts


def _is_saber_label(value):
    """True for saberParam1 / Saber Parameter 1, NOT saberParam10 etc."""
    import re
    phrase = str(value or '')
    return bool(re.search(r'\bsaber\s*(?:parameter|param)\s*[_-]?\s*1\b',
                          phrase, re.I))


def _native_text_type_number(value):
    """Convert ONLY native scalars, never a live win32com dynamic dispatch.

    Calling int() on some COM dispatch objects implicitly invokes their default
    COM method. On E3, that may throw a pywintypes.com_error (not ValueError).
    """
    if type(value) is int:
        return value if value > 0 else None
    if type(value) is float:
        return int(value) if math.isfinite(value) and value.is_integer() and value > 0 else None
    if type(value) is str and value.strip().isdecimal():
        return int(value.strip())
    return None


def project_saber_text_types(prj):
    """READ-ONLY, conservative Saber Parameter 1 type discovery.

    e3Job.GetTextTypes takes an OUT dictionary; depending on the COM wrapper,
    its result can contain a Scripting.Dictionary, a plain Python dict, or
    even a COM dispatch that cannot be cast to int/str. No best-guess types.
    """
    try:
        method = getattr(prj, 'GetTextTypes')
    except Exception:
        log('  E3 job.GetTextTypes not available; using attached text hints.')
        return set()

    results = []
    # Zuken documents this as GetTextTypes(textTypeDefinitions), where
    # textTypeDefinitions is a dictionary OUT parameter. Try that form first.
    try:
        import win32com.client as win32
        definitions = win32.Dispatch('Scripting.Dictionary')
        returned = method(definitions)
        results.append(definitions)
        results.append(returned)
        try:
            count = definitions.Count
        except Exception:
            count = '<unavailable>'
        log(f'  job.GetTextTypes(Scripting.Dictionary): returned '
            f'{type(returned).__name__}; dictionary items={count}')
    except Exception as exc:
        log(f'  job.GetTextTypes(Scripting.Dictionary) unavailable: '
            f'{type(exc).__name__}: {str(exc)[:150]}')

    # COM out-argument handling is version-dependent. These fallback calls
    # are READ-ONLY. Preserve their results without coercing COM dispatches.
    for args in ((None,), (), ({},)):
        try:
            returned = method(*args)
            results.append(returned)
            log(f'  job.GetTextTypes{args}: returned {type(returned).__name__}')
            break
        except Exception as exc:
            log(f'  job.GetTextTypes{args}: '
                f'{type(exc).__name__}: {str(exc)[:150]}')

    candidates = set()
    seen = set()
    inspected = 0

    def accept(raw_id, raw_description):
        nonlocal inspected
        inspected += 1
        type_number = _native_text_type_number(raw_id)
        # Crucially DO NOT call str()/int() on a COM object. Attribute names
        # can only be established from actual text/string definition fields.
        if type_number is None or type(raw_description) is not str:
            return
        if _is_saber_label(raw_description):
            candidates.add(type_number)

    def visit(value, depth=0):
        if depth > 7 or value is None:
            return
        if type(value) not in (str, int, float, dict, tuple, list):
            identity = id(value)
            if identity in seen:
                return
            seen.add(identity)
            # Scripting.Dictionary exposes Keys and Items as SAFE read calls.
            # A dynamic COM object without these methods cannot be trusted.
            try:
                keys = value.Keys()
                items = value.Items()
                if isinstance(keys, (tuple, list)) and isinstance(items, (tuple, list)):
                    log(f'  E3 COM dictionary: {len(keys)} text-type definitions')
                    for key, desc in zip(keys, items):
                        accept(key, desc)
                        accept(desc, key)
                        visit(desc, depth+1)
                else:
                    log('  E3 COM text-type object has no usable Keys/Items arrays')
            except Exception as exc:
                log(f'  E3 text-type result is a COM object, not an integer '
                    f'({type(exc).__name__}); skipping unparseable object safely')
            return
        if type(value) is dict:
            # A dictionary is only trustworthy if *the same entry* associates
            # an actual numeric type ID with a label mentioning saberParam1.
            for key, desc in value.items():
                accept(key, desc)
                accept(desc, key)
            # Structured text-type definition with a separate ID/label pair.
            numeric_fields = ('type', 'Type', 'type_id', 'typeId', 'id', 'number')
            descriptive_fields = ('name', 'Name', 'label', 'Label', 'description',
                                  'Description', 'attribute', 'attributeName')
            for nkey in numeric_fields:
                if nkey in value:
                    for dkey in descriptive_fields:
                        if dkey in value:
                            accept(value[nkey], value[dkey])
            for desc in value.values():
                visit(desc, depth+1)
        elif type(value) in (tuple, list):
            if len(value) == 2:
                accept(value[0], value[1]); accept(value[1], value[0])
            for entry in value:
                visit(entry, depth+1)
        elif type(value) is str and value.lstrip().startswith(('{', '[')):
            try:
                import ast
                parsed = ast.literal_eval(value)
                if parsed is not value:
                    visit(parsed, depth+1)
            except (ValueError, SyntaxError, MemoryError, RecursionError):
                pass

    for result in results:
        visit(result)
    if candidates:
        log(f'  Confirmed Saber Parameter 1 text-type candidate(s): '
            f'{sorted(candidates)}')
    else:
        log(f'  No proven Saber Parameter 1 text type from E3 definitions '
            f'({inspected} scalar pair(s) considered). No type was guessed.')
    return candidates

def determine_mapping(prj, all_texts, override_saber=None,
                      override_equipment=None, override_designation=None):
    """Resolve ONLY a confirmed ``saberParam1``-linked node.

    System Device Function / Higher Level Assignment type 14 must NEVER be used.
    """
    prohibited = {14, 15, 1000, 212, 1010}
    if override_saber is not None:
        saber_type = int(override_saber)
        log(f'  Saber Parameter 1 type explicitly specified: {saber_type}')
    else:
        candidates = set()
        for attached in all_texts.values():
            for typ, rows in attached.items():
                if any(_is_saber_label(e.get('hint', '')) for e in rows):
                    candidates.add(int(typ))
        candidates |= project_saber_text_types(prj)
        # The whole point of this revision is that some selected CABBLOCKs
        # DO NOT yet have the Saber text node attached.  Do not filter
        # candidate types by presence on every selected block.
        if len(candidates) == 1:
            saber_type = next(iter(candidates))
        else:
            available = {
                block_id: [(typ, [(e['id'], e['value'][:35]) for e in rows])
                           for typ, rows in sorted(group.items())]
                for block_id, group in all_texts.items()
            }
            raise RuntimeError(
                'Could not uniquely identify the attribute-linked Saber Parameter 1 '
                f'text type in the selected CABBLOCKs. Candidates={sorted(candidates)}.\n'
                f'Available attached types: {available}\n'
                'NEXT STEP: In E3 Database Editor > Format > Text Types, find '
                'the type associated with the attribute saberParam1. If it is '
                'NOT among the attached types, this version can attempt to '
                'create the node if your COM interface supports Symbol.AddText. '
                'Run with --saber-type NUMBER after verifying the type is '
                'assigned to saberParam1 in your E3 text-type configuration. '
                'NO E3 changes were made.'
            )
    if saber_type in prohibited or saber_type <= 0:
        raise RuntimeError(
            f'Saber type {saber_type} is a protected header/combined type or '
            'invalid (14 is Device Function, 1010 is combined text). '
            'Refusing to edit it; NO CHANGES MADE.'
        )
    mapping = {
        'location': LOCATION_TYPE,
        'saber': saber_type,
        'equipment': int(override_equipment) if override_equipment is not None else EQUIPMENT_TYPE,
        'designation': int(override_designation) if override_designation is not None else DEVICE_DESIGNATION_TYPE,
    }
    if len(set(mapping.values())) != 4:
        raise RuntimeError(f'Header text type mappings overlap: {mapping}. NO CHANGES MADE.')
    log('E3 HEADER TEXT TYPE MAPPING: ' + ', '.join(
        f'{FIELD_LABELS[k]}={mapping[k]}' for k in HEADER_ORDER))
    log('PROTECTION: Device Function VALUE (type 14) and combined text 1010 are never changed; old type-14 DISPLAY is hidden.')
    for block_id, entries in all_texts.items():
        missing = [(FIELD_LABELS[field], typ) for field, typ in mapping.items()
                   if field != 'saber' and typ not in entries]
        if missing:
            raise RuntimeError(
                f'CABBLOCK {block_id}: required existing header text missing: {missing}. '
                'NO CHANGES MADE. Only Saber Parameter 1 can be automatically added.'
            )
    missing_saber = [bid for bid, groups in all_texts.items()
                     if saber_type not in groups]
    if missing_saber:
        log(f'  Saber Parameter 1 text node MISSING on block(s): {missing_saber}. '
            'These require a new attribute-linked text node, not just formatting.')
    return mapping


def available_method(obj, name):
    try:
        return callable(getattr(obj, name))
    except Exception:
        return False


def get_symbol_saber_value(sym, allow_missing=False):
    try:
        raw = sym.GetAttributeValue(SABER_ATTRIBUTE_NAME)
    except Exception as exc:
        if allow_missing:
            log(f'  Missing/inaccessible Saber attribute: {type(exc).__name__}: {str(exc)[:120]}')
            return None
        raise RuntimeError(f'Cannot read existing symbol saberParam1: {exc}') from exc
    if raw is None:
        return ''
    if isinstance(raw, (tuple, list)):
        if len(raw) != 1 or not isinstance(raw[0], str):
            raise RuntimeError(f'Ambiguous symbol GetAttributeValue response: {raw!r}')
        raw = raw[0]
    return str(raw).strip()


def preflight_saber_attributes(prj, plan):
    """Before ANY writes, check all selected blocks and needed COM methods."""
    sym = prj.CreateSymbolObject()
    for item in plan:
        if item['field'] != 'saber':
            continue
        if not select_id(sym, item['block']['id']):
            raise RuntimeError(f"CABBLOCK {item['block']['id']} disappeared.")
        before_attr = get_symbol_saber_value(sym, allow_missing=True)
        item['before_attribute_value'] = before_attr
        item['set_xxx'] = not ((before_attr or '').strip() or item['before_value'].strip())
        item['new_attr_needed'] = (before_attr is None)
        if item['new_attr_needed'] and not available_method(sym, 'AddAttributeValue'):
            raise RuntimeError(
                f"Block {item['block']['id']} lacks saberParam1 and E3 symbol "
                'has no AddAttributeValue method. The known Device.AddAttributeValue '
                'is NOT used because it can change other block views. '
                'NO E3 changes made. Send this log so the correct instance-level '
                'attribute method can be implemented for your E3 COM build.'
            )
        if item['set_xxx'] and not item['new_attr_needed'] and not available_method(sym, 'SetAttributeValue'):
            raise RuntimeError(
                f"CABBLOCK {item['block']['id']}: SetAttributeValue unavailable. "
                'NO CHANGES MADE.'
            )
        if item['id'] is None and not available_method(sym, 'AddText'):
            names = [name for name in ('AddText','AddTextNode','AddTextType',
                                       'CreateText','InsertText','AddAttributeText')
                     if available_method(sym, name)]
            log(f'  Available possible text-insertion methods: {names}')
            raise RuntimeError(
                f"CABBLOCK {item['block']['id']}: Saber Parameter 1 needs an "
                'attached text node, but Symbol.AddText is not supported here. '
                'NO E3 CHANGES. We cannot safely replace it with unrelated '
                'graphic text or a shared device property. '
                'Use E3 Database Editor to add an attribute-linked text type to '
                'the CABBLOCK template, or send the above COM method diagnostics.'
            )
        action = ('ADD ATTRIBUTE' if item['new_attr_needed'] else
                  'SET XXX' if item['set_xxx'] else 'PRESERVE')
        log(f"  BLOCK {item['block']['id']}: saberParam1={before_attr!r}, "
            f"text ID={item['id']!r} -> {action} "
            f"{'and INSERT LINKED TEXT' if item['id'] is None else ''}")
    return plan


def build_plan(blocks, all_texts, mapping):
    """Allow Saber type to be absent; other three fields must already exist."""
    plan = []
    target_ids = set()
    for b in blocks:
        available = all_texts[b['id']]
        requested = {}
        for field in HEADER_ORDER:
            typ = mapping[field]
            matches = available.get(typ, [])
            if field == 'saber' and not matches:
                requested[field] = None
            elif len(matches) != 1:
                raise RuntimeError(
                    f"CABBLOCK {b['id']}: {FIELD_LABELS[field]} (type {typ}) "
                    f'has {len(matches)} text node(s), expected one. NO CHANGES.'
                )
            else:
                requested[field] = matches[0]
        height = float(b['top']) - float(b['bottom'])
        needed = TOP_INSET + 3 * TEXT_PITCH + BOTTOM_CLEARANCE
        if height < needed:
            raise RuntimeError(f"CABBLOCK {b['id']} too short ({height:.3f} < {needed:.3f}).")
        x = (float(b['left']) + float(b['right'])) / 2
        for index, field in enumerate(HEADER_ORDER):
            e = requested[field]
            if e is not None:
                if e['id'] in target_ids:
                    raise RuntimeError(f"Text {e['id']} owned by multiple blocks; NO CHANGES.")
                target_ids.add(e['id'])
            plan.append({'block': b, 'field': field, 'type': mapping[field],
                         'id': e['id'] if e else None,
                         'before_value': e['value'] if e else '',
                         'x': x, 'y': float(b['top']) - TOP_INSET - index * TEXT_PITCH})
    return plan


def preflight_positions(prj, plan):
    txt = prj.CreateTextObject()
    for item in plan:
        if item['id'] is None:
            item['before'] = None
            item['move_needed'] = True
            log(f"  BLOCK {item['block']['id']} | Saber Parameter 1 "
                f"type={item['type']} INSERT at ({item['x']:.3f}, {item['y']:.3f})")
            continue
        if not select_id(txt, item['id']):
            raise RuntimeError(f"Text {item['id']} disappeared in preflight.")
        item['before'] = get_text_xy(txt, item['block'])
        item['move_needed'] = (
            abs(item['before']['x'] - item['x']) > POSITION_TOLERANCE or
            abs(item['before']['y'] - item['y']) > POSITION_TOLERANCE)
        log(f"  BLOCK {item['block']['id']} | {FIELD_LABELS[item['field']]} "
            f"text={item['id']} MOVE={item['move_needed']}")
    return plan


def insert_linked_saber_text(prj, item):
    """Create only through selected CABBLOCK, verify ownership by GetTextIds.

    Important: AddText is version-specific. Failed attempts are checked for
    side effects before trying the next signature; never insert free text.
    """
    sym = prj.CreateSymbolObject()
    b = item['block']
    if not select_id(sym, b['id']):
        raise RuntimeError(f"CABBLOCK {b['id']} unavailable during text insertion.")
    old_ids = set(get_ids(sym.GetTextIds))
    args_list = [
        (int(item['type']), float(item['x']), float(item['y'])),
        (int(item['type']), float(item['x']), float(item['y']), ''),
    ]
    errors = []
    for args in args_list:
        try:
            result = sym.AddText(*args)
            log(f'  Symbol.AddText{args} returned {result!r}')
        except Exception as exc:
            errors.append(f'{len(args)} arguments: {type(exc).__name__}: {str(exc)[:150]}')
        new_ids = set(get_ids(sym.GetTextIds)) - old_ids
        if len(new_ids) == 1:
            tid = next(iter(new_ids))
            txt = prj.CreateTextObject()
            if not select_id(txt, tid) or int(txt.GetType()) != int(item['type']):
                raise RuntimeError(f'New text {tid} attached to block {b["id"]}, '
                                   'but its E3 text type is unexpected. INSPECT THE DRAWING.')
            # A mere attachment is not proof that this type is linked to saberParam1.
            value = str(optional_call(txt, 'GetText') or '').strip()
            if value != 'XXX':
                raise RuntimeError(f'New text {tid} attached to block {b["id"]}, '
                                   f'but displays {value!r} instead of XXX. '
                                   'Verify this text type is ACTUALLY assigned to saberParam1. '
                                   'STOPPED; inspect drawing before rerunning.')
            log(f'  CREATED bound Saber text ID {tid} on CABBLOCK {b["id"]}')
            return tid
        if new_ids:
            raise RuntimeError(f'AddText created {len(new_ids)} node(s) on block '
                               f'{b["id"]}; expected one. INSPECT THE DRAWING.')
    raise RuntimeError(f'Could not create bound Saber text on CABBLOCK {b["id"]}. '
                       f'AddText signatures failed: {errors}. No free text substituted.')

def old_function_nodes_to_hide(blocks, all_texts, plan):
    """Hide ONLY existing Device Function display nodes on selected blocks.

    The old formatter left these visible at the second header row. We do not
    modify their text content, property value, type, position or formatting.
    """
    target_ids = {item['id'] for item in plan}
    old_nodes = []
    seen = set()
    for block in blocks:
        for record in all_texts[block['id']].get(14, []):
            tid = record['id']
            if tid in target_ids or tid in seen:
                raise RuntimeError(
                    f'Text {tid} appears in multiple selected blocks or overlaps '
                    'a target header text; batch refused before changes.'
                )
            seen.add(tid)
            old_nodes.append({'id': tid, 'block': block, 'value': record['value']})
    log(f'  OLD DEVICE FUNCTION LABELS: {len(old_nodes)} will be hidden '
        '(their values and E3 attributes are not changed).')
    return old_nodes


def _same_xy(txt, block, x, y):
    try:
        actual = get_text_xy(txt, block)
        return (abs(actual['x'] - x) <= POSITION_TOLERANCE and
                abs(actual['y'] - y) <= POSITION_TOLERANCE)
    except Exception:
        return False


def move_text(txt, item):
    """Only try signatures already used by the working signal relocator."""
    method = getattr(txt, 'SetSchemaLocation', None)
    if method is None:
        raise RuntimeError('E3 Text.SetSchemaLocation is unavailable.')
    old = item['before']
    b = item['block']
    attempts = (
        ((item['x'], item['y']), (old['x'], old['y']), 'x,y'),
        ((b['sheet_id'], item['x'], item['y']),
         (b['sheet_id'], old['x'], old['y']), 'sheet,x,y'),
    )
    errors = []
    for new, rollback, signature in attempts:
        try:
            result = method(*new)
        except Exception as exc:
            errors.append(f'{signature}: {exc}')
            continue
        if _same_xy(txt, b, item['x'], item['y']):
            return
        try:
            method(*rollback)
        except Exception as exc:
            raise RuntimeError(f"Text {item['id']} moved but readback failed "
                               f"and rollback also failed: {exc}; CHECK E3") from exc
        raise RuntimeError(f"Text {item['id']}: SetSchemaLocation({signature}) "
                           'returned but new position was not verified. '
                           f'Rolled back (reported result={result!r}); batch stopped.')
    raise RuntimeError(f"Text {item['id']}: unable to set position: {errors}")


def checked_style_write(txt, method, value, field):
    try:
        result = getattr(txt, method)(value)
    except Exception as exc:
        raise RuntimeError(f"Text {field}: {method}({value}) failed: {exc}") from exc
    # E3 methods sometimes return None on success; negative status means error.
    if isinstance(result, (int, float)) and result < 0:
        raise RuntimeError(f"Text {field}: {method}({value}) returned {result}.")


def apply_plan(prj, plan, old_function_labels=None):
    """Apply planned changes; creation and formatting remain verifiable."""
    txt = prj.CreateTextObject()
    sym = prj.CreateSymbolObject()
    completed = 0
    updated_saber = 0
    try:
        # Create missing Saber nodes BEFORE formatting any of the other labels.
        # If E3 does not support insertion, minimize partially formatted headers.
        ordered_plan = sorted(plan, key=lambda p: 0 if p['field'] == 'saber' and p['id'] is None else 1)
        for item in ordered_plan:
            field = FIELD_LABELS[item['field']]
            if item['field'] == 'saber':
                if not select_id(sym, item['block']['id']):
                    raise RuntimeError(f"CABBLOCK {item['block']['id']} disappeared.")
                if item['set_xxx']:
                    method = ('AddAttributeValue' if item['new_attr_needed']
                              else 'SetAttributeValue')
                    result = getattr(sym, method)(SABER_ATTRIBUTE_NAME, 'XXX')
                    if type(result) is int and result < 0:
                        raise RuntimeError(f'Block {item["block"]["id"]}: {method} '
                                           f'failed with {result}')
                    actual = get_symbol_saber_value(sym)
                    if actual != 'XXX':
                        raise RuntimeError('Symbol saberParam1 did not read back as XXX. '
                                           'STOPPED; inspect E3 for partial changes.')
                    updated_saber += 1
                if item['id'] is None:
                    item['id'] = insert_linked_saber_text(prj, item)
                    item['before'] = get_text_xy(_select_text(prj, item['id']), item['block'])
                    item['move_needed'] = (abs(item['before']['x'] - item['x']) > POSITION_TOLERANCE or
                                            abs(item['before']['y'] - item['y']) > POSITION_TOLERANCE)
            if not select_id(txt, item['id']):
                raise RuntimeError(f"Text {item['id']} unavailable for formatting.")
            if item['field'] == 'saber':
                displayed = str(optional_call(txt, 'GetText') or '').strip()
                if displayed != 'XXX' and item['set_xxx']:
                    raise RuntimeError(f'Linked Saber text reads {displayed!r}, expected XXX. '
                                       'STOPPED before text formatting.')
            for method, value in (
                    ('SetVisibility', 1), ('SetStyle', BOLD_STYLE),
                    ('SetColour', BLACK_COLOUR), ('SetHeight', TEXT_HEIGHT),
                    ('SetAlignment', CENTER_ALIGNMENT)):
                checked_style_write(txt, method, value, field)
            if item['move_needed']:
                move_text(txt, item)
            completed += 1
            log(f"  DONE block {item['block']['id']} | {field}, text {item['id']}")
        for old in (old_function_labels or []):
            if not select_id(txt, old['id']):
                raise RuntimeError(f"Function label {old['id']} disappeared.")
            checked_style_write(txt, 'SetVisibility', 0, 'Old Function display')
        return completed, updated_saber
    except Exception:
        log(f'PARTIAL FAILURE: {completed}/{len(plan)} completed; '
            'changes already written to E3 may remain. INSPECT THE DRAWING.')
        raise


def _select_text(prj, tid):
    obj = prj.CreateTextObject()
    if not select_id(obj, tid):
        raise RuntimeError(f'Cannot select newly created text {tid}')
    return obj

def run(preview=False, saber_type=None, equipment_type=None,
        designation_type=None):
    global LOG_STREAM
    path = choose_log_path()
    try:
        LOG_STREAM = open(path, 'a', encoding='utf-8')
        log('\n' + '=' * 80)
        log('NEW SABER HEADER BATCH: ' + datetime.now().astimezone().strftime('%Y-%m-%d %H:%M:%S'))
        log('FORMAT: LOCATION > SABER PARAMETER 1 > EQUIPMENT ID > DEVICE DESIGNATION')
        log('KNOWN TYPES: LOCATION=15; EQUIPMENT=1000; DESIGNATION=212')
        log('Saber Parameter 1: attribute saberParam1, linked text-type dynamically resolved')
        log('NEVER MODIFY DEVICE FUNCTION (14) OR COMBINED TEXT (1010)')
        log('BOLD, BLACK (0), CENTER (2), HEIGHT 0.25, VISIBLE')
        log('Using current E3 GUI selection. No spreadsheet required.')
        log('=' * 80)
        import win32com.client as win32
        app = win32.gencache.EnsureDispatch('CT.Application')
        prj = app.CreateJobObject()
        sht = prj.CreateSheetObject()
        blocks = get_selected_cabblocks(prj, sht, app)
        all_texts = inspect_block_texts(prj, blocks)
        mapping = determine_mapping(prj, all_texts, saber_type, equipment_type,
                                    designation_type)
        plan = build_plan(blocks, all_texts, mapping)
        preflight_positions(prj, plan)
        preflight_saber_attributes(prj, plan)
        old_function_labels = old_function_nodes_to_hide(blocks, all_texts, plan)
        changes = sum(p['move_needed'] for p in plan)
        blanks = sum(p['set_xxx'] for p in plan)
        log(f'BATCH READY: {len(blocks)} block(s); {len(plan)} linked text(s); '
            f'{changes} reposition(s)/insert(s); {blanks} blank per-block Saber parameters -> XXX; '
            f'{len(old_function_labels)} old Function labels will be hidden.')
        if preview:
            log('DRY RUN: NO CHANGES MADE.')
            popup(f'{len(blocks)} selected block(s) validated.\n'
                  f'{len(plan)} text nodes; {changes} moves, {blanks} blank '
                  'symbol Saber Parameter 1 values would become XXX.\n\n'
                  f'{len(old_function_labels)} old Function labels would be hidden.\n'
                  f'No changes made.\nLog: {path}')
            return 0
        completed, filled = apply_plan(prj, plan, old_function_labels)
        log(f'SUCCESS: {len(blocks)} CABBLOCK(s), {completed} text nodes; '
            f'{filled} per-symbol Saber Parameter 1 values set to XXX; '
            f'{len(old_function_labels)} old Function displays hidden.')
        popup(f'Formatted {len(blocks)} selected block(s).\n'
              'Location → Saber Parameter 1 → Equipment ID → Device Designation\n'
              f'{completed} labels formatted; {filled} per-symbol Saber Parameter 1 '
              f'values filled with XXX; {len(old_function_labels)} old Function labels hidden.\n\n'
              f'Log: {path}\n\nSave the E3 project.')
        return 0
    except Exception:
        log('ERROR: ' + traceback.format_exc())
        raise
    finally:
        if LOG_STREAM is not None:
            LOG_STREAM.close()
        LOG_STREAM = None


def main(argv=None):
    parser = argparse.ArgumentParser(
        description='Select any number of E3 CABBLOCKs and press Enter to '
                    'format a 4-line header on each block.')
    parser.add_argument('--dry-run', action='store_true', help='Preview only.')
    parser.add_argument('--saber-type', type=int, default=None,
                        help='Project-specific type ID linked to saberParam1 (after verifying in E3).')
    parser.add_argument('--equipment-type', type=int, default=None,
                        help='Override observed Equipment Identification type 1000.')
    parser.add_argument('--designation-type', type=int, default=None,
                        help='Override observed CABBLOCK designation type 212.')
    args = parser.parse_args(argv)
    while True:
        try:
            accepted = wait_for_block_selection(preview=args.dry_run)
        except Exception as exc:
            print('Could not open selection instructions:', exc)
            return 1
        if not accepted:
            print('CLOSED: no additional E3 changes.')
            return 0
        try:
            run(preview=args.dry_run,
                saber_type=args.saber_type,
                equipment_type=args.equipment_type,
                designation_type=args.designation_type)
        except Exception as exc:
            print('FAILED:', exc)
            popup(str(exc) + '\n\nThe instruction window will reopen.\n'
                  'Log: E3_Saber_Header_Formatter_Log.txt', error=True)


if __name__ == '__main__':
    sys.exit(main())
