# -*- coding: utf-8 -*-
"""E3 top-level PLUG MIL-placement A/B test. Standalone. No main.py imports.

USE ONLY WITH A COPY OF YOUR E3 PROJECT. This script creates two small test
blocks and two connector devices per run; it does not delete existing objects.

1) In E3: Tools > Settings > MIL-Standard > Place Block Connectors:
   choose "on the block's outline". Then, from a terminal, run:
       py e3_top_plug_mode_test.py outline
2) AFTER the Python process exits, set that option to "inside block".
   Run again, choosing a DIFFERENT empty area:
       py e3_top_plug_mode_test.py inside

Both runs request MilBlockPlugInt¶FULL. The script compares the ACTUAL E3
symbol families and GetDisplayMILStandard() settings. It tests RIGHT and LEFT
block edges because main.py uses both orientations.

No edits to main.py, utils.py, test.py, or launcher.py.
"""

import json
import sys
from datetime import datetime
from pathlib import Path
from tkinter import Tk, filedialog

SCRIPT_DIR = Path(__file__).resolve().parent
REPORT = SCRIPT_DIR / "e3_top_plug_mode_results.json"
REQUESTED_SYMBOL = "MilBlockPlugInt¶FULL"
SHEET_NAME = "1"
BLOCK_SYMBOL = "CABBLOCK"
BLOCK_VERSION = 1
BLOCK_W, BLOCK_H = 3.0, 5.0
GAP_X = 8.0


def positive_ids(value):
    if isinstance(value, (list, tuple)):
        if len(value) >= 2 and isinstance(value[1], (list, tuple)):
            value = value[1]
        return [v for v in value if type(v) is int and v > 0]
    return []


def positive_id(value):
    if isinstance(value, (list, tuple)):
        value = value[0] if value else None
    return value if type(value) is int and value > 0 else None


def safe(obj, method, *args):
    try:
        return getattr(obj, method)(*args)
    except Exception as exc:
        return "ERROR: " + repr(exc)


def get_input():
    """All file dialogs and prompts happen BEFORE connecting to E3."""
    from openpyxl import load_workbook
    root = Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    try:
        path = filedialog.askopenfilename(
            parent=root, title="Choose Block Maker workbook (PLUG in column F)",
            filetypes=[("Excel workbooks", "*.xlsm *.xlsx")]
        )
    finally:
        root.destroy()
    if not path:
        raise SystemExit("Workbook selection canceled")
    wb = load_workbook(path, read_only=True, data_only=True)
    try:
        ws = wb.active
        n = int(ws["C5"].value)
        choices = []
        for index in range(n):
            row = index + 7
            name = str(ws.cell(row, 2).value or "").strip()
            part = str(ws.cell(row, 3).value or "").strip()
            gender = str(ws.cell(row, 6).value or "").strip().upper()
            if name and part and gender == "PLUG":
                choices.append((name, part))
        if not choices:
            raise RuntimeError("No PLUG connectors found in Excel column F (rows 7+).")
        print("\nPLUG connectors from Excel:")
        for i, (name, component) in enumerate(choices, start=1):
            print(f"  {i}: {name} | {component}")
        selection = input("Number [Enter=1]: ").strip()
        ix = (int(selection) if selection else 1) - 1
        if ix < 0 or ix >= len(choices):
            raise ValueError("Invalid connector number")
        name, part = choices[ix]
    finally:
        wb.close()

    answer = input("Empty test area: LOWER-LEFT X,Y [Enter=3,4]: ").strip()
    if answer:
        x, y = [float(k.strip()) for k in answer.split(",")]
    else:
        x, y = 3.0, 4.0
    if x < 0 or y < 0:
        raise ValueError("Use nonnegative sheet coordinates")
    print(f"Test blocks: RIGHT-edge case at ({x}, {y}); "
          f"LEFT-edge case at ({x + GAP_X}, {y})")
    print("USE A PROJECT COPY. Two test blocks and connectors will be created.")
    return (name, part, x, y)


def get_sheet_id(job):
    sh = job.CreateSheetObject()
    for shid in positive_ids(job.GetAllSheetIds(None)):
        sh.SetId(shid)
        if str(sh.GetName()).strip() == SHEET_NAME:
            return shid
    raise RuntimeError(f"Cannot find E3 sheet called {SHEET_NAME!r}")


def create_block(job, sheet_id, x, y):
    sym = job.CreateSymbolObject()
    if not sym.Load(BLOCK_SYMBOL, BLOCK_VERSION):
        raise RuntimeError(f"Could not load {BLOCK_SYMBOL} v{BLOCK_VERSION}")
    answer = sym.PlaceBlock(sheet_id, x, y, BLOCK_W, BLOCK_H)
    sid = positive_id(answer)
    if not sid:
        raise RuntimeError(f"PlaceBlock returned {answer!r}")
    return sid


def new_connector(job, app, part, unique_name):
    check = app.GetDatabaseTableSelectedComponents(part, 1)
    if isinstance(check, (list, tuple)) and check and check[0] != 0:
        raise RuntimeError(f"Database component not found: {part!r}: {check!r}")
    device = job.CreateDeviceObject()
    raw = device.Create(unique_name, "", "", part, 1, 0)
    if not positive_id(raw):
        raise RuntimeError(f"Device.Create({unique_name!r}) returned {raw!r}")
    dev_id = None
    for did in positive_ids(job.GetAllDeviceIds(None)):
        device.SetId(did)
        if str(device.GetName()) == unique_name:
            dev_id = did
            break
    if not dev_id:
        raise RuntimeError(f"Could not resolve new connector {unique_name!r}")
    device.SetId(dev_id)
    pins = positive_ids(device.GetPinIds(None))
    if not pins:
        raise RuntimeError(f"No connector pins for {unique_name!r}")
    return dev_id, pins


def place_and_inspect(job, pins, sheet_id, x, y, flip, unique_name):
    symbol = job.CreateSymbolObject()
    raw = symbol.PlacePins(list(pins), REQUESTED_SYMBOL, 1,
                           sheet_id, x, y, flip, 1.0)
    sid = positive_id(raw)
    if not sid:
        raise RuntimeError(f"PlacePins({unique_name}): returned {raw!r}")
    symbol.SetId(sid)
    actual = safe(symbol, "GetSymbolTypeName")
    record = {
        "connector": unique_name, "symbol_id": sid,
        "x": x, "y": y, "flip": flip,
        "requested": REQUESTED_SYMBOL,
        "actual": actual,
        "characteristic": safe(symbol, "GetCharacteristic"),
        "type_id": safe(symbol, "GetSymbolTypeId"),
        "pin_count": len(pins),
    }
    print("  REQUESTED:", REQUESTED_SYMBOL)
    print("  ACTUAL:   ", actual)
    print("  MATCH:    ", "YES" if actual == REQUESTED_SYMBOL else "NO (E3 substituted)")
    print("  SYMBOL ID:", sid, "CHARACTERISTIC:", record["characteristic"])
    return record


def run(label):
    name, component, x, y = get_input()
    import win32com.client as win32
    app = win32.GetActiveObject("CT.Application")
    job = app.CreateJobObject()
    run_data = {
        "label": label,
        "timestamp": datetime.now().isoformat(),
        "source_connector": name,
        "component": component,
        "mil_standard": safe(job, "GetDisplayMILStandard"),
        "connector_symbol_format": str(safe(job, "GetConnectorSymbolFormat")),
        "results": {},
        "errors": [],
    }
    print("\nGetDisplayMILStandard():", run_data["mil_standard"])
    print("Project MIL mode is a setting; this script does NOT modify it.")
    try:
        sheet_id = get_sheet_id(job)
        stamp = datetime.now().strftime("%H%M%S%f")
        for side, bx, by, flip, edge_x in (
            ("right", x, y, 0, x + BLOCK_W),
            ("left", x + GAP_X, y, "/1", x + GAP_X),
        ):
            print(f"\n{side.upper()} EDGE:")
            unique_name = f"MILTEST_{label}_{side}_{stamp}"
            create_block(job, sheet_id, bx, by)
            _, pin_ids = new_connector(job, app, component, unique_name)
            try:
                result = place_and_inspect(
                    job, pin_ids, sheet_id, edge_x, by + 3.8, flip, unique_name
                )
                run_data["results"][side] = result
            except Exception as ex:
                run_data["errors"].append(f"{side} edge: {ex!r}")
                print("  FAILED:", repr(ex))
    except Exception as ex:
        run_data["errors"].append(repr(ex))
        print("TEST ERROR:", repr(ex))
    finally:
        del job, app
        if REPORT.is_file():
            try:
                all_runs = json.loads(REPORT.read_text(encoding="utf-8"))
            except Exception:
                all_runs = {}
        else:
            all_runs = {}
        all_runs[label] = run_data
        REPORT.write_text(json.dumps(all_runs, indent=2, ensure_ascii=False, default=str),
                          encoding="utf-8")

    print("\nSaved:", REPORT)
    if "outline" in all_runs and "inside" in all_runs:
        a, b = all_runs["outline"], all_runs["inside"]
        print("\n===== MIL SETTING COMPARISON =====")
        print("OUTLINE GetDisplayMILStandard:", a["mil_standard"])
        print("INSIDE  GetDisplayMILStandard:", b["mil_standard"])
        if a["mil_standard"] == b["mil_standard"]:
            print("WARNING: settings are identical; this is not yet a proven MIL-mode A/B test.")
        elif type(a["mil_standard"]) is int and type(b["mil_standard"]) is int:
            print("Changed bits (XOR):", hex(a["mil_standard"] ^ b["mil_standard"]))
        for side in ("right", "left"):
            old = a.get("results", {}).get(side, {}).get("actual", "NOT PLACED")
            new = b.get("results", {}).get(side, {}).get("actual", "NOT PLACED")
            print(f"{side.upper():5}: OUTLINE {old}  ->  INSIDE {new}")
    print("\nPython process ending: inspect/move objects in E3 ONLY AFTER it exits.")
    if run_data["errors"]:
        raise SystemExit(1)


if __name__ == "__main__":
    if len(sys.argv) != 2 or sys.argv[1].lower() not in ("outline", "inside"):
        raise SystemExit("Usage: py e3_top_plug_mode_test.py outline|inside")
    run(sys.argv[1].lower())
