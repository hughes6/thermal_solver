# -*- coding: utf-8 -*-
"""AUTOMATED E3 TOP-LEVEL PLUG MIL SYMBOL TEST - ONE FILE.

Run:   py e3_top_plug_autonomous_test.py

- No Excel file, user prompts, second pass, or manual MIL-setting changes.
- Uses known PLUG database component: M24308/4-1F_9P SD, version 1.
- Auto-tests the requested MilBlockPlugInt¶FULL at BOTH block edges.
- Changes one project MIL-setting bit at a time, verifies readback, then places.
- ALWAYS attempts to restore the original MIL setting in a finally block.
- Writes a JSON report next to this script.
- Creates CABBLOCK blocks and temporary connector devices: RUN ONLY ON A COPY
  of your E3 project. It does not delete existing objects.

IMPORTANT: E3's public documentation confirms Get/SetDisplayMILStandard but
DOES NOT publish the bit positions for 'outline', 'inside', and 'outside'.
The candidate values below are therefore EXPERIMENTAL bit flips; the script
reports actual numbers/results, NOT guessed names for modes.
"""

import json
import sys
import traceback
from datetime import datetime
from pathlib import Path

# Existing verified plug component. No workbook selection required.
COMPONENT_CODE = "M24308/4-1F_9P SD"
COMPONENT_VERSION = 1
REQUESTED_SYMBOL = "MilBlockPlugInt¶FULL"
BLOCK_SYMBOL = "CABBLOCK"
SHEET_NAME = "1"

# Fresh copy of your E3 project recommended. Layout fits 42.9 x 32.9-inch sheet.
START_X, START_Y = 2.0, 2.0
COL_GAP, ROW_GAP = 10.0, 6.8
BLOCK_WIDTH, BLOCK_HEIGHT = 3.0, 5.0
LEFT_BLOCK_OFFSET = 4.5
PINS_Y_OFFSET = 3.8

# Toggle a single bit at a time from the ACTUAL current setting. We don't
# pretend to know which bits encode inside/outline/outside. Stop early if both
# right and left edges place the exact requested symbol.
BITS_TO_TEST = (1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384)
STOP_ON_BOTH_EXACT = True

REPORT = Path(__file__).with_name("e3_top_plug_autonomous_results.json")


def positive_id(raw):
    if isinstance(raw, (list, tuple)):
        raw = raw[0] if raw else None
    return int(raw) if type(raw) is int and raw > 0 else 0


def positive_ids(raw):
    if isinstance(raw, (list, tuple)):
        if len(raw) >= 2 and isinstance(raw[1], (tuple, list)):
            raw = raw[1]
        return [v for v in raw if type(v) is int and v > 0]
    return []


def try_call(obj, method, *args):
    try:
        return getattr(obj, method)(*args)
    except Exception as exc:
        return "ERROR: " + repr(exc)


def save_json(report):
    REPORT.write_text(json.dumps(report, indent=2, ensure_ascii=False, default=str), encoding="utf-8")


def get_sheet_id(job):
    sheet = job.CreateSheetObject()
    ids = positive_ids(job.GetAllSheetIds(None))
    for sheet_id in ids:
        sheet.SetId(sheet_id)
        if str(sheet.GetName()).strip() == SHEET_NAME:
            return sheet_id
    raise RuntimeError(f"No drawing sheet named {SHEET_NAME!r}. Available sheet IDs: {ids!r}")


def place_block(job, sheet_id, x, y):
    sym = job.CreateSymbolObject()
    if not sym.Load(BLOCK_SYMBOL, 1):
        raise RuntimeError(f"Could not Load {BLOCK_SYMBOL!r} version 1")
    reply = sym.PlaceBlock(sheet_id, x, y, BLOCK_WIDTH, BLOCK_HEIGHT)
    result = positive_id(reply)
    if not result:
        raise RuntimeError(f"PlaceBlock({x}, {y}) returned {reply!r}")
    return result


def create_connector(job, app, name):
    dev = job.CreateDeviceObject()
    raw = dev.Create(name, "", "", COMPONENT_CODE, COMPONENT_VERSION, 0)
    if not positive_id(raw):
        raise RuntimeError(f"Device.Create({name}) returned {raw!r}")

    # Tested in the existing block-maker workflow: resolve created device by name.
    for did in positive_ids(job.GetAllDeviceIds(None)):
        dev.SetId(did)
        if str(dev.GetName()).strip() == name:
            dev.SetId(did)
            pins = positive_ids(dev.GetPinIds(None))
            if not pins:
                raise RuntimeError(f"Connector {name} contains no pins")
            return did, pins
    raise RuntimeError(f"Created connector device {name!r} was not found")


def place_connector(job, sheet_id, pin_ids, x, y, flip):
    sym = job.CreateSymbolObject()
    raw = sym.PlacePins(list(pin_ids), REQUESTED_SYMBOL, 1,
                        sheet_id, x, y, flip, 1.0)
    sid = positive_id(raw)
    if not sid:
        raise RuntimeError(f"PlacePins at ({x}, {y}) returned {raw!r}")
    sym.SetId(sid)
    actual = try_call(sym, "GetSymbolTypeName")
    type_id = try_call(sym, "GetSymbolTypeId")
    return {
        "symbol_id": sid,
        "requested": REQUESTED_SYMBOL,
        "actual": actual,
        "exact_match": actual == REQUESTED_SYMBOL,
        "symbol_type_id": type_id,
        "characteristic": try_call(sym, "GetCharacteristic"),
        "x": x, "y": y, "flip": flip,
        "pin_count": len(pin_ids),
    }


def run_case(job, app, sheet_id, idx, mil_value, label, timestamp):
    # Four cases per row, each with two blocks to compare RIGHT and LEFT.
    col, row = idx % 4, idx // 4
    bx = START_X + col * COL_GAP
    by = START_Y + row * ROW_GAP
    case = {"label": label, "requested_setting": mil_value,
            "setting_readback": None, "results": {}, "errors": []}

    raw = try_call(job, "SetDisplayMILStandard", mil_value)
    case["setter_result"] = raw
    now = try_call(job, "GetDisplayMILStandard")
    case["setting_readback"] = now
    if type(now) is not int or now != mil_value:
        msg = f"SetDisplayMILStandard({mil_value}) not verified: result={raw!r}, readback={now!r}"
        print("  SKIPPED:", msg, flush=True)
        case["errors"].append(msg)
        return case

    for side, block_x, connector_x, flip in (
        ("right", bx, bx + BLOCK_WIDTH, 0),
        ("left", bx + LEFT_BLOCK_OFFSET, bx + LEFT_BLOCK_OFFSET, "/1"),
    ):
        item = {"block_x": block_x, "block_y": by}
        case["results"][side] = item
        try:
            item["block_symbol_id"] = place_block(job, sheet_id, block_x, by)
            devname = f"MILTEST_{timestamp}_{idx:02d}_{side[0].upper()}"
            device_id, pin_ids = create_connector(job, app, devname)
            item["device_name"] = devname
            item["device_id"] = device_id
            item["placed"] = place_connector(
                job, sheet_id, pin_ids, connector_x, by + PINS_Y_OFFSET, flip
            )
            actual = item["placed"]["actual"]
            match = item["placed"]["exact_match"]
            print(f"  {side.upper():5} requested={REQUESTED_SYMBOL} actual={actual!r} "
                  f"{'EXACT' if match else 'SUBSTITUTED'}", flush=True)
        except Exception as exc:
            msg = f"{side}: {type(exc).__name__}: {exc}"
            case["errors"].append(msg)
            item["error"] = msg
            print("  ERROR:", msg, flush=True)
    return case


def main():
    import win32com.client as win32

    report = {
        "date": datetime.now().isoformat(),
        "component": COMPONENT_CODE,
        "requested_symbol": REQUESTED_SYMBOL,
        "sheet_name": SHEET_NAME,
        "original_mil_setting": None,
        "restored_mil_setting": None,
        "cases": [],
        "best_case": None,
        "errors": [],
        "note": "Experimental bit flips are NOT confirmed inside/outside/outline mode names."
    }

    # Do not use Dispatch() here: GetActiveObject requires already-running E3.
    app = win32.GetActiveObject("CT.Application")
    job = app.CreateJobObject()
    original = None
    try:
        original = job.GetDisplayMILStandard()
        if type(original) is not int or original < 0:
            raise RuntimeError(f"Unexpected GetDisplayMILStandard value: {original!r}")
        report["original_mil_setting"] = original
        print("ORIGINAL MIL SETTING:", original, hex(original), flush=True)
        print("REQUESTED SYMBOL:", REQUESTED_SYMBOL, flush=True)
        print("Using project sheet", SHEET_NAME, "and creating new test blocks/devices.", flush=True)

        # Verify our known component exists before making project changes.
        available = app.GetDatabaseTableSelectedComponents(COMPONENT_CODE, COMPONENT_VERSION)
        if isinstance(available, (tuple, list)) and available and available[0] != 0:
            raise RuntimeError(f"DB component {COMPONENT_CODE} not available: {available!r}")
        sheet_id = get_sheet_id(job)
        report["sheet_id"] = sheet_id

        # Baseline first, then one-bit-at-a-time candidate masks. Remove dupes.
        candidates = [("original", original)]
        candidates += [(f"xor_0x{b:X}", original ^ b) for b in BITS_TO_TEST]
        unique = []
        used = set()
        for label, value in candidates:
            if value not in used:
                used.add(value)
                unique.append((label, value))

        if len(unique) > 16:
            raise RuntimeError("More than 16 tests would exceed the configured grid")
        for idx, (label, value) in enumerate(unique):
            print(f"\nCASE {idx + 1}/{len(unique)}: {label}; MIL {value} ({value:#x})", flush=True)
            case = run_case(job, app, sheet_id, idx, value, label, datetime.now().strftime('%H%M%S'))
            report["cases"].append(case)
            save_json(report)

            # Success requires the requested internal plug symbol on BOTH edges.
            both = all(case.get("results", {}).get(side, {}).get("placed", {}).get("exact_match")
                       for side in ("right", "left"))
            if both:
                report["best_case"] = {"label": label, "mil_value": value,
                                       "decimal": value, "hex": hex(value),
                                       "right_and_left_exact": True}
                print("\nSUCCESS: Both edges placed exactly", REQUESTED_SYMBOL, flush=True)
                if STOP_ON_BOTH_EXACT:
                    break

    except Exception as exc:
        report["errors"].append(f"{type(exc).__name__}: {exc}")
        report["errors"].append(traceback.format_exc())
        print("\nFATAL ERROR:", traceback.format_exc(), flush=True)
    finally:
        # Restore the setting even if any placement, inspection, or write fails.
        if original is not None:
            print("\nRESTORING MIL SETTING TO:", original, hex(original), flush=True)
            try:
                result = job.SetDisplayMILStandard(original)
                readback = job.GetDisplayMILStandard()
                report["restored_mil_setting"] = readback
                report["restore_setter_result"] = result
                report["restore_verified"] = readback == original
                print("RESTORE RESULT:", result, "READBACK:", readback,
                      "VERIFIED:", report["restore_verified"], flush=True)
                if not report["restore_verified"]:
                    report["errors"].append("MIL setting restore FAILED; check E3 Tools -> Settings")
            except Exception as exc:
                report["errors"].append("MIL setting restore ERROR: " + repr(exc))
                print("WARNING: MIL setting restore ERROR:", repr(exc), flush=True)
        save_json(report)
        del job, app

    print("\n" + "=" * 65)
    print("AUTOMATIC TEST COMPLETE")
    print("Results saved:", REPORT)
    if report["best_case"]:
        b = report["best_case"]
        print("FOUND exact internal PLUG on both edges with setting", b["mil_value"], b["hex"])
    else:
        print("No candidate yielded an exact internal PLUG on BOTH edges.")
        print("Next likely area: component database MIL plug-internal symbol assignment.")
    print("Setting restored:", report.get("restore_verified", False))
    print("Open E3 to inspect the test blocks only AFTER this Python process exits.")
    if report["errors"]:
        print("Errors:", *report["errors"][:4], sep="\n - ")
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())
