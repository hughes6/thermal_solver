import time
import win32com.client as win32


SYMBOL_NAME = "Shield_Demo"
SYMBOL_VERSION = 1
SHEET_ID = 204828


app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sym = prj.CreateSymbolObject()


print("")
print("============================================================")
print("SHIELD WIDTH TEST")
print("============================================================")


load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)
print("LOAD RESULT:", load_result)


placed_id = sym.Place(SHEET_ID, 10.0, 2.0, 0, 1.0, 0)
print("PLACE RESULT:", placed_id)


time.sleep(0.10)


sym.SetId(placed_id)


try:
    length_before = float(sym.GetDisplayLength())
except Exception:
    length_before = None


try:
    width_before = float(sym.GetDisplayWidth())
except Exception:
    width_before = None


print("DISPLAY LENGTH BEFORE:", length_before)
print("DISPLAY WIDTH BEFORE:", width_before)


try:
    width_result = sym.SetDisplayWidth(2.0)
except Exception as error:
    width_result = None
    print("SET WIDTH ERROR:", repr(error))


print("SET DISPLAY WIDTH RESULT:", width_result)


time.sleep(0.10)


sym.SetId(placed_id)


try:
    length_after = float(sym.GetDisplayLength())
except Exception:
    length_after = None


try:
    width_after = float(sym.GetDisplayWidth())
except Exception:
    width_after = None


print("DISPLAY LENGTH AFTER:", length_after)
print("DISPLAY WIDTH AFTER:", width_after)


print("")
print("============================================================")
print("DONE")
print("============================================================")


del sym
del prj
del app
