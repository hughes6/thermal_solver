def running_as_packaged_exe() -> bool:
    """Recognize the actual executable, not only the optional sys.frozen flag.

    The previous menu loop happened when a packaged child was mistakenly
    treated as a Python interpreter and launched as [sys.executable, file.py].
    A onefile exe does not interpret file.py: it starts the menu again.
    """
    if bool(getattr(sys, 'frozen', False)) or bool(getattr(sys, '_MEIPASS', None)):
        return True
    if sys.platform != 'win32':
        return False
    # Normalize either Windows or POSIX-style separators for reliability.
    basename = str(sys.executable).replace('\\', '/').rsplit('/', 1)[-1].casefold()
    if not basename.endswith('.exe'):
        return False
    stem = basename[:-4]
    # Python itself is also an .exe; only non-interpreter executables count.
    return stem not in {
        'python', 'pythonw', 'py', 'pyw', 'python3', 'python3w',
    } and not stem.startswith(('python3.', 'python3_'))

def should_show_launcher_error(key, *, error, rc, windows_opened, elapsed):
    """Keep the original handled error; show a fallback only for unhandled failures."""
    # Closing a file picker (or exiting a quick one-shot script) is not a
    # crash. No artificial elapsed-time or window-count threshold is valid.
    if error is None and not rc:
        return False
    # These tools catch operational errors and display their own dialog. Their
    # nonzero return code is NOT a second failure requiring another popup.
    # A Python exception escaping into the launcher still deserves a report.
    if key in TOOLS_WITH_OWN_ERROR_DIALOG and error is None:
        return False
    return True


def trace_exe_start(event: str, **details) -> None:
    """Write durable diagnostics to the Windows temporary directory.

    Avoid resolving runtime_directory here: when the EXE was incorrectly
    identified, old log locations moved and the user couldn't find the log.
    """
    try:
        root = diagnostics_directory()
        with (root / 'launcher_debug.log').open('a', encoding='utf-8', errors='replace') as stream:
            entry = {'at': datetime.now().isoformat(timespec='seconds'),
                     'event': event, 'version': SUITE_VERSION,
                     'pid': os.getpid(), 'sys_frozen': bool(getattr(sys,'frozen',False)),
                     'packaged_exe': running_as_packaged_exe(),
                     'argv': sys.argv[:12], 'tool_env': os.environ.get(EXE_TOOL_ENV),
                     'exe': sys.executable,
                     'meipass': str(getattr(sys, '_MEIPASS', ''))}
            entry.update(details)
            stream.write(json.dumps(entry, ensure_ascii=False, default=str) + '\n')
    except Exception:
        # Do not block legitimate work due to a noncritical diagnostic failure.
        pass

def source_digest(source: str) -> str:
    return hashlib.sha256(source.encode('utf-8')).hexdigest()


def runtime_directory() -> Path:
    """Use a stable, writable tool folder in EXE mode; preserve .py behavior."""
    if running_as_packaged_exe():
        home = Path(os.environ.get('LOCALAPPDATA') or Path.home())
        target = home / 'E3_Mega_Runtime_V26'
        target.mkdir(parents=True, exist_ok=True)
        # Recover the previous pin-identity state automatically when upgrading.
        # Do not rewrite or delete anything from the previous installation.
        for filename in ('main_recovery.json', 'main_pin_identity.json'):
            dst = target / 'cable_pin_spacer' / filename
            if dst.exists():
                continue
            # Newest previous runtime first; never overwrite existing recovery data.
            for old_version in ('V25', 'V24', 'V23', 'V22', 'V21', 'V20', 'V19', 'V18', 'V13'):
                src = home / ('E3_Mega_Runtime_' + old_version) / 'cable_pin_spacer' / filename
                if src.is_file():
                    try:
                        dst.parent.mkdir(parents=True, exist_ok=True)
                        shutil.copy2(src, dst)
                        break
                    except OSError:
                        continue
        return target
    preferred = Path(__file__).resolve().parent / 'E3_Mega_Runtime_V26'
    try:
        preferred.mkdir(parents=True, exist_ok=True)
        with tempfile.NamedTemporaryFile(dir=str(preferred), delete=True):
            pass
        return preferred
    except OSError:
        home = Path(os.environ.get('LOCALAPPDATA') or Path.home())
        fallback = home / 'E3_Mega_Runtime_V26'
        fallback.mkdir(parents=True, exist_ok=True)
        return fallback


def prefs_file() -> Path:
    folder = Path(os.environ.get('LOCALAPPDATA') or Path.home()) / 'E3_Mega_Tools'
    folder.mkdir(parents=True,exist_ok=True)
    return folder / 'prefs.json'


def read_prefs() -> dict:
    try:
        data=json.loads(prefs_file().read_text(encoding='utf-8'))
        if isinstance(data,dict):
            return {key:str(data.get(key,'')).strip() for key in WORKBOOKS}
    except (OSError,ValueError,TypeError):
        pass
    return {key:'' for key in WORKBOOKS}


def write_prefs(data: dict) -> None:
    dest=prefs_file()
    tmp=dest.with_suffix('.tmp')
    tmp.write_text(json.dumps({k:data.get(k,'') for k in WORKBOOKS}, indent=2)+'\n',encoding='utf-8')
    os.replace(tmp,dest)


def validate_excel_template(path: Path) -> None:
    """Check the Excel container but NEVER recalculate or rewrite the workbook."""
    if path.suffix.lower() not in ('.xlsx', '.xlsm'):
        raise ValueError('Choose an Excel .xlsx or .xlsm template.')
    if not path.is_file():
        raise FileNotFoundError(str(path))
    if not zipfile.is_zipfile(path):
        raise ValueError('The selected file is not a valid Excel workbook (.xlsx/.xlsm).')
    try:
        with zipfile.ZipFile(path,'r') as archive:
            required = {'[Content_Types].xml','xl/workbook.xml'}
            if not required.issubset(archive.namelist()):
                raise ValueError('Missing Excel workbook files; choose a genuine template.')
            if archive.testzip() is not None:
                raise ValueError('The selected Excel template is damaged.')
    except zipfile.BadZipFile as exc:
        raise ValueError('The selected Excel template is damaged.') from exc


def template_source_directories() -> tuple[Path, ...]:
    """Read original templates only from the application source/bundle.

    In a PyInstaller executable, only packaged templates under _MEIPASS are
    accepted. This deliberately ignores old user-imported template overrides
    and loose files beside the executable. In Python source mode, templates
    are maintained in the project's templates/ folder.
    """
    if running_as_packaged_exe():
        bundle_dir = getattr(sys, '_MEIPASS', None)
        return (Path(bundle_dir) / 'templates',) if bundle_dir else ()
    return (Path(__file__).resolve().parent / 'templates',)


def get_template_path(key: str) -> Path | None:
    """Return only the original template bundled with the application."""
    if key not in TEMPLATE_GROUPS:
        raise KeyError(key)
    filename = TEMPLATE_GROUPS[key]['filename']
    for directory in template_source_directories():
        candidate = directory / filename
        if candidate.is_file():
            return candidate
    return None


def save_template_copy(key: str, output_path: Path) -> Path:
    """Write an exact workbook copy, preserving formatting, formulas, macros."""
    if key not in TEMPLATE_GROUPS:raise KeyError(key)
    output_path=Path(output_path)
    source=get_template_path(key)
    if source is None:
        raise FileNotFoundError('No original template found in the templates folder for '+TEMPLATE_GROUPS[key]['title'])
    validate_excel_template(source)
    if output_path.suffix.lower()!=source.suffix.lower():
        raise ValueError('Keep the template extension '+source.suffix+' when saving.')
    if output_path.resolve() == source.resolve():
        raise ValueError('Choose another destination; the master template cannot be overwritten.')
    output_path.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source,output_path)
    return output_path


def install_one(relative_path: str, base: Path) -> tuple[Path,str]:
    """Materialize original source; keep local edits if any."""
    if relative_path not in SOURCE_FILES:
        raise KeyError(relative_path)
    path=base / relative_path
    path.parent.mkdir(parents=True,exist_ok=True)
    raw=SOURCE_FILES[relative_path].encode('utf-8')
    digest=hashlib.sha256(raw).hexdigest()
    compile(SOURCE_FILES[relative_path],relative_path,'exec')
    tracker=path.with_name(path.name+'.bundled.sha256')
    if path.exists():
        current=hashlib.sha256(path.read_bytes()).hexdigest()
        if current==digest:
            if not tracker.exists():tracker.write_text(digest+'\n',encoding='utf-8')
            return path,'unchanged'
        previous=tracker.read_text(encoding='utf-8').strip() if tracker.exists() else ''
        if current!=previous:
            return path,'kept user-edited file'
        backup=path.with_name(path.name+'.backup.'+datetime.now().strftime('%Y%m%d_%H%M%S'))
        shutil.copy2(path,backup)
    tmp=path.with_name(path.name+'.updating')
    try:
        tmp.write_bytes(raw)
        os.replace(tmp,path)
        tracker.write_text(digest+'\n',encoding='utf-8')
    finally:
        if tmp.exists():tmp.unlink()
    return path,'installed'


def install_tool(tool_id: str, base: Path) -> tuple[Path,list[str]]:
    tool=TOOLS[tool_id]
    entry=tool['entry']
    folder=entry.split('/')[0]+'/'
    files=[k for k in SOURCE_FILES if k.startswith(folder)]
    actions=[]
    for relative in files:
        _,action=install_one(relative,base)
        actions.append(f'{relative}: {action}')
    return base/entry,actions


def open_folder(folder: Path) -> None:
    folder.mkdir(parents=True,exist_ok=True)
    if sys.platform=='win32':
        os.startfile(str(folder))
    elif sys.platform=='darwin':
        subprocess.Popen(['open',str(folder)])
    else:
        subprocess.Popen(['xdg-open',str(folder)])

    def launch_local(self, key):
        """Open any top-level tool directly in the frozen EXE (no subprocess).

        This does not start a second copy of E3_Mega_Tools.exe. The launcher
        is hidden during the selected operation and restored on close.
        """
        if self.running_local is not None:
            messagebox.showinfo('One tool at a time',
                'Close the running drawing tool first.', parent=self.root)
            return
        group = TOOLS[key]['workbook']
        workbook = self.prefs.get(group, '') if group else ''
        if workbook and not Path(workbook).is_file():
            messagebox.showerror('Workbook not found',
                'The saved workbook no longer exists:\n' + workbook +
                '\n\nChoose a new workbook or clear this selection.', parent=self.root)
            return
        self.running_local = key
        self.disable_buttons(False)
        self.status.set('Opening ' + TOOLS[key]['name'])
        self.root.attributes('-topmost', False)
        self.root.withdraw()
        trace_exe_start('inprocess-start-request', selected=key)
        # Return from the launch button callback before entering the tool's
        # own Tk event loop; both windows stay on the same UI thread.
        self.root.after(20, lambda tool_id=key: self.run_local_tool(tool_id))

    def run_local_tool(self, key):
        import traceback
        script = None
        error = None
        rc = 0
        import time
        started_at = time.monotonic()
        tool_windows = [0]
        trace_exe_start('real-tool-entry', selected=key)
        env_names = ['E3_MEGA_' + name.upper() + '_WORKBOOK' for name in WORKBOOKS]
        original_env = {name: os.environ.get(name) for name in env_names}
        original_hook = sys.excepthook
        try:
            for name in env_names:
                os.environ.pop(name, None)
            group = TOOLS[key]['workbook']
            if group:
                workbook = self.prefs.get(group, '')
                if workbook:
                    os.environ['E3_MEGA_' + group.upper() + '_WORKBOOK'] = workbook
            script, notes = install_tool(key, self.runtime)
            output = script.parent / 'launcher_output.log'
            with output.open('a', encoding='utf-8', errors='replace', buffering=1) as log:
                log.write('\n===== IN-PROCESS ' + datetime.now().isoformat(timespec='seconds')
                          + ' ' + key + ' (' + SUITE_VERSION + ') =====\n')
                for note in notes:
                    log.write('install: ' + note + '\n')
                log.write('STARTUP: entering tool interpreter (not the launcher menu)\n')
                log.flush()
                trace_exe_start('tool-startup-after-install', selected=key, script=str(script))
                original_stdout, original_stderr = sys.stdout, sys.stderr
                try:
                    # --windowed EXEs may have sys.stdout/sys.stderr = None.
                    # Always supply valid streams for the extracted scripts.
                    sys.stdout = log
                    sys.stderr = log
                    # The Pin Spacer must enter GUI worker mode directly;
                    # otherwise its own main() starts an EXE subprocess.
                    args = ['--worker'] if key == 'pin' else []
                    # A second Tk() would normally call mainloop(0), which
                    # cannot return after its window closes if the hidden
                    # launcher Tk() still exists. Threshold 1 means the
                    # tool's own Tk event loop exits on child.destroy(),
                    # while root.quit() continues working as normal. This
                    # changes Tk.mainloop ONLY during the embedded tool call.
                    original_mainloop = tk.Tk.mainloop
                    original_tk_init = tk.Tk.__init__
                    def child_window_mainloop(window, n=0):
                        return original_mainloop(window, max(1, n))
                    def count_tool_window(window, *args, **kwargs):
                        original_tk_init(window, *args, **kwargs)
                        tool_windows[0] += 1
                        trace_exe_start('tool-window-created', selected=key, count=tool_windows[0])
                    tk.Tk.mainloop = child_window_mainloop
                    tk.Tk.__init__ = count_tool_window
                    try:
                        try:
                            run_embedded_tool(key, args)
                        except SystemExit as exc:
                            # Most old tools end with sys.exit(main()). Never
                            # let that terminate the entire mega launcher.
                            rc = int(exc.code) if isinstance(exc.code, int) else (0 if exc.code is None else 1)
                            if rc:
                                log.write('Tool exited with status ' + str(rc) + '\n')
                    finally:
                        tk.Tk.mainloop = original_mainloop
                        tk.Tk.__init__ = original_tk_init
                finally:
                    sys.stdout, sys.stderr = original_stdout, original_stderr
        except BaseException as exc:
            if isinstance(exc, KeyboardInterrupt):
                raise
            error = exc
            rc = 1
            trace_exe_start('inprocess-tool-error', selected=key, error=repr(exc))
            if script is not None:
                try:
                    with (script.parent / 'launcher_output.log').open('a', encoding='utf-8') as log:
                        traceback.print_exception(type(exc), exc, exc.__traceback__, file=log)
                except OSError:
                    pass
        finally:
            sys.excepthook = original_hook
            for name, previous in original_env.items():
                if previous is None:
                    os.environ.pop(name, None)
                else:
                    os.environ[name] = previous
            elapsed = round(time.monotonic() - started_at, 2)
            trace_exe_start('inprocess-tool-end', selected=key, returncode=rc, elapsed_sec=elapsed, windows_opened=tool_windows[0], error=repr(error))
            self.running_local = None
            self.disable_buttons(True)
            try:
                self.root.deiconify()
                self.root.lift()
                self.root.attributes('-topmost', True)
            except tk.TclError:
                pass
            # A returned launcher is NOT necessarily a second EXE. Do not hide
            # an early failure; expose the actual tool log and dispatcher events.
            if should_show_launcher_error(key, error=error, rc=rc,
                                          windows_opened=tool_windows[0], elapsed=elapsed):
                why = ('Python exception' if error is not None else
                       'tool exited with status '+str(rc) if rc else
                       'tool never created a GUI window' if not tool_windows[0] else
                       'tool GUI closed immediately')
                self.status.set(TOOLS[key]['name'] + ': '+why)
                self.show_startup_failure(key,why,script,elapsed,tool_windows[0],error)
            elif rc and key in TOOLS_WITH_OWN_ERROR_DIALOG:
                # Diagnostic information stays available, but the user sees
                # exactly the original tool's friendly error, not duplicates.
                trace_exe_start('handled-error-dialog-used', selected=key, rc=rc)
                self.status.set(TOOLS[key]['name'] + ' stopped — see tool error')
            else:
                self.status.set(TOOLS[key]['name'] + ' closed')

    def show_startup_failure(self, key, reason, script, elapsed, windows_opened, error):
        """Always show the underlying tool startup evidence instead of silently returning to the menu."""
        lines = [
            'E3 Mega Tools V26 — tool startup report',
            'Tool: ' + TOOLS[key]['name'],
            'Reason: ' + str(reason),
            'Tool GUI windows created: ' + str(windows_opened),
            'Run duration: ' + str(round(elapsed, 3)) + ' seconds',
            'Executable: ' + sys.executable,
            'Packaged: ' + str(running_as_packaged_exe()),
            'Source: ' + str(script),
            'Exception: ' + repr(error),
        ]
        if script is not None:
            try:
                logpath = script.parent / 'launcher_output.log'
                lines.append('Tool log: ' + str(logpath))
                if logpath.exists():
                    lines.extend(['', '--- LAST TOOL LOG LINES ---'])
                    lines.extend(logpath.read_text(encoding='utf-8', errors='replace').splitlines()[-95:])
            except Exception as exc:
                lines.append('Could not read tool log: ' + repr(exc))
        # Block Maker has two separate child stages. Their actionable tracebacks
        # are in a second stage log, not necessarily in launcher_output.log.
        if key == 'block_maker':
            try:
                stage_dir = Path(tempfile.gettempdir()) / 'E3_Block_Maker_Logs'
                stage_files = sorted(stage_dir.glob('block_maker_*.log'),
                                     key=lambda p: p.stat().st_mtime, reverse=True)
                if stage_files:
                    stage_log = stage_files[0]
                    lines.extend(['', '--- BLOCK MAKER STAGE LOG ---', str(stage_log)])
                    lines.extend(stage_log.read_text(
                        encoding='utf-8', errors='replace').splitlines()[-125:])
            except Exception as exc:
                lines.append('Could not read Block Maker stage log: ' + repr(exc))
        try:
            diag = diagnostics_directory() / 'launcher_debug.log'
            lines.extend(['', '--- LAUNCHER EVENTS ---', 'Diagnostic log: ' + str(diag)])
            if diag.exists():
                events = diag.read_text(encoding='utf-8', errors='replace').splitlines()[-16:]
                lines.extend(events)
        except Exception as exc:
            lines.append('Could not read diagnostic log: ' + repr(exc))
        report = '\n'.join(lines)
        try:
            report_path = diagnostics_directory() / 'last_tool_failure.txt'
            report_path.write_text(report + '\n', encoding='utf-8')
        except Exception:
            pass
        dialog = tk.Toplevel(self.root)
        dialog.title('E3 Mega Tools — Startup issue')
        dialog.geometry('800x560')
        dialog.attributes('-topmost', True)
        tk.Label(dialog, text='Tool did not stay open — startup details',
                 font=('Segoe UI', 12, 'bold'), anchor='w').pack(fill='x', padx=13, pady=(12,5))
        view = tk.Text(dialog, wrap='word', font=('Consolas', 9), height=24)
        view.pack(fill='both', expand=True, padx=13, pady=(2,6))
        view.insert('1.0', report)
        view.configure(state='disabled')
        buttons = tk.Frame(dialog)
        buttons.pack(fill='x', padx=13, pady=(0,12))
        def copy_report():
            dialog.clipboard_clear()
            dialog.clipboard_append(report)
            dialog.update()
        tk.Button(buttons,text='Copy failure report',command=copy_report).pack(side='left')
        tk.Button(buttons,text='Open diagnostic folder',
                  command=lambda:open_folder(diagnostics_directory())).pack(side='left',padx=(9,0))
        tk.Button(buttons,text='Close',command=dialog.destroy).pack(side='right')

    def launch(self,key):
        trace_exe_start('button-click', selected=key, route='inprocess' if running_as_packaged_exe() else 'python-subprocess')
        if self.running_local is not None:
            messagebox.showinfo('One tool at a time',
                'Close the running drawing tool first.', parent=self.root)
            return
        if self.running is not None and self.running.poll() is None:
            messagebox.showinfo('One tool at a time','Close the running E3 tool first.',parent=self.root)
            return
        # Bypass the PyInstaller subprocess/bootloader for ALL top-level
        # tools. Nested worker processes still use the explicit dispatcher.
        if running_as_packaged_exe() and key in FROZEN_INPROCESS_TOOLS:
            self.launch_local(key)
            return
        tool=TOOLS[key]
        group=tool['workbook']
        workbook=self.prefs.get(group,'') if group else ''
        if workbook and not Path(workbook).is_file():
            messagebox.showerror('Workbook not found',
                'The saved workbook no longer exists:\n'+workbook+
                '\n\nChoose a new workbook or clear this selection.',parent=self.root)
            return
        try:
            script,notes=install_tool(key,self.runtime)
            env=os.environ.copy()
            # Clear stale inherited choices when a workbook is not selected.
            for group_name in WORKBOOKS:
                env.pop('E3_MEGA_'+group_name.upper()+'_WORKBOOK',None)
            if workbook:
                env['E3_MEGA_'+group.upper()+'_WORKBOOK']=workbook
            # sys.executable is python.exe when running .py, but this launcher
            # itself when PyInstaller-frozen. Explicitly route EXE children to
            # the intended embedded tool instead of reopening this menu.
            if running_as_packaged_exe():
                # Double-route the worker: argv AND inherited environment.
                # The parent menu is never a legitimate destination for this
                # process. The child checks this marker BEFORE constructing Tk.
                child_args = ['--worker'] if key == 'pin' else []
                env[EXE_TOOL_ENV] = key
                env[EXE_ARGS_ENV] = json.dumps(child_args)
                cmd = [sys.executable, '--run-tool', key] + child_args
                trace_exe_start('launch-worker', child=key, command=cmd)
            else:
                if key == 'cable_placer':
                    # Route source-mode Cable Placer through the same saved-
                    # workbook adapter as the EXE. Its own .py stays unchanged.
                    entry = Path(__file__).resolve().parent / 'Zuken_Mega_Script.py'
                    cmd = [sys.executable, str(entry), '--run-tool', key]
                else:
                    cmd=[sys.executable,str(script)]
                    if key=='pin':cmd.append('--worker')
            flags=getattr(subprocess,'CREATE_NO_WINDOW',0) if sys.platform=='win32' else 0
            logfile=script.parent/'launcher_output.log'
            with logfile.open('a',encoding='utf-8') as log:
                log.write('\n===== '+datetime.now().isoformat(timespec='seconds')+' '+key+
                         ' ('+SUITE_VERSION+') =====\n')
                for note in notes:log.write('install: '+note+'\n')
                if workbook:log.write('workbook: '+workbook+'\n')
            with logfile.open('a',encoding='utf-8') as log:
                self.running=subprocess.Popen(cmd,cwd=str(script.parent),env=env,
                    stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT,
                    creationflags=flags,close_fds=True)
            self.running_id=key
            self.disable_buttons(False)
            self.status.set(tool['name']+' is running')
            self.root.attributes('-topmost',False)
        except Exception as exc:
            self.running=None;self.running_id=None
            self.disable_buttons(True)
            self.status.set('Failed to launch '+tool['name'])
            messagebox.showerror('E3 Tools launcher error',str(exc),parent=self.root)

    def poll_process(self):
        try:
            if self.running is not None:
                returncode=self.running.poll()
                if returncode is not None:
                    name=TOOLS[self.running_id]['name']
                    self.running=None;self.running_id=None
                    self.disable_buttons(True)
                    self.status.set(name+(' completed / closed' if returncode==0 else ' failed — see logs'))
                    self.root.attributes('-topmost',True)
        finally:
            self.root.after(500,self.poll_process)

        tk.Button(buttons, text='Copy Diagnostics', command=copy_all).pack(side='left')
        tk.Button(buttons, text='Open Log Folder',
                  command=lambda:open_folder(diagnostics_directory())).pack(side='left',padx=(8,0))
        tk.Button(buttons, text='Close', command=dialog.destroy).pack(side='right')

    def import_pin_data(self):
        directory=filedialog.askdirectory(parent=self.root,title='Locate previous pin spacer recovery files')
        if not directory:return
        destination=self.runtime/'cable_pin_spacer'
        destination.mkdir(parents=True,exist_ok=True)
        imported=[]
        for suffix in ('_recovery.json','_pin_identity.json'):
            for name in ('main'+suffix,'E3_Cable_To_Block_Pin_Spacing_V15_ZUKEN_LIGHT_GUI'+suffix):
                original=Path(directory)/name
                dest=destination/('main'+suffix)
                if original.is_file() and not dest.exists():
                    data=json.loads(original.read_text(encoding='utf-8-sig'))
                    if not isinstance(data,dict):raise ValueError('Expected JSON object in '+name)
                    shutil.copy2(original,dest)
                    imported.append(dest.name)
                    break
        messagebox.showinfo('Pin Spacer recovery data',
            'Imported: '+(', '.join(imported) or 'none')+'\nExisting files were left untouched.',
            parent=self.root)

    def close(self):
        if self.running_local is not None:
            messagebox.showinfo('Tool is open',
                'Close the drawing tool first.', parent=self.root)
            return
        if self.running is not None and self.running.poll() is None:
            if not messagebox.askyesno('Close launcher?',
                'A tool is still running. Close only the launcher and let it finish?',parent=self.root):
                return
        self.root.destroy()


@contextlib.contextmanager
def launcher_workbook_override(tool_id: str):
    """Honor the saved Cable Maker/Placer workbook without modifying Cable Placer.

    Both the EXE in-process route and the normal Python subprocess route carry
    E3_MEGA_CABLE_WORKBOOK. Cable Placer's legacy ask_save_path() always calls
    tkinter.filedialog.askopenfilename(), so supply that saved path to its FIRST
    Excel-open dialog. All other dialogs and tools keep their original behavior.
    """
    saved = (os.environ.get('E3_MEGA_CABLE_WORKBOOK') or '').strip() if tool_id == 'cable_placer' else ''
    if not saved:
        yield
        return

    workbook = Path(saved).expanduser()
    if not workbook.is_file():
        raise FileNotFoundError('The saved Cable Maker / Placer workbook is missing: ' + str(workbook))
    if workbook.suffix.lower() not in ('.xlsx', '.xlsm'):
        raise ValueError('The saved Cable Maker / Placer workbook must be .xlsx or .xlsm: ' + str(workbook))

    original_picker = filedialog.askopenfilename
    supplied = False

    def choose_workbook_or_prompt(*args, **kwargs):
        nonlocal supplied
        # Only intercept an Excel *open* dialog. Do not intercept a PDF picker,
        # a save-as dialog, or subsequent requests for a different workbook.
        filetypes = kwargs.get('filetypes') or ()
        filetypes_text = ' '.join(map(str, filetypes)).lower()
        if not supplied and ('.xls' in filetypes_text or 'excel' in filetypes_text):
            supplied = True
            print('Using Cable Maker / Placer workbook saved in Mega Tools:', workbook)
            return str(workbook)
        return original_picker(*args, **kwargs)

    filedialog.askopenfilename = choose_workbook_or_prompt
    try:
        yield
    finally:
        # Always restore tkinter for the launcher and all other tools,
        # including when the legacy script calls sys.exit() or raises.
        filedialog.askopenfilename = original_picker


def run_embedded_tool(tool_id: str, tool_args: list[str]) -> None:
    """EXE subprocess entry. Execute a tool *in this process*, not the menu.

    Each tool still receives its normal __file__, import directory, cwd,
    arguments and inherited workbook environment. The GUI remains in its
    own parent process with independent E3 COM access.
    """
    if tool_id not in TOOLS:
        trace_exe_start('invalid-worker-id', invalid_id=tool_id)
        raise SystemExit('Unknown E3 tool: '+str(tool_id))
    # Crucial: nested Pin Spacer / Block Maker workers launch new processes.
    # They must not inherit this top-level role and be accidentally routed
    # to the WRONG tool. Their own --run-tool arguments remain authoritative.
    os.environ.pop(EXE_TOOL_ENV, None)
    os.environ.pop(EXE_ARGS_ENV, None)
    trace_exe_start('dispatch-start', selected=tool_id, args=tool_args)
    import runpy
    script, _actions = install_tool(tool_id, runtime_directory())
    trace_exe_start('worker-file-ready', selected=tool_id, script=str(script), install=_actions)
    # PyInstaller --windowed can leave sys.stdout/sys.stderr as None even
    # when the parent has redirected OS-level handles. Many original tools
    # use print() and traceback.print_exc(), so supply a writable log stream.
    if sys.stdout is None:
        sys.stdout = (script.parent / 'launcher_output.log').open(
            'a', encoding='utf-8', errors='replace', buffering=1)
    if sys.stderr is None:
        sys.stderr = sys.stdout
    tool_dir = str(script.parent)
    prior_argv = sys.argv[:]
    prior_cwd = os.getcwd()
    sys.path.insert(0, tool_dir)
    sys.argv = [str(script)] + list(tool_args)
    try:
        os.chdir(tool_dir)
        trace_exe_start('executing-tool-source', selected=tool_id, real_file=str(script))
        if sys.stdout is not None:
            print('STARTUP: executing '+str(script), flush=True)
        with launcher_workbook_override(tool_id):
            runpy.run_path(str(script), run_name='__main__')
    finally:
        sys.argv = prior_argv
        os.chdir(prior_cwd)
        if sys.path and sys.path[0] == tool_dir:
            sys.path.pop(0)


def acquire_main_window_lock():
    """Prevent any mistakenly-routed worker from opening another main menu."""
    if sys.platform != 'win32' or not running_as_packaged_exe():
        return None
    import ctypes
    kernel32 = ctypes.WinDLL('kernel32', use_last_error=True)
    create = kernel32.CreateMutexW
    create.argtypes = (ctypes.c_void_p, ctypes.c_int, ctypes.c_wchar_p)
    create.restype = ctypes.c_void_p
    handle = create(None, 0, 'Local\\E3MegaTools_MainWindow_Singleton_V26')
    if not handle:
        raise ctypes.WinError(ctypes.get_last_error())
    if ctypes.get_last_error() == 183:  # ERROR_ALREADY_EXISTS
        trace_exe_start('duplicate-menu-suppressed', reason='another main window is open')
        kernel32.CloseHandle(ctypes.c_void_p(handle))
        return False
    return (kernel32, handle)


def main():
    # Two independent dispatch channels: command line AND Windows process
    # environment. The environment marker survives even when a frozen EXE
    # is restarted with unexpected argv. Never allow a child to open MegaApp.
    trace_exe_start('program-entry')
    is_frozen = running_as_packaged_exe()
    if is_frozen:
        os.environ['E3_MEGA_PACKAGED_EXE'] = '1'
    env_tool = os.environ.get(EXE_TOOL_ENV, '').strip()
    cli_tool = None
    cli_args = []
    if '--run-tool' in sys.argv:
        i = sys.argv.index('--run-tool')
        if len(sys.argv) <= i+1:
            trace_exe_start('bad-command-line', problem='missing-tool-id')
            raise SystemExit('Missing tool ID after --run-tool')
        cli_tool, cli_args = sys.argv[i+1], sys.argv[i+2:]
    if cli_tool or env_tool:
        # Prefer explicit CLI for nested worker commands. A bad environment
        # marker fails CLOSED rather than accidentally reopening the menu.
        if cli_tool:
            tool_id, tool_args = cli_tool, cli_args
        else:
            tool_id = env_tool
            try:
                tool_args = json.loads(os.environ.get(EXE_ARGS_ENV, '[]'))
                if not isinstance(tool_args, list) or not all(isinstance(a, str) for a in tool_args):
                    raise ValueError('Invalid worker arguments')
            except (ValueError, TypeError) as exc:
                trace_exe_start('bad-worker-args', error=str(exc))
                raise SystemExit('Invalid E3 tool subprocess arguments') from exc
        if tool_id not in TOOLS:
            trace_exe_start('unknown-tool', selected=tool_id)
            raise SystemExit('Unknown E3 child tool: '+str(tool_id))
        try:
            run_embedded_tool(tool_id, tool_args)
        except Exception as exc:
            # Worker errors must exit with status 1, not bubble to PyInstaller's
            # generic "failed to execute script" popup. The parent Block Maker
            # or Pin Arranger already has its own actionable error dialog.
            import traceback as _traceback
            trace_exe_start('worker-error-handled', selected=tool_id, error=repr(exc))
            if sys.stderr is not None:
                _traceback.print_exc(file=sys.stderr)
                sys.stderr.flush()
            raise SystemExit(1) from None
        return
    if '--self-test' in sys.argv or '--diagnose' in sys.argv:
        assert len(TOOLS)==14 and len(SOURCE_FILES)==18
        for filename,code in SOURCE_FILES.items():
            compile(code,filename,'exec')
            print('OK:',filename,len(code.splitlines()),'lines',source_digest(code)[:14])
        print('PASS: 14 tool entries / 18 separate Python source files. No E3/SQL invoked.')
        return
    if '--verify-templates' in sys.argv:
        for key, info in TEMPLATE_GROUPS.items():
            path=get_template_path(key)
            if path is None:
                raise FileNotFoundError('Missing '+info['filename']+' in templates folder')
            validate_excel_template(path)
            print('OK:',info['filename'],'('+str(path)+')')
        print('PASS: all four templates found and verified. No E3/SQL invoked.')
        return
    if '--list-templates' in sys.argv:
        for key,info in TEMPLATE_GROUPS.items():
            path=get_template_path(key)
            print(key+': '+(str(path) if path else 'not found'))
        return
    if '--list-tools' in sys.argv:
        for key,info in TOOLS.items():print(key+': '+info['entry'])
        return
    if '--extract' in sys.argv:
        output=runtime_directory()
        for key in TOOLS:
            entry,notes=install_tool(key,output)
            print(key+': '+str(entry))
            for note in notes:print('  '+note)
        return
    # No worker marker means this process is deliberately the main menu.
    # Only one main menu is permitted; a misrouted worker cannot duplicate it.
    lock = acquire_main_window_lock()
    if lock is False:
        return
    trace_exe_start('launcher-menu-open')
    try:
        MegaApp().root.mainloop()
    finally:
        if lock is not None:
            kernel32, handle = lock
            kernel32.CloseHandle(handle)

if __name__=='__main__':
    main()
