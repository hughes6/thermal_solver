"""Two-stage Block Maker text formatting.

Stage 1: existing connector and pin formatting (without forcing SCH headers on).
Stage 2: wait for the user to activate Saber Parameter 1 in Zuken; Equipment
         Identification is optional and formatted ONLY where already active.
"""
import json
import os
import time
import tkinter as tk
from tkinter import messagebox

from utils import runn, read_schematic_header_activation


def wait_for_schematic_header_activation(metadata):
    """Keep the launcher paused until the required SCH Saber texts are ON.

    Closing the dialog means cancel; we must not format any hidden header text.
    E3 COM is opened only briefly when DONE is clicked, so E3 remains usable
    while the operator is enabling the attribute text.
    """
    root = tk.Tk()
    root.title('SCH Block Header Attributes')
    root.attributes('-topmost', True)
    root.resizable(False, False)

    width, height = 690, 295
    x = (root.winfo_screenwidth() - width) // 2
    y = (root.winfo_screenheight() - height) // 2
    root.geometry(f'{width}x{height}+{x}+{y}')

    instructions = (
        'In Zuken, turn ON Saber Parameter 1 for every SCH block with a name in Excel '
        '(K3, P3, U3, ...).\n\n'
        'Turn ON Equipment Identification only for the SCH blocks where you '
        'want it displayed.\n'
        'Equipment Identification is optional and will NOT be enabled automatically.\n\n'
        'Once the attributes are visible in E3, click DONE to format them.'
    )
    tk.Label(root, text=instructions, font=('Arial', 11), justify='center',
             wraplength=640).pack(padx=22, pady=(24, 12))
    status = tk.StringVar(value='Waiting for SCH header attribute activation in E3.')
    tk.Label(root, textvariable=status, font=('Arial', 10), fg='#295f91',
             wraplength=635).pack(padx=18, pady=(0, 10))
    approved = {'done': False}

    def finish():
        try:
            current = read_schematic_header_activation(metadata)
        except Exception as exc:
            messagebox.showerror('E3 Attribute Check',
                                 f'Could not check the active E3 attributes:\n{exc}', parent=root)
            return
        missing = [item['name'] for item in current
                   if item['saber_required'] and not item['saber_active']]
        if missing:
            status.set('Waiting for Saber Parameter 1 on: ' + ', '.join(missing))
            messagebox.showwarning(
                'Saber Parameter 1 Not Active',
                'Turn ON Saber Parameter 1 attribute text in E3 for:\n\n'
                + '\n'.join(missing)
                + '\n\nEquipment Identification is optional.',
                parent=root,
            )
            return
        equipment_count = sum(1 for item in current if item['equipment_active'])
        print('SCH HEADER READY: Saber text active where required; '
              f'Equipment text active on {equipment_count} block(s)', flush=True)
        approved['done'] = True
        root.destroy()

    def cancel():
        approved['done'] = False
        root.destroy()

    buttons = tk.Frame(root)
    buttons.pack(pady=(2, 12))
    tk.Button(buttons, text='DONE - FORMAT ACTIVE TEXT', width=32, height=2,
              command=finish).pack(side='left', padx=8)
    tk.Button(buttons, text='CANCEL', width=14, height=2,
              command=cancel).pack(side='left', padx=8)
    root.protocol('WM_DELETE_WINDOW', cancel)
    root.lift()
    root.focus_force()
    root.mainloop()
    return approved['done']


def main():
    print('\n' + '=' * 60, flush=True)
    print('FIRST FORMATTER PASS', flush=True)
    print('=' * 60, flush=True)

    metadata_path = os.environ.get('E3_SECONDARY_BLOCK_FILE')
    if not metadata_path or not os.path.isfile(metadata_path):
        raise RuntimeError('Block Maker metadata not found. Run through launcher.py.')
    with open(metadata_path, 'r', encoding='utf-8') as handle:
        metadata = json.load(handle)
    size = float(metadata[0].get('text_size', 0.2)) if metadata else 0.2

    runn(False, size)
    print('FIRST FORMATTER PASS COMPLETE', flush=True)
    print('WAITING FOR USER TO ACTIVATE SCH SABER / OPTIONAL EQUIPMENT TEXT', flush=True)

    if not wait_for_schematic_header_activation(metadata):
        print('SCH HEADER FORMATTING CANCELED BY USER; NO HIDDEN ATTRIBUTES CHANGED',
              flush=True)
        return 0

    print('USER CONFIRMED ACTIVE SCH HEADER ATTRIBUTES', flush=True)
    print('WAITING BRIEFLY FOR E3 TO SETTLE', flush=True)
    time.sleep(1.0)
    print('SECOND FORMATTER PASS: ACTIVE SCH HEADERS', flush=True)
    formatted = runn(True, size)
    print('SECOND FORMATTER PASS COMPLETE:', formatted, flush=True)
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
