def normalize_pdf_line(line):
    line = line.replace("\t", " ")
    line = re.sub(r"\s+", " ", line)

    return line.strip()


def is_integer_value(value):
    value = normalize_pdf_line(value)

    return re.fullmatch(r"\d+", value) is not None


# =========================================
# PDF CONNECTOR PINOUT TABLE DETECTION
#
# Handles copied PDF tables such as:
#
# Ref
# Des Pin Signal Name Direction Type Description
# J1R
# 1 NC NC NC No Connect
# 11 GND - GND Secondary ground
# 12 P5_1000BT_D4- Input BASE-T BUS ... pair 4/D, negative
#
# The important detail is that "Ref Des" is often a merged cell in the PDF,
# so the connector designator appears once and the following data rows begin
# with the pin number. Descriptions are allowed to contain commas.
# =========================================

PINOUT_HEADERS = [
    "REF DES",
    "PIN",
    "SIGNAL NAME",
    "DIRECTION",
    "TYPE",
    "DESCRIPTION"
]


def get_nonempty_normalized_lines(input_text):
    lines = []

    for raw_line in input_text.split("\n"):
        line = normalize_pdf_line(raw_line)

        if line != "":
            lines.append(line)

    return lines




# =========================================
# PDF 4-COLUMN PIN / DESCRIPTION / FIGURE / LOGIC TABLE
#
# Handles tables such as the J5 Clock Signals table:
#   Pin Number | Description     | Figure    | Logic
#   1          | Chassis GND     | -         | -
#   6          | 10 MHz+ Input   | Figure 10 | LVDS
#   2          | 10 MHz- Input   | Figure 10 | LVDS
#
# This parser intentionally runs BEFORE the older 3-column Figure-table parser.
# Otherwise a 4-column table can be mistaken for Pin/Description/Figure and the
# Logic column is silently lost.
#
# It supports both normal row-wise clipboard text and Acrobat's tagged/interleaved
# form where the next pin can be emitted alongside the previous row's Figure and
# Logic cells, for example:
#
#   Pin
#   Number
#   Description
#   1
#   Figure
#   Logic
#   6 - -
#   Chassis GND
#   2 Figure 10 LVDS
#   10 MHz+ Input
#   7 Figure 10 LVDS
#   10 MHz- Input
#
# Note that Description values are allowed to BEGIN WITH NUMBERS (10 MHz...), so
# this table family must never locate rows by blindly treating every numeric token
# as a new pin.
# =========================================

FIGURE_LOGIC_PINOUT_HEADERS = [
    "PIN NUMBER",
    "DESCRIPTION",
    "FIGURE",
    "LOGIC"
]


def normalize_dash_cell(value):
    text = normalize_pdf_line(value)
    if text in {"-", "–", "—"}:
        return "-"
    return text


# A description/signal is allowed to START with a number.  The J5 clock table
# contains values such as "10 MHz+ Input" and "10 MHz- Input".  Those leading
# 10s are part of Column 2; they are NOT pin numbers.
#
# Keep this deliberately unit-based instead of saying "every number followed by
# text is a description", because a normal row is also "6 10 MHz+ Input ...".
# In that normal row, the first number (6) is the pin and the *second* number is
# the start of the description.
ENGINEERING_DESCRIPTION_UNIT_RE = re.compile(
    r"(?:Hz|kHz|MHz|GHz|THz|bps|kbps|Mbps|Gbps|Tbps|"
    r"V|mV|uV|µV|A|mA|uA|µA|W|mW|kW|Ohm|ohms|Ω)"
    r"[+\-]?",
    re.IGNORECASE
)


def looks_like_numeric_leading_description(value):
    """
    True when the beginning of a cell is an engineering value such as:
        10 MHz+ Input
        10 MHz- Input
        3.3 V Logic

    This check happens BEFORE pin-row detection so the leading numeric value is
    preserved in Column 2 instead of being stolen and emitted as a pin.
    """
    text = normalize_pdf_line(value)
    if text == "":
        return False

    match = re.match(r"^(\d+(?:\.\d+)?)\s+([^\s]+)(?:\s|$)", text)
    if match is None:
        return False

    unit_token = match.group(2).strip().rstrip(",;:")
    return ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(unit_token) is not None


def next_line_continues_numeric_description(lines, index):
    """
    Handle the even uglier Acrobat form where one cell is broken as:
        10
        MHz+ Input

    An exact integer followed immediately by an engineering-unit fragment is
    treated as description text rather than a new pin.
    """
    if index < 0 or index >= len(lines):
        return False

    current = normalize_pdf_line(lines[index])
    if re.fullmatch(r"\d+(?:\.\d+)?", current) is None:
        return False

    lookahead = index + 1
    while lookahead < len(lines):
        nxt = normalize_pdf_line(lines[lookahead])
        if nxt == "":
            lookahead += 1
            continue
        first = nxt.split()[0].rstrip(",;:")
        return ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(first) is not None

    return False


def is_logic_cell_text(value):
    """Return True for the short electrical-interface values used in Logic columns."""
    text = normalize_dash_cell(value)
    if text == "-":
        return True

    upper = text.upper().replace(" ", "")

    known = {
        "LVDS", "LVTTL", "TTL", "CMOS", "LVCMOS", "ECL", "PECL", "LVPECL",
        "CML", "RS422", "RS-422", "RS485", "RS-485", "SINGLEENDED",
        "SINGLE-ENDED", "DIFFERENTIAL"
    }
    if upper in known:
        return True

    # Voltage-style logic labels such as 3.3V / 5V / 1.8 V.
    if re.fullmatch(r"\d+(?:\.\d+)?V", upper):
        return True

    return False


def is_figure_logic_table_header_fragment(value):
    upper = normalize_pdf_line(value).upper()
    if upper == "":
        return True

    if upper in {
        "PIN", "NUMBER", "PIN NUMBER", "DESCRIPTION", "FIGURE", "LOGIC",
        "LOGIC TYPE", "LOGIC LEVEL"
    }:
        return True

    if upper.startswith("TABLE ") and ("PIN" in upper or "CONNECTOR" in upper):
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True
    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True

    return False


def get_figure_logic_header_window(input_text):
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 4:
        return lines, None

    search_limit = min(len(lines), 40)
    pin_seen = False
    number_seen = False
    description_seen = False
    figure_seen = False
    logic_seen = False
    first_header_index = None
    last_header_index = None

    for index, line in enumerate(lines[:search_limit]):
        compact = re.sub(r"\s+", " ", line.upper()).strip()

        # These tests intentionally work for either individual header cells or
        # a whole header row copied as one line.
        if compact == "PIN" or "PIN NUMBER" in compact:
            pin_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if compact == "NUMBER" or "PIN NUMBER" in compact:
            number_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if re.search(r"\bDESCRIPTION\b", compact):
            description_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        # Require the standalone word Figure, not Figure 10 in a data cell.
        if compact == "FIGURE" or re.search(r"\bDESCRIPTION\b.*\bFIGURE\b", compact):
            figure_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if compact == "LOGIC" or compact.startswith("LOGIC ") or re.search(r"\bFIGURE\b.*\bLOGIC\b", compact):
            logic_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if pin_seen and number_seen and description_seen and figure_seen and logic_seen:
            break

    if not (pin_seen and number_seen and description_seen and figure_seen and logic_seen):
        return lines, None

    return lines, {
        "first": first_header_index,
        "last": last_header_index
    }


def extract_one_figure_cell_prefix(value):
    """
    Return (figure_cell, remainder) when value starts with one Figure-column cell.
    A comma-separated reference such as 'Figure 5, Figure 8' is one cell.
    """
    text = normalize_pdf_line(value)
    match = re.match(
        r"^(Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*)(?:\s+(.*))?$",
        text,
        re.IGNORECASE
    )
    if match is None:
        return "", text
    return normalize_pdf_line(match.group(1)), normalize_pdf_line(match.group(2) or "")


def split_pin_description_figure_logic_payload(payload):
    """
    Split one 4-column payload into Description / Figure / Logic.

    Important: Figure and Logic are independent cells. A dash is a REAL cell
    value, not punctuation to discard. Partial clipboard rows are allowed too,
    for example '10 MHz+ Input Figure 10' with Logic on the next line.
    """
    text = normalize_pdf_line(payload)
    if text == "":
        return "", "", ""

    figure_match = re.search(
        r"\bFigure\s+\d+(?:\s*,\s*Figure\s+\d+)*",
        text,
        re.IGNORECASE
    )
    if figure_match is not None:
        description = text[:figure_match.start()].strip()
        figure = normalize_pdf_line(figure_match.group(0))
        tail = normalize_dash_cell(text[figure_match.end():].strip())
        logic = tail
        return description, figure, logic

    dash_pair = re.match(r"^(.*?)\s+[-–—]\s+[-–—]$", text)
    if dash_pair is not None:
        return dash_pair.group(1).strip(), "-", "-"

    if re.fullmatch(r"[-–—]\s+[-–—]", text):
        return "", "-", "-"

    dash_logic = re.match(r"^(.*?)\s+[-–—]\s+(\S(?:.*\S)?)$", text)
    if dash_logic is not None and is_logic_cell_text(dash_logic.group(2)):
        return dash_logic.group(1).strip(), "-", normalize_dash_cell(dash_logic.group(2))

    single_dash = re.match(r"^(.*?)\s+[-–—]$", text)
    if single_dash is not None and single_dash.group(1).strip() != "":
        return single_dash.group(1).strip(), "-", ""

    return text, "", ""


def parse_normal_pin_description_figure_logic_rows(lines):
    rows = []

    for raw_line in lines:
        line = normalize_pdf_line(raw_line)
        if line == "" or is_figure_logic_table_header_fragment(line):
            continue

        # IMPORTANT: "10 MHz+ Input Figure 10 LVDS" is a Description/Figure/
        # Logic fragment, not pin 10.  Preserve the 10 in Column 2.
        if looks_like_numeric_leading_description(line):
            continue

        match = re.match(r"^(\d+)\s+(.+)$", line)
        if match is None:
            continue

        pin = match.group(1)
        description, figure, logic = split_pin_description_figure_logic_payload(match.group(2))

        # Require actual trailing column evidence.  This is what keeps a line
        # such as '10 MHz+ Input' from being interpreted as pin 10.
        if description == "" or figure == "" or logic == "":
            continue

        rows.append([pin, description, figure, logic])

    return rows


def split_figure_logic_only_payload(value):
    """Return (figure, logic) only when the entire value is those two cells."""
    text = normalize_pdf_line(value)
    if text == "":
        return "", ""

    if re.fullmatch(r"[-–—]\s+[-–—]", text):
        return "-", "-"

    figure, remainder = extract_one_figure_cell_prefix(text)
    if figure and remainder and is_logic_cell_text(remainder):
        return figure, normalize_dash_cell(remainder)

    return "", ""


def parse_stream_pin_description_figure_logic_rows(lines, header_info):
    """
    Reconstruct the 4-column table as independent ordered streams.

    Standalone '-' cells are preserved and assigned in physical column order:
    Figure first, then Logic. Description is always split away from Figure/Logic
    before rows are zipped back together.
    """
    pins = []
    descriptions = []
    figures = []
    logics = []

    start_index = max(0, header_info["first"])

    def append_figure(value):
        value = normalize_dash_cell(value)
        if value != "":
            figures.append(value)

    def append_logic(value):
        value = normalize_dash_cell(value)
        if value != "":
            logics.append(value)

    def append_ambiguous_dash():
        if len(figures) <= len(logics):
            figures.append("-")
        else:
            logics.append("-")

    line_index = start_index
    while line_index < len(lines):
        line = normalize_pdf_line(lines[line_index])
        if line == "" or is_figure_logic_table_header_fragment(line):
            line_index += 1
            continue

        # Protect engineering descriptions that start with a number.  This must
        # run before the generic ^\d+ pin matcher.
        if looks_like_numeric_leading_description(line):
            description, figure, logic = split_pin_description_figure_logic_payload(line)
            if description:
                descriptions.append(description)
            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)
            line_index += 1
            continue

        # Acrobat can split the same description into two physical lines:
        #     10
        #     MHz+ Input [Figure 10] [LVDS]
        # Rejoin those two pieces before pin detection.
        if next_line_continues_numeric_description(lines, line_index):
            next_text = normalize_pdf_line(lines[line_index + 1])
            combined = normalize_pdf_line(line + " " + next_text)
            description, figure, logic = split_pin_description_figure_logic_payload(combined)
            if description:
                descriptions.append(description)
            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)
            line_index += 2
            continue

        dash_tokens = re.findall(r"(?<!\S)[-–—](?!\S)", line)
        if dash_tokens:
            remainder_without_dashes = re.sub(r"(?<!\S)[-–—](?!\S)", " ", line)
            if normalize_pdf_line(remainder_without_dashes) == "":
                for _ in dash_tokens:
                    append_ambiguous_dash()
                line_index += 1
                continue

        pin_match = re.match(r"^(\d+)(?:\s+(.*))?$", line)
        if pin_match is not None:
            pin = pin_match.group(1)
            remainder = normalize_pdf_line(pin_match.group(2) or "")

            if remainder == "":
                pins.append(pin)
                line_index += 1
                continue

            description, figure, logic = split_pin_description_figure_logic_payload(remainder)

            # IMPORTANT FIX:
            # A PDF row can be split after column 2, for example:
            #     6 10 MHz+ Input
            #     Figure 10
            #     LVDS
            # or:
            #     8 1PPS+ Input
            #     Figure 10
            #     LVDS
            #
            # At this point the WHOLE line has already passed the
            # looks_like_numeric_leading_description(line) guard above, so the
            # first number is the pin.  Do not throw the pin away just because
            # Figure/Logic were copied onto later physical lines.
            pins.append(pin)

            if description:
                descriptions.append(description)

            # Do NOT append `remainder` when it contained only Figure/Logic
            # cells (for example the transition marker "2 Figure 10 LVDS").
            # Those trailing cells belong to the previous row in Acrobat's
            # interleaved clipboard order; treating them as Description is what
            # caused Column 3 to leak into Column 2.

            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)

            line_index += 1
            continue

        description, figure, logic = split_pin_description_figure_logic_payload(line)
        if figure or logic:
            if description:
                descriptions.append(description)
            if figure:
                append_figure(figure)
            if logic:
                append_logic(logic)
            line_index += 1
            continue

        figure_only, logic_only = split_figure_logic_only_payload(line)
        if figure_only or logic_only:
            if figure_only:
                append_figure(figure_only)
            if logic_only:
                append_logic(logic_only)
            line_index += 1
            continue

        if re.fullmatch(r"Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*", line, re.IGNORECASE):
            append_figure(line)
            line_index += 1
            continue

        if is_logic_cell_text(line) and normalize_dash_cell(line) != "-":
            append_logic(line)
            line_index += 1
            continue

        descriptions.append(line)
        line_index += 1

    if len(pins) < 2:
        return []

    if len(descriptions) < len(pins):
        expanded = []
        for description in descriptions:
            expanded.extend(split_merged_description_cell(description))
        if len(expanded) >= len(descriptions):
            descriptions = expanded

    descriptions = descriptions[:len(pins)]
    figures = figures[:len(pins)]
    logics = logics[:len(pins)]

    while len(descriptions) < len(pins):
        descriptions.append("")
    while len(figures) < len(pins):
        figures.append("")
    while len(logics) < len(pins):
        logics.append("")

    return [[pin, descriptions[i], figures[i], logics[i]] for i, pin in enumerate(pins)]


def expand_figure_logic_trailing_cells(value):
    """
    Expand a physical clipboard line into Figure/Logic cells.  This is used by
    the column-major fallback where Acrobat copies the table one column at a
    time.  Dashes are kept as real cells.
    """
    text = normalize_pdf_line(value)
    if text == "":
        return []

    cells = []
    position = 0
    token_re = re.compile(
        r"Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*|(?<!\S)[-–—](?!\S)|"
        r"LVDS|LVTTL|TTL|CMOS|LVCMOS|ECL|PECL|LVPECL|CML|RS-?422|RS-?485",
        re.IGNORECASE
    )

    for match in token_re.finditer(text):
        gap = normalize_pdf_line(text[position:match.start()])
        if gap:
            # Unknown text means this is probably not a pure trailing-cell line.
            return []
        cells.append(normalize_dash_cell(match.group(0)))
        position = match.end()

    if normalize_pdf_line(text[position:]):
        return []

    return cells


def parse_column_major_pin_description_figure_logic_rows(lines, header_info):
    """
    Fallback for PDFs that copy the J5-style table by columns rather than rows:

        1
        6
        2
        7
        ...
        Chassis GND
        10 MHz+ Input
        10 MHz- Input
        ...
        -
        Figure 10
        Figure 10
        ...
        -
        LVDS
        LVDS
        ...

    The key rule is that once the leading pin block has ended, numeric-leading
    descriptions such as "10 MHz+ Input" are collected as Description cells
    and can never re-enter the pin stream.
    """
    start_index = max(0, header_info["first"])
    data = []
    for line in lines[start_index:]:
        line = normalize_pdf_line(line)
        if line == "" or is_figure_logic_table_header_fragment(line):
            continue
        data.append(line)

    if len(data) < 4:
        return []

    # Column-major copies start with a contiguous run of exact pin cells.
    pins = []
    index = 0
    while index < len(data):
        line = data[index]
        if re.fullmatch(r"\d+", line) is None:
            break

        # Do not steal the first half of a split engineering description.
        if index + 1 < len(data):
            nxt = normalize_pdf_line(data[index + 1])
            first = nxt.split()[0].rstrip(",;:") if nxt else ""
            if ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(first):
                break

        pins.append(line)
        index += 1

    if len(pins) < 2:
        return []

    row_count = len(pins)

    # Collect exactly one Description cell per pin.  Numeric-leading engineering
    # descriptions are preserved verbatim, including the leading number.
    descriptions = []
    pending_trailing = []

    while index < len(data) and len(descriptions) < row_count:
        line = data[index]

        # Handle a description Acrobat split as "10" / "MHz+ Input".
        if re.fullmatch(r"\d+(?:\.\d+)?", line) and index + 1 < len(data):
            nxt = normalize_pdf_line(data[index + 1])
            first = nxt.split()[0].rstrip(",;:") if nxt else ""
            if ENGINEERING_DESCRIPTION_UNIT_RE.fullmatch(first):
                line = normalize_pdf_line(line + " " + nxt)
                index += 2
            else:
                index += 1
        else:
            index += 1

        description, figure, logic = split_pin_description_figure_logic_payload(line)

        # A pure Figure/Logic cell means Description collection ended too early;
        # keep it for the trailing columns and stop rather than contaminating col 2.
        if description == "" and (figure or logic):
            if figure:
                pending_trailing.append(figure)
            if logic:
                pending_trailing.append(logic)
            break

        # A pure trailing cell like '-' or 'LVDS' should not become Description.
        pure_cells = expand_figure_logic_trailing_cells(line)
        if pure_cells and (line == "-" or is_logic_cell_text(line) or line.upper().startswith("FIGURE ")):
            pending_trailing.extend(pure_cells)
            break

        if description:
            descriptions.append(description)
            if figure:
                pending_trailing.append(figure)
            if logic:
                pending_trailing.append(logic)
        else:
            descriptions.append(line)

    if len(descriptions) != row_count:
        return []

    trailing_cells = list(pending_trailing)
    while index < len(data):
        line = data[index]
        cells = expand_figure_logic_trailing_cells(line)
        if not cells:
            # Do not accept a candidate that would have to concatenate unknown
            # text into Description/Figure/Logic.
            return []
        trailing_cells.extend(cells)
        index += 1

    # We need exactly N Figure cells followed by N Logic cells.  Extra page text
    # invalidates this conservative fallback instead of shifting columns.
    if len(trailing_cells) < row_count * 2:
        return []

    trailing_cells = trailing_cells[:row_count * 2]
    figures = [normalize_dash_cell(v) for v in trailing_cells[:row_count]]
    logics = [normalize_dash_cell(v) for v in trailing_cells[row_count:row_count * 2]]

    if not all(is_figure_cell_text(v) for v in figures):
        return []
    if not all(is_logic_cell_text(v) for v in logics):
        return []

    return [[pins[i], descriptions[i], figures[i], logics[i]] for i in range(row_count)]


def figure_logic_candidate_score(rows):
    if not rows:
        return (0, 0, 0, 0, 0)

    complete = sum(
        1 for row in rows
        if len(row) >= 4
        and str(row[1]).strip() != ""
        and is_figure_cell_text(row[2])
        and is_logic_cell_text(row[3])
    )
    incomplete = len(rows) - complete
    descriptions = sum(1 for row in rows if len(row) >= 2 and str(row[1]).strip() != "")
    figures = sum(1 for row in rows if len(row) >= 3 and is_figure_cell_text(row[2]))
    logics = sum(1 for row in rows if len(row) >= 4 and is_logic_cell_text(row[3]))

    # Prefer candidates that reconstruct COMPLETE rows.  This prevents a parser
    # that invented an extra numeric pin from beating a shorter but correct
    # candidate merely because it returned more rows.
    return (complete, -incomplete, descriptions, min(figures, logics), len(rows))


def is_valid_figure_logic_rows(rows):
    if len(rows) < 2:
        return False

    complete = sum(
        1 for row in rows
        if len(row) >= 4
        and str(row[1]).strip() != ""
        and is_figure_cell_text(row[2])
        and is_logic_cell_text(row[3])
    )
    minimum_complete = max(2, (len(rows) + 1) // 2)
    return complete >= minimum_complete


def looks_like_pdf_pin_description_figure_logic_table(input_text):
    lines, header_info = get_figure_logic_header_window(input_text)
    if header_info is None:
        return False

    # Use line starts, not every numeric token, because Description values can
    # legitimately begin with numbers (10 MHz+ Input / 10 MHz- Input).
    candidates = 0
    for line in lines[header_info["first"] + 1:]:
        if re.match(r"^\d+(?:\s|$)", line):
            candidates += 1

    return candidates >= 2


def parse_pdf_pin_description_figure_logic_table(input_text):
    lines, header_info = get_figure_logic_header_window(input_text)
    rows = [list(FIGURE_LOGIC_PINOUT_HEADERS)]

    if header_info is None:
        return rows

    normal_rows = parse_normal_pin_description_figure_logic_rows(lines[header_info["first"]:])
    stream_rows = parse_stream_pin_description_figure_logic_rows(lines, header_info)
    column_major_rows = parse_column_major_pin_description_figure_logic_rows(lines, header_info)

    candidates = [
        candidate
        # Column-major reconstruction is deliberately ahead of the generic
        # stream candidate.  When both are equally complete, the column-major
        # parser has stronger structural evidence and keeps dash cells in their
        # real Figure/Logic columns.
        for candidate in (normal_rows, column_major_rows, stream_rows)
        if candidate
    ]
    if not candidates:
        return rows

    selected_rows = max(candidates, key=figure_logic_candidate_score)
    rows.extend(selected_rows)
    return rows


def parse_headerless_pin_description_figure_logic_rows(input_text):
    """Conservative body-only recovery for rows such as '1 Chassis GND - -'."""
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 2:
        return []

    # A real header belongs to the normal parser.
    _, header_info = get_figure_logic_header_window(input_text)
    if header_info is not None:
        return []

    normal_rows = parse_normal_pin_description_figure_logic_rows(lines)
    if is_valid_figure_logic_rows(normal_rows):
        return normal_rows

    # Headerless interleaved recovery starts at the first exact pin line.  It is
    # intentionally conservative so this path does not steal unrelated text.
    first_pin_index = None
    for index, line in enumerate(lines):
        if re.fullmatch(r"\d+", line):
            if next_line_continues_numeric_description(lines, index):
                continue
            first_pin_index = index
            break

    if first_pin_index is None:
        return []

    stream_rows = parse_stream_pin_description_figure_logic_rows(
        lines,
        {"first": first_pin_index, "last": first_pin_index}
    )
    if is_valid_figure_logic_rows(stream_rows):
        return stream_rows

    return []


def looks_like_headerless_pin_description_figure_logic_table(input_text):
    return len(parse_headerless_pin_description_figure_logic_rows(input_text)) >= 2


def parse_headerless_pin_description_figure_logic_table(input_text):
    rows = [list(FIGURE_LOGIC_PINOUT_HEADERS)]
    rows.extend(parse_headerless_pin_description_figure_logic_rows(input_text))
    return rows


# =========================================
# PDF 3-COLUMN PIN / DESCRIPTION / FIGURE TABLE
#
# Handles tables such as:
#   Pin Number | Description | Figure
#
# Some tagged PDFs copy this table in a staggered order.  A typical clipboard
# stream looks like:
#
#   Pin
#   Number
#   Description
#   1
#   Figure
#   19 -
#   Chassis GND
#   36 Figure 5, Figure 8
#   TEMP (Thermistor)
#
# In that form the NEXT pin is emitted on the same line as the PREVIOUS row's
# Figure cell, and the previous row's Description follows on the next line(s).
# The parser below reconstructs the rows instead of treating every newline as a
# new spreadsheet row.
# =========================================

FIGURE_PINOUT_HEADERS = [
    "PIN NUMBER",
    "DESCRIPTION",
    "FIGURE"
]


def is_figure_cell_text(value):
    text = normalize_pdf_line(value)
    if text == "":
        return False

    # The Figure column commonly uses a plain dash for "no figure".  Some PDFs
    # copy that glyph as an en/em dash, so accept those forms too.
    if text in {"-", "–", "—"}:
        return True

    return re.match(r"^FIGURE\b", text, re.IGNORECASE) is not None


def is_figure_table_header_fragment(value):
    upper = normalize_pdf_line(value).upper()
    if upper in {"PIN", "NUMBER", "PIN NUMBER", "DESCRIPTION", "FIGURE"}:
        return True
    if upper.startswith("TABLE ") and ("PIN" in upper or "CONNECTOR" in upper):
        return True
    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True
    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True
    return False


def get_figure_pinout_header_window(input_text):
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 5:
        return lines, None

    search_limit = min(len(lines), 30)
    upper_lines = [line.upper() for line in lines[:search_limit]]

    pin_seen = False
    number_seen = False
    description_seen = False
    figure_seen = False
    first_header_index = None
    last_header_index = None

    for index, upper in enumerate(upper_lines):
        compact = re.sub(r"\s+", " ", upper).strip()

        if compact == "PIN" or compact.startswith("PIN NUMBER") or "PIN NUMBER" in compact:
            pin_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if compact == "NUMBER" or "PIN NUMBER" in compact:
            number_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if re.search(r"\bDESCRIPTION\b", compact):
            description_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        if re.search(r"\bFIGURE\b", compact):
            figure_seen = True
            first_header_index = index if first_header_index is None else first_header_index
            last_header_index = index

        # Do not scan forever into the data.  Once every required header has
        # appeared, the window is known even if Figure was emitted after pin 1.
        if pin_seen and number_seen and description_seen and figure_seen:
            break

    if not (pin_seen and number_seen and description_seen and figure_seen):
        return lines, None

    return lines, {
        "first": first_header_index,
        "last": last_header_index
    }


def split_pin_description_figure_payload(payload):
    payload = normalize_pdf_line(payload)
    if payload == "":
        return "", ""

    figure_match = re.search(r"\bFIGURE\b", payload, re.IGNORECASE)
    if figure_match is not None:
        description = payload[:figure_match.start()].strip()
        figure = payload[figure_match.start():].strip()
        return description, figure

    # A dash is a legitimate Figure-column value in this table family.  Accept
    # the common Unicode dash variants Acrobat may put on the clipboard.
    dash_match = re.search(r"(?:^|\s)[-–—](?:\s*)$", payload)
    if dash_match is not None:
        description = payload[:dash_match.start()].strip()
        return description, "-"

    return payload, ""


def parse_normal_pin_description_figure_rows(lines):
    rows = []

    for line in lines:
        if is_figure_table_header_fragment(line):
            continue

        # A Description such as "10 MHz+ Input Figure 10" must not create
        # phantom pin 10 or lose the leading 10 from Column 2.
        if looks_like_numeric_leading_description(line):
            continue

        match = re.match(r"^(\d+)\s+(.+)$", line)
        if match is None:
            continue

        pin = match.group(1)
        payload = match.group(2).strip()
        description, figure = split_pin_description_figure_payload(payload)

        # A line such as "19 -" or "36 Figure 5, Figure 8" is the staggered
        # form's transition marker, not a complete normal row.
        if description == "":
            continue

        rows.append([pin, description, figure])

    return rows


def get_staggered_figure_markers(lines, start_index):
    markers = []

    for index in range(start_index, len(lines)):
        line = lines[index]

        if is_figure_table_header_fragment(line):
            continue

        exact_pin = re.fullmatch(r"\d+", line)
        if exact_pin is not None:
            markers.append({"index": index, "pin": exact_pin.group(0), "figure": ""})
            continue

        match = re.match(r"^(\d+)\s+(.+)$", line)
        if match is None:
            continue

        remainder = normalize_pdf_line(match.group(2))
        if is_figure_cell_text(remainder):
            markers.append({
                "index": index,
                "pin": match.group(1),
                "figure": remainder
            })

    return markers


def clean_figure_description_lines(values):
    cleaned = []
    for value in values:
        text = normalize_pdf_line(value)
        if text == "" or is_figure_table_header_fragment(text):
            continue
        cleaned.append(text)
    return " ".join(cleaned).strip()


def extract_figure_cells_from_text(value):
    """
    Split one clipboard line into Figure-column cells without confusing a
    multi-reference cell such as "Figure 5, Figure 8" with two rows.

    Also handles PDFs that collapse many blank Figure cells onto one line:
        "- - - -" -> ["-", "-", "-", "-"]
    """
    text = normalize_pdf_line(value)
    if text == "":
        return []

    # One or more dash-only Figure cells.  This is important for continuation
    # tables like the Chassis GND block where every Figure cell is '-'.
    dash_tokens = re.findall(r"(?<!\S)[-–—](?!\S)", text)
    if dash_tokens:
        remainder = re.sub(r"(?<!\S)[-–—](?!\S)", " ", text)
        remainder = re.sub(r"\s+", "", remainder)
        if remainder == "":
            return ["-"] * len(dash_tokens)

    # A comma joins references that belong to ONE Figure cell.  A new Figure
    # token not introduced by a comma starts the next cell.
    pattern = re.compile(
        r"Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*",
        re.IGNORECASE
    )
    matches = [normalize_pdf_line(match.group(0)) for match in pattern.finditer(text)]
    return matches


def description_start_positions(text):
    """
    Return strong candidate starts of a second Description cell on a line.

    Some tagged PDFs collapse TWO description cells onto one clipboard line.
    We only split at signal-like all-caps identifiers, and only use this as a
    recovery step when there are fewer description cells than pin numbers.
    """
    positions = []

    # Signal-like identifiers seen in connector tables.  Require either an
    # underscore-containing identifier or TEMP/TEMP_RTN so ordinary prose is
    # not split merely because it contains a capitalized word.
    pattern = re.compile(
        r"(?<!\S)(?:[A-Z][A-Z0-9]*_[A-Z0-9_+\-./]+|TEMP(?:_RTN)?)(?=\s|$|\()"
    )

    for match in pattern.finditer(text):
        if match.start() > 0:
            positions.append(match.start())

    return positions


def split_repeated_description_cells(value):
    """
    Recover description cells when a PDF merges repeated neighboring cells.

    Example from the continuation table shown by the user:
        "Chassis GND Chassis GND Chassis GND"
    becomes:
        ["Chassis GND", "Chassis GND", "Chassis GND"]

    Only exact repeated token phrases are split, which keeps ordinary prose
    intact.
    """
    text = normalize_pdf_line(value)
    if text == "":
        return []

    tokens = text.split()
    if len(tokens) < 2:
        return [text]

    # Description cells in these pinout tables are normally short.  Limiting
    # the repeated unit to eight tokens keeps this recovery conservative.
    max_unit = min(8, len(tokens) // 2)

    for unit_length in range(1, max_unit + 1):
        if len(tokens) % unit_length != 0:
            continue

        repeat_count = len(tokens) // unit_length
        if repeat_count < 2:
            continue

        unit = tokens[:unit_length]
        if unit * repeat_count == tokens:
            phrase = " ".join(unit)
            return [phrase] * repeat_count

    return [text]


def split_merged_description_cell(value):
    text = normalize_pdf_line(value)
    if text == "":
        return []

    # First handle exact repeated phrases such as a whole run of "Chassis GND"
    # cells merged onto one clipboard line.
    repeated_parts = split_repeated_description_cells(text)
    if len(repeated_parts) > 1:
        return repeated_parts

    positions = description_start_positions(text)
    if not positions:
        return [text]

    # Split at every strong later signal start.  The first chunk remains intact
    # (e.g. "Chassis GND"), then each following signal begins a new cell.
    parts = []
    start = 0
    for position in positions:
        part = text[start:position].strip()
        if part:
            parts.append(part)
        start = position

    tail = text[start:].strip()
    if tail:
        parts.append(tail)

    return parts if parts else [text]


def parse_stream_pin_description_figure_rows(lines, header_info):
    """
    Reconstruct the 3-column table as independent ordered streams.

    Description is split from Figure before row reconstruction so Figure text
    and dash cells cannot leak into Column 2.
    """
    pins = []
    descriptions = []
    figures = []
    start_index = max(0, header_info["first"])

    line_index = start_index
    while line_index < len(lines):
        line = normalize_pdf_line(lines[line_index])
        if line == "" or is_figure_table_header_fragment(line):
            line_index += 1
            continue

        if looks_like_numeric_leading_description(line):
            description, figure = split_pin_description_figure_payload(line)
            if description:
                descriptions.append(description)
            if figure:
                figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
            line_index += 1
            continue

        if next_line_continues_numeric_description(lines, line_index):
            next_text = normalize_pdf_line(lines[line_index + 1])
            combined = normalize_pdf_line(line + " " + next_text)
            description, figure = split_pin_description_figure_payload(combined)
            if description:
                descriptions.append(description)
            if figure:
                figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
            line_index += 2
            continue

        pin_match = re.match(r"^(\d+)(?:\s+(.*))?$", line)
        if pin_match is not None:
            pin = pin_match.group(1)
            remainder = normalize_pdf_line(pin_match.group(2) or "")

            if remainder == "":
                pins.append(pin)
                line_index += 1
                continue

            description, figure = split_pin_description_figure_payload(remainder)
            if figure:
                pins.append(pin)
                if description:
                    descriptions.append(description)
                figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
                line_index += 1
                continue

            descriptions.append(line)
            line_index += 1
            continue

        description, figure = split_pin_description_figure_payload(line)
        if figure:
            if description:
                descriptions.append(description)
            figures.extend(extract_figure_cells_from_text(figure) or [normalize_dash_cell(figure)])
            line_index += 1
            continue

        figure_cells = extract_figure_cells_from_text(line)
        if figure_cells:
            figures.extend(figure_cells)
            line_index += 1
            continue

        if line in {"-", "–", "—"}:
            figures.append("-")
            line_index += 1
            continue

        descriptions.append(line)
        line_index += 1

    if not pins:
        return []

    if len(descriptions) < len(pins):
        expanded = []
        for description in descriptions:
            expanded.extend(split_merged_description_cell(description))
        if len(expanded) >= len(descriptions):
            descriptions = expanded

    descriptions = descriptions[:len(pins)]
    figures = figures[:len(pins)]
    while len(descriptions) < len(pins):
        descriptions.append("")
    while len(figures) < len(pins):
        figures.append("")

    return [[pin, descriptions[i], figures[i]] for i, pin in enumerate(pins)]

def parse_staggered_pin_description_figure_rows(lines, header_info):
    """Legacy adjacency parser retained only as a fallback."""
    start_index = max(0, header_info["first"])
    markers = get_staggered_figure_markers(lines, start_index)

    if len(markers) < 2:
        return []

    rows = []

    for marker_index in range(1, len(markers)):
        previous_marker = markers[marker_index - 1]
        current_marker = markers[marker_index]
        next_line_index = markers[marker_index + 1]["index"] if marker_index + 1 < len(markers) else len(lines)
        description_slice = lines[current_marker["index"] + 1:next_line_index]

        terminal_figure_offset = None
        if marker_index == len(markers) - 1:
            for offset, value in enumerate(description_slice):
                if is_figure_cell_text(value):
                    terminal_figure_offset = offset
                    break

        if terminal_figure_offset is None:
            previous_description_lines = description_slice
        else:
            previous_description_lines = description_slice[:terminal_figure_offset]

        description = clean_figure_description_lines(previous_description_lines)
        figure = current_marker["figure"]

        if description != "":
            rows.append([previous_marker["pin"], description, figure])

        if marker_index == len(markers) - 1 and terminal_figure_offset is not None:
            terminal_figure = normalize_pdf_line(description_slice[terminal_figure_offset])
            terminal_description = clean_figure_description_lines(description_slice[terminal_figure_offset + 1:])
            if terminal_description != "":
                rows.append([current_marker["pin"], terminal_description, terminal_figure])

    return rows


def looks_like_pdf_pin_description_figure_table(input_text):
    lines, header_info = get_figure_pinout_header_window(input_text)
    if header_info is None:
        return False

    # A visible Logic header means this is the 4-column family handled above.
    header_probe = " ".join(lines[:40]).upper()
    if re.search(r"\bLOGIC\b", header_probe):
        return False

    data_candidates = 0
    for line in lines[header_info["first"] + 1:]:
        if re.match(r"^\d+(?:\s|$)", line):
            data_candidates += 1

    return data_candidates >= 2


def parse_compact_pin_description_figure_columns(lines, start_index=0):
    """
    Recover a 3-column pin table when the PDF collapses an entire visual column
    onto one clipboard line, for example:

        47 13 31 48 14 32
        Chassis GND Chassis GND Chassis GND Chassis GND Chassis GND Chassis GND
        - - - - - -

    The visual pin order is preserved exactly; pin numbers do NOT need to be
    ascending or consecutive.
    """
    pins = []
    descriptions = []
    figures = []
    pin_block_found = False

    for raw_line in lines[start_index:]:
        line = normalize_pdf_line(raw_line)
        if line == "" or is_figure_table_header_fragment(line):
            continue

        tokens = line.split()

        # A compact pin-number column: at least two integer cells on one line.
        if not pin_block_found and len(tokens) >= 2 and all(is_integer_value(token) for token in tokens):
            pins.extend(tokens)
            pin_block_found = True
            continue

        if not pin_block_found:
            continue

        figure_cells = extract_figure_cells_from_text(line)
        if figure_cells:
            # Only accept the line as a Figure block when all of its visible
            # content is made from Figure cells/dashes.  This prevents a normal
            # description containing the word "Figure" from being stolen.
            dash_only = re.fullmatch(r"(?:\s*[-–—]\s*)+", line) is not None
            figure_only = re.fullmatch(
                r"\s*Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*(?:\s+Figure\s+\d+(?:\s*,\s*Figure\s+\d+)*)*\s*",
                line,
                re.IGNORECASE
            ) is not None

            if dash_only or figure_only:
                figures.extend(figure_cells)
                continue

        descriptions.extend(split_merged_description_cell(line))

    if len(pins) < 2:
        return []

    descriptions = descriptions[:len(pins)]
    figures = figures[:len(pins)]

    while len(descriptions) < len(pins):
        descriptions.append("")
    while len(figures) < len(pins):
        figures.append("")

    return [
        [pin, descriptions[index], figures[index]]
        for index, pin in enumerate(pins)
    ]


def figure_table_candidate_score(rows):
    """Score a possible 3-column Pin / Description / Figure reconstruction."""
    if not rows:
        return (0, 0, 0)

    description_count = sum(
        1 for row in rows
        if len(row) >= 2 and str(row[1]).strip() != ""
    )
    figure_count = sum(
        1 for row in rows
        if len(row) >= 3 and is_figure_cell_text(row[2])
    )

    return (len(rows), description_count, figure_count)


def is_valid_headerless_figure_table_rows(rows):
    """
    Conservative validation for a headerless continuation table.

    Requiring real Figure-column markers is what keeps an arbitrary list of
    numbers and text from being misdetected as this special 3-column format.
    """
    if len(rows) < 3:
        return False

    description_count = sum(
        1 for row in rows
        if len(row) >= 2 and str(row[1]).strip() != ""
    )
    figure_count = sum(
        1 for row in rows
        if len(row) >= 3 and is_figure_cell_text(row[2])
    )

    minimum_descriptions = max(2, (len(rows) + 1) // 2)
    minimum_figures = max(2, (len(rows) + 1) // 2)

    return (
        description_count >= minimum_descriptions
        and figure_count >= minimum_figures
    )


def parse_headerless_pin_description_figure_rows(input_text):
    """
    Parse a continuation/body-only Pin Number | Description | Figure table.

    This is specifically for selections like the screenshot where the visible
    rows are only:
        47   Chassis GND   -
        13   Chassis GND   -
        31   Chassis GND   -
        ...
    and the Pin Number / Description / Figure header is not part of the copy.
    """
    lines = get_nonempty_normalized_lines(input_text)
    if len(lines) < 3:
        return []

    # If the real header is present, the normal parser should own this input.
    _, header_info = get_figure_pinout_header_window(input_text)
    if header_info is not None:
        return []

    # Do not steal the 4-column Figure/Logic body-only family.
    if looks_like_headerless_pin_description_figure_logic_table(input_text):
        return []

    # Do not steal the other known pinout families when their headers are clear.
    header_probe = " ".join(lines[:20]).upper()
    if "SIGNAL NAME" in header_probe and (
        "SIGNAL TYPE" in header_probe
        or "DIRECTION" in header_probe
    ):
        return []

    candidates = []

    # 1) Normal row-wise copy: 47 Chassis GND -
    normal_rows = parse_normal_pin_description_figure_rows(lines)
    if is_valid_headerless_figure_table_rows(normal_rows):
        candidates.append(normal_rows)

    # 2) Tagged/interleaved/column-major copy.  Start at the first numeric row so
    # page titles before the continuation block cannot shift the descriptions.
    first_pin_index = None
    for index, line in enumerate(lines):
        if re.match(r"^\d+(?:\s|$)", line):
            first_pin_index = index
            break

    if first_pin_index is not None:
        synthetic_header = {"first": first_pin_index, "last": first_pin_index}
        stream_rows = parse_stream_pin_description_figure_rows(lines, synthetic_header)
        if is_valid_headerless_figure_table_rows(stream_rows):
            candidates.append(stream_rows)

        # 3) Entire visual columns collapsed onto single lines.
        compact_rows = parse_compact_pin_description_figure_columns(lines, first_pin_index)
        if is_valid_headerless_figure_table_rows(compact_rows):
            candidates.append(compact_rows)

    if not candidates:
        return []

    # Prefer the candidate with the most complete Figure cells when row counts
    # are equal; those markers are the strongest evidence of correct alignment.
    return max(
        candidates,
        key=lambda rows: (
            figure_table_candidate_score(rows)[0],
            figure_table_candidate_score(rows)[2],
            figure_table_candidate_score(rows)[1]
        )
    )


def looks_like_headerless_pin_description_figure_table(input_text):
    return len(parse_headerless_pin_description_figure_rows(input_text)) >= 3


def parse_headerless_pin_description_figure_table(input_text):
    rows = [list(FIGURE_PINOUT_HEADERS)]
    rows.extend(parse_headerless_pin_description_figure_rows(input_text))
    return rows


def parse_pdf_pin_description_figure_table(input_text):
    lines, header_info = get_figure_pinout_header_window(input_text)
    rows = [list(FIGURE_PINOUT_HEADERS)]

    if header_info is None:
        return rows

    normal_rows = parse_normal_pin_description_figure_rows(lines[header_info["first"]:])
    stream_rows = parse_stream_pin_description_figure_rows(lines, header_info)
    legacy_rows = parse_staggered_pin_description_figure_rows(lines, header_info)
    compact_rows = parse_compact_pin_description_figure_columns(lines, header_info["first"])

    # Prefer stream reconstruction whenever it found the same or more rows than
    # the alternatives.  It is specifically designed for PDFs whose clipboard
    # tags interleave the three physical columns.
    candidates = [normal_rows, legacy_rows, stream_rows, compact_rows]
    selected_rows = max(
        candidates,
        key=lambda candidate: (
            len(candidate),
            sum(1 for row in candidate if len(row) >= 2 and str(row[1]).strip() != ""),
            1 if candidate is stream_rows else 0
        )
    )

    rows.extend(selected_rows)
    return rows


# =========================================
# PDF 4-COLUMN POWER / SIMPLE PINOUT TABLE DETECTION
#
# Handles copied PDF tables such as:
#
# TABLE 5 - POWER CONNECTOR PINOUT
# Pin #   Signal Name   Signal Type   Signal Description
# 1       VBUS          Power         22-36Vdc Power Input
# 2       VBUS          Power         22-36Vdc Power Input
# 3       INHIBIT       Power         Enable
# 4       BUS_RTN       Power         Board Ground
#
# This table family has NO Ref Des column and NO Direction column.  It is
# intentionally detected before the 6-column pinout parser so the words
# "Signal Type" / "Signal Description" cannot be mistaken for that format.
# =========================================

SIMPLE_PINOUT_HEADERS = [
    "PIN #",
    "SIGNAL NAME",
    "SIGNAL TYPE",
    "SIGNAL DESCRIPTION"
]


def get_flat_pdf_tokens(input_text):
    """Return a whitespace token stream regardless of how Acrobat inserted newlines."""
    tokens = []

    for raw_line in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        line = normalize_pdf_line(raw_line)

        if line == "":
            continue

        tokens.extend(line.split())

    return tokens


def normalized_header_token(value):
    value = str(value or "").strip().upper()
    value = value.replace("–", "-").replace("—", "-")
    return value.strip(":")


def find_token_pair(tokens, first, second, start=0, stop=None):
    upper_tokens = [normalized_header_token(token) for token in tokens]
    if stop is None:
        stop = len(tokens)

    for index in range(max(0, start), min(stop, len(tokens) - 1)):
        if upper_tokens[index] == first and upper_tokens[index + 1] == second:
            return index

    return None


def find_simple_pinout_header_end(tokens):
    """
    Find the end of:
        Pin # | Signal Name | Signal Type | Signal Description

    This works whether those headers arrive as four normal lines, individual
    cells on separate lines, or literally one word per line.
    """
    upper_tokens = [normalized_header_token(token) for token in tokens]
    search_stop = min(len(tokens), 120)

    for description_index in range(search_stop - 1):
        if not (
            upper_tokens[description_index] == "SIGNAL"
            and upper_tokens[description_index + 1] == "DESCRIPTION"
        ):
            continue

        before = upper_tokens[max(0, description_index - 20):description_index]
        before_text = " ".join(before)

        if "SIGNAL NAME" not in before_text:
            continue

        if "SIGNAL TYPE" not in before_text:
            continue

        if "PIN" not in before:
            continue

        return description_index + 2

    return None


def find_connector_pinout_header_end(tokens):
    """
    Find the end of:
        Ref Des | Pin | Signal Name | Direction | Type | Description

    The header can be split at arbitrary newline boundaries.
    """
    upper_tokens = [normalized_header_token(token) for token in tokens]
    search_stop = min(len(tokens), 140)

    for description_index in range(search_stop):
        if upper_tokens[description_index] != "DESCRIPTION":
            continue

        before = upper_tokens[max(0, description_index - 24):description_index]
        before_text = " ".join(before)

        if "SIGNAL NAME" not in before_text:
            continue

        if "DIRECTION" not in before:
            continue

        if "TYPE" not in before:
            continue

        if "PIN" not in before:
            continue

        if "REF" not in before or "DES" not in before:
            continue

        # A 4-column table has the phrase SIGNAL TYPE.  Do not steal it.
        if "SIGNAL TYPE" in before_text:
            continue

        return description_index + 1

    return None


def looks_highly_verticalized(input_text):
    """Heuristic used only for the status text, not for deciding correctness."""
    lines = get_nonempty_normalized_lines(input_text)

    if len(lines) < 6:
        return False

    short_lines = sum(1 for line in lines if len(line.split()) <= 1)
    return short_lines / len(lines) >= 0.55


def select_pin_token_indices(tokens, data_start, minimum_payload_tokens):
    """
    Pick the most plausible increasing pin-number sequence from a flattened PDF
    token stream.  This prevents an isolated numeric value in a description from
    normally being mistaken for a new pin row.
    """
    candidates = []

    for index in range(data_start, len(tokens)):
        value = tokens[index]

        if not is_integer_value(value):
            continue

        candidates.append((index, int(value)))

    if not candidates:
        return []

    # DP state: best path ending at each candidate.  Prefer more rows first,
    # then consecutive pin numbering, then a smaller first pin number.
    states = []

    for candidate_index, (token_index, pin_number) in enumerate(candidates):
        best_path = [candidate_index]
        best_count = 1
        best_consecutive = 0
        best_first_pin = pin_number

        for previous_candidate_index in range(candidate_index):
            previous_token_index, previous_pin_number = candidates[previous_candidate_index]

            if pin_number <= previous_pin_number:
                continue

            payload_end = token_index

            # If a connector designator sits immediately before this pin, it
            # belongs to this row, not the previous row's Description.
            if token_index - 1 > previous_token_index:
                possible_ref = tokens[token_index - 1]
                if is_pinout_ref_designator(possible_ref):
                    payload_end -= 1

            payload_count = payload_end - previous_token_index - 1

            if payload_count < minimum_payload_tokens:
                continue

            previous_state = states[previous_candidate_index]
            path = previous_state["path"] + [candidate_index]
            count = previous_state["count"] + 1
            consecutive = previous_state["consecutive"] + (1 if pin_number == previous_pin_number + 1 else 0)
            first_pin = previous_state["first_pin"]

            score = (count, consecutive, 1 if first_pin == 1 else 0, -first_pin)
            best_score = (best_count, best_consecutive, 1 if best_first_pin == 1 else 0, -best_first_pin)

            if score > best_score:
                best_path = path
                best_count = count
                best_consecutive = consecutive
                best_first_pin = first_pin

        states.append({
            "path": best_path,
            "count": best_count,
            "consecutive": best_consecutive,
            "first_pin": best_first_pin
        })

    valid_states = []

    for state in states:
        last_candidate_index = state["path"][-1]
        last_token_index = candidates[last_candidate_index][0]
        remaining_payload = len(tokens) - last_token_index - 1

        if remaining_payload >= minimum_payload_tokens:
            valid_states.append(state)

    if not valid_states:
        valid_states = states

    valid_states.sort(
        key=lambda state: (
            state["count"],
            state["consecutive"],
            1 if state["first_pin"] == 1 else 0,
            -state["first_pin"]
        ),
        reverse=True
    )

    selected = valid_states[0]["path"]
    return [candidates[index][0] for index in selected]


def looks_like_pdf_simple_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_simple_pinout_header_end(tokens)

    if header_end is None:
        return False

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=2)
    return len(pin_indices) >= 2


def is_simple_pinout_header_line(line):
    upper_text = normalize_pdf_line(line).upper()

    if upper_text == "":
        return True

    if upper_text in {
        "PIN",
        "PIN #",
        "SIGNAL NAME",
        "SIGNAL TYPE",
        "SIGNAL DESCRIPTION"
    }:
        return True

    if (
        "PIN" in upper_text
        and "SIGNAL NAME" in upper_text
        and "SIGNAL TYPE" in upper_text
        and "SIGNAL DESCRIPTION" in upper_text
    ):
        return True

    if upper_text.startswith("TABLE ") and "PINOUT" in upper_text:
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    return False


SIGNAL_START_JOIN_CHARS = "_-/#+.:"
SIGNAL_END_JOIN_CHARS = "_/#.:"
SIGNAL_STANDALONE_JOIN_CHARS = "_-/#+.:"


def token_starts_with_signal_joiner(value):
    text = str(value or "").strip()
    return text != "" and text[0] in SIGNAL_START_JOIN_CHARS


def token_ends_with_signal_joiner(value):
    text = str(value or "").strip()
    # Do not treat a trailing + or - as proof that the next word belongs to the
    # signal.  Differential signals commonly end in + / -, e.g. D4- Input.
    return text != "" and text[-1] in SIGNAL_END_JOIN_CHARS


def token_is_signal_joiner(value):
    text = str(value or "").strip()
    return text != "" and all(char in SIGNAL_STANDALONE_JOIN_CHARS for char in text)


def signal_tokens_are_continuous(previous_value, next_value):
    """
    Acrobat occasionally breaks an identifier at punctuation, for example:

        BUS_  RTN
        BUS   _RTN
        BUS   _   RTN
        P5_   1000BT_D4-

    Those are one Signal Name, not extra PDF columns.  Only fuse tokens when
    punctuation gives us strong evidence that the identifier continues.  This
    avoids swallowing an ordinary Signal Type such as "Power".
    """
    previous_text = str(previous_value or "").strip()
    next_text = str(next_value or "").strip()

    if previous_text == "" or next_text == "":
        return False

    if token_ends_with_signal_joiner(previous_text):
        return True

    if token_starts_with_signal_joiner(next_text):
        return True

    if token_is_signal_joiner(previous_text) or token_is_signal_joiner(next_text):
        return True

    return False


def normalize_signal_name_fragments(fragments):
    fragments = [str(value or "").strip() for value in fragments if str(value or "").strip() != ""]

    if not fragments:
        return ""

    # Preserve a real two-word mnemonic that occurs in the older table family.
    if len(fragments) == 2 and fragments[0].upper() == "CTU" and fragments[1].upper() == "RTN":
        return "CTU RTN"

    text = " ".join(fragments)

    # Rejoin identifiers that Acrobat split immediately before/after connector
    # punctuation.  Repeating the substitution handles forms such as
    # "BUS _ RTN" as well as "P5_ 1000BT_D4-".
    previous = None
    while previous != text:
        previous = text
        text = re.sub(r"\s*([_\-/#+.:])\s*", r"\1", text)

    return text.strip()


def consume_signal_name_tokens(payload):
    """Return (signal_name, number_of_payload_tokens_consumed)."""
    if not payload:
        return "", 0

    fragments = [payload[0]]
    index = 1

    while index < len(payload):
        current_signal = normalize_signal_name_fragments(fragments)
        next_value = payload[index]

        if signal_tokens_are_continuous(current_signal, next_value):
            fragments.append(next_value)
            index += 1
            continue

        # Existing PDFs also contain the legitimate two-word signal CTU RTN.
        if current_signal.upper() == "CTU" and str(next_value).strip().upper() == "RTN":
            fragments.append(next_value)
            index += 1
            continue

        break

    return normalize_signal_name_fragments(fragments), index


def parse_simple_pinout_data_line(line):
    line = normalize_pdf_line(line)

    if line == "":
        return None

    tokens = line.split()

    if len(tokens) < 3:
        return None

    if not is_integer_value(tokens[0]):
        return None

    pin = tokens.pop(0)
    signal_name, consumed = consume_signal_name_tokens(tokens)

    if consumed >= len(tokens):
        return None

    signal_type = tokens[consumed]
    description = " ".join(tokens[consumed + 1:])

    return [pin, signal_name, signal_type, description]


def parse_pdf_simple_pinout_table(input_text):
    # Flattening first is intentional.  The same parser therefore works when
    # Acrobat gives us normal rows, one cell per line, or one word per line.
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_simple_pinout_header_end(tokens)
    rows = [list(SIMPLE_PINOUT_HEADERS)]

    if header_end is None:
        return rows

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=2)

    for row_index, pin_index in enumerate(pin_indices):
        next_pin_index = pin_indices[row_index + 1] if row_index + 1 < len(pin_indices) else len(tokens)
        payload_end = next_pin_index

        payload = tokens[pin_index + 1:payload_end]

        if len(payload) < 2:
            continue

        pin = tokens[pin_index]
        signal_name, consumed = consume_signal_name_tokens(payload)

        if consumed >= len(payload):
            continue

        signal_type = payload[consumed]
        description = " ".join(payload[consumed + 1:])

        rows.append([
            pin,
            signal_name,
            signal_type,
            description
        ])

    return rows


def looks_like_pdf_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_connector_pinout_header_end(tokens)

    if header_end is None:
        return False

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=3)
    return len(pin_indices) >= 2


def is_pinout_header_line(line):
    upper_text = normalize_pdf_line(line).upper()

    if upper_text == "":
        return True

    if upper_text in {
        "REF",
        "DES",
        "REF DES",
        "PIN",
        "SIGNAL NAME",
        "DIRECTION",
        "TYPE",
        "DESCRIPTION"
    }:
        return True

    if (
        "PIN" in upper_text
        and "SIGNAL NAME" in upper_text
        and "DIRECTION" in upper_text
        and "TYPE" in upper_text
        and "DESCRIPTION" in upper_text
    ):
        return True

    if upper_text.startswith("TABLE ") and "PINOUT" in upper_text:
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    return False




def is_pinout_ref_designator(value):
    """Stricter Ref Des test so signal names like P5_1000BT_D4- are not connectors."""
    text = normalize_pdf_line(value).upper().replace(" ", "")

    if text == "":
        return False

    # Normal connector names used by these pinout tables, including a compact
    # shared label such as J1R/J2R.
    return re.fullmatch(r"J\d+[A-Z]*(?:[/,&+]J\d+[A-Z]*)*", text) is not None

def normalize_pinout_direction(value):
    upper = str(value or "").strip().upper().replace("\\", "/")

    mapping = {
        "INPUT": "Input",
        "IN": "In",
        "OUTPUT": "Output",
        "OUT": "Out",
        "INOUT": "Inout",
        "IN/OUT": "In/Out",
        "BIDIR": "Bidir",
        "BIDIRECTIONAL": "Bidirectional",
        "NC": "NC",
        "N/C": "N/C",
        "-": "-"
    }

    return mapping.get(upper, str(value or "").strip())


def find_direction_token_index(payload):
    direction_values = {
        "INPUT",
        "IN",
        "OUTPUT",
        "OUT",
        "INOUT",
        "IN/OUT",
        "BIDIR",
        "BIDIRECTIONAL",
        "NC",
        "N/C",
        "-"
    }

    # Start at 1 so a signal named NC is not mistaken for the direction when
    # the row is "NC NC NC No Connect".
    for index in range(1, len(payload)):
        upper = str(payload[index]).strip().upper().replace("\\", "/")

        if upper not in direction_values:
            continue

        previous_value = payload[index - 1] if index > 0 else ""
        next_value = payload[index + 1] if index + 1 < len(payload) else ""

        # Do not mistake IN / OUT for the Direction column when Acrobat split a
        # signal identifier around an underscore, e.g. DIG_ / IN / _1.
        if signal_tokens_are_continuous(previous_value, payload[index]):
            continue

        if signal_tokens_are_continuous(payload[index], next_value):
            continue

        # A standalone '-' can itself be part of a signal name.  If a stronger
        # direction word immediately follows, keep '-' with the signal and use
        # the later token as Direction.
        if upper == "-":
            for lookahead in range(index + 1, min(index + 3, len(payload))):
                later = str(payload[lookahead]).strip().upper().replace("\\", "/")
                if later in direction_values and later != "-":
                    break
            else:
                return index
            continue

        return index

    return None


def parse_pinout_data_line(line, current_ref_des):
    line = normalize_pdf_line(line)

    if line == "":
        return None

    tokens = line.split()

    if len(tokens) < 4:
        return None

    ref_des = current_ref_des

    if (
        len(tokens) >= 2
        and is_connector_designator(tokens[0])
        and is_integer_value(tokens[1])
    ):
        ref_des = tokens.pop(0)

    if len(tokens) == 0 or not is_integer_value(tokens[0]):
        return None

    pin = tokens.pop(0)
    direction_index = find_direction_token_index(tokens)

    if direction_index is not None and direction_index >= 1 and direction_index + 1 < len(tokens):
        signal_name = normalize_signal_name_fragments(tokens[:direction_index])
        direction = normalize_pinout_direction(tokens[direction_index])
        signal_type = tokens[direction_index + 1]
        description = " ".join(tokens[direction_index + 2:])
    else:
        if len(tokens) < 3:
            return None

        signal_name, consumed = consume_signal_name_tokens(tokens)

        if consumed + 1 >= len(tokens):
            return None

        direction = normalize_pinout_direction(tokens[consumed])
        signal_type = tokens[consumed + 1]
        description = " ".join(tokens[consumed + 2:])

    return [ref_des, pin, signal_name, direction, signal_type, description]


def parse_pdf_pinout_table(input_text):
    tokens = get_flat_pdf_tokens(input_text)
    header_end = find_connector_pinout_header_end(tokens)
    rows = [list(PINOUT_HEADERS)]

    if header_end is None:
        return rows

    pin_indices = select_pin_token_indices(tokens, header_end, minimum_payload_tokens=3)
    current_ref_des = ""

    for row_index, pin_index in enumerate(pin_indices):
        next_pin_index = pin_indices[row_index + 1] if row_index + 1 < len(pin_indices) else len(tokens)

        # Ref Des can be a vertically merged value once before the first pin,
        # or it can be repeated immediately before each pin.
        if row_index == 0:
            # Only search the small pre-data area before the first pin.  Do not
            # scan the previous row payload for later pins, because signal names
            # such as P5_1000BT_D4- also contain letters and digits.
            for token in tokens[header_end:pin_index]:
                if is_pinout_ref_designator(token):
                    current_ref_des = token

        if pin_index > header_end and is_pinout_ref_designator(tokens[pin_index - 1]):
            current_ref_des = tokens[pin_index - 1]

        payload_end = next_pin_index

        if row_index + 1 < len(pin_indices):
            # A repeated connector name immediately before the next pin belongs
            # to the next row, not this row's Description.
            if payload_end - 1 > pin_index and is_pinout_ref_designator(tokens[payload_end - 1]):
                payload_end -= 1

        payload = tokens[pin_index + 1:payload_end]

        if len(payload) < 3:
            continue

        direction_index = find_direction_token_index(payload)

        if direction_index is not None and direction_index >= 1 and direction_index + 1 < len(payload):
            signal_name = normalize_signal_name_fragments(payload[:direction_index])
            direction = normalize_pinout_direction(payload[direction_index])
            signal_type = payload[direction_index + 1]
            description = " ".join(payload[direction_index + 2:])
        else:
            signal_name, consumed = consume_signal_name_tokens(payload)

            if consumed + 1 >= len(payload):
                continue

            direction = normalize_pinout_direction(payload[consumed])
            signal_type = payload[consumed + 1]
            description = " ".join(payload[consumed + 2:])

        rows.append([
            current_ref_des,
            tokens[pin_index],
            signal_name,
            direction,
            signal_type,
            description
        ])

    return rows


# =========================================
# SAFE CSV DETECTION
#
# Do NOT treat a block as CSV merely because a description contains a comma.
# Real CSV should show comma-separated structure on the first meaningful rows.
# =========================================

def looks_like_real_csv(input_text):
    raw_lines = [line for line in input_text.split("\n") if line.strip() != ""]

    if len(raw_lines) == 0:
        return False

    sample_lines = raw_lines[:6]
    comma_sample_count = sum(1 for line in sample_lines if "," in line)

    # A normal CSV header/data block will contain commas immediately. A PDF
    # table whose descriptions merely contain punctuation usually will not.
    if comma_sample_count < min(2, len(sample_lines)):
        return False

    widths = []

    for line in raw_lines:
        try:
            parsed = next(csv.reader([line]))
        except csv.Error:
            continue

        if len(parsed) > 1:
            widths.append(len(parsed))

    if len(widths) < 2:
        return False

    # Require a repeated multi-column width. This rejects isolated prose commas.
    width_counts = {}

    for width in widths:
        width_counts[width] = width_counts.get(width, 0) + 1

    most_common_width = max(width_counts, key=width_counts.get)
    most_common_count = width_counts[most_common_width]

    return most_common_width > 1 and most_common_count >= 2

# PDF WIRING TABLE DETECTION
# =========================================

def is_connector_designator(value):
    value = normalize_pdf_line(value)

    if value == "":
        return False

    if " " in value:
        return False

    if re.fullmatch(r"[A-Za-z][A-Za-z0-9_.#/-]*", value) is None:
        return False

    if re.search(r"[A-Za-z]", value) is None:
        return False

    if re.search(r"\d", value) is None:
        return False

    return True


def looks_like_inline_wiring_row(line):
    line = normalize_pdf_line(line)

    if line == "":
        return False

    match = re.match(
        r"^([A-Za-z][A-Za-z0-9_.#/-]*)\s+(\d+)\s+(\S+)(?:\s+.*)?$",
        line
    )

    if match is None:
        return False

    return is_connector_designator(match.group(1))


def is_pdf_junk_line(line):
    text = normalize_pdf_line(line)
    upper_text = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    exact_junk = {
        "FROM",
        "TO",
        "SIGNAL NAME",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE",
        "CODE",
        "WIRE CODE",
        "NOTE",
        "COMMENT",
        "WIRE CODES"
    }

    if upper_text in exact_junk:
        return True

    if upper_text.startswith("DES POSN DES POSN"):
        return True

    if upper_text.startswith("CODE NOTE COMMENT"):
        return True

    if "DES" in upper_text and "POSN" in upper_text and "MNEMONIC" in upper_text:
        return True

    if "MNEMONIC" in upper_text and "DESCRIPTION" in upper_text:
        return True

    if "WIRE CODE" in upper_text and "NOTE" in upper_text:
        return True

    return False


# =========================================
# FIND PDF RECORDS
#
# Supports both:
#
# J12 18 114 SPARE
#
# AND:
#
# J12
# 19
# 114
# MAG1-X-
# Magnetometer 1 X axis negative
# differential input
# IN
# =========================================

def find_pdf_records(input_text):
    raw_lines = input_text.split("\n")

    lines = []

    for raw_line in raw_lines:
        line = normalize_pdf_line(raw_line)

        if is_pdf_junk_line(line):
            continue

        lines.append(line)

    records = []
    current_record = None

    index = 0

    while index < len(lines):
        line = lines[index]

        # =========================================
        # INLINE RECORD
        #
        # J12 19 114 MAG1-X- ...
        # =========================================

        if looks_like_inline_wiring_row(line):

            if current_record is not None:
                records.append(current_record)

            match = re.match(
                r"^(\S+)\s+(\d+)\s+(\S+)(?:\s+(.*))?$",
                line
            )

            current_record = {
                "from_des": match.group(1),
                "from_posn": match.group(2),
                "to_des": match.group(3),
                "payload": []
            }

            remainder = match.group(4)

            if remainder is not None:
                remainder = remainder.strip()

                if remainder != "":
                    current_record["payload"].append(remainder)

            index += 1
            continue

        # =========================================
        # VERTICAL / BROKEN PDF RECORD
        #
        # J12
        # 19
        # 114
        # MAG1-X-
        # ...
        # =========================================

        if is_connector_designator(line):

            if index + 2 < len(lines):
                possible_pin = lines[index + 1]
                possible_destination = lines[index + 2]

                if is_integer_value(possible_pin):

                    if current_record is not None:
                        records.append(current_record)

                    current_record = {
                        "from_des": line,
                        "from_posn": possible_pin,
                        "to_des": possible_destination,
                        "payload": []
                    }

                    index += 3
                    continue

        # =========================================
        # CONTINUATION OF CURRENT ROW
        # =========================================

        if current_record is not None:
            current_record["payload"].append(line)

        index += 1

    if current_record is not None:
        records.append(current_record)

    return records


# =========================================
# PARSE CONTENT OF ONE PDF ROW
# =========================================

def parse_wiring_payload(payload_lines):
    to_posn = ""
    mnemonic = ""
    description = ""
    wire_code = ""
    note = ""
    comment = ""

    if len(payload_lines) == 0:
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    payload_text = " ".join(payload_lines)
    payload_text = normalize_pdf_line(payload_text)

    if payload_text == "":
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    tokens = payload_text.split()

    # =========================================
    # COMMENTS
    #
    # A SPARE row must produce a BLANK mnemonic.
    # =========================================

    comment_values = {
        "SPARE",
        "UNUSED",
        "NC",
        "N/C"
    }

    remaining = []

    for token in tokens:

        if token.upper() in comment_values:

            if comment == "":
                comment = token

        else:
            remaining.append(token)

    tokens = remaining

    if len(tokens) == 0:
        return (
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        )

    # =========================================
    # NOTE / DIRECTION
    # =========================================

    note_values = {
        "IN",
        "OUT",
        "IN/OUT",
    }

    remaining = []

    for token in tokens:

        if token in note_values:

            if note == "":
                note = token

        else:
            remaining.append(token)

    tokens = remaining

    if note not in note_values:
        note = ""

    # =========================================
    # OPTIONAL TO POSITION
    # =========================================

    if len(tokens) >= 2:

        if is_integer_value(tokens[0]):
            to_posn = tokens.pop(0)

    # =========================================
    # MNEMONIC
    #
    # First remaining item is normally the
    # mnemonic.
    #
    # CTU RTN is a valid two-word mnemonic.
    # Preserve it instead of treating RTN as
    # the start of the description.
    # =========================================

    if len(tokens) >= 2 and tokens[0].upper() == "CTU" and tokens[1].upper() == "RTN":
        mnemonic = "CTU RTN"
        tokens = tokens[2:]
    elif len(tokens) >= 1 and tokens[0].upper() == "CTURTN":
        mnemonic = "CTU RTN"
        tokens.pop(0)
    elif len(tokens) >= 1:
        mnemonic = tokens.pop(0)

        if len(tokens) >= 1 and tokens[0].startswith("("):
            continuation_parts = []

            while len(tokens) >= 1:
                continuation_token = tokens.pop(0)
                continuation_parts.append(continuation_token)

                if continuation_token.endswith(")"):
                    break

            if len(continuation_parts) > 0:
                mnemonic = mnemonic + " " + " ".join(continuation_parts)

    # =========================================
    # DESCRIPTION
    #
    # Preserve all remaining text. This is
    # important because Acrobat frequently wraps
    # descriptions onto multiple clipboard lines.
    # =========================================

    if len(tokens) > 0:
        description = " ".join(tokens)

    return (
        to_posn,
        mnemonic,
        description,
        wire_code,
        note,
        comment
    )

    records = find_pdf_records(input_text)

    rows = []

# =========================================
# GENERIC PDF TABLE PARSER (3-10 COLUMNS)
#
# This is the fallback for PDF tables that do not match one of the specialized
# pinout/wiring formats above.  Set the "Columns" box to the physical number of
# columns in the PDF (3 through 10).  The parser deliberately works with CELLS
# and ROW ANCHORS instead of flattening the whole table into words.  That keeps
# literal '-' cells in their own columns and prevents the next column from being
# concatenated onto the selected output column.
# =========================================


# =========================================
# CONTACT ALLOCATION / UNIT SIGNAL TABLE
# =========================================
# Handles tables whose headers are split across arbitrary PDF clipboard lines:
#
#   Contact Designation | Unit Signal Designation | I/F Direction |
#   Signal Type | Description
#
# The important difference from the generic parser is that row keys such as
# 01, 02, 03 ... are detected in the FLATTENED token stream, not only when they
# occupy a physical line by themselves.  This prevents Column 2 from swallowing
# "Output" plus the next row's Unit Signal Designation.

CONTACT_ALLOCATION_HEADERS = [
    "CONTACT DESIGNATION",
    "UNIT SIGNAL DESIGNATION",
    "I/F DIRECTION",
    "SIGNAL TYPE",
    "DESCRIPTION",
]

CONTACT_DIRECTION_VALUES = {
    "INPUT", "OUTPUT", "IN", "OUT", "INOUT", "IN/OUT", "I/O",
    "BIDIR", "BIDIRECTIONAL", "NA", "N/A", "NC", "N/C", "-",
}

CONTACT_SIGNAL_TYPE_VALUES = {
    "ANALOG", "DIGITAL", "DISCRETE", "POWER", "GROUND", "GND",
    "LVDS", "TTL", "CMOS", "RS422", "RS-422", "RS485", "RS-485",
    "CAN", "UART", "SPI", "I2C", "NA", "N/A", "NC", "N/C", "-",
}


def contact_allocation_token(value):
    return generic_pdf_normalized_cell(value).strip(",;:")


def contact_allocation_upper(value):
    return contact_allocation_token(value).upper().replace("\\", "/")


def looks_like_contact_allocation_table(input_text):
    """
    Detect the 5-column Contact Allocation family even when Acrobat drops the
    "Signal Type" header from the clipboard stream.

    The stable header evidence is:
        Contact Designation
        Unit Signal Designation
        I/F Direction
        Description

    Signal Type is useful when present but is NOT required.
    """
    tokens = [
        contact_allocation_token(v)
        for v in re.split(r"\s+", input_text)
        if contact_allocation_token(v)
    ]
    prefix = " ".join(tokens[:160]).upper()

    has_contact = "CONTACT" in prefix and "DESIGNATION" in prefix
    has_unit_signal = "UNIT" in prefix and "SIGNAL" in prefix and "DESIGNATION" in prefix
    has_direction = "DIRECTION" in prefix and (
        "I/F" in prefix or "I F" in prefix or "IF" in prefix
    )
    has_description = "DESCRIPTION" in prefix

    return has_contact and has_unit_signal and has_direction and has_description

def contact_allocation_longest_anchor_path(tokens):
    """Return token indices for the strongest 01,02,03... / 1,2,3... run."""
    numeric = []
    for index, token in enumerate(tokens):
        text = contact_allocation_token(token)
        if re.fullmatch(r"\d+", text) is None:
            continue
        numeric.append((index, text, int(text)))

    if len(numeric) < 2:
        return []

    # Prefer a zero-padded family (01,02,03...) when present.  This is extremely
    # strong evidence for a Contact Designation column and cannot be confused
    # with the 10 in a signal such as "10 MHz+ Input".
    families = {}
    for index, text, number in numeric:
        if len(text) > 1 and text.startswith("0"):
            family = ("ZERO", len(text))
        else:
            family = ("PLAIN",)
        families.setdefault(family, []).append((index, text, number))

    best = []
    best_score = None

    for family, values in families.items():
        states = []
        for pos, (index, text, number) in enumerate(values):
            state = [index]
            for prev_pos in range(pos):
                prev_index, prev_text, prev_number = values[prev_pos]
                if prev_number + 1 != number:
                    continue
                candidate = states[prev_pos] + [index]
                if len(candidate) > len(state):
                    state = candidate
            states.append(state)

            first_value = int(contact_allocation_token(tokens[state[0]]))
            score = (
                len(state),
                1 if family[0] == "ZERO" else 0,
                1 if first_value in (0, 1) else 0,
                -state[0],
            )
            if best_score is None or score > best_score:
                best_score = score
                best = state

    return best if len(best) >= 2 else []


def contact_allocation_is_direction(value):
    return contact_allocation_upper(value) in CONTACT_DIRECTION_VALUES


def contact_allocation_is_signal_type(value):
    return contact_allocation_upper(value) in CONTACT_SIGNAL_TYPE_VALUES


def contact_allocation_join(values):
    values = [contact_allocation_token(v) for v in values if contact_allocation_token(v)]
    if not values:
        return ""
    return normalize_signal_name_fragments(values)


def parse_contact_allocation_payload(payload, allow_large_tail=False):
    """
    Parse everything between two contact anchors.

    Direction is a hard boundary.  Critically, the search starts at token 1 so
    a Unit Signal value of NA is not mistaken for I/F Direction.  Therefore:

        01 NA NA NA SPARE

    becomes NA | NA | NA | SPARE, while:

        03 RA0_OSW_D_A Output Analog

    becomes RA0_OSW_D_A | Output | Analog | <blank>.
    """
    values = [contact_allocation_token(v) for v in payload if contact_allocation_token(v)]
    if not values:
        return ["", "", "", ""], []

    direction_index = None
    for index in range(1, len(values)):
        if contact_allocation_is_direction(values[index]):
            direction_index = index
            break

    if direction_index is None:
        # No trustworthy column boundary.  Keep the entire payload in Unit
        # Signal rather than contaminating multiple columns.
        return [contact_allocation_join(values), "", "", ""], []

    signal = contact_allocation_join(values[:direction_index])
    direction = contact_allocation_token(values[direction_index])
    remainder = values[direction_index + 1:]

    signal_type = ""
    description = ""
    overflow = []

    if remainder:
        if contact_allocation_is_signal_type(remainder[0]):
            signal_type = contact_allocation_token(remainder[0])
            remainder = remainder[1:]

        # For ordinary anchor-delimited rows, any residual text is Description.
        # On the FINAL row a large residual block can instead contain one or
        # more trailing whole-column streams, so return it separately for the
        # caller to distribute.
        if remainder:
            if allow_large_tail:
                overflow = remainder
            else:
                description = " ".join(remainder).strip()

    return [signal, direction, signal_type, description], overflow


def contact_allocation_is_ambiguous_blank_scalar(value):
    return contact_allocation_upper(value) in {"NA", "N/A", "NC", "N/C", "-"}


def contact_allocation_signal_score(value):
    """Semantic score for a Unit Signal Designation token/cell."""
    text = contact_allocation_token(value)
    upper = contact_allocation_upper(value)

    if text == "" or re.fullmatch(r"\d+", text):
        return -1000.0

    # Hard exclusions: these belong to other physical columns.
    if upper in CONTACT_DIRECTION_VALUES and upper not in {"NA", "N/A", "NC", "N/C", "-"}:
        return -1000.0
    if upper in CONTACT_SIGNAL_TYPE_VALUES and upper not in {"NA", "N/A", "NC", "N/C", "-"}:
        return -1000.0

    if contact_allocation_is_ambiguous_blank_scalar(text):
        return 2.0

    if upper in {"RESERVED", "SPARE", "UNUSED", "OPEN"}:
        return 14.0

    # Typical unit-signal identifiers are the strongest evidence.
    if "_" in text:
        return 18.0
    if re.fullmatch(r"[A-Za-z][A-Za-z0-9_.#/+:\-]*", text):
        # Code-like punctuation/digits are only signal evidence when the token
        # itself is uppercase-ish.  This prevents Description words such as
        # "electro-mechanical" from being discarded as fake signal names.
        if text.upper() == text and re.search(r"\d|[._#/+:\-]", text):
            return 12.0
        if text.isupper() and len(text) >= 2:
            return 7.0

    return -20.0


def contact_allocation_direction_score(value):
    upper = contact_allocation_upper(value)
    if upper in CONTACT_DIRECTION_VALUES:
        return 5.0 if upper in {"NA", "N/A", "NC", "N/C", "-"} else 18.0
    return -1000.0


def contact_allocation_type_score(value):
    upper = contact_allocation_upper(value)
    if upper in CONTACT_SIGNAL_TYPE_VALUES:
        return 5.0 if upper in {"NA", "N/A", "NC", "N/C", "-"} else 18.0
    return -1000.0


def contact_allocation_select_stream(tokens, row_count, score_function, anchor_indices, forbidden=None):
    """
    Select exactly one item per row for one logical column while preserving the
    source order.  PDF tagged text may interleave columns, so we do NOT require
    the item for row N to physically sit between contact N and contact N+1.

    A local-row bonus is used when Acrobat *does* preserve adjacency, but the
    semantic score remains strong enough to recover column-major/hybrid streams.
    """
    forbidden = set(forbidden or ())
    anchors = list(anchor_indices)
    anchor_set = set(anchors)

    candidates = []
    for index, token in enumerate(tokens):
        if index in forbidden or index in anchor_set:
            continue
        base = score_function(token)
        if base <= -900:
            continue
        candidates.append((index, base))

    if len(candidates) < row_count:
        return []

    # dp[k] = best (score, path) after choosing k values.
    dp = {0: (0.0, [])}

    for index, base in candidates:
        previous = list(dp.items())
        for chosen_count, (prior_score, prior_path) in reversed(previous):
            if chosen_count >= row_count:
                continue

            row_index = chosen_count
            score = base

            # Bonus when this token lies inside the matching contact interval.
            row_start = anchors[row_index]
            row_end = anchors[row_index + 1] if row_index + 1 < len(anchors) else len(tokens)
            if row_start < index < row_end:
                score += 7.0

            # Mild preference for earlier equally-good evidence.  This prevents
            # a word such as "output" inside a prose Description from beating
            # the real Direction token that appeared earlier.
            score -= index * 0.00001

            candidate_score = prior_score + score
            candidate_path = prior_path + [index]
            current = dp.get(chosen_count + 1)
            if current is None or candidate_score > current[0]:
                dp[chosen_count + 1] = (candidate_score, candidate_path)

    if row_count not in dp:
        return []

    total_score, path = dp[row_count]
    # Reject a stream assembled mostly from weak ambiguous placeholders.
    if total_score < row_count * 2.0:
        return []
    return path


def contact_allocation_description_from_indices(tokens, indices):
    values = []
    for idx in indices:
        value = contact_allocation_token(tokens[idx])
        if value:
            values.append(value)
    return " ".join(values).strip()


def parse_contact_allocation_table(input_text):
    """
    Reconstruct the 5-column Contact Allocation table using INDEPENDENT ordered
    column streams instead of assuming that newline order == row order.

    This specifically fixes Acrobat streams like:

        01
        RA0_THERM_P
        02
        Output
        Analog
        RA0 thermistor positive lead.
        RA0_SW_P
        ...

    where a row-boundary parser incorrectly produced Column 2 values such as
    "RA0_THERM_P Output RA0_SW_P".

    The only hard row frame is the sequential Contact Designation stream.  Unit
    Signal, I/F Direction, and Signal Type are each recovered globally in their
    own preserved order, then zipped back onto those contacts.
    """
    flat_tokens = [
        contact_allocation_token(v)
        for v in re.split(r"\s+", input_text.replace("\r", "\n"))
        if contact_allocation_token(v)
    ]

    anchor_indices = contact_allocation_longest_anchor_path(flat_tokens)
    if len(anchor_indices) < 2:
        return []

    row_count = len(anchor_indices)
    contacts = [contact_allocation_token(flat_tokens[index]) for index in anchor_indices]

    # Recover Unit Signal first because it is the user's key extracted column
    # and identifiers such as RA0_THERM_P are semantically unambiguous.
    signal_indices = contact_allocation_select_stream(
        flat_tokens,
        row_count,
        contact_allocation_signal_score,
        anchor_indices,
    )
    if len(signal_indices) != row_count:
        return []

    used = set(signal_indices)

    direction_indices = contact_allocation_select_stream(
        flat_tokens,
        row_count,
        contact_allocation_direction_score,
        anchor_indices,
        forbidden=used,
    )
    if len(direction_indices) != row_count:
        return []
    used.update(direction_indices)

    type_indices = contact_allocation_select_stream(
        flat_tokens,
        row_count,
        contact_allocation_type_score,
        anchor_indices,
        forbidden=used,
    )
    if len(type_indices) != row_count:
        return []
    used.update(type_indices)

    # Build Description without ever affecting Columns 1-4.  First determine
    # whether the first four columns are locally row-major or are being emitted
    # as separate whole-column streams.  In a column-major stream, assigning all
    # prose after the last Type token to row 7 is the exact failure we want to
    # avoid, so descriptions are recovered globally instead.
    descriptions = [""] * row_count
    anchor_set = set(anchor_indices)
    consumed_description_indices = set()
    selected_signal_values = {
        contact_allocation_upper(flat_tokens[idx]) for idx in signal_indices
    }

    local_order_hits = 0
    local_anchor_hits = 0
    for row_index in range(row_count):
        if signal_indices[row_index] < direction_indices[row_index] < type_indices[row_index]:
            local_order_hits += 1
        row_start = anchor_indices[row_index]
        row_end = anchor_indices[row_index + 1] if row_index + 1 < row_count else len(flat_tokens)
        if (
            row_start < signal_indices[row_index] < row_end
            and row_start < direction_indices[row_index] < row_end
            and row_start < type_indices[row_index] < row_end
        ):
            local_anchor_hits += 1

    use_local_descriptions = (
        local_order_hits >= max(2, (row_count + 1) // 2)
        and local_anchor_hits >= max(2, row_count // 3)
    )

    if use_local_descriptions:
        for row_index in range(row_count):
            start = max(
                signal_indices[row_index],
                direction_indices[row_index],
                type_indices[row_index],
            ) + 1

            # In row-major data the next CONTACT anchor is a safer Description
            # boundary than the next signal, because a signal may itself be
            # emitted late by Acrobat tagging.
            end = (
                anchor_indices[row_index + 1]
                if row_index + 1 < row_count
                else len(flat_tokens)
            )

            local = []
            for idx in range(start, end):
                if idx in used or idx in anchor_set:
                    continue
                token = contact_allocation_token(flat_tokens[idx])
                if token == "":
                    continue
                upper = contact_allocation_upper(token)

                if upper in CONTACT_DIRECTION_VALUES or upper in CONTACT_SIGNAL_TYPE_VALUES:
                    continue
                if upper in selected_signal_values:
                    continue
                if re.fullmatch(r"\d+", token):
                    continue

                local.append(idx)

            if local:
                descriptions[row_index] = contact_allocation_description_from_indices(flat_tokens, local)
                consumed_description_indices.update(local)

    # Recover all still-unclaimed prose.  This also handles column-major tagged
    # PDFs where Description is emitted as a final stream after Contacts,
    # Signals, Directions, and Types.
    leftovers = []
    for idx, token in enumerate(flat_tokens):
        if idx < anchor_indices[0]:
            continue
        if idx in used or idx in anchor_set or idx in consumed_description_indices:
            continue
        value = contact_allocation_token(token)
        if not value:
            continue
        upper = contact_allocation_upper(value)
        if upper in CONTACT_DIRECTION_VALUES or upper in CONTACT_SIGNAL_TYPE_VALUES:
            continue
        if upper in selected_signal_values:
            continue
        if re.fullmatch(r"\d+", value):
            continue
        leftovers.append((idx, value))

    # Sentence punctuation is the strongest safe delimiter for a prose column.
    # If punctuation is absent, retain line-sized chunks below instead of
    # concatenating prose into Unit Signal or another physical column.
    chunks = []
    current = []
    current_start = None
    for idx, value in leftovers:
        if current_start is None:
            current_start = idx
        current.append(value)
        if re.search(r"[.;!?]$", value):
            chunks.append((current_start, " ".join(current).strip()))
            current = []
            current_start = None
    if current:
        chunks.append((current_start, " ".join(current).strip()))

    blank_rows = [i for i, value in enumerate(descriptions) if value == ""]
    if chunks and blank_rows and len(chunks) <= len(blank_rows) + 2:
        # If every Description chunk appears after every Unit Signal token, the
        # PDF emitted a true trailing Description column: distribute in row
        # order.  Otherwise associate a chunk with the nearest preceding Unit
        # Signal.  This handles hybrid tagged streams where descriptions for
        # rows 1 and 4 appear early even though Direction/Type are elsewhere.
        if all(start_idx > max(signal_indices) for start_idx, _ in chunks):
            for row_index, (_, chunk) in zip(blank_rows, chunks):
                descriptions[row_index] = chunk
        else:
            unassigned = []
            for start_idx, chunk in chunks:
                candidate_rows = [
                    i for i, sig_idx in enumerate(signal_indices)
                    if sig_idx < start_idx
                ]
                target = candidate_rows[-1] if candidate_rows else None
                if target is not None and descriptions[target] == "":
                    descriptions[target] = chunk
                else:
                    unassigned.append(chunk)

            if unassigned:
                remaining_blank = [i for i, value in enumerate(descriptions) if value == ""]
                for row_index, chunk in zip(remaining_blank, unassigned):
                    descriptions[row_index] = chunk

    rows = [list(CONTACT_ALLOCATION_HEADERS)]
    for i in range(row_count):
        rows.append([
            contacts[i],
            contact_allocation_token(flat_tokens[signal_indices[i]]),
            contact_allocation_token(flat_tokens[direction_indices[i]]),
            contact_allocation_token(flat_tokens[type_indices[i]]),
            descriptions[i],
        ])

    # Final confidence gate: the first four physical columns must all have the
    # expected row count and Unit Signal must not contain Direction/Type text.
    for row in rows[1:]:
        if row[1] == "" or row[2] == "" or row[3] == "":
            return []
        if any(
            contact_allocation_upper(part) in CONTACT_DIRECTION_VALUES
            for part in row[1].split()
        ):
            return []

    return rows


GENERIC_PDF_MIN_COLUMNS = 3
GENERIC_PDF_MAX_COLUMNS = 10


def generic_pdf_columns_supported(column_count):
    return (
        column_count is not None
        and GENERIC_PDF_MIN_COLUMNS <= column_count <= GENERIC_PDF_MAX_COLUMNS
    )


def generic_pdf_normalized_cell(value):
    text = str(value or "")
    text = text.replace("\t", " ")
    text = re.sub(r"\s+", " ", text)
    return text.strip()


def generic_pdf_is_junk_line(value):
    text = generic_pdf_normalized_cell(value)
    upper = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True
    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True

    # Common PDF page/footer fragments.  Keep this intentionally conservative;
    # '-' and NA/N/A are REAL CELLS and must never be treated as junk.
    if upper in {"CONTINUED", "CONTINUED ON NEXT PAGE"}:
        return True

    return False


def generic_pdf_raw_lines(input_text):
    values = []
    for raw in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        if raw.strip() == "":
            continue
        if generic_pdf_is_junk_line(raw):
            continue
        values.append(raw.rstrip())
    return values


def generic_pdf_anchor_signature(value):
    """
    Return a conservative signature for a likely first-column row key.

    Requiring anchors to share a signature is what prevents a numeric value in
    another column (for example the 10 in "10 MHz+ Input") from being promoted
    to a new row when the real first column is 01, 02, 03, ... .
    """
    text = generic_pdf_normalized_cell(value)
    if text == "" or " " in text:
        return None

    # 01, 02, 03 ... -- preserve the leading zero and exact width.
    if re.fullmatch(r"\d+", text):
        if len(text) > 1 and text.startswith("0"):
            return ("DIGITS_ZERO_PADDED", len(text))
        return ("DIGITS",)

    # A01, J12, P003 etc.  Keep the alphabetic prefix stable.
    match = re.fullmatch(r"([A-Za-z]+)(\d+)([A-Za-z]*)", text)
    if match is not None:
        return (
            "ALPHA_DIGITS",
            match.group(1).upper(),
            match.group(3).upper(),
        )

    # Compact designators with punctuation, but only when a digit is present.
    if (
        len(text) <= 18
        and re.fullmatch(r"[A-Za-z0-9_.#/+\-]+", text) is not None
        and re.search(r"\d", text) is not None
    ):
        prefix = re.match(r"^[A-Za-z]+", text)
        return ("COMPACT_DESIGNATOR", prefix.group(0).upper() if prefix else "")

    return None


def generic_pdf_same_anchor_signature(value, signature):
    return generic_pdf_anchor_signature(value) == signature


def generic_pdf_choose_anchor_path(lines, column_count):
    """
    Find the longest plausible sequence of first-column cells in a vertical PDF
    clipboard stream.  A normal unwrapped row occupies exactly column_count
    physical cell lines.  Wrapped cells may make it a little longer.
    """
    candidates_by_signature = {}

    for index, raw in enumerate(lines):
        cell = generic_pdf_normalized_cell(raw)
        signature = generic_pdf_anchor_signature(cell)
        if signature is None:
            continue
        candidates_by_signature.setdefault(signature, []).append(index)

    best_path = []
    best_score = None

    # Allow wrapped text, but not an unlimited distance.  The limit is generous
    # enough for long Description cells while still rejecting unrelated numbers.
    max_gap = max(column_count + 24, column_count * 8)

    for signature, indices in candidates_by_signature.items():
        if len(indices) < 2:
            continue

        states = []

        for pos, line_index in enumerate(indices):
            cell_value = generic_pdf_normalized_cell(lines[line_index])
            best = {
                "path": [line_index],
                "rows": 1,
                "gap_error": 0,
                "wrapped": 0,
                "quality": 0.0,
                "values": [cell_value],
            }

            for prev_pos in range(pos):
                prev_index = indices[prev_pos]
                gap = line_index - prev_index

                # PDF clipboard text often OMITS empty table cells entirely.
                # Therefore a real N-column row can occupy fewer than N physical
                # lines.  Do not require gap >= column_count here; that was the
                # bug that skipped contacts such as 02/04/06 when Description was
                # blank and merged the next row into Column 2.
                #
                # A one-line gap is allowed for strong anchor families where the
                # first column itself is enough to identify the next row (e.g.
                # 01, 02, 03 or J1, J2, J3).  For plain unpadded integers keep a
                # two-line minimum so a numeric value inside another cell is less
                # likely to become a phantom row anchor.
                minimum_gap = 2
                if signature[0] in {
                    "DIGITS_ZERO_PADDED",
                    "ALPHA_DIGITS",
                    "COMPACT_DESIGNATOR",
                }:
                    minimum_gap = 1

                if gap < minimum_gap:
                    continue
                if gap > max_gap:
                    continue

                prev_state = states[prev_pos]

                # Score how plausibly the physical lines between these two
                # anchors reconstruct into the remaining columns of ONE row.
                # This is what rejects a numeric token inside a signal/figure
                # (for example the 10 in "10 MHz- Input" or "Figure 10") from
                # becoming a phantom first-column value.
                transition_payload = lines[prev_index + 1:line_index]
                transition_cells = generic_pdf_partition_payload(
                    transition_payload,
                    column_count - 1,
                )
                transition_quality = 0.0
                if len(transition_cells) == column_count - 1:
                    for transition_col, transition_cell in enumerate(transition_cells):
                        transition_quality += generic_pdf_group_score(
                            [transition_cell],
                            transition_col,
                            column_count - 1,
                        )

                state_values = prev_state["values"] + [cell_value]
                state = {
                    "path": prev_state["path"] + [line_index],
                    "rows": prev_state["rows"] + 1,
                    # A gap SMALLER than column_count is not an error: it
                    # usually means one or more blank PDF cells were not emitted
                    # to the clipboard.  Only penalize overflow/wrapping.
                    "gap_error": prev_state["gap_error"] + max(0, gap - column_count),
                    "wrapped": prev_state["wrapped"] + max(0, gap - column_count),
                    "quality": prev_state["quality"] + transition_quality,
                    "values": state_values,
                }

                unique_count = len(set(state_values))
                duplicate_count = len(state_values) - unique_count
                best_unique = len(set(best["values"]))
                best_duplicates = len(best["values"]) - best_unique

                state_score = (
                    state["rows"],
                    unique_count,
                    -duplicate_count,
                    round(state["quality"], 6),
                    -state["gap_error"],
                    -state["wrapped"],
                    -state["path"][0],
                )
                best_score_local = (
                    best["rows"],
                    best_unique,
                    -best_duplicates,
                    round(best["quality"], 6),
                    -best["gap_error"],
                    -best["wrapped"],
                    -best["path"][0],
                )

                if state_score > best_score_local:
                    best = state

            states.append(best)

        for state in states:
            unique_count = len(set(state["values"]))
            duplicate_count = len(state["values"]) - unique_count
            score = (
                state["rows"],
                unique_count,
                -duplicate_count,
                round(state["quality"], 6),
                -state["gap_error"],
                -state["wrapped"],
                # A zero-padded numeric contact/pin column is especially strong.
                1 if signature[0] == "DIGITS_ZERO_PADDED" else 0,
                -state["path"][0],
            )
            if best_score is None or score > best_score:
                best_score = score
                best_path = state["path"]

    return best_path


GENERIC_STRONG_SCALARS = {
    "-", "NA", "N/A", "NC", "N/C", "NONE", "NULL",
    "IN", "OUT", "INPUT", "OUTPUT", "INOUT", "IN/OUT", "I/O",
    "POWER", "GROUND", "GND", "ANALOG", "DIGITAL",
    "LVDS", "TTL", "CMOS", "RS422", "RS-422", "RS485", "RS-485",
    "SPARE", "RESERVED", "UNUSED", "OPEN",
    "YES", "NO", "TRUE", "FALSE",
}

GENERIC_ENGINEERING_UNITS = (
    "HZ", "KHZ", "MHZ", "GHZ",
    "V", "MV", "UV", "KV",
    "A", "MA", "UA",
    "OHM", "OHMS", "KOHM", "KOHMS", "MOHM", "MOHMS", "Ω",
    "W", "MW", "KW",
    "F", "UF", "NF", "PF",
)


def generic_pdf_join_cell_lines(values):
    return generic_pdf_normalized_cell(" ".join(values))


def generic_pdf_is_strong_scalar(text):
    upper = generic_pdf_normalized_cell(text).upper()
    if upper in GENERIC_STRONG_SCALARS:
        return True
    if re.fullmatch(r"(?:LVDS|TTL|CMOS|RS[- ]?4(?:22|85)|CAN|UART|SPI|I2C)", upper):
        return True
    return False


def generic_pdf_group_score(values, column_index, target_count):
    """Score one possible reconstructed cell made from consecutive PDF lines."""
    values = [generic_pdf_normalized_cell(v) for v in values if generic_pdf_normalized_cell(v) != ""]
    if not values:
        return -100000.0

    text = generic_pdf_join_cell_lines(values)
    upper = text.upper()
    line_count = len(values)
    word_count = len(text.split())
    is_last = column_index == target_count - 1
    is_first_payload = column_index == 0

    score = 0.0

    # If Acrobat already emitted a whole cell on one line, strongly prefer not
    # to merge it with a neighbor.  This is the normal "newline between every
    # input/cell" case.
    if line_count == 1:
        score += 3.0
    else:
        score -= 0.20 * (line_count - 1)

    # Literal dash / NA / direction / logic-type values are strong standalone
    # cells.  Merging them with neighboring text is almost always a column shift.
    if generic_pdf_is_strong_scalar(text):
        score += 7.0
        if line_count > 1:
            score -= 2.0
    else:
        scalar_parts = sum(1 for value in values if generic_pdf_is_strong_scalar(value))
        if scalar_parts:
            score -= 5.0 * scalar_parts

    # Figure references are commonly split by Acrobat as "Figure" then "10".
    if re.fullmatch(r"FIGURE\s+\d+(?:\s*,\s*FIGURE\s+\d+)*", upper):
        score += 8.0

    # Numeric-leading signal names such as "10 MHz+ Input" MUST stay together.
    unit_pattern = "|".join(re.escape(unit) for unit in GENERIC_ENGINEERING_UNITS)
    if re.match(
        rf"^\d+(?:\.\d+)?\s*(?:{unit_pattern})(?:[+\-])?(?:\s+.*)?$",
        upper,
    ):
        score += 8.0

    # Common two-word electrical cells that PDFs often split one word per line.
    if upper in {
        "CHASSIS GND", "CHASSIS GROUND", "SIGNAL GND", "SIGNAL GROUND",
        "POWER GND", "POWER GROUND", "FRAME GND", "FRAME GROUND",
        "NOT USED", "NO CONNECT", "DO NOT CONNECT",
    }:
        score += 6.0

    # Signal/designator-like text belongs comfortably in the first payload
    # column, even when Acrobat split it across several lines.
    if is_first_payload:
        if re.search(r"[A-Z0-9][A-Z0-9_./+\-]*", upper):
            score += 1.0
        score += min(2.5, 0.45 * max(0, line_count - 1))

    # The final column is often Description/Comment and can legitimately wrap.
    if is_last:
        score += min(4.0, 0.55 * max(0, line_count - 1))
        score += min(2.0, 0.20 * max(0, word_count - 1))
    elif not is_first_payload and line_count > 1:
        # Middle scalar columns are usually short.  Only a strong compound such
        # as Figure 10 should overcome this penalty.
        score -= 1.10 * (line_count - 1)

    return score



def generic_pdf_empty_cell_score(column_index, target_count):
    """
    Score an omitted/blank logical cell.

    Acrobat/PDF viewers commonly do not emit blank cells at all.  Empty cells
    must therefore be legal during reconstruction.  The final payload column is
    most often Description/Comment/Figure and is especially likely to be blank,
    while a blank middle cell is possible but gets a small penalty.
    """
    if target_count <= 0:
        return -3.0

    if column_index == target_count - 1:
        return 0.75

    if column_index == 0:
        return -3.0

    return -1.25

def generic_pdf_partition_payload(payload, target_count):
    """
    Partition one anchor-delimited PDF row payload into exactly target_count
    logical cells.

    Important behaviors:
      * blank PDF cells may be omitted from clipboard text entirely;
      * a newline may appear between every real cell;
      * a newline may also appear inside one real cell;
      * '-' / NA / Output / Analog / LVDS remain standalone cells;
      * Figure + 10 may be rejoined as "Figure 10";
      * 10 + MHz+ + Input may be rejoined as "10 MHz+ Input".

    Unlike the older implementation, EMPTY logical groups are allowed.  That is
    what keeps a five-column row such as

        01
        RA0_THERM_P
        Output
        Analog
        02

    aligned as

        01 | RA0_THERM_P | Output | Analog | <blank>

    instead of skipping 02 and concatenating the next row into Column 2.
    """
    values = [generic_pdf_normalized_cell(value) for value in payload]
    values = [value for value in values if value != ""]

    if target_count <= 0:
        return []

    if not values:
        return [""] * target_count

    if target_count == 1:
        return [generic_pdf_join_cell_lines(values)]

    n = len(values)
    neg = -10**18

    # dp[g][i] = best score after creating g logical cells and consuming the
    # first i physical clipboard lines.  A logical cell may consume zero lines.
    dp = [[neg] * (n + 1) for _ in range(target_count + 1)]
    prev = [[None] * (n + 1) for _ in range(target_count + 1)]
    dp[0][0] = 0.0

    for groups in range(1, target_count + 1):
        column_index = groups - 1

        for i in range(0, n + 1):
            # Option 1: this logical column was blank/omitted by the PDF copy.
            if dp[groups - 1][i] != neg:
                empty_score = dp[groups - 1][i] + generic_pdf_empty_cell_score(
                    column_index,
                    target_count,
                )
                if empty_score > dp[groups][i]:
                    dp[groups][i] = empty_score
                    prev[groups][i] = ("EMPTY", i)

            # Option 2: consume one or more consecutive physical lines.
            max_group_len = n if groups == target_count else min(12, i)
            j_min = max(0, i - max_group_len)

            for j in range(j_min, i):
                if dp[groups - 1][j] == neg:
                    continue

                group_values = values[j:i]
                group_score = generic_pdf_group_score(
                    group_values,
                    column_index,
                    target_count,
                )

                # Extra evidence for identifiers split at punctuation.
                joined = generic_pdf_join_cell_lines(group_values)
                if len(group_values) >= 2:
                    if any(
                        str(group_values[k]).rstrip().endswith(("_", "/", "-", "+", "."))
                        or str(group_values[k + 1]).lstrip().startswith(("_", "/", "-", "+", "."))
                        for k in range(len(group_values) - 1)
                    ):
                        group_score += 6.0

                    # "Figure" + "10" and engineering-unit fragments are
                    # especially common PDF line breaks.
                    upper_parts = [str(v).upper() for v in group_values]
                    if (
                        upper_parts[0] == "FIGURE"
                        and len(upper_parts) >= 2
                        and re.fullmatch(r"\d+", upper_parts[1]) is not None
                    ):
                        group_score += 5.0

                score = dp[groups - 1][j] + group_score

                if score > dp[groups][i]:
                    dp[groups][i] = score
                    prev[groups][i] = ("VALUES", j)

    if dp[target_count][n] == neg:
        # Extremely conservative fallback.
        values = values[:target_count]
        while len(values) < target_count:
            values.append("")
        return values

    cells = []
    g = target_count
    i = n

    while g > 0:
        entry = prev[g][i]

        if entry is None:
            cells.append("")
            g -= 1
            continue

        kind, j = entry

        if kind == "EMPTY":
            cells.append("")
            # i is unchanged
        else:
            cells.append(generic_pdf_join_cell_lines(values[j:i]))
            i = j

        g -= 1

    cells.reverse()

    if len(cells) != target_count:
        cells = cells[:target_count]
        while len(cells) < target_count:
            cells.append("")

    return cells


def generic_pdf_merge_payload_to_width(payload, target_count):
    """
    Convert the physical lines after a row anchor into target_count cells.

    Unlike the old implementation, overflow is NOT blindly appended to the
    final column.  A DP partitioner reconstructs wrapped/interleaved cells while
    preserving explicit '-' / NA / Figure / Logic / Direction cells.
    """
    return generic_pdf_partition_payload(payload, target_count)




def generic_pdf_choose_sequential_numeric_anchor_path(lines):
    """
    Find the longest physically ordered run of consecutive numeric row keys.

    This is intentionally stronger than the generic anchor scorer for tables
    whose first column is 01, 02, 03 ... (or 1, 2, 3 ...).  PDF tagged-text
    extraction can reorder later columns, so the distance between row keys is
    not reliable.  The numeric sequence itself is reliable and prevents values
    such as the 10 in "Figure 10" or "10 MHz+ Input" from stealing a row.
    """
    candidates = []
    for index, raw in enumerate(lines):
        text = generic_pdf_normalized_cell(raw)
        if re.fullmatch(r"\d+", text) is None:
            continue
        candidates.append((index, text, int(text)))

    if len(candidates) < 3:
        return []

    best_path = []

    # Keep zero-padded and non-padded families separate.  Width is part of the
    # family only for values beginning with zero so 01..09 cannot jump into a
    # random 10 embedded in another column.
    families = {}
    for pos, (index, text, number) in enumerate(candidates):
        if len(text) > 1 and text.startswith("0"):
            key = ("ZERO_PADDED", len(text))
        else:
            key = ("PLAIN",)
        families.setdefault(key, []).append((pos, index, text, number))

    for family, values in families.items():
        # DP: best consecutive-value path ending at this candidate.
        states = []
        for local_pos, (_global_pos, line_index, text_value, number) in enumerate(values):
            best = [line_index]
            for prev_local in range(local_pos):
                _p, prev_line, prev_text, prev_number = values[prev_local]
                if prev_number + 1 != number:
                    continue
                candidate_path = states[prev_local] + [line_index]
                if len(candidate_path) > len(best):
                    best = candidate_path
            states.append(best)
            if len(best) > len(best_path):
                best_path = best

    return best_path if len(best_path) >= 3 else []


def generic_pdf_mode_int(values):
    if not values:
        return None
    counts = {}
    for value in values:
        counts[value] = counts.get(value, 0) + 1
    return max(counts, key=lambda value: (counts[value], -value))


def generic_pdf_partition_nonempty_payload(payload, target_count):
    """Partition into target_count non-empty cells; used for columns known present."""
    values = [generic_pdf_normalized_cell(v) for v in payload]
    values = [v for v in values if v != ""]
    if target_count <= 0:
        return []
    if len(values) < target_count:
        return []
    if target_count == 1:
        return [generic_pdf_join_cell_lines(values)]

    n = len(values)
    neg = -10**18
    dp = [[neg] * (n + 1) for _ in range(target_count + 1)]
    prev = [[None] * (n + 1) for _ in range(target_count + 1)]
    dp[0][0] = 0.0

    for groups in range(1, target_count + 1):
        col = groups - 1
        for i in range(1, n + 1):
            for j in range(groups - 1, i):
                if dp[groups - 1][j] == neg:
                    continue
                vals = values[j:i]
                score = dp[groups - 1][j] + generic_pdf_group_score(vals, col, target_count)
                joined = generic_pdf_join_cell_lines(vals)
                if len(vals) >= 2:
                    if any(
                        vals[k].rstrip().endswith(("_", "/", "-", "+", "."))
                        or vals[k + 1].lstrip().startswith(("_", "/", "-", "+", "."))
                        for k in range(len(vals) - 1)
                    ):
                        score += 6.0
                    upper_parts = [v.upper() for v in vals]
                    if upper_parts[0] == "FIGURE" and len(upper_parts) >= 2 and re.fullmatch(r"\d+", upper_parts[1]):
                        score += 5.0
                if score > dp[groups][i]:
                    dp[groups][i] = score
                    prev[groups][i] = j

    if dp[target_count][n] == neg:
        return []

    cells = []
    g = target_count
    i = n
    while g > 0:
        j = prev[g][i]
        if j is None:
            return []
        cells.append(generic_pdf_join_cell_lines(values[j:i]))
        i = j
        g -= 1
    cells.reverse()
    return cells


def parse_generic_pdf_hybrid_anchor_stream(input_text, column_count):
    """
    Recover PDF tables that mix row-major and column-major tagged-text order.

    A common Acrobat stream for a five-column table is effectively:

        01
        SIGNAL_1
        Output
        02
        SIGNAL_2
        Output
        ...
        07
        RESERVED
        NA
        Analog
        Analog
        Analog
        Analog
        Analog
        Analog
        NA

    The first few columns are emitted row-by-row, while a later column is emitted
    as one trailing column stream.  Earlier parsers treated the trailing stream
    as part of the final row and then shifted/merged Column 2.  This routine uses
    a consecutive first-column sequence, learns how many payload columns are
    present before each next row key, then distributes complete trailing streams
    back across ALL rows.
    """
    lines = generic_pdf_raw_lines(input_text)
    if len(lines) < max(6, column_count + 2):
        return []

    path = generic_pdf_choose_sequential_numeric_anchor_path(lines)
    if len(path) < 3:
        return []

    row_count = len(path)
    payload_width = column_count - 1

    # Physical payload slices for every row except the last one.  Their repeated
    # size tells us how many early columns Acrobat is keeping row-major.
    between = [lines[path[i] + 1:path[i + 1]] for i in range(row_count - 1)]
    between_lengths = [len([v for v in part if generic_pdf_normalized_cell(v) != ""]) for part in between]
    if not between_lengths:
        return []

    typical_physical = generic_pdf_mode_int(between_lengths)
    if typical_physical is None:
        return []

    # If anchors are adjacent, the body is column-major: no payload columns were
    # emitted row-major before the next anchor.  Otherwise, when the repeated
    # between-anchor payload has <= payload_width lines, each line is normally a
    # complete early-column cell.  For wrapped cells, fall back to the old anchor
    # parser instead of guessing.
    if typical_physical == 0:
        early_columns = 0
    elif typical_physical <= payload_width:
        early_columns = typical_physical
    else:
        return []

    # Require most rows to agree with the detected physical pattern.  This keeps
    # the recovery conservative on ordinary prose containing numeric sequences.
    agreeing = sum(1 for value in between_lengths if value == typical_physical)
    if agreeing < max(2, (len(between_lengths) + 1) // 2):
        return []

    rows = [[generic_pdf_normalized_cell(lines[index])] + [""] * payload_width for index in path]

    # Fill the row-major prefix for all rows except the final row.
    for row_index, part in enumerate(between):
        if early_columns == 0:
            continue
        cells = generic_pdf_partition_nonempty_payload(part, early_columns)
        if len(cells) != early_columns:
            return []
        for col, cell in enumerate(cells, start=1):
            rows[row_index][col] = cell

    # Everything after the final anchor contains the final row's early cells,
    # followed by zero or more complete column streams.
    tail = [generic_pdf_normalized_cell(v) for v in lines[path[-1] + 1:]]
    tail = [v for v in tail if v != ""]

    # Determine how many physical lines belong to the final row's early prefix.
    # For the newline-per-cell case this is exactly typical_physical.  We also
    # try a small range and prefer a remainder divisible by row_count, because a
    # complete trailing column stream contains one cell per row.
    prefix_candidates = [0] if early_columns == 0 else []
    if early_columns > 0:
        low = early_columns
        high = min(len(tail), max(early_columns, typical_physical + 6))
        for prefix_len in range(low, high + 1):
            cells = generic_pdf_partition_nonempty_payload(tail[:prefix_len], early_columns)
            if len(cells) != early_columns:
                continue
            remainder = len(tail) - prefix_len
            full_streams = remainder // row_count
            leftover = remainder % row_count
            score = (
                1 if leftover == 0 else 0,
                min(full_streams, payload_width - early_columns),
                -leftover,
                -abs(prefix_len - typical_physical),
            )
            prefix_candidates.append((score, prefix_len, cells))

    if early_columns == 0:
        final_prefix_len = 0
        final_prefix_cells = []
    else:
        if not prefix_candidates:
            return []
        prefix_candidates.sort(reverse=True, key=lambda item: item[0])
        _score, final_prefix_len, final_prefix_cells = prefix_candidates[0]
        for col, cell in enumerate(final_prefix_cells, start=1):
            rows[-1][col] = cell

    remaining = tail[final_prefix_len:]
    next_col = 1 + early_columns

    # Distribute complete trailing streams.  One stream == one logical column
    # with one value for every row.  This is the key fix for Signal Type / Figure
    # / Logic columns that Acrobat emits after all row anchors.
    while next_col < column_count and len(remaining) >= row_count:
        stream = remaining[:row_count]

        # A real trailing stream normally has a consistent semantic shape
        # (Analog/Analog/..., LVDS/LVDS/..., '-'/'-'/..., etc.).  Reject a chunk
        # that looks like arbitrary prose so we do not manufacture columns.
        shapes = [generic_pdf_cell_shape(v) for v in stream]
        shape_counts = {}
        for shape in shapes:
            shape_counts[shape] = shape_counts.get(shape, 0) + 1
        dominant = max(shape_counts.values())
        scalar_count = sum(1 for value in stream if generic_pdf_is_strong_scalar(value) or value == "-")

        if dominant < max(2, (row_count + 1) // 2) and scalar_count < max(2, (row_count + 1) // 2):
            break

        for row_index, value in enumerate(stream):
            rows[row_index][next_col] = value

        remaining = remaining[row_count:]
        next_col += 1

    # If a small remainder exists and only the final logical column is still
    # empty, it is usually wrapped Description text for the last row.  Preserve
    # it there rather than allowing it to contaminate Column 2.
    if remaining and next_col == column_count - 1:
        rows[-1][next_col] = generic_pdf_join_cell_lines(remaining)

    return rows

def parse_generic_pdf_anchor_stream(input_text, column_count):
    lines = generic_pdf_raw_lines(input_text)
    if len(lines) < column_count * 2:
        return []

    path = generic_pdf_choose_anchor_path(lines, column_count)
    if len(path) < 2:
        return []

    rows = []

    for row_index, start in enumerate(path):
        end = path[row_index + 1] if row_index + 1 < len(path) else len(lines)
        anchor = generic_pdf_normalized_cell(lines[start])
        payload = lines[start + 1:end]

        # The final row may be followed by a footer/title.  Once we have at
        # least enough cells, only let modest overflow be merged into the last
        # column.  Very large overflow is more likely outside the table.
        max_payload = (column_count - 1) + 8
        if len(payload) > max_payload:
            payload = payload[:max_payload]

        cells = [anchor] + generic_pdf_merge_payload_to_width(
            payload,
            column_count - 1,
        )

        if len(cells) == column_count:
            rows.append(cells)

    return rows


def generic_pdf_split_explicit_columns(raw_line, column_count):
    """
    Split one physical row only when the PDF preserved an actual column
    delimiter: tabs, pipes, or runs of 2+ whitespace.  Never split ordinary
    single spaces here because they belong inside values such as 10 MHz+ Input.
    """
    raw = raw_line.rstrip("\r\n")

    if "\t" in raw:
        parts = [generic_pdf_normalized_cell(value) for value in raw.split("\t")]
        if len(parts) == column_count:
            return parts

    if "|" in raw:
        parts = [generic_pdf_normalized_cell(value) for value in raw.split("|")]
        parts = [value for value in parts if value != "" or len(parts) == column_count]
        if len(parts) == column_count:
            return parts

    parts = [
        generic_pdf_normalized_cell(value)
        for value in re.split(r"\s{2,}", raw.strip())
    ]
    if len(parts) == column_count:
        return parts

    return None


def parse_generic_pdf_explicit_rows(input_text, column_count):
    rows = []
    for raw in input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n"):
        if raw.strip() == "" or generic_pdf_is_junk_line(raw):
            continue
        row = generic_pdf_split_explicit_columns(raw, column_count)
        if row is not None:
            rows.append(row)

    if not rows:
        return []

    # A clean aligned header is also a column_count-wide row.  Do not return it
    # as DATA when the following rows share a clear first-column key pattern.
    signatures = {}
    anchored_rows = []
    for row in rows:
        signature = generic_pdf_anchor_signature(row[0])
        if signature is None:
            continue
        signatures[signature] = signatures.get(signature, 0) + 1
        anchored_rows.append((signature, row))

    if signatures:
        dominant = max(signatures, key=signatures.get)
        selected = [row for signature, row in anchored_rows if signature == dominant]
        if len(selected) >= 2:
            return selected

    return rows


def generic_pdf_partition_single_space_row(raw_line, column_count):
    """
    Last-resort row parser for PDFs that collapse column spacing to one space.

    It is conservative: the first column is required to look like a row key.
    Extra words are kept together in the LAST column instead of being truncated.
    Therefore `02 RESERVED NA NA do not use this contact` remains five cells.
    Specialized parsers still run before this fallback for layouts where a
    multi-word middle cell needs format-specific treatment.
    """
    text = generic_pdf_normalized_cell(raw_line)
    tokens = text.split()

    if len(tokens) < column_count:
        return None
    if generic_pdf_anchor_signature(tokens[0]) is None:
        return None

    if len(tokens) == column_count:
        return tokens

    if column_count == 1:
        return [text]

    return tokens[:column_count - 1] + [" ".join(tokens[column_count - 1:])]


def parse_generic_pdf_single_space_rows(input_text, column_count):
    rows = []
    signatures = {}

    for raw in generic_pdf_raw_lines(input_text):
        row = generic_pdf_partition_single_space_row(raw, column_count)
        if row is None:
            continue
        signature = generic_pdf_anchor_signature(row[0])
        signatures[signature] = signatures.get(signature, 0) + 1
        rows.append((signature, row))

    if not rows:
        return []

    # Keep the dominant first-column signature so numeric prose elsewhere in
    # the selection cannot become a phantom table row.
    dominant = max(signatures, key=signatures.get)
    selected = [row for signature, row in rows if signature == dominant]
    return selected if len(selected) >= 2 else []


def generic_pdf_cell_shape(value):
    text = generic_pdf_normalized_cell(value)
    upper = text.upper()
    if text == "":
        return "BLANK"
    if text == "-":
        return "DASH"
    if generic_pdf_is_strong_scalar(text):
        return "SCALAR"
    if re.fullmatch(r"\d+", text):
        return "DIGITS"
    if re.fullmatch(r"FIGURE\s+\d+(?:\s*,\s*FIGURE\s+\d+)*", upper):
        return "FIGURE"
    if re.fullmatch(r"[A-Z0-9_.#/+\-]+", upper):
        return "CODE"
    if len(text.split()) >= 2:
        return "PHRASE"
    return "TEXT"


GENERIC_HEADER_KEYWORDS = {
    "PIN", "PIN NUMBER", "NUMBER", "CONTACT", "CONTACT DESIGNATION",
    "DESIGNATION", "UNIT", "UNIT SIGNAL", "UNIT SIGNAL DESIGNATION",
    "SIGNAL", "SIGNAL NAME", "SIGNAL TYPE", "DESCRIPTION",
    "DIRECTION", "I/F", "I/F DIRECTION", "TYPE", "FIGURE", "LOGIC",
    "FUNCTION", "NAME", "REF", "REF DES", "DES", "POSN", "POSITION",
    "FROM", "TO", "MNEMONIC", "WIRE", "WIRE CODE", "NOTE", "COMMENT",
    "CODE", "VALUE", "PARAMETER", "CHANNEL", "PORT", "CONTACT NO",
}


def generic_pdf_row_looks_header_like(row):
    hits = 0
    for value in row:
        upper = generic_pdf_normalized_cell(value).upper()
        if upper in GENERIC_HEADER_KEYWORDS:
            hits += 1
            continue
        # Also recognize wrapped/composite labels such as "Signal Type" or
        # "Contact Designation" without matching arbitrary data prose.
        if any(
            keyword in upper
            for keyword in (
                "DESCRIPTION", "DIRECTION", "DESIGNATION", "SIGNAL TYPE",
                "SIGNAL NAME", "PIN NUMBER", "CONTACT", "FIGURE", "LOGIC",
            )
        ):
            hits += 1
    return hits >= max(2, (len(row) + 1) // 2)


def parse_generic_pdf_modulo_cell_stream(input_text, column_count):
    """
    Recover a pure row-major stream where Acrobat inserted a newline between
    every cell and the first column does not provide a usable numeric/designator
    anchor.  We try each possible starting offset and keep the alignment whose
    per-column value shapes are most consistent across rows.
    """
    lines = [generic_pdf_normalized_cell(v) for v in generic_pdf_raw_lines(input_text)]
    lines = [v for v in lines if v != ""]
    if len(lines) < column_count * 2:
        return []

    best_rows = []
    best_score = None
    max_start = min(len(lines) - column_count * 2, column_count * 3)
    max_start = max(0, max_start)

    for start in range(max_start + 1):
        remaining = lines[start:]
        full_row_count = len(remaining) // column_count
        if full_row_count < 2:
            continue

        # Ignore an incomplete trailing fragment/footer.
        rows = [
            remaining[i * column_count:(i + 1) * column_count]
            for i in range(full_row_count)
        ]

        # Score column-shape consistency.  Real cell-stream alignment tends to
        # keep each physical column semantically similar from row to row.
        consistency = 0.0
        for col in range(column_count):
            counts = {}
            for row in rows:
                shape = generic_pdf_cell_shape(row[col])
                counts[shape] = counts.get(shape, 0) + 1
            dominant_count = max(counts.values())
            consistency += dominant_count / len(rows)

        dash_cells = sum(1 for row in rows for v in row if v == "-")
        scalar_cells = sum(1 for row in rows for v in row if generic_pdf_is_strong_scalar(v))

        # Prefer more complete rows, consistent columns, and starts close to the
        # beginning without requiring that headers themselves be parseable.
        score = (
            full_row_count,
            round(consistency, 6),
            dash_cells + scalar_cells,
            -start,
        )
        if best_score is None or score > best_score:
            best_score = score
            best_rows = rows

    # If the first row looks header-like while the following rows have a much
    # stronger repeated shape, drop it from the data candidate.  A synthetic
    # header will be added later.
    if len(best_rows) >= 3:
        first_shapes = [generic_pdf_cell_shape(v) for v in best_rows[0]]
        next_shapes = [generic_pdf_cell_shape(v) for v in best_rows[1]]
        if generic_pdf_row_looks_header_like(best_rows[0]):
            best_rows = best_rows[1:]
        elif first_shapes.count("PHRASE") + first_shapes.count("TEXT") >= column_count - 1:
            if next_shapes != first_shapes:
                best_rows = best_rows[1:]

    return best_rows


def generic_pdf_candidate_score(rows, column_count, method_priority):
    if not rows:
        return (-1, -1, -1, -1, -1)

    valid = [row for row in rows if len(row) == column_count]
    if len(valid) < 2:
        return (-1, -1, -1, -1, -1)

    nonblank = sum(
        1
        for row in valid
        for value in row
        if generic_pdf_normalized_cell(value) != ""
    )
    dash_cells = sum(
        1
        for row in valid
        for value in row
        if generic_pdf_normalized_cell(value) == "-"
    )

    shape_consistency = 0.0
    for col in range(column_count):
        counts = {}
        for row in valid:
            shape = generic_pdf_cell_shape(row[col])
            counts[shape] = counts.get(shape, 0) + 1
        if counts:
            shape_consistency += max(counts.values()) / len(valid)

    return (len(valid), round(shape_consistency, 6), nonblank, dash_cells, method_priority)


def generic_pdf_extract_aligned_headers(input_text, column_count, first_data_row):
    """Use a clean aligned header when one exists; otherwise use safe labels."""
    if first_data_row is None:
        return [f"COLUMN {index + 1}" for index in range(column_count)]

    raw_lines = input_text.replace("\r\n", "\n").replace("\r", "\n").split("\n")
    data_key = generic_pdf_normalized_cell(first_data_row[0])

    data_line_index = None
    for index, raw in enumerate(raw_lines):
        normalized = generic_pdf_normalized_cell(raw)
        if normalized == data_key or normalized.startswith(data_key + " "):
            data_line_index = index
            break

    if data_line_index is None:
        return [f"COLUMN {index + 1}" for index in range(column_count)]

    # Search only the nearby pre-data lines.  If one clean row already has the
    # correct width, it is a much better header than guessing from wrapped text.
    for raw in reversed(raw_lines[max(0, data_line_index - 6):data_line_index]):
        row = generic_pdf_split_explicit_columns(raw, column_count)
        if row is not None and any(row):
            return row

    return [f"COLUMN {index + 1}" for index in range(column_count)]


def parse_generic_pdf_table(input_text, column_count):
    """
    Generic 3-10 column PDF fallback.

    Candidate order:
      1. explicit/aligned rows (tabs, pipes, 2+ spaces)
      2. vertical/cell-stream rows reconstructed from first-column anchors
      3. pure newline-per-cell row-major stream
      4. conservative single-space rows

    The best reconstruction is returned with one header row.  Synthetic headers
    are intentional when the PDF header itself is wrapped/interleaved; they keep
    Extract Column aligned and let the existing Skip Header checkbox work.
    """
    if not generic_pdf_columns_supported(column_count):
        return []

    explicit_rows = parse_generic_pdf_explicit_rows(input_text, column_count)
    hybrid_rows = parse_generic_pdf_hybrid_anchor_stream(input_text, column_count)
    anchor_rows = parse_generic_pdf_anchor_stream(input_text, column_count)
    modulo_rows = parse_generic_pdf_modulo_cell_stream(input_text, column_count)
    single_space_rows = parse_generic_pdf_single_space_rows(input_text, column_count)

    candidates = [
        (explicit_rows, 5),
        # Hybrid row/column stream recovery is deliberately ahead of the old
        # anchor parser when it succeeds; it is the format that caused adjacent
        # columns to be concatenated in the user's 01..07 contact table.
        (hybrid_rows, 4),
        (anchor_rows, 3),
    ]

    # A first-column anchor is much stronger evidence than modulo grouping.
    # Only use the pure modulo/cell-stream fallback when anchor reconstruction
    # is unavailable; otherwise extra newlines can make an incorrect modulo
    # alignment appear to contain more rows than the real table.
    if not hybrid_rows and not anchor_rows and not explicit_rows:
        candidates.append((modulo_rows, 2))

    candidates.append((single_space_rows, 1))

    best_rows = []
    best_score = None

    for candidate_rows, priority in candidates:
        score = generic_pdf_candidate_score(candidate_rows, column_count, priority)
        if best_score is None or score > best_score:
            best_score = score
            best_rows = candidate_rows

    if not best_rows:
        return []

    headers = generic_pdf_extract_aligned_headers(
        input_text,
        column_count,
        best_rows[0] if best_rows else None,
    )

    return [headers] + best_rows


# =========================================
# GENERAL TABLE PARSER
# =========================================

def parse_table():
    input_text = input_box.get("1.0", "end-1c")

    if input_text == "":
        return [], 0, "NONE"

    input_text = input_text.replace("\r\n", "\n")
    input_text = input_text.replace("\r", "\n")

    manual_columns = get_manual_column_count()

    rows = []
    delimiter_name = ""

    # =========================================
    # CONTACT ALLOCATION / UNIT SIGNAL TABLE
    #
    # Run this before the generic manual-column parser.  The PDF frequently
    # inserts newlines at arbitrary token boundaries, so physical-line grouping
    # is not reliable.  Sequential Contact Designations provide the row frame.
    # =========================================

    if (manual_columns in (None, 5)) and looks_like_contact_allocation_table(input_text):
        rows = parse_contact_allocation_table(input_text)
        if len(rows) > 2:
            return rows, 5, "PDF CONTACT ALLOCATION TABLE / TOKEN-ANCHORED"

    # =========================================
    # MANUAL 3-10 COLUMN OVERRIDE
    #
    # If the user explicitly entered a column count, honor that before any
    # format-specific detector.  This prevents a 5-column table from being
    # accidentally stolen by a 3- or 4-column pinout parser just because some
    # header words overlap.  The generic parser is newline/cell-stream aware.
    # =========================================

    if generic_pdf_columns_supported(manual_columns):
        rows = parse_generic_pdf_table(input_text, manual_columns)
        if len(rows) > 2:
            label = f"GENERIC PDF TABLE / {manual_columns} COLUMNS"
            if looks_highly_verticalized(input_text):
                label += " / NEWLINE CELL STREAM"
            return rows, manual_columns, label

    # =========================================
    # PDF 4-COLUMN PIN / DESCRIPTION / FIGURE / LOGIC TABLE FIRST
    #
    # This must run before the 3-column Figure-table parser because both formats
    # contain Pin Number / Description / Figure headers.
    # =========================================

    if looks_like_pdf_pin_description_figure_logic_table(input_text):

        rows = parse_pdf_pin_description_figure_logic_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE / LOGIC TABLE"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 4, label

    # =========================================
    # HEADERLESS 4-COLUMN PIN / DESCRIPTION / FIGURE / LOGIC CONTINUATION
    # =========================================

    if looks_like_headerless_pin_description_figure_logic_table(input_text):

        rows = parse_headerless_pin_description_figure_logic_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE / LOGIC TABLE / HEADERLESS CONTINUATION"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 4, label

    # =========================================
    # PDF 3-COLUMN PIN / DESCRIPTION / FIGURE TABLE FIRST
    #
    # This format can be copied from the PDF in a staggered tagged-text order,
    # so it must be reconstructed before the generic spaced-table fallback.
    # =========================================

    if looks_like_pdf_pin_description_figure_table(input_text):

        rows = parse_pdf_pin_description_figure_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE TABLE"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 3, label

    # =========================================
    # HEADERLESS 3-COLUMN PIN / DESCRIPTION / FIGURE CONTINUATION
    #
    # Handles body-only selections such as:
    #   47   Chassis GND   -
    #   13   Chassis GND   -
    #   31   Chassis GND   -
    # where the table header is on a previous page/section and is not copied.
    # This also handles PDF clipboard streams that merge repeated descriptions
    # or many '-' Figure cells onto a single line.
    # =========================================

    if looks_like_headerless_pin_description_figure_table(input_text):

        rows = parse_headerless_pin_description_figure_table(input_text)

        if len(rows) > 1:
            label = "PDF PIN / DESCRIPTION / FIGURE TABLE / HEADERLESS CONTINUATION"
            if looks_highly_verticalized(input_text):
                label += " / INTERLEAVED PDF PASTE"
            return rows, 3, label

    # =========================================
    # PDF 4-COLUMN SIMPLE / POWER PINOUT TABLE FIRST
    #
    # Example: Pin # | Signal Name | Signal Type | Signal Description.
    # This has no Ref Des and no Direction column.
    # =========================================

    if looks_like_pdf_simple_pinout_table(input_text):

        rows = parse_pdf_simple_pinout_table(input_text)

        if len(rows) > 1:
            label = "PDF SIMPLE PINOUT TABLE"
            if looks_highly_verticalized(input_text):
                label += " / VERTICAL PASTE"
            return rows, 4, label

    # =========================================
    # PDF 6-COLUMN CONNECTOR PINOUT TABLE
    #
    # This must run before the wiring-table and CSV checks. Pinout tables can
    # contain commas inside Description and can have a vertically merged Ref Des.
    # =========================================

    if looks_like_pdf_pinout_table(input_text):

        rows = parse_pdf_pinout_table(input_text)

        if len(rows) > 1:
            label = "PDF PINOUT TABLE"
            if looks_highly_verticalized(input_text):
                label += " / VERTICAL PASTE"
            return rows, 6, label

    # =========================================
    # PDF WIRING TABLE
    #
    # This MUST happen before generic TAB parsing because PDF clipboard data can
    # contain tabs without actually preserving table structure.
    # =========================================

    if looks_like_flat_pdf_wiring_table(input_text):

        rows = parse_flat_pdf_wiring_table(input_text)

        if len(rows) > 1:
            return rows, 9, "PDF WIRING TABLE"

    # =========================================
    # TRUE TAB-SEPARATED DATA
    # =========================================

    if "\t" in input_text:
        delimiter_name = "TAB"

        for line in input_text.split("\n"):
            rows.append(line.split("\t"))

    # =========================================
    # CSV
    #
    # Use structural CSV detection instead of `if "," in input_text` so a
    # normal Description such as "pair 4/D, negative" does not hijack parsing.
    # =========================================

    elif looks_like_real_csv(input_text):
        delimiter_name = "COMMA"

        reader = csv.reader(io.StringIO(input_text))

        for row in reader:
            rows.append(row)

    # =========================================
    # GENERIC SPACE-ALIGNED TABLE
    # =========================================

    elif manual_columns is not None and manual_columns > 1:
        delimiter_name = "SPACED TABLE"

        for line in input_text.split("\n"):

            if line.strip() == "":
                continue

            row = re.split(r" {2,}", line.strip())

            rows.append(row)

    # =========================================
    # SINGLE COLUMN
    # =========================================

    else:
        delimiter_name = "LINES"

        for line in input_text.split("\n"):
            rows.append([line])

    # =========================================
    # COLUMN COUNT
    # =========================================

    if manual_columns is not None:
        column_count = manual_columns

    else:
        column_count = 0

        for row in rows:

            if len(row) > column_count:
                column_count = len(row)

    # =========================================
    # NORMALIZE ALL ROW WIDTHS
    # =========================================

    normalized_rows = []

    for row in rows:

        normalized_row = list(row)

        while len(normalized_row) < column_count:
            normalized_row.append("")

        if len(normalized_row) > column_count:
            normalized_row = normalized_row[:column_count]

        normalized_rows.append(normalized_row)

    return normalized_rows, column_count, delimiter_name


# =========================================
# DETECT TABLE
# =========================================

def detect_table():
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        column_selector["values"] = []
        column_selector.set("")

        status_label.config(
            text="No table data found."
        )

        return

    column_choices = []

    first_row = rows[0]

    for index in range(column_count):
        header = ""

        if index < len(first_row):
            header = str(first_row[index]).strip()

        if header == "":
            column_choices.append(str(index + 1))

        else:
            column_choices.append(
                f"{index + 1}: {header}"
            )

    column_selector["values"] = column_choices

    if column_count > 0:
        column_selector.current(0)

    structured_pdf_table = (
        delimiter_name == "PDF WIRING TABLE"
        or delimiter_name.startswith("PDF PINOUT TABLE")
        or delimiter_name.startswith("PDF SIMPLE PINOUT TABLE")
        or delimiter_name.startswith("PDF PIN / DESCRIPTION / FIGURE TABLE")
        or delimiter_name.startswith("PDF PIN / DESCRIPTION / FIGURE / LOGIC TABLE")
        or delimiter_name.startswith("GENERIC PDF TABLE")
    )

    detected_row_count = len(rows) - 1 if structured_pdf_table else len(rows)

    status_label.config(
        text=f"Detected {detected_row_count} rows x {column_count} columns | {delimiter_name}"
    )


def get_selected_column():
    selected = column_selector.get().strip()

    if selected == "":
        return None

    selected_number = selected.split(":")[0].strip()

    try:
        return int(selected_number)

    except ValueError:
        return None


# =========================================
# EXTRACT SELECTED COLUMN
# =========================================

def extract_column():
    global current_count
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        status_label.config(
            text="No table data found."
        )

        return

    selected_column = get_selected_column()

    if selected_column is None:

        detect_table()

        selected_column = get_selected_column()

    if selected_column is None:
        status_label.config(
            text="Select a column."
        )

        return

    if selected_column < 1 or selected_column > column_count:
        status_label.config(
            text=f"Column must be between 1 and {column_count}."
        )

        return

    start_row = 0

    if skip_header_var.get():
        start_row = 1

    output_values = []

    for row_index in range(start_row, len(rows)):
        row = rows[row_index]

        if selected_column - 1 < len(row):
            value = str(row[selected_column - 1]).strip()

        else:
            value = ""

        output_values.append(value)

    output_text = "\n".join(output_values)

    output_box.delete("1.0", tk.END)
    output_box.insert(tk.END, output_text)

    current_count = len(output_values)

    blank_count = 0

    for value in output_values:

        if value.strip() == "":
            blank_count += 1

    count_label.config(
        text=f"Items: {current_count}"
    )

    status_label.config(
        text=f"Column {selected_column} extracted | Blank cells: {blank_count}"
    )

def find_missing_number(nums):
    nums = sorted(nums)
    for i in range(len(nums) - 1):
        if nums[i+1] != nums[i] + 1:
            return nums[i] + 1
    return None  # No missing number found

#!/usr/bin/env python3
"""Compare two Excel/CSV spreadsheets and generate HTML and CSV diff reports.

Requires Python 3.9+; no third-party packages.

Usage:
    python spreadsheet_diff.py                   # Pick files with dialog boxes
    python spreadsheet_diff.py old.xlsx new.xlsx
    python spreadsheet_diff.py old.xlsx new.xlsx -o my_report.html
    python spreadsheet_diff.py old.xlsx new.xlsx --ignore-cached

Supported inputs: .xlsx, .xlsm, .csv, .tsv (not legacy .xls or .xlsb).
Compares nonempty cells at their exact coordinates, including formulas and
cached formula results by default. Cell formatting is not compared.
"""

from __future__ import annotations

import argparse
import csv
from dataclasses import dataclass
from datetime import datetime, timedelta
from decimal import Decimal, InvalidOperation
from collections import Counter
from html import escape
from pathlib import Path
import posixpath
import re
import sys
import zipfile
from xml.etree import ElementTree as ET

MAIN = "http://schemas.openxmlformats.org/spreadsheetml/2006/main"
DOCREL = "http://schemas.openxmlformats.org/officeDocument/2006/relationships"
PKGREL = "http://schemas.openxmlformats.org/package/2006/relationships"
MAX_HTML_ROWS = 5000


def tag(local):
    return f"{{{MAIN}}}{local}"


@dataclass(frozen=True)
class Cell:
    kind: str             # text / number / boolean / error / formula / date
    raw: str              # normalized comparable value
    shown: str            # readable representation
    cached: str = ""      # formula result (may be absent / stale)
    formula_kind: str = ""  # shared formula information when applicable


def number_key(s):
    try:
        return str(Decimal(s).normalize())
    except (InvalidOperation, ValueError):
        return s


def col_number(cell_ref):
    m = re.match(r"([A-Za-z]+)", cell_ref)
    if not m:
        return 0
    n = 0
    for ch in m.group(1).upper():
        n = 26 * n + ord(ch) - ord("A") + 1
    return n


def address_key(ref):
    m = re.fullmatch(r"([A-Za-z]+)([0-9]+)", ref)
    return (int(m.group(2)), col_number(ref)) if m else (float("inf"), ref)


def date_style_formats(z):
    """Recognize conventional Excel date formats for legible output."""
    try:
        root = ET.fromstring(z.read("xl/styles.xml"))
    except KeyError:
        return {}
    builtins = set(range(14, 23)) | set(range(27, 37)) | set(range(45, 48)) | set(range(50, 59))
    custom = {}
    numfmts = root.find(tag("numFmts"))
    if numfmts is not None:
        for f in numfmts.findall(tag("numFmt")):
            custom[int(f.get("numFmtId", "0"))] = f.get("formatCode", "")
    result = {}
    xfs = root.find(tag("cellXfs"))
    if xfs is None:
        return result
    for index, xf in enumerate(xfs.findall(tag("xf"))):
        n = int(xf.get("numFmtId", "0"))
        fmt = custom.get(n, "")
        # Strip format literals and color/condition brackets before matching.
        fmt_clean = re.sub(r'"[^"]*"|\\.|\[[^\]]*\]', "", fmt)
        is_date = n in builtins or bool(re.search(r"[ymdhHsS]", fmt_clean))
        if is_date:
            result[index] = "time" if (n in {18, 19, 20, 21, 45, 46, 47} or
                                          ("h" in fmt_clean.lower() and "d" not in fmt_clean.lower() and "y" not in fmt_clean.lower())) else "date"
    return result


def excel_date(raw, date_kind, use_1904):
    try:
        d = Decimal(raw)
        serial = float(d)
        if not (-100000 < serial < 2000000):
            return None
        if use_1904:
            origin = datetime(1904, 1, 1)
        else:
            origin = datetime(1899, 12, 30) if serial >= 60 else datetime(1899, 12, 31)
        dt = origin + timedelta(days=serial)
        if date_kind == "time":
            return dt.strftime("%H:%M:%S")
        if d % 1:
            return dt.strftime("%Y-%m-%d %H:%M:%S")
        return dt.strftime("%Y-%m-%d")
    except (ValueError, OverflowError, InvalidOperation):
        return None


def shared_strings(z):
    try:
        with z.open("xl/sharedStrings.xml") as stream:
            root = ET.parse(stream).getroot()
    except KeyError:
        return []
    strings = []
    for item in root.findall(tag("si")):
        strings.append("".join(t.text or "" for t in item.iter(tag("t"))))
    return strings


def column_label(index):
    letters = ""
    while index:
        index, digit = divmod(index - 1, 26)
        letters = chr(65 + digit) + letters
    return letters


def expand_shared_formula(formula, base_ref, target_ref):
    """Translate the usual A1/$A$1 references used in Excel shared formulas."""
    old_row, old_col = address_key(base_ref)
    new_row, new_col = address_key(target_ref)
    dr, dc = new_row - old_row, new_col - old_col
    if dr == 0 and dc == 0:
        return formula
    pattern = re.compile(r"(?<![A-Za-z0-9_.])(\$?)([A-Za-z]{1,3})(\$?)([0-9]{1,7})(?![A-Za-z0-9_.]|\s*\()")

    def replace_ref(match):
        dollar_col, letters, dollar_row, row = match.groups()
        col = col_number(letters)
        nrow = int(row)
        if not 1 <= col <= 16384 or not 1 <= nrow <= 1048576:
            return match.group(0)
        ncol = col if dollar_col else col + dc
        nrow = nrow if dollar_row else nrow + dr
        if not (1 <= ncol <= 16384 and 1 <= nrow <= 1048576):
            return "#REF!"
        return f"{dollar_col}{column_label(ncol)}{dollar_row}{nrow}"

    # Preserve quoted text and quoted worksheet names.
    parts = re.split(r'''("(?:[^"]|"")*"|'(?:[^']|'')*')''', formula)
    return "".join(p if p.startswith(('"', "'")) else pattern.sub(replace_ref, p) for p in parts)


def xml_cell(element, strings, date_styles, use_1904, shared_masters):
    typ = element.get("t", "n")
    value_el = element.find(tag("v"))
    raw = value_el.text if value_el is not None and value_el.text is not None else ""
    formula_el = element.find(tag("f"))
    if formula_el is not None:
        formula = formula_el.text or ""
        ftype = formula_el.get("t", "")
        si = formula_el.get("si", "")
        if ftype == "shared":
            if formula:
                shared_masters[si] = (element.get("r", ""), formula)
            elif si in shared_masters:
                base_ref, base_formula = shared_masters[si]
                formula = expand_shared_formula(base_formula, base_ref, element.get("r", ""))
            else:
                formula = f"[shared formula group {si}; expression not stored here]"
        shown = "=" + formula
        if raw:
            shown += "  [cached: " + raw + "]"
        return Cell("formula", formula, shown, raw, ftype)
    if typ == "inlineStr":
        inline = element.find(tag("is"))
        raw = "" if inline is None else "".join(t.text or "" for t in inline.iter(tag("t")))
        return Cell("text", raw, raw) if raw else None
    if typ == "s":
        try:
            text = strings[int(raw)]
        except (IndexError, ValueError):
            text = f"[invalid shared-string index: {raw}]"
        return Cell("text", text, text) if text else None
    if typ in ("str", "e", "d"):
        if not raw:
            return None
        return Cell({"str": "text", "e": "error", "d": "date"}[typ], raw, raw)
    if typ == "b":
        return Cell("boolean", "1" if raw in ("1", "true", "TRUE") else "0",
                    "TRUE" if raw in ("1", "true", "TRUE") else "FALSE")
    if raw:
        style_num = int(element.get("s", "0"))
        rendered_date = excel_date(raw, date_styles[style_num], use_1904) if style_num in date_styles else None
        display = f"{rendered_date} (serial {raw})" if rendered_date else raw
        return Cell("number", number_key(raw), display)
    return None  # A formatted but empty cell isn't a content change.


def load_excel(path):
    sheets = {}
    with zipfile.ZipFile(path) as z:
        strings = shared_strings(z)
        dates = date_style_formats(z)
        workbook = ET.fromstring(z.read("xl/workbook.xml"))
        wbpr = workbook.find(tag("workbookPr"))
        use_1904 = wbpr is not None and wbpr.get("date1904", "0").lower() in ("1", "true")
        relationships = ET.fromstring(z.read("xl/_rels/workbook.xml.rels"))
        targets = {}
        for r in relationships.findall(f"{{{PKGREL}}}Relationship"):
            dest = r.get("Target", "")
            if r.get("TargetMode") == "External":
                continue
            if dest.startswith("/"):
                part = dest.lstrip("/")
            else:
                part = posixpath.normpath(posixpath.join("xl", dest))
            targets[r.get("Id")] = part
        ss = workbook.find(tag("sheets"))
        if ss is None:
            return sheets
        for sheet_el in ss.findall(tag("sheet")):
            name = sheet_el.get("name", "Unnamed")
            rel = sheet_el.get(f"{{{DOCREL}}}id")
            part = targets.get(rel)
            if not part or not part.startswith("xl/worksheets/"):
                continue  # Ignore non-worksheet entries such as chartsheets.
            cells = {}
            shared_masters = {}
            with z.open(part) as stream:
                # Parsing only cell nodes keeps memory reasonable for large sheets.
                for event, elem in ET.iterparse(stream, events=("end",)):
                    if elem.tag == tag("c"):
                        ref = elem.get("r", "")
                        cell = xml_cell(elem, strings, dates, use_1904, shared_masters)
                        if ref and cell is not None:
                            cells[ref.upper()] = cell
                        elem.clear()
            sheets[name] = cells
    return sheets


def load_delimited(path, delimiter):
    cells = {}
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        for r, row in enumerate(csv.reader(stream, delimiter=delimiter), start=1):
            for c, value in enumerate(row, start=1):
                if not value:
                    continue
                n = c
                letters = ""
                while n:
                    n, digit = divmod(n - 1, 26)
                    letters = chr(65 + digit) + letters
                cells[f"{letters}{r}"] = Cell("text", value, value)
    return {"Data": cells}


def load_spreadsheet(path):
    suffix = path.suffix.lower()
    if suffix in (".xlsx", ".xlsm"):
        return load_excel(path)
    if suffix in (".csv", ".tsv"):
        return load_delimited(path, "," if suffix == ".csv" else "\t")
    raise ValueError("Unsupported file type. Use .xlsx, .xlsm, .csv, or .tsv (not .xls/.xlsb).")


def classify(old, new, ignore_cached):
    if old is None:
        return "ADDED"
    if new is None:
        return "REMOVED"
    if old.kind == new.kind and old.raw == new.raw:
        if old.kind == "formula" and not ignore_cached and old.cached != new.cached:
            return "CACHED RESULT CHANGED"
        return None
    if old.kind == "formula" or new.kind == "formula":
        return "FORMULA CHANGED"
    return "CHANGED"


def excel_safe(value):
    """Prevent CSV cells from becoming executable spreadsheet formulas."""
    value = str(value)
    if value and value.lstrip(" \t\r\n").startswith(("=", "+", "-", "@")):
        return "'" + value
    return value


def choose_paths(args):
    if args.old and args.new:
        old = Path(args.old).expanduser()
        new = Path(args.new).expanduser()
    elif not args.old and not args.new:
        try:
            from tkinter import Tk, filedialog
            root = Tk()
            root.withdraw()
            root.attributes("-topmost", True)
            allowed = [("Excel and CSV files", "*.xlsx *.xlsm *.csv *.tsv"), ("All files", "*.*")]
            old_name = filedialog.askopenfilename(title="Select the OLD spreadsheet", filetypes=allowed, parent=root)
            if not old_name:
                root.destroy()
                raise SystemExit("No old spreadsheet selected.")
            new_name = filedialog.askopenfilename(title="Select the NEW spreadsheet", filetypes=allowed, parent=root)
            if not new_name:
                root.destroy()
                raise SystemExit("No new spreadsheet selected.")
            old, new = Path(old_name), Path(new_name)
            if args.output is None:
                out_name = filedialog.asksaveasfilename(title="Save the diff report", parent=root,
                    defaultextension=".html", filetypes=[("HTML report", "*.html")],
                    initialfile="spreadsheet_diff_report.html")
                if not out_name:
                    root.destroy()
                    raise SystemExit("Report save canceled.")
                args.output = out_name
            root.destroy()
        except (ImportError, RuntimeError, OSError):
            print("File dialogs unavailable; enter file paths instead.")
            old = Path(input("Path to OLD spreadsheet: ").strip().strip('"')).expanduser()
            new = Path(input("Path to NEW spreadsheet: ").strip().strip('"')).expanduser()
    else:
        raise SystemExit("Provide both old and new paths, or neither to use file dialogs.")
    for file in (old, new):
        if not file.is_file():
            raise FileNotFoundError(f"File not found: {file}")
    out = Path(args.output).expanduser() if args.output else new.parent / "spreadsheet_diff_report.html"
    if out.suffix.lower() != ".html":
        out = out.with_suffix(".html")
    return old, new, out

def make_report(old_path, new_path, html_path, ignore_cached=False):
    old_sheets = load_spreadsheet(old_path)
    new_sheets = load_spreadsheet(new_path)
    all_sheets = sorted(set(old_sheets) | set(new_sheets), key=str.casefold)
    stats = Counter()
    summary = []
    changes = []
    for sheet in all_sheets:
        old = old_sheets.get(sheet, {})
        new = new_sheets.get(sheet, {})
        sheet_count = 0
        for addr in sorted(set(old) | set(new), key=address_key):
            oc = old.get(addr)
            nc = new.get(addr)
            change = classify(oc, nc, ignore_cached)
            if change is None:
                continue
            stats[change] += 1
            sheet_count += 1
            changes.append((sheet, addr, change, oc.shown if oc else "", nc.shown if nc else ""))
        state = ("ADDED SHEET" if sheet not in old_sheets else
                 "REMOVED SHEET" if sheet not in new_sheets else "EXISTING")
        summary.append((sheet, state, len(old), len(new), sheet_count))
    total = len(changes)
    html_path.parent.mkdir(parents=True, exist_ok=True)
    csv_path = html_path.with_suffix(".csv")
    with csv_path.open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.writer(stream)
        writer.writerow(["Worksheet", "Cell", "Change type", "Old value", "New value"])
        for row in changes:
            writer.writerow([excel_safe(v) for v in row])
    with html_path.open("w", encoding="utf-8") as stream:
        stream.write(html_top(old_path, new_path, stats, summary, total))
        for sheet, addr, change, ov, nv in changes[:MAX_HTML_ROWS]:
            klass = change.replace(" ", "-")
            stream.write("<tr>" + f"<td>{escape(sheet)}</td><td>{escape(addr)}</td>"
                         + f'<td><span class="badge {klass}">{escape(change)}</span></td>'
                         + f'<td class="value">{escape(ov)}</td><td class="value">{escape(nv)}</td></tr>\n')
        stream.write(html_end(min(MAX_HTML_ROWS, total), total))
    return csv_path, total, stats, summary


def main():
    parser = argparse.ArgumentParser(description="Compare every cell in two Excel or CSV workbooks.")
    parser.add_argument("old", nargs="?", help="Old workbook (.xlsx/.xlsm/.csv/.tsv)")

        print(f"Error: {exc}", file=sys.stderr)

        print(f"  {name}: {nchanges:,} changes ({state.lower()})")
    print(f"HTML report: {report.resolve()}")
    print(f"Full CSV log: {csv_report.resolve()}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

def parse_pasted_column_names(raw_text):
    raw_text = raw_text.replace("\r\n", "\n")
    raw_text = raw_text.replace("\r", "\n")
    raw_text = raw_text.strip()
    if raw_text == "":
        return []
    lines = [line.strip() for line in raw_text.split("\n") if line.strip() != ""]
    if len(lines) > 1:
        return lines
    one_line = lines[0] if lines else ""
    if "\t" in one_line:
        return [value.strip() for value in one_line.split("\t") if value.strip() != ""]
    if "," in one_line:
        return [value.strip() for value in one_line.split(",") if value.strip() != ""]
    spaced = [value.strip() for value in re.split(r"\s{2,}", one_line) if value.strip() != ""]
    if len(spaced) > 1:
        return spaced
    return [one_line] if one_line else []

def extract_connector_names(value):
    """
    Return every connector designator found in a PDF/Excel label.

    Examples handled:
        J1R
        J1R / J2R
        J1R, J2R
        J1R & J2R
        SCP-J1R / SCP-J2R

    Keeping this as a list is important because some PDF tables use one
    shared pinout for two (or more) connectors.
    """
    text = str(value or "").strip().upper()
    connectors = []

    for match in re.finditer(r"J\s*0*(\d+)\s*([A-Z]*)\b", text):
        number = int(match.group(1))
        suffix = match.group(2)
        connector = f"J{number}{suffix}"

        if connector not in connectors:
            connectors.append(connector)

    return connectors


def canonical_connector_name(value):
    connectors = extract_connector_names(value)

    if connectors:
        return connectors[0]

    return str(value or "").strip().upper()


def connector_sort_key(value):

    return 999999, str(canonical)

def normalize_direction_value(value):
    text = str(value or "").strip().upper()
    text = text.replace("\\", "/")
    text = re.sub(r"\s+", " ", text)
    compact = re.sub(r"[\s_-]+", "", text)

    # Common wiring-table values
    if compact in {"IN/OUT", "INOUT", "I/O", "IO", "BIDIR", "BIDIRECTIONAL"}:
        return "IN/OUT"
    if compact in {"IN", "INPUT"}:
        return "IN"
    if compact in {"OUT", "OUTPUT"}:
        return "OUT"

    if re.fullmatch(r"IN\s*/\s*OUT", text):
        return "IN/OUT"

    # NC, "-", blank, etc. intentionally remain blank.
    return ""


def is_signal_prefix_candidate(value):
    text = str(value or "").strip()

    if text == "":
        return False

    if re.fullmatch(r"[+-]?\d+(?:\.\d+)?", text):
        return False

    if re.fullmatch(r"J\d+", canonical_connector_name(text), re.IGNORECASE):
        return False

    if normalize_direction_value(text) != "":
        return False

    return True


def combine_local_signal_fragments(fragments):
    cleaned = []

    for fragment in fragments:
        value = normalize_local_signal_value(fragment)

        if value == "":
            continue

        if normalize_direction_value(value) != "":
            continue

        cleaned.append(value)

    if len(cleaned) == 0:
        return ""

    combined = cleaned[0]

    for fragment in cleaned[1:]:
        combined_upper = combined.upper()
        fragment_upper = fragment.upper()

        if combined_upper == "CTU" and fragment_upper == "RTN":
            combined = combined + " " + fragment
        elif fragment.startswith("("):
            combined = combined + " " + fragment
        elif combined.endswith("("):
            combined = combined + fragment
        elif combined.endswith(("_", "-", "/", "#")):
            combined = combined + fragment
        elif fragment.startswith(("_", "-", "/", "#")):
            combined = combined + fragment
        else:
            combined = combined + fragment

    return normalize_local_signal_value(combined)


def extract_local_signal_from_words(row_words, boundaries, signal_index):
    if signal_index < 0 or signal_index >= len(boundaries) - 1:
        return ""

    left = boundaries[signal_index]
    right = boundaries[signal_index + 1]
    sorted_words = sorted(row_words, key=lambda word: (float(word["top"]), float(word["x0"])))
    line_groups = []

    for word in sorted_words:
        top = float(word["top"])

        if not line_groups or abs(top - line_groups[-1]["top"]) > 3.0:
            line_groups.append({"top": top, "words": [word]})
        else:
            line_groups[-1]["words"].append(word)

    fragments = []

    for line in line_groups:
        words = sorted(line["words"], key=lambda word: float(word["x0"]))
        seed_indexes = []

        for index, word in enumerate(words):
            value = str(word.get("text", "")).strip()

            if value == "":
                continue

            if re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
                continue

            x0 = float(word["x0"])
            x1 = float(word["x1"])

            if x1 > left and x0 < right:
                seed_indexes.append(index)

        if not seed_indexes:
            continue

        start_index = min(seed_indexes)
        end_index = max(seed_indexes)

        while start_index > 0:
            previous_word = words[start_index - 1]
            current_word = words[start_index]
            previous_value = str(previous_word.get("text", "")).strip()

            if not is_signal_prefix_candidate(previous_value):
                break

            previous_x1 = float(previous_word["x1"])
            current_x0 = float(current_word["x0"])
            gap = current_x0 - previous_x1

            if gap > 8.0:
                break

            if previous_x1 < left - 35.0:
                break

            start_index -= 1

        selected_words = words[start_index:end_index + 1]
        line_text = ""
        previous_x1 = None

        for word in selected_words:
            value = str(word.get("text", "")).strip()

            if value == "":
                continue

            if re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
                continue

            x0 = float(word["x0"])
            x1 = float(word["x1"])

            if line_text != "" and previous_x1 is not None:
                gap = x0 - previous_x1

                if gap > 2.5:
                    line_text += " "

            line_text += value
            previous_x1 = x1

        line_text = normalize_local_signal_value(line_text)

        if line_text == "":
            continue

        if normalize_direction_value(line_text) != "":
            continue

        fragments.append({"top": float(line["top"]), "text": line_text})

    fragments.sort(key=lambda item: item["top"])
    fragment_values = [item["text"] for item in fragments]

    return combine_local_signal_fragments(fragment_values)


def find_direction_in_column_from_words(row_words, boundaries, column_index):
    if column_index < 0 or column_index >= len(boundaries) - 1:
        return ""

    left = boundaries[column_index]
    right = boundaries[column_index + 1]
    target_center = (left + right) / 2.0
    nearby = []

    for word in row_words:
        value = str(word.get("text", "")).strip()

        if value == "":
            continue

        x0 = float(word["x0"])
        x1 = float(word["x1"])
        center = (x0 + x1) / 2.0

        if center < left or center >= right:
            continue

        nearby.append({"text": value, "top": float(word["top"]), "x0": x0, "x1": x1, "center": center})

    nearby.sort(key=lambda item: (item["top"], item["x0"]))
    candidates = []

    for item in nearby:
        normalized = normalize_direction_value(item["text"])

        if normalized != "":
            candidates.append((abs(item["center"] - target_center), normalized))

    for index in range(len(nearby)):
        for length in (2, 3):
            end = index + length

            if end > len(nearby):
                continue

            group = nearby[index:end]

            if max(item["top"] for item in group) - min(item["top"] for item in group) > 3.0:
                continue

            combined = "".join(item["text"] for item in group)
            normalized = normalize_direction_value(combined)

            if normalized == "":
                continue

            group_center = sum(item["center"] for item in group) / len(group)
            candidates.append((abs(group_center - target_center), normalized))

    if not candidates:
        return ""

    candidates.sort(key=lambda item: item[0])

    return candidates[0][1]


def find_direction_from_row_words(row_words, boundaries, direction_index):
    normal_value = find_direction_in_column_from_words(row_words, boundaries, direction_index)

    if normal_value != "":
        return normal_value

    previous_column_index = direction_index - 1

    if previous_column_index < 0:
        return ""

    previous_value = find_direction_in_column_from_words(row_words, boundaries, previous_column_index)

    if previous_value != "":
        return previous_value

    return ""


def extract_role_text_from_words(row_words, boundaries, column_index, left_pad=0.0, right_pad=0.0, skip_numeric=False):
    if column_index < 0 or column_index >= len(boundaries) - 1:
        return ""
    left = boundaries[column_index] - left_pad
    right = boundaries[column_index + 1] + right_pad
    candidates = []
    for word in row_words:
        value = str(word.get("text", "")).strip()
        if value == "":
            continue
        x0 = float(word["x0"])
        if x0 < left or x0 >= right:
            continue
        if skip_numeric and re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
            continue
        candidates.append(word)
    if not candidates:
        return ""
    candidates.sort(key=lambda word: (float(word["top"]), float(word["x0"])))
    line_groups = []
    for word in candidates:
        top = float(word["top"])
        if not line_groups or abs(top - line_groups[-1]["top"]) > 3.0:
            line_groups.append({"top": top, "words": [word]})
        else:
            line_groups[-1]["words"].append(word)
    rebuilt_lines = []
    for line in line_groups:
        line["words"].sort(key=lambda word: float(word["x0"]))
        line_text = ""
        previous_x1 = None
        for word in line["words"]:
            value = str(word.get("text", "")).strip()
            if value == "":
                continue
            x0 = float(word["x0"])
            x1 = float(word["x1"])
            if line_text != "" and previous_x1 is not None:
                gap = x0 - previous_x1
                if gap > 2.5:
                    line_text += " "
            line_text += value
            previous_x1 = x1
        if line_text != "":
            rebuilt_lines.append(line_text)
    return clean_extracted_value(" ".join(value for value in rebuilt_lines if value != ""))

def word_center_x(word):
    return (float(word["x0"]) + float(word["x1"])) / 2.0


def char_center_x(char):
    return (float(char["x0"]) + float(char["x1"])) / 2.0


def group_words_into_lines(words, y_tolerance=3.0):
    lines = []
    sorted_words = sorted(words, key=lambda word: (float(word["top"]), float(word["x0"])))
    for word in sorted_words:
        word_top = float(word["top"])
        target_line = None
        for line in lines:
            if abs(line["top"] - word_top) <= y_tolerance:
                target_line = line
                break
        if target_line is None:
            target_line = {"top": word_top, "words": []}
            lines.append(target_line)
        target_line["words"].append(word)
    for line in lines:
        line["words"] = sorted(line["words"], key=lambda word: float(word["x0"]))
        line["text"] = " ".join(str(word["text"]) for word in line["words"]).strip()
    return lines


def group_chars_into_lines(chars, y_tolerance=2.5):
    lines = []
    sorted_chars = sorted(chars, key=lambda char: (float(char["top"]), float(char["x0"])))
    for char in sorted_chars:
        char_top = float(char["top"])
        target_line = None
        for line in lines:
            if abs(line["top"] - char_top) <= y_tolerance:
                target_line = line
                break
        if target_line is None:
            target_line = {"top": char_top, "chars": []}
            lines.append(target_line)
        target_line["chars"].append(char)
    for line in lines:
        line["chars"] = sorted(line["chars"], key=lambda char: float(char["x0"]))
    return lines


def rebuild_cell_from_chars(chars, y_tolerance=2.5):
    if not chars:
        return ""
    lines = group_chars_into_lines(chars, y_tolerance=y_tolerance)
    rebuilt_lines = []
    for line in lines:
        line_chars = line["chars"]
        widths = [float(char["x1"]) - float(char["x0"]) for char in line_chars if str(char.get("text", "")).strip() != ""]
        median_width = statistics.median(widths) if widths else 3.0
        space_gap = max(1.25, median_width * 0.55)
        line_text = ""
        previous_x1 = None
        for char in line_chars:
            value = str(char.get("text", ""))
            x0 = float(char["x0"])
            x1 = float(char["x1"])
            if value == "":
                continue
            if value.isspace():
                if line_text and not line_text.endswith(" "):
                    line_text += " "
                previous_x1 = x1
                continue
            if previous_x1 is not None:
                gap = x0 - previous_x1
                if gap > space_gap and line_text and not line_text.endswith(" "):
                    line_text += " "
            line_text += value
            previous_x1 = x1
        line_text = clean_extracted_value(line_text)
        if line_text:
            rebuilt_lines.append(line_text)
    return clean_extracted_value(" ".join(rebuilt_lines))


def find_exact_word(words, wanted_text, target_y=None, y_tolerance=60.0):
    candidates = []
    for word in words:
        text = str(word["text"]).strip().upper()
        if text != wanted_text.upper():
            continue
        if target_y is not None and abs(float(word["top"]) - target_y) > y_tolerance:
            continue
        candidates.append(word)
    if not candidates:
        return None
    if target_y is None:
        return candidates[0]
    candidates.sort(key=lambda word: abs(float(word["top"]) - target_y))
    return candidates[0]

def find_page_footer_top(lines):
    footer_tops = []
    for line in lines:
        line_text = str(line.get("text", "")).strip()
        if re.fullmatch(r"Page\s+\d+\s+of\s+\d+", line_text, re.IGNORECASE):
            footer_tops.append(float(line["top"]))
    return min(footer_tops) if footer_tops else None


def load_pdf_cache(pdf_path, start_page, end_page=None):
    start_time = time.perf_counter()
    pdf_cache = {}
    with pdfplumber.open(pdf_path) as pdf:
        total_pages = len(pdf.pages)
        if start_page > total_pages:
            raise ValueError(f"Start page {start_page} is greater than PDF page count {total_pages}")
        actual_end_page = total_pages if end_page is None else min(end_page, total_pages)
        for page_number in range(start_page, actual_end_page + 1):
            page = pdf.pages[page_number - 1]
            text = page.extract_text() or ""
            words = page.extract_words(x_tolerance=3, y_tolerance=3, keep_blank_chars=False, use_text_flow=False)
            chars = list(page.chars)
            lines = group_words_into_lines(words)
            edges = list(page.edges)
            pdf_cache[page_number] = {"text": text, "words": words, "chars": chars, "lines": lines, "edges": edges, "footer_top": find_page_footer_top(lines), "width": float(page.width), "height": float(page.height)}
    elapsed = time.perf_counter() - start_time
    print(f"PDF CACHE BUILT: {len(pdf_cache)} PAGES IN {elapsed:.2f} SECONDS")
    return pdf_cache

def detect_default_column_names(pdf_cache):
    # Prefer strong header geometry/text signatures over raw comma/space parsing.
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data.get("words", [])
        lines = page_data.get("lines", [])

        if not words:
            continue

        for line in lines:
            upper_values = {
                str(word.get("text", "")).strip().upper()
                for word in line.get("words", [])
            }

            # 6-column pinout:
            # REF DES | PIN | SIGNAL NAME | DIRECTION | TYPE | DESCRIPTION
            if {"DIRECTION", "TYPE", "DESCRIPTION"}.issubset(upper_values):
                header_y = float(line["top"])
                nearby = [
                    str(word.get("text", "")).strip().upper()
                    for word in words
                    if abs(float(word["top"]) - header_y) <= 40.0
                ]

                if (
                    "PIN" in nearby
                    and "SIGNAL" in nearby
                    and "REF" in nearby
                    and "DES" in nearby
                ):
                    print("AUTO-DETECTED TABLE FORMAT: 6-COLUMN CONNECTOR PINOUT")
                    return list(SIX_COLUMN_PINOUT_NAMES)

            # 9-column wiring:
            # FROM DES/POSN | TO DES/POSN | MNEMONIC | DESCRIPTION | WIRE | NOTE | COMMENT
            if "MNEMONIC" in upper_values and "DESCRIPTION" in upper_values:
                header_y = float(line["top"])
                nearby = [
                    str(word.get("text", "")).strip().upper()
                    for word in words
                    if abs(float(word["top"]) - header_y) <= 60.0
                ]

                if (
                    nearby.count("DES") >= 2
                    and nearby.count("POSN") >= 2
                    and "WIRE" in nearby
                    and "NOTE" in nearby
                    and "COMMENT" in nearby
                ):
                    print("AUTO-DETECTED TABLE FORMAT: 9-COLUMN WIRING TABLE")
                    return list(NINE_COLUMN_NAMES)

    # Text-level fallback for PDFs whose header words are oddly grouped.
    for page_number in sorted(pdf_cache):
        upper_text = str(pdf_cache[page_number].get("text", "") or "").upper()

        if (
            "SIGNAL NAME" in upper_text
            and "DIRECTION" in upper_text
            and "TYPE" in upper_text
            and "DESCRIPTION" in upper_text
            and "PIN" in upper_text
        ):
            print("AUTO-DETECTED TABLE FORMAT: 6-COLUMN CONNECTOR PINOUT (TEXT FALLBACK)")
            return list(SIX_COLUMN_PINOUT_NAMES)

        if (
            "MNEMONIC" in upper_text
            and "WIRE" in upper_text
            and "NOTE" in upper_text
            and "COMMENT" in upper_text
        ):
            print("AUTO-DETECTED TABLE FORMAT: 9-COLUMN WIRING TABLE (TEXT FALLBACK)")
            return list(NINE_COLUMN_NAMES)

    print("WARNING: TABLE FORMAT NOT AUTO-DETECTED; DEFAULTING TO 9-COLUMN WIRING")
    return list(DEFAULT_COLUMN_NAMES)

def extract_sections_by_keyword(pdf_cache, separator_keyword):

    for page_number in sorted(pdf_cache):

    return sections

# LEARN COLUMN GEOMETRY FROM PDF HEADER
# ================================================
def edge_vertical_length(edge):
    if "top" in edge and "bottom" in edge:
        return abs(float(edge["bottom"]) - float(edge["top"]))
    if "y0" in edge and "y1" in edge:
        return abs(float(edge["y1"]) - float(edge["y0"]))
    return 0.0


def edge_x_position(edge):
    if "x0" not in edge or "x1" not in edge:
        return None
    x0 = float(edge["x0"])
    x1 = float(edge["x1"])
    if abs(x1 - x0) > 1.5:
        return None
    return (x0 + x1) / 2.0


def cluster_vertical_edges(edges, tolerance=1.5):
    raw = []
    for edge in edges:
        x = edge_x_position(edge)
        if x is None:
            continue
        length = edge_vertical_length(edge)
        if length < 8.0:
            continue
        raw.append((x, length))
    raw.sort(key=lambda item: item[0])
    clusters = []
    for x, length in raw:
        if not clusters or abs(x - clusters[-1]["x"]) > tolerance:
            clusters.append({"x": x, "count": 1, "length": length})
        else:
            cluster = clusters[-1]
            total = cluster["count"] + 1
            cluster["x"] = ((cluster["x"] * cluster["count"]) + x) / total
            cluster["count"] = total
            cluster["length"] += length
    return clusters


def choose_grid_boundary(clusters, left_center, right_center):
    candidates = [cluster for cluster in clusters if left_center < cluster["x"] < right_center]
    if not candidates:
        return None
    midpoint = (left_center + right_center) / 2.0
    candidates.sort(key=lambda cluster: (-cluster["length"], abs(cluster["x"] - midpoint)))
    return candidates[0]["x"]


def print_learned_columns(page_number, column_names, centers):
    print("")
    print("============================================================")
    print("TABLE COLUMNS LEARNED")
    print("============================================================")
    print("PAGE:", page_number)
    for name, center in zip(column_names, centers):
        print(name, "X =", round(center, 2))
    print("============================================================")


def find_nine_column_centers(pdf_cache, column_names):
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data["words"]
        lines = page_data["lines"]
        if not words:
            continue
        for line in lines:
            upper_values = [str(word["text"]).strip().upper() for word in line["words"]]
            if "MNEMONIC" not in upper_values or "DESCRIPTION" not in upper_values:
                continue
            header_y = line["top"]
            nearby_words = [word for word in words if abs(float(word["top"]) - header_y) <= 60.0]
            des_words = [word for word in nearby_words if str(word["text"]).strip().upper() == "DES"]
            posn_words = [word for word in nearby_words if str(word["text"]).strip().upper() == "POSN"]
            des_words.sort(key=lambda word: float(word["x0"]))
            posn_words.sort(key=lambda word: float(word["x0"]))
            if len(des_words) < 2 or len(posn_words) < 2:
                continue
            mnemonic_word = find_exact_word(nearby_words, "MNEMONIC", header_y)
            description_word = find_exact_word(nearby_words, "DESCRIPTION", header_y)
            wire_word = find_exact_word(nearby_words, "WIRE", header_y)
            note_word = find_exact_word(nearby_words, "NOTE", header_y)
            comment_word = find_exact_word(nearby_words, "COMMENT", header_y)
            if mnemonic_word is None or description_word is None or wire_word is None or note_word is None or comment_word is None:
                continue
            centers = [
                word_center_x(des_words[0]),
                word_center_x(posn_words[0]),
                word_center_x(des_words[1]),
                word_center_x(posn_words[1]),
                word_center_x(mnemonic_word),
                word_center_x(description_word),
                word_center_x(wire_word),
                word_center_x(note_word),
                word_center_x(comment_word),
            ]
            if any(centers[index] >= centers[index + 1] for index in range(len(centers) - 1)):
                continue
            print_learned_columns(page_number, column_names, centers)
            return centers, page_data["width"], page_number
    raise Exception("Could not automatically determine the 9-column wiring-table locations from PDF headers")


def find_six_column_centers(pdf_cache, column_names):
    """
    Learn the 6-column connector pinout layout:
        REF DES | PIN | SIGNAL NAME | DIRECTION | TYPE | DESCRIPTION

    REF/DES can be stacked on two lines, and SIGNAL/NAME can be split into
    separate PDF words. We therefore use a vertical header window instead of
    requiring every header word to live on one text line.
    """
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data["words"]
        lines = page_data["lines"]

        if not words:
            continue

        for line in lines:
            upper_values = {str(word["text"]).strip().upper() for word in line["words"]}

            # The right side of this header is the most stable identifying pattern.
            if not {"DIRECTION", "TYPE", "DESCRIPTION"}.issubset(upper_values):
                continue

            header_y = float(line["top"])
            nearby_words = [
                word for word in words
                if abs(float(word["top"]) - header_y) <= 40.0
            ]

            pin_word = find_exact_word(nearby_words, "PIN", header_y, y_tolerance=40.0)
            direction_word = find_exact_word(nearby_words, "DIRECTION", header_y, y_tolerance=40.0)
            type_word = find_exact_word(nearby_words, "TYPE", header_y, y_tolerance=40.0)
            description_word = find_exact_word(nearby_words, "DESCRIPTION", header_y, y_tolerance=40.0)

            signal_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "SIGNAL"
            ]
            name_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "NAME"
            ]
            ref_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "REF"
            ]
            des_candidates = [
                word for word in nearby_words
                if str(word["text"]).strip().upper() == "DES"
            ]

            if pin_word is None or direction_word is None or type_word is None or description_word is None:
                continue
            if not signal_candidates or not ref_candidates or not des_candidates:
                continue

            # Pick the REF/DES pair that lives to the left of PIN.
            pin_center = word_center_x(pin_word)
            ref_candidates = [word for word in ref_candidates if word_center_x(word) < pin_center]
            des_candidates = [word for word in des_candidates if word_center_x(word) < pin_center]
            if not ref_candidates or not des_candidates:
                continue

            ref_word = min(ref_candidates, key=lambda word: abs(word_center_x(word) - pin_center))
            des_word = min(des_candidates, key=lambda word: abs(word_center_x(word) - word_center_x(ref_word)))
            ref_des_center = (word_center_x(ref_word) + word_center_x(des_word)) / 2.0

            # SIGNAL is enough to establish the column. If NAME is present nearby,
            # average the two centers for a better estimate of the visual header center.
            signal_word = min(signal_candidates, key=lambda word: abs(float(word["top"]) - header_y))
            signal_center = word_center_x(signal_word)
            if name_candidates:
                plausible_names = [
                    word for word in name_candidates
                    if abs(word_center_x(word) - signal_center) < 80.0
                ]
                if plausible_names:
                    name_word = min(plausible_names, key=lambda word: abs(word_center_x(word) - signal_center))
                    signal_center = (signal_center + word_center_x(name_word)) / 2.0

            centers = [
                ref_des_center,
                pin_center,
                signal_center,
                word_center_x(direction_word),
                word_center_x(type_word),
                word_center_x(description_word),
            ]

            if any(centers[index] >= centers[index + 1] for index in range(len(centers) - 1)):
                continue

            print_learned_columns(page_number, column_names, centers)
            return centers, page_data["width"], page_number

    raise Exception("Could not automatically determine the 6-column pinout locations from PDF headers")


def find_column_centers(pdf_cache, column_names):
    if len(column_names) == 9:
        return find_nine_column_centers(pdf_cache, column_names)

    if len(column_names) == 6:
        return find_six_column_centers(pdf_cache, column_names)

    raise ValueError(
        f"Unsupported physical column count: {len(column_names)}. "
        "This scraper supports 6-column pinout tables and 9-column wiring tables."
    )


def build_column_boundaries(pdf_cache, column_centers, page_width, header_page):
    page_data = pdf_cache[header_page]
    clusters = cluster_vertical_edges(page_data.get("edges", []))
    boundaries = []
    left_candidates = [cluster for cluster in clusters if cluster["x"] < column_centers[0]]
    right_candidates = [cluster for cluster in clusters if cluster["x"] > column_centers[-1]]
    left_boundary = max(left_candidates, key=lambda cluster: cluster["x"])["x"] if left_candidates else 0.0
    right_boundary = min(right_candidates, key=lambda cluster: cluster["x"])["x"] if right_candidates else page_width
    boundaries.append(left_boundary)
    grid_success = True
    for index in range(len(column_centers) - 1):
        boundary = choose_grid_boundary(clusters, column_centers[index], column_centers[index + 1])
        if boundary is None:
            grid_success = False
            boundary = (column_centers[index] + column_centers[index + 1]) / 2.0
        boundaries.append(boundary)
    boundaries.append(right_boundary)
    if any(boundaries[index] >= boundaries[index + 1] for index in range(len(boundaries) - 1)):
        boundaries = [0.0]
        for index in range(len(column_centers) - 1):
            boundaries.append((column_centers[index] + column_centers[index + 1]) / 2.0)
        boundaries.append(page_width)
        grid_success = False
    print("")
    print("============================================================")
    print("TABLE COLUMN BOUNDARIES")
    print("============================================================")
    print("METHOD:", "PDF GRID LINES" if grid_success else "GRID + MIDPOINT FALLBACK")
    for index, boundary in enumerate(boundaries):
        print("BOUNDARY", index, "X =", round(boundary, 2))
    print("============================================================")
    return boundaries


def get_column_index_for_x(x_value, boundaries):
    for index in range(len(boundaries) - 1):
        if boundaries[index] <= x_value < boundaries[index + 1]:
            return index
    return None

def is_connector_name(text):
    return re.fullmatch(r"J\d+[A-Z]*", canonical_connector_name(text), re.IGNORECASE) is not None


def find_connector_anchors(chars, boundaries):
    first_column_right = boundaries[1]
    first_column_chars = [char for char in chars if char_center_x(char) < first_column_right]
    lines = group_chars_into_lines(first_column_chars, y_tolerance=2.5)
    anchors = []

    for line in lines:
        value = rebuild_cell_from_chars(line["chars"])
        connectors = extract_connector_names(value)

        if not connectors:
            continue

        line_top = min(float(char["top"]) for char in line["chars"])
        line_bottom = max(float(char["bottom"]) for char in line["chars"])

        # One physical row can explicitly apply to multiple connectors, e.g.
        # "J1R / J2R".  Keep one row anchor and carry all connector aliases.
        anchors.append({
            "connector": connectors[0],
            "connectors": connectors,
            "top": line_top,
            "bottom": line_bottom,
        })

    anchors.sort(key=lambda anchor: anchor["top"])
    deduped = []

    for anchor in anchors:
        if deduped and abs(anchor["top"] - deduped[-1]["top"]) <= 2.0:
            # Merge aliases rather than dropping a second connector printed on
            # the same physical line.
            for connector in anchor.get("connectors", []):
                if connector not in deduped[-1]["connectors"]:
                    deduped[-1]["connectors"].append(connector)
            continue

        deduped.append(anchor)

    return deduped


# ================================================
# ROW GRID GEOMETRY
# ================================================
def edge_horizontal_y(edge):
    orientation = str(edge.get("orientation", "")).lower()
    if "top" in edge and "bottom" in edge:
        top = float(edge["top"])
        bottom = float(edge["bottom"])
        if orientation == "h" or abs(bottom - top) <= 1.5:
            return (top + bottom) / 2.0
    return None


def get_horizontal_table_grid_lines(edges, table_left, table_right, tolerance=1.25):
    table_width = max(1.0, table_right - table_left)
    raw = []
    for edge in edges:
        y = edge_horizontal_y(edge)
        if y is None:
            continue
        if "x0" not in edge or "x1" not in edge:
            continue
        x0 = min(float(edge["x0"]), float(edge["x1"]))
        x1 = max(float(edge["x0"]), float(edge["x1"]))
        overlap = max(0.0, min(x1, table_right) - max(x0, table_left))
        if overlap <= 0.0:
            continue
        raw.append((y, overlap))
    raw.sort(key=lambda item: item[0])
    clusters = []
    for y, overlap in raw:
        if not clusters or abs(y - clusters[-1]["y"]) > tolerance:
            clusters.append({"y": y, "overlap": overlap, "count": 1})
        else:
            cluster = clusters[-1]
            total = cluster["count"] + 1
            cluster["y"] = ((cluster["y"] * cluster["count"]) + y) / total
            cluster["overlap"] += overlap
            cluster["count"] = total
    minimum_coverage = table_width * 0.55
    return [cluster["y"] for cluster in clusters if cluster["overlap"] >= minimum_coverage]


def find_grid_row_bounds(anchor, horizontal_grid_lines):
    anchor_top = float(anchor["top"])
    anchor_bottom = float(anchor.get("bottom", anchor_top + 6.0))
    above = [y for y in horizontal_grid_lines if y < anchor_top - 0.25]
    below = [y for y in horizontal_grid_lines if y > anchor_bottom + 0.25]
    if not above or not below:
        return None
    row_top = max(above)
    row_bottom = min(below)
    row_height = row_bottom - row_top
    if row_height < 6.0 or row_height > 80.0:
        return None
    return row_top, row_bottom

# PARSE ONE ROW USING CHARACTER GEOMETRY
# ================================================
def parse_row_chars(row_chars, boundaries, column_names):
    column_chars = [[] for _ in column_names]
    for char in row_chars:
        column_index = get_column_index_for_x(char_center_x(char), boundaries)
        if column_index is None or column_index < 0 or column_index >= len(column_names):
            continue
        column_chars[column_index].append(char)
    output = {}
    for index, column_name in enumerate(column_names):
        output[column_name] = rebuild_cell_from_chars(column_chars[index])
    return output

# 6-COLUMN PINOUT ROW HELPERS
# ================================================
def find_pin_anchors(chars, boundaries, pin_index):
    if pin_index < 0 or pin_index >= len(boundaries) - 1:
        return []

    left = boundaries[pin_index]
    right = boundaries[pin_index + 1]

    pin_chars = [
        char for char in chars
        if left <= char_center_x(char) < right
    ]

    lines = group_chars_into_lines(pin_chars, y_tolerance=2.5)
    anchors = []

    for line in lines:
        value = rebuild_cell_from_chars(line["chars"])

        if not re.fullmatch(r"\d+", value):
            continue

        line_top = min(float(char["top"]) for char in line["chars"])
        line_bottom = max(float(char["bottom"]) for char in line["chars"])

        anchors.append({
            "pin": value,
            "top": line_top,
            "bottom": line_bottom,
        })

    anchors.sort(key=lambda anchor: anchor["top"])

    deduped = []

    for anchor in anchors:
        if deduped and abs(anchor["top"] - deduped[-1]["top"]) <= 2.0:
            # Keep the first numeric value found on this text line.
            continue
        deduped.append(anchor)

    return deduped


def find_connector_labels(chars, boundaries, connector_index):
    if connector_index < 0 or connector_index >= len(boundaries) - 1:
        return []

    left = boundaries[connector_index]
    right = boundaries[connector_index + 1]

    connector_chars = [
        char for char in chars
        if left <= char_center_x(char) < right
    ]

    lines = group_chars_into_lines(connector_chars, y_tolerance=2.5)
    labels = []

    for line in lines:
        raw_value = rebuild_cell_from_chars(line["chars"])
        connectors = extract_connector_names(raw_value)

        if not connectors:
            continue

        line_top = min(float(char["top"]) for char in line["chars"])
        line_bottom = max(float(char["bottom"]) for char in line["chars"])

        labels.append({
            "connector": connectors[0],
            "connectors": connectors,
            "top": line_top,
            "bottom": line_bottom,
            "center": (line_top + line_bottom) / 2.0,
        })

    labels.sort(key=lambda item: item["center"])

    deduped = []

    for label in labels:
        if deduped and abs(label["center"] - deduped[-1]["center"]) <= 2.0:
            # A combined label may be split into multiple text fragments by the
            # PDF. Merge all connector names at the same vertical location.
            for connector in label.get("connectors", []):
                if connector not in deduped[-1]["connectors"]:
                    deduped[-1]["connectors"].append(connector)
            continue

        deduped.append(label)

    return deduped


def split_pin_rows_into_groups(rows):
    if not rows:
        return []

    tops = [float(row["_anchor_top"]) for row in rows]
    gaps = [
        tops[index + 1] - tops[index]
        for index in range(len(tops) - 1)
        if tops[index + 1] - tops[index] > 0.0
    ]

    normal_gap = statistics.median(gaps) if gaps else 20.0
    large_gap_threshold = max(normal_gap * 2.5, 35.0)

    groups = [[rows[0]]]

    for row in rows[1:]:
        previous = groups[-1][-1]

        current_pin = int(row["_pin_number"])
        previous_pin = int(previous["_pin_number"])
        vertical_gap = float(row["_anchor_top"]) - float(previous["_anchor_top"])

        starts_new_group = (
            current_pin <= previous_pin
            or vertical_gap > large_gap_threshold
        )

        if starts_new_group:
            groups.append([row])
        else:
            groups[-1].append(row)

    return groups


def _unique_connectors(values):
    output = []

    for value in values:
        for connector in extract_connector_names(value):
            if connector not in output:
                output.append(connector)

    return output


def choose_connectors_for_pin_group(group, connector_labels, last_connectors=None):
    if last_connectors is None:
        last_connectors = []

    # First prefer any valid connector text that was physically inside a row.
    row_connectors = []

    for row in group:
        for connector in row.get("_parsed_connectors", []):
            if is_connector_name(connector) and connector not in row_connectors:
                row_connectors.append(connector)

    if row_connectors:
        return row_connectors

    group_top = min(float(row["_row_top"]) for row in group)
    group_bottom = max(float(row["_row_bottom"]) for row in group)
    group_center = (group_top + group_bottom) / 2.0

    inside = [
        label for label in connector_labels
        if group_top - 3.0 <= label["center"] <= group_bottom + 3.0
    ]

    if inside:
        connectors = []

        # All labels physically inside the same merged REF DES area apply to
        # this pin group. This is the key behavior for shared pinouts such as
        # J1R + J2R using one common table.
        for label in inside:
            for connector in label.get("connectors", [label.get("connector", "")]):
                if is_connector_name(connector) and connector not in connectors:
                    connectors.append(connector)

        if connectors:
            return connectors

    # If this page has one merged REF DES label, it belongs to the visible pin table.
    if len(connector_labels) == 1:
        return list(connector_labels[0].get("connectors", [connector_labels[0]["connector"]]))

    # Multiple tables can share a page. Choose the closest label cluster.
    if connector_labels:
        nearest = min(
            connector_labels,
            key=lambda label: abs(label["center"] - group_center)
        )

        # Avoid borrowing a connector from a very distant table.
        if abs(nearest["center"] - group_center) <= max(group_bottom - group_top, 80.0):
            cluster_tolerance = 12.0
            connectors = []

            for label in connector_labels:
                if abs(label["center"] - nearest["center"]) <= cluster_tolerance:
                    for connector in label.get("connectors", [label.get("connector", "")]):
                        if is_connector_name(connector) and connector not in connectors:
                            connectors.append(connector)

            if connectors:
                return connectors

    # Continuation-page fallback: if pins continue above 1 and the REF DES merged
    # cell was not repeated on the new page, keep every connector from the prior
    # shared table.
    first_pin = int(group[0]["_pin_number"])

    if last_connectors and first_pin > 1:
        return list(last_connectors)

    return []


# ================================================
# EXTRACT 6-COLUMN PINOUT TABLE
#
# REF DES is often one merged cell spanning many
# pin rows, so PIN is the row anchor instead of
# requiring the connector name on every row.
# ================================================
def extract_structured_rows_six_column(pdf_cache, separator_keyword, column_names, pdf_roles):
    start_time = time.perf_counter()

    column_centers, page_width, header_page = find_column_centers(pdf_cache, column_names)
    reference_boundaries = build_column_boundaries(
        pdf_cache,
        column_centers,
        page_width,
        header_page
    )

    section_markers = find_section_markers(pdf_cache, separator_keyword)

    connector_index = column_names.index(pdf_roles["connector"])
    pin_index = column_names.index(pdf_roles["pin"])
    signal_index = column_names.index(pdf_roles["signal"])
    skip_direction = should_skip_direction(pdf_roles)
    direction_index = None if skip_direction else column_names.index(pdf_roles["direction"])

    structured_rows = []
    grid_bounded_rows = 0
    fallback_bounded_rows = 0
    last_connectors = []

    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        chars = page_data["chars"]
        words = page_data["words"]

        if not chars:
            continue

        boundaries = reference_boundaries

        pin_anchors = find_pin_anchors(chars, boundaries, pin_index)

        if not pin_anchors:
            continue

        connector_labels = find_connector_labels(
            chars,
            boundaries,
            connector_index
        )

        horizontal_grid_lines = get_horizontal_table_grid_lines(
            page_data.get("edges", []),
            boundaries[0],
            boundaries[-1]
        )

        anchor_tops = [float(anchor["top"]) for anchor in pin_anchors]
        gaps = [
            anchor_tops[index + 1] - anchor_tops[index]
            for index in range(len(anchor_tops) - 1)
            if anchor_tops[index + 1] - anchor_tops[index] > 0
        ]

        normal_row_height = statistics.median(gaps) if gaps else 22.0
        maximum_last_row_height = max(normal_row_height * 2.75, 50.0)

        page_bottom = (
            page_data["footer_top"] - 2.0
            if page_data.get("footer_top") is not None
            else page_data["height"]
        )

        page_rows = []

        for anchor_index, anchor in enumerate(pin_anchors):
            grid_bounds = find_grid_row_bounds(anchor, horizontal_grid_lines)

            if grid_bounds is not None:
                row_top = grid_bounds[0] + 0.35
                row_bottom = grid_bounds[1] - 0.35
                grid_bounded_rows += 1
            else:
                row_top = float(anchor["top"]) - 2.0
                row_bottom = (
                    float(pin_anchors[anchor_index + 1]["top"]) - 1.0
                    if anchor_index + 1 < len(pin_anchors)
                    else float(anchor["top"]) + maximum_last_row_height
                )
                fallback_bounded_rows += 1

            row_bottom = min(row_bottom, page_bottom)

            if row_bottom <= row_top:
                continue

            row_chars = [
                char for char in chars
                if row_top <= float(char["top"]) < row_bottom
            ]

            row_words = [
                word for word in words
                if row_top <= float(word["top"]) < row_bottom
            ]

            parsed_row = parse_row_chars(
                row_chars,
                boundaries,
                column_names
            )

            # PIN is our authoritative row anchor.
            parsed_row[pdf_roles["pin"]] = str(anchor["pin"])

            signal_from_words = extract_local_signal_from_words(
                row_words,
                boundaries,
                signal_index
            )

            if signal_from_words != "":
                parsed_row[pdf_roles["signal"]] = normalize_local_signal_value(
                    signal_from_words
                )
            else:
                parsed_row[pdf_roles["signal"]] = normalize_local_signal_value(
                    parsed_row.get(pdf_roles["signal"], "")
                )

            if not skip_direction:
                direction_from_words = extract_role_text_from_words(
                    row_words,
                    boundaries,
                    direction_index,
                    left_pad=8.0,
                    right_pad=8.0,
                    skip_numeric=False
                )

                direction_value = normalize_direction_value(direction_from_words)

                if direction_value == "":
                    direction_value = normalize_direction_value(
                        parsed_row.get(pdf_roles["direction"], "")
                    )

                if direction_value == "":
                    direction_value = find_direction_from_row_words(
                        row_words,
                        boundaries,
                        direction_index
                    )

                parsed_row[pdf_roles["direction"]] = direction_value

            parsed_connectors = extract_connector_names(
                parsed_row.get(pdf_roles["connector"], "")
            )

            section_marker = find_section_for_row(
                page_number,
                float(anchor["top"]),
                section_markers
            )

            parsed_row["PAGE"] = page_number
            parsed_row["SECTION"] = (
                section_marker["section_number"]
                if section_marker
                else None
            )
            parsed_row["SECTION_TITLE"] = (
                section_marker["title"]
                if section_marker
                else ""
            )

            # Temporary geometry fields for merged REF DES assignment.
            parsed_row["_anchor_top"] = float(anchor["top"])
            parsed_row["_row_top"] = float(row_top)
            parsed_row["_row_bottom"] = float(row_bottom)
            parsed_row["_pin_number"] = int(anchor["pin"])
            parsed_row["_parsed_connectors"] = parsed_connectors

            page_rows.append(parsed_row)

        pin_groups = split_pin_rows_into_groups(page_rows)

        for group in pin_groups:
            connectors = choose_connectors_for_pin_group(
                group,
                connector_labels,
                last_connectors=last_connectors
            )

            if not connectors:
                print(
                    "WARNING: Could not assign REF DES to pin group on page",
                    page_number,
                    "pins",
                    group[0]["_pin_number"],
                    "through",
                    group[-1]["_pin_number"]
                )
                continue

            if len(connectors) > 1:
                print(
                    "SHARED CONNECTOR PINOUT:",
                    ", ".join(connectors),
                    "| PAGE",
                    page_number,
                    "| PINS",
                    group[0]["_pin_number"],
                    "THROUGH",
                    group[-1]["_pin_number"]
                )

            for parsed_row in group:
                base_row = dict(parsed_row)
                base_row.pop("_anchor_top", None)
                base_row.pop("_row_top", None)
                base_row.pop("_row_bottom", None)
                base_row.pop("_pin_number", None)
                base_row.pop("_parsed_connectors", None)

                # Duplicate the same physical PDF row under every connector
                # named by the shared REF DES cell. Excel matching is by name,
                # so J1R and J2R can appear in any spreadsheet order.
                for connector in connectors:
                    output_row = dict(base_row)
                    output_row[pdf_roles["connector"]] = connector
                    structured_rows.append(output_row)

            last_connectors = list(connectors)

    elapsed = time.perf_counter() - start_time

    print("PARSER MODE: 6-COLUMN PINOUT")
    print("GRID-BOUNDED ROWS:", grid_bounded_rows)
    print("FALLBACK-BOUNDED ROWS:", fallback_bounded_rows)
    print(f"STRUCTURED ROW EXTRACTION TIME: {elapsed:.2f} SECONDS")

    return structured_rows

def extract_structured_rows(pdf_cache, separator_keyword, column_names, pdf_roles):
    if len(column_names) == 6:
        return extract_structured_rows_six_column(
            pdf_cache,
            separator_keyword,
            column_names,
            pdf_roles
        )

    if len(column_names) == 9:
        return extract_structured_rows_nine_column(
            pdf_cache,
            separator_keyword,
            column_names,
            pdf_roles
        )

    raise ValueError(
        f"Unsupported physical column count: {len(column_names)}. "
        "Use 6 columns for pinout tables or 9 columns for wiring tables."
    )


# ================================================
# STRUCTURED DATA HELPERS
# ================================================
def validate_row(row, pdf_roles):
    problems = []
    connectors = extract_connector_names(get_role_value(row, pdf_roles, "connector"))
    pin_text = get_role_value(row, pdf_roles, "pin")

    if not connectors:
        problems.append("invalid connector")

    if not re.fullmatch(r"\d+", pin_text):
        problems.append("invalid pin")

    return problems


def validate_structured_rows(structured_rows, pdf_roles):
    good_rows = []
    suspicious_rows = []
    for row in structured_rows:
        problems = validate_row(row, pdf_roles)
        if problems:
            suspicious_rows.append({"row": row, "problems": problems})
        else:
            good_rows.append(row)
    return good_rows, suspicious_rows


def build_pdf_connector_map(structured_rows, pdf_roles):
    pdf_connectors = {}

    for row in structured_rows:
        connectors = extract_connector_names(
            get_role_value(row, pdf_roles, "connector")
        )

        for connector in connectors:
            if not is_connector_name(connector):
                continue

            # Store a connector-specific copy so a shared PDF table can safely
            # feed multiple independent spreadsheet blocks.
            mapped_row = dict(row)
            mapped_row[pdf_roles["connector"]] = connector
            pdf_connectors.setdefault(connector, []).append(mapped_row)

    return pdf_connectors


def get_connector_pin_map(rows, pdf_roles):
    pin_map = {}
    for row in rows:
        pin_text = get_role_value(row, pdf_roles, "pin")
        if not re.fullmatch(r"\d+", pin_text):
            continue
        pin_map[int(pin_text)] = row
    return pin_map

    for connector in sorted(pdf_connectors, key=connector_sort_key):
        pin_map = get_connector_pin_map(pdf_connectors[connector], pdf_roles)
        expected = expected_counts.get(connector) if expected_counts is not None else None
        if expected is not None and expected > 0:
            missing = [pin_number for pin_number in range(1, expected + 1) if pin_number not in pin_map]
            print(connector, "| FOUND:", len(pin_map), "| EXPECTED:", expected, "| MISSING:", missing)
        elif pin_map:
            highest = max(pin_map)
            missing = [pin_number for pin_number in range(1, highest + 1) if pin_number not in pin_map]
            print(connector, "| FOUND:", len(pin_map), "| HIGHEST:", highest, "| MISSING:", missing)
        else:
            print(connector, "| NO VALID PINS")
    print("============================================================")


def print_structured_rows(structured_rows, column_names):
    current_section = object()
    for row in structured_rows:
        if row["SECTION"] != current_section:
            current_section = row["SECTION"]
            print("")
            print("############################################################")
            print("STRUCTURED SECTION:", row["SECTION"])
            print("TITLE:", row["SECTION_TITLE"])
            print("############################################################")
            print(" | ".join(column_names))
        print(" | ".join(str(row.get(column_name, "") or "").strip() for column_name in column_names))

def read_excel_connector_info(excel_path):
    import win32com.client as win32
    start_time = time.perf_counter()
    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    workbook = None
    try:
        workbook = excel.Workbooks.Open(os.path.abspath(excel_path), UpdateLinks=0, ReadOnly=True, AddToMru=False)
        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)
        raw_count = worksheet.Range(EXCEL_CONNECTOR_COUNT_CELL).Value2
        connector_count = int(raw_count) if raw_count is not None and str(raw_count).strip() != "" else 0
        if connector_count < 0:
            connector_count = 0
        if connector_count > MAX_BLOCKS:
            connector_count = MAX_BLOCKS
        if connector_count == 0:
            return [], {}
        data_range = worksheet.Range(worksheet.Cells(EXCEL_CONNECTOR_FIRST_ROW, EXCEL_CONNECTOR_NAME_COLUMN), worksheet.Cells(EXCEL_CONNECTOR_FIRST_ROW + connector_count - 1, EXCEL_EXPECTED_PIN_COUNT_COLUMN))
        data = data_range.Value2
        connector_info = []
        expected_counts = {}
        for index, row_values in enumerate(data):
            excel_name = str(row_values[0] or "").strip()
            canonical = canonical_connector_name(excel_name)
            raw_expected = row_values[EXCEL_EXPECTED_PIN_COUNT_COLUMN - EXCEL_CONNECTOR_NAME_COLUMN]
            try:
                expected = int(raw_expected)
            except Exception:
                expected = None
            connector_info.append({"block_index": index, "excel_row": EXCEL_CONNECTOR_FIRST_ROW + index, "excel_name": excel_name, "canonical": canonical, "expected_pins": expected})
            if canonical:
                expected_counts[canonical] = expected
        elapsed = time.perf_counter() - start_time
        print(f"EXCEL CONNECTOR READ TIME: {elapsed:.2f} SECONDS")
        return connector_info, expected_counts
    finally:
        if workbook is not None:
            workbook.Close(SaveChanges=False)
        excel.Quit()


def build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info):
    jobs = []
    skip_direction = should_skip_direction(pdf_roles)

    for connector in connector_info:
        block_index = connector["block_index"]
        excel_name = connector["excel_name"]
        canonical = connector["canonical"]
        expected_pins = connector["expected_pins"]

        if canonical not in pdf_connectors:
            jobs.append({
                "block_index": block_index,
                "excel_name": excel_name,
                "canonical": canonical,
                "expected_pins": expected_pins,
                "skip": True,
                "reason": "connector not found in PDF"
            })
            continue

        pin_map = get_connector_pin_map(pdf_connectors[canonical], pdf_roles)

        if expected_pins is not None and expected_pins > BLOCK_DATA_ROW_CAPACITY:
            raise ValueError(
                f"{excel_name} expects {expected_pins} pins but "
                f"BLOCK_DATA_ROW_CAPACITY is only {BLOCK_DATA_ROW_CAPACITY}"
            )

        # Normal mode writes PIN + SIGNAL + IN/OUT.
        # Skip-direction mode intentionally writes only PIN + SIGNAL so the
        # existing spreadsheet IN/OUT column is never cleared or overwritten.
        matrix_width = 2 if skip_direction else 3
        matrix = [[""] * matrix_width for _ in range(BLOCK_DATA_ROW_CAPACITY)]

        for pin_number, row in pin_map.items():
            if pin_number < 1 or pin_number > BLOCK_DATA_ROW_CAPACITY:
                continue

            matrix[pin_number - 1][0] = pin_number
            matrix[pin_number - 1][1] = normalize_local_signal_value(
                get_role_value(row, pdf_roles, "signal")
            )

            if not skip_direction:
                matrix[pin_number - 1][2] = normalize_direction_value(
                    get_role_value(row, pdf_roles, "direction")
                )

        missing = []
        if expected_pins is not None and expected_pins > 0:
            missing = [
                pin_number
                for pin_number in range(1, expected_pins + 1)
                if pin_number not in pin_map
            ]

        block_start_column = BLOCK_FIRST_COLUMN + block_index * BLOCK_WIDTH
        pin_column = block_start_column + BLOCK_PIN_COLUMN_OFFSET
        signal_column = block_start_column + BLOCK_SIGNAL_COLUMN_OFFSET
        direction_column = block_start_column + BLOCK_DIRECTION_COLUMN_OFFSET

        jobs.append({
            "block_index": block_index,
            "excel_name": excel_name,
            "canonical": canonical,
            "expected_pins": expected_pins,
            "skip": False,
            "pin_count": len(pin_map),
            "missing": missing,
            "first_row": BLOCK_DATA_FIRST_ROW,
            "last_row": BLOCK_DATA_FIRST_ROW + BLOCK_DATA_ROW_CAPACITY - 1,
            "first_column": pin_column,
            "last_column": signal_column if skip_direction else direction_column,
            "matrix": tuple(tuple(row) for row in matrix),
            "write_direction": not skip_direction,
        })

    return jobs


def write_pdf_data_to_excel(source_excel_path, output_excel_path, pdf_connectors, pdf_roles, connector_info):
    import win32com.client as win32
    total_start = time.perf_counter()
    source_excel_path = os.path.abspath(source_excel_path)
    output_excel_path = os.path.abspath(output_excel_path)
    if source_excel_path.lower() != output_excel_path.lower():
        shutil.copy2(source_excel_path, output_excel_path)
    jobs = build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info)
    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    excel.AskToUpdateLinks = False
    workbook = None
    previous_calculation = None
    calculation_changed = False
    try:
        open_start = time.perf_counter()
        workbook = excel.Workbooks.Open(output_excel_path, UpdateLinks=0, ReadOnly=False, AddToMru=False)
        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)
        open_elapsed = time.perf_counter() - open_start
        print(f"EXCEL OPEN TIME: {open_elapsed:.2f} SECONDS")
        try:
            previous_calculation = excel.Calculation
            excel.Calculation = XL_CALCULATION_MANUAL
            calculation_changed = True
            print("EXCEL CALCULATION MODE: MANUAL")
        except Exception as e:
            print("WARNING: COULD NOT SET MANUAL CALCULATION:", repr(e))
        print("")
        print("============================================================")
        print("EXCEL CONNECTOR CHECK")
        print("============================================================")
        print("EXCEL CONNECTORS:", len(connector_info))
        print("PDF CONNECTORS:", len(pdf_connectors))
        print("MATCH MODE: CONNECTOR NAME (PDF/EXCEL ORDER DOES NOT MATTER)")
        print(
            "IN/OUT WRITE MODE:",
            "SKIPPED - EXISTING EXCEL VALUES LEFT UNCHANGED"
            if should_skip_direction(pdf_roles)
            else "WRITE FROM PDF"
        )
        print("============================================================")
        write_start = time.perf_counter()
        written_blocks = 0
        written_pin_rows = 0
        for job in jobs:
            print("")
            print("BLOCK:", job["block_index"] + 1, "EXCEL CONNECTOR:", repr(job["excel_name"]), "PDF KEY:", repr(job["canonical"]))
            if job["skip"]:
                print("SKIPPED -", job["reason"], ":", job["excel_name"])
                continue
            destination_range = worksheet.Range(worksheet.Cells(job["first_row"], job["first_column"]), worksheet.Cells(job["last_row"], job["last_column"]))
            destination_range.Value2 = job["matrix"]
            written_blocks += 1
            written_pin_rows += job["pin_count"]
            print("WRITTEN PINS:", job["pin_count"], "EXPECTED:", job["expected_pins"], "MISSING:", job["missing"])
            if not job.get("write_direction", True):
                print("IN/OUT COLUMN: SKIPPED (LEFT UNCHANGED IN EXCEL)")
        write_elapsed = time.perf_counter() - write_start
        print(f"BULK EXCEL WRITE TIME: {write_elapsed:.2f} SECONDS")
        calc_start = time.perf_counter()
        try:
            worksheet.Calculate()
            print(f"SELECTION SHEET CALCULATION TIME: {time.perf_counter() - calc_start:.2f} SECONDS")
        except Exception as e:
            print("WARNING: SELECTION SHEET CALCULATION FAILED:", repr(e))
        if calculation_changed and previous_calculation is not None:
            try:
                excel.Calculation = previous_calculation
            except Exception as e:
                print("WARNING: COULD NOT RESTORE CALCULATION MODE:", repr(e))
        save_start = time.perf_counter()
        workbook.Save()
        save_elapsed = time.perf_counter() - save_start
        print(f"EXCEL SAVE TIME: {save_elapsed:.2f} SECONDS")
        print("")
        print("============================================================")
        print("EXCEL WRITE COMPLETE")
        print("============================================================")
        print("OUTPUT:", output_excel_path)
        print("BLOCKS WRITTEN:", written_blocks)
        print("PIN ROWS WRITTEN:", written_pin_rows)
        print("============================================================")
    finally:
        if workbook is not None:
            try:
                workbook.Close(SaveChanges=False)
            except Exception:
                pass
        try:
            excel.EnableEvents = True
            excel.ScreenUpdating = True
        except Exception:
            pass
        try:
            excel.Quit()
        except Exception:
            pass
    total_elapsed = time.perf_counter() - total_start
    print(f"TOTAL EXCEL WRITE PHASE: {total_elapsed:.2f} SECONDS")
