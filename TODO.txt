def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog
    default_folder = os.path.dirname(default_full_path)
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    try:
        selected_path = filedialog.askopenfilename(initialdir=default_folder, filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")])
    except Exception:
        root.destroy()
        return None
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    return os.path.abspath(selected_path)

def is_missing_value(value):
    if value is None:
        return True
    try:
        return int(value) == COM_MISSING
    except Exception:
        return False
def clean_id_list(values):
    cleaned = []
    if values is None:
        return cleaned
    if not isinstance(values, (tuple, list)):
        values = [values]
    for value in values:
        if is_missing_value(value):
            continue
        try:
            value = int(value)
        except Exception:
            continue
        if value > 0:
            cleaned.append(value)
    return cleaned
def parse_count_array_result(result):
    if result is None:
        return []
    if isinstance(result, (tuple, list)):
        items = list(result)
        if len(items) >= 2 and isinstance(items[1], (tuple, list)):
            try:
                count = int(items[0])
            except Exception:
                count = None
            values = clean_id_list(items[1])
            if count is not None and len(values) > count:
                values = values[-count:]
            return values
        return clean_id_list(items)
    return clean_id_list(result)
def call_id_array(method):
    try:
        result = method(None)
        values = parse_count_array_result(result)
        if values:
            return values
    except Exception:
        pass
    try:
        result = method()
        values = parse_count_array_result(result)
        if values:
            return values
    except Exception:
        pass
    return []
def call_noarg(obj, method_name):
    if not hasattr(obj, method_name):
        return None
    try:
        return getattr(obj, method_name)()
    except Exception:
        return None
def normalize_pin_value(value):
    if value is None:
        return None
    if isinstance(value, bool):
        return str(value)
    if isinstance(value, int):
        return str(value)
    if isinstance(value, float):
        if math.isfinite(value) and value.is_integer():
            return str(int(value))
        return str(value).strip()
    return str(value).strip()
def median(values):
    if not values:
        return None
    ordered = sorted(values)
    middle = len(ordered) // 2
    if len(ordered) % 2 == 0:
        return (ordered[middle - 1] + ordered[middle]) / 2.0
    return ordered[middle]

def get_symbol_schema_location(sym):
    try:
        result = sym.GetSchemaLocation(None, None, None, None, None)
    except Exception:
        return None
    if not isinstance(result, (tuple, list)):
        return None
    if len(result) < 3:
        return None
    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None
    return {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "raw": result,
    }
def get_symbol_placed_area(sym):
    try:
        result = sym.GetPlacedArea(None, None, None, None)
    except Exception:
        return None
    if not isinstance(result, (tuple, list)):
        return None
    if len(result) < 5:
        return None
    try:
        success = int(result[0])
        left = float(result[1])
        bottom = float(result[2])
        right = float(result[3])
        top = float(result[4])
    except Exception:
        return None
    if success <= 0:
        return None
    return {
        "left": left,
        "bottom": bottom,
        "right": right,
        "top": top,
        "width": right - left,
        "height": top - bottom,
        "center_x": (left + right) / 2.0,
        "center_y": (bottom + top) / 2.0,
        "raw": result,
    }
def get_pin_schema_location(pin, pin_view_id):
    pin.SetId(pin_view_id)
    try:
        result = pin.GetSchemaLocation(None, None, None, None, None)
    except Exception:
        return None
    if not isinstance(result, (tuple, list)):
        return None
    numeric_values = []
    for value in result:
        if is_missing_value(value):
            continue
        try:
            numeric_values.append(float(value))
        except Exception:
            pass
    if len(numeric_values) < 3:
        return None
    return numeric_values
def resolve_live_symbol_from_pin_view(prj, pin_view_id):
    local_sym = prj.CreateSymbolObject()
    try:
        live_symbol_id = local_sym.SetId(pin_view_id)
    except Exception:
        live_symbol_id = 0
    try:
        live_symbol_id = int(live_symbol_id)
    except Exception:
        live_symbol_id = 0
    if live_symbol_id <= 0:
        return None
    return {
        "symbol_object": local_sym,
        "live_symbol_id": live_symbol_id,
    }
def inspect_live_symbol_by_id(prj, symbol_id):
    local_sym = prj.CreateSymbolObject()
    try:
        set_result = local_sym.SetId(symbol_id)
    except Exception:
        return None
    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0
    if set_result <= 0:
        return None
    return {
        "symbol_id": call_noarg(local_sym, "GetId"),
        "symbol_type_name": call_noarg(local_sym, "GetSymbolTypeName"),
        "characteristic": call_noarg(local_sym, "GetCharacteristic"),
        "rotation": call_noarg(local_sym, "GetRotation"),
        "scaling": call_noarg(local_sym, "GetScaling"),
        "schema_location": get_symbol_schema_location(local_sym),
        "bound_ids": get_bound_ids(local_sym),
        "text_ids": get_text_ids(local_sym),
    }
def location_is_inside_block(location, block_geometry, tolerance=0.10):
    if location is None:
        return False
    try:
        x = float(location["x"])
        y = float(location["y"])
        sheet_id = int(location["sheet_id"])
    except Exception:
        return False
    if sheet_id != int(block_geometry["sheet_id"]):
        return False
    if x < block_geometry["block_left"] - tolerance:
        return False
    if x > block_geometry["block_right"] + tolerance:
        return False
    if y < block_geometry["block_bottom"] - tolerance:
        return False
    if y > block_geometry["block_top"] + tolerance:
        return False
    return True
def discover_current_connector_pin_symbols(prj, sht, connector_name, block_geometry):
    candidate_ids = collect_live_symbol_candidates(prj, sht, block_geometry["sheet_id"])
    discovered = {}
    for candidate_id in candidate_ids:
        identity = read_symbol_identity(prj, candidate_id)
        if identity is None:
            continue
        if (identity["location"] is None or
                identity["location"]["sheet_id"] != block_geometry["sheet_id"]):
            continue
        pin_number = identity["pin_number"]
        device_text = identity["device_text"]
        if pin_number in (None, ""):
            continue
        if not isinstance(device_text, str):
            continue
        if not match_connector_designation(device_text, connector_name):
            continue
        if not location_is_inside_block(identity["location"], block_geometry):
            continue
        metadata = inspect_live_symbol_by_id(prj, identity["symbol_id"])
        if metadata is None:
            continue
        candidate = {
            "symbol_id": identity["symbol_id"],
            "location": identity["location"],
            "symbol_type_name": identity["symbol_type_name"],
            "characteristic": identity["characteristic"],
            "rotation": metadata["rotation"],
            "scaling": metadata["scaling"],
            "bound_ids": metadata["bound_ids"],
        }
        if pin_number not in discovered:
            discovered[pin_number] = candidate
            continue
        old = discovered[pin_number]
        old_type = old["symbol_type_name"]
        new_type = candidate["symbol_type_name"]
        preferred_types = {
            "MilSymbolReceptacle¶END_TOP",
            "MilSymbolReceptacle¶END_SPLIT",
            "MilSymbolReceptacle¶SPLIT_BOTTOM",
            "MilSymbolReceptacle¶CONNECTOR_PIN",
            "MilSymbolReceptacle¶END_BOTTOM",
        }
        if new_type in preferred_types and old_type not in preferred_types:
            discovered[pin_number] = candidate
    return discovered

def snapshot_connector(prj, dev, pin, sht, connector_device_id, connector_name, block_geometry):
    snapshot = []
    discovered_live_symbols = discover_current_connector_pin_symbols(prj, sht, connector_name, block_geometry)
    dev.SetId(connector_device_id)
    for logical_pin_id in get_pin_ids(dev):
        pin.SetId(logical_pin_id)
        pin_name = normalize_pin_value(pin.GetName())
        pin_view_ids = get_view_ids(pin)
        record = {
            "pin_name": pin_name,
            "logical_pin_id": logical_pin_id,
            "view_ids": pin_view_ids,
            "views": [],
        }
        for pin_view_id in pin_view_ids:
            location = get_pin_schema_location(pin, pin_view_id)
            node_ids = get_node_ids(pin)
            live = resolve_live_symbol_from_pin_view(prj, pin_view_id)
            view_record = {
                "pin_view_id": pin_view_id,
                "location": location,
                "node_ids": node_ids,
                "live_symbol_id": None,
                "symbol_type_name": None,
                "characteristic": None,
                "rotation": None,
                "scaling": None,
                "bound_ids": [],
                "resolution_source": None,
            }
            if live is not None:
                live_sym = live["symbol_object"]
                view_record["live_symbol_id"] = live["live_symbol_id"]
                view_record["symbol_type_name"] = call_noarg(live_sym, "GetSymbolTypeName")
                view_record["characteristic"] = call_noarg(live_sym, "GetCharacteristic")
                view_record["rotation"] = call_noarg(live_sym, "GetRotation")
                view_record["scaling"] = call_noarg(live_sym, "GetScaling")
                view_record["bound_ids"] = get_bound_ids(live_sym)
                view_record["resolution_source"] = "stable pin-view ID"
                live_location = get_symbol_schema_location(live_sym)
                if live_location is not None:
                    view_record["location"] = [
                        live_location["sheet_id"],
                        live_location["x"],
                        live_location["y"],
                    ]
            else:
                discovered = discovered_live_symbols.get(pin_name)
                if discovered is not None:
                    view_record["live_symbol_id"] = discovered["symbol_id"]
                    view_record["symbol_type_name"] = discovered["symbol_type_name"]
                    view_record["characteristic"] = discovered["characteristic"]
                    view_record["rotation"] = discovered["rotation"]
                    view_record["scaling"] = discovered["scaling"]
                    view_record["bound_ids"] = discovered["bound_ids"]
                    view_record["resolution_source"] = "sheet scan by connector + pin + block"
                    discovered_location = discovered["location"]
                    if discovered_location is not None:
                        view_record["location"] = [
                            discovered_location["sheet_id"],
                            discovered_location["x"],
                            discovered_location["y"],
                        ]
            record["views"].append(view_record)
        snapshot.append(record)
    return snapshot
def get_sorted_live_pin_records(snapshot):
    records = []
    for pin_record in snapshot:
        if len(pin_record["views"]) != 1:
            raise RuntimeError(f"Pin {pin_record['pin_name']} does not have exactly one pin view.")
        view = pin_record["views"][0]
        if view["live_symbol_id"] is None:
            raise RuntimeError(f"Pin {pin_record['pin_name']} does not have a live symbol. Stable view ID and connector/block sheet scan both failed.")
        location = view["location"]
        if location is None or len(location) < 3:
            raise RuntimeError(f"Pin {pin_record['pin_name']} does not have a valid schema location.")
        records.append({
            "pin_name": pin_record["pin_name"],
            "logical_pin_id": pin_record["logical_pin_id"],
            "pin_view_id": view["pin_view_id"],
            "live_symbol_id": view["live_symbol_id"],
            "sheet_id": int(location[0]),
            "x": float(location[1]),
            "y": float(location[2]),
            "symbol_type_name": view["symbol_type_name"],
            "characteristic": view["characteristic"],
            "rotation": view["rotation"],
            "scaling": view["scaling"],
        })
    records.sort(key=lambda item: item["y"], reverse=True)
    return records
def normalize_optional_text(value):
    """Blank Excel cells mean no restriction; numbers like 2.0 mean '2'."""
    value = normalize_pin_value(value)
    return value if value not in (None, "") else None


def find_device_ids(prj, dev, target_name, all_device_ids=None):
    """Unlike find_device_id, keep duplicates so they cannot be selected silently."""
    ids = all_device_ids if all_device_ids is not None else get_device_ids(prj)
    matches = []
    for device_id in ids:
        try:
            dev.SetId(device_id)
            name = str(dev.GetName()).strip()
        except Exception:
            continue
        if name == target_name:
            matches.append(device_id)
    return matches


def get_sheet_metadata(prj, sht):
    """Resolve user-facing sheet names/numbers to E3's internal sheet IDs."""
    sheets = []
    for sheet_id in call_id_array(prj.GetAllSheetIds):
        try:
            sht.SetId(sheet_id)
            name = str(sht.GetName()).strip()
        except Exception:
            continue
        aliases = [name]
        # The installed E3 API version may or may not provide these methods.
        for method_name in ("GetSheetNumber", "GetNumber"):
            raw = call_noarg(sht, method_name)
            alias = normalize_optional_text(raw)
            if alias is not None:
                aliases.append(alias)
        sheets.append({"sheet_id": sheet_id, "name": name, "aliases": aliases})
    if not sheets:
        raise RuntimeError("No sheets were found in the current E3 project.")
    return sheets


def same_sheet_label(left, right):
    a = str(left).strip().casefold()
    b = str(right).strip().casefold()
    if a == b:
        return True
    # Only treat whole-number names like '02' and '2' as equivalent.
    try:
        return a.isdecimal() and b.isdecimal() and int(a) == int(b)
    except Exception:
        return False


def resolve_optional_sheet_id(sheet_metadata, requested_page):
    if requested_page is None:
        return None
    matches = [
        sheet for sheet in sheet_metadata
        if any(same_sheet_label(alias, requested_page) for alias in sheet["aliases"])
    ]
    if len(matches) == 1:
        return matches[0]["sheet_id"]
    if not matches:
        available = ", ".join(repr(item["name"]) for item in sheet_metadata)
        raise RuntimeError(
            f"Excel J2 page {requested_page!r} does not match an E3 sheet name/number. "
            f"Available E3 sheets: {available}"
        )
    raise RuntimeError(
        f"Excel J2 page {requested_page!r} matches multiple E3 sheets. "
        "Use a unique sheet name/number in J2."
    )


def get_live_connector_pin_locations(prj, dev, pin, connector_device_id):
    """Find where THIS logical connector's live pin views are actually placed.

    Avoid connector text scans for initial disambiguation: a populated E3
    project may reuse both pin numbers and connector names.
    """
    locations = []
    dev.SetId(connector_device_id)
    logical_pin_ids = get_pin_ids(dev)
    for logical_pin_id in logical_pin_ids:
        try:
            pin.SetId(logical_pin_id)
            pin_name = normalize_pin_value(pin.GetName())
            pin_view_ids = get_view_ids(pin)
        except Exception:
            continue
        for pin_view_id in pin_view_ids:
            live = resolve_live_symbol_from_pin_view(prj, pin_view_id)
            if live is None:
                continue
            location = get_symbol_schema_location(live["symbol_object"])
            if location is not None:
                locations.append({"pin_name": pin_name, "location": location})
    return locations


def collect_candidate_cabblocks(prj, dev, main_device_ids, requested_sheet_id=None):
    """Collect EVERY CABBLOCK in matching device views, never the first one."""
    blocks = []
    for main_device_id in main_device_ids:
        for main_view_id in get_device_view_ids(dev, main_device_id):
            main_view_number = get_view_number(dev, main_view_id)
            if main_view_number is None:
                continue
            for symbol_id in get_device_symbol_ids(dev, main_view_id):
                local_sym = prj.CreateSymbolObject()
                try:
                    if not local_sym.SetId(symbol_id):
                        continue
                    symbol_type_name = str(local_sym.GetSymbolTypeName())
                except Exception:
                    continue
                if "CABBLOCK" not in symbol_type_name.upper():
                    continue
                location = get_symbol_schema_location(local_sym)
                if location is None:
                    continue
                sheet_id = location["sheet_id"]
                if requested_sheet_id is not None and sheet_id != requested_sheet_id:
                    continue
                area = get_symbol_placed_area(local_sym)
                if area is None:
                    continue
                blocks.append({
                    "main_device_id": main_device_id,
                    "main_view_id": main_view_id,
                    "connector_view_number": main_view_number,
                    "block_symbol_id": symbol_id,
                    "block_symbol_type": symbol_type_name,
                    "block_schema_location": location,
                    "block_left": area["left"],
                    "block_right": area["right"],
                    "block_bottom": area["bottom"],
                    "block_top": area["top"],
                    "block_width": area["width"],
                    "block_height": area["height"],
                    "block_center_x": area["center_x"],
                    "block_center_y": area["center_y"],
                    "sheet_id": sheet_id,
                })
    return blocks

def get_device_debug_identity(dev, device_id):
    """Capture the real E3 fields; absent methods are shown as unavailable."""
    try:
        # Some E3 COM builds return None on success. Read the actual
        # identity rather than interpreting a falsy return as failure.
        dev.SetId(device_id)
    except Exception as exc:
        return {"name": f"<SetId error: {exc}>"}
    identity = {"name": call_noarg(dev, "GetName")}
    for name in ("GetDesignation", "GetDeviceDesignation", "GetAssignment", "GetLocation"):
        value = call_noarg(dev, name)
        if value not in (None, ""):
            identity[name] = str(value)
    return identity


def get_symbol_debug_texts(prj, symbol_id, limit=18):
    """Return text type + text as E3 actually stores them on a symbol."""
    local_sym = prj.CreateSymbolObject()
    try:
        if not local_sym.SetId(symbol_id):
            return []
    except Exception:
        return []
    try:
        local_text = prj.CreateTextObject()
        text_ids = get_text_ids(local_sym)
    except Exception:
        return []
    rows = []
    for text_id in text_ids[:limit]:
        try:
            local_text.SetId(text_id)
            typ = local_text.GetType()
            value = str(local_text.GetText())
            if value.strip():
                rows.append((typ, value.replace("\n", " ").strip()[:160]))
        except Exception:
            continue
    return rows


def get_sheet_debug_symbol_ids(prj, sht, sheet_id):
    """Use sheet symbol IDs if provided; otherwise fall back to project IDs."""
    try:
        sht.SetId(sheet_id)
        sheet_ids = call_id_array(sht.GetSymbolIds)
    except Exception:
        sheet_ids = []
    if sheet_ids:
        return sheet_ids
    try:
        return call_id_array(prj.GetSymbolIds)
    except Exception:
        return []


def print_candidate_block_details(prj, dev, sht, block_candidates, sheet_names,
                                  excel_connector_names, nearby_limit=18):
    """Print all prospective blocks and a small sample of nearby pin labels.

    This runs before any pins are moved and is deliberately diagnostic only.
    It prints visible text rather than assuming the E3 device's GetName()
    equals the connector label printed on the schematic.
    """
    section("PROSPECTIVE CABBLOCK CANDIDATES - E3 DEBUG")
    log(f"Excel connector names: {excel_connector_names}")
    log(f"Number of CABBLOCK candidates: {len(block_candidates)}")
    sheet_cache = {}
    for index, block in enumerate(block_candidates, 1):
        main_id = block["main_device_id"]
        ident = get_device_debug_identity(dev, main_id)
        sheet_id = block["sheet_id"]
        sheet_name = sheet_names.get(sheet_id, f"<unknown sheet: {sheet_id}>")
        symbol_id = block["block_symbol_id"]
        center = block["block_schema_location"]
        log()
        log(f"CABBLOCK CANDIDATE {index}/{len(block_candidates)}")
        log(f"  E3 main device ID: {main_id}")
        log(f"  E3 main device GetName(): {ident.get('name')!r}")
        for field in ("GetDesignation", "GetDeviceDesignation", "GetAssignment", "GetLocation"):
            if field in ident:
                log(f"  E3 main device {field}(): {ident[field]!r}")
        log(f"  Main device view ID: {block['main_view_id']}")
        log(f"  Main device view number: {block['connector_view_number']}")
        log(f"  Sheet/page name: {sheet_name!r} | internal sheet ID: {sheet_id}")
        log(f"  CABBLOCK symbol ID: {symbol_id}")
        log(f"  CABBLOCK symbol type: {block['block_symbol_type']!r}")
        if center is not None:
            log(f"  Symbol origin: ({center['x']:.3f}, {center['y']:.3f})")
        log(f"  Block bounds: L={block['block_left']:.3f}, R={block['block_right']:.3f}, "
            f"B={block['block_bottom']:.3f}, T={block['block_top']:.3f}")
        local_sym = prj.CreateSymbolObject()
        try:
            if local_sym.SetId(symbol_id):
                log(f"  CABBLOCK bound IDs: {get_bound_ids(local_sym)}")
        except Exception:
            pass
        block_text = get_symbol_debug_texts(prj, symbol_id)
        if block_text:
            log(f"  Text attached to CABBLOCK: {block_text}")
        else:
            log("  Text attached to CABBLOCK: <none/read failed>")

        # Print visible connector / pin texts inside the actual block. These
        # are often more useful than logical-device names in large jobs.
        if sheet_id not in sheet_cache:
            ids = get_sheet_debug_symbol_ids(prj, sht, sheet_id)
            sheet_cache[sheet_id] = ids
            log(f"  Sheet symbol scan size: {len(ids)}")
        candidate_symbol_ids = sheet_cache[sheet_id]
        inside = []
        pin_entries = []
        device_labels = {}
        for nearby_id in candidate_symbol_ids:
            if nearby_id == symbol_id:
                continue
            local = prj.CreateSymbolObject()
            try:
                if not local.SetId(nearby_id):
                    continue
            except Exception:
                continue
            loc = get_symbol_schema_location(local)
            if not location_is_inside_block(loc, block, tolerance=0.03):
                continue
            inside.append(nearby_id)
            texts = get_symbol_debug_texts(prj, nearby_id, limit=14)
            label = [v for t, v in texts if t == 12]
            pin_name = [v for t, v in texts if t == 203]
            if label:
                for val in label:
                    device_labels[val] = device_labels.get(val, 0) + 1
            if (label or pin_name) and len(pin_entries) < nearby_limit:
                pin_entries.append((nearby_id, call_noarg(local, "GetSymbolTypeName"),
                                    label[:2], pin_name[:2]))
        log(f"  Other symbols whose origins are inside block: {len(inside)}")
        if device_labels:
            labels = sorted(device_labels.items(), key=lambda pair: (-pair[1], pair[0]))
            log(f"  DEVICE DESIGNATION LABELS (text type 12): {labels[:20]}")
        else:
            log("  DEVICE DESIGNATION LABELS (text type 12): <none found>")
        if pin_entries:
            log(f"  Nearby pin / connector symbols (up to {nearby_limit}):")
            for sid, typ, labels, pins in pin_entries:
                log(f"    SYMBOL {sid} | type {typ!r} | device labels {labels} | pin numbers {pins}")
        else:
            log("  Nearby pin / connector symbols: <none with readable labels>")
    log("END OF CABBLOCK CANDIDATE DETAILS")


def print_connector_name_lookup_failure(prj, dev, connector_name, device_ids,
                                        block_candidates, sheet_names, top_n=30):
    """Explain an unsuccessful *exact* GetName lookup using real E3 names."""
    import difflib
    section(f"CONNECTOR LOOKUP FAILED - {connector_name!r} - NAME DIAGNOSTICS")
    log(f"  Excel requested connector name: {connector_name!r}")
    log("  Lookup currently requires EXACT dev.GetName() == Excel name.")
    requested = connector_name.strip().casefold()
    device_info = []
    for device_id in device_ids:
        ident = get_device_debug_identity(dev, device_id)
        raw_name = ident.get("name")
        name = str(raw_name).strip() if raw_name is not None else ""
        if not name:
            continue
        lower = name.casefold()
        exact_ignoring_case = lower == requested
        contains = bool(requested) and (requested in lower or lower in requested)
        similarity = difflib.SequenceMatcher(None, requested, lower).ratio()
        # Favor contained names before loose fuzzy similarity.
        score = (200 if exact_ignoring_case else 100 if contains else 0) + similarity
        device_info.append((score, device_id, name, ident, exact_ignoring_case, contains))
    device_info.sort(key=lambda info: (-info[0], info[2]))
    log(f"  Nonempty E3 device GetName() values scanned: {len(device_info)}")
    log(f"  Closest E3 device-name matches (top {top_n}):")
    for _, device_id, name, ident, same, contains in device_info[:top_n]:
        flag = 'CASE DIFFERENCE' if same else 'CONTAINS NAME' if contains else 'FUZZY'
        log(f"    E3 device ID {device_id} | GetName()={name!r} | match={flag}")
        for field in ("GetDesignation", "GetDeviceDesignation"):
            if field in ident:
                log(f"      {field}()={ident[field]!r}")
    log("  CABBLOCK owners on selected page/device:")
    for block in block_candidates:
        ident = get_device_debug_identity(dev, block["main_device_id"])
        page = sheet_names.get(block["sheet_id"], str(block["sheet_id"]))
        log(f"    PAGE {page!r} | BLOCK {block['block_symbol_id']} | "
            f"OWNER ID {block['main_device_id']} | OWNER NAME {ident.get('name')!r} "
            f"| VIEW NUMBER {block['connector_view_number']}")
    log("  NO CHANGES MADE: E3 connector resolution stopped before pin movement.")

def select_connector_and_block(prj, dev, pin, connector_name,
                               connector_device_ids, block_candidates,
                               requested_page=None, requested_device=None,
                               sheet_names=None):
    """Use view number + live pin location + optional C4/J2 to pick one target.

    This is intentionally strict. Picking the first candidate can silently
    change an unrelated block when the project contains repeated names.
    """
    matches = []
    for connector_device_id in connector_device_ids:
        connector_view_ids = get_device_view_ids(dev, connector_device_id)
        view_numbers = set()
        for connector_view_id in connector_view_ids:
            view_number = get_view_number(dev, connector_view_id)
            if view_number is not None:
                view_numbers.add(view_number)
        if not view_numbers:
            continue
        anchors = get_live_connector_pin_locations(prj, dev, pin, connector_device_id)
        for block in block_candidates:
            if block["connector_view_number"] not in view_numbers:
                continue
            # A pin on another sheet or outside this CABBLOCK is positive
            # evidence this is NOT the connector/block combination we want.
            if anchors and not all(
                location_is_inside_block(anchor["location"], block)
                for anchor in anchors
            ):
                continue
            geometry = dict(block)
            geometry["connector_view_ids"] = connector_view_ids
            matches.append({
                "connector_device_id": connector_device_id,
                "main_device_id": block["main_device_id"],
                "block_geometry": geometry,
                "confirmed_pin_count": len(anchors),
            })

    # A block can sometimes surface twice from the E3 COM view structure.
    unique = {}
    for match in matches:
        key = (match["connector_device_id"], match["block_geometry"]["block_symbol_id"])
        unique[key] = match
    matches = list(unique.values())

    if len(matches) == 1:
        selected = matches[0]
        if selected["confirmed_pin_count"] == 0:
            log(f"WARNING: {connector_name}: no live pin locations were available for "
                "preselection; the single block match will be verified in the snapshot.")
        return selected

    filters = f"C4={requested_device or '<blank>'}, J2={requested_page or '<blank>'}"
    if not matches:
        raise RuntimeError(
            f"No uniquely matching placed block for connector {connector_name!r} "
            f"({filters}). Check the connector name, device designation, page, "
            "and whether its E3 pin views are placed inside the CABBLOCK."
        )

    details = []
    for match in matches:
        block = match["block_geometry"]
        sheet_name = (sheet_names or {}).get(block["sheet_id"], str(block["sheet_id"]))
        details.append(
            f"connector device ID {match['connector_device_id']}, "
            f"main device ID {match['main_device_id']}, "
            f"sheet {sheet_name!r}, view {block['connector_view_number']}, "
            f"block ID {block['block_symbol_id']}, "
            f"live pins verified {match['confirmed_pin_count']}"
        )
    raise RuntimeError(
        f"AMBIGUOUS connector {connector_name!r} ({filters}): {len(matches)} matches. "
        "Specify C4 and/or J2 rather than modifying an arbitrary block. "
        "Matches: " + " | ".join(details)
    )

def match_connector_designation(actual, requested):
    """Match J1A or the E3 view-designated J1A#1; never J1A2/A2J1A.

    E3 presents a connector's printed designation (text type 12) separately
    from the owning device name (GetName). A component PNL2 may contain views
    J1A#1, J2A#2 and J3A#3 without any device named exactly J1A.
    """
    if actual is None or requested is None:
        return False
    label = str(actual).strip().casefold()
    wanted = str(requested).strip().casefold()
    if not wanted:
        return False
    if label == wanted:
        return True
    if label.startswith(wanted + '#'):
        trailing = label[len(wanted) + 1:]
        return trailing.isdecimal()
    return False


def collect_labeled_block_pins(prj, sht, block, connector_name, sheet_symbol_cache):
    """Read actual pin symbols inside this placed CABBLOCK, read-only."""
    sheet_id = block['sheet_id']
    if sheet_id not in sheet_symbol_cache:
        sheet_symbol_cache[sheet_id] = get_sheet_debug_symbol_ids(prj, sht, sheet_id)
    pins = {}
    seen_labels = set()
    duplicates = []
    for symbol_id in sheet_symbol_cache[sheet_id]:
        if symbol_id == block['block_symbol_id']:
            continue
        ident = read_symbol_identity(prj, symbol_id)
        if ident is None:
            continue
        if not location_is_inside_block(ident['location'], block, tolerance=0.03):
            continue
        if not match_connector_designation(ident['device_text'], connector_name):
            continue
        name = ident['pin_number']
        if name in (None, ''):
            continue
        seen_labels.add(str(ident['device_text']).strip())
        if name in pins:
            duplicates.append((name, pins[name]['symbol_id'], ident['symbol_id']))
        else:
            pins[name] = ident
    return {'pins': pins, 'labels': sorted(seen_labels), 'duplicates': duplicates}


def select_labeled_block(prj, sht, connector_record, candidates, sheet_names):
    """Choose by exact connector label + expected pin inventory, not GetName().

    Even if only one block is located, reject it unless all the expected pin
    names appear under that connector designation inside that exact block.
    """
    name = connector_record['connector name']
    expected = connector_record['left pins'] + connector_record['right pins']
    if len(set(expected)) != len(expected):
        raise RuntimeError(f'{name}: duplicate pin names in Excel, refusing ambiguous lookup.')
    expected_set = set(expected)
    sheet_symbol_cache = {}
    valid = []
    section(f'CONNECTOR DESIGNATION / BLOCK MATCH - {name}')
    for block in candidates:
        scan = collect_labeled_block_pins(prj, sht, block, name, sheet_symbol_cache)
        found = set(scan['pins'])
        missing = sorted(expected_set - found)
        extra = sorted(found - expected_set)
        sheet = sheet_names.get(block['sheet_id'], str(block['sheet_id']))
        log(f"  CABBLOCK {block['block_symbol_id']} | owner {block['main_device_id']} "
            f"| page {sheet!r} | view {block['connector_view_number']} "
            f"| matching labels {scan['labels']} | pins {len(found)}/{len(expected)}")
        if scan['duplicates']:
            log(f"      DUPLICATE PIN SYMBOLS: {scan['duplicates'][:12]}")
        if missing:
            log(f"      MISSING FROM BLOCK: {missing[:15]}")
        if extra:
            log(f"      EXTRA IN BLOCK: {extra[:15]}")
        if not scan['duplicates'] and found == expected_set and len(found) == len(expected):
            valid.append({'block_geometry': dict(block), 'pin_symbols': scan['pins'],
                          'labels': scan['labels'], 'main_device_id': block['main_device_id']})
            log('      RESULT: EXACT connector designation and pin inventory MATCH')
        elif found:
            log('      RESULT: connector label found, but pin inventory does NOT match')
        else:
            log('      RESULT: no matching connector designation within block')
    if len(valid) == 1:
        selected = valid[0]
        log(f"SELECTED {name} -> CABBLOCK {selected['block_geometry']['block_symbol_id']} "
            f"| labels {selected['labels']} | {len(selected['pin_symbols'])} pins")
        return selected
    if not valid:
        raise RuntimeError(f'{name}: no CABBLOCK has the matching connector designation '
                           'AND the complete Excel pin inventory. No pins modified.')
    details = ', '.join(f"symbol {v['block_geometry']['block_symbol_id']} "
                        f"page {sheet_names.get(v['block_geometry']['sheet_id'])!r} "
                        f"owner {v['main_device_id']}" for v in valid)
    raise RuntimeError(f'{name}: {len(valid)} IDENTICAL eligible connector blocks: {details}. '
                       'Specify C4 and/or J2 to narrow them. No pins modified.')


def read_pin_view_schema_location(pin, view_id):
    """Optional fallback for COM builds that do not resolve view -> symbol ID."""
    try:
        result = pin.SetId(view_id)
        raw = pin.GetSchemaLocation(None, None, None, None, None)
        if isinstance(raw, (tuple, list)) and len(raw) >= 3:
            return {'sheet_id': int(raw[0]), 'x': float(raw[1]), 'y': float(raw[2])}
    except Exception:
        pass
    return None


def locate_logical_pin_bindings(prj, dev, pin, selected, all_device_ids):
    """Match actual E3 symbol IDs back to the stable logical pin/view IDs.

    Try the selected CABBLOCK's owning device first (fast). Search the other
    devices only if necessary. Never guess from a similar device name.
    """
    symbol_by_pin = selected['pin_symbols']
    main_id = selected['main_device_id']
    unmatched = set(symbol_by_pin)
    bindings = {}
    candidate_devices = [main_id] + [d for d in all_device_ids if d != main_id]
    log(f'REVERSE PIN LOOKUP: try owning device {main_id} first; '
        f'{len(candidate_devices)} total E3 devices available')
    tested_pins = 0
    devices_scanned = 0
    for device_id in candidate_devices:
        devices_scanned += 1
        try:
            dev.SetId(device_id)
            logical_ids = get_pin_ids(dev)
        except Exception:
            continue
        for logical_id in logical_ids:
            tested_pins += 1
            try:
                pin.SetId(logical_id)
                name = normalize_pin_value(pin.GetName())
                if name not in unmatched:
                    continue
                view_ids = get_view_ids(pin)
            except Exception:
                continue
            target = symbol_by_pin[name]
            for view_id in view_ids:
                live = resolve_live_symbol_from_pin_view(prj, view_id)
                live_id = live['live_symbol_id'] if live is not None else None
                if live_id is None:
                    # Some E3 COM versions use distinct ghost/view and symbol IDs;
                    # this positional fallback is still checked against the
                    # exact previously identified block symbol and pin number.
                    loc = read_pin_view_schema_location(pin, view_id)
                    target_loc = target['location']
                    if (loc is None or target_loc is None or
                            loc['sheet_id'] != target_loc['sheet_id'] or
                            not coordinates_close(loc, target_loc['x'], target_loc['y'])):
                        continue
                    log(f"  {name}: view {view_id} resolved through verified pin-view location")
                elif int(live_id) != int(target['symbol_id']):
                    continue
                bindings[name] = {
                    'pin_name': name,
                    'logical_pin_id': logical_id,
                    'pin_view_id': view_id,
                    'live_symbol_id': target['symbol_id'],
                    'device_id': device_id,
                }
                unmatched.discard(name)
                break
        if not unmatched:
            break
        if devices_scanned % 250 == 0:
            log(f'  REVERSE LOOKUP PROGRESS: {devices_scanned} devices, '
                f'{tested_pins} logical pins, {len(bindings)}/{len(symbol_by_pin)} matched')
    log(f'REVERSE LOOKUP RESULT: {len(bindings)}/{len(symbol_by_pin)} logical pin views '
        f'mapped; {devices_scanned} devices and {tested_pins} logical pins checked.')
    if unmatched:
        log(f"  UNRESOLVED PINS: {sorted(unmatched)}")
        log('  Each target symbol exists and is labeled correctly, but its stable '
            'pin-view ID could not be established from the E3 logical pins.')
        raise RuntimeError('Cannot safely move pins without confirmed E3 pin-view IDs. '
                           'No pins have been modified.')
    owner_ids = sorted(set(value['device_id'] for value in bindings.values()))
    log(f'  Logical-pin owning device IDs: {owner_ids}')
    for name in sorted(bindings):
        record = bindings[name]
        log(f"    PIN {name}: owner {record['device_id']} | logical {record['logical_pin_id']} "
            f"| view {record['pin_view_id']} | symbol {record['live_symbol_id']}")
    return bindings


def snapshot_selected_pin_bindings(prj, bindings, selected):
    """Snapshot only the actual chosen pin views, never unrelated device pins."""
    records = []
    for name, binding in bindings.items():
        symbol = selected['pin_symbols'][name]
        meta = inspect_live_symbol_by_id(prj, symbol['symbol_id'])
        if meta is None:
            raise RuntimeError(f'Pin {name}: selected symbol has disappeared; no pins modified.')
        location = symbol['location']
        if location is None:
            raise RuntimeError(f'Pin {name}: missing symbol location; no pins modified.')
        records.append({
            'pin_name': name, 'logical_pin_id': binding['logical_pin_id'],
            'view_ids': [binding['pin_view_id']],
            'views': [{
                'pin_view_id': binding['pin_view_id'],
                'location': [location['sheet_id'], location['x'], location['y']],
                'node_ids': [], 'live_symbol_id': symbol['symbol_id'],
                'symbol_type_name': meta['symbol_type_name'],
                'characteristic': meta['characteristic'],
                'rotation': meta['rotation'], 'scaling': meta['scaling'],
                'bound_ids': meta['bound_ids'],
                'resolution_source': 'exact CABBLOCK label + pin + verified logical view',
            }],
        })
    return records

def validate_connector_snapshot_location(connector_name, sorted_pin_records, block_geometry):
    """Final read-only preflight before anything is deleted or replaced."""
    if not sorted_pin_records:
        raise RuntimeError(f"{connector_name}: no live connector pins were found.")
    for record in sorted_pin_records:
        location = {
            "sheet_id": record["sheet_id"],
            "x": record["x"],
            "y": record["y"],
        }
        if not location_is_inside_block(location, block_geometry):
            raise RuntimeError(
                f"{connector_name} pin {record['pin_name']}: live pin at "
                f"sheet {record['sheet_id']} ({record['x']}, {record['y']}) "
                f"is not inside selected CABBLOCK {block_geometry['block_symbol_id']} "
                f"on sheet {block_geometry['sheet_id']}. No pins were modified."
            )
def derive_pin_geometry(block_geometry, sorted_pin_records, connector_excel_record):
    y_differences = []
    for index in range(len(sorted_pin_records) - 1):
        difference = abs(sorted_pin_records[index]["y"] - sorted_pin_records[index + 1]["y"])
        if difference > 0.000001:
            y_differences.append(difference)
    pin_pitch = median(y_differences)
    if pin_pitch is None:
        pin_pitch = 0.25
    original_top_y = sorted_pin_records[0]["y"]
    left_x = block_geometry["block_left"] + PIN_EDGE_INSET
    right_x = block_geometry["block_right"] - PIN_EDGE_INSET
    left_count = len(connector_excel_record["left pins"])
    right_count = len(connector_excel_record["right pins"])
    max_side_count = max(left_count, right_count)
    if max_side_count > 0:
        required_bottom_y = original_top_y - ((max_side_count - 1) * pin_pitch)
    else:
        required_bottom_y = original_top_y
    minimum_allowed_y = block_geometry["block_bottom"] + BLOCK_BOTTOM_CLEARANCE
    if required_bottom_y < minimum_allowed_y:
        raise RuntimeError(f"Block is not tall enough for the requested pin order. Required bottom Y is {required_bottom_y}, allowed bottom Y is {minimum_allowed_y}. Automatic block resizing is not implemented yet.")
    if original_top_y > block_geometry["block_top"]:
        raise RuntimeError(f"Existing top pin Y {original_top_y} is above block top {block_geometry['block_top']}.")
    return {
        "pin_pitch": pin_pitch,
        "original_top_y": original_top_y,
        "final_top_y": original_top_y,
        "left_x": left_x,
        "right_x": right_x,
        "staging_center_x": block_geometry["block_center_x"],
        "staging_center_y": block_geometry["block_center_y"],
        "block_left": block_geometry["block_left"],
        "block_right": block_geometry["block_right"],
        "block_bottom": block_geometry["block_bottom"],
        "block_top": block_geometry["block_top"],
        "sheet_id": block_geometry["sheet_id"],
    }
def build_global_final_sequence(final_plan):
    left_plan = [dict(item) for item in final_plan if item["side"] == "LEFT"]
    right_plan = [dict(item) for item in final_plan if item["side"] == "RIGHT"]
    left_plan.sort(key=lambda item: item["row"])
    right_plan.sort(key=lambda item: item["row"])
    sequence = []
    left_index = 0
    right_index = 0
    while left_index < len(left_plan) or right_index < len(right_plan):
        if left_index < len(left_plan):
            sequence.append(left_plan[left_index])
            left_index += 1
        if right_index < len(right_plan):
            sequence.append(right_plan[right_index])
            right_index += 1
    for sequence_index, item in enumerate(sequence):
        item["sequence_index"] = sequence_index
        item["stage_lane_index"] = sequence_index % 2
        item["stage_row"] = sequence_index // 2
        if item["stage_lane_index"] == 0:
            item["stage_lane"] = "CENTER A"
        else:
            item["stage_lane"] = "CENTER B"
    return sequence
def build_edge_staging_map(global_final_sequence, pin_geometry, connector_excel_record=None, sorted_pin_records=None):
    """Stage on the UNUSED CABBLOCK boundary, never inside the block.

    Zuken only allows block-connector pins at the block border. This guarded
    workflow supports an all-left or all-right connector, with all *current*
    pins on that same occupied edge. Mixed-sided connectors need a separate
    non-overlapping placement algorithm, so they are rejected BEFORE edits.
    """
    staging_map = {}
    if not global_final_sequence:
        return staging_map
    if connector_excel_record is None or sorted_pin_records is None:
        raise RuntimeError('Edge staging requires Excel side and current pin-location preflight.')
    left_count = len(connector_excel_record['left pins'])
    right_count = len(connector_excel_record['right pins'])
    connector_name = connector_excel_record['connector name']
    if bool(left_count) == bool(right_count):
        raise RuntimeError(
            f'{connector_name}: edge staging only supports pins on ONE final side; '
            f'Excel has {left_count} left and {right_count} right. '
            'Edge fallback is unavailable for mixed final-side layouts.'
        )

    if left_count:
        source_x = pin_geometry['left_x']
        stage_x = pin_geometry['right_x']
        stage_flip = RIGHT_FLIP
        stage_name = 'TEMP RIGHT EDGE'
    else:
        source_x = pin_geometry['right_x']
        stage_x = pin_geometry['left_x']
        stage_flip = LEFT_FLIP
        stage_name = 'TEMP LEFT EDGE'

    # Do not risk placement collisions on a half-reordered block. The initial
    # pin-origin coordinates should all be on the currently occupied edge.
    edge_tolerance = 0.12
    outliers = [
        (rec['pin_name'], rec['x'], rec['y']) for rec in sorted_pin_records
        if abs(float(rec['x']) - float(source_x)) > edge_tolerance
    ]
    if outliers:
        raise RuntimeError(
            f'{connector_name}: cannot reserve {stage_name}. '
            f'{len(outliers)} existing pin(s) are not on the expected opposite '
            f'edge X={source_x:.4f}; examples: {outliers[:8]}. '
            'No changes made; inspect for partially staged pins or an unexpected symbol offset.'
        )

    # Use one column on the unused CABBLOCK boundary, from top down.
    # Interior staging is never attempted: E3 pin views reject it.
    stage_top_y = pin_geometry['original_top_y']
    pitch = pin_geometry['pin_pitch']
    bottom_y = stage_top_y - (len(global_final_sequence) - 1) * pitch
    if bottom_y < pin_geometry['block_bottom'] + BLOCK_BOTTOM_CLEARANCE:
        raise RuntimeError(
            f'{connector_name}: {len(global_final_sequence)} pins do not fit on '
            f'{stage_name} at pitch {pitch:.5f}. '
            f'Last stage Y={bottom_y:.5f}, block bottom={pin_geometry["block_bottom"]:.5f}. '
            'No changes made.'
        )
    if stage_top_y > pin_geometry['block_top'] - 0.10:
        raise RuntimeError(f'{connector_name}: stage top exceeds block boundary. No changes made.')

    for index, item in enumerate(global_final_sequence):
        # Do NOT mutate global_final_sequence: it is also used by the primary
        # center-staging strategy and the final-placement sequence.
        y = stage_top_y - index * pitch
        staging_map[item['pin_name']] = {
            'x': round(stage_x, 6),
            'y': round(y, 6),
            'flip': stage_flip,
            'lane_index': 0,
            'lane': stage_name,
            'row': index,
            'sequence_index': item['sequence_index'],
        }
    log(f'{connector_name}: guarded staging on {stage_name} '
        f'(X={stage_x:.5f}, Y={stage_top_y:.5f} to {bottom_y:.5f}); '
        'no interior placement.')
    return staging_map
def connector_already_matches_final_plan(sorted_pin_records, final_plan, tolerance=0.06):
    current_by_pin = {}
    for record in sorted_pin_records:
        current_by_pin[record["pin_name"]] = record
    if len(current_by_pin) != len(final_plan):
        return False
    for final_item in final_plan:
        pin_name = final_item["pin_name"]
        if pin_name not in current_by_pin:
            return False
        current = current_by_pin[pin_name]
        if abs(float(current["x"]) - float(final_item["x"])) > tolerance:
            return False
        if abs(float(current["y"]) - float(final_item["y"])) > tolerance:
            return False
    return True
def validate_excel_against_e3(connector_excel_record, sorted_pin_records):
    excel_pins = connector_excel_record["left pins"] + connector_excel_record["right pins"]
    e3_pins = [record["pin_name"] for record in sorted_pin_records]
    if len(e3_pins) != connector_excel_record["number of pins"]:
        raise RuntimeError(f"{connector_excel_record['connector name']}: E3 has {len(e3_pins)} pins but Excel expects {connector_excel_record['number of pins']}.")
    if set(excel_pins) != set(e3_pins):
        missing_from_excel = sorted(set(e3_pins) - set(excel_pins))
        missing_from_e3 = sorted(set(excel_pins) - set(e3_pins))
        raise RuntimeError(f"{connector_excel_record['connector name']}: Excel/E3 mismatch. Missing from Excel: {missing_from_excel}. Missing from E3: {missing_from_e3}.")
def verify_detached_symbol_load(prj):
    probe = prj.CreateSymbolObject()
    result = probe.Load(DETACHED_SYMBOL_NAME, 1)
    if result == 0:
        raise RuntimeError(f"Could not load {DETACHED_SYMBOL_NAME}.")
def delete_live_symbol(prj, live_symbol_id):
    local_sym = prj.CreateSymbolObject()
    set_result = local_sym.SetId(live_symbol_id)
    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0
    if set_result <= 0:
        raise RuntimeError(f"Could not resolve live symbol ID {live_symbol_id}.")
    return local_sym.Delete()
def verify_ghost(prj, pin_view_id):
    local_sym = prj.CreateSymbolObject()
    try:
        set_result = local_sym.SetId(pin_view_id)
    except Exception:
        set_result = 0
    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0
    return set_result == 0
def choose_compatible_pin_symbol(prj, sorted_pin_records, connector_name):
    """Find a loadable symbol from the placed block-pin family, if possible.

    The old hardcoded detached symbol belongs to MilSymbolReceptacle; the
    supplied E3 project uses MilBlockSymbolReceptacleIInt for block pins.
    An incompatible symbol is another reason why PlacePinView may fail.
    """
    original_types = [str(item.get('symbol_type_name') or '').strip()
                      for item in sorted_pin_records]
    candidates = []
    for kind in original_types:
        if not kind:
            continue
        if '¶' in kind:
            family, suffix = kind.rsplit('¶', 1)
            if 'BLOCK' in family.upper():
                candidates.extend((family + '¶CONNECTOR_PIN',
                                   family + '¶END_SPLIT', kind))
        else:
            candidates.append(kind)
    is_block_connector = any('BLOCK' in kind.upper() for kind in original_types)
    if not is_block_connector:
        candidates.append(DETACHED_SYMBOL_NAME)
    if len({kind.split('¶')[0] for kind in original_types if kind}) > 1:
        raise RuntimeError(f'{connector_name}: mixed pin-symbol families; refusing to alter pins.')
    seen = set()
    for name in candidates:
        if name in seen:
            continue
        seen.add(name)
        probe = prj.CreateSymbolObject()
        try:
            load_result = probe.Load(name, int(DETACHED_SYMBOL_VERSION))
        except Exception:
            load_result = 0
        if load_result:
            log(f'{connector_name}: stage/restore symbol {name!r} loaded '
                f'(source pin types {sorted(set(original_types))})')
            return name
    raise RuntimeError(
        f'{connector_name}: no compatible connector pin symbol loaded in E3. '
        f'Original pin symbol types: {sorted(set(original_types))}. '
        'Aborting before pin deletion.'
    )


def place_ghost_pin_view(prj, pin_view_id, sheet_id, x, y, rotation, scaling, symbol_name=None):
    placement = prj.CreateSymbolObject()
    name = symbol_name or DETACHED_SYMBOL_NAME
    new_symbol_id = placement.PlacePinView(pin_view_id, sheet_id, name, DETACHED_SYMBOL_VERSION, x, y, rotation, scaling, MAINTAIN_TEXT_SIZE)
    try:
        new_symbol_id = int(new_symbol_id)
    except Exception:
        new_symbol_id = 0
    if new_symbol_id <= 0:
        raise RuntimeError(
            f"PlacePinView failed for pin-view ID {pin_view_id} on sheet {sheet_id} "
            f"at ({x}, {y}), symbol {name!r}. "
            "For block connectors, E3 requires a compatible attachment. "
            "Also verify that this detached symbol is a block-pin type supported "
            "by the project's E3 database; the pin is now unplaced/ghost."
        )
    return new_symbol_id
class StagingPlacementRejected(RuntimeError):
    """E3 refused placement but successfully restored the original pin."""


def move_pin_by_delete_recreate(prj, pin_view_id, current_live_symbol_id, sheet_id, x, y, rotation, scaling, symbol_name=None):
    # Cache the original position/orientation *before* the destructive COM call.
    # If the placement is disallowed, attempt to restore the original symbol.
    before_sym = prj.CreateSymbolObject()
    if not before_sym.SetId(current_live_symbol_id):
        raise RuntimeError(f"Cannot inspect live symbol {current_live_symbol_id} before move.")
    before_location = get_symbol_schema_location(before_sym)
    before_rotation = call_noarg(before_sym, 'GetRotation')
    before_scale = call_noarg(before_sym, 'GetScaling')
    if before_location is None:
        raise RuntimeError(f"Symbol {current_live_symbol_id} has no original location; refusing deletion.")
    if not isinstance(before_rotation, str) or not before_rotation:
        before_rotation = rotation
    try:
        before_scale = float(before_scale)
    except Exception:
        before_scale = scaling
    delete_live_symbol(prj, current_live_symbol_id)
    time.sleep(0.15)
    if not verify_ghost(prj, pin_view_id):
        raise RuntimeError(f"Pin-view ID {pin_view_id} did not become a ghost after delete.")
    try:
        new_symbol_id = place_ghost_pin_view(prj, pin_view_id, sheet_id, x, y, rotation, scaling, symbol_name)
        time.sleep(0.15)
        metadata = inspect_live_symbol_by_id(prj, new_symbol_id)
        if metadata is None:
            raise RuntimeError(f"Returned symbol ID {new_symbol_id} could not be resolved.")
        return new_symbol_id, metadata
    except Exception as stage_exc:
        try:
            restored_id = place_ghost_pin_view(
                prj, pin_view_id, before_location['sheet_id'],
                before_location['x'], before_location['y'],
                before_rotation, before_scale, symbol_name)
            log(f"RESTORED original pin view {pin_view_id} as symbol {restored_id} "
                f"at ({before_location['x']}, {before_location['y']}).")
            raise StagingPlacementRejected(
                f"STAGING FAILED for pin view {pin_view_id}, but original "
                f"location was restored. Original error: {stage_exc}"
            ) from stage_exc
        except RuntimeError as restore_exception:
            if isinstance(restore_exception, StagingPlacementRejected):
                raise
            raise RuntimeError(
                f"CRITICAL: failed to stage pin view {pin_view_id} AND failed to "
                f"restore it automatically. Pin may be UNPLACED. "
                f"Staging error: {stage_exc}; restoration error: {restore_exception}. "
                "Do not rerun until you recover the pin in the saved project."
            ) from restore_exception
        except Exception as restore_exception:
            raise RuntimeError(
                f"CRITICAL: failed to stage pin view {pin_view_id} AND failed to "
                f"restore it automatically. Pin may be UNPLACED. "
                f"Staging error: {stage_exc}; restoration error: {restore_exception}. "
                "Do not rerun until you recover the pin in the saved project."
            ) from restore_exception
def find_logical_pin(dev, pin, connector_device_id, pin_name):
    target_name = normalize_pin_value(pin_name)
    dev.SetId(connector_device_id)
    for logical_pin_id in get_pin_ids(dev):
        pin.SetId(logical_pin_id)
        current_name = normalize_pin_value(pin.GetName())
        if current_name == target_name:
            return {
                "pin_name": current_name,
                "logical_pin_id": logical_pin_id,
                "view_ids": get_view_ids(pin),
            }
    return None
def peel_all_pins_to_staging(prj, dev, pin, sht, connector_device_id, connector_name, block_geometry, sorted_pin_records, staging_map, pin_geometry, placement_symbol_name, staged_records=None):
    if staged_records is None:
        staged_records = []
    allowed_characteristics = {
        "END_TOP",
        "END_BOTTOM",
        "SPLIT_BOTTOM",
        "END_SPLIT",
    }
    allowed_staged_characteristics = {
        "END_SPLIT",
        "END_TOP",
        "CONNECTOR_PIN",
        "SPLIT_BOTTOM",
        "END_BOTTOM",
    }
    for index, original_record in enumerate(sorted_pin_records):
        pin_name = original_record["pin_name"]
        # The selected block may be a VIEW of the owning main device, not a
        # separately named connector device. Use the exact preverified ID.
        pin_view_id = original_record['pin_view_id']
        try:
            pin.SetId(original_record['logical_pin_id'])
            current_view_ids = get_view_ids(pin)
        except Exception as exc:
            raise RuntimeError(f"Pin {pin_name}: logical pin vanished: {exc}")
        if pin_view_id not in current_view_ids:
            raise RuntimeError(f"Pin {pin_name}: selected view {pin_view_id} "
                               "no longer belongs to the verified logical pin.")
        live = resolve_live_symbol_from_pin_view(prj, pin_view_id)
        live_symbol_id = None
        live_sym = None
        live_resolution_source = None
        if live is not None:
            live_symbol_id = live["live_symbol_id"]
            live_sym = live["symbol_object"]
            current_location = get_symbol_schema_location(live_sym)
            if not location_is_inside_block(current_location, block_geometry):
                raise RuntimeError(
                    f"Refusing to move {connector_name} pin {pin_name}: "
                    f"the live symbol {live_symbol_id} is outside selected block "
                    f"{block_geometry['block_symbol_id']}."
                )
            live_resolution_source = "stable pin-view ID"
        else:
            scanned = resolve_current_live_pin_symbol(prj, sht, connector_name, pin_name, block_geometry)
            if scanned is None:
                raise RuntimeError(f"Pin {pin_name} has no live symbol when its peel turn arrived. Stable pin-view resolution and fresh sheet scan both failed.")
            live_symbol_id = scanned["symbol_id"]
            live_sym = prj.CreateSymbolObject()
            set_result = live_sym.SetId(live_symbol_id)
            if not set_result:
                raise RuntimeError(f"Fresh sheet scan found symbol {live_symbol_id} for pin {pin_name}, but sym.SetId failed.")
            live_resolution_source = "fresh sheet scan by connector + pin + block"
        characteristic = call_noarg(live_sym, "GetCharacteristic")
        rotation = call_noarg(live_sym, "GetRotation")
        scaling = call_noarg(live_sym, "GetScaling")
        if characteristic not in allowed_characteristics:
            raise RuntimeError(f"Pin {pin_name} is not peelable. Current characteristic is {characteristic}.")
        if not isinstance(rotation, str) or rotation == "":
            rotation = "y"
        try:
            scaling = float(scaling)
        except Exception:
            scaling = 1.0
        stage = staging_map[pin_name]
        log(f"STAGE {index + 1}/{len(sorted_pin_records)} | PIN {pin_name} | VIEW {pin_view_id} | LIVE {live_symbol_id} | SOURCE {live_resolution_source} | {characteristic} -> {stage['lane']} ROW {stage['row']} -> ({stage['x']}, {stage['y']})")
        stage_flip = stage['flip'] if stage.get('flip') is not None else rotation
        new_symbol_id, metadata = move_pin_by_delete_recreate(
            prj, pin_view_id, live_symbol_id, pin_geometry['sheet_id'],
            stage['x'], stage['y'], stage_flip, scaling, placement_symbol_name)
        # Record *before* validating so even an unexpected E3 return is
        # included in recovery if subsequent validation fails.
        staged_records.append({
            "pin_name": pin_name,
            "logical_pin_id": original_record["logical_pin_id"],
            "pin_view_id": pin_view_id,
            "last_returned_stage_symbol_id": new_symbol_id,
            "rotation": rotation,
            "stage_flip": stage_flip,
            "placement_symbol_name": placement_symbol_name,
            "scaling": scaling,
            "sheet_id": pin_geometry["sheet_id"],
            "stage_x": stage["x"],
            "stage_y": stage["y"],
            "stage_lane_index": stage["lane_index"],
            "stage_lane": stage["lane"],
            "stage_row": stage["row"],
            "stage_metadata_immediate": metadata,
        })
        if metadata['characteristic'] not in allowed_staged_characteristics:
            raise RuntimeError(f"Pin {pin_name} returned unexpected staged characteristic {metadata['characteristic']}.")
        actual_location = metadata['schema_location']
        if (actual_location is None or
                actual_location['sheet_id'] != int(pin_geometry['sheet_id']) or
                not coordinates_close(actual_location, stage['x'], stage['y'])):
            raise RuntimeError(
                f"Pin {pin_name} was reported placed at a different location "
                f"than expected: requested ({stage['x']}, {stage['y']}), "
                f"actual {actual_location!r}. Must restore before trying a fallback.")
        log(f"    RETURNED SYMBOL {new_symbol_id} | TYPE {metadata['symbol_type_name']} | CHARACTERISTIC {metadata['characteristic']} | BOUNDS {metadata['bound_ids']}")
    return staged_records
def check_original_pin_positions(prj, sht, connector_name, sorted_pin_records):
    """Verify restoration using original per-pin view associations and locations."""
    differences = []
    for original in sorted_pin_records:
        view_id = original['pin_view_id']
        live = resolve_live_symbol_from_pin_view(prj, view_id)
        location = None
        if live is not None:
            location = get_symbol_schema_location(live['symbol_object'])
        expected = (int(original['sheet_id']), float(original['x']), float(original['y']))
        if location is None or location['sheet_id'] != expected[0] or not coordinates_close(
                location, expected[1], expected[2]):
            # Stable view association isn't reliable on all E3 builds;
            # scan specifically for the unique labeled connector/pin/location.
            identity = resolve_symbol_at_location(
                prj, sht, expected[0], connector_name, original['pin_name'],
                expected[1], expected[2])
            if identity is None:
                differences.append((original['pin_name'], view_id, location, expected))
    return differences


def rollback_staged_connector(prj, sht, connector_name, staged_records,
                              sorted_pin_records, placement_symbol_name):
    """Restore all successfully staged pins before changing staging method.

    If a restore or any position verification fails, ABORT. Never try the
    fallback from a mixed original/partially staged block.
    """
    original_by_pin = {r['pin_name']: r for r in sorted_pin_records}
    log(f'{connector_name}: RESTORING {len(staged_records)} already-staged pin(s) '
        'in reverse peel order before changing strategies.')
    for staged in reversed(staged_records):
        pin_name = staged['pin_name']
        original = original_by_pin[pin_name]
        view_id = original['pin_view_id']
        live = resolve_live_symbol_from_pin_view(prj, view_id)
        live_id = None
        if live is not None:
            location = get_symbol_schema_location(live['symbol_object'])
            if (location is not None and
                    int(location['sheet_id']) == int(staged['sheet_id']) and
                    coordinates_close(location, staged['stage_x'], staged['stage_y'])):
                live_id = live['live_symbol_id']
        if live_id is None:
            scanned = resolve_symbol_at_location(
                prj, sht, staged['sheet_id'], connector_name, pin_name,
                staged['stage_x'], staged['stage_y'])
            if scanned is not None:
                live_id = scanned['symbol_id']
        if live_id is None:
            raise RuntimeError(
                f'CRITICAL: rollback cannot identify staged {connector_name} '
                f'pin {pin_name} at ({staged["stage_x"]}, {staged["stage_y"]}). '
                'Do not run the fallback; recover the saved project.')
        try:
            move_pin_by_delete_recreate(
                prj, view_id, live_id, original['sheet_id'],
                original['x'], original['y'],
                (original['rotation'] if isinstance(original['rotation'], str)
                 and original['rotation'] else 'y'),
                float(original['scaling']) if original['scaling'] else 1.0,
                placement_symbol_name)
        except Exception as exc:
            raise RuntimeError(
                f'CRITICAL: rollback failed for {connector_name} pin {pin_name}. '
                'Fallback was NOT attempted; recover the saved project. '
                f'Reason: {exc}') from exc
        log(f'    RESTORED {pin_name} to ({original["x"]}, {original["y"]})')
    issues = check_original_pin_positions(prj, sht, connector_name, sorted_pin_records)
    if issues:
        raise RuntimeError(
            f'CRITICAL: {connector_name} rollback could not verify original '
            f'locations for {len(issues)} pin(s): {issues[:10]!r}. '
            'Fallback was NOT attempted; recover the saved project.')
    log(f'{connector_name}: restored original pin positions VERIFIED.')


def build_final_plan(connector_excel_record, pin_geometry):
    final_plan = []
    for row_index, pin_name in enumerate(connector_excel_record["left pins"]):
        final_plan.append({
            "pin_name": pin_name,
            "side": "LEFT",
            "row": row_index,
            "x": pin_geometry["left_x"],
            "y": pin_geometry["final_top_y"] - (row_index * pin_geometry["pin_pitch"]),
            "flip": LEFT_FLIP,
        })
    for row_index, pin_name in enumerate(connector_excel_record["right pins"]):
        final_plan.append({
            "pin_name": pin_name,
            "side": "RIGHT",
            "row": row_index,
            "x": pin_geometry["right_x"],
            "y": pin_geometry["final_top_y"] - (row_index * pin_geometry["pin_pitch"]),
            "flip": RIGHT_FLIP,
        })
    return final_plan
def find_staged_record(staged_records, pin_name):
    for record in staged_records:
        if record["pin_name"] == pin_name:
            return record
    raise RuntimeError(f"Could not find staged record for pin {pin_name}.")
def resolve_symbol_at_location(prj, sht, sheet_id, connector_name, pin_name, x, y):
    candidate_ids = collect_live_symbol_candidates(prj, sht, sheet_id)
    matches = []
    for candidate_id in candidate_ids:
        identity = read_symbol_identity(prj, candidate_id)
        if identity is None:
            continue
        if (identity["location"] is None or
                identity["location"]["sheet_id"] != sheet_id):
            continue
        if identity["pin_number"] != pin_name:
            continue
        if not coordinates_close(identity["location"], x, y):
            continue
        if not match_connector_designation(identity['device_text'], connector_name):
            continue
        score = 230
        if identity["symbol_type_name"] in ("MilSymbolReceptacle¶END_TOP", "MilSymbolReceptacle¶END_SPLIT", "MilSymbolReceptacle¶SPLIT_BOTTOM", "MilSymbolReceptacle¶CONNECTOR_PIN", "MilSymbolReceptacle¶END_BOTTOM"):
            score += 10
        matches.append({
            "score": score,
            "identity": identity,
        })
    matches.sort(key=lambda item: item["score"], reverse=True)
    if len(matches) != 1:
        return None
    return matches[0]["identity"]
def move_center_stacks_to_final(prj, sht, connector_name, staged_records, global_final_sequence):
    allowed_center_endpoint_characteristics = {
        "END_TOP",
        "END_SPLIT",
        "END_BOTTOM",
        # The MilBlockSymbolReceptacleIInt family can return connector-pin
        # and split-bottom variants. Pin name + sheet + exact location must
        # still match uniquely before touching the symbol.
        "CONNECTOR_PIN",
        "SPLIT_BOTTOM",
    }
    for index, final_item in enumerate(global_final_sequence):
        staged_record = find_staged_record(staged_records, final_item["pin_name"])
        log(f"FINAL {index + 1}/{len(global_final_sequence)} | PIN {final_item['pin_name']} | FROM {staged_record['stage_lane']} ROW {staged_record['stage_row']} -> {final_item['side']} ROW {final_item['row']} | FLIP {final_item['flip']} -> ({final_item['x']}, {final_item['y']})")
        identity = resolve_symbol_at_location(prj, sht, staged_record["sheet_id"], connector_name, staged_record["pin_name"], staged_record["stage_x"], staged_record["stage_y"])
        if identity is None:
            raise RuntimeError(f"Could not rediscover live temporary edge symbol for pin {staged_record['pin_name']} at ({staged_record['stage_x']}, {staged_record['stage_y']}).")
        if identity["characteristic"] not in allowed_center_endpoint_characteristics:
            raise RuntimeError(f"Pin {staged_record['pin_name']} is not exposed at the top/end of {staged_record['stage_lane']}. Current characteristic is {identity['characteristic']}.")
        current_live_symbol_id = identity["symbol_id"]
        log(f"    CURRENT STAGED EDGE SYMBOL: {current_live_symbol_id} | TYPE {identity['symbol_type_name']} | CHARACTERISTIC {identity['characteristic']}")
        new_symbol_id, metadata = move_pin_by_delete_recreate(
            prj, staged_record["pin_view_id"], current_live_symbol_id,
            staged_record["sheet_id"], final_item["x"], final_item["y"],
            final_item["flip"], staged_record["scaling"],
            staged_record["placement_symbol_name"])
        staged_record["final_returned_symbol_id"] = new_symbol_id
        staged_record["final_side"] = final_item["side"]
        staged_record["final_row"] = final_item["row"]
        staged_record["final_x"] = final_item["x"]
        staged_record["final_y"] = final_item["y"]
        staged_record["final_flip"] = final_item["flip"]
        staged_record["final_metadata_immediate"] = metadata
        log(f"    FINAL RETURNED SYMBOL: {new_symbol_id} | TYPE {metadata['symbol_type_name']} | CHARACTERISTIC {metadata['characteristic']} | BOUNDS {metadata['bound_ids']}")
        time.sleep(0.20)
def collect_live_symbol_candidates(prj, sht, sheet_id):
    candidate_ids = set()
    if hasattr(prj, "GetSymbolIds"):
        try:
            for symbol_id in call_id_array(prj.GetSymbolIds):
                candidate_ids.add(symbol_id)
        except Exception:
            pass
    sht.SetId(sheet_id)
    if hasattr(sht, "GetSymbolIds"):
        try:
            for symbol_id in call_id_array(sht.GetSymbolIds):
                candidate_ids.add(symbol_id)
        except Exception:
            pass
    return sorted(candidate_ids)
def read_symbol_identity(prj, symbol_id):
    local_sym = prj.CreateSymbolObject()
    local_text = prj.CreateTextObject()
    try:
        set_result = local_sym.SetId(symbol_id)
    except Exception:
        return None
    if not set_result:
        return None
    pin_number = None
    device_text = None
    for text_id in get_text_ids(local_sym):
        try:
            local_text.SetId(text_id)
            text_type = local_text.GetType()
            text_value = str(local_text.GetText())
        except Exception:
            continue
        if text_type == 203:
            pin_number = normalize_pin_value(text_value)
        if text_type == 12:
            device_text = text_value
    return {
        "symbol_id": int(local_sym.GetId()),
        "pin_number": pin_number,
        "device_text": device_text,
        "location": get_symbol_schema_location(local_sym),
        "symbol_type_name": call_noarg(local_sym, "GetSymbolTypeName"),
        "characteristic": call_noarg(local_sym, "GetCharacteristic"),
    }
def coordinates_close(location, x, y):
    if location is None:
        return False
    try:
        current_x = float(location["x"])
        current_y = float(location["y"])
    except Exception:
        return False
    return abs(current_x - float(x)) <= SYMBOL_SCAN_COORDINATE_TOLERANCE and abs(current_y - float(y)) <= SYMBOL_SCAN_COORDINATE_TOLERANCE
def resolve_final_symbol(prj, candidate_ids, connector_name, staged_record):
    matches = []
    for candidate_id in candidate_ids:
        identity = read_symbol_identity(prj, candidate_id)
        if identity is None:
            continue
        if (identity["location"] is None or
                identity["location"]["sheet_id"] != staged_record["sheet_id"]):
            continue
        if identity["pin_number"] != staged_record["pin_name"]:
            continue
        if not coordinates_close(identity['location'], staged_record['final_x'], staged_record['final_y']):
            continue
        if not match_connector_designation(identity['device_text'], connector_name):
            continue
        score = 230
        if identity["symbol_type_name"] in ("MilSymbolReceptacle¶END_TOP", "MilSymbolReceptacle¶END_SPLIT", "MilSymbolReceptacle¶SPLIT_BOTTOM", "MilSymbolReceptacle¶CONNECTOR_PIN"):
            score += 10
        matches.append({
            "score": score,
            "identity": identity,
        })
    matches.sort(key=lambda item: item["score"], reverse=True)
    if len(matches) != 1:
        return None
    return matches[0]["identity"]
def apply_text_tweaks(prj, symbol_id, side, is_top_of_side):
    local_sym = prj.CreateSymbolObject()
    local_text = prj.CreateTextObject()
    set_result = local_sym.SetId(symbol_id)
    if not set_result:
        raise RuntimeError(f"Could not resolve symbol {symbol_id} for formatting.")
    if side == "LEFT":
        signal_text_alignment = 1
    elif side == "RIGHT":
        signal_text_alignment = 3
    else:
        raise RuntimeError(f"Unknown final pin side {side}.")
    local_sym.SetBlockHatchColour(58)
    for text_id in get_text_ids(local_sym):
        local_text.SetId(text_id)
        try:
            text_type = local_text.GetType()
        except Exception:
            continue
        if text_type == 5:
            local_text.SetVisibility(0)
        elif text_type == 12:
            if is_top_of_side:
                local_text.SetVisibility(1)
                local_text.SetStyle(1)
                local_text.SetHeight(0.2)
            else:
                local_text.SetVisibility(0)
        elif text_type == 617:
            local_text.SetVisibility(1)
            local_text.SetColour(0)
            local_text.SetAlignment(signal_text_alignment)
            local_text.SetStyle(1)
            local_text.SetHeight(0.15)
        elif text_type == 203:
            local_text.SetVisibility(1)
            local_text.SetStyle(1)
def resolve_and_format_final_pins(prj, sht, connector_name, staged_records):
    if not staged_records:
        return
    candidate_ids = collect_live_symbol_candidates(prj, sht, staged_records[0]["sheet_id"])
    for staged_record in staged_records:
        identity = resolve_final_symbol(prj, candidate_ids, connector_name, staged_record)
        if identity is None:
            raise RuntimeError(f"Could not rediscover final symbol for {connector_name} pin {staged_record['pin_name']}.")
        side = staged_record["final_side"]
        is_top_of_side = staged_record["final_row"] == 0
        if side == "LEFT":
            signal_text_alignment = 1
        else:
            signal_text_alignment = 3
        log(f"FORMAT PIN {staged_record['pin_name']} | SIDE {side} | ROW {staged_record['final_row']} | SYMBOL {identity['symbol_id']} | SIGNAL ALIGNMENT {signal_text_alignment} | DEVICE DESIGNATION {'ON' if is_top_of_side else 'OFF'}")
        apply_text_tweaks(prj, identity["symbol_id"], side, is_top_of_side)
        staged_record["final_resolved_symbol_id"] = identity["symbol_id"]
