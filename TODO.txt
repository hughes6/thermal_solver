def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def generate_master_schematic_layout(PIN_SIGNAL_INFO,x_block,main_block_width):
    SHEET_DICT = {1:sheet_1_id}

    # PAGE BOUNDS
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    # PAGE MARGINS
    TOP_MARGIN = 1.0
    BOTTOM_MARGIN = 1.0
    LEFT_MARGIN = 1.0
    RIGHT_MARGIN = 1.0

    # BLOCK SPACING
    BLOCK_X_SPACING = 1.0
    BLOCK_Y_SPACING = 1.0
    SCHEMATIC_BLOCK_WIDTH = 3.0
    PIN_SPACING = 0.25
    PIN_START_OFFSET = 2.25

    # USABLE PAGE LIMITS
    PAGE_TOP = TOP_OF_PAGE - TOP_MARGIN
    PAGE_BOTTOM = BOTTOM_OF_PAGE + BOTTOM_MARGIN
    PAGE_LEFT = LEFT_OF_PAGE + LEFT_MARGIN
    PAGE_RIGHT = RIGHT_OF_PAGE - RIGHT_MARGIN
    USABLE_PAGE_HEIGHT = PAGE_TOP - PAGE_BOTTOM

    # OUTPUT
    SPACING_DICT = []
    sheet = 1
    # On sheet 1, start to the RIGHT of the main block
    current_x = (x_block + main_block_width + BLOCK_X_SPACING)

    # Current Y is the TOP edge available for placement
    current_top_y = PAGE_TOP
    # Tracks widest block in current column
    current_column_width = 0.0

    # ================================================
    # BUILD LAYOUT
    # ================================================
    for i in range(len(PIN_SIGNAL_INFO)):
        connector_name = (PIN_SIGNAL_INFO[i][0]["connector name"])
        number_of_pins = (PIN_SIGNAL_INFO[i][0]["number of pins"])
        width = SCHEMATIC_BLOCK_WIDTH
        height = (PIN_START_OFFSET + (PIN_SPACING * number_of_pins))

        # --------------------------------------------
        # SAFETY CHECK:
        # Block cannot physically fit on ANY page
        # --------------------------------------------
        if height > USABLE_PAGE_HEIGHT:
            raise Exception(f"Schematic block for {connector_name} is too tall for a page. Height = {height}, usable page height = {USABLE_PAGE_HEIGHT}")

        # CALCULATE BOTTOM Y IF PLACED HERE
        candidate_y = current_top_y - height

        # DOES IT FIT VERTICALLY?
        if candidate_y < PAGE_BOTTOM:
            next_x = (current_x + current_column_width + BLOCK_X_SPACING)

            # DOES NEXT COLUMN FIT ON THIS PAGE?
            if next_x + width <= PAGE_RIGHT:
                current_x = next_x
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            else:
                # PAGE IS FULL -> NEW PAGE
                sheet += 1
                sheet_symbol = "ANSI Size E - BAE"
                res = sht.Create(0, str(sheet), sheet_symbol, 0, 0)
                if res == 0:
                    app.PutMessage("failed to create new sheet")
                    raise Exception("failed to create new sheet")
                SHEET_DICT[sheet] = res
                current_x = PAGE_LEFT
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            # Recalculate Y at new column/page
            candidate_y = current_top_y - height

        # CHECK HORIZONTAL FIT
        if current_x + width > PAGE_RIGHT:
            sheet += 1
            res = sht.Create(0, sheet, "DINA3", 0)
            if res == 0:
                app.PutMessage("failed to create new sheet")
                raise Exception("failed to create new sheet")
            SHEET_DICT[sheet] = res
            current_x = PAGE_LEFT
            current_top_y = PAGE_TOP
            current_column_width = 0.0
            candidate_y = current_top_y - height

        # SAVE PLACEMENT
        x_start = current_x
        y_start = candidate_y
        SPACING_DICT.append({"connector name": connector_name, "connector index": i, "sheet": sheet, "x start": x_start, "y start": y_start, "x end": x_start + width, "y end": y_start + height, "height": height, "width": width, "number of pins": number_of_pins})
        # UPDATE CURSOR FOR NEXT BLOCK
        current_top_y = (y_start - BLOCK_Y_SPACING)
        current_column_width = max(current_column_width, width)

    return SPACING_DICT, SHEET_DICT



def generate_schematic_pin_map(connector_info,newpindict,layout,pin_spacing=0.25,pin_start_offset=2.0,edge_offset=0.3):
    x=layout["x start"]+layout["width"]-0.3
    y_start=layout["y start"]+layout["height"]-pin_start_offset
    pin_map=[]
    for pos,pin_data in enumerate(connector_info[1:]):
        pin_name=str(pin_data[0]).strip()
        resolved_pin_name = pin_name

        # Normal case: the Excel pin name exactly matches the E3 view pin name.
        if resolved_pin_name not in newpindict:
            # Minimal fallback for library pins such as "1-C" / "2-S" when
            # Excel identifies those same pins as "1" / "2".  Only use the
            # fallback when there is exactly one unambiguous match.
            prefix_matches = [
                e3_pin_name
                for e3_pin_name in newpindict
                if str(e3_pin_name).strip().split("-", 1)[0] == pin_name
            ]

            if len(prefix_matches) == 1:
                resolved_pin_name = prefix_matches[0]
                print(
                    "PIN NAME FALLBACK:",
                    connector_info[0]['connector name'],
                    "EXCEL:", pin_name,
                    "E3:", resolved_pin_name
                )
            else:
                raise Exception(
                    f"Pin {pin_name} not found in new view for {connector_info[0]['connector name']}. "
                    f"Available E3 pins: {list(newpindict.keys())}"
                )

        pin_map.append({"pin name":pin_name,"pin id":newpindict[resolved_pin_name],"x":x,"y":y_start-(pos*pin_spacing),"flip":"/1"})
    return pin_map

def format_main_block_view(device_view_id, x, y, width, height, block_symbol_id=None):
    print("FORMATTING DEVICE VIEW:", device_view_id)

    x_center = x + (width / 2.0)

    # Split blocks share one logical device view.  Do NOT reformat every
    # symbol from that view for each connector's independent block layout.
    # The saved exact symbol ID ensures each split block keeps its own text.
    if block_symbol_id is not None:
        symids = [block_symbol_id]
    else:
        dev.SetId(device_view_id)
        syminfo = dev.GetSymbolIds(None)
        symids = [s for s in syminfo[1] if s is not None]

    for sym_id in symids:
        sym.SetId(sym_id)

        sym.SetBlockHatchColour(58)
        sym.SetBlockOutlineColour(0)

        #=========================================
        # NORMAL SYMBOL TEXT
        #=========================================
        textinfo = sym.GetTextIds(None)
        textids = [t for t in textinfo[1] if t is not None]

        for text_id in textids:
            text.SetId(text_id)
            txt = str(text.GetText()).strip()

            print("SYMBOL TEXT:", "VIEW:", device_view_id, "SYM:", sym_id, "TEXT ID:", text_id, "VALUE:", repr(txt))

            if txt == "BxRxUx":
                text.SetColour(0)
                text.SetMode(1)
                text.SetAlignment(2)
                text.SetStyle(1)
                text.SetHeight(TEXT_SIZE)
                text.SetSchemaLocation(x_center, y + height - 0.3)

            elif txt == str(DEVICE_DESIGNATION).strip() or str(DEVICE_DESIGNATION).strip() in txt:
                text.SetColour(0)
                text.SetMode(1)
                text.SetAlignment(2)
                text.SetStyle(1)
                text.SetHeight(TEXT_SIZE)
                text.SetSchemaLocation(x_center, y + height - 0.7)

                print("MOVED DEVICE DESIGNATION:", repr(txt), "TO:", x_center, y + height - 0.7)

            elif DEVICE_FUNCTION is not None and txt == str(DEVICE_FUNCTION).strip():
                text.SetVisibility(0)

        #=========================================
        # ATTRIBUTE TEXT
        #=========================================
        attrinfo = sym.GetAttributeIds(None)
        attrids = [a for a in attrinfo[1] if a is not None]

        for attr_id in attrids:
            attr.SetId(attr_id)
            attr_name = str(attr.GetName()).strip()

            print("ATTRIBUTE:", "VIEW:", device_view_id, "SYM:", sym_id, "ATTR ID:", attr_id, "NAME:", repr(attr_name))

            if attr_name == "Equipment Identifacation":
                attrtextinfo = attr.GetTextIds(None)
                attrtextids = [t for t in attrtextinfo[1] if t is not None]

                print("EQUIPMENT IDENTIFICATION TEXT IDS:", attrtextids)

                for text_id in attrtextids:
                    text.SetId(text_id)
                    txt = str(text.GetText()).strip()

                    print("ATTRIBUTE TEXT:", "ID:", text_id, "VALUE:", repr(txt))

                    text.SetVisibility(1)
                    text.SetColour(0)
                    text.SetMode(1)
                    text.SetAlignment(2)
                    text.SetStyle(1)
                    text.SetHeight(0.2)
                    text.SetSchemaLocation(x_center, y + height - 1.1)

                    print("MOVED EQUIPMENT IDENTIFICATION:", repr(txt), "TO:", x_center, y + height - 1.1)

def runn(secondary=False, TEXT_SIZE=0.2):
    # ================================================
    # CONNECT TO ACTIVE E3 PROJECT
    # ================================================
    app = win32.GetActiveObject("CT.Application")
    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()
    sym = prj.CreateSymbolObject()
    text = prj.CreateTextObject()
    attr = prj.CreateAttributeObject()


    # ================================================
    # HELPERS
    # ================================================
    def extract_ids(result):
        if result is None:
            return []

        if isinstance(result, (tuple, list)):
            if len(result) >= 2 and isinstance(result[1], (tuple, list)):
                return [x for x in result[1] if x is not None and x != 0]

            return [x for x in result if isinstance(x, int) and x != 0]

        if isinstance(result, int):
            if result != 0:
                return [result]

        return []


    def get_visibility(text_id):
        text.SetId(text_id)

        try:
            return text.GetVisibility()
        except Exception as e:
            return "GET VIS ERROR: " + repr(e)


    def get_pin_view_ids(pin_id):
        pin.SetId(pin_id)

        try:
            result = pin.GetViewIds(None)
            return extract_ids(result)
        except Exception:
            pass

        try:
            result = pin.GetViewIds()
            return extract_ids(result)
        except Exception as e:
            print("GetViewIds ERROR:", pin_id, repr(e))
            return []


    def get_schema_location(text_id):
        text.SetId(text_id)

        xloc = [0.0]
        yloc = [0.0]
        grid = [""]
        column = [""]
        row = [""]

        try:
            text.GetSchemaLocation(xloc, yloc, grid, column, row)
            return float(xloc[0]), float(yloc[0])
        except Exception as e:
            print("GET SCHEMA LOCATION ERROR:", text_id, repr(e))
            return None, None


    def format_equipment_identification(TEXT_SIZE):
        print("")
        print("============================================================")
        print("FORMATTING EQUIPMENT IDENTIFACATION")
        print("USING EXACT BLOCK SYMBOL IDS FROM MAIN.PY")
        print("============================================================")

        metadata_path = os.environ.get("E3_SECONDARY_BLOCK_FILE")


        if not metadata_path:
            print("E3_SECONDARY_BLOCK_FILE WAS NOT PROVIDED BY LAUNCHER.PY")
            return 0

        if not os.path.isfile(metadata_path):
            print("SECONDARY BLOCK METADATA FILE NOT FOUND:", metadata_path)
            return 0

        try:
            with open(metadata_path, "r", encoding="utf-8") as metadata_handle:
                secondary_blocks = json.load(metadata_handle)
        except Exception as e:
            print("FAILED TO READ SECONDARY BLOCK METADATA:", repr(e))
            return 0

        formatted_count = 0
        block_count = 0
        found_attribute_count = 0

        for block_data in secondary_blocks:
            block_count += 1

            block_name = str(block_data.get("name", ""))
            block_symbol_id = int(block_data["block_symbol_id"])
            parent_block = int(block_data["parent_block"])
            x = float(block_data["x"])
            y = float(block_data["y"])
            width = float(block_data["width"])
            height = float(block_data["height"])

            x_center = x + (width / 2.0)
            equipment_y = y + height - 1.1

            print("")
            print("SECONDARY BLOCK:", block_name, "VIEW:", parent_block, "SYMBOL:", block_symbol_id)

            try:
                sym.SetId(block_symbol_id)
            except Exception as e:
                print("FAILED TO SET BLOCK SYMBOL ID:", block_symbol_id, repr(e))
                continue

            try:
                attribute_result = sym.GetAttributeIds(None)
                attribute_ids = extract_ids(attribute_result)
            except Exception as e:
                print("FAILED TO GET BLOCK ATTRIBUTES:", block_symbol_id, repr(e))
                continue

            print("BLOCK ATTRIBUTE IDS:", attribute_ids)

            equipment_attribute_id = None

            for attribute_id in attribute_ids:
                attr.SetId(attribute_id)

                try:
                    attribute_name = str(attr.GetName()).strip()
                except Exception:
                    continue

                print("BLOCK ATTRIBUTE:", attribute_id, repr(attribute_name))

                if attribute_name == "Equipment Identifacation":
                    equipment_attribute_id = attribute_id
                    break

            if equipment_attribute_id is None:
                print("EQUIPMENT IDENTIFACATION NOT FOUND ON EXACT BLOCK SYMBOL:", block_symbol_id)
                continue

            found_attribute_count += 1
            attr.SetId(equipment_attribute_id)

            equipment_value = ""

                equipment_value = str(attr.GetValue()).strip()
            except Exception:
                try:
