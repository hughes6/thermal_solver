def ask_column_names(default_names):
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    default_text = ",".join(default_names)
    entered_text = simpledialog.askstring("Column Names", "Enter the 9 column names from LEFT to RIGHT, separated by commas:", initialvalue=default_text, parent=root)
    root.destroy()
    if entered_text is None:
        return None

    column_names = []

    for value in entered_text.split(","):
        value = value.strip()
        if value != "":
            column_names.append(value)

    if len(column_names) != EXPECTED_COLUMN_COUNT:
        raise ValueError(f"Exactly {EXPECTED_COLUMN_COUNT} column names are required. You entered {len(column_names)}.")

    return column_names


def excel_column_to_number(column_letter):
    column_letter = column_letter.strip().upper()
    result = 0

    for character in column_letter:
        result = result * 26 + ord(character) - ord("A") + 1

    return result


def normalize_connector_name(value):
    if value is None:
        return ""

    value = str(value).strip().upper()
    value = re.sub(r"\s+", "", value)

    return value

# RAW SECTION EXTRACTION
# ================================================
def extract_sections_by_keyword(pdf_path, start_page, separator_keyword):
    sections = []
    current_section = []
    current_start_page = None
    keyword_lower = separator_keyword.lower()

    with pdfplumber.open(pdf_path) as pdf:
        if start_page > len(pdf.pages):
            raise ValueError(f"Start page {start_page} is greater than PDF page count {len(pdf.pages)}")

        for page_number in range(start_page, len(pdf.pages) + 1):
            page = pdf.pages[page_number - 1]
            text = page.extract_text()

            if not text:
                continue

            lines = text.splitlines()

            for line in lines:
                clean_line = line.strip()

                if keyword_lower in clean_line.lower():
                    if current_section:
                        sections.append({
                            "section_number": len(sections) + 1,
                            "start_page": current_start_page,
                            "text": "\n".join(current_section).strip()
                        })

                    current_section = [clean_line]
                    current_start_page = page_number

                else:
                    if current_section:
                        current_section.append(clean_line)

        if current_section:
            sections.append({
                "section_number": len(sections) + 1,
                "start_page": current_start_page,
                "text": "\n".join(current_section).strip()
            })

    return sections

def extract_page_words(page):
    return page.extract_words(x_tolerance=3, y_tolerance=3, keep_blank_chars=False, use_text_flow=False)


def word_center_x(word):
    return (float(word["x0"]) + float(word["x1"])) / 2.0


def group_words_into_lines(words, y_tolerance=3.0):
    lines = []
    sorted_words = sorted(words, key=lambda word: (float(word["top"]), float(word["x0"])))

    for word in sorted_words:
        word_top = float(word["top"])
        target_line = None

        for line in lines:
            if abs(line["top"] - word_top) <= y_tolerance:
                target_line = line
                break

        if target_line is None:
            target_line = {"top": word_top, "words": []}
            lines.append(target_line)

        target_line["words"].append(word)

    for line in lines:
        line["words"] = sorted(line["words"], key=lambda word: float(word["x0"]))
        line["text"] = " ".join(str(word["text"]) for word in line["words"]).strip()

    return lines


def find_exact_word(words, wanted_text, target_y=None, y_tolerance=60.0):
    candidates = []

    for word in words:
        text = str(word["text"]).strip().upper()

        if text != wanted_text.upper():
            continue

        if target_y is not None and abs(float(word["top"]) - target_y) > y_tolerance:
            continue

        candidates.append(word)

    if not candidates:
        return None

    if target_y is None:
        return candidates[0]

    candidates.sort(key=lambda word: abs(float(word["top"]) - target_y))

    return candidates[0]


# ================================================
# MALFORMED TEXT REPAIR
# ================================================
def repair_fragmented_text(value):
    value = str(value).strip()

    if value == "":
        return ""

    tokens = value.split()

    if len(tokens) < 3:
        return value

    repaired = []
    index = 0

    while index < len(tokens):
        token = tokens[index]

        if len(token) <= 2 and re.fullmatch(r"[A-Za-z0-9]+", token):
            run = [token]
            run_index = index + 1

            while run_index < len(tokens):
                next_token = tokens[run_index]

                if len(next_token) <= 2 and re.fullmatch(r"[A-Za-z0-9]+", next_token):
                    run.append(next_token)
                    run_index += 1
                else:
                    break

            joined_length = sum(len(item) for item in run)

            if len(run) >= 3 and joined_length >= 4:
                repaired.append("".join(run))
                index = run_index
                continue

        repaired.append(token)
        index += 1

    return " ".join(repaired)


def rebuild_cell_text(words, same_line_tolerance=3.0, join_gap=2.5):
    if not words:
        return ""

    words = sorted(words, key=lambda word: (float(word["top"]), float(word["x0"])))

    lines = []
    current_line = []
    current_y = None

    for word in words:
        word_y = float(word["top"])

        if current_y is None:
            current_y = word_y
            current_line = [word]
            continue

        if abs(word_y - current_y) <= same_line_tolerance:
            current_line.append(word)

        else:
            lines.append(current_line)
            current_line = [word]
            current_y = word_y

    if current_line:
        lines.append(current_line)

    rebuilt_lines = []

    for line in lines:
        line = sorted(line, key=lambda word: float(word["x0"]))

        text = ""
        previous_x1 = None

        for word in line:
            value = str(word["text"]).strip()

            if value == "":
                continue

            x0 = float(word["x0"])
            x1 = float(word["x1"])

            if text == "":
                text = value

            else:
                gap = x0 - previous_x1

                if gap <= join_gap:
                    text += value
                else:
                    text += " " + value

            previous_x1 = x1

        if text != "":
            rebuilt_lines.append(text)

    value = " ".join(rebuilt_lines).strip()
    value = repair_fragmented_text(value)

    return value


def clean_extracted_value(value):
    value = str(value).strip()
    value = re.sub(r"\s+", " ", value)
    value = re.sub(r"\s+([,.;:])", r"\1", value)
    value = repair_fragmented_text(value)

    return value.strip()

# LEARN TABLE COLUMN LOCATIONS
# ================================================
def find_column_centers(pdf_path, start_page):
    with pdfplumber.open(pdf_path) as pdf:
        for page_number in range(start_page, len(pdf.pages) + 1):
            page = pdf.pages[page_number - 1]
            words = extract_page_words(page)

            if not words:
                continue

            lines = group_words_into_lines(words)

            for line in lines:
                upper_values = [str(word["text"]).strip().upper() for word in line["words"]]

                if "MNEMONIC" not in upper_values:
                    continue

                if "DESCRIPTION" not in upper_values:
                    continue

                header_y = line["top"]
                nearby_words = []

                for word in words:
                    if abs(float(word["top"]) - header_y) <= 60.0:
                        nearby_words.append(word)

                des_words = []
                posn_words = []

                for word in nearby_words:
                    text = str(word["text"]).strip().upper()

                    if text == "DES":
                        des_words.append(word)

                    if text == "POSN":
                        posn_words.append(word)

                des_words.sort(key=lambda word: float(word["x0"]))
                posn_words.sort(key=lambda word: float(word["x0"]))

                if len(des_words) < 2:
                    continue

                if len(posn_words) < 2:
                    continue

                mnemonic_word = find_exact_word(nearby_words, "MNEMONIC", header_y)
                description_word = find_exact_word(nearby_words, "DESCRIPTION", header_y)
                wire_word = find_exact_word(nearby_words, "WIRE", header_y)
                note_word = find_exact_word(nearby_words, "NOTE", header_y)
                comment_word = find_exact_word(nearby_words, "COMMENT", header_y)

                if mnemonic_word is None:
                    continue

                if description_word is None:
                    continue

                if wire_word is None:
                    continue

                if note_word is None:
                    continue

                if comment_word is None:
                    continue

                centers = [
                    word_center_x(des_words[0]),
                    word_center_x(posn_words[0]),
                    word_center_x(des_words[1]),
                    word_center_x(posn_words[1]),
                    word_center_x(mnemonic_word),
                    word_center_x(description_word),
                    word_center_x(wire_word),
                    word_center_x(note_word),
                    word_center_x(comment_word)
                ]

                centers_are_valid = True

                for index in range(len(centers) - 1):
                    if centers[index] >= centers[index + 1]:
                        centers_are_valid = False
                        break

                if not centers_are_valid:
                    continue

                print("")
                print("============================================================")
                print("TABLE COLUMNS LEARNED")
                print("============================================================")
                print("PAGE:", page_number)

                for index, center in enumerate(centers, start=1):
                    print("COLUMN", index, "X =", round(center, 2))

                print("============================================================")

                return centers, float(page.width)

    raise Exception("Could not automatically determine table column locations from PDF headers")


def build_column_boundaries(column_centers, page_width):
    boundaries = [0.0]

    for index in range(len(column_centers) - 1):
        midpoint = (column_centers[index] + column_centers[index + 1]) / 2.0
        boundaries.append(midpoint)

    boundaries.append(page_width)

    return boundaries


def get_column_index_for_word(word, boundaries):
    x = float(word["x0"])

    for index in range(len(boundaries) - 1):
        if boundaries[index] <= x < boundaries[index + 1]:
            return index

    return None


# ================================================
# SECTION MARKERS
# ================================================
def find_section_markers(pdf_path, start_page, separator_keyword):
    markers = []
    keyword_lower = separator_keyword.lower()

    with pdfplumber.open(pdf_path) as pdf:
        for page_number in range(start_page, len(pdf.pages) + 1):
            page = pdf.pages[page_number - 1]
            words = extract_page_words(page)

            if not words:
                continue

            lines = group_words_into_lines(words)

            for line in lines:
                if keyword_lower in line["text"].lower():
                    markers.append({
                        "section_number": len(markers) + 1,
                        "page": page_number,
                        "top": float(line["top"]),
                        "title": line["text"]
                    })

    return markers


def find_section_for_row(page_number, row_top, section_markers):
    selected_marker = None

    for marker in section_markers:
        marker_position = (marker["page"], marker["top"])
        row_position = (page_number, row_top)

        if marker_position <= row_position:
            selected_marker = marker
        else:
            break

    return selected_marker


# ================================================
# CONNECTOR ROW DETECTION
# ================================================
def is_connector_name(text):
    text = str(text).strip()

    return re.fullmatch(r"J\d+", text, re.IGNORECASE) is not None


def find_connector_anchors(words, boundaries):
    anchors = []

    for word in words:
        text = str(word["text"]).strip()

        if not is_connector_name(text):
            continue

        column_index = get_column_index_for_word(word, boundaries)

        if column_index != 0:
            continue

        anchors.append(word)

    anchors.sort(key=lambda word: float(word["top"]))

    return anchors


# ================================================
# FILTER NON-TABLE TEXT
# ================================================
def is_junk_line(line_text, separator_keyword):
    text = line_text.strip()
    upper = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper):
        return True

    if upper in {
        "FROM",
        "TO",
        "SIGNAL NAME",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE",
        "CODE",
        "WIRE CODE",
        "NOTE",
        "COMMENT",
        "WIRE CODES"
    }:
        return True

    if upper.startswith("DES POSN DES POSN"):
        return True

    if separator_keyword.lower() in text.lower():
        return True

    return False


# ================================================
# PARSE ONE TABLE ROW
# ================================================
def parse_row_words(row_words, boundaries, column_names):
    column_words = [[] for _ in column_names]

    for word in row_words:
        column_index = get_column_index_for_word(word, boundaries)

        if column_index is None:
            continue

        if column_index < 0 or column_index >= len(column_names):
            continue

        column_words[column_index].append(word)

    output = {}

    for index, column_name in enumerate(column_names):
        value = rebuild_cell_text(column_words[index])
        value = clean_extracted_value(value)
        output[column_name] = value

    return output


# ================================================
# EXTRACT STRUCTURED ROWS
# ================================================
def extract_structured_rows(pdf_path, start_page, separator_keyword, column_names):
    column_centers, page_width = find_column_centers(pdf_path, start_page)
    boundaries = build_column_boundaries(column_centers, page_width)
    section_markers = find_section_markers(pdf_path, start_page, separator_keyword)
    structured_rows = []

    with pdfplumber.open(pdf_path) as pdf:
        for page_number in range(start_page, len(pdf.pages) + 1):
            page = pdf.pages[page_number - 1]
            words = extract_page_words(page)

            if not words:
                continue

            lines = group_words_into_lines(words)
            anchors = find_connector_anchors(words, boundaries)

            if not anchors:
                continue

            for anchor_index, anchor in enumerate(anchors):
                row_top = float(anchor["top"]) - 2.0

                if anchor_index + 1 < len(anchors):
                    row_bottom = float(anchors[anchor_index + 1]["top"]) - 1.0
                else:
                    # IMPORTANT:
                    # Last row on the page is allowed to continue all
                    # the way to the bottom of the page.
                    row_bottom = float(page.height)

                row_words = []

                for line in lines:
                    if line["top"] < row_top:
                        continue

                    if line["top"] >= row_bottom:
                        continue

                    if is_junk_line(line["text"], separator_keyword):
                        continue

                    for word in line["words"]:
                        row_words.append(word)

                parsed_row = parse_row_words(row_words, boundaries, column_names)
                connector_key = column_names[CONNECTOR_SOURCE_INDEX]

                if not is_connector_name(parsed_row.get(connector_key, "")):
                    continue

                section_marker = find_section_for_row(page_number, float(anchor["top"]), section_markers)

                parsed_row["PAGE"] = page_number

                if section_marker is None:
                    parsed_row["SECTION"] = None
                    parsed_row["SECTION_TITLE"] = ""
                else:
                    parsed_row["SECTION"] = section_marker["section_number"]
                    parsed_row["SECTION_TITLE"] = section_marker["title"]

                structured_rows.append(parsed_row)

    return structured_rows


# ================================================
# VALIDATION
# ================================================
def validate_row(row, column_names):
    problems = []

    connector_key = column_names[CONNECTOR_SOURCE_INDEX]
    pin_key = column_names[PIN_SOURCE_INDEX]

    connector = str(row.get(connector_key, "")).strip()
    pin_number = str(row.get(pin_key, "")).strip()

    if not re.fullmatch(r"J\d+", connector, re.IGNORECASE):
        problems.append("invalid connector")

    if not re.fullmatch(r"\d+", pin_number):
        problems.append("invalid pin number")

    return problems


def validate_structured_rows(structured_rows, column_names):
    good_rows = []
    suspicious_rows = []

    for row in structured_rows:
        problems = validate_row(row, column_names)

        if problems:
            suspicious_rows.append({
                "row": row,
                "problems": problems
            })
        else:
            good_rows.append(row)

    return good_rows, suspicious_rows

def pin_sort_key(row, pin_field):
    value = str(row.get(pin_field, "")).strip()

    try:
        return int(value)
    except Exception:
        return 999999


def group_rows_by_connector(structured_rows, column_names):
    connector_field = column_names[CONNECTOR_SOURCE_INDEX]
    pin_field = column_names[PIN_SOURCE_INDEX]
    connector_groups = {}

    for row in structured_rows:
        connector_name = normalize_connector_name(row.get(connector_field, ""))

        if connector_name == "":
            continue

        if connector_name not in connector_groups:
            connector_groups[connector_name] = []

        connector_groups[connector_name].append(row)

    for connector_name in connector_groups:
        connector_groups[connector_name].sort(key=lambda row: pin_sort_key(row, pin_field))

    return connector_groups


# ================================================
# REMOVE DUPLICATE CONNECTOR/PIN ROWS
# ================================================
def remove_duplicate_pins(connector_groups, column_names):
    pin_field = column_names[PIN_SOURCE_INDEX]
    cleaned_groups = {}

    print("")
    print("============================================================")
    print("DUPLICATE PIN CHECK")
    print("============================================================")

    for connector_name, rows in connector_groups.items():
        seen_pins = set()
        cleaned_rows = []

        for row in rows:
            pin_number = str(row.get(pin_field, "")).strip()

            if pin_number in seen_pins:
                print("DUPLICATE SKIPPED:", connector_name, "PIN:", pin_number, "PAGE:", row.get("PAGE"))
                continue

            seen_pins.add(pin_number)
            cleaned_rows.append(row)

        cleaned_groups[connector_name] = cleaned_rows

    print("============================================================")

    return cleaned_groups

def get_excel_application_and_workbook(excel_path):
    excel_path = os.path.abspath(excel_path)
    workbook = None

    try:
        excel = win32.GetActiveObject("Excel.Application")
    except Exception:
        excel = win32.Dispatch("Excel.Application")

    excel.Visible = True

    for open_workbook in excel.Workbooks:
        try:
            if os.path.abspath(str(open_workbook.FullName)).lower() == excel_path.lower():
                workbook = open_workbook
                break
        except Exception:
            continue

    if workbook is None:
        workbook = excel.Workbooks.Open(excel_path)

    return excel, workbook


def get_excel_connector_names(worksheet):
    connector_count_value = worksheet.Range(CONNECTOR_COUNT_CELL).Value

    if connector_count_value is None:
        raise Exception(f"{CONNECTOR_COUNT_CELL} is blank")

    connector_count = int(connector_count_value)

    if connector_count <= 0:
        raise Exception(f"{CONNECTOR_COUNT_CELL} must be greater than zero")

    connector_column_number = excel_column_to_number(EXCEL_CONNECTOR_COLUMN)
    connectors = []

    for index in range(connector_count):
        row_number = EXCEL_CONNECTOR_FIRST_ROW + index
        connector_value = worksheet.Cells(row_number, connector_column_number).Value
        connector_name = normalize_connector_name(connector_value)

        connectors.append({
            "block_index": index,
            "excel_row": row_number,
            "raw_name": "" if connector_value is None else str(connector_value).strip(),
            "normalized_name": connector_name
        })

    return connectors


def clear_excel_block_data(worksheet, block_index):
    block_start_column = BLOCK_FIRST_COLUMN + block_index * BLOCK_WIDTH
    pin_column = block_start_column + PIN_COLUMN_OFFSET
    signal_column = block_start_column + SIGNAL_COLUMN_OFFSET
    note_column = block_start_column + NOTE_COLUMN_OFFSET

    worksheet.Range(worksheet.Cells(DATA_FIRST_ROW, pin_column), worksheet.Cells(DATA_LAST_ROW, note_column)).ClearContents()


def write_connector_to_excel_block(worksheet, block_index, connector_rows, column_names):
    pin_field = column_names[PIN_SOURCE_INDEX]
    signal_field = column_names[SIGNAL_SOURCE_INDEX]
    note_field = column_names[NOTE_SOURCE_INDEX]

    block_start_column = BLOCK_FIRST_COLUMN + block_index * BLOCK_WIDTH
    pin_column = block_start_column + PIN_COLUMN_OFFSET
    signal_column = block_start_column + SIGNAL_COLUMN_OFFSET
    note_column = block_start_column + NOTE_COLUMN_OFFSET

    maximum_rows = DATA_LAST_ROW - DATA_FIRST_ROW + 1

    if len(connector_rows) > maximum_rows:
        raise Exception(f"Connector has {len(connector_rows)} rows but block only supports {maximum_rows}")

    clear_excel_block_data(worksheet, block_index)

    if not connector_rows:
        return

    excel_values = []

    for row in connector_rows:
        pin_value = str(row.get(pin_field, "")).strip()
        signal_value = str(row.get(signal_field, "")).strip()
        note_value = str(row.get(note_field, "")).strip()

        try:
            pin_value = int(pin_value)
        except Exception:
            pass

        excel_values.append((pin_value, signal_value, note_value))

    start_row = DATA_FIRST_ROW
    end_row = DATA_FIRST_ROW + len(excel_values) - 1

    destination = worksheet.Range(worksheet.Cells(start_row, pin_column), worksheet.Cells(end_row, note_column))
    destination.Value = tuple(excel_values)

# WRITE ALL CONNECTORS TO EXCEL
# ================================================
def populate_excel(excel_path, connector_groups, column_names):
    excel, workbook = get_excel_application_and_workbook(excel_path)

    try:
        worksheet = workbook.Worksheets(EXCEL_SHEET_NAME)
    except Exception:
        raise Exception(f"Worksheet '{EXCEL_SHEET_NAME}' was not found")

    excel_connectors = get_excel_connector_names(worksheet)

    print("")
    print("============================================================")
    print("EXCEL CONNECTOR CHECK")
    print("============================================================")
    print("EXCEL CONNECTORS:", len(excel_connectors))
    print("PDF CONNECTORS:", len(connector_groups))
    print("============================================================")

    previous_events = excel.EnableEvents
    previous_screen_updating = excel.ScreenUpdating

    excel.EnableEvents = False
    excel.ScreenUpdating = False

    written_count = 0
    missing_count = 0

    try:
        for connector_data in excel_connectors:
            block_index = connector_data["block_index"]
            excel_name = connector_data["normalized_name"]
            raw_name = connector_data["raw_name"]

            print("")
            print("BLOCK:", block_index + 1, "EXCEL CONNECTOR:", repr(raw_name))

            if excel_name == "":
                print("SKIPPED - CONNECTOR NAME IS BLANK")
                missing_count += 1
                continue

            if excel_name not in connector_groups:
                print("SKIPPED - CONNECTOR NOT FOUND IN PDF:", excel_name)
                missing_count += 1
                continue

            connector_rows = connector_groups[excel_name]

            write_connector_to_excel_block(worksheet, block_index, connector_rows, column_names)

            print("ROWS WRITTEN:", len(connector_rows))

            written_count += 1

        workbook.Save()

    finally:
        excel.EnableEvents = previous_events
        excel.ScreenUpdating = previous_screen_updating

    print("")
    print("============================================================")
    print("EXCEL POPULATION COMPLETE")
    print("============================================================")
    print("CONNECTORS WRITTEN:", written_count)
    print("CONNECTORS NOT MATCHED:", missing_count)
    print("WORKBOOK:", excel_path)
    print("SHEET:", EXCEL_SHEET_NAME)
    print("============================================================")
