import win32com.client as win32
import inspect
import time


#====================================================================================================================================
# SETTINGS
#====================================================================================================================================
SYMBOL_NAME = "Cable Shield"
SYMBOL_VERSION = 1

TEST_Y = 12.0

TEST_1_X = 20.0
TEST_2_X = 24.0
TEST_3_X = 28.0
TEST_4_X = 32.0

DESIRED_LENGTH = 10.0

FLIP = 0


#====================================================================================================================================
# CREATE E3 OBJECTS
#====================================================================================================================================
app = win32.gencache.EnsureDispatch("CT.Application")

prj = app.CreateJobObject()

sht = prj.CreateSheetObject()
sym = prj.CreateSymbolObject()
pin = prj.CreatePinObject()


#====================================================================================================================================
# START
#====================================================================================================================================
app.PutMessage("Starting Cable Shield Method Test")

print("")
print("============================================================")
print("CABLE SHIELD METHOD TEST")
print("============================================================")


#====================================================================================================================================
# STEP 1 - FIND SHEET 1
#====================================================================================================================================
sheet_info = prj.GetAllSheetIds(None)

sheet_ids = [x for x in sheet_info[1] if x is not None]

sheet_1_id = None

for sheet_id in sheet_ids:
    sht.SetId(sheet_id)

    if sht.GetName() == "1":
        sheet_1_id = sheet_id
        break


if sheet_1_id is None:
    raise Exception("Could not find Sheet 1")


print("SHEET 1 ID:", sheet_1_id)


#====================================================================================================================================
# HELPER - LOAD AND PLACE A FRESH SYMBOL
#====================================================================================================================================
def place_test_symbol(x, y):
    load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)

    if load_result == 0:
        raise Exception("Could not load Cable Shield symbol")

    placed_id = sym.Place(sheet_1_id, x, y, FLIP, 1.0, 0)

    if placed_id == 0:
        raise Exception("Could not place Cable Shield symbol")

    sym.SetId(placed_id)

    return placed_id


#====================================================================================================================================
# HELPER - GET NODE LOCATIONS
#====================================================================================================================================
def get_symbol_node_locations(symbol_id):
    sym.SetId(symbol_id)

    pin_info = sym.GetPinIds(None)

    pin_ids = [x for x in pin_info[1] if x is not None]

    output = []

    for pin_id in pin_ids:
        pin.SetId(pin_id)

        xloc = [0.0]
        yloc = [0.0]
        grid = [""]
        column = [""]
        row = [""]

        result = pin.GetSchemaLocation(xloc, yloc, grid, column, row)

        node_x = None
        node_y = None

        try:
            node_x = float(result[1])
            node_y = float(result[2])
        except Exception:
            pass

        output.append(
            {
                "id": pin_id,
                "x": node_x,
                "y": node_y,
                "raw": result
            }
        )

    return output


#====================================================================================================================================
# HELPER - PRINT NODE LOCATIONS
#====================================================================================================================================
def print_symbol_nodes(symbol_id, title):
    print("")
    print("============================================================")
    print(title)
    print("============================================================")

    nodes = get_symbol_node_locations(symbol_id)

    for node in nodes:
        print("NODE ID:", node["id"], "| X:", node["x"], "| Y:", node["y"])

    if len(nodes) >= 2:
        valid_nodes = [node for node in nodes if node["y"] is not None]

        if len(valid_nodes) >= 2:
            valid_nodes.sort(key=lambda item: item["y"])

            bottom_y = valid_nodes[0]["y"]
            top_y = valid_nodes[-1]["y"]

            print("GEOMETRIC VERTICAL LENGTH:", top_y - bottom_y)

    return nodes

# STEP 2 - INSPECT AVAILABLE SYMBOL METHODS
#====================================================================================================================================

print("============================================================")
print("AVAILABLE SYMBOL METHODS")
print("============================================================")

keywords = [
    "loc",
    "position",
    "move",
    "length",
    "display",
    "size",
    "stretch",
    "schema",
    "pin",
    "place",
    "transform"
]

method_names = []

for name in dir(sym):
    lower_name = name.lower()

    if any(keyword in lower_name for keyword in keywords):
        method_names.append(name)


method_names.sort()


for name in method_names:
    try:
        member = getattr(sym, name)
    except Exception as e:
        print(name, "| GET FAILED:", repr(e))
        continue

    if callable(member):
        try:
            signature = inspect.signature(member)
            print(name, signature)
        except Exception:
            print(name, "| CALLABLE")
    else:
        print(name, "| PROPERTY:", repr(member))


print("============================================================")
