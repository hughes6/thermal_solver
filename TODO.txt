# VALUE HELPERS
#====================================================================================================================================
def clean_excel_value(value):
    if value is None:
        return ""

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def normalize_pin_name(value):
    return clean_excel_value(value).upper()


def normalize_connector_token(value):
    text = clean_excel_value(value).upper().replace(" ", "")

    if text == "":
        return ""

    root_prefix = CONNECTOR_ROOT_NAME.upper() + "."

    if text.startswith(root_prefix):
        text = text[len(root_prefix):]

    if "#" in text:
        text = text.split("#", 1)[0]

    match = re.fullmatch(r"P?0*(\d+)([A-Z]*)", text)

    if match:
        number = int(match.group(1))
        suffix = match.group(2)
        return f"P{number}{suffix}"

    return text


def connector_base_name(connector_token):
    connector = normalize_connector_token(connector_token)

    if connector == "":
        return ""

    return f"{CONNECTOR_ROOT_NAME}.{connector}"


def cell_has_bar(value):
    return "|" in clean_excel_value(value)


#====================================================================================================================================
# YES / NO
#====================================================================================================================================
def parse_yes_no(value, setting_name):
    text = clean_excel_value(value).upper()

    if text == "":
        print("WARNING:", setting_name, "IS BLANK - DEFAULTING TO YES")
        return True

    if text in ["YES", "Y", "TRUE", "1"]:
        return True

    if text in ["NO", "N", "FALSE", "0"]:
        return False

    raise Exception(f"{setting_name} must contain YES or NO. Current value: {value}")


def read_twist_offset_settings(ws):

    right_enabled = parse_yes_no(ws[RIGHT_TWIST_OFFSET_CELL].value, "RIGHT TWIST OFFSET R2:S2")

    return {"LEFT": left_enabled, "RIGHT": right_enabled}

def get_last_data_row(ws):
    columns = [LEFT_OUTER_SHIELD_COL, LEFT_INNER_SHIELD_COL, LEFT_TWIST_A_COL, LEFT_TWIST_B_COL, LEFT_CONNECTOR_COL, PIN_1_COL, PIN_2_COL, RIGHT_CONNECTOR_COL, RIGHT_TWIST_A_COL, RIGHT_TWIST_B_COL, RIGHT_INNER_SHIELD_COL, RIGHT_OUTER_SHIELD_COL]

    last_row = DATA_FIRST_ROW - 1

    for row in range(DATA_FIRST_ROW, scan_end + 1):
        for col in columns:
            value = clean_excel_value(ws[f"{col}{row}"].value)

            if value != "":
                last_row = row
                break

    return last_row

#====================================================================================================================================
# SPLIT CONNECTOR SECTIONS
#
# This is used for twists and inner shields ONLY.
#
# Connector changes intentionally split these jobs.
#====================================================================================================================================
def split_into_sections(rows):
    sections = []
    current = []

    for row in rows:
        if not current:
            current = [row]
            continue

        previous = current[-1]
        connector_changed = row["connector"] != previous["connector"]
        row_gap = row["excel_row"] != previous["excel_row"] + 1

        if connector_changed or row_gap:
            sections.append(current)
            current = [row]
        else:
            current.append(row)

    if current:
        sections.append(current)

    return sections


#====================================================================================================================================
# CONTIGUOUS TRUE RANGES WITHIN ONE CONNECTOR SECTION
#====================================================================================================================================
def true_ranges(rows, key):
    ranges = []
    start = None

    for index, row in enumerate(rows):
        marked = bool(row[key])

        if marked and start is None:
            start = index

        if not marked and start is not None:
            ranges.append((start, index - 1))
            start = None

    if start is not None:
        ranges.append((start, len(rows) - 1))

    return ranges


def range_pins(rows, start, end):
    pins = []

    for index in range(start, end + 1):
        pins.append(rows[index]["pin"])

    return pins


def range_excel_rows(rows, start, end):
    excel_rows = []

    for index in range(start, end + 1):
        excel_rows.append(rows[index]["excel_row"])

    return excel_rows


#====================================================================================================================================
# BUILD CONNECTOR-SPECIFIC JOBS
#
# IMPORTANT:
#
# OUTER SHIELDS ARE NOT CREATED HERE.
#
# Twists and inner shields remain connector-specific.
#====================================================================================================================================
def build_section_jobs(rows):
    jobs = []
    warnings = []

    if not rows:
        return jobs, warnings

    side = rows[0]["side"]
    connector = rows[0]["connector"]

    if connector == "":
        warnings.append(f"{side}: rows {rows[0]['excel_row']}-{rows[-1]['excel_row']} have pins but no connector")
        return jobs, warnings

    conflicts = []

    for row in rows:
        if row["twist_a"] and row["twist_b"]:
            conflicts.append(row["excel_row"])

    if conflicts:
        message = f"{side} {connector}: same wire marked in both twist columns on Excel rows {conflicts}"

        if STRICT_TWIST_VALIDATION:
            raise Exception(message)

        warnings.append(message)

    twist_jobs = []

    for key, twist_column in [("twist_a", "A"), ("twist_b", "B")]:
        ranges = true_ranges(rows, key)

        for start, end in ranges:
            pins = range_pins(rows, start, end)
            excel_rows = range_excel_rows(rows, start, end)

            if len(pins) < 2:
                warnings.append(f"{side} {connector}: twist {twist_column} only contains one wire on Excel row {excel_rows[0]}")
                continue

            job = {"type": "twist", "side": side, "connector": connector, "twist_column": twist_column, "start": start, "end": end, "pins": pins, "excel_rows": excel_rows, "symbol_name": TWIST_SYMBOL_NAME}

            twist_jobs.append(job)
            jobs.append(job)

    inner_consumed = set()

    for twist_job in twist_jobs:
        indexes = list(range(twist_job["start"], twist_job["end"] + 1))
        fully_inner_shielded = all(rows[index]["inner_shield"] for index in indexes)

        if fully_inner_shielded:
            shield_job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": twist_job["twist_column"], "pins": list(twist_job["pins"]), "excel_rows": list(twist_job["excel_rows"]), "symbol_name": SHIELD_SYMBOL_NAME}

            jobs.append(shield_job)

            for index in indexes:
                inner_consumed.add(index)

    remaining = []

    for index, row in enumerate(rows):
        remaining.append(bool(row["inner_shield"] and index not in inner_consumed))

    start = None

    for index, marked in enumerate(remaining):
        if marked and start is None:
            start = index

        if not marked and start is not None:
            end = index - 1
            job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME}
            jobs.append(job)
            start = None

    if start is not None:
        end = len(rows) - 1
        job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME}
        jobs.append(job)

    return jobs, warnings


#====================================================================================================================================
# BUILD SIDE-LEVEL OUTER SHIELDS
#
# THIS IS THE IMPORTANT MULTI-CONNECTOR FIX.
#
# Connector changes DO NOT terminate an outer-shield run.
#
# Only:
#
# - outer-shield marker ending
# - an actual Excel row gap
#
# terminates the outer-shield run.
#
# Therefore:
#
# P1
# P3
# P4
#
# can all belong to ONE left-side outer shield.
#====================================================================================================================================
def build_side_outer_jobs(rows):
    jobs = []
    current = []

    for row in rows:
        if not row["outer_shield"]:
            if current:
                jobs.append(make_outer_job(current))
                current = []

            continue

        if not current:
            current = [row]
            continue

        previous = current[-1]
        row_gap = row["excel_row"] != previous["excel_row"] + 1

        if row_gap:
            jobs.append(make_outer_job(current))
            current = [row]
        else:
            current.append(row)

    if current:
        jobs.append(make_outer_job(current))

    return jobs


def make_outer_job(rows):
    side = rows[0]["side"]
    entries = []
    connectors = []
    excel_rows = []

    for row in rows:
        entries.append({"connector": row["connector"], "pin": row["pin"], "excel_row": row["excel_row"]})
        excel_rows.append(row["excel_row"])

        if row["connector"] not in connectors:
            connectors.append(row["connector"])

    return {"type": "shield_outer", "side": side, "connector": None, "connectors": connectors, "twist_column": None, "entries": entries, "excel_rows": excel_rows, "symbol_name": SHIELD_SYMBOL_NAME}


#====================================================================================================================================
# E3 ID EXTRACTOR
#====================================================================================================================================
def extract_e3_ids(result):
    if result is None:
        return []

    if isinstance(result, int):
        if result > 0:
            return [result]

        return []

    if isinstance(result, list):
        output = []

        for value in result:
            if value is not None:
                try:
                    if int(value) > 0:
                        output.append(int(value))
                except Exception:
                    pass

        return output

    if not isinstance(result, tuple):
        return []

    if len(result) == 0:
        return []

    if len(result) >= 2:
        values = result[1]

        if values is None:
            return []

        if isinstance(values, int):
            if values > 0:
                return [values]

            return []

        try:
            return [int(value) for value in values if value is not None and int(value) > 0]
        except Exception:
            pass

    output = []

    for value in result:
        try:
            if int(value) > 0:
                output.append(int(value))
        except Exception:
            pass

    return output

#====================================================================================================================================
# PIN LOCATION
#====================================================================================================================================
def get_pin_location(pin, pin_id):
    pin.SetId(pin_id)

    xloc = [0.0]
    yloc = [0.0]
    grid = [""]
    column = [""]
    row = [""]

    try:
        result = pin.GetSchemaLocation(xloc, yloc, grid, column, row)
    except Exception:
        return None

    if not isinstance(result, tuple):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None

    if sheet_id <= 0:
        return None

    return {"sheet_id": sheet_id, "x": x, "y": y}


#====================================================================================================================================
# LOAD DEVICE PIN MAP
#====================================================================================================================================
def load_device_pin_map(dev, pin, device_id):
    dev.SetId(device_id)

    try:
        info = dev.GetAllPinIds(None)
    except Exception:
        return {}

    pin_ids = extract_e3_ids(info)
    pin_map = {}

    for pin_id in pin_ids:
        pin.SetId(pin_id)

        try:
            pin_name = normalize_pin_name(pin.GetName())
        except Exception:
            continue

        location = get_pin_location(pin, pin_id)

        if pin_name not in pin_map:
            pin_map[pin_name] = []

        if location is not None:
            record = {"pin_id": pin_id, "sheet_id": location["sheet_id"], "x": location["x"], "y": location["y"]}
            pin_map[pin_name].append(record)

    return pin_map

# SAME NAME ASSEMBLY DEVICES
#====================================================================================================================================
def get_same_name_assembly_devices(dev, base_device_id, expected_name):
    found_ids = []

    dev.SetId(base_device_id)

    try:
        assembly_id = dev.GetAssemblyId()
    except Exception:
        return found_ids

    if assembly_id is None:
        return found_ids

    try:
        assembly_id = int(assembly_id)
    except Exception:
        return found_ids

    if assembly_id <= 0:
        return found_ids

    dev.SetId(assembly_id)

    try:
        device_info = dev.GetDeviceIds(None)
    except Exception:
        return found_ids

    device_ids = extract_e3_ids(device_info)

    for device_id in device_ids:
        dev.SetId(device_id)

        try:
            name = clean_excel_value(dev.GetName())
        except Exception:
            continue

        if name.upper() == expected_name.upper():
            found_ids.append(device_id)

    return found_ids


#====================================================================================================================================
# CONNECTOR CANDIDATES
#====================================================================================================================================
def collect_connector_candidate_ids(connector_token, all_devices, dev):
    expected_name = connector_base_name(connector_token)
    base_records = find_base_connector_devices(connector_token, all_devices)

    if not base_records:
        raise Exception(f"Could not find base E3 connector {expected_name}")

    candidate_sources = {}

    for record in base_records:
        base_id = record["device_id"]

        candidate_sources[base_id] = "BASE"

        view_ids = get_device_view_ids(dev, base_id)

        for view_id in view_ids:
            candidate_sources[view_id] = "VIEW"

        child_ids = get_child_device_ids(dev, base_id)

        for child_id in child_ids:
            if child_id not in candidate_sources:
                candidate_sources[child_id] = "CHILD"

        assembly_ids = get_same_name_assembly_devices(dev, base_id, expected_name)

        for assembly_device_id in assembly_ids:
            if assembly_device_id not in candidate_sources:
                candidate_sources[assembly_device_id] = "ASSEMBLY"

    return candidate_sources


#====================================================================================================================================
# ANALYZE CANDIDATE
#====================================================================================================================================
def analyze_candidate(required_pins, pin_map):
    sheets = {}

    for pin_name in required_pins:
        locations = pin_map.get(pin_name, [])

        for location in locations:
            sheet_id = location["sheet_id"]

            if sheet_id not in sheets:
                sheets[sheet_id] = {"pins": set(), "xs": [], "ys": []}

            sheets[sheet_id]["pins"].add(pin_name)
            sheets[sheet_id]["xs"].append(location["x"])
            sheets[sheet_id]["ys"].append(location["y"])

    if not sheets:
        return None

    results = []

    for sheet_id, data in sheets.items():
        placed_count = len(data["pins"])
        unique_x_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["xs"])
        unique_y_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["ys"])
        unique_x_count = len(unique_x_values)
        unique_y_count = len(unique_y_values)
        x_span = max(data["xs"]) - min(data["xs"]) if data["xs"] else 0.0
        y_span = max(data["ys"]) - min(data["ys"]) if data["ys"] else 0.0
        has_real_y_spread = y_span >= SCHEMATIC_MIN_Y_SPAN and unique_y_count > 1
        result = {"sheet_id": sheet_id, "placed_count": placed_count, "unique_x_count": unique_x_count, "unique_y_count": unique_y_count, "x_span": x_span, "y_span": y_span, "has_real_y_spread": has_real_y_spread}

        results.append(result)

    results.sort(key=lambda item: (item["has_real_y_spread"], item["placed_count"], item["unique_y_count"], item["y_span"]), reverse=True)

    return results[0]


#====================================================================================================================================
# RESOLVE CONNECTOR SCHEMATIC VIEW
#====================================================================================================================================
def resolve_connector_schematic_view(connector_token, required_pins, all_devices, dev, pin, pin_map_cache):
    expected_name = connector_base_name(connector_token)
    candidate_sources = collect_connector_candidate_ids(connector_token, all_devices, dev)
    candidates = []

    print("")
    print("============================================================")
    print("E3 CANDIDATES FOR:", expected_name)
    print("============================================================")

    for candidate_id, source in candidate_sources.items():
        dev.SetId(candidate_id)

        try:
            candidate_name = clean_excel_value(dev.GetName())
        except Exception:
            candidate_name = "UNKNOWN"

        if candidate_id not in pin_map_cache:
            try:
                pin_map_cache[candidate_id] = load_device_pin_map(dev, pin, candidate_id)
            except Exception as error:
                print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "PIN MAP FAILED:", repr(error))
                continue

        pin_map = pin_map_cache[candidate_id]
        analysis = analyze_candidate(required_pins, pin_map)

        if analysis is None:
            print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "NO PLACED REQUIRED PINS")
            continue

        print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "SHEET:", analysis["sheet_id"], "PLACED:", analysis["placed_count"], "UNIQUE X:", analysis["unique_x_count"], "UNIQUE Y:", analysis["unique_y_count"], "X SPAN:", analysis["x_span"], "Y SPAN:", analysis["y_span"])

        candidate = {"device_id": candidate_id, "name": candidate_name, "source": source, "pin_map": pin_map, "analysis": analysis}

        candidates.append(candidate)

    if not candidates:
        raise Exception(f"No usable E3 device/view candidates found for {expected_name}")

    candidates.sort(key=lambda item: (item["analysis"]["has_real_y_spread"], item["analysis"]["placed_count"], item["analysis"]["unique_y_count"], item["analysis"]["y_span"]), reverse=True)

    best = candidates[0]

    if not best["analysis"]["has_real_y_spread"]:
        raise Exception(f"Could not locate the split schematic view for {expected_name}")

    sheet_id = best["analysis"]["sheet_id"]
    missing_pins = []

    for pin_name in required_pins:
        locations = best["pin_map"].get(pin_name, [])
        found_on_sheet = False

        for location in locations:
            if location["sheet_id"] == sheet_id:
                found_on_sheet = True
                break

        if not found_on_sheet:
            missing_pins.append(pin_name)

    if missing_pins:
        raise Exception(f"Selected schematic view for {expected_name} is missing pins on sheet {sheet_id}: {missing_pins}")

    print("")
    print("SELECTED SCHEMATIC VIEW ID:", best["device_id"])
    print("SELECTED SOURCE:", best["source"])
    print("SELECTED NAME:", best["name"])
    print("SELECTED SHEET:", sheet_id)
    print("SELECTED PLACED PIN COUNT:", best["analysis"]["placed_count"])
    print("SELECTED UNIQUE Y COUNT:", best["analysis"]["unique_y_count"])
    print("SELECTED Y SPAN:", best["analysis"]["y_span"])
    print("============================================================")

    return {"device_id": best["device_id"], "name": best["name"], "pin_map": best["pin_map"], "sheet_id": sheet_id}

#====================================================================================================================================
# JOB OFFSET
#====================================================================================================================================
def job_offset(job, twist_offset_settings):
    offset_enabled = twist_offset_settings[job["side"]]

    if job["type"] == "twist":
        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if job["type"] == "shield_inner":
        if job["twist_column"] is None:
            return INNER_SHIELD_ONLY_OFFSET

        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if not offset_enabled:
        return OFFSET_NO_OUTER_SHIELD

    return OFFSET_YES_OUTER_SHIELD

def connector_job_priority(job):
    if job["type"] == "twist":
        return 0

    return 1


#====================================================================================================================================
# CALCULATE CONNECTOR JOB PLACEMENT
#====================================================================================================================================
def calculate_connector_placement(job, pin_map, sheet_id, twist_offset_settings):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} has no schematic location on sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    if top_y - bottom_y < SCHEMATIC_MIN_Y_SPAN and len(job["pins"]) > 1:
        raise Exception(f"{job['side']} {job['connector']} {job['type']}: pins {job['pins']} all have effectively the same Y coordinate")

    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job, twist_offset_settings)
    x = anchor_x + direction * offset

    if job["type"] == "twist":
        y = bottom_y - TWIST_BOTTOM_OVERLAP
        length = None
    else:
        y = bottom_y - SHIELD_BOTTOM_OVERLAP
        length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

        if length < MIN_SYMBOL_LENGTH:
            length = MIN_SYMBOL_LENGTH

    return {"sheet_id": sheet_id, "x": x, "y": y, "length": length, "anchor_x": anchor_x, "bottom_y": bottom_y, "top_y": top_y, "offset_enabled": twist_offset_settings[job["side"]]}


#====================================================================================================================================
# RESOLVE ALL CONNECTORS USED BY AN OUTER SHIELD
#====================================================================================================================================
def resolve_outer_contexts(job, all_devices, dev, pin, pin_map_cache, connector_context_cache):
    connector_pin_requirements = {}

    for entry in job["entries"]:
        connector = entry["connector"]
        pin_name = entry["pin"]

        if connector == "":
            raise Exception(f"{job['side']} outer shield contains a row with no connector")

        if connector not in connector_pin_requirements:
            connector_pin_requirements[connector] = []

        if pin_name not in connector_pin_requirements[connector]:
            connector_pin_requirements[connector].append(pin_name)

    contexts = {}

    for connector, required_pins in connector_pin_requirements.items():
        contexts[connector] = get_connector_context(connector, required_pins, all_devices, dev, pin, pin_map_cache, connector_context_cache)

    return contexts


#====================================================================================================================================
# CALCULATE MULTI-CONNECTOR OUTER SHIELD PLACEMENT
#====================================================================================================================================
def calculate_outer_placement(job, contexts, twist_offset_settings):
    locations = []
    sheet_ids = set()

    for entry in job["entries"]:
        connector = entry["connector"]
        pin_name = entry["pin"]
        context = contexts[connector]
        location = pin_location_on_sheet(context["pin_map"], pin_name, context["sheet_id"])

        if location is None:
            raise Exception(f"{job['side']} outer shield could not locate {connector} pin {pin_name}")

        locations.append({"connector": connector, "pin": pin_name, "x": location["x"], "y": location["y"], "sheet_id": location["sheet_id"]})
        sheet_ids.add(location["sheet_id"])

    if len(sheet_ids) != 1:
        raise Exception(f"{job['side']} outer shield crosses multiple E3 sheets: {sorted(sheet_ids)}")

    sheet_id = list(sheet_ids)[0]
    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]
    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)
    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job, twist_offset_settings)
    x = anchor_x + direction * offset
    y = bottom_y - SHIELD_BOTTOM_OVERLAP
    length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

    if length < MIN_SYMBOL_LENGTH:
        length = MIN_SYMBOL_LENGTH

    return {"sheet_id": sheet_id, "x": x, "y": y, "length": length, "anchor_x": anchor_x, "bottom_y": bottom_y, "top_y": top_y, "offset_enabled": twist_offset_settings[job["side"]], "locations": locations}

#====================================================================================================================================
# INNER SHIELD STAGING CANDIDATES
#====================================================================================================================================
def build_inner_stage_candidates(final_y):
    candidates = []

    x = INNER_STAGE_X_MIN

    while x <= INNER_STAGE_X_MAX + 0.000001:
        candidates.append((round(x, 6), round(final_y, 6)))
        x += INNER_STAGE_X_STEP

    for y in INNER_STAGE_Y_VALUES:
        x = INNER_STAGE_X_MIN

        while x <= INNER_STAGE_X_MAX + 0.000001:
            candidate = (round(x, 6), round(y, 6))

            if candidate not in candidates:
                candidates.append(candidate)

            x += INNER_STAGE_X_STEP

    return candidates[:INNER_STAGE_MAX_ATTEMPTS]


#====================================================================================================================================
# PLACE INNER SHIELD
#
# Existing stage -> resize -> move behavior retained.
#
# Search is capped to avoid thousands of attempts.
#====================================================================================================================================
def place_inner_shield(sym, job, placement):
    requested_length = float(placement["length"])
    final_x = float(placement["x"])
    final_y = float(placement["y"])
    candidates = build_inner_stage_candidates(final_y)

    print("INNER SHIELD FINAL X:", final_x)
    print("INNER SHIELD FINAL Y:", final_y)
    print("INNER SHIELD REQUESTED LENGTH:", requested_length)
    print("INNER SHIELD MAX STAGE ATTEMPTS:", len(candidates))

    for attempt, candidate in enumerate(candidates, start=1):
        stage_x = candidate[0]
        stage_y = candidate[1]

        print("")
        print("INNER SHIELD STAGE ATTEMPT:", attempt, "OF", len(candidates))
        print("INNER SHIELD STAGE X:", stage_x)
        print("INNER SHIELD STAGE Y:", stage_y)

        load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

        if load_result == 0:
            raise Exception(f"Could not load library shield {job['symbol_name']}")

        staged_id = sym.Place(placement["sheet_id"], stage_x, stage_y, 0, 1.0, 0)

        print("INNER SHIELD STAGE PLACE RESULT:", staged_id)

        if staged_id == 0:
            continue

        time.sleep(PLACEMENT_DELAY_SECONDS)

        before = get_display_length_safe(sym, staged_id)

        print("INNER SHIELD LENGTH BEFORE:", before)

        if before is not None and abs(before - requested_length) <= 0.001:
            resize_success = True
        else:
            sym.SetId(staged_id)

            try:
                resize_result = sym.SetDisplayLength(requested_length)
            except Exception as error:
                resize_result = None
                print("INNER SHIELD RESIZE EXCEPTION:", repr(error))

            print("INNER SHIELD RESIZE RESULT:", resize_result)

            time.sleep(STRETCH_DELAY_SECONDS)

            after = get_display_length_safe(sym, staged_id)

            print("INNER SHIELD LENGTH AFTER:", after)

            resize_success = after is not None and abs(after - requested_length) <= 0.001

        if not resize_success:
            delete_script_symbol(sym, staged_id)
            continue

        sym.SetId(staged_id)

        move_result = sym.Place(placement["sheet_id"], final_x, final_y, 0, 1.0, 0)

        print("INNER SHIELD FINAL MOVE RESULT:", move_result)

        if move_result == 0:
            delete_script_symbol(sym, staged_id)
            continue

        final_id = move_result

        time.sleep(MOVE_DELAY_SECONDS)

        final_length = get_display_length_safe(sym, final_id)

        print("INNER SHIELD FINAL LENGTH:", final_length)

        if final_length is not None and abs(final_length - requested_length) <= 0.001:
            print("INNER SHIELD PLACED SUCCESSFULLY")
            return final_id

        delete_script_symbol(sym, final_id)

    raise Exception(f"Could not place inner shield at requested length {requested_length}")


#====================================================================================================================================
# TRY TO PRE-SIZE LOADED OUTER SHIELD
#
# THIS HAPPENS BEFORE Place().
#====================================================================================================================================
def try_presize_loaded_outer_shield(sym, requested_length):
    print("")
    print("OUTER SHIELD PRE-SIZE TEST")

    before = get_display_length_safe(sym)

    print("LOADED SHIELD LENGTH BEFORE:", before)

    try:
        result = sym.SetDisplayLength(requested_length)
    except Exception as error:
        result = None
        print("PRE-SIZE EXCEPTION:", repr(error))

    print("PRE-SIZE SetDisplayLength RESULT:", result)

    time.sleep(STRETCH_DELAY_SECONDS)

    after = get_display_length_safe(sym)

    print("LOADED SHIELD REQUESTED LENGTH:", requested_length)
    print("LOADED SHIELD LENGTH AFTER:", after)

    if after is not None and abs(after - requested_length) <= 0.001:
        print("OUTER SHIELD PRE-SIZE SUCCESSFUL")
        return True

    print("OUTER SHIELD PRE-SIZE NOT ACCEPTED")

    return False


#====================================================================================================================================
# RESIZE ALREADY-PLACED OUTER SHIELD
#====================================================================================================================================
def resize_placed_outer_shield(sym, placed_id, requested_length):
    for attempt in range(1, OUTER_DIRECT_STRETCH_ATTEMPTS + 1):
        sym.SetId(placed_id)

        before = get_display_length_safe(sym, placed_id)

        print("")
        print("OUTER DIRECT RESIZE ATTEMPT:", attempt, "OF", OUTER_DIRECT_STRETCH_ATTEMPTS)
        print("OUTER DIRECT LENGTH BEFORE:", before)

        if before is not None and abs(before - requested_length) <= 0.001:
            return True

        try:
            result = sym.SetDisplayLength(requested_length)
        except Exception as error:
            result = None
            print("OUTER DIRECT RESIZE EXCEPTION:", repr(error))

        print("OUTER DIRECT SetDisplayLength RESULT:", result)

        time.sleep(STRETCH_DELAY_SECONDS)

        after = get_display_length_safe(sym, placed_id)

        print("OUTER DIRECT REQUESTED LENGTH:", requested_length)
        print("OUTER DIRECT LENGTH AFTER:", after)

        if after is not None and abs(after - requested_length) <= 0.001:
            return True

        time.sleep(PLACEMENT_DELAY_SECONDS)

    return False


#====================================================================================================================================
# PLACE OUTER SHIELD
#
# ORDER:
#
# 1. Load Shield_Demo.
# 2. Try SetDisplayLength BEFORE placement.
# 3. If accepted, place pre-sized shield directly at final position.
# 4. Verify final size.
# 5. If pre-sizing fails, reload a clean Shield_Demo.
# 6. Place directly on real conductors.
# 7. Retry SetDisplayLength there.
#
# NO LARGE STAGING SEARCH.
#====================================================================================================================================
def place_outer_shield(sym, job, placement):
    requested_length = float(placement["length"])
    final_x = float(placement["x"])
    final_y = float(placement["y"])

    print("")
    print("============================================================")
    print("OUTER SHIELD PLACEMENT")
    print("============================================================")
    print("SIDE:", job["side"])
    print("CONNECTORS COVERED:", job["connectors"])
    print("EXCEL ROWS:", job["excel_rows"][0], "TO", job["excel_rows"][-1])
    print("FINAL X:", final_x)
    print("FINAL Y:", final_y)
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("REQUESTED LENGTH:", requested_length)

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("OUTER SHIELD LOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not load library outer shield {job['symbol_name']}")

    presized = try_presize_loaded_outer_shield(sym, requested_length)

    if presized:
        placed_id = sym.Place(placement["sheet_id"], final_x, final_y, 0, 1.0, 0)

        print("PRE-SIZED OUTER SHIELD PLACE RESULT:", placed_id)

        if placed_id > 0:
            time.sleep(PLACEMENT_DELAY_SECONDS)

            final_length = get_display_length_safe(sym, placed_id)

            print("PRE-SIZED OUTER SHIELD FINAL LENGTH:", final_length)

            if final_length is not None and abs(final_length - requested_length) <= 0.001:
                print("PRE-SIZED OUTER SHIELD PLACED SUCCESSFULLY")
                return placed_id

            print("PRE-SIZED SHIELD LOST REQUESTED LENGTH AFTER PLACEMENT")

            if resize_placed_outer_shield(sym, placed_id, requested_length):
                print("PRE-SIZED OUTER SHIELD CORRECTED AFTER PLACEMENT")
                return placed_id

            delete_script_symbol(sym, placed_id)

    print("")
    print("FALLING BACK TO DIRECT OUTER SHIELD PLACEMENT")

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("OUTER SHIELD RELOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not reload library outer shield {job['symbol_name']}")

    placed_id = sym.Place(placement["sheet_id"], final_x, final_y, 0, 1.0, 0)

    print("DIRECT OUTER SHIELD PLACE RESULT:", placed_id)

    if placed_id == 0:
        raise Exception(f"Could not place outer shield directly at X={final_x} Y={final_y}")

    time.sleep(PLACEMENT_DELAY_SECONDS)

    sym.SetId(placed_id)

    try:
        is_shield = sym.IsShield()
    except Exception:
        is_shield = None

    try:
        is_connected = sym.IsConnected()
    except Exception:
        is_connected = None

    initial_length = get_display_length_safe(sym, placed_id)

    print("DIRECT OUTER SHIELD ID:", placed_id)
    print("DIRECT OUTER SHIELD IsShield:", is_shield)
    print("DIRECT OUTER SHIELD IsConnected:", is_connected)
    print("DIRECT OUTER SHIELD INITIAL LENGTH:", initial_length)

    if resize_placed_outer_shield(sym, placed_id, requested_length):
        final_length = get_display_length_safe(sym, placed_id)

        print("DIRECT OUTER SHIELD FINAL LENGTH:", final_length)
        print("DIRECT OUTER SHIELD PLACED SUCCESSFULLY")

        return placed_id

    delete_script_symbol(sym, placed_id)

    raise Exception(f"Outer shield covering {job['connectors']} could not resize to {requested_length}")

# PROCESS ONE CONNECTOR PACKAGE
#====================================================================================================================================
def process_connector_package(package, sym, all_devices, dev, pin, pin_map_cache, connector_context_cache, twist_offset_settings, placed_ids, app):
    section_rows = package["rows"]
    jobs = package["jobs"]
    side = section_rows[0]["side"]
    connector = section_rows[0]["connector"]
    required_pins = list(dict.fromkeys(row["pin"] for row in section_rows))

    print("")
    print("============================================================")
    print("PROCESSING CONNECTOR SECTION")
    print("============================================================")
    print("SIDE:", side)
    print("CONNECTOR:", connector)
    print("EXPECTED E3 CONNECTOR:", connector_base_name(connector))
    print("EXCEL ROWS:", section_rows[0]["excel_row"], "TO", section_rows[-1]["excel_row"])
    print("REQUIRED PINS:", required_pins)

    context = get_connector_context(connector, required_pins, all_devices, dev, pin, pin_map_cache, connector_context_cache)

    print_section_pin_locations(required_pins, context["pin_map"], context["sheet_id"])

    jobs = sorted(jobs, key=connector_job_priority)

    for job in jobs:
        try:
            placement = calculate_connector_placement(job, context["pin_map"], context["sheet_id"], twist_offset_settings)
            placed_id = place_connector_job(sym, job, placement)

            if placed_id is not None:
                placed_ids.append(placed_id)

        except Exception as error:
            print("ERROR:", str(error))
            app.PutMessage(f"Cable overlay warning: {error}")

        time.sleep(PLACEMENT_DELAY_SECONDS)


#====================================================================================================================================
# PROCESS ONE OUTER SHIELD JOB
#====================================================================================================================================
def process_outer_job(job, sym, all_devices, dev, pin, pin_map_cache, connector_context_cache, twist_offset_settings, placed_ids, app):
    try:
        contexts = resolve_outer_contexts(job, all_devices, dev, pin, pin_map_cache, connector_context_cache)
        placement = calculate_outer_placement(job, contexts, twist_offset_settings)
        placed_id = place_outer_job(sym, job, placement)

        if placed_id is not None:
            placed_ids.append(placed_id)

    except Exception as error:
        print("ERROR:", str(error))
        app.PutMessage(f"Cable overlay warning: {error}")

    time.sleep(PLACEMENT_DELAY_SECONDS)

    print("DATA ROWS:", DATA_FIRST_ROW, "TO", last_row)

    left_rows = read_side_rows(ws, "LEFT", last_row)
    right_rows = read_side_rows(ws, "RIGHT", last_row)

    side_rows = {"LEFT": left_rows, "RIGHT": right_rows}
    packages_by_side = {"LEFT": [], "RIGHT": []}
    outer_jobs_by_side = {"LEFT": build_side_outer_jobs(left_rows), "RIGHT": build_side_outer_jobs(right_rows)}
    warnings = []

    for side in ["LEFT", "RIGHT"]:
        sections = split_into_sections(side_rows[side])

        for section_rows in sections:
            jobs, section_warnings = build_section_jobs(section_rows)

            if jobs:
                packages_by_side[side].append({"rows": section_rows, "jobs": jobs})

            warnings.extend(section_warnings)

    print("")
    print("LEFT CONNECTOR JOB SECTIONS:", len(packages_by_side["LEFT"]))
    print("RIGHT CONNECTOR JOB SECTIONS:", len(packages_by_side["RIGHT"]))
    print("LEFT OUTER SHIELD JOBS:", len(outer_jobs_by_side["LEFT"]))
    print("RIGHT OUTER SHIELD JOBS:", len(outer_jobs_by_side["RIGHT"]))

    for side in ["LEFT", "RIGHT"]:
        for index, job in enumerate(outer_jobs_by_side[side], start=1):
            print(side, "OUTER JOB", index, "CONNECTORS:", job["connectors"], "ROWS:", job["excel_rows"][0], "TO", job["excel_rows"][-1])

    if warnings:
        print("")
        print("============================================================")
        print("EXCEL WARNINGS")
        print("============================================================")

        for warning in warnings:
            print("WARNING:", warning)

    app = win32.gencache.EnsureDispatch("CT.Application")
    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()
    sym = prj.CreateSymbolObject()

    app.PutMessage("Starting Cable Shield / Twist Overlay Script")

    all_devices = get_all_devices(prj, dev)

    print("")
    print("E3 BASE DEVICE ID COUNT:", len(all_devices))

    pin_map_cache = {}
    connector_context_cache = {}
    placed_ids = []

    for side in ["LEFT", "RIGHT"]:
        packages = packages_by_side[side]
        outer_jobs = outer_jobs_by_side[side]
        all_connector_jobs = []

        for package in packages:
            all_connector_jobs.extend(package["jobs"])

        has_twists = any(job["type"] == "twist" for job in all_connector_jobs)
        has_inner_shields = any(job["type"] == "shield_inner" for job in all_connector_jobs)
        has_outer_shields = len(outer_jobs) > 0
        outer_first = has_twists and has_outer_shields and not has_inner_shields

        print("")
        print("============================================================")
        print("SIDE PROCESSING PLAN:", side)
        print("============================================================")
        print("HAS TWISTS:", has_twists)
        print("HAS INNER SHIELDS:", has_inner_shields)
        print("HAS OUTER SHIELDS:", has_outer_shields)
        print("OUTER SHIELD FIRST:", outer_first)

        if outer_first:
            for outer_job in outer_jobs:
                process_outer_job(outer_job, sym, all_devices, dev, pin, pin_map_cache, connector_context_cache, twist_offset_settings, placed_ids, app)

        for package in packages:
            process_connector_package(package, sym, all_devices, dev, pin, pin_map_cache, connector_context_cache, twist_offset_settings, placed_ids, app)

        if not outer_first:
            for outer_job in outer_jobs:
                process_outer_job(outer_job, sym, all_devices, dev, pin, pin_map_cache, connector_context_cache, twist_offset_settings, placed_ids, app)

    print("")
    print("============================================================")
    print("OVERLAY COMPLETE")
    print("============================================================")
    print("PLACED SYMBOL COUNT:", len(placed_ids))
    print("PLACED SYMBOL IDS:", placed_ids)
    print("============================================================")

    app.PutMessage(f"Cable Shield / Twist Overlay complete. Placed {len(placed_ids)} symbols.")

    workbook.close()

    del sym
    del pin
    del dev
    del prj
    del app


if __name__ == "__main__":
    main()
