import win32com.client as win32


#====================================================================================================================================
# SETTINGS
#====================================================================================================================================
SYMBOL_NAME = "Shield_Demo"
SYMBOL_VERSION = 1

PLACE_X = 20.0
PLACE_Y = 20.0

TEST_LENGTH = 1.0


#====================================================================================================================================
# CONNECT TO E3
#====================================================================================================================================
app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
sht = prj.CreateSheetObject()
sym = prj.CreateSymbolObject()


#====================================================================================================================================
# FIND SHEET 1
#====================================================================================================================================
sheet_info = prj.GetAllSheetIds(None)
sheet_ids = [value for value in sheet_info[1] if value is not None]
sheet_1_id = None

for sheet_id in sheet_ids:
    sht.SetId(sheet_id)

    if sht.GetName() == "1":
        sheet_1_id = sheet_id
        break

if sheet_1_id is None:
    raise Exception("Could not find sheet 1")


#====================================================================================================================================
# LOAD SYMBOL
#====================================================================================================================================
load_result = sym.Load(SYMBOL_NAME, SYMBOL_VERSION)

print("")
print("============================================================")
print("SHIELD GEOMETRY TEST")
print("============================================================")
print("LOAD RESULT:", load_result)

if load_result == 0:
    raise Exception("Could not load Shield_Demo")


#====================================================================================================================================
# PLACE SYMBOL IN EMPTY AREA
#====================================================================================================================================
placed_id = sym.Place(sheet_1_id, PLACE_X, PLACE_Y, 0, 1.0, 0)

print("PLACE RESULT:", placed_id)

if placed_id == 0:
    raise Exception("Could not place Shield_Demo")


sym.SetId(placed_id)


#====================================================================================================================================
# READ BASIC INFORMATION
#====================================================================================================================================
print("PLACED ID:", placed_id)
print("SYMBOL TYPE:", sym.GetSymbolType())
print("IS SHIELD:", sym.IsShield())
print("DISPLAY LENGTH BEFORE:", sym.GetDisplayLength())


#====================================================================================================================================
# GET AREA BEFORE STRETCH
#====================================================================================================================================
try:
    area_before = sym.GetPlacedArea(None, None, None, None)
    print("")
    print("PLACED AREA BEFORE:", area_before)
except Exception as error:
    print("GET PLACED AREA BEFORE FAILED:", repr(error))


#====================================================================================================================================
# GET POLYGON BEFORE STRETCH
#====================================================================================================================================
try:
    polygon_before = sym.GetPlacedPolygon(None, None, None)
    print("PLACED POLYGON BEFORE:", polygon_before)
except Exception as error:
    print("GET PLACED POLYGON BEFORE FAILED:", repr(error))


#====================================================================================================================================
# STRETCH
#====================================================================================================================================
stretch_result = sym.SetDisplayLength(TEST_LENGTH)

print("")
print("SET DISPLAY LENGTH RESULT:", stretch_result)


sym.SetId(placed_id)

print("DISPLAY LENGTH AFTER:", sym.GetDisplayLength())


#====================================================================================================================================
# GET AREA AFTER STRETCH
#====================================================================================================================================
try:
    area_after = sym.GetPlacedArea(None, None, None, None)
    print("")
    print("PLACED AREA AFTER:", area_after)
except Exception as error:
    print("GET PLACED AREA AFTER FAILED:", repr(error))


#====================================================================================================================================
# GET POLYGON AFTER STRETCH
#====================================================================================================================================
try:
    polygon_after = sym.GetPlacedPolygon(None, None, None)
    print("PLACED POLYGON AFTER:", polygon_after)
except Exception as error:
    print("GET PLACED POLYGON AFTER FAILED:", repr(error))


#====================================================================================================================================
# ORIGIN COMPARISON
#====================================================================================================================================
print("")
print("============================================================")
print("INSERTION POINT")
print("============================================================")
print("INSERTION X:", PLACE_X)
print("INSERTION Y:", PLACE_Y)
print("============================================================")


#====================================================================================================================================
# CLEANUP
#====================================================================================================================================
del sym
del sht
del prj
del app
