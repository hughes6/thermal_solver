def looks_like_wiring_row(line):
    line = line.strip()

    if line == "":
        return False

    match = re.match(r"^[A-Za-z][A-Za-z0-9_.#/-]*\s+\d+\s+[A-Za-z0-9_.#/-]+(?:\s+.*)?$", line)

    return match is not None


def is_pdf_junk_line(line):
    text = line.strip()
    upper_text = text.upper()

    if text == "":
        return True

    if re.fullmatch(r"PAGE\s+\d+\s+OF\s+\d+", upper_text):
        return True

    if re.fullmatch(r"\d+\s+OF\s+\d+", upper_text):
        return True

    if upper_text == "WIRE CODES":
        return True

    if upper_text == "FROM":
        return True

    if upper_text == "TO":
        return True

    if upper_text == "SIGNAL NAME":
        return True

    if upper_text == "MNEMONIC":
        return True

    if upper_text == "DESCRIPTION":
        return True

    if upper_text == "WIRE":
        return True

    if upper_text == "CODE":
        return True

    if upper_text == "WIRE CODE":
        return True

    if upper_text == "NOTE":
        return True

    if upper_text == "COMMENT":
        return True

    if "DES" in upper_text and "POSN" in upper_text and "MNEMONIC" in upper_text:
        return True

    if upper_text.startswith("DES POSN DES POSN"):
        return True

    if upper_text.startswith("CODE NOTE COMMENT"):
        return True

    return False


def looks_like_flat_pdf_wiring_table(input_text):
    row_count = 0

    for line in input_text.split("\n"):
        if looks_like_wiring_row(line):
            row_count += 1

    manual_columns = get_manual_column_count()

    if manual_columns == 9 and row_count >= 1:
        return True

    if row_count >= 2:
        return True

    return False


def parse_wiring_record(record_lines):
    first_line = record_lines[0].strip()

    match = re.match(
        r"^([A-Za-z][A-Za-z0-9_.#/-]*)\s+(\d+)\s+([A-Za-z0-9_.#/-]+)(?:\s+(.*))?$",
        first_line
    )

    if match is None:
        return None

    from_des = match.group(1)
    from_posn = match.group(2)
    to_des = match.group(3)

    first_remainder = match.group(4)

    remaining_parts = []

    if first_remainder is not None:
        if first_remainder.strip() != "":
            remaining_parts.append(first_remainder.strip())

    for line in record_lines[1:]:
        line = line.strip()

        if line != "":
            remaining_parts.append(line)

    combined_remaining = " ".join(remaining_parts).strip()

    to_posn = ""
    mnemonic = ""
    description = ""
    wire_code = ""
    note = ""
    comment = ""

    if combined_remaining == "":
        return [
            from_des,
            from_posn,
            to_des,
            to_posn,
            mnemonic,
            description,
            wire_code,
            note,
            comment
        ]

    tokens = combined_remaining.split()

    # =========================================
    # REMOVE COMMENT VALUES
    # =========================================

    comment_values = {
        "SPARE",
        "UNUSED",
        "NC",
        "N/C"
    }

    cleaned_tokens = []

    for token in tokens:
        if token.upper() in comment_values:
            if comment == "":
                comment = token
        else:
            cleaned_tokens.append(token)

    tokens = cleaned_tokens

    # =========================================
    # REMOVE NOTE / IN-OUT VALUES
    # =========================================

    note_values = {
        "IN",
        "OUT",
        "I/O",
        "INPUT",
        "OUTPUT"
    }

    cleaned_tokens = []

    for token in tokens:
        if token.upper() in note_values:
            if note == "":
                note = token
        else:
            cleaned_tokens.append(token)

    tokens = cleaned_tokens

    # =========================================
    # OPTIONAL TO-POSITION
    #
    # If the first remaining token is numeric
    # and there are other signal tokens after it,
    # treat it as TO POSN.
    # =========================================

    if len(tokens) >= 2:
        if re.fullmatch(r"\d+", tokens[0]):
            to_posn = tokens.pop(0)

    # =========================================
    # WIRE CODE
    #
    # Usually a standalone numeric value near
    # the end, for example 24.
    # =========================================

    if len(tokens) >= 2:

        for index in range(len(tokens) - 1, 0, -1):

            if re.fullmatch(r"\d+", tokens[index]):
                wire_code = tokens[index]
                tokens.pop(index)
                break

    # =========================================
    # SIGNAL DATA
    #
    # First remaining token = mnemonic.
    # Everything after = description.
    # =========================================

    if len(tokens) >= 1:
        mnemonic = tokens[0]

    if len(tokens) >= 2:
        description = " ".join(tokens[1:])

    return [
        from_des,
        from_posn,
        to_des,
        to_posn,
        mnemonic,
        description,
        wire_code,
        note,
        comment
    ]


def parse_flat_pdf_wiring_table(input_text):
    raw_lines = input_text.split("\n")

    records = []
    current_record = []

    for raw_line in raw_lines:

        line = raw_line.strip()

        if is_pdf_junk_line(line):
            continue

        if looks_like_wiring_row(line):

            if len(current_record) > 0:
                records.append(current_record)

            current_record = [line]

        else:

            if len(current_record) > 0:
                current_record.append(line)

    if len(current_record) > 0:
        records.append(current_record)

    rows = []

    rows.append([
        "FROM DES",
        "FROM POSN",
        "TO DES",
        "TO POSN",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE CODE",
        "NOTE",
        "COMMENT"
    ])

    for record in records:

        parsed_row = parse_wiring_record(record)

        if parsed_row is not None:
            rows.append(parsed_row)

    return rows


def parse_table():
    input_text = input_box.get("1.0", "end-1c")

    if input_text == "":
        return [], 0, "NONE"

    input_text = input_text.replace("\r\n", "\n")
    input_text = input_text.replace("\r", "\n")

    manual_columns = get_manual_column_count()

    rows = []
    delimiter_name = ""

    # =========================================
    # REAL EXCEL TABLE
    # =========================================

    if "\t" in input_text:
        delimiter_name = "TAB"

        for line in input_text.split("\n"):
            rows.append(line.split("\t"))

    # =========================================
    # CSV
    # =========================================

    elif "," in input_text:
        delimiter_name = "COMMA"

        reader = csv.reader(io.StringIO(input_text))

        for row in reader:
            rows.append(row)

    # =========================================
    # PDF WIRING TABLE
    #
    # DOES NOT REQUIRE HEADERS.
    #
    # It detects rows such as:
    #
    # J3 1 101 ABI000 ABI CH 0 IN
    #
    # and handles continuation lines.
    # =========================================

    elif looks_like_flat_pdf_wiring_table(input_text):

        rows = parse_flat_pdf_wiring_table(input_text)

        delimiter_name = "PDF WIRING TABLE"

        return rows, 9, delimiter_name

    # =========================================
    # GENERIC SPACE-ALIGNED TABLE
    # =========================================

    elif manual_columns is not None and manual_columns > 1:

        delimiter_name = "SPACED TABLE"

        for line in input_text.split("\n"):

            if line.strip() == "":
                continue

            row = re.split(r" {2,}", line.strip())

            rows.append(row)

    # =========================================
    # SINGLE COLUMN
    # =========================================

    else:

        delimiter_name = "LINES"

        for line in input_text.split("\n"):
            rows.append([line])

    # =========================================
    # COLUMN COUNT
    # =========================================

    if manual_columns is not None:
        column_count = manual_columns
    else:
        column_count = 0

        for row in rows:

            if len(row) > column_count:
                column_count = len(row)

    # =========================================
    # NORMALIZE
    # =========================================

    normalized_rows = []

    for row in rows:

        normalized_row = list(row)

        while len(normalized_row) < column_count:
            normalized_row.append("")

        if len(normalized_row) > column_count:
            normalized_row = normalized_row[:column_count]

        normalized_rows.append(normalized_row)

    return normalized_rows, column_count, delimiter_name
