def ask_pdf_path():
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    pdf_path = filedialog.askopenfilename(
        filetypes=[
            ("PDF Files", "*.pdf"),
            ("All Files", "*.*")
        ]
    )

    root.destroy()

    if not pdf_path:
        return None

    return pdf_path


def ask_extraction_settings():
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    start_page = simpledialog.askinteger(
        "PDF Start Page",
        "Enter the page where connector data begins:",
        minvalue=1,
        parent=root
    )

    if start_page is None:
        root.destroy()
        return None, None

    separator_keyword = simpledialog.askstring(
        "Section Separator",
        "Enter the keyword that starts each connector section:",
        parent=root
    )

    root.destroy()

    if separator_keyword is None:
        return None, None

    separator_keyword = separator_keyword.strip()

    if separator_keyword == "":
        return None, None

    return start_page, separator_keyword


def ask_column_names(default_names):
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    default_text = ",".join(default_names)

    entered_text = simpledialog.askstring(
        "Column Names",
        "Enter the 9 column names from LEFT to RIGHT, separated by commas:",
        initialvalue=default_text,
        parent=root
    )

    root.destroy()

    if entered_text is None:
        return None

    column_names = []

    for value in entered_text.split(","):
        value = value.strip()

        if value != "":
            column_names.append(value)

    if len(column_names) != EXPECTED_COLUMN_COUNT:
        raise ValueError(
            f"Exactly {EXPECTED_COLUMN_COUNT} column names are required. "
            f"You entered {len(column_names)}."
        )

    return column_names


def extract_sections_by_keyword(
    pdf_path,
    start_page,
    separator_keyword
):
    sections = []
    current_section = []
    current_start_page = None

    keyword_lower = separator_keyword.lower()

    with pdfplumber.open(pdf_path) as pdf:

        if start_page > len(pdf.pages):
            raise ValueError(
                f"Start page {start_page} is greater than "
                f"PDF page count {len(pdf.pages)}"
            )

        for page_number in range(
            start_page,
            len(pdf.pages) + 1
        ):
            page = pdf.pages[page_number - 1]

            text = page.extract_text()

            if not text:
                continue

            lines = text.splitlines()

            for line in lines:
                clean_line = line.strip()

                if keyword_lower in clean_line.lower():

                    if current_section:
                        sections.append({
                            "section_number": len(sections) + 1,
                            "start_page": current_start_page,
                            "text": "\n".join(
                                current_section
                            ).strip()
                        })

                    current_section = [clean_line]
                    current_start_page = page_number

                else:

                    if current_section:
                        current_section.append(
                            clean_line
                        )

        if current_section:
            sections.append({
                "section_number": len(sections) + 1,
                "start_page": current_start_page,
                "text": "\n".join(
                    current_section
                ).strip()
            })

    return sections


# ================================================
# BASIC PDF GEOMETRY HELPERS
# ================================================

def extract_page_words(page):
    return page.extract_words(
        x_tolerance=3,
        y_tolerance=3,
        keep_blank_chars=False,
        use_text_flow=False
    )


def word_center_x(word):
    return (
        float(word["x0"]) +
        float(word["x1"])
    ) / 2.0


def group_words_into_lines(
    words,
    y_tolerance=3.0
):
    lines = []

    sorted_words = sorted(
        words,
        key=lambda word: (
            float(word["top"]),
            float(word["x0"])
        )
    )

    for word in sorted_words:
        word_top = float(word["top"])

        target_line = None

        for line in lines:

            if abs(
                line["top"] - word_top
            ) <= y_tolerance:
                target_line = line
                break

        if target_line is None:
            target_line = {
                "top": word_top,
                "words": []
            }

            lines.append(
                target_line
            )

        target_line["words"].append(
            word
        )

    for line in lines:
        line["words"] = sorted(
            line["words"],
            key=lambda word: float(
                word["x0"]
            )
        )

        line["text"] = " ".join(
            str(word["text"])
            for word in line["words"]
        ).strip()

    return lines


def find_exact_word(
    words,
    wanted_text,
    target_y=None,
    y_tolerance=60.0
):
    candidates = []

    for word in words:
        text = str(
            word["text"]
        ).strip().upper()

        if text != wanted_text.upper():
            continue

        if target_y is not None:

            if abs(
                float(word["top"]) -
                target_y
            ) > y_tolerance:
                continue

        candidates.append(
            word
        )

    if not candidates:
        return None

    if target_y is None:
        return candidates[0]

    candidates.sort(
        key=lambda word: abs(
            float(word["top"]) -
            target_y
        )
    )

    return candidates[0]


# ================================================
# REBUILD MALFORMED / FRAGMENTED CELL TEXT
# ================================================

def rebuild_cell_text(
    words,
    same_line_tolerance=3.0,
    join_gap=2.5
):
    if not words:
        return ""

    words = sorted(
        words,
        key=lambda word: (
            float(word["top"]),
            float(word["x0"])
        )
    )

    lines = []
    current_line = []
    current_y = None

    for word in words:
        word_y = float(
            word["top"]
        )

        if current_y is None:
            current_y = word_y
            current_line = [word]
            continue

        if abs(
            word_y - current_y
        ) <= same_line_tolerance:
            current_line.append(
                word
            )

        else:
            lines.append(
                current_line
            )

            current_line = [word]
            current_y = word_y

    if current_line:
        lines.append(
            current_line
        )

    rebuilt_lines = []

    for line in lines:
        line = sorted(
            line,
            key=lambda word: float(
                word["x0"]
            )
        )

        text = ""
        previous_x1 = None

        for word in line:
            value = str(
                word["text"]
            ).strip()

            if value == "":
                continue

            x0 = float(
                word["x0"]
            )

            x1 = float(
                word["x1"]
            )

            if text == "":
                text = value

            else:
                gap = (
                    x0 -
                    previous_x1
                )

                if gap <= join_gap:
                    text += value

                else:
                    text += " " + value

            previous_x1 = x1

        if text != "":
            rebuilt_lines.append(
                text
            )

    return " ".join(
        rebuilt_lines
    ).strip()


def clean_extracted_value(value):
    value = str(
        value
    ).strip()

    value = re.sub(
        r"\s+",
        " ",
        value
    )

    value = re.sub(
        r"\s+([,.;:])",
        r"\1",
        value
    )

    return value.strip()

def find_column_centers(
    pdf_path,
    start_page
):
    with pdfplumber.open(
        pdf_path
    ) as pdf:

        for page_number in range(
            start_page,
            len(pdf.pages) + 1
        ):
            page = pdf.pages[
                page_number - 1
            ]

            words = extract_page_words(
                page
            )

            if not words:
                continue

            lines = group_words_into_lines(
                words
            )

            for line in lines:
                upper_values = [
                    str(
                        word["text"]
                    ).strip().upper()
                    for word in line["words"]
                ]

                if "MNEMONIC" not in upper_values:
                    continue

                if "DESCRIPTION" not in upper_values:
                    continue

                header_y = line["top"]

                nearby_words = []

                for word in words:

                    if abs(
                        float(word["top"]) -
                        header_y
                    ) <= 60.0:
                        nearby_words.append(
                            word
                        )

                des_words = []
                posn_words = []

                for word in nearby_words:
                    text = str(
                        word["text"]
                    ).strip().upper()

                    if text == "DES":
                        des_words.append(
                            word
                        )

                    if text == "POSN":
                        posn_words.append(
                            word
                        )

                des_words.sort(
                    key=lambda word: float(
                        word["x0"]
                    )
                )

                posn_words.sort(
                    key=lambda word: float(
                        word["x0"]
                    )
                )

                if len(des_words) < 2:
                    continue

                if len(posn_words) < 2:
                    continue

                mnemonic_word = find_exact_word(
                    nearby_words,
                    "MNEMONIC",
                    header_y
                )

                description_word = find_exact_word(
                    nearby_words,
                    "DESCRIPTION",
                    header_y
                )

                wire_word = find_exact_word(
                    nearby_words,
                    "WIRE",
                    header_y
                )

                note_word = find_exact_word(
                    nearby_words,
                    "NOTE",
                    header_y
                )

                comment_word = find_exact_word(
                    nearby_words,
                    "COMMENT",
                    header_y
                )

                if mnemonic_word is None:
                    continue

                if description_word is None:
                    continue

                if wire_word is None:
                    continue

                if note_word is None:
                    continue

                if comment_word is None:
                    continue

                centers = [
                    word_center_x(
                        des_words[0]
                    ),
                    word_center_x(
                        posn_words[0]
                    ),
                    word_center_x(
                        des_words[1]
                    ),
                    word_center_x(
                        posn_words[1]
                    ),
                    word_center_x(
                        mnemonic_word
                    ),
                    word_center_x(
                        description_word
                    ),
                    word_center_x(
                        wire_word
                    ),
                    word_center_x(
                        note_word
                    ),
                    word_center_x(
                        comment_word
                    )
                ]

                centers_are_valid = True

                for index in range(
                    len(centers) - 1
                ):

                    if (
                        centers[index] >=
                        centers[index + 1]
                    ):
                        centers_are_valid = False
                        break

                if not centers_are_valid:
                    continue

                print("")
                print(
                    "============================================================"
                )
                print(
                    "TABLE COLUMNS LEARNED"
                )
                print(
                    "============================================================"
                )
                print(
                    "PAGE:",
                    page_number
                )

                for index, center in enumerate(
                    centers,
                    start=1
                ):
                    print(
                        "COLUMN",
                        index,
                        "X =",
                        round(
                            center,
                            2
                        )
                    )

                print(
                    "============================================================"
                )

                return (
                    centers,
                    float(page.width)
                )

    raise Exception(
        "Could not automatically determine "
        "table column locations from PDF headers"
    )


def build_column_boundaries(
    column_centers,
    page_width
):
    boundaries = [0.0]

    for index in range(
        len(column_centers) - 1
    ):
        midpoint = (
            column_centers[index] +
            column_centers[index + 1]
        ) / 2.0

        boundaries.append(
            midpoint
        )

    boundaries.append(
        page_width
    )

    return boundaries


def get_column_index_for_word(
    word,
    boundaries
):
    x = float(
        word["x0"]
    )

    for index in range(
        len(boundaries) - 1
    ):

        if (
            boundaries[index] <= x <
            boundaries[index + 1]
        ):
            return index

    return None


# ================================================
# SECTION MARKERS USING PDF POSITION
# ================================================

def find_section_markers(
    pdf_path,
    start_page,
    separator_keyword
):
    markers = []

    keyword_lower = (
        separator_keyword.lower()
    )

    with pdfplumber.open(
        pdf_path
    ) as pdf:

        for page_number in range(
            start_page,
            len(pdf.pages) + 1
        ):
            page = pdf.pages[
                page_number - 1
            ]

            words = extract_page_words(
                page
            )

            if not words:
                continue

            lines = group_words_into_lines(
                words
            )

            for line in lines:

                if keyword_lower in line["text"].lower():
                    markers.append({
                        "section_number": len(markers) + 1,
                        "page": page_number,
                        "top": float(
                            line["top"]
                        ),
                        "title": line["text"]
                    })

    return markers


def find_section_for_row(
    page_number,
    row_top,
    section_markers
):
    selected_marker = None

    for marker in section_markers:
        marker_position = (
            marker["page"],
            marker["top"]
        )

        row_position = (
            page_number,
            row_top
        )

        if (
            marker_position <=
            row_position
        ):
            selected_marker = marker

        else:
            break

    return selected_marker


# ================================================
# CONNECTOR ROW DETECTION
# ================================================

def is_connector_name(text):
    text = str(
        text
    ).strip()

    return (
        re.fullmatch(
            r"J\d+",
            text,
            re.IGNORECASE
        )
        is not None
    )


def find_connector_anchors(
    words,
    boundaries
):
    anchors = []

    for word in words:
        text = str(
            word["text"]
        ).strip()

        if not is_connector_name(
            text
        ):
            continue

        column_index = (
            get_column_index_for_word(
                word,
                boundaries
            )
        )

        if column_index != 0:
            continue

        anchors.append(
            word
        )

    anchors.sort(
        key=lambda word: float(
            word["top"]
        )
    )

    return anchors


# ================================================
# FILTER KNOWN NON-TABLE TEXT
# ================================================

def is_junk_line(
    line_text,
    separator_keyword
):
    text = line_text.strip()
    upper = text.upper()

    if text == "":
        return True

    if re.fullmatch(
        r"PAGE\s+\d+\s+OF\s+\d+",
        upper
    ):
        return True

    if upper in {
        "FROM",
        "TO",
        "SIGNAL NAME",
        "MNEMONIC",
        "DESCRIPTION",
        "WIRE",
        "CODE",
        "WIRE CODE",
        "NOTE",
        "COMMENT",
        "WIRE CODES"
    }:
        return True

    if upper.startswith(
        "DES POSN DES POSN"
    ):
        return True

    if (
        separator_keyword.lower()
        in text.lower()
    ):
        return True

    return False


# ================================================
# PARSE ONE PHYSICAL TABLE ROW
# ================================================

def parse_row_words(
    row_words,
    boundaries,
    column_names
):
    column_words = [
        []
        for _ in column_names
    ]

    for word in row_words:
        column_index = (
            get_column_index_for_word(
                word,
                boundaries
            )
        )

        if column_index is None:
            continue

        if (
            column_index < 0 or
            column_index >= len(
                column_names
            )
        ):
            continue

        column_words[
            column_index
        ].append(
            word
        )

    output = {}

    for index, column_name in enumerate(
        column_names
    ):
        value = rebuild_cell_text(
            column_words[index]
        )

        value = clean_extracted_value(
            value
        )

        output[
            column_name
        ] = value

    return output


# ================================================
# EXTRACT STRUCTURED TABLE ROWS
# ================================================

def extract_structured_rows(
    pdf_path,
    start_page,
    separator_keyword,
    column_names
):
    column_centers, page_width = (
        find_column_centers(
            pdf_path,
            start_page
        )
    )

    boundaries = (
        build_column_boundaries(
            column_centers,
            page_width
        )
    )

    section_markers = (
        find_section_markers(
            pdf_path,
            start_page,
            separator_keyword
        )
    )

    structured_rows = []

    with pdfplumber.open(
        pdf_path
    ) as pdf:

        for page_number in range(
            start_page,
            len(pdf.pages) + 1
        ):
            page = pdf.pages[
                page_number - 1
            ]

            words = extract_page_words(
                page
            )

            if not words:
                continue

            lines = group_words_into_lines(
                words
            )

            anchors = find_connector_anchors(
                words,
                boundaries
            )

            if not anchors:
                continue

            anchor_tops = [
                float(
                    anchor["top"]
                )
                for anchor in anchors
            ]

            gaps = []

            for index in range(
                len(anchor_tops) - 1
            ):
                gap = (
                    anchor_tops[index + 1] -
                    anchor_tops[index]
                )

                if gap > 0:
                    gaps.append(
                        gap
                    )

            if gaps:
                normal_row_height = (
                    statistics.median(
                        gaps
                    )
                )

            else:
                normal_row_height = 22.0

            maximum_last_row_height = max(
                normal_row_height * 2.75,
                50.0
            )

            for anchor_index, anchor in enumerate(
                anchors
            ):
                row_top = (
                    float(anchor["top"]) -
                    2.0
                )

                if (
                    anchor_index + 1 <
                    len(anchors)
                ):
                    row_bottom = (
                        float(
                            anchors[
                                anchor_index + 1
                            ]["top"]
                        ) -
                        1.0
                    )

                else:
                    row_bottom = (
                        float(
                            anchor["top"]
                        ) +
                        maximum_last_row_height
                    )

                row_words = []

                for line in lines:

                    if (
                        line["top"] <
                        row_top
                    ):
                        continue

                    if (
                        line["top"] >=
                        row_bottom
                    ):
                        continue

                    if is_junk_line(
                        line["text"],
                        separator_keyword
                    ):
                        continue

                    for word in line["words"]:
                        row_words.append(
                            word
                        )

                parsed_row = (
                    parse_row_words(
                        row_words,
                        boundaries,
                        column_names
                    )
                )

                from_des_key = (
                    column_names[0]
                )

                if not is_connector_name(
                    parsed_row[
                        from_des_key
                    ]
                ):
                    continue

                section_marker = (
                    find_section_for_row(
                        page_number,
                        float(
                            anchor["top"]
                        ),
                        section_markers
                    )
                )

                parsed_row[
                    "PAGE"
                ] = page_number

                if section_marker is None:
                    parsed_row[
                        "SECTION"
                    ] = None

                    parsed_row[
                        "SECTION_TITLE"
                    ] = ""

                else:
                    parsed_row[
                        "SECTION"
                    ] = section_marker[
                        "section_number"
                    ]

                    parsed_row[
                        "SECTION_TITLE"
                    ] = section_marker[
                        "title"
                    ]

                structured_rows.append(
                    parsed_row
                )

    return structured_rows


# ================================================
# GROUP STRUCTURED ROWS BY SECTION
# ================================================

def group_rows_by_section(
    structured_rows
):
    structured_sections = {}

    for row in structured_rows:
        section_number = row[
            "SECTION"
        ]

        if (
            section_number not in
            structured_sections
        ):
            structured_sections[
                section_number
            ] = []

        structured_sections[
            section_number
        ].append(
            row
        )

    return structured_sections


# ================================================
# VALIDATION
# ================================================

def validate_row(
    row,
    column_names
):
    problems = []

    from_des_key = (
        column_names[0]
    )

    from_posn_key = (
        column_names[1]
    )

    from_des = str(
        row.get(
            from_des_key,
            ""
        )
    ).strip()

    from_posn = str(
        row.get(
            from_posn_key,
            ""
        )
    ).strip()

    if not re.fullmatch(
        r"J\d+",
        from_des,
        re.IGNORECASE
    ):
        problems.append(
            "invalid connector"
        )

    if not re.fullmatch(
        r"\d+",
        from_posn
    ):
        problems.append(
            "invalid pin number"
        )

    return problems


def validate_structured_rows(
    structured_rows,
    column_names
):
    good_rows = []
    suspicious_rows = []

    for row in structured_rows:
        problems = validate_row(
            row,
            column_names
        )

        if problems:
            suspicious_rows.append({
                "row": row,
                "problems": problems
            })

        else:
            good_rows.append(
                row
            )

    return (
        good_rows,
        suspicious_rows
    )

                "############################################################"

                str(

                    )
                )

            )
        )
