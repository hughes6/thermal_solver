import os
import sys
import tempfile
import subprocess
from pathlib import Path


ROLE_ENV = "BLOCK_MAKER_ROLE"
METADATA_ENV = "E3_SECONDARY_BLOCK_FILE"


def get_application_directory():
    if getattr(sys, "frozen", False):
        return Path(sys.executable).resolve().parent

    return Path(__file__).resolve().parent


def get_child_command():
    if getattr(sys, "frozen", False):
        return [sys.executable]

    return [sys.executable, str(Path(__file__).resolve())]


def run_builder_child():
    import main


def run_formatter_child():
    import test


role = os.environ.get(ROLE_ENV, "").strip().lower()

if role == "main":
    run_builder_child()
    sys.exit(0)

if role == "formatter":
    run_formatter_child()
    sys.exit(0)


application_directory = get_application_directory()
child_command = get_child_command()

with tempfile.TemporaryDirectory(prefix="e3_block_maker_") as temp_directory:
    metadata_path = str(Path(temp_directory) / "secondary_blocks.json")

    main_environment = os.environ.copy()
    main_environment[ROLE_ENV] = "main"
    main_environment[METADATA_ENV] = metadata_path

    print("")
    print("============================================================")
    print("STARTING MAIN E3 BUILDER")
    print("============================================================")
    print("TEMP METADATA:", metadata_path)

    main_result = subprocess.run(child_command, cwd=str(application_directory), env=main_environment)

    if main_result.returncode != 0:
        sys.exit(main_result.returncode)

    formatter_environment = os.environ.copy()
    formatter_environment[ROLE_ENV] = "formatter"
    formatter_environment[METADATA_ENV] = metadata_path

    print("")
    print("============================================================")
    print("MAIN BUILDER COMPLETE")
    print("STARTING FORMATTER")
    print("============================================================")

    formatter_result = subprocess.run(child_command, cwd=str(application_directory), env=formatter_environment)

    sys.exit(formatter_result.returncode)
