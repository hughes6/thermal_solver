#====================================================================================================================================
# CALCULATE SYMBOL POSITION FROM EXISTING CABLE PIN LOCATIONS
#====================================================================================================================================
def calculate_placement(job, pin_map, sheet_id):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} has no schematic location on sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    if top_y - bottom_y < SCHEMATIC_MIN_Y_SPAN and len(job["pins"]) > 1:
        raise Exception(f"{job['side']} {job['connector']} {job['type']}: pins {job['pins']} all have effectively the same Y coordinate")

    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job)
    x = anchor_x + direction * offset

    if job["type"] == "twist":
        padding = TWIST_VERTICAL_PADDING
        y = bottom_y - padding
        length = top_y - bottom_y + 2.0 * padding
    elif job["type"] == "shield_inner":
        padding = INNER_SHIELD_VERTICAL_PADDING
        y = bottom_y
        length = top_y - bottom_y + padding
    else:
        padding = OUTER_SHIELD_VERTICAL_PADDING
        y = bottom_y
        length = top_y - bottom_y + padding

    length = max(length, MIN_SYMBOL_LENGTH)

    placement = {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y
    }

    return placement


#====================================================================================================================================
# PLACE REAL LIBRARY SYMBOL
#
# TWISTS:
#     Load
#     Place
#     SetDisplayLength
#
# SHIELDS:
#     Load
#     PRE-SIZE with SetDisplayLength
#     Place directly on the bottom conductor
#     Verify final size
#====================================================================================================================================
def place_job(sym, job, placement):
    print("")
    print("------------------------------------------------------------")
    print("TYPE:", job["type"])
    print("SIDE:", job["side"])
    print("CONNECTOR:", job["connector"])
    print("PINS:", job["pins"])
    print("EXCEL ROWS:", job["excel_rows"])
    print("LIBRARY SYMBOL:", job["symbol_name"])
    print("SHEET:", placement["sheet_id"])
    print("CONNECTOR PIN X:", placement["anchor_x"])
    print("SYMBOL X:", placement["x"])
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("SYMBOL Y:", placement["y"])
    print("REQUESTED SYMBOL LENGTH:", placement["length"])

    if DRY_RUN:
        print("DRY RUN - NOT PLACED")
        return None

    requested_length = float(placement["length"])

    if requested_length < MIN_SYMBOL_LENGTH:
        requested_length = MIN_SYMBOL_LENGTH

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("LOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not load library symbol {job['symbol_name']}")

    #================================================================================================================================
    # SHIELD
    #================================================================================================================================
    if job["type"] == "shield_inner" or job["type"] == "shield_outer":
        print("SHIELD PLACEMENT MODE: PRE-SIZE THEN PLACE")

        try:
            loaded_length_before = float(sym.GetDisplayLength())
        except Exception:
            loaded_length_before = None

        print("LOADED SHIELD LENGTH BEFORE:", loaded_length_before)

        pre_stretch_result = sym.SetDisplayLength(requested_length)

        print("PRE-PLACE SET DISPLAY LENGTH RESULT:", pre_stretch_result)

        try:
            loaded_length_after = float(sym.GetDisplayLength())
        except Exception:
            loaded_length_after = None

        print("LOADED SHIELD LENGTH AFTER:", loaded_length_after)

        placed_id = sym.Place(placement["sheet_id"], placement["x"], placement["y"], 0, 1.0, 0)

        print("SHIELD PLACE RESULT / SYMBOL ID:", placed_id)

        if placed_id == 0:
            raise Exception(f"Could not place library shield {job['symbol_name']} at X={placement['x']} Y={placement['y']}")

        sym.SetId(placed_id)

        try:
            placed_name = sym.GetName()
        except Exception:
            placed_name = "UNKNOWN"

        try:
            symbol_type = sym.GetSymbolType()
        except Exception:
            symbol_type = "UNKNOWN"

        try:
            final_length = float(sym.GetDisplayLength())
        except Exception:
            final_length = None

        print("PLACED SHIELD ID:", placed_id)
        print("PLACED SHIELD NAME:", placed_name)
        print("PLACED SHIELD TYPE:", symbol_type)
        print("FINAL SHIELD DISPLAY LENGTH:", final_length)

        if final_length is None:
            raise Exception(f"Could not verify placed shield {job['symbol_name']}")

        length_error = abs(final_length - requested_length)

        if length_error <= 0.001:
            print("REAL SHIELD PLACED AT REQUESTED LENGTH")
            return placed_id

        print("PRE-SIZED SHIELD DID NOT RETAIN REQUESTED LENGTH")
        print("REQUESTED:", requested_length)
        print("ACTUAL:", final_length)

        post_stretch_result = sym.SetDisplayLength(requested_length)

        print("POST-PLACE SHIELD SET DISPLAY LENGTH RESULT:", post_stretch_result)

        sym.SetId(placed_id)

        try:
            final_length = float(sym.GetDisplayLength())
        except Exception:
            final_length = None

        print("SHIELD LENGTH AFTER SECOND ATTEMPT:", final_length)

        if final_length is not None and abs(final_length - requested_length) <= 0.001:
            print("REAL SHIELD PLACED AND STRETCHED SUCCESSFULLY")
            return placed_id

        raise Exception(f"Cable Shield placed but could not stretch to {requested_length}")

    #================================================================================================================================
    # TWIST
    #================================================================================================================================
    print("TWIST PLACEMENT MODE: PLACE THEN STRETCH")

    placed_id = sym.Place(placement["sheet_id"], placement["x"], placement["y"], 0, 1.0, 0)

    print("TWIST PLACE RESULT / SYMBOL ID:", placed_id)

    if placed_id == 0:
        raise Exception(f"Could not place library twist {job['symbol_name']}")

    sym.SetId(placed_id)

    try:
        placed_name = sym.GetName()
    except Exception:
        placed_name = "UNKNOWN"

    try:
        symbol_type = sym.GetSymbolType()
    except Exception:
        symbol_type = "UNKNOWN"

    try:
        length_before = float(sym.GetDisplayLength())
    except Exception:
        length_before = None

    print("PLACED TWIST ID:", placed_id)
    print("PLACED TWIST NAME:", placed_name)
    print("PLACED TWIST TYPE:", symbol_type)
    print("TWIST DISPLAY LENGTH BEFORE:", length_before)

    stretch_result = sym.SetDisplayLength(requested_length)

    print("TWIST SET DISPLAY LENGTH RESULT:", stretch_result)

    sym.SetId(placed_id)

    try:
        length_after = float(sym.GetDisplayLength())
    except Exception:
        length_after = None

    print("REQUESTED TWIST LENGTH:", requested_length)
    print("TWIST DISPLAY LENGTH AFTER:", length_after)

    if length_after is None:
        raise Exception(f"Could not verify display length for {job['symbol_name']}")

    if abs(length_after - requested_length) > 0.001:
        raise Exception(f"{job['symbol_name']} placed but failed to stretch from {length_before} to {requested_length}")

    print("REAL TWIST SYMBOL PLACED AND STRETCHED SUCCESSFULLY")

    return placed_id
