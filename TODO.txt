def parse_pasted_column_names(raw_text):
    raw_text = raw_text.replace("\r\n", "\n")
    raw_text = raw_text.replace("\r", "\n")
    raw_text = raw_text.strip()
    if raw_text == "":
        return []
    lines = [line.strip() for line in raw_text.split("\n") if line.strip() != ""]
    if len(lines) > 1:
        return lines
    one_line = lines[0] if lines else ""
    if "\t" in one_line:
        return [value.strip() for value in one_line.split("\t") if value.strip() != ""]
    if "," in one_line:
        return [value.strip() for value in one_line.split(",") if value.strip() != ""]
    spaced = [value.strip() for value in re.split(r"\s{2,}", one_line) if value.strip() != ""]
    if len(spaced) > 1:
        return spaced
    return [one_line] if one_line else []

def guess_role_column(column_names, possible_names):
    for column_name in column_names:
        upper_name = column_name.upper()
        for possible_name in possible_names:
            if possible_name.upper() == upper_name:
                return column_name
    for column_name in column_names:
        upper_name = column_name.upper()
        for possible_name in possible_names:
            if possible_name.upper() in upper_name:
                return column_name
    return column_names[0]

def canonical_connector_name(value):
    value = str(value or "").strip().upper()
    match = re.search(r"J\s*0*(\d+)\b", value)
    if match:
        return f"J{int(match.group(1))}"
    return value


def connector_sort_key(value):
    match = re.search(r"\d+", str(value))
    if match:
        return int(match.group())
    return 999999


def clean_extracted_value(value):
    value = str(value or "").strip()
    value = re.sub(r"\s+", " ", value)
    value = re.sub(r"\s+([,.;:])", r"\1", value)
    return value


def get_role_value(row, pdf_roles, role_name):
    column_name = pdf_roles[role_name]
    return str(row.get(column_name, "") or "").strip()


def normalize_direction_value(value):
    text = str(value or "").strip().upper()
    text = re.sub(r"\s+", " ", text)
    if re.search(r"(?<![A-Z0-9_])IN\s*/\s*OUT(?![A-Z0-9_])", text):
        return "IN/OUT"
    if re.search(r"(?<![A-Z0-9_])OUT(?![A-Z0-9_])", text):
        return "OUT"
    if re.search(r"(?<![A-Z0-9_])IN(?![A-Z0-9_])", text):
        return "IN"
    return ""


def extract_role_text_from_words(row_words, boundaries, column_index, left_pad=0.0, right_pad=0.0, skip_numeric=False):
    if column_index < 0 or column_index >= len(boundaries) - 1:
        return ""
    left = boundaries[column_index] - left_pad
    right = boundaries[column_index + 1] + right_pad
    candidates = []
    for word in row_words:
        value = str(word.get("text", "")).strip()
        if value == "":
            continue
        x0 = float(word["x0"])
        if x0 < left or x0 >= right:
            continue
        if skip_numeric and re.fullmatch(r"[+-]?\d+(?:\.\d+)?", value):
            continue
        candidates.append(word)
    if not candidates:
        return ""
    candidates.sort(key=lambda word: (float(word["top"]), float(word["x0"])))
    line_groups = []
    for word in candidates:
        top = float(word["top"])
        if not line_groups or abs(top - line_groups[-1]["top"]) > 3.0:
            line_groups.append({"top": top, "words": [word]})
        else:
            line_groups[-1]["words"].append(word)
    rebuilt_lines = []
    for line in line_groups:
        line["words"].sort(key=lambda word: float(word["x0"]))
        line_text = ""
        previous_x1 = None
        for word in line["words"]:
            value = str(word.get("text", "")).strip()
            if value == "":
                continue
            x0 = float(word["x0"])
            x1 = float(word["x1"])
            if line_text != "" and previous_x1 is not None:
                gap = x0 - previous_x1
                if gap > 2.5:
                    line_text += " "
            line_text += value
            previous_x1 = x1
        if line_text != "":
            rebuilt_lines.append(line_text)
    return clean_extracted_value(" ".join(value for value in rebuilt_lines if value != ""))

def word_center_x(word):
    return (float(word["x0"]) + float(word["x1"])) / 2.0


def char_center_x(char):
    return (float(char["x0"]) + float(char["x1"])) / 2.0


def group_words_into_lines(words, y_tolerance=3.0):
    lines = []
    sorted_words = sorted(words, key=lambda word: (float(word["top"]), float(word["x0"])))
    for word in sorted_words:
        word_top = float(word["top"])
        target_line = None
        for line in lines:
            if abs(line["top"] - word_top) <= y_tolerance:
                target_line = line
                break
        if target_line is None:
            target_line = {"top": word_top, "words": []}
            lines.append(target_line)
        target_line["words"].append(word)
    for line in lines:
        line["words"] = sorted(line["words"], key=lambda word: float(word["x0"]))
        line["text"] = " ".join(str(word["text"]) for word in line["words"]).strip()
    return lines


def group_chars_into_lines(chars, y_tolerance=2.5):
    lines = []
    sorted_chars = sorted(chars, key=lambda char: (float(char["top"]), float(char["x0"])))
    for char in sorted_chars:
        char_top = float(char["top"])
        target_line = None
        for line in lines:
            if abs(line["top"] - char_top) <= y_tolerance:
                target_line = line
                break
        if target_line is None:
            target_line = {"top": char_top, "chars": []}
            lines.append(target_line)
        target_line["chars"].append(char)
    for line in lines:
        line["chars"] = sorted(line["chars"], key=lambda char: float(char["x0"]))
    return lines


def rebuild_cell_from_chars(chars, y_tolerance=2.5):
    if not chars:
        return ""
    lines = group_chars_into_lines(chars, y_tolerance=y_tolerance)
    rebuilt_lines = []
    for line in lines:
        line_chars = line["chars"]
        widths = [float(char["x1"]) - float(char["x0"]) for char in line_chars if str(char.get("text", "")).strip() != ""]
        median_width = statistics.median(widths) if widths else 3.0
        space_gap = max(1.25, median_width * 0.55)
        line_text = ""
        previous_x1 = None
        for char in line_chars:
            value = str(char.get("text", ""))
            x0 = float(char["x0"])
            x1 = float(char["x1"])
            if value == "":
                continue
            if value.isspace():
                if line_text and not line_text.endswith(" "):
                    line_text += " "
                previous_x1 = x1
                continue
            if previous_x1 is not None:
                gap = x0 - previous_x1
                if gap > space_gap and line_text and not line_text.endswith(" "):
                    line_text += " "
            line_text += value
            previous_x1 = x1
        line_text = clean_extracted_value(line_text)
        if line_text:
            rebuilt_lines.append(line_text)
    return clean_extracted_value(" ".join(rebuilt_lines))


def find_exact_word(words, wanted_text, target_y=None, y_tolerance=60.0):
    candidates = []
    for word in words:
        text = str(word["text"]).strip().upper()
        if text != wanted_text.upper():
            continue
        if target_y is not None and abs(float(word["top"]) - target_y) > y_tolerance:
            continue
        candidates.append(word)
    if not candidates:
        return None
    if target_y is None:
        return candidates[0]
    candidates.sort(key=lambda word: abs(float(word["top"]) - target_y))
    return candidates[0]

def find_page_footer_top(lines):
    footer_tops = []
    for line in lines:
        line_text = str(line.get("text", "")).strip()
        if re.fullmatch(r"Page\s+\d+\s+of\s+\d+", line_text, re.IGNORECASE):
            footer_tops.append(float(line["top"]))
    return min(footer_tops) if footer_tops else None


def load_pdf_cache(pdf_path, start_page):
    start_time = time.perf_counter()
    pdf_cache = {}
    with pdfplumber.open(pdf_path) as pdf:
        if start_page > len(pdf.pages):
            raise ValueError(f"Start page {start_page} is greater than PDF page count {len(pdf.pages)}")
        for page_number in range(start_page, len(pdf.pages) + 1):
            page = pdf.pages[page_number - 1]
            text = page.extract_text() or ""
            words = page.extract_words(x_tolerance=3, y_tolerance=3, keep_blank_chars=False, use_text_flow=False)
            chars = list(page.chars)
            lines = group_words_into_lines(words)
            edges = list(page.edges)
            pdf_cache[page_number] = {"text": text, "words": words, "chars": chars, "lines": lines, "edges": edges, "footer_top": find_page_footer_top(lines), "width": float(page.width), "height": float(page.height)}
    elapsed = time.perf_counter() - start_time
    print(f"PDF CACHE BUILT: {len(pdf_cache)} PAGES IN {elapsed:.2f} SECONDS")
    return pdf_cache

def extract_sections_by_keyword(pdf_cache, separator_keyword):
    sections = []
    current_section = []
    current_start_page = None
    keyword_lower = separator_keyword.lower()
    for page_number in sorted(pdf_cache):
        text = pdf_cache[page_number]["text"]
        if not text:
            continue
        for line in text.splitlines():
            clean_line = line.strip()
            if keyword_lower in clean_line.lower():
                if current_section:
                    sections.append({"section_number": len(sections) + 1, "start_page": current_start_page, "text": "\n".join(current_section).strip()})
                current_section = [clean_line]
                current_start_page = page_number
            elif current_section:
                current_section.append(clean_line)
    if current_section:
        sections.append({"section_number": len(sections) + 1, "start_page": current_start_page, "text": "\n".join(current_section).strip()})
    return sections

# LEARN COLUMN GEOMETRY FROM PDF HEADER
# ================================================
def edge_vertical_length(edge):
    if "top" in edge and "bottom" in edge:
        return abs(float(edge["bottom"]) - float(edge["top"]))
    if "y0" in edge and "y1" in edge:
        return abs(float(edge["y1"]) - float(edge["y0"]))
    return 0.0


def edge_x_position(edge):
    if "x0" not in edge or "x1" not in edge:
        return None
    x0 = float(edge["x0"])
    x1 = float(edge["x1"])
    if abs(x1 - x0) > 1.5:
        return None
    return (x0 + x1) / 2.0


def cluster_vertical_edges(edges, tolerance=1.5):
    raw = []
    for edge in edges:
        x = edge_x_position(edge)
        if x is None:
            continue
        length = edge_vertical_length(edge)
        if length < 8.0:
            continue
        raw.append((x, length))
    raw.sort(key=lambda item: item[0])
    clusters = []
    for x, length in raw:
        if not clusters or abs(x - clusters[-1]["x"]) > tolerance:
            clusters.append({"x": x, "count": 1, "length": length})
        else:
            cluster = clusters[-1]
            total = cluster["count"] + 1
            cluster["x"] = ((cluster["x"] * cluster["count"]) + x) / total
            cluster["count"] = total
            cluster["length"] += length
    return clusters


def choose_grid_boundary(clusters, left_center, right_center):
    candidates = [cluster for cluster in clusters if left_center < cluster["x"] < right_center]
    if not candidates:
        return None
    midpoint = (left_center + right_center) / 2.0
    candidates.sort(key=lambda cluster: (-cluster["length"], abs(cluster["x"] - midpoint)))
    return candidates[0]["x"]


def find_column_centers(pdf_cache, column_names):
    if len(column_names) != 9:
        raise ValueError("This connector-table parser currently expects exactly 9 physical columns")
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        words = page_data["words"]
        lines = page_data["lines"]
        if not words:
            continue
        for line in lines:
            upper_values = [str(word["text"]).strip().upper() for word in line["words"]]
            if "MNEMONIC" not in upper_values or "DESCRIPTION" not in upper_values:
                continue
            header_y = line["top"]
            nearby_words = [word for word in words if abs(float(word["top"]) - header_y) <= 60.0]
            des_words = [word for word in nearby_words if str(word["text"]).strip().upper() == "DES"]
            posn_words = [word for word in nearby_words if str(word["text"]).strip().upper() == "POSN"]
            des_words.sort(key=lambda word: float(word["x0"]))
            posn_words.sort(key=lambda word: float(word["x0"]))
            if len(des_words) < 2 or len(posn_words) < 2:
                continue
            mnemonic_word = find_exact_word(nearby_words, "MNEMONIC", header_y)
            description_word = find_exact_word(nearby_words, "DESCRIPTION", header_y)
            wire_word = find_exact_word(nearby_words, "WIRE", header_y)
            note_word = find_exact_word(nearby_words, "NOTE", header_y)
            comment_word = find_exact_word(nearby_words, "COMMENT", header_y)
            if mnemonic_word is None or description_word is None or wire_word is None or note_word is None or comment_word is None:
                continue
            centers = [word_center_x(des_words[0]), word_center_x(posn_words[0]), word_center_x(des_words[1]), word_center_x(posn_words[1]), word_center_x(mnemonic_word), word_center_x(description_word), word_center_x(wire_word), word_center_x(note_word), word_center_x(comment_word)]
            if any(centers[index] >= centers[index + 1] for index in range(len(centers) - 1)):
                continue
            print("")
            print("============================================================")
            print("TABLE COLUMNS LEARNED")
            print("============================================================")
            print("PAGE:", page_number)
            for name, center in zip(column_names, centers):
                print(name, "X =", round(center, 2))
            print("============================================================")
            return centers, page_data["width"], page_number
    raise Exception("Could not automatically determine table column locations from PDF headers")


def build_column_boundaries(pdf_cache, column_centers, page_width, header_page):
    page_data = pdf_cache[header_page]
    clusters = cluster_vertical_edges(page_data.get("edges", []))
    boundaries = []
    left_candidates = [cluster for cluster in clusters if cluster["x"] < column_centers[0]]
    right_candidates = [cluster for cluster in clusters if cluster["x"] > column_centers[-1]]
    left_boundary = max(left_candidates, key=lambda cluster: cluster["x"])["x"] if left_candidates else 0.0
    right_boundary = min(right_candidates, key=lambda cluster: cluster["x"])["x"] if right_candidates else page_width
    boundaries.append(left_boundary)
    grid_success = True
    for index in range(len(column_centers) - 1):
        boundary = choose_grid_boundary(clusters, column_centers[index], column_centers[index + 1])
        if boundary is None:
            grid_success = False
            boundary = (column_centers[index] + column_centers[index + 1]) / 2.0
        boundaries.append(boundary)
    boundaries.append(right_boundary)
    if any(boundaries[index] >= boundaries[index + 1] for index in range(len(boundaries) - 1)):
        boundaries = [0.0]
        for index in range(len(column_centers) - 1):
            boundaries.append((column_centers[index] + column_centers[index + 1]) / 2.0)
        boundaries.append(page_width)
        grid_success = False
    print("")
    print("============================================================")
    print("TABLE COLUMN BOUNDARIES")
    print("============================================================")
    print("METHOD:", "PDF GRID LINES" if grid_success else "GRID + MIDPOINT FALLBACK")
    for index, boundary in enumerate(boundaries):
        print("BOUNDARY", index, "X =", round(boundary, 2))
    print("============================================================")
    return boundaries


def get_column_index_for_x(x_value, boundaries):
    for index in range(len(boundaries) - 1):
        if boundaries[index] <= x_value < boundaries[index + 1]:
            return index
    return None

def find_section_markers(pdf_cache, separator_keyword):
    markers = []
    keyword_lower = separator_keyword.lower()
    for page_number in sorted(pdf_cache):
        for line in pdf_cache[page_number]["lines"]:
            if keyword_lower in line["text"].lower():
                markers.append({"section_number": len(markers) + 1, "page": page_number, "top": float(line["top"]), "title": line["text"]})
    return markers


def find_section_for_row(page_number, row_top, section_markers):
    selected_marker = None
    for marker in section_markers:
        if (marker["page"], marker["top"]) <= (page_number, row_top):
            selected_marker = marker
        else:
            break
    return selected_marker


# ================================================
# CONNECTOR ROW ANCHOR DETECTION
# ================================================
def is_connector_name(text):
    return re.fullmatch(r"J\d+", canonical_connector_name(text), re.IGNORECASE) is not None


def find_connector_anchors(chars, boundaries):
    first_column_right = boundaries[1]
    first_column_chars = [char for char in chars if char_center_x(char) < first_column_right]
    lines = group_chars_into_lines(first_column_chars, y_tolerance=2.5)
    anchors = []
    for line in lines:
        value = rebuild_cell_from_chars(line["chars"])
        canonical = canonical_connector_name(value)
        if not is_connector_name(canonical):
            continue
        anchors.append({"connector": canonical, "top": float(line["top"])})
    anchors.sort(key=lambda anchor: anchor["top"])
    deduped = []
    for anchor in anchors:
        if deduped and abs(anchor["top"] - deduped[-1]["top"]) <= 2.0:
            continue
        deduped.append(anchor)
    return deduped


# ================================================
# PARSE ONE ROW USING CHARACTER GEOMETRY
# ================================================
def parse_row_chars(row_chars, boundaries, column_names):
    column_chars = [[] for _ in column_names]
    for char in row_chars:
        column_index = get_column_index_for_x(char_center_x(char), boundaries)
        if column_index is None or column_index < 0 or column_index >= len(column_names):
            continue
        column_chars[column_index].append(char)
    output = {}
    for index, column_name in enumerate(column_names):
        output[column_name] = rebuild_cell_from_chars(column_chars[index])
    return output

def extract_structured_rows(pdf_cache, separator_keyword, column_names, pdf_roles):
    start_time = time.perf_counter()
    column_centers, page_width, header_page = find_column_centers(pdf_cache, column_names)
    reference_boundaries = build_column_boundaries(pdf_cache, column_centers, page_width, header_page)
    section_markers = find_section_markers(pdf_cache, separator_keyword)
    signal_index = column_names.index(pdf_roles["signal"])
    direction_index = column_names.index(pdf_roles["direction"])
    structured_rows = []
    for page_number in sorted(pdf_cache):
        page_data = pdf_cache[page_number]
        chars = page_data["chars"]
        words = page_data["words"]
        if not chars:
            continue
        boundaries = reference_boundaries
        anchors = find_connector_anchors(chars, boundaries)
        if not anchors:
            continue
        anchor_tops = [float(anchor["top"]) for anchor in anchors]
        gaps = [anchor_tops[index + 1] - anchor_tops[index] for index in range(len(anchor_tops) - 1) if anchor_tops[index + 1] - anchor_tops[index] > 0]
        normal_row_height = statistics.median(gaps) if gaps else 22.0
        maximum_last_row_height = max(normal_row_height * 2.75, 50.0)
        page_bottom = page_data["footer_top"] - 2.0 if page_data.get("footer_top") is not None else page_data["height"]
        for anchor_index, anchor in enumerate(anchors):
            row_top = float(anchor["top"]) - 2.0
            row_bottom = float(anchors[anchor_index + 1]["top"]) - 1.0 if anchor_index + 1 < len(anchors) else float(anchor["top"]) + maximum_last_row_height
            row_bottom = min(row_bottom, page_bottom)
            if row_bottom <= row_top:
                continue
            row_chars = [char for char in chars if row_top <= float(char["top"]) < row_bottom]
            row_words = [word for word in words if row_top <= float(word["top"]) < row_bottom]
            parsed_row = parse_row_chars(row_chars, boundaries, column_names)
            signal_from_words = extract_role_text_from_words(row_words, boundaries, signal_index, left_pad=10.0, right_pad=0.0, skip_numeric=True)
            direction_from_words = extract_role_text_from_words(row_words, boundaries, direction_index, left_pad=4.0, right_pad=4.0, skip_numeric=False)
            if signal_from_words != "":
                parsed_row[pdf_roles["signal"]] = signal_from_words
            parsed_row[pdf_roles["direction"]] = normalize_direction_value(direction_from_words if direction_from_words != "" else parsed_row.get(pdf_roles["direction"], ""))
            parsed_row[column_names[0]] = anchor["connector"]
            if not is_connector_name(parsed_row[column_names[0]]):
                continue
            section_marker = find_section_for_row(page_number, float(anchor["top"]), section_markers)
            parsed_row["PAGE"] = page_number
            parsed_row["SECTION"] = section_marker["section_number"] if section_marker else None
            parsed_row["SECTION_TITLE"] = section_marker["title"] if section_marker else ""
            structured_rows.append(parsed_row)
    elapsed = time.perf_counter() - start_time
    print(f"STRUCTURED ROW EXTRACTION TIME: {elapsed:.2f} SECONDS")
    return structured_rows

def validate_row(row, pdf_roles):
    problems = []
    connector = canonical_connector_name(get_role_value(row, pdf_roles, "connector"))
    pin_text = get_role_value(row, pdf_roles, "pin")
    if not re.fullmatch(r"J\d+", connector, re.IGNORECASE):
        problems.append("invalid connector")
    if not re.fullmatch(r"\d+", pin_text):
        problems.append("invalid pin")
    return problems


def validate_structured_rows(structured_rows, pdf_roles):
    good_rows = []
    suspicious_rows = []
    for row in structured_rows:
        problems = validate_row(row, pdf_roles)
        if problems:
            suspicious_rows.append({"row": row, "problems": problems})
        else:
            good_rows.append(row)
    return good_rows, suspicious_rows


def build_pdf_connector_map(structured_rows, pdf_roles):
    pdf_connectors = {}
    for row in structured_rows:
        connector = canonical_connector_name(get_role_value(row, pdf_roles, "connector"))
        if not is_connector_name(connector):
            continue
        pdf_connectors.setdefault(connector, []).append(row)
    return pdf_connectors


def get_connector_pin_map(rows, pdf_roles):
    pin_map = {}
    for row in rows:
        pin_text = get_role_value(row, pdf_roles, "pin")
        if not re.fullmatch(r"\d+", pin_text):
            continue
        pin_map[int(pin_text)] = row
    return pin_map

def read_excel_connector_info(excel_path):
    import win32com.client as win32
    start_time = time.perf_counter()
    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    workbook = None
    try:
        workbook = excel.Workbooks.Open(os.path.abspath(excel_path), UpdateLinks=0, ReadOnly=True, AddToMru=False)
        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)
        raw_count = worksheet.Range(EXCEL_CONNECTOR_COUNT_CELL).Value2
        connector_count = int(raw_count) if raw_count is not None and str(raw_count).strip() != "" else 0
        if connector_count < 0:
            connector_count = 0
        if connector_count > MAX_BLOCKS:
            connector_count = MAX_BLOCKS
        if connector_count == 0:
            return [], {}
        data_range = worksheet.Range(worksheet.Cells(EXCEL_CONNECTOR_FIRST_ROW, EXCEL_CONNECTOR_NAME_COLUMN), worksheet.Cells(EXCEL_CONNECTOR_FIRST_ROW + connector_count - 1, EXCEL_EXPECTED_PIN_COUNT_COLUMN))
        data = data_range.Value2
        connector_info = []
        expected_counts = {}
        for index, row_values in enumerate(data):
            excel_name = str(row_values[0] or "").strip()
            canonical = canonical_connector_name(excel_name)
            raw_expected = row_values[EXCEL_EXPECTED_PIN_COUNT_COLUMN - EXCEL_CONNECTOR_NAME_COLUMN]
            try:
                expected = int(raw_expected)
            except Exception:
                expected = None
            connector_info.append({"block_index": index, "excel_row": EXCEL_CONNECTOR_FIRST_ROW + index, "excel_name": excel_name, "canonical": canonical, "expected_pins": expected})
            if canonical:
                expected_counts[canonical] = expected
        elapsed = time.perf_counter() - start_time
        print(f"EXCEL CONNECTOR READ TIME: {elapsed:.2f} SECONDS")
        return connector_info, expected_counts
    finally:
        if workbook is not None:
            workbook.Close(SaveChanges=False)
        excel.Quit()


def build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info):
    jobs = []
    for connector in connector_info:
        block_index = connector["block_index"]
        excel_name = connector["excel_name"]
        canonical = connector["canonical"]
        expected_pins = connector["expected_pins"]
        if canonical not in pdf_connectors:
            jobs.append({"block_index": block_index, "excel_name": excel_name, "canonical": canonical, "expected_pins": expected_pins, "skip": True, "reason": "connector not found in PDF"})
            continue
        pin_map = get_connector_pin_map(pdf_connectors[canonical], pdf_roles)
        if expected_pins is not None and expected_pins > BLOCK_DATA_ROW_CAPACITY:
            raise ValueError(f"{excel_name} expects {expected_pins} pins but BLOCK_DATA_ROW_CAPACITY is only {BLOCK_DATA_ROW_CAPACITY}")
        matrix = [["", "", ""] for _ in range(BLOCK_DATA_ROW_CAPACITY)]
        for pin_number, row in pin_map.items():
            if pin_number < 1 or pin_number > BLOCK_DATA_ROW_CAPACITY:
                continue
            matrix[pin_number - 1][0] = pin_number
            matrix[pin_number - 1][1] = get_role_value(row, pdf_roles, "signal")
            matrix[pin_number - 1][2] = normalize_direction_value(get_role_value(row, pdf_roles, "direction"))
        missing = []
        if expected_pins is not None and expected_pins > 0:
            missing = [pin_number for pin_number in range(1, expected_pins + 1) if pin_number not in pin_map]
        block_start_column = BLOCK_FIRST_COLUMN + block_index * BLOCK_WIDTH
        pin_column = block_start_column + BLOCK_PIN_COLUMN_OFFSET
        direction_column = block_start_column + BLOCK_DIRECTION_COLUMN_OFFSET
        jobs.append({"block_index": block_index, "excel_name": excel_name, "canonical": canonical, "expected_pins": expected_pins, "skip": False, "pin_count": len(pin_map), "missing": missing, "first_row": BLOCK_DATA_FIRST_ROW, "last_row": BLOCK_DATA_FIRST_ROW + BLOCK_DATA_ROW_CAPACITY - 1, "first_column": pin_column, "last_column": direction_column, "matrix": tuple(tuple(row) for row in matrix)})
    return jobs


def write_pdf_data_to_excel(source_excel_path, output_excel_path, pdf_connectors, pdf_roles, connector_info):
    import win32com.client as win32
    total_start = time.perf_counter()
    source_excel_path = os.path.abspath(source_excel_path)
    output_excel_path = os.path.abspath(output_excel_path)
    if source_excel_path.lower() != output_excel_path.lower():
        shutil.copy2(source_excel_path, output_excel_path)
    jobs = build_excel_write_jobs(pdf_connectors, pdf_roles, connector_info)
    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    excel.AskToUpdateLinks = False
    workbook = None
    previous_calculation = None
    calculation_changed = False
    try:
        open_start = time.perf_counter()
        workbook = excel.Workbooks.Open(output_excel_path, UpdateLinks=0, ReadOnly=False, AddToMru=False)
        worksheet = workbook.Worksheets(TARGET_SHEET_NAME)
        open_elapsed = time.perf_counter() - open_start
        print(f"EXCEL OPEN TIME: {open_elapsed:.2f} SECONDS")
        try:
            previous_calculation = excel.Calculation
            excel.Calculation = XL_CALCULATION_MANUAL
            calculation_changed = True
            print("EXCEL CALCULATION MODE: MANUAL")
        except Exception as e:
            print("WARNING: COULD NOT SET MANUAL CALCULATION:", repr(e))
        print("")
        print("============================================================")
        print("EXCEL CONNECTOR CHECK")
        print("============================================================")
        print("EXCEL CONNECTORS:", len(connector_info))
        print("PDF CONNECTORS:", len(pdf_connectors))
        print("============================================================")
        write_start = time.perf_counter()
        written_blocks = 0
        written_pin_rows = 0
        for job in jobs:
            print("")
            print("BLOCK:", job["block_index"] + 1, "EXCEL CONNECTOR:", repr(job["excel_name"]), "PDF KEY:", repr(job["canonical"]))
            if job["skip"]:
                print("SKIPPED -", job["reason"], ":", job["excel_name"])
                continue
            destination_range = worksheet.Range(worksheet.Cells(job["first_row"], job["first_column"]), worksheet.Cells(job["last_row"], job["last_column"]))
            destination_range.Value2 = job["matrix"]
            written_blocks += 1
            written_pin_rows += job["pin_count"]
            print("WRITTEN PINS:", job["pin_count"], "EXPECTED:", job["expected_pins"], "MISSING:", job["missing"])
        write_elapsed = time.perf_counter() - write_start
        print(f"BULK EXCEL WRITE TIME: {write_elapsed:.2f} SECONDS")
        calc_start = time.perf_counter()
        try:
            worksheet.Calculate()
            print(f"SELECTION SHEET CALCULATION TIME: {time.perf_counter() - calc_start:.2f} SECONDS")
        except Exception as e:
            print("WARNING: SELECTION SHEET CALCULATION FAILED:", repr(e))
        if calculation_changed and previous_calculation is not None:
            try:
                excel.Calculation = previous_calculation
            except Exception as e:
                print("WARNING: COULD NOT RESTORE CALCULATION MODE:", repr(e))
        save_start = time.perf_counter()
        workbook.Save()
        save_elapsed = time.perf_counter() - save_start
        print(f"EXCEL SAVE TIME: {save_elapsed:.2f} SECONDS")
        print("")
        print("============================================================")
        print("EXCEL WRITE COMPLETE")
        print("============================================================")
        print("OUTPUT:", output_excel_path)
        print("BLOCKS WRITTEN:", written_blocks)
        print("PIN ROWS WRITTEN:", written_pin_rows)
        print("============================================================")
    finally:
        if workbook is not None:
            try:
                workbook.Close(SaveChanges=False)
            except Exception:
                pass
        try:
            excel.EnableEvents = True
            excel.ScreenUpdating = True
        except Exception:
            pass
        try:
            excel.Quit()
        except Exception:
            pass
    total_elapsed = time.perf_counter() - total_start
    print(f"TOTAL EXCEL WRITE PHASE: {total_elapsed:.2f} SECONDS")

def excel_column_letter(column_number):
    result = ""
    number = int(column_number)

    while number > 0:
        number, remainder = divmod(number - 1, 26)
        result = chr(65 + remainder) + result

    return result


def cell_address(row, column):
    return f"{excel_column_letter(column)}{row}"


def safe_text(value):
    if value is None:
        return ""

    if isinstance(value, bool):
        return "TRUE" if value else "FALSE"

    if isinstance(value, float):
        if math.isnan(value):
            return "NaN"
        if value.is_integer():
            return str(int(value))

    return str(value)


def report_text(value):
    text = safe_text(value)

    if text.startswith("=") or text.startswith("+") or text.startswith("-") or text.startswith("@"):
        return "'" + text

    return text


def normalize_blank(value):
    if value is None:
        return ""

    return value


def values_equal(left, right):
    left = normalize_blank(left)
    right = normalize_blank(right)

    if left == right:
        return True

    if isinstance(left, (int, float)) and isinstance(right, (int, float)):
        try:
            return abs(float(left) - float(right)) <= VALUE_TOLERANCE
        except Exception:
            return False

    return False


def formula_text(value):
    if isinstance(value, str) and value.startswith("="):
        return value

    return ""


def formulas_equal(left, right):
    return formula_text(left) == formula_text(right)


def matrix_from_com(value, row_count, column_count):
    if row_count <= 0 or column_count <= 0:
        return []

    if row_count == 1 and column_count == 1:
        return [[value]]

    if row_count == 1:
        if isinstance(value, tuple):
            return [list(value)]
        return [[value]]

    if column_count == 1:
        if isinstance(value, tuple):
            result = []

            for item in value:
                if isinstance(item, tuple):
                    result.append([item[0] if item else None])
                else:
                    result.append([item])

            return result

        return [[value]]

    if not isinstance(value, tuple):
        return [[value for _ in range(column_count)] for _ in range(row_count)]

    return [list(row) if isinstance(row, tuple) else [row] for row in value]


def get_matrix_value(matrix, row, column):
    if row < 1 or column < 1:
        return None

    row_index = row - 1
    column_index = column - 1

    if row_index >= len(matrix):
        return None

    if column_index >= len(matrix[row_index]):
        return None

    return matrix[row_index][column_index]

def meaningful_bounds(sheet):
    used = sheet.UsedRange

    first_row = int(used.Row)
    first_column = int(used.Column)
    last_row = int(used.Row + used.Rows.Count - 1)
    last_column = int(used.Column + used.Columns.Count - 1)

    if last_row < 1:
        last_row = 1

    if last_column < 1:
        last_column = 1

    return first_row, first_column, last_row, last_column


def read_sheet_snapshot(sheet):
    first_row, first_column, last_row, last_column = meaningful_bounds(sheet)

    row_count = last_row
    column_count = last_column

    data_range = sheet.Range(sheet.Cells(1, 1), sheet.Cells(row_count, column_count))

    values_raw = data_range.Value2

    try:
        formulas_raw = data_range.Formula2
    except Exception:
        formulas_raw = data_range.Formula

    values = matrix_from_com(values_raw, row_count, column_count)
    formulas = matrix_from_com(formulas_raw, row_count, column_count)

    return {
        "name": str(sheet.Name),
        "rows": row_count,
        "cols": column_count,
        "values": values,
        "formulas": formulas,
        "used_first_row": first_row,
        "used_first_col": first_column,
        "used_last_row": last_row,
        "used_last_col": last_column,
    }


def read_workbook_snapshot(workbook):
    sheets = {}
    order = []

    for sheet in workbook.Worksheets:
        snapshot = read_sheet_snapshot(sheet)
        sheets[snapshot["name"]] = snapshot
        order.append(snapshot["name"])

    return {
        "path": os.path.abspath(workbook.FullName),
        "sheets": sheets,
        "order": order,
    }

def selection_connector_count(sheet_snapshot):
    value = get_matrix_value(sheet_snapshot["values"], 5, 3)

    try:
        return max(0, int(value))
    except Exception:
        return 0


def value_from_either(old_sheet, new_sheet, row, column):
    new_value = get_matrix_value(new_sheet["values"], row, column) if new_sheet is not None else None

    if new_value not in (None, ""):
        return new_value

    old_value = get_matrix_value(old_sheet["values"], row, column) if old_sheet is not None else None

    return old_value


def describe_selection_connector_table(old_sheet, new_sheet, row, column):
    if row < SELECTION_CONNECTOR_FIRST_ROW:
        return None

    if column not in SELECTION_CONNECTOR_COLUMNS:
        return None

    old_count = selection_connector_count(old_sheet) if old_sheet is not None else 0
    new_count = selection_connector_count(new_sheet) if new_sheet is not None else 0
    connector_count = max(old_count, new_count)

    if connector_count <= 0:
        return None

    last_row = SELECTION_CONNECTOR_FIRST_ROW + connector_count - 1

    if row > last_row:
        return None

    block_number = row - SELECTION_CONNECTOR_FIRST_ROW + 1
    connector = value_from_either(old_sheet, new_sheet, row, 2)
    field = SELECTION_CONNECTOR_COLUMNS[column]

    return {
        "kind": "Connector Selection",
        "block": block_number,
        "connector": safe_text(connector),
        "pin": "",
        "field": field,
        "location": f"Connector Selection / Block {block_number} / {safe_text(connector)} / {field}",
    }


def block_number_for_column(column):
    if column < BLOCK_FIRST_COLUMN:
        return None

    return ((column - BLOCK_FIRST_COLUMN) // BLOCK_WIDTH) + 1


def block_start_column(block_number):
    return BLOCK_FIRST_COLUMN + ((block_number - 1) * BLOCK_WIDTH)


def describe_block_cell(old_sheet, new_sheet, row, column):
    block_number = block_number_for_column(column)

    if block_number is None:
        return None

    start_column = block_start_column(block_number)
    relative_column = column - start_column

    if relative_column < 0 or relative_column >= BLOCK_WIDTH:
        return None

    connector_column = start_column + BLOCK_CENTER_OFFSET
    connector = value_from_either(old_sheet, new_sheet, BLOCK_CONNECTOR_ROW, connector_column)
    connector_text = safe_text(connector)

    pin = ""
    field = "Block Cell"

    if row == BLOCK_PIN_COUNT_ROW:
        field = "Number of Pins"
    elif row == BLOCK_CONNECTOR_ROW:
        field = "Connector Name"
    elif row == BLOCK_HEADER_ROW:
        field = "Header"
    elif row >= BLOCK_DATA_FIRST_ROW:
        if relative_column == BLOCK_PIN_OFFSET:
            field = "Pin"
        elif relative_column == BLOCK_SIGNAL_OFFSET:
            field = "Local Signal"
        elif relative_column == BLOCK_DIRECTION_OFFSET:
            field = "IN/OUT"
        elif relative_column == 0:
            field = "Left Helper"
        elif relative_column == BLOCK_WIDTH - 1:
            field = "Right Helper"

        pin_value = value_from_either(old_sheet, new_sheet, row, start_column + BLOCK_PIN_OFFSET)

        if pin_value in (None, ""):
            pin = row - BLOCK_DATA_FIRST_ROW + 1
        else:
            pin = pin_value

    location_parts = [f"Block {block_number}"]

    if connector_text:
        location_parts.append(connector_text)

    if pin != "":
        location_parts.append(f"Pin {safe_text(pin)}")

    location_parts.append(field)

    return {
        "kind": "Block",
        "block": block_number,
        "connector": connector_text,
        "pin": safe_text(pin),
        "field": field,
        "location": " / ".join(location_parts),
    }


def generic_row_label(old_sheet, new_sheet, row):
    if row < 1:
        return ""

    label = value_from_either(old_sheet, new_sheet, row, 1)

    return safe_text(label)


def generic_column_label(old_sheet, new_sheet, column):
    if column < 1:
        return ""

    label = value_from_either(old_sheet, new_sheet, 1, column)

    return safe_text(label)


def describe_cell(sheet_name, old_sheet, new_sheet, row, column):
    if sheet_name.lower() == "selection":
        selection_info = describe_selection_connector_table(old_sheet, new_sheet, row, column)

        if selection_info is not None:
            return selection_info

        block_info = describe_block_cell(old_sheet, new_sheet, row, column)

        if block_info is not None:
            return block_info

    block_info = describe_block_cell(old_sheet, new_sheet, row, column)

    if block_info is not None:
        header_value = value_from_either(old_sheet, new_sheet, BLOCK_HEADER_ROW, block_start_column(block_info["block"]) + BLOCK_PIN_OFFSET)
        signal_header = value_from_either(old_sheet, new_sheet, BLOCK_HEADER_ROW, block_start_column(block_info["block"]) + BLOCK_SIGNAL_OFFSET)

        if safe_text(header_value).lower() == "pins" or "signal" in safe_text(signal_header).lower():
            return block_info

    row_label = generic_row_label(old_sheet, new_sheet, row)
    column_label = generic_column_label(old_sheet, new_sheet, column)
    address = cell_address(row, column)

    location = f"{sheet_name} / {address}"

    if row_label:
        location += f" / Row: {row_label}"

    if column_label:
        location += f" / Column: {column_label}"

    return {
        "kind": "Generic",
        "block": "",
        "connector": "",
        "pin": "",
        "field": column_label,
        "location": location,
    }

def classify_diff(old_value, new_value, old_formula, new_formula):
    value_changed = not values_equal(old_value, new_value)
    formula_changed = not formulas_equal(old_formula, new_formula)

    old_formula_text = formula_text(old_formula)
    new_formula_text = formula_text(new_formula)

    if not value_changed and not formula_changed:
        return None

    if old_value in (None, "") and new_value not in (None, "") and not old_formula_text and not new_formula_text:
        return "Added Value"

    if new_value in (None, "") and old_value not in (None, "") and not old_formula_text and not new_formula_text:
        return "Removed Value"

    if formula_changed and value_changed:
        return "Formula + Value"

    if formula_changed:
        return "Formula Only"

    if old_formula_text and old_formula_text == new_formula_text and value_changed:
        return "Same Formula / Different Result"

    return "Value"


def compare_matching_sheet(sheet_name, old_sheet, new_sheet):
    max_rows = max(old_sheet["rows"], new_sheet["rows"])
    max_cols = max(old_sheet["cols"], new_sheet["cols"])

    diffs = []

    for row in range(1, max_rows + 1):
        for column in range(1, max_cols + 1):
            old_value = get_matrix_value(old_sheet["values"], row, column)
            new_value = get_matrix_value(new_sheet["values"], row, column)
            old_formula_raw = get_matrix_value(old_sheet["formulas"], row, column)
            new_formula_raw = get_matrix_value(new_sheet["formulas"], row, column)
            diff_type = classify_diff(old_value, new_value, old_formula_raw, new_formula_raw)

            if diff_type is None:
                continue

            location_info = describe_cell(sheet_name, old_sheet, new_sheet, row, column)

            diffs.append({
                "sheet": sheet_name,
                "row": row,
                "column": column,
                "cell": cell_address(row, column),
                "kind": location_info["kind"],
                "block": location_info["block"],
                "connector": location_info["connector"],
                "pin": location_info["pin"],
                "field": location_info["field"],
                "location": location_info["location"],
                "diff_type": diff_type,
                "old_value": old_value,
                "new_value": new_value,
                "old_formula": formula_text(old_formula_raw),
                "new_formula": formula_text(new_formula_raw),
            })

    return diffs


def compare_workbooks(old_snapshot, new_snapshot):
    old_sheet_names = set(old_snapshot["sheets"].keys())
    new_sheet_names = set(new_snapshot["sheets"].keys())

    added_sheets = sorted(new_sheet_names - old_sheet_names)
    removed_sheets = sorted(old_sheet_names - new_sheet_names)
    common_sheets = [name for name in old_snapshot["order"] if name in new_sheet_names]

    diffs = []

    for sheet_name in common_sheets:
        print("COMPARING SHEET:", sheet_name)
        old_sheet = old_snapshot["sheets"][sheet_name]
        new_sheet = new_snapshot["sheets"][sheet_name]
        diffs.extend(compare_matching_sheet(sheet_name, old_sheet, new_sheet))

    return {
        "diffs": diffs,
        "added_sheets": added_sheets,
        "removed_sheets": removed_sheets,
    }

def build_block_summary(block_diffs):
    groups = {}

    for diff in block_diffs:
        key = (diff["sheet"], diff["block"], diff["connector"])

        if key not in groups:
            groups[key] = {
                "sheet": diff["sheet"],
                "block": diff["block"],
                "connector": diff["connector"],
                "count": 0,
                "pins": set(),
                "fields": set(),
                "types": set(),
            }

        group = groups[key]
        group["count"] += 1

        if diff["pin"] != "":
            group["pins"].add(str(diff["pin"]))

        if diff["field"]:
            group["fields"].add(str(diff["field"]))

        group["types"].add(str(diff["diff_type"]))

    result = []

    def pin_sort(value):
        try:
            return int(value)
        except Exception:
            return 999999

    for key in sorted(groups, key=lambda item: (item[0], int(item[1]) if str(item[1]).isdigit() else 999999, item[2])):
        group = groups[key]
        pins = sorted(group["pins"], key=pin_sort)
        fields = sorted(group["fields"])
        types = sorted(group["types"])

        result.append([
            group["sheet"],
            group["block"],
            group["connector"],
            group["count"],
            ", ".join(pins),
            ", ".join(fields),
            ", ".join(types),
        ])

    return result


def build_sheet_summary(old_snapshot, new_snapshot, comparison):
    rows = []

    all_names = []

    for name in old_snapshot["order"]:
        if name not in all_names:
            all_names.append(name)

    for name in new_snapshot["order"]:
        if name not in all_names:
            all_names.append(name)

    for sheet_name in all_names:
        if sheet_name in comparison["added_sheets"]:
            status = "Added Sheet"
        elif sheet_name in comparison["removed_sheets"]:
            status = "Removed Sheet"
        else:
            status = "Compared"

        sheet_diffs = [diff for diff in comparison["diffs"] if diff["sheet"] == sheet_name]
        block_diffs = [diff for diff in sheet_diffs if diff["kind"] in ("Block", "Connector Selection")]
        generic_diffs = [diff for diff in sheet_diffs if diff["kind"] == "Generic"]

        rows.append([sheet_name, status, len(sheet_diffs), len(block_diffs), len(generic_diffs)])

    return rows

def write_matrix(sheet, start_row, start_col, matrix):
    if not matrix:
        return

    row_count = len(matrix)
    col_count = max(len(row) for row in matrix)

    normalized = []

    for row in matrix:
        new_row = list(row)

        while len(new_row) < col_count:
            new_row.append("")

        normalized.append(tuple(new_row))

    target = sheet.Range(sheet.Cells(start_row, start_col), sheet.Cells(start_row + row_count - 1, start_col + col_count - 1))
    target.Value2 = tuple(normalized)


def style_report_sheet(sheet, column_count, freeze_row=1):
    header = sheet.Range(sheet.Cells(1, 1), sheet.Cells(1, column_count))
    header.Font.Bold = True
    header.Interior.Color = 0xD9EAF7
    header.Borders.LineStyle = 1
    header.AutoFilter()

    used = sheet.UsedRange
    used.VerticalAlignment = -4108
    used.WrapText = False

    sheet.Activate()

    window = sheet.Application.ActiveWindow

    if window is not None:
        window.SplitRow = freeze_row
        window.FreezePanes = True


def set_widths(sheet, widths):
    for column_number, width in enumerate(widths, start=1):
        sheet.Columns(column_number).ColumnWidth = width


def add_report_sheet(workbook, name, headers, rows, widths):
    sheet = workbook.Worksheets.Add(After=workbook.Worksheets(workbook.Worksheets.Count))
    sheet.Name = name

    data = [headers]

    for row in rows[:MAX_REPORT_ROWS_PER_SHEET]:
        data.append(row)

    write_matrix(sheet, 1, 1, data)
    style_report_sheet(sheet, len(headers), 1)
    set_widths(sheet, widths)

    return sheet


def create_diff_report(excel, report_path, old_snapshot, new_snapshot, comparison):
    report_path = os.path.abspath(report_path)
    report_folder = os.path.dirname(report_path)
    os.makedirs(report_folder, exist_ok=True)
    if os.path.isfile(report_path):
        try:
            os.remove(report_path)
        except Exception:
            pass
    workbook = excel.Workbooks.Add()
    actual_saved_path = None
    try:
        while workbook.Worksheets.Count > 1:
            workbook.Worksheets(workbook.Worksheets.Count).Delete()
        summary_sheet = workbook.Worksheets(1)
        summary_sheet.Name = "Summary"
        block_diffs = [diff for diff in comparison["diffs"] if diff["kind"] in ("Block", "Connector Selection")]
        generic_diffs = [diff for diff in comparison["diffs"] if diff["kind"] == "Generic"]
        summary_rows = [["Reference Workbook", old_snapshot["path"]], ["Comparison Workbook", new_snapshot["path"]], ["Total Cell Diffs", len(comparison["diffs"])], ["Block-Maker Diffs", len(block_diffs)], ["Generic Diffs", len(generic_diffs)], ["Added Sheets", ", ".join(comparison["added_sheets"])], ["Removed Sheets", ", ".join(comparison["removed_sheets"])]]
        write_matrix(summary_sheet, 1, 1, [["EXCEL DIFF REPORT", ""]] + summary_rows)
        summary_sheet.Range("A1:B1").Font.Bold = True
        summary_sheet.Range("A1:B1").Interior.Color = 0xD9EAF7
        summary_sheet.Columns(1).ColumnWidth = 24
        summary_sheet.Columns(2).ColumnWidth = 90
        sheet_summary_headers = ["Sheet", "Status", "Total Diffs", "Block-Maker Diffs", "Generic Diffs"]
        sheet_summary_rows = build_sheet_summary(old_snapshot, new_snapshot, comparison)
        write_matrix(summary_sheet, 11, 1, [sheet_summary_headers] + sheet_summary_rows)
        summary_sheet.Range(summary_sheet.Cells(11, 1), summary_sheet.Cells(11, len(sheet_summary_headers))).Font.Bold = True
        summary_sheet.Range(summary_sheet.Cells(11, 1), summary_sheet.Cells(11, len(sheet_summary_headers))).Interior.Color = 0xD9EAF7
        block_summary_headers = ["Sheet", "Block #", "Connector", "Diff Count", "Pins Affected", "Fields Affected", "Diff Types"]
        block_summary_rows = build_block_summary(block_diffs)
        add_report_sheet(workbook, "Block Summary", block_summary_headers, block_summary_rows, [18, 10, 20, 12, 35, 35, 30])
        block_headers = ["Sheet", "Block #", "Connector", "Pin", "Field", "Diff Type", "Location", "Cell", "Old Value", "New Value", "Old Formula", "New Formula"]
        block_rows = []
        for diff in block_diffs:
            block_rows.append([diff["sheet"], diff["block"], diff["connector"], diff["pin"], diff["field"], diff["diff_type"], diff["location"], diff["cell"], report_text(diff["old_value"]), report_text(diff["new_value"]), report_text(diff["old_formula"]), report_text(diff["new_formula"])])
        add_report_sheet(workbook, "Block Diffs", block_headers, block_rows, [16, 9, 20, 9, 22, 28, 65, 10, 35, 35, 45, 45])
        generic_headers = ["Sheet", "Cell", "Row", "Column", "Field/Header", "Diff Type", "Location", "Old Value", "New Value", "Old Formula", "New Formula"]
        generic_rows = []
        for diff in generic_diffs:
            generic_rows.append([diff["sheet"], diff["cell"], diff["row"], diff["column"], diff["field"], diff["diff_type"], diff["location"], report_text(diff["old_value"]), report_text(diff["new_value"]), report_text(diff["old_formula"]), report_text(diff["new_formula"])])
        add_report_sheet(workbook, "Other Diffs", generic_headers, generic_rows, [18, 10, 9, 9, 28, 28, 70, 35, 35, 45, 45])
        sheet_change_headers = ["Change", "Sheet"]
        sheet_change_rows = []
        for name in comparison["added_sheets"]:
            sheet_change_rows.append(["Added", name])
        for name in comparison["removed_sheets"]:
            sheet_change_rows.append(["Removed", name])
        add_report_sheet(workbook, "Sheet Changes", sheet_change_headers, sheet_change_rows, [16, 30])
        summary_sheet.Activate()
        print("SAVING DIFF REPORT TO:", report_path)
        workbook.SaveAs(Filename=report_path, FileFormat=REPORT_FILE_FORMAT_XLSX, CreateBackup=False, AddToMru=False, Local=True)
        workbook.Save()
        actual_saved_path = os.path.abspath(str(workbook.FullName))
        print("EXCEL REPORTED SAVED PATH:", actual_saved_path)
        workbook.Close(SaveChanges=True)
        workbook = None
        time.sleep(0.5)
        if not os.path.isfile(report_path) and actual_saved_path and os.path.isfile(actual_saved_path):
            shutil.copy2(actual_saved_path, report_path)
        if not os.path.isfile(report_path):
            raise Exception(f"Diff report was not created at {report_path}")
        print("DIFF REPORT SAVED:", report_path)
        print("DIFF REPORT SIZE:", os.path.getsize(report_path), "BYTES")
        return report_path
    finally:
        if workbook is not None:
            try:
                workbook.Close(SaveChanges=False)
            except Exception:
                pass
