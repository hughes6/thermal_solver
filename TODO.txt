# Uses openpyxl to manipulate the Excel spreadsheets.
# Note: This tool uses 1-based indexing, not 0-based.
#
# The spreadsheets that run through this depend on these requirements:
#   1) The header row MUST always be row 5 in the spreadsheet!
#   2) The individual cell values must be text and cannot contain spaces or commas.
#   3) Cell A5 MUST contain "LINE NO." for the cable list files.
#   4) Cell A5 MUST contain "WIRE NO." for the wire list files.
#   5) Cell A5 MUST contain ? for the mistery files.
#
# Brad Brookes - (C) BAE Systems, March 2026
import openpyxl
import textwrap
from wire_table import WireTable
import logging
LOGGER_NAME = "ZukenChangeAnalyzerLog"   # Just a name. When used with the getLogger() method,
logger = logging.getLogger(LOGGER_NAME)

###############################################################################
class WorkSheet:
    """ WorkSheet class 'has a' openpyxl Worksheet object. """
    HEADER_ROW = 5      # Sheet's header row. Sheets have 1-based index of rows.
    MIN_COLUMNS = 8     # If less than this, then this sheet is invalid.
    MAX_COLUMNS = 15    # Arbitrary number.
    MAX_ROWS = 10000     # Arbitrary value to prevent runaway

    def __init__(self, cfg_tables, detection_mode="Default Detection"):
        self.cfg_tables = cfg_tables
        self.cols = 0
        self.rows = 0
        self.mode = detection_mode
        self.table_type = ''

    def loadWorksheet(self, work_sheet, file_path):
        """ Load the WorkSheet class object from an openpyxl Worksheet object that
        was read from a file.
        """
        self.ws = work_sheet
        self.file_path = file_path

    def generateRowTable(self):
        """ This must be called after the initial load AND after any updates to a worksheet.
        Get the worksheet from the workbook, determine its type, create a new
        row table with sheet row 5 (headers) being the first row enter, dict[0].
        The worksheet column and row  indexes are 1-based, not 0-based. Starting with
        the header row, insert all the rows into the row table. """
        self._getSheetType()
        self._calcSheetSize()    # Scan and get number of rows in sheet.
        self._loadRowTable()

    def _getSheetType(self):
        """ Base the sheet type on the text in 'B1' of the title row.
            WARNING: This value may change! I was using A5, but it changed.
            (They other way was to count columns.)
        """
        logging.info(f"Loading workbook: {self.file_path}")
        # commented code below because manual sheet detection added for this reason
        # if self.ws['B1'] == None or self.ws['B1'].value == None:
        #     self.table_type = 'NONE'
        #     logging.error('Value in cell B1 is None. Unknown type of spreadsheet!')
        #     raise ValueError("Unknown type of List Spreadsheet!")
        
        if self.mode == "Default Detection":
            # Print-label exports can have a blank B1. Treat titles as optional
            # and use the existing print-label signature before checking B1.
            title = str(self.ws['B1'].value or '')
            print_title = str(self.ws['A1'].value or '')
            if 'Cable' in title:
                self.table_type = 'cable'
            elif 'Wire' in title:
                self.table_type = 'wire'
            elif 'End' in title:
                self.table_type = 'end2end'
            elif 'PRINT' in print_title.upper() or self.ws['X1'].value is not None:
                self.table_type = 'print'
            else:
                raise ValueError(
                    "Cannot detect the Zuken report type from this workbook. "
                    "Choose Wire List, Cable Table, Print Label, or End to End explicitly."
                )
        elif self.mode == "Wire List":
            self.table_type = 'wire'
        elif self.mode == "Cable Table":
            self.table_type = 'cable'
        elif self.mode == "Print Label":
            self.table_type = 'print'
        elif self.mode == "End to End":
            self.table_type = 'end2end'
        else:
            print(self.mode)
            self.table_type = 'UNKNOWN'
            raise ValueError("Value in cell [B1] does not match search strings. Unknown type of List Spreadsheet!")
        logging.info(f"Table type: {self.table_type}")

    def _calcSheetSize(self):
        """ Instead of calculating the size, we will get the max dimenstion from the openpyxl.Worksheet class."""
        self.cols = self.ws.max_column
        self.rows = self.ws.max_row

    def _loadRowTable(self):
        """ Starting with the header row, insert all the rows into the row table. """
        logging.info(f"  |--> Processing workbook {self.file_path}, which has {self.rows} rows.")
        # self.table = WireTable(self.cfg_tables[self.table_type], self.table_type)
        self.table = WireTable(self.cfg_tables, self.table_type)
        # print('==============')
        # print(self.cfg_tables[self.table_type])
        for row in self.ws.iter_rows(min_row=WorkSheet.HEADER_ROW,
                                     max_row=self.rows,
                                     max_col=self.cols,
                                     values_only=True):
            self.table.insert_wire(row) # Treat everything like a wire.
        # self.table.print_table_info()

    def col_row_to_string(self, col, row):
        # Convert the col and row 0-based index to the Excel A1 to Z99 format.
        if not (0 <= col <= 25):
            raise ValueError("Index must be between 0 and 25 for the A to Z conversion.")
        return chr(ord('A') + col) + chr(ord('1') + row)

    def addSheetRow(self, row_data):
        self.ws.append(row_data)

    def clearOutSheet(self):
        # Use this to start a diff sheet using the first 5 rows with header formatting
        # and other information from the rev-2 sheet being compared to the old rev-1 sheet.
        # That gives me a copy of the rev-2 to use as a template.
        # Then clear the data out so that we can then write in the difference data.
        self.ws.delete_rows(WorkSheet.HEADER_ROW+1, self.rows)
        self._calcSheetSize()
        data = self.ws['B1'].value
        if data == None:    # If there is not a sheet Title,
            data = 'No Title'
        data = data + ' - CHANGE LOG'
        self.ws['B1'].value = data
        cell_index_str = self.col_row_to_string(self.cols-1, WorkSheet.HEADER_ROW)

    def getSheetRow(self, row_num):
        return self.ws[row_num]
    
    def getOpenpyxlWs(self):
        return self.ws

    def getSheetRowValues(self, row_num):
        return self.ws[row_num].values

    def getRowTable(self):
        return self.table

    def getTitle(self):
        return self.ws.title

    def setTitle(self, title):
        self.ws.title = title

    def printColumnCountError(self):
        err_string = textwrap.dedent('''\
            ===========================================================================
            ERROR: Invalid Excel Spreadsheet for comparison!
            ===========================================================================
            Each spreadsheet MUST have a header row in the fifth row (1-based index) of
            the spreadsheet that defines the number of columns of data it contains. If
            it is not in that location, this tool is unable to determine the column
            count correctly. If the header row must be moved, please call Brad Brookes
            to update the tool to work with the new format.
            ---------------------------------------------------------------------------
            ''')
        print(err_string)

    #--------------------------------------------------------------------------
    def saveWorksheet(self):
        """ Write the workbook to the file."""
        self.workbook.save(self.filename)


    # -=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=-=--==--=-=-=
    #
    # Debugging stuff:
    #
    #
    #--------------------------------------------------------------------------
    def print_table(self):
        # self.table.printTable()
        self.table.print_table_info()

    #--------------------------------------------------------------------------
    def print_table_row(self, row_key ):
        row = self.table.getRow(row_key)
        print(f'Table row[{row_key:>30}] = {row}')

    #--------------------------------------------------------------------------
    def print_sheet_row(self, row_key ):
        """ Only use this for debugging. This prints the Speadsheet row info. """
        print(f'\n============= Sheet row {row_key + WorkSheet.HEADER_ROW} =========')
        row = self.ws[row_key + WorkSheet.HEADER_ROW]
        # print(row)
        self.print_sheet_row_details(row)

    #--------------------------------------------------------------------------
    def print_sheet_row_details(self, row_data):
        # print('-----------  ROW %d  -----------' % row_data[0].row)
        # format_str = 'Cell: Coord = {:<8}, type = {:20}, data_type = {}, cell_encoding = {}, cell_value = {}'
        format_str = '{:5} {:^8}, {:15}, {:10}, {:10}, {:20}'
        # print('row type: ', type(row_data))
        # print('row_data raw =', row_data)
        print(format_str.format(' ', 'Coord','Type','Data type', 'Encoding', 'Value'))
        print(format_str.format('', '------', '------', '------', '------', '------', '------'))
        for cell in row_data:
            cell_coord = cell.coordinate
            cell_encoding = 'none'
            if type(cell) == openpyxl.cell.cell.MergedCell:
                cell_type = 'Merged cell'
            elif type(cell) == openpyxl.cell.cell.Cell:
                cell_type = 'Cell'
                cell_encoding = cell.encoding
            else:
                cell_type = 'Unknown'
            cell_data_type = cell.data_type
            if cell.value == None:
                cell_value = 'none'
            else:
                cell_value = cell.value
            # print('Cell: Coord = {0:<8}, type = {1:20}, data_type = {2}, cell_encoding = {3}, cell_value = {4}'.format(
            print(format_str.format( 'Cell:', cell_coord, cell_type, cell_data_type, cell_encoding, cell_value ))
