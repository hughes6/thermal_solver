# VALUE HELPERS
#====================================================================================================================================
def clean_excel_value(value):
    if value is None:
        return ""

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):
        if value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def normalize_pin_name(value):
    return clean_excel_value(value).upper()


def normalize_connector_token(value):
    text = clean_excel_value(value).upper().replace(" ", "")

    if text == "":
        return ""

    root_prefix = CONNECTOR_ROOT_NAME.upper() + "."

    if text.startswith(root_prefix):
        text = text[len(root_prefix):]

    if "#" in text:
        text = text.split("#", 1)[0]

    match = re.fullmatch(r"P?0*(\d+)([A-Z]*)", text)

    if match:
        number = int(match.group(1))
        suffix = match.group(2)
        return f"P{number}{suffix}"

    return text


def connector_base_name(connector_token):
    connector = normalize_connector_token(connector_token)

    if connector == "":
        return ""

    return f"{CONNECTOR_ROOT_NAME}.{connector}"


def cell_has_bar(value):
    return "|" in clean_excel_value(value)


#====================================================================================================================================
# YES / NO SETTING
#====================================================================================================================================
def parse_yes_no(value, setting_name):
    text = clean_excel_value(value).upper()

    if text == "":
        print("WARNING:", setting_name, "IS BLANK - DEFAULTING TO YES")
        return True

    if text in ["YES", "Y", "TRUE", "1"]:
        return True

    if text in ["NO", "N", "FALSE", "0"]:
        return False

    raise Exception(f"{setting_name} must contain YES or NO. Current value: {value}")

def get_last_data_row(ws):
    columns = [LEFT_OUTER_SHIELD_COL, LEFT_INNER_SHIELD_COL, LEFT_TWIST_A_COL, LEFT_TWIST_B_COL, LEFT_CONNECTOR_COL, PIN_1_COL, PIN_2_COL, RIGHT_CONNECTOR_COL, RIGHT_TWIST_A_COL, RIGHT_TWIST_B_COL, RIGHT_INNER_SHIELD_COL, RIGHT_OUTER_SHIELD_COL]

    last_row = DATA_FIRST_ROW - 1

    for row in range(DATA_FIRST_ROW, scan_end + 1):
        for col in columns:

            if value != "":
                last_row = row

    return last_row

        }

        rows.append(row_info)

    return rows


#====================================================================================================================================
# SPLIT INTO CONNECTOR SECTIONS
#====================================================================================================================================
def split_into_sections(rows):
    sections = []
    current = []

    for row in rows:
        if not current:
            current = [row]
            continue

        previous = current[-1]
        connector_changed = row["connector"] != previous["connector"]
        row_gap = row["excel_row"] != previous["excel_row"] + 1

        if connector_changed or row_gap:
            sections.append(current)
            current = [row]
        else:
            current.append(row)

    if current:
        sections.append(current)

    return sections


#====================================================================================================================================
# CONTIGUOUS MARKED RANGES
#====================================================================================================================================
def true_ranges(rows, key):
    ranges = []
    start = None

    for index, row in enumerate(rows):
        marked = bool(row[key])

        if marked and start is None:
            start = index

        if not marked and start is not None:
            ranges.append((start, index - 1))
            start = None

    if start is not None:
        ranges.append((start, len(rows) - 1))

    return ranges


def range_pins(rows, start, end):
    pins = []

    for index in range(start, end + 1):
        pins.append(rows[index]["pin"])

    return pins


def range_excel_rows(rows, start, end):
    excel_rows = []

    for index in range(start, end + 1):
        excel_rows.append(rows[index]["excel_row"])

    return excel_rows


#====================================================================================================================================
# BUILD TWIST / SHIELD JOBS
#====================================================================================================================================
def build_section_jobs(rows):
    jobs = []
    warnings = []

    if not rows:
        return jobs, warnings

    side = rows[0]["side"]
    connector = rows[0]["connector"]

    if connector == "":
        warnings.append(f"{side}: rows {rows[0]['excel_row']}-{rows[-1]['excel_row']} have pins but no connector")
        return jobs, warnings

    conflicts = []

    for row in rows:
        if row["twist_a"] and row["twist_b"]:
            conflicts.append(row["excel_row"])

    if conflicts:
        message = f"{side} {connector}: same wire marked in both twist columns on Excel rows {conflicts}"

        if STRICT_TWIST_VALIDATION:
            raise Exception(message)

        warnings.append(message)

    twist_jobs = []

    for key, twist_column in [("twist_a", "A"), ("twist_b", "B")]:
        ranges = true_ranges(rows, key)

        for start, end in ranges:
            pins = range_pins(rows, start, end)
            excel_rows = range_excel_rows(rows, start, end)

            if len(pins) < 2:
                warnings.append(f"{side} {connector}: twist {twist_column} only contains one wire on Excel row {excel_rows[0]}")
                continue

            job = {
                "type": "twist",
                "side": side,
                "connector": connector,
                "twist_column": twist_column,
                "start": start,
                "end": end,
                "pins": pins,
                "excel_rows": excel_rows,
                "symbol_name": TWIST_SYMBOL_NAME,
                "direct_outer_first": False
            }

            twist_jobs.append(job)
            jobs.append(job)

    inner_consumed = set()

    for twist_job in twist_jobs:
        indexes = list(range(twist_job["start"], twist_job["end"] + 1))
        fully_inner_shielded = all(rows[index]["inner_shield"] for index in indexes)

        if fully_inner_shielded:
            shield_job = {
                "type": "shield_inner",
                "side": side,
                "connector": connector,
                "twist_column": twist_job["twist_column"],
                "pins": list(twist_job["pins"]),
                "excel_rows": list(twist_job["excel_rows"]),
                "symbol_name": SHIELD_SYMBOL_NAME,
                "direct_outer_first": False
            }

            jobs.append(shield_job)

            for index in indexes:
                inner_consumed.add(index)

    remaining = []

    for index, row in enumerate(rows):
        remaining.append(bool(row["inner_shield"] and index not in inner_consumed))

    start = None

    for index, marked in enumerate(remaining):
        if marked and start is None:
            start = index

        if not marked and start is not None:
            end = index - 1
            job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME, "direct_outer_first": False}
            jobs.append(job)
            start = None

    if start is not None:
        end = len(rows) - 1
        job = {"type": "shield_inner", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME, "direct_outer_first": False}
        jobs.append(job)

    outer_ranges = true_ranges(rows, "outer_shield")

    for start, end in outer_ranges:
        job = {"type": "shield_outer", "side": side, "connector": connector, "twist_column": None, "pins": range_pins(rows, start, end), "excel_rows": range_excel_rows(rows, start, end), "symbol_name": SHIELD_SYMBOL_NAME, "direct_outer_first": False}
        jobs.append(job)

    return jobs, warnings


#====================================================================================================================================
# E3 ID EXTRACTOR
#====================================================================================================================================
def extract_e3_ids(result):
    if result is None:
        return []

    if isinstance(result, int):
        if result > 0:
            return [result]

        return []

    if isinstance(result, list):
        output = []

        for value in result:
            if value is not None:
                try:
                    if int(value) > 0:
                        output.append(int(value))
                except Exception:
                    pass

        return output

    if not isinstance(result, tuple):
        return []

    if len(result) == 0:
        return []

    if len(result) >= 2:
        values = result[1]

        if values is None:
            return []

        if isinstance(values, int):
            if values > 0:
                return [values]

            return []

        try:
            return [int(value) for value in values if value is not None and int(value) > 0]
        except Exception:
            pass

    output = []

    for value in result:
        try:
            if int(value) > 0:
                output.append(int(value))
        except Exception:
            pass

    return output

# READ PIN SCHEMATIC LOCATION
#====================================================================================================================================
def get_pin_location(pin, pin_id):
    pin.SetId(pin_id)

    xloc = [0.0]
    yloc = [0.0]
    grid = [""]
    column = [""]
    row = [""]

    try:
        result = pin.GetSchemaLocation(xloc, yloc, grid, column, row)
    except Exception:
        return None

    if not isinstance(result, tuple):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None

    if sheet_id <= 0:
        return None

    return {"sheet_id": sheet_id, "x": x, "y": y}

# LOAD DEVICE PIN MAP
#====================================================================================================================================
def load_device_pin_map(dev, pin, device_id):
    dev.SetId(device_id)

    try:
        info = dev.GetAllPinIds(None)
    except Exception:
        return {}

    pin_ids = extract_e3_ids(info)
    pin_map = {}

    for pin_id in pin_ids:
        pin.SetId(pin_id)

        try:
            pin_name = normalize_pin_name(pin.GetName())
        except Exception:
            continue

        location = get_pin_location(pin, pin_id)

        if pin_name not in pin_map:
            pin_map[pin_name] = []

        if location is not None:
            record = {"pin_id": pin_id, "sheet_id": location["sheet_id"], "x": location["x"], "y": location["y"]}
            pin_map[pin_name].append(record)

    return pin_map

# GET SAME NAME ASSEMBLY DEVICES
#====================================================================================================================================
def get_same_name_assembly_devices(dev, base_device_id, expected_name):
    found_ids = []

    dev.SetId(base_device_id)

    try:
        assembly_id = dev.GetAssemblyId()
    except Exception:
        return found_ids

    if assembly_id is None:
        return found_ids

    try:
        assembly_id = int(assembly_id)
    except Exception:
        return found_ids

    if assembly_id <= 0:
        return found_ids

    dev.SetId(assembly_id)

    try:
        device_info = dev.GetDeviceIds(None)
    except Exception:
        return found_ids

    device_ids = extract_e3_ids(device_info)

    for device_id in device_ids:
        dev.SetId(device_id)

        try:
            name = clean_excel_value(dev.GetName())
        except Exception:
            continue

        if name.upper() == expected_name.upper():
            found_ids.append(device_id)

    return found_ids


#====================================================================================================================================
# COLLECT CONNECTOR CANDIDATES
#====================================================================================================================================
def collect_connector_candidate_ids(connector_token, all_devices, dev):
    expected_name = connector_base_name(connector_token)
    base_records = find_base_connector_devices(connector_token, all_devices)

    if not base_records:
        raise Exception(f"Could not find base E3 connector {expected_name}")

    candidate_sources = {}

    for record in base_records:
        base_id = record["device_id"]

        candidate_sources[base_id] = "BASE"

        view_ids = get_device_view_ids(dev, base_id)

        for view_id in view_ids:
            candidate_sources[view_id] = "VIEW"

        child_ids = get_child_device_ids(dev, base_id)

        for child_id in child_ids:
            if child_id not in candidate_sources:
                candidate_sources[child_id] = "CHILD"

        assembly_ids = get_same_name_assembly_devices(dev, base_id, expected_name)

        for assembly_device_id in assembly_ids:
            if assembly_device_id not in candidate_sources:
                candidate_sources[assembly_device_id] = "ASSEMBLY"

    return candidate_sources


#====================================================================================================================================
# ANALYZE CANDIDATE VIEW
#====================================================================================================================================
def analyze_candidate(required_pins, pin_map):
    sheets = {}

    for pin_name in required_pins:
        locations = pin_map.get(pin_name, [])

        for location in locations:
            sheet_id = location["sheet_id"]

            if sheet_id not in sheets:
                sheets[sheet_id] = {"pins": set(), "xs": [], "ys": []}

            sheets[sheet_id]["pins"].add(pin_name)
            sheets[sheet_id]["xs"].append(location["x"])
            sheets[sheet_id]["ys"].append(location["y"])

    if not sheets:
        return None

    results = []

    for sheet_id, data in sheets.items():
        placed_count = len(data["pins"])
        unique_x_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["xs"])
        unique_y_values = set(round(value, COORDINATE_ROUND_DIGITS) for value in data["ys"])
        unique_x_count = len(unique_x_values)
        unique_y_count = len(unique_y_values)
        x_span = max(data["xs"]) - min(data["xs"]) if data["xs"] else 0.0
        y_span = max(data["ys"]) - min(data["ys"]) if data["ys"] else 0.0
        has_real_y_spread = y_span >= SCHEMATIC_MIN_Y_SPAN and unique_y_count > 1
        result = {"sheet_id": sheet_id, "placed_count": placed_count, "unique_x_count": unique_x_count, "unique_y_count": unique_y_count, "x_span": x_span, "y_span": y_span, "has_real_y_spread": has_real_y_spread}

        results.append(result)

    results.sort(key=lambda item: (item["has_real_y_spread"], item["placed_count"], item["unique_y_count"], item["y_span"]), reverse=True)

    return results[0]


#====================================================================================================================================
# RESOLVE REAL SCHEMATIC VIEW
#====================================================================================================================================
def resolve_connector_schematic_view(connector_token, required_pins, all_devices, dev, pin, cache):
    expected_name = connector_base_name(connector_token)
    candidate_sources = collect_connector_candidate_ids(connector_token, all_devices, dev)
    candidates = []

    print("")
    print("============================================================")
    print("E3 CANDIDATES FOR:", expected_name)
    print("============================================================")

    for candidate_id, source in candidate_sources.items():
        dev.SetId(candidate_id)

        try:
            candidate_name = clean_excel_value(dev.GetName())
        except Exception:
            candidate_name = "UNKNOWN"

        if candidate_id not in cache:
            try:
                cache[candidate_id] = load_device_pin_map(dev, pin, candidate_id)
            except Exception as error:
                print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "PIN MAP FAILED:", repr(error))
                continue

        pin_map = cache[candidate_id]
        analysis = analyze_candidate(required_pins, pin_map)

        if analysis is None:
            print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "NO PLACED REQUIRED PINS")
            continue

        print("CANDIDATE ID:", candidate_id, "SOURCE:", source, "NAME:", candidate_name, "SHEET:", analysis["sheet_id"], "PLACED:", analysis["placed_count"], "UNIQUE X:", analysis["unique_x_count"], "UNIQUE Y:", analysis["unique_y_count"], "X SPAN:", analysis["x_span"], "Y SPAN:", analysis["y_span"])

        candidate = {"device_id": candidate_id, "name": candidate_name, "source": source, "pin_map": pin_map, "analysis": analysis}

        candidates.append(candidate)

    if not candidates:
        raise Exception(f"No usable E3 device/view candidates found for {expected_name}")

    candidates.sort(key=lambda item: (item["analysis"]["has_real_y_spread"], item["analysis"]["placed_count"], item["analysis"]["unique_y_count"], item["analysis"]["y_span"]), reverse=True)

    best = candidates[0]

    if not best["analysis"]["has_real_y_spread"]:
        raise Exception(f"Could not locate the split schematic view for {expected_name}")

    sheet_id = best["analysis"]["sheet_id"]
    missing_pins = []

    for pin_name in required_pins:
        locations = best["pin_map"].get(pin_name, [])
        found_on_sheet = False

        for location in locations:
            if location["sheet_id"] == sheet_id:
                found_on_sheet = True
                break

        if not found_on_sheet:
            missing_pins.append(pin_name)

    if missing_pins:
        raise Exception(f"Selected schematic view for {expected_name} is missing pins on sheet {sheet_id}: {missing_pins}")

    print("")
    print("SELECTED SCHEMATIC VIEW ID:", best["device_id"])
    print("SELECTED SOURCE:", best["source"])
    print("SELECTED NAME:", best["name"])
    print("SELECTED SHEET:", sheet_id)
    print("SELECTED PLACED PIN COUNT:", best["analysis"]["placed_count"])
    print("SELECTED UNIQUE Y COUNT:", best["analysis"]["unique_y_count"])
    print("SELECTED Y SPAN:", best["analysis"]["y_span"])
    print("============================================================")

    record = {"device_id": best["device_id"], "name": best["name"]}

    return record, best["pin_map"], sheet_id

#====================================================================================================================================
# JOB X OFFSET
#====================================================================================================================================
def job_offset(job, twist_offset_settings):
    offset_enabled = twist_offset_settings[job["side"]]

    if job["type"] == "twist":
        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if job["type"] == "shield_inner":
        if job["twist_column"] is None:
            return INNER_SHIELD_ONLY_OFFSET

        if not offset_enabled:
            return OFFSET_NO_TWIST_LINE

        if job["twist_column"] == "A":
            return OFFSET_YES_TWIST_A

        return OFFSET_YES_TWIST_B

    if not offset_enabled:
        return OFFSET_NO_OUTER_SHIELD

    return OFFSET_YES_OUTER_SHIELD


#====================================================================================================================================
# DRAW ORDER
#
# SPECIAL CASE:
#
# Outer-only shield around twists:
#     outer shield first
#     twists second
#
# NORMAL CASE:
#
#     twists first
#     inner shields second
#     outer shields last
#====================================================================================================================================
def job_priority(job):
    if job.get("direct_outer_first", False):
        return 0

    if job["type"] == "twist":
        return 1

    if job["type"] == "shield_inner":
        return 2

    return 3


#====================================================================================================================================
# CALCULATE FINAL SYMBOL PLACEMENT
#====================================================================================================================================
def calculate_placement(job, pin_map, sheet_id, twist_offset_settings):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} has no schematic location on sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    if top_y - bottom_y < SCHEMATIC_MIN_Y_SPAN and len(job["pins"]) > 1:
        raise Exception(f"{job['side']} {job['connector']} {job['type']}: pins {job['pins']} all have effectively the same Y coordinate")

    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job, twist_offset_settings)
    x = anchor_x + direction * offset

    if job["type"] == "twist":
        y = bottom_y - TWIST_BOTTOM_OVERLAP
        length = None
    else:
        y = bottom_y - SHIELD_BOTTOM_OVERLAP
        length = top_y - bottom_y + SHIELD_BOTTOM_OVERLAP + SHIELD_TOP_OVERLAP

        if length < MIN_SYMBOL_LENGTH:
            length = MIN_SYMBOL_LENGTH

    placement = {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y,
        "direction": direction,
        "offset_enabled": twist_offset_settings[job["side"]]
    }

    return placement


#====================================================================================================================================
# FLOAT RANGE
#====================================================================================================================================
def float_range(start, stop, step):
    values = []
    value = float(start)

    while value <= float(stop) + 0.000001:
        values.append(round(value, 6))
        value += float(step)

    return values

# DELETE SCRIPT-CREATED SHIELD

def delete_staged_shield(sym, symbol_id):
    if symbol_id is None:
        return

    print("DELETING FAILED SCRIPT-CREATED SHIELD:", symbol_id)

    try:

        delete_result = sym.Delete()
        print("FAILED SHIELD DELETE RESULT:", delete_result)
    except Exception as error:
        print("FAILED SHIELD DELETE EXCEPTION:", repr(error))

    time.sleep(PLACEMENT_DELAY_SECONDS)

    try:
        verify_result = sym.SetId(symbol_id)
        print("FAILED SHIELD DELETE VERIFY SETID:", verify_result)
    except Exception as error:
        print("FAILED SHIELD DELETE VERIFY EXCEPTION:", repr(error))


#====================================================================================================================================
# TRY TO SIZE ONE SHIELD
#====================================================================================================================================
def try_size_shield(sym, symbol_id, requested_length):
    sym.SetId(symbol_id)

    before = get_display_length_safe(sym, symbol_id)

    print("SHIELD LENGTH BEFORE:", before)

    if before is not None and abs(before - requested_length) <= 0.001:
        print("SHIELD ALREADY HAS REQUESTED LENGTH")
        return True

    try:
        stretch_result = sym.SetDisplayLength(requested_length)
    except Exception as error:
        stretch_result = None
        print("SHIELD STRETCH EXCEPTION:", repr(error))

    print("SHIELD STRETCH RESULT:", stretch_result)

    time.sleep(STRETCH_DELAY_SECONDS)

    after = get_display_length_safe(sym, symbol_id)

    print("SHIELD REQUESTED LENGTH:", requested_length)
    print("SHIELD LENGTH AFTER:", after)

    if after is None:
        return False

    if abs(after - requested_length) <= 0.001:
        return True

    return False


#====================================================================================================================================
# BUILD DENSE SHIELD STAGING LOCATIONS
#====================================================================================================================================
def build_shield_stage_candidates(final_y, requested_length):
    x_values = float_range(SHIELD_STAGE_X_MIN, SHIELD_STAGE_X_MAX, SHIELD_STAGE_X_STEP)

    minimum_insertion_y = SHIELD_STAGE_Y_MIN + SHIELD_STAGE_EDGE_MARGIN
    maximum_insertion_y = SHIELD_STAGE_Y_MAX - requested_length - SHIELD_STAGE_EDGE_MARGIN

    candidates = []
    used = set()

    if final_y >= minimum_insertion_y and final_y <= maximum_insertion_y:
        for stage_x in x_values:
            key = (round(stage_x, 6), round(final_y, 6))

            if key not in used:
                candidates.append((round(stage_x, 6), round(final_y, 6)))
                used.add(key)

    if maximum_insertion_y < minimum_insertion_y:
        return candidates

    stage_y = maximum_insertion_y

    while stage_y >= minimum_insertion_y - 0.000001:
        rounded_y = round(stage_y, 6)

        for stage_x in x_values:
            key = (round(stage_x, 6), rounded_y)

            if key not in used:
                candidates.append((round(stage_x, 6), rounded_y))
                used.add(key)

        stage_y -= SHIELD_STAGE_Y_STEP

    return candidates

#====================================================================================================================================
# DIRECT OUTER SHIELD PLACEMENT
#
# Used only when:
#
# - twists exist
# - outer shield exists
# - no inner shields exist
#
# The shield is placed on the actual conductor region FIRST.
#
# Then SetDisplayLength() is called while the shield has actual conductor
# context rather than sitting in empty staging space.
#====================================================================================================================================
def place_outer_shield_direct(sym, job, placement):
    requested_length = float(placement["length"])
    final_x = float(placement["x"])
    final_y = float(placement["y"])

    print("")
    print("============================================================")
    print("DIRECT OUTER SHIELD MODE")
    print("============================================================")
    print("FINAL X:", final_x)
    print("FINAL Y:", final_y)
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("REQUESTED LENGTH:", requested_length)

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("SHIELD LOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not load library shield {job['symbol_name']}")

    placed_id = sym.Place(placement["sheet_id"], final_x, final_y, 0, 1.0, 0)

    print("DIRECT SHIELD PLACE RESULT:", placed_id)

    if placed_id == 0:
        raise Exception(f"Could not place outer shield directly at X={final_x} Y={final_y}")

    time.sleep(PLACEMENT_DELAY_SECONDS)

    sym.SetId(placed_id)

    try:
        is_shield = sym.IsShield()
    except Exception:
        is_shield = None

    try:
        is_connected = sym.IsConnected()
    except Exception:
        is_connected = None

    try:
        initial_length = float(sym.GetDisplayLength())
    except Exception:
        initial_length = None

    print("DIRECT SHIELD ID:", placed_id)
    print("DIRECT SHIELD IS SHIELD:", is_shield)
    print("DIRECT SHIELD IS CONNECTED:", is_connected)
    print("DIRECT SHIELD INITIAL LENGTH:", initial_length)

    if initial_length is not None and abs(initial_length - requested_length) <= 0.001:
        print("DIRECT OUTER SHIELD ALREADY CORRECT LENGTH")
        return placed_id

    for attempt in range(1, DIRECT_OUTER_STRETCH_ATTEMPTS + 1):
        sym.SetId(placed_id)

        try:
            length_before = float(sym.GetDisplayLength())
        except Exception:
            length_before = None

        print("")
        print("DIRECT SHIELD STRETCH ATTEMPT:", attempt, "OF", DIRECT_OUTER_STRETCH_ATTEMPTS)
        print("DIRECT SHIELD LENGTH BEFORE:", length_before)

        try:
            stretch_result = sym.SetDisplayLength(requested_length)
        except Exception as error:
            stretch_result = None
            print("DIRECT SHIELD STRETCH EXCEPTION:", repr(error))

        print("DIRECT SHIELD SET DISPLAY LENGTH RESULT:", stretch_result)

        time.sleep(STRETCH_DELAY_SECONDS)

        sym.SetId(placed_id)

        try:
            length_after = float(sym.GetDisplayLength())
        except Exception:
            length_after = None

        print("DIRECT SHIELD REQUESTED LENGTH:", requested_length)
        print("DIRECT SHIELD LENGTH AFTER:", length_after)

        if length_after is not None and abs(length_after - requested_length) <= 0.001:
            print("DIRECT OUTER SHIELD STRETCH SUCCESSFUL")
            print("============================================================")
            return placed_id

        time.sleep(PLACEMENT_DELAY_SECONDS)

    print("DIRECT OUTER SHIELD COULD NOT BE STRETCHED")

    delete_staged_shield(sym, placed_id)

    raise Exception(f"Outer shield placed over conductors but could not resize from {initial_length} to {requested_length}")


#====================================================================================================================================
# NORMAL STAGED SHIELD PLACEMENT
#
# This remains for:
#
# - inner shields
# - normal shield arrangements
# - outer shields where inner shields also exist
#
# Every staging attempt uses a fresh Shield_Demo.
#====================================================================================================================================
def place_shield_staged(sym, job, placement):
    requested_length = float(placement["length"])
    final_x = float(placement["x"])
    final_y = float(placement["y"])

    print("SHIELD FINAL X:", final_x)
    print("SHIELD FINAL Y:", final_y)
    print("SHIELD REQUESTED LENGTH:", requested_length)

    candidates = build_shield_stage_candidates(final_y, requested_length)

    print("SHIELD STAGING CANDIDATE COUNT:", len(candidates))

    if not candidates:
        raise Exception(f"No staging coordinates can physically fit shield length {requested_length}")

    for candidate_index, candidate in enumerate(candidates):
        stage_x = candidate[0]
        stage_y = candidate[1]

        print("")
        print("------------------------------------------------------------")
        print("SHIELD STAGING ATTEMPT:", candidate_index + 1, "OF", len(candidates))
        print("STAGE X:", stage_x)
        print("STAGE Y:", stage_y)
        print("STAGED TOP Y:", stage_y + requested_length)

        load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

        print("SHIELD LOAD RESULT:", load_result)

        if load_result == 0:
            raise Exception(f"Could not load library shield {job['symbol_name']}")

        staged_id = sym.Place(placement["sheet_id"], stage_x, stage_y, 0, 1.0, 0)

        print("SHIELD STAGE PLACE RESULT:", staged_id)

        if staged_id == 0:
            time.sleep(PLACEMENT_DELAY_SECONDS)
            continue

        time.sleep(PLACEMENT_DELAY_SECONDS)

        print("FRESH STAGED SHIELD ID:", staged_id)

        staged_length_before = get_display_length_safe(sym, staged_id)

        print("FRESH STAGED SHIELD LENGTH:", staged_length_before)

        size_success = try_size_shield(sym, staged_id, requested_length)

        if not size_success:
            print("STAGING POSITION REJECTED RESIZE")
            delete_staged_shield(sym, staged_id)
            continue

        staged_length_after = get_display_length_safe(sym, staged_id)

        print("SHIELD RESIZED SUCCESSFULLY")
        print("SUCCESSFUL STAGE X:", stage_x)
        print("SUCCESSFUL STAGE Y:", stage_y)
        print("SUCCESSFUL STAGED LENGTH:", staged_length_after)

        time.sleep(MOVE_DELAY_SECONDS)

        sym.SetId(staged_id)

        final_move_result = sym.Place(placement["sheet_id"], final_x, final_y, 0, 1.0, 0)

        print("SHIELD FINAL MOVE RESULT:", final_move_result)

        if final_move_result == 0:
            print("FINAL MOVE FAILED FOR THIS STAGED SHIELD")
            delete_staged_shield(sym, staged_id)
            continue

        final_id = final_move_result

        time.sleep(PLACEMENT_DELAY_SECONDS)

        final_length = get_display_length_safe(sym, final_id)

        print("SHIELD FINAL ID:", final_id)
        print("SHIELD FINAL X:", final_x)
        print("SHIELD FINAL Y:", final_y)
        print("SHIELD FINAL LENGTH:", final_length)

        if final_length is None:
            print("FINAL SHIELD LENGTH COULD NOT BE READ")
            delete_staged_shield(sym, final_id)
            continue

        if abs(final_length - requested_length) > 0.001:
            print("FINAL SHIELD LENGTH CHANGED AFTER MOVE")
            print("EXPECTED FINAL LENGTH:", requested_length)
            print("ACTUAL FINAL LENGTH:", final_length)
            delete_staged_shield(sym, final_id)
            continue

        print("SHIELD FINAL LENGTH VERIFIED")
        print("SHIELD STAGED, SIZED, AND MOVED SUCCESSFULLY")

        return final_id

    raise Exception(f"Could not find a staging area where {job['symbol_name']} could resize to {requested_length}")

def place_shield(sym, job, placement):

    last_row = get_last_data_row(ws)

    if last_row < DATA_FIRST_ROW:

    print("")
    print("============================================================")
    print("ZUKEN CABLE SHIELD / TWIST OVERLAY")
    print("============================================================")
    print("EXCEL:", excel_path)
    print("WORKSHEET:", ws.title)
    print("DEVICE REFERENCE FROM B1:", sheet_device_reference)
    print("CONNECTOR ROOT:", CONNECTOR_ROOT_NAME)
    print("SHIELD SYMBOL:", SHIELD_SYMBOL_NAME)
    print("TWIST SYMBOL:", TWIST_SYMBOL_NAME)
    print("LEFT TWIST OFFSET J2:K2:", "YES" if twist_offset_settings["LEFT"] else "NO")

    print("PLACEMENT DELAY:", PLACEMENT_DELAY_SECONDS)
    print("TWIST STRETCHING: DISABLED")
    print("SPECIAL OUTER-ONLY MODE: ENABLED")
    print("DATA ROWS:", DATA_FIRST_ROW, "TO", last_row)

    left_rows = read_side_rows(ws, "LEFT", last_row)
    right_rows = read_side_rows(ws, "RIGHT", last_row)

    left_sections = split_into_sections(left_rows)
    right_sections = split_into_sections(right_rows)

    sections = left_sections + right_sections

    packages = []
    warnings = []

    for section_rows in sections:
        jobs, section_warnings = build_section_jobs(section_rows)

        if jobs:
            packages.append({"rows": section_rows, "jobs": jobs})

        warnings.extend(section_warnings)

    print("")
    print("CONNECTOR SECTIONS WITH SYMBOLS:", len(packages))

    if warnings:
        print("")
        print("============================================================")
        print("EXCEL WARNINGS")
        print("============================================================")

        for warning in warnings:
            print("WARNING:", warning)

    app = win32.gencache.EnsureDispatch("CT.Application")
    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()
    sym = prj.CreateSymbolObject()

    app.PutMessage("Starting Cable Shield / Twist Overlay Script")

    all_devices = get_all_devices(prj, dev)

    print("")
    print("E3 BASE DEVICE ID COUNT:", len(all_devices))

    pin_map_cache = {}
    placed_ids = []

    for package in packages:
        section_rows = package["rows"]
        jobs = package["jobs"]

        side = section_rows[0]["side"]
        connector = section_rows[0]["connector"]

        required_pins = list(dict.fromkeys(row["pin"] for row in section_rows))

        print("")
        print("============================================================")
        print("PROCESSING CONNECTOR SECTION")
        print("============================================================")
        print("SIDE:", side)
        print("TWIST OFFSET ENABLED:", "YES" if twist_offset_settings[side] else "NO")
        print("EXCEL CONNECTOR:", connector)
        print("EXPECTED E3 CONNECTOR:", connector_base_name(connector))
        print("EXCEL ROWS:", section_rows[0]["excel_row"], "TO", section_rows[-1]["excel_row"])
        print("REQUIRED PINS:", required_pins)

        device_record, pin_map, sheet_id = resolve_connector_schematic_view(connector, required_pins, all_devices, dev, pin, pin_map_cache)

        print_section_pin_locations(required_pins, pin_map, sheet_id)

        has_twists = any(job["type"] == "twist" for job in jobs)
        has_inner_shields = any(job["type"] == "shield_inner" for job in jobs)
        has_outer_shields = any(job["type"] == "shield_outer" for job in jobs)

        outer_only_shield_case = has_twists and has_outer_shields and not has_inner_shields

        print("")
        print("SECTION JOB ANALYSIS")
        print("HAS TWISTS:", has_twists)
        print("HAS INNER SHIELDS:", has_inner_shields)
        print("HAS OUTER SHIELDS:", has_outer_shields)
        print("OUTER-ONLY SHIELD SPECIAL CASE:", outer_only_shield_case)

        if outer_only_shield_case:
            print("SPECIAL MODE ACTIVE: OUTER SHIELD WILL BE PLACED BEFORE TWISTS")

            for job in jobs:
                if job["type"] == "shield_outer":
                    job["direct_outer_first"] = True

        jobs = sorted(jobs, key=job_priority)

        print("")
        print("FINAL JOB ORDER")

        for job_index, job in enumerate(jobs):
            print(job_index + 1, job["type"], job.get("twist_column"), "DIRECT OUTER:", job.get("direct_outer_first", False))

        for job in jobs:
            try:
                placement = calculate_placement(job, pin_map, sheet_id, twist_offset_settings)
                placed_id = place_job(sym, job, placement)

                if placed_id is not None:
                    placed_ids.append(placed_id)

            except Exception as error:
                print("ERROR:", str(error))
                app.PutMessage(f"Cable overlay warning: {error}")

            time.sleep(PLACEMENT_DELAY_SECONDS)

    print("")
    print("============================================================")
    print("OVERLAY COMPLETE")
    print("============================================================")
    print("PLACED SYMBOL COUNT:", len(placed_ids))
    print("PLACED SYMBOL IDS:", placed_ids)
    print("============================================================")

    app.PutMessage(f"Cable Shield / Twist Overlay complete. Placed {len(placed_ids)} symbols.")

    workbook.close()

    del sym
    del pin
    del dev
    del prj
    del app


if __name__ == "__main__":
