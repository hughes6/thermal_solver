#====================================================================================================================================
# CALCULATE SYMBOL POSITION FROM EXISTING CABLE PIN LOCATIONS
#====================================================================================================================================
def calculate_placement(job, pin_map, sheet_id):
    locations = []

    for pin_name in job["pins"]:
        location = pin_location_on_sheet(pin_map, pin_name, sheet_id)

        if location is None:
            raise Exception(f"{job['side']} {job['connector']}: pin {pin_name} has no schematic location on sheet {sheet_id}")

        locations.append(location)

    x_values = [location["x"] for location in locations]
    y_values = [location["y"] for location in locations]

    anchor_x = statistics.median(x_values)
    bottom_y = min(y_values)
    top_y = max(y_values)

    if top_y - bottom_y < SCHEMATIC_MIN_Y_SPAN and len(job["pins"]) > 1:
        raise Exception(f"{job['side']} {job['connector']} {job['type']}: pins {job['pins']} all have effectively the same Y coordinate")

    direction = 1.0 if job["side"] == "LEFT" else -1.0
    offset = job_offset(job)
    x = anchor_x + direction * offset

    if job["type"] == "twist":
        y = bottom_y
        length = top_y - bottom_y

    if job["type"] == "shield_inner":
        y = top_y
        length = top_y - bottom_y

    if job["type"] == "shield_outer":
        y = top_y
        length = top_y - bottom_y

    if length < MIN_SYMBOL_LENGTH:
        length = MIN_SYMBOL_LENGTH

    placement = {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "length": length,
        "anchor_x": anchor_x,
        "bottom_y": bottom_y,
        "top_y": top_y
    }

    return placement


#====================================================================================================================================
# PLACE REAL LIBRARY SYMBOL
#
# Both Shield_Demo and MIL_Twist_ArrowsDown now use:
#
#     Load
#     Place
#     SetId
#     SetDisplayLength
#
# The only geometry difference:
#
#     Shield starts at TOP wire Y
#     Twist starts at BOTTOM wire Y
#====================================================================================================================================
def place_job(sym, job, placement):
    print("")
    print("------------------------------------------------------------")
    print("TYPE:", job["type"])
    print("SIDE:", job["side"])
    print("CONNECTOR:", job["connector"])
    print("PINS:", job["pins"])
    print("EXCEL ROWS:", job["excel_rows"])
    print("LIBRARY SYMBOL:", job["symbol_name"])
    print("SHEET:", placement["sheet_id"])
    print("CONNECTOR PIN X:", placement["anchor_x"])
    print("SYMBOL X:", placement["x"])
    print("BOTTOM WIRE Y:", placement["bottom_y"])
    print("TOP WIRE Y:", placement["top_y"])
    print("SYMBOL INSERTION Y:", placement["y"])
    print("REQUESTED SYMBOL LENGTH:", placement["length"])

    if DRY_RUN:
        print("DRY RUN - NOT PLACED")
        return None

    requested_length = float(placement["length"])

    if requested_length < MIN_SYMBOL_LENGTH:
        requested_length = MIN_SYMBOL_LENGTH

    load_result = sym.Load(job["symbol_name"], SYMBOL_VERSION)

    print("LOAD RESULT:", load_result)

    if load_result == 0:
        raise Exception(f"Could not load library symbol {job['symbol_name']}")

    placed_id = sym.Place(placement["sheet_id"], placement["x"], placement["y"], 0, 1.0, 0)

    print("PLACE RESULT / SYMBOL ID:", placed_id)

    if placed_id == 0:
        raise Exception(f"Could not place library symbol {job['symbol_name']} at X={placement['x']} Y={placement['y']}")

    sym.SetId(placed_id)

    try:
        placed_name = sym.GetName()
    except Exception:
        placed_name = "UNKNOWN"

    try:
        symbol_type = sym.GetSymbolType()
    except Exception:
        symbol_type = "UNKNOWN"

    try:
        is_shield = sym.IsShield()
    except Exception:
        is_shield = None

    try:
        is_twisted = sym.IsTwisted()
    except Exception:
        is_twisted = None

    try:
        length_before = float(sym.GetDisplayLength())
    except Exception:
        length_before = None

    print("PLACED SYMBOL ID:", placed_id)
    print("PLACED SYMBOL NAME:", placed_name)
    print("PLACED SYMBOL TYPE:", symbol_type)
    print("IS SHIELD:", is_shield)
    print("IS TWISTED:", is_twisted)
    print("DISPLAY LENGTH BEFORE:", length_before)

    stretch_result = sym.SetDisplayLength(requested_length)

    print("SET DISPLAY LENGTH RESULT:", stretch_result)

    sym.SetId(placed_id)

    try:
        length_after = float(sym.GetDisplayLength())
    except Exception:
        length_after = None

    print("REQUESTED DISPLAY LENGTH:", requested_length)
    print("DISPLAY LENGTH AFTER:", length_after)

    if length_after is None:
        raise Exception(f"Could not verify display length for {job['symbol_name']}")

    if abs(length_after - requested_length) > 0.001:
        raise Exception(f"{job['symbol_name']} placed but failed to stretch from {length_before} to {requested_length}")

    print("REAL LIBRARY SYMBOL PLACED AND STRETCHED SUCCESSFULLY")

    return placed_id
