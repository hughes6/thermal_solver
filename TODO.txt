# -*- coding: utf-8 -*-

import wx
import wx.xrc
import wirelist

import gettext
_ = gettext.gettext


class MainFrame(wx.Frame):

    def __init__(self, parent, on_generate_callback):

        wx.Frame.__init__(
            self,
            parent,
            id=wx.ID_ANY,
            title="Sheet Generator",
            pos=wx.DefaultPosition,
            size=wx.Size(290, 310),
            style=wx.DEFAULT_FRAME_STYLE | wx.TAB_TRAVERSAL
        )

        self.SetSizeHints(wx.Size(290, 310), wx.DefaultSize)

        bSizer1 = wx.BoxSizer(wx.VERTICAL)

        # --------------------------------------------------------------
        # Sheet Selection
        # --------------------------------------------------------------

        m_checkList1Choices = [
            _(u"Wirelist"),
            _(u"Cable Table"),
            _(u"Print Labels"),
            _(u"End to End"),
            _(u"All")
        ]

        self.m_checkList1 = wx.CheckListBox(
            self,
            wx.ID_ANY,
            wx.DefaultPosition,
            wx.Size(200, 145),
            m_checkList1Choices,
            0
        )

        self.m_checkList1.SetFont(
            wx.Font(
                14,
                wx.FONTFAMILY_SWISS,
                wx.FONTSTYLE_NORMAL,
                wx.FONTWEIGHT_BOLD,
                False,
                "Arial"
            )
        )

        self.m_checkList1.SetForegroundColour(
            wx.SystemSettings.GetColour(wx.SYS_COLOUR_BTNTEXT)
        )

        self.m_checkList1.SetBackgroundColour(
            wx.SystemSettings.GetColour(wx.SYS_COLOUR_BTNFACE)
        )

        bSizer1.Add(
            self.m_checkList1,
            0,
            wx.ALL | wx.ALIGN_CENTER_HORIZONTAL,
            10
        )

        # --------------------------------------------------------------
        # Export Options (Separate from Select All)
        # --------------------------------------------------------------

        exportBox = wx.StaticBox(
            self,
            wx.ID_ANY,
            "Export Options"
        )

        exportSizer = wx.StaticBoxSizer(
            exportBox,
            wx.VERTICAL
        )

        self.m_savePdfCheckbox = wx.CheckBox(
            self,
            wx.ID_ANY,
            "Save Excel Sheets as PDF",
            wx.DefaultPosition,
            wx.DefaultSize,
            0
        )

        # Enabled by default
        self.m_savePdfCheckbox.SetValue(True)

        exportSizer.Add(
            self.m_savePdfCheckbox,
            0,
            wx.ALL,
            8
        )

        bSizer1.Add(
            exportSizer,
            0,
            wx.EXPAND | wx.LEFT | wx.RIGHT | wx.TOP,
            12
        )

        # --------------------------------------------------------------
        # Generate Button
        # --------------------------------------------------------------

        self.m_button1 = wx.Button(
            self,
            wx.ID_ANY,
            _(u"Generate"),
            wx.DefaultPosition,
            wx.DefaultSize,
            0
        )

        bSizer1.Add(
            self.m_button1,
            0,
            wx.ALL | wx.ALIGN_CENTER_HORIZONTAL,
            10
        )

        self.on_generate_callback = on_generate_callback

        self.SetSizer(bSizer1)
        self.Layout()
        self.Centre(wx.BOTH)

        # --------------------------------------------------------------
        # Events
        # --------------------------------------------------------------

        self.m_checkList1.Bind(
            wx.EVT_CHECKLISTBOX,
            self.OnCheck
        )

        self.Bind(
            wx.EVT_ACTIVATE,
            self.OnActivate_Main
        )

        self.Bind(
            wx.EVT_CLOSE,
            self.OnClose_Main
        )

        self.m_button1.Bind(
            wx.EVT_BUTTON,
            self.Generate
        )

    def __del__(self):
        pass

    # --------------------------------------------------------------
    # Event Handlers
    # --------------------------------------------------------------

    def OnActivate_Main(self, event):
        event.Skip()

    def OnClose_Main(self, event):
        event.Skip()

    def Generate(self, event):

        # Get selected sheets
        checked = self.m_checkList1.GetCheckedItems()

        labels = [
            self.m_checkList1.GetString(i)
            for i in checked
        ]

        # Get PDF export setting
        save_pdf = self.m_savePdfCheckbox.GetValue()

        # Send both selections to main program
        self.on_generate_callback(labels, save_pdf)

        self.Close()

    def OnCheck(self, event):

        idx = event.GetInt()
        label = self.m_checkList1.GetString(idx)

        count = self.m_checkList1.GetCount()
        all_idx = count - 1

        if label == "All":

            is_checked = self.m_checkList1.IsChecked(idx)

            # Only affect sheet selections
            for i in range(all_idx):
                self.m_checkList1.Check(i, is_checked)

        else:

            # Keep "All" synchronized with individual selections
            all_checked = all(
                self.m_checkList1.IsChecked(i)
                for i in range(all_idx)
            )

            self.m_checkList1.Check(all_idx, all_checked)

        event.Skip()

import wx
import wirelist
import endtoend
import cabletable


def on_generate(labels, save_pdf):

    if "Wirelist" in labels:
        objects["Wirelist"] = True

    if "Cable Table" in labels:
        objects["Cable Table"] = True

    if "Print Labels" in labels:
        objects["Print Labels"] = True

    if "End to End" in labels:
        objects["End to End"] = True

    # Save the PDF export option
    objects["Save PDF"] = save_pdf


def generate_sheets(objects):

    if objects["Wirelist"] == True:
        w = wirelist.WireList()
        w.run()

    if objects["End to End"] == True:
        e = endtoend.EndtoEnd()
        e.run()

    if objects["Cable Table"] == True and objects["Print Labels"] == True:
        c = cabletable.CableTable(True)
        c.run()

    elif objects["Cable Table"] == True:
        c = cabletable.CableTable(False)
        c.run()

    # PDF export will be implemented here later
    # Setting is available as objects["Save PDF"]


#------------------------------------------------------------------------------
# GUI main
#------------------------------------------------------------------------------

if __name__ == "__main__":

    try:

        objects = {
            "Wirelist": False,
            "Cable Table": False,
            "Print Labels": False,
            "End to End": False,
            "Save PDF": True
        }

        app = wx.App(False)

        frame = AppGuiCode.MainFrame(None, on_generate)
        frame.Show(True)

        app.MainLoop()

        generate_sheets(objects)

    except Exception as err:
        print(err)
