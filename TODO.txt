T="$(for d in processor0/*; do test -f "$d/fluid/U" && basename "$d"; done)"
test -n "$T" || { echo "ERROR: solved checkpoint not found"; exit 1; }

for r in 0 1 2 3; do
  d="processor${r}/${T}/uniform"
  command mkdir -p -- "$d" || exit 1
  printf 'FoamFile\n{\n    version 2.0;\n    format ascii;\n    class dictionary;\n    location "%s/uniform";\n    object time;\n}\n\nvalue %s;\ndeltaT 0.0001089;\ndeltaT0 0.0001089;\nindex 0;\n' "$T" "$T" > "$d/time" || exit 1
  test -s "$d/time" || exit 1
done

echo "Fixed restart metadata for solved checkpoint t=$T"
for r in 0 1 2 3; do ls -l "processor${r}/${T}/fluid/U" "processor${r}/${T}/uniform/time"; done
