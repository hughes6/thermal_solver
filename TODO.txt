v"""E3.series: copy cable-side pin spacing to one side of a CABBLOCK.

Workflow: choose cable symbol -> choose one of its pins (left/right) -> choose
CABBLOCK -> preview -> apply. E3 versions with inaccessible cable-pin
relationships can instead use Ctrl-selected 2+ single cable pins on one side. Uses existing placed connector-pin symbols; it
never edits logical pin names, device/connector assignments, or block geometry.

IMPORTANT: E3 COM differs by installation. This script deliberately REFUSES
writes if it cannot verify per-pin symbol ownership, a location setter or the
post-write pin position. Test in a disposable copy of the project.

Run from E3 Add-ons or with: py E3_Cable_To_Block_Pin_Spacing.py
Requires e3series (preferred) or pywin32.
"""

VERSION = 'CABLE-BLOCK-PIN-SPACING-V2-SELECTION-20261123'
MISSING = -2147352572
EDGE_ALLOWANCE = 0.65   # maximum distance, drawing units, from block edge
POINT_TOL = 0.015      # position readback tolerance in E3 drawing units
LOGFILE = Path(__file__).with_suffix('.log')


def log(text):
    with LOGFILE.open('a', encoding='utf-8') as f:
        f.write(f'{datetime.now().isoformat(timespec="seconds")} {text}\n')


def ids_from_return(raw):
    """Never treat an E3 count/status integer as an object ID."""
    if raw is None:
        return []
    if isinstance(raw, (tuple, list)):
        if len(raw) >= 2 and isinstance(raw[0], int) and not isinstance(raw[1], (int, float)):
            raw = raw[1:]
        else:
            # E3 can return (count, id_1, id_2, ...) or (count, single_id)
            # instead of (count, [ids]). Discard the count, not actual IDs.
            if (len(raw) >= 3 and isinstance(raw[0], int) and raw[0] == len(raw)-1
                    and all(isinstance(v, int) for v in raw[1:])):
                raw = raw[1:]
            elif len(raw) == 2 and isinstance(raw[0], int) and isinstance(raw[1], int):
                count, object_id = raw
                if count == 1 and object_id > 0 and object_id != MISSING:
                    return [object_id]
                # Direct (id1, id2) only when neither looks like an E3 count.
                if count > 100 and object_id > 100:
                    return list(dict.fromkeys((count, object_id)))
                return []
        values = []
        for v in raw:
            if isinstance(v, (tuple, list)):
                values.extend(v)
            else:
                values.append(v)
        return list(dict.fromkeys(int(v) for v in values
                   if isinstance(v, (int, float)) and int(v) > 0 and int(v) != MISSING))
    return []


def get_ids(obj, method):
    fn = getattr(obj, method, None)
    if fn is None:
        return []
    for args in ((None,), (), ([],)):
        try:
            result = fn(*args)
            values = ids_from_return(result)
            if values:
                return values
            # If the argument was an out-array, it may have been populated.
            if args and isinstance(args[0], list):
                values = ids_from_return([args[0]])
                if values:
                    return values
        except Exception:
            pass
    # Pywin32-only output array fallback.
    try:
        import pythoncom
        import win32com.client as w
        for flags, seed in ((pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None),
                            (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, [])):
            try:
                arg = w.VARIANT(flags, seed)
                result = fn(arg)
                values = ids_from_return(result) or ids_from_return([arg.value])
                if values:
                    return values
            except Exception:
                continue
    except ImportError:
        pass
    return []


def com_method_names(obj):
    names = set(n for n in dir(obj) if not n.startswith('_'))
    try:
        info = obj._oleobj_.GetTypeInfo()
        attr = info.GetTypeAttr()
        for i in range(attr.cFuncs):
            desc = info.GetFuncDesc(i)
            name_list = info.GetNames(desc.memid)
            if name_list:
                names.add(str(name_list[0]))
    except Exception:
        pass
    return sorted(names)


def optional(obj, method, *args):
    try:
        return getattr(obj, method)(*args)
    except Exception:
        return None


def open_e3():
    try:
        import e3series
        app = e3series.Application()
        job = app.CreateJobObject()
        if job.GetId():
            return app, job, 'official e3series wrapper'
    except Exception as exc:
        log(f'e3series wrapper unavailable: {exc}')
    import win32com.client
    app = win32com.client.Dispatch('CT.Application')
    job = app.CreateJobObject()
    if not job.GetId():
        raise RuntimeError('No active E3 project.')
    return app, job, 'pywin32'


def obj(job, category, oid):
    maker = {'symbol': 'CreateSymbolObject', 'pin': 'CreatePinObject',
             'device': 'CreateDeviceObject'}[category]
    result = getattr(job, maker)()
    result.SetId(int(oid))
    return result


def symbol_location(sym):
    # Confirmed in the user's existing working CABBLOCK header formatter:
    # e3Symbol.GetSchemaLocation(None,None,None,None,None) -> (sheet,x,y,...).
    attempts = ((None, None, None, None, None), (), (None, None, None))
    for args in attempts:
        r = optional(sym, 'GetSchemaLocation', *args)
        if not isinstance(r, (tuple, list)) or len(r) < 3:
            continue
        try:
            sheet, x, y = int(r[0]), float(r[1]), float(r[2])
            if sheet > 0 and all(math.isfinite(v) for v in (x, y)):
                return (sheet, x, y)
        except (ValueError, TypeError):
            continue
    return None


def pin_location(pin):
    # Some versions give individual PIN coordinates even for multi-pin symbols.
    # This is only used as a READ of source pins; do not move logical pins.
    for args in ((None, None, None, None, None), (), (None, None, None)):
        r = optional(pin, 'GetSchemaLocation', *args)
        if isinstance(r, (tuple, list)) and len(r) >= 3:
            try:
                v = (int(r[0]), float(r[1]), float(r[2]))
                if v[0] > 0 and math.isfinite(v[1]) and math.isfinite(v[2]):
                    return v
            except (ValueError, TypeError):
                pass
    return None


def symbol_area(sym):
    r = optional(sym, 'GetPlacedArea', None, None, None, None)
    if isinstance(r, (tuple, list)) and len(r) >= 5:
        try:
            a = tuple(float(v) for v in r[1:5])
            if int(r[0]) > 0 and a[2] > a[0] and a[3] > a[1]:
                return a  # left,bottom,right,top
        except (TypeError, ValueError):
            pass
    return None


def selected(job):
    return {'symbols': get_ids(job, 'GetSelectedSymbolIds'),
            'pins': get_ids(job, 'GetSelectedPinIds'),
            'blocks': get_ids(job, 'GetSelectedBlockIds')}



def selected_pin_points(job, sel, target_sheet=None):
    """Read *explicitly selected* single-pin symbols or pin locations.

    This is a source-only fallback. Its coordinates are measured, never moved.
    Selecting two or more cable pins is a reliable alternative when E3 does
    not expose a parent cable -> placed pin relationship.
    """
    points = []
    seen_pids = set()
    seen_sids = set()

    # First, accept selected pin IDs with real schematic coordinates.
    for pid in sel['pins']:
        try:
            pin = obj(job, 'pin', pid)
            loc = pin_location(pin)
            if not loc or (target_sheet is not None and loc[0] != target_sheet):
                continue
            sid = optional(pin, 'GetSymbolId')
            if not isinstance(sid, int) or sid <= 0:
                sid = None
            points.append({'sid': sid, 'pid': pid, 'sheet': loc[0],
                           'x': loc[1], 'y': loc[2], 'moveable': False})
            seen_pids.add(pid)
            if sid:
                seen_sids.add(sid)
        except Exception as exc:
            log(f'SOURCE selection pin {pid}: cannot read: {exc}')

    for sid in sel['symbols']:
        if sid in seen_sids:
            continue
        try:
            sym = obj(job, 'symbol', sid)
            loc = symbol_location(sym)
            if not loc or (target_sheet is not None and loc[0] != target_sheet):
                continue
            # A CABBLOCK or a multi-pin graphic is not an individual marker.
            if is_block(sym):
                log(f'SOURCE selection {sid}: CABBLOCK body is not a cable pin')
                continue
            pids = get_ids(sym, 'GetPinIds')
            if len(pids) > 1:
                log(f'SOURCE selection {sid}: multi-pin symbol ({len(pids)} pins); not one selected pin')
                continue
            pid = pids[0] if pids else None
            if pid and pid in seen_pids:
                continue
            points.append({'sid': sid, 'pid': pid, 'sheet': loc[0],
                           'x': loc[1], 'y': loc[2], 'moveable': False})
            seen_sids.add(sid)
            if pid:
                seen_pids.add(pid)
        except Exception as exc:
            log(f'SOURCE selection symbol {sid}: cannot read: {exc}')

    # For a selected pin with no schema location, follow its single placed
    # symbol. This is READ ONLY; we do not use this in destination writes.
    for pid in sel['pins']:
        if pid in seen_pids:
            continue
        try:
            pin = obj(job, 'pin', pid)
            sids = get_ids(pin, 'GetSymbolIds')
            direct = optional(pin, 'GetSymbolId')
            if isinstance(direct, int) and direct > 0:
                sids = list(dict.fromkeys([direct] + sids))
            if len(sids) != 1 or sids[0] in seen_sids:
                continue
            sym = obj(job, 'symbol', sids[0])
            loc = symbol_location(sym)
            if not loc or (target_sheet is not None and loc[0] != target_sheet):
                continue
            pids = get_ids(sym, 'GetPinIds')
            if len(pids) != 1 or pids[0] != pid:
                continue
            points.append({'sid': sids[0], 'pid': pid, 'sheet': loc[0],
                           'x': loc[1], 'y': loc[2], 'moveable': False})
            seen_pids.add(pid)
            seen_sids.add(sids[0])
        except Exception as exc:
            log(f'SOURCE selected pin {pid} symbol lookup: {exc}')

    # Deduplicate pins from pin and symbol selectors; also avoid interpreting
    # two identifiers for the same physical point as two cable pins.
    unique=[]
    for pt in sorted(points, key=lambda z: (-z['y'], z['x'])):
        if all(pt['sheet'] != other['sheet'] or
               math.hypot(pt['x']-other['x'], pt['y']-other['y']) > POINT_TOL
               for other in unique):
            unique.append(pt)
    return unique


def manual_source_from_points(points):
    if len(points) < 2:
        raise ValueError('Select at least TWO individual cable pins on the SAME edge in E3.')
    sheets = {p['sheet'] for p in points}
    if len(sheets) != 1:
        raise ValueError('Selected cable pins are on different E3 sheets.')
    # Manual mode is specifically for a vertical row on one cable side.
    span=max(p['x'] for p in points)-min(p['x'] for p in points)
    if span > max(POINT_TOL * 4, 0.10):
        raise ValueError(f'Selected pins have different X positions (spread {span:.3f}). '
                         'Ctrl-select pins from ONE vertical side of the cable.')
    ordered=sorted(points,key=lambda x:-x['y'])
    if any(abs(ordered[i]['y']-ordered[i+1]['y']) <= POINT_TOL
           for i in range(len(ordered)-1)):
        raise ValueError('Two selected cable pins have overlapping Y coordinates.')
    return {'sid': None, 'did': None, 'sheet': ordered[0]['sheet'],
            'x': statistics.mean(p['x'] for p in ordered),
            'y': statistics.mean(p['y'] for p in ordered),
            'area': None, 'block': False, 'points': ordered,
            'manual_side_only': True}

def selection_symbol(job, sel):
    sids = sel['symbols']
    if len(sids) == 1:
        return sids[0]
    if not sids and len(sel['pins']) == 1:
        p = obj(job, 'pin', sel['pins'][0])
        sid = optional(p, 'GetSymbolId')
        if isinstance(sid, int) and sid > 0:
            return sid
        sids = get_ids(p, 'GetSymbolIds')
        if len(sids) == 1:
            return sids[0]
    raise ValueError('Select exactly ONE placed E3 symbol or individual pin, then press the button.')


def is_block(sym):
    flag = optional(sym, 'IsBlock')
    typ = optional(sym, 'GetSymbolType')
    hints = ' '.join(str(optional(sym, name) or '') for name in
                     ('GetName', 'GetSymbolTypeName', 'GetSymbolName')).upper()
    return flag in (True, 1) or typ == 4 or 'CABBLOCK' in hints


def source_or_block(job, sid, sel):
    s = obj(job, 'symbol', sid)
    loc = symbol_location(s)
    if not loc:
        raise RuntimeError(f'Symbol {sid}: E3 did not return its sheet/X/Y.')
    area = symbol_area(s)
    did = optional(s, 'GetDeviceId') or optional(s, 'GetOwnerDeviceId')
    # In E3 a CABBLOCK is often a device, not a standalone Block object.
    if not (isinstance(did, int) and did > 0) and len(sel['blocks']) == 1:
        did = sel['blocks'][0]
    return {'sid': sid, 'did': did if isinstance(did, int) and did > 0 else None,
            'sheet': loc[0], 'x': loc[1], 'y': loc[2], 'area': area,
            'block': is_block(s)}


def gather_markers(job, parent):
    """View-scoped connector pin markers; never scan the entire sheet.

    Only connector symbols owned by the selected parent device/block are used.
    If multiple placed block views share one device, accept the symbols near
    this placed view only. A connector GROUP is never a movable single pin.
    """
    root = obj(job, 'symbol', parent['sid'])
    owner = obj(job, 'device', parent['did']) if parent['did'] else None
    connector_ids = set()
    for thing in (root, owner):
        if thing is None: continue
        for name in ('GetConnectorIds', 'GetBlockConnectorIds'):
            connector_ids.update(get_ids(thing, name))

    symbols = set()
    pin_candidates = set()
    for thing in (root, owner):
        if thing is None: continue
        pin_candidates.update(get_ids(thing, 'GetPinIds'))
    for cid in connector_ids:
        try:
            c = obj(job, 'device', cid)
            symbols.update(get_ids(c, 'GetSymbolIds'))
            pin_candidates.update(get_ids(c, 'GetPinIds'))
        except Exception as exc:
            log(f'Connector device {cid} cannot be read: {exc}')
    for pid in pin_candidates:
        try:
            p = obj(job, 'pin', pid)
            symbols.update(get_ids(p, 'GetSymbolIds'))
            psid = optional(p, 'GetSymbolId')
            if isinstance(psid, int) and psid > 0:
                symbols.add(psid)
        except Exception:
            pass
    # A cable may itself be represented as a set of pin symbols, not a block.
    if owner and not symbols:
        symbols.update(get_ids(owner, 'GetSymbolIds'))
    symbols.discard(parent['sid'])
    a = parent['area']
    out = []
    for sid in sorted(symbols):
        try:
            s = obj(job, 'symbol', sid)
            loc = symbol_location(s)
            if not loc or loc[0] != parent['sheet']:
                continue
            _, x, y = loc
            if a:
                if not (a[0]-EDGE_ALLOWANCE <= x <= a[2]+EDGE_ALLOWANCE and
                        a[1]-EDGE_ALLOWANCE <= y <= a[3]+EDGE_ALLOWANCE):
                    continue
            else:
                # Without a source outline, do not gather distant device views.
                if math.hypot(x-parent['x'], y-parent['y']) > 8:
                    continue
            pids = get_ids(s, 'GetPinIds')
            if len(pids) == 0:
                # This could be a multi-pin graphic symbol; it is not safe to
                # use unless a unique connected pin identifies it.
                linked = []
                for pid in pin_candidates:
                    try:
                        pp = obj(job, 'pin', pid)
                        sid_from_pin = optional(pp, 'GetSymbolId')
                        if sid_from_pin == sid or sid in get_ids(pp, 'GetSymbolIds'):
                            linked.append(pid)
                    except Exception:
                        pass
                pids = list(dict.fromkeys(linked))
            if len(pids) != 1:
                log(f'SKIP candidate symbol {sid}: {len(pids)} pins (not a single-pin symbol)')
                continue
            out.append({'sid': sid, 'pid': pids[0], 'x': x, 'y': y,
                        'sheet': loc[0], 'moveable': True})
        except Exception as exc:
            log(f'SKIP linked symbol {sid}: {exc}')
    # If the selected source symbol has internal pins (one symbol housing
    # several actual nodes), try reading individual E3 pin coordinates.
    if not out:
        for pid in sorted(pin_candidates):
            try:
                p = obj(job, 'pin', pid)
                loc = pin_location(p)
                if loc and loc[0] == parent['sheet']:
                    out.append({'sid': None, 'pid': pid, 'x': loc[1],
                                'y': loc[2], 'sheet': loc[0], 'moveable': False})
            except Exception:
                pass
    return sorted(out, key=lambda x: (-x['y'], x['x'], x['pid']))


def side_of_pin(parent, points, pin):
    a = parent['area']
    if a:
        middle = (a[0] + a[2])/2
    elif len(points) >= 2:
        middle = (min(p['x'] for p in points) + max(p['x'] for p in points))/2
        if abs(min(p['x'] for p in points) - max(p['x'] for p in points)) < POINT_TOL:
            raise ValueError('Unable to distinguish left and right side of cable pins.')
    else:
        raise ValueError('Need a source outline or pins on both sides to determine side.')
    return 'left' if pin['x'] < middle else 'right', middle


def pick_edge(points, side, mid, area=None):
    p = [v for v in points if (v['x'] < mid if side == 'left' else v['x'] >= mid)]
    if area:
        edge = area[0] if side == 'left' else area[2]
        p = [v for v in p if abs(v['x'] - edge) <= EDGE_ALLOWANCE]
    # Multiple pins at identical Y positions are ambiguous even if ordered.
    ys = [v['y'] for v in p]
    if any(abs(ys[i] - ys[j]) <= POINT_TOL for i in range(len(ys)) for j in range(i+1, len(ys))):
        raise ValueError(f'{side} edge has multiple pins at the same Y position. Cannot determine pin order safely.')
    return sorted(p, key=lambda v: -v['y'])


def make_plan(src, dst, src_side, target_side):
    if not dst['block']:
        raise ValueError('Target must be a placed CABBLOCK/block symbol.')
    if not dst['area']:
        raise ValueError('Target block outline GetPlacedArea could not be read.')
    if not src['points']:
        raise ValueError('No view-scoped source pin symbols were identified.')
    if not dst['points']:
        raise ValueError('No view-scoped target pin symbols were identified.')
    smid = (src['area'][0]+src['area'][2])/2 if src['area'] else (min(p['x'] for p in src['points'])+max(p['x'] for p in src['points']))/2
    dmid = (dst['area'][0]+dst['area'][2])/2
    # In manual mode the user explicitly Ctrl-selected just one cable edge.
    # The source may have no body outline or opposite-edge points at all.
    source_pins = (sorted(src['points'], key=lambda v: -v['y'])
                   if src.get('manual_side_only') else
                   pick_edge(src['points'], src_side, smid, src['area']))
    targets = pick_edge(dst['points'], target_side, dmid, dst['area'])
    if len(source_pins) < 2:
        raise ValueError('Need at least TWO cable pins on the selected edge.')
    if len(targets) < 2:
        raise ValueError('Need at least TWO individual connector-pin symbols on the block edge.')
    if any(not p['moveable'] for p in targets):
        raise ValueError('Target has non-individual pin geometry; cannot reposition as single-pin symbols.')
    if len({p['sid'] for p in targets}) != len(targets):
        raise ValueError('Two target pins share a placed symbol. Refusing to move a whole pin group.')
    src_gaps = [source_pins[i]['y']-source_pins[i+1]['y'] for i in range(len(source_pins)-1)]
    if any(g <= POINT_TOL for g in src_gaps):
        raise ValueError('Source pins do not have distinct vertical coordinates.')
    mode = 'exact offset pattern' if len(source_pins) == len(targets) else 'median pitch (pin counts differ)'
    offsets = [source_pins[0]['y']-p['y'] for p in source_pins] if len(source_pins)==len(targets) else [i*statistics.median(src_gaps) for i in range(len(targets))]
    top = targets[0]['y']
    output = []
    for pin, off in zip(targets, offsets):
        new_y = top-off
        if not (dst['area'][1]-0.03 <= new_y <= dst['area'][3]+0.03):
            raise ValueError(f'New pin Y={new_y:.3f} is outside target block top/bottom. Block size will NOT be changed.')
        output.append({'sid': pin['sid'], 'pid': pin['pid'], 'sheet': pin['sheet'],
                       'x': pin['x'], 'old_y': pin['y'], 'new_y': new_y})
    if len({round(p['new_y'],3) for p in output}) != len(output):
        raise ValueError('Calculated overlapping target pins. Aborting.')
    return {'source_count': len(source_pins), 'target_count': len(targets),
            'mode': mode, 'source_pitch': statistics.median(src_gaps),
            'target_side': target_side, 'source_side': src_side,
            'source': src, 'target': dst, 'changes': output}


def move_symbol(job, item):
    """Only attempt absolute location setters, verify new position; fail closed.

    No e3Symbol move-by-offset calls, no unplace/re-place, and no E3 geometry
    resize calls. COM moves connected objects only as E3 normally permits.
    """
    s = obj(job, 'symbol', item['sid'])
    old = symbol_location(s)
    if not old or old[0] != item['sheet'] or abs(old[1]-item['x'])>POINT_TOL or abs(old[2]-item['old_y'])>POINT_TOL:
        raise RuntimeError(f'Pin symbol {item["sid"]} has changed since Preview. Refresh before applying.')
    if abs(item['new_y']-item['old_y']) <= POINT_TOL:
        return 'already aligned'
    # SetSchemaLocation is an E3 text method in previous scripts; symbol
    # implementations may or may not expose it. Do NOT presume it exists.
    attempts = [ ('SetSchemaLocation', (item['x'], item['new_y']),
                 (item['x'], item['old_y'])),
                 ('SetSchemaLocation', (item['sheet'],item['x'],item['new_y']),
                 (item['sheet'],item['x'],item['old_y'])),
                 ('SetPosition', (item['x'],item['new_y']),
                 (item['x'],item['old_y'])) ]
    errors = []
    for method, args, rollback in attempts:
        fn = getattr(s, method, None)
        if fn is None:
            continue
        try:
            result = fn(*args)
        except Exception as exc:
            errors.append(f'{method}{args}: {exc}')
            continue
        new = symbol_location(s)
        if new and new[0]==old[0] and abs(new[1]-item['x'])<=POINT_TOL and abs(new[2]-item['new_y'])<=POINT_TOL:
            return f'{method}{args}, result={result!r}'
        # If a COM call succeeded but position was not verified, DO NOT try
        # another call with a different signature: attempt rollback and stop.
        try:
            fn(*rollback)
            restored = symbol_location(s)
            if not restored or math.hypot(restored[1]-old[1], restored[2]-old[2]) > POINT_TOL:
                raise RuntimeError('rollback readback does not match original')
        except Exception as exc:
            raise RuntimeError(f'Symbol {item["sid"]}: attempted {method}, location not verified, rollback FAILED: {exc}. Inspect E3 immediately.') from exc
        raise RuntimeError(f'Symbol {item["sid"]}: {method} did not change position; rolled back. E3 setter needs version-specific mapping.')
    log('AVAILABLE SYMBOL MOVE/PLACE/SET METHODS: '+repr([n for n in com_method_names(s) if any(k in n.lower() for k in ('move','place','schema','position','location'))]))
    raise RuntimeError('No verified automatic symbol location setter available for this E3 version. '
                       'E3 e3Symbol.PlaceInteractively exists for manual moving, but does not give this script an automatic coordinate move. '
                       f'Tried absolute methods; errors={errors[:3]}')


def run_changes(job, plan):
    done=[]
    try:
        for item in plan['changes']:
            result=move_symbol(job, item)
            log(f'APPLY pin={item["pid"]} symbol={item["sid"]} {item["old_y"]:.3f}->{item["new_y"]:.3f}: {result}')
            if result != 'already aligned':
                done.append(item)
        return len(done)
    except Exception:
        log('WARNING: apply interrupted. Attempting to restore moved symbols.')
        for item in reversed(done):
            try:
                back=dict(item)
                back['old_y'],back['new_y']=item['new_y'],item['old_y']
                move_symbol(job,back)
                log(f'ROLLBACK OK {item["sid"]}')
            except Exception as exc:
                log(f'ROLLBACK FAILED {item["sid"]}: {exc}; manually inspect project')
        raise

        self.manual_source_side=tk.StringVar(value='')

    def write(self, s):
        self.output.insert('end',str(s)+'\n')
        self.output.see('end')
        log(str(s))

    def error(self,e):
        self.write('ERROR: '+str(e))
        log(traceback.format_exc())
        self.status.set(str(e))
        messagebox.showerror('Cable pin spacing',str(e),parent=self.root)

    def clear_plan(self):
        self.plan=None
        self.apply_button.configure(state='disabled')

    def reset(self):
        self.src=self.dst=self.clicked=None
        self.src_side=None
        self.manual_source_side.set('')
        self.clear_plan()
        self.status.set('Select a CABLE body OR 2+ individual cable pins, then click 1.')
        self.write('--- Cleared; ready for a new cable ---')

    def read_selection(self):
        app,job,backend=open_e3()
        sel=selected(job)
        self.write(f'Selection: {sel} ({backend})')
        return app,job,sel

    def capture_cable(self):
        try:
            app,job,sel=self.read_selection()
            selected_points=selected_pin_points(job,sel)
            if len(selected_points) >= 2:
                # Explicitly selected 2+ pins: no cable-device relationship
                # lookup is required. This also fixes unlinked cable symbols.
                manual=manual_source_from_points(selected_points)
                self.src=manual
                self.clicked=self.dst=None
                self.src_side=None
                self.clear_plan()
                self.write(f'MANUAL CABLE PIN SELECTION: {len(manual["points"])} pins, '
                           f'sheet {manual["sheet"]}; X spread '
                           f'{max(p["x"] for p in manual["points"])-min(p["x"] for p in manual["points"]):.3f}')
                self.status.set('Cable pins captured. Choose Left/Right above, then click 2; '
                                'NO need to change selection in E3.')
                return
            sid=selection_symbol(job,sel)
            parent=source_or_block(job,sid,sel)
            parent['points']=gather_markers(job,parent)
            parent['manual_side_only']=False
            self.src=parent
            self.clicked=self.dst=None
            self.src_side=None
            self.clear_plan()
            self.write(f'CABLE {sid} / device {parent["did"]}: '
                       f'{len(parent["points"])} auto-detected pins, sheet {parent["sheet"]}, '
                       f'bounds={parent["area"]}')
            if len(parent['points']) >= 2:
                self.status.set('Now select ONE cable pin on the left or right and click 2. '
                                'If one pin is not reported, Ctrl-select 2+ pins on one edge and click 2.')
            else:
                self.status.set('Cable captured, but E3 did not report 2 linked pins. '
                                'Ctrl-select TWO OR MORE cable pins on ONE edge in E3, '
                                'choose Left/Right above, then click 2.')
                self.write('FALLBACK REQUIRED: E3 cannot expose the cable-side row from this symbol. '
                           'Select 2+ individual cable-pin SYMBOLS, then click button 2.')
        except Exception as exc:self.error(exc)
        finally:gc.collect()

    def capture_side(self):
        try:
            if not self.src:
                raise ValueError('First click button 1 with the cable BODY or 2+ cable pins selected.')
            # In direct selection mode, step 1 already captured the full
            # source row. Step 2 simply records which physical cable edge it is.
            if self.src.get('manual_side_only'):
                side=self.manual_source_side.get()
                if side not in ('left','right'):
                    raise ValueError('For a manually selected cable-pin row, '
                                     'choose LEFT or RIGHT in the new option above.')
                self.src_side=side
                self.clear_plan()
                self.write(f'CABLE SIDE: {side.upper()} / manually selected '
                           f'{len(self.src["points"])} cable-side pins.')
                self.status.set('Now select ONE CABBLOCK in E3 and click 3.')
                return

            app,job,sel=self.read_selection()
            selected_points=selected_pin_points(job,sel,target_sheet=self.src['sheet'])
            candidate_sids=set(sel['symbols'])
            candidate_pids=set(sel['pins'])
            matches=[p for p in self.src['points'] if
                     ((p['sid'] in candidate_sids if p['sid'] is not None else False)
                     or p['pid'] in candidate_pids)]

            if len(matches)==1 and len(self.src['points'])>=2:
                self.clicked=matches[0]
                try:
                    side,_=side_of_pin(self.src,self.src['points'],self.clicked)
                except ValueError:
                    side=self.manual_source_side.get()
                if side not in ('left','right'):
                    raise ValueError('Cable pin detected but its side is ambiguous. '
                                     'Choose LEFT or RIGHT above and click 2 again.')
                self.src_side=side
                self.clear_plan()
                self.write(f'CABLE SIDE: {side.upper()} / clicked pin '
                           f'{self.clicked["pid"]}, Y={self.clicked["y"]:.3f}; '
                           f'{len(self.src["points"])} linked pins available')
                self.status.set('Now select ONE CABBLOCK in E3 and click 3.')
                return

            # Multiple selected individual pin symbols on one edge are a
            # valid spacing reference even if E3 fails to link a cable body.
            if len(selected_points)>=2:
                manual=manual_source_from_points(selected_points)
                side=None
                if self.src['area']:
                    mid=(self.src['area'][0]+self.src['area'][2])/2
                    inferred={('left' if p['x']<mid else 'right') for p in manual['points']}
                    if len(inferred)==1:
                        side=next(iter(inferred))
                if side is None:
                    side=self.manual_source_side.get()
                if side not in ('left','right'):
                    raise ValueError('Captured pin spacing, but the cable side cannot be inferred. '
                                     'Choose LEFT or RIGHT above, then click 2 again.')
                self.src['points']=manual['points']
                self.src['manual_side_only']=True
                self.src_side=side
                self.clear_plan()
                self.write(f'CABLE SIDE: {side.upper()}; '
                           f'manually measured {len(manual["points"])} selected cable pins.')
                self.status.set('Now select ONE CABBLOCK in E3 and click 3.')
                return

            if len(selected_points)==1:
                self.write(f'Only one physical pin position was returned: '
                           f'X={selected_points[0]["x"]:.3f}, Y={selected_points[0]["y"]:.3f}.')
            raise ValueError('E3 did not identify one of the automatically linked cable pins. '
                             f'Matched {len(matches)} linked pins, identified '
                             f'{len(selected_points)} individually selected positions. '
                             'Ctrl-select TWO OR MORE separate cable pin SYMBOLS on the SAME edge, '
                             'then click button 2. Step 1 has been preserved.')
        except Exception as exc:self.error(exc)
        finally:gc.collect()

    def capture_block(self):
        try:
            if not self.src or not self.src_side:raise ValueError('Capture cable and a side pin first.')
            app,job,sel=self.read_selection()
            sid=selection_symbol(job,sel)
            parent=source_or_block(job,sid,sel)
            if not parent['block']:
                raise ValueError(f'Symbol {sid} does not report itself as a CABBLOCK.')
            if sid==self.src['sid']:
                raise ValueError('Cable and target block must be different symbols.')
            parent['points']=gather_markers(job,parent)
            self.dst=parent
            self.clear_plan()
            self.write(f'BLOCK {sid} / device {parent["did"]}: {len(parent["points"])} positioned single pins, bounds={parent["area"]}')
            if parent['points']:
                sample = obj(job, 'symbol', parent['points'][0]['sid'])
                methods = [n for n in com_method_names(sample) if any(k in n.lower() for k in ('move','place','schema','position','location'))]
                self.write(f'PIN SYMBOL COM positioning methods: {methods}')
            self.status.set('Press PREVIEW. No E3 locations have been changed.')
        except Exception as exc:self.error(exc)
        finally:gc.collect()

    def preview(self):
        try:
            if not self.src or not self.dst or not self.src_side:
                raise ValueError('Complete cable, side pin, and block selection (steps 1–3) first.')
            side=(('left' if self.src_side=='right' else 'right')
                  if self.target_side.get()=='opposite' else self.src_side)
            self.plan=make_plan(self.src,self.dst,self.src_side,side)
            p=self.plan
            self.write('\n--- READ-ONLY PREVIEW ---')
            self.write(f'Cable {p["source"]["sid"]} {p["source_side"]}: {p["source_count"]} pins; median pitch {p["source_pitch"]:.4f}')
            self.write(f'Block {p["target"]["sid"]} {p["target_side"]}: {p["target_count"]} pins; {p["mode"]}')
            self.write('Pin SID       X        current Y    proposed Y    delta Y')
            for x in p['changes']:
                self.write(f'{x["sid"]:10d} {x["x"]:8.3f} {x["old_y"]:12.3f} {x["new_y"]:13.3f} {x["new_y"]-x["old_y"]:10.3f}')
            self.write('Block outline and pin X coordinates remain unchanged. NOT connected/pluggable automatically.')
            self.apply_button.configure(state='normal')
            self.status.set('Review the pin IDs/Y values before choosing APPLY. Save a project copy first.')
        except Exception as exc:self.clear_plan();self.error(exc)

    def apply(self):
        if not self.plan:return
        p=self.plan
        n=sum(abs(v['new_y']-v['old_y'])>POINT_TOL for v in p['changes'])
        if not messagebox.askyesno('Apply pin spacing',
            f'Move {n} individual CABBLOCK connector-pin symbols?\n\n'+
            'Save a separate project copy first.\nThe block size will not change.\n'+
            'If E3 does not support a verified symbol-position setter, the script will STOP.',parent=self.root):
            return
        try:
            app,job,backend=open_e3()
            current=source_or_block(job,p['target']['sid'],{'blocks':[]})
            current['points']=gather_markers(job,current)
            # Re-generate preview from live target positions to avoid stale edits.
            refreshed=make_plan(p['source'],current,p['source_side'],p['target_side'])
            if [(v['sid'],round(v['old_y'],3),round(v['new_y'],3)) for v in refreshed['changes']] != [
                (v['sid'],round(v['old_y'],3),round(v['new_y'],3)) for v in p['changes']]:
                raise RuntimeError('Target pin layout changed since preview. Capture block and preview again.')
            if current['area']!=p['target']['area']:
                raise RuntimeError('Target block geometry changed since preview. Aborting.')
            count=run_changes(job,p)
            self.write(f'APPLY COMPLETE: moved {count} connector pin symbols.')
            self.status.set(f'Finished: moved {count} pins. Verify E3 visually before saving.')
            self.clear_plan()
        except Exception as exc:self.clear_plan();self.error(exc)
        finally:gc.collect()


def main():
    if '--worker' not in sys.argv:
        # E3's Add-on launcher may lock selection until its process exits.
        # Launch detached child, as in the user's working Match Properties GUI.
        flags=getattr(subprocess,'DETACHED_PROCESS',0)|getattr(subprocess,'CREATE_NEW_PROCESS_GROUP',0)
        try:
            subprocess.Popen([sys.executable,str(Path(__file__).resolve()),'--worker'],
                             cwd=str(Path(__file__).resolve().parent),creationflags=flags,
                             stdin=subprocess.DEVNULL,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,
                             close_fds=True)
            return
        except Exception:
            log('DETACHED GUI LAUNCH FAILED: '+traceback.format_exc())
            raise
    try:
        log(f'=== Start {VERSION} ===')
        gui=App();gui.root.mainloop()
    except Exception:
        log('FATAL: '+traceback.format_exc())
        raise

if __name__=='__main__':
    main()
