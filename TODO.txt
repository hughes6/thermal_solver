"""Zuken E3.series - Match Properties (v10: block and connector-pin text properties)

Select one TEXT or BLOCK SYMBOL on a sheet and click Copy Source.
The separate window STAYS OPEN after copying. Click back into E3 to select
target(s), return to this window, Preview Targets, then Apply.

IMPORTANT: This file first launches a DETACHED GUI process and then exits.
This avoids keeping the E3 Add-on launcher waiting in an active script, which
can cause E3's drawing window to appear locked. COM objects are also released
after every click, instead of being retained for the lifetime of the GUI.

This script traverses SYMBOL -> PINS/CONNECTORS -> PIN-OWNED TEXT, in
addition to the symbol's own text. Pin text type #7 (local signal name) and
#12 (device designation) can be attached to top connector/pin symbols rather
than to the main block symbol. Only formatting is copied; no text values,
connections, pin names or positions are written.

This script copies FONT/STYLE only for matched E3 text types. Standard type IDs
are built in from the supplied E3 text-type table; no user text-type entry or
learning step is required. E3 attribute VALUES are NEVER copied. E3 type 1000
is generic Attribute, so Equipment Identification aliases must be checked for
ambiguity, rather than assuming every type-1000 attribute is equipment ID.

This script does NOT change block dimensions, scaling, width/height of symbol,
names, IDs, text content, object locations, pin information, connections, or component assignments. The only write calls are paired
format setters, and only after the user approves the preview.

Requires Windows with E3.series running and pywin32 installed:
    py -m pip install pywin32

For best COM output-array handling, optionally install Zuken's official
E3.series Python wrapper, matching your E3.series version:
    py -m pip install e3series

Automatic color (-1) is preserved for color properties. Other -1 format
values are treated as potentially meaningful defaults/inheritance rather
than automatically skipped. A target reporting -1 for hatch width/style WILL
be offered as a target. An E3 setter that does not stick is logged accurately.
The hatch pattern is copied before hatch colour/width/style/distance when
its getter/setter values can be read safely. E3 support varies by symbol.

Not executable/testable here: actual COM behavior depends on the user's
installed version. Test in a disposable copy of a project first.
"""

AUTO_COLOUR = -1  # E3's automatic/default color on tested block outline.

# Do not include content/position/orientation or geometry by default.
# E3 text width may mean glyph ratio, not measured text width; no change
# to text width is attempted until that distinction is established.
# Built-in E3 text-type codes from the user's supplied E3 type list.
# Important distinctions: 12 = DEVICE designation, not 32 (sheet designation);
# 15 = DEVICE location, not 35 (sheet location).
DEVICE_DESIGNATION_TYPE = 12
LOCATION_TYPE = 15
SIGNAL_NAME_TYPE = 7
DEVICE_FUNCTION_TYPE = 420
DEVICE_SUB_FUNCTION_TYPE = 421
ADDITIONAL_PART_ID_TYPE = 482       # E3's standard "Additional Part"
ATTRIBUTE_TYPE = 1000              # Generic "Attribute", NOT a unique Equipment ID type
SABER_TYPE_IDS = (5000, 5001, 5002, 5003)

# The user's cable drawings used text type 1000 for an additional part ID.
# Equipment Identification can be implemented as an attribute on some drawings.
# Treat 482 and 1000 as possible formatting equivalents ONLY if a unique
# compatible counterpart exists. A 1000 text with an explicitly different
# attribute name is NOT treated as equipment identification.
KNOWN_TYPE_CONFIG = {
    'Equipment Identification': (ATTRIBUTE_TYPE,),
    'Device Function': (DEVICE_FUNCTION_TYPE, DEVICE_SUB_FUNCTION_TYPE),
    'Saber Parameters': SABER_TYPE_IDS,
    'Local Signal Name': (SIGNAL_NAME_TYPE,),
}
PREFERRED_TEXT_TYPES = (
    DEVICE_DESIGNATION_TYPE, LOCATION_TYPE, SIGNAL_NAME_TYPE,
    ADDITIONAL_PART_ID_TYPE, ATTRIBUTE_TYPE,
    DEVICE_FUNCTION_TYPE, DEVICE_SUB_FUNCTION_TYPE, *SABER_TYPE_IDS,
)
TEXT_TYPE_NAMES = {
    7: 'Local Signal Name',
    12: 'Device Designation',
    15: 'Location',
    420: 'Device Function',
    421: 'Device Sub-function',
    482: 'Additional Part',
    1000: 'Attribute (possible Equipment Identification / Part ID)',
    5000: 'Saber Model Name',

def normalize_attribute_name(raw):
    """Normalize just an attribute NAME, never its value, for safe comparison."""
    if raw is None:
        return ''
    return ''.join(ch.lower() for ch in str(raw) if ch.isalnum())


def equipment_attribute_name(raw):
    """Accept only names that actually identify equipment/part ID fields."""
    key = normalize_attribute_name(raw)
    return key in {
        'equipmentid', 'equipmentidentification', 'equipmentidentifier',
        'additionalpartid', 'additionalpart', 'additionalpartnumber',
    }


def text_role(typ, config=None, attribute_name=None):
    """Role for safe source-to-target FONT matching, never data matching."""
    if typ == ADDITIONAL_PART_ID_TYPE:
        return 'Equipment / Additional Part ID'
    if typ == ATTRIBUTE_TYPE:
        if attribute_name and not equipment_attribute_name(attribute_name):
            return f'Other named attribute ({normalize_attribute_name(attribute_name)})'
        # Attribute names cannot always be queried via COM. Unique, otherwise
        # unidentifiable #1000 text is a candidate and never matched by order.
        return 'Equipment / Additional Part ID'
    if typ == DEVICE_DESIGNATION_TYPE:
        return 'Device Designation'
    if typ == LOCATION_TYPE:
        return 'Location'
    if typ == SIGNAL_NAME_TYPE:
        return 'Local Signal Name'
    if typ in (DEVICE_FUNCTION_TYPE, DEVICE_SUB_FUNCTION_TYPE):
        # Keep function and sub-function distinct (exact type match only).
        return TEXT_TYPE_NAMES[typ]
    if typ in SABER_TYPE_IDS:
        # Never copy a Saber Library font to Saber Parameter 1 by group order.
        return TEXT_TYPE_NAMES[typ]
    return f'Text type {typ}'


def text_type_description(typ, config=None):
    return TEXT_TYPE_NAMES.get(typ, f'Text type {typ}')


def read_attribute_name(text_obj, names):
    """Best-effort read-only attribute identifier for generic type 1000 text.

    No value/content getters are called; attribute values must remain private
    to the document and never be used to match visual formatting.
    """
    for method in ('GetAttributeName', 'GetAssignedAttributeName',
                   'GetTextAttributeName', 'GetAssignedAttribute'):
        if names and method not in names:
            continue
        try:
            value = getattr(text_obj, method)()
        except Exception:
            continue
        if isinstance(value, str) and value.strip():
            return value.strip()
    return None

def palette_override(selection):
    """Return a concrete hatch color index or None to copy the source color."""
    if selection == 'source':
        return None
    if selection not in PALETTE_BY_INDEX:
        raise ValueError(f'Unrecognized palette selection: {selection!r}')
    return int(selection)

def is_int_id(value):
    return isinstance(value, int) and not isinstance(value, bool) and value > 0


def display(value, maxlen=175):
    s = repr(value)
    return s if len(s) <= maxlen else s[:maxlen] + ' ...'


def collect_ids(value):
    """Turn E3's variable COM output shapes into a list of positive IDs."""
    if isinstance(value, (list, tuple)):
        items = []
        for val in value:
            items.extend(collect_ids(val))
        return items
    return [int(value)] if is_int_id(value) else []


def count_and_ids(result):
    """Recognize (count, ids) without mistaking a count for an ID."""
    if isinstance(result, (tuple, list)) and result:
        if isinstance(result[0], int) and not isinstance(result[0], bool):
            return int(result[0]), collect_ids(result[1:])
    # Sometimes pywin32 returns a count alone, but it is NOT an object ID.
    return (result if isinstance(result, int) else None), []


def get_selected(job, method_name, is_official_wrapper=False):
    """Read current selection without making changes to the drawing.

    Strategies: default COM call, then explicit by-reference variants.
    A value of -2147352572 is NOT treated as a selected object ID.
    """
    fn = getattr(job, method_name)
    attempts = []
    best_count = 0
    try:
        raw = fn()
        count, ids = count_and_ids(raw)
        attempts.append(f'no args: {display(raw)}')
        best_count = max(0, count or 0)
        if ids:
            return ids[:best_count] if best_count else ids, attempts
        if best_count == 0:
            return [], attempts
    except Exception as exc:
        attempts.append(f'no args: {type(exc).__name__}: {exc}')

    # The official e3series wrapper should already return IDs as an output
    # tuple; COM VARIANT objects are for direct pywin32 only.
    if is_official_wrapper:
        return [], attempts

    # E3's VBScript signatures often use a byref output array. Different
    # versions/type-library registrations need different Python VARIANTs.
    for type_flags, seed, desc in (
        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None, 'BYREF VARIANT'),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, [], 'BYREF LONG ARRAY'),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_VARIANT, [], 'BYREF VARIANT ARRAY'),
    ):
        try:
            arg = win32.VARIANT(type_flags, seed)
            raw = fn(arg)
            count, embedded = count_and_ids(raw)
            out_ids = collect_ids(arg.value)
            # COM output can also be placed in the returned tuple.
            ids = embedded or out_ids
            best_count = max(best_count, max(0, count or 0))
            attempts.append(f'{desc}: return={display(raw)}, out={display(arg.value)}')
            if ids:
                return ids[:best_count] if best_count else ids, attempts
        except Exception as exc:
            attempts.append(f'{desc}: {type(exc).__name__}: {exc}')
    return [], attempts


def get_com_names(obj):
    names = set()
    try:
        info = obj._oleobj_.GetTypeInfo()
        attr = info.GetTypeAttr()
        for i in range(attr.cFuncs):
            info_fn = info.GetFuncDesc(i)
            name_set = info.GetNames(info_fn.memid)
            if name_set:
                names.add(str(name_set[0]))
    except Exception:
        pass
    try:
        names.update(n for n in dir(obj) if not n.startswith('_'))
    except Exception:
        pass
    return names


def valid_property(value):
    if value is None or value == '':
        return False
    if isinstance(value, int) and value == MISSING_COM_PARAM:
        return False
    if isinstance(value, (tuple, list)):
        # Most scalar getter functions should not return a multi-value tuple.
        # An output-marker tuple is not a format value to copy.
        if not value or MISSING_COM_PARAM in value:
            return False
    return isinstance(value, (str, bool, int, float)) and (not isinstance(value, float) or math.isfinite(value))


def valid_property_for(name, value):
    """Pattern getters can return (pattern, angle1, angle2) on some versions.

    E3 has several COM signatures; do not treat output placeholders as data.
    """
    if name == 'Hatch pattern' and isinstance(value, (tuple, list)):
        return len(value) == 3 and all(valid_property(item) for item in value)
    return valid_property(value)


def signature_hint(method):
    try:
        return str(inspect.signature(method))
    except (ValueError, TypeError, Exception):
        return '(signature not available through COM)'


def read_hatch_diagnostics(symbol):
    """Only read existing values/methods; no E3 writes."""
    lines = []
    names = get_com_names(symbol)
    for m in (
        'GetBlockHatchPattern', 'GetBlockHatchDistance',
        'GetBlockHatchColour', 'GetBlockHatchWidth', 'GetBlockHatchStyle',
    ):
        if names and m not in names:
            lines.append(f'{m}: NOT EXPOSED')
            continue
        try:
            raw = getattr(symbol, m)()
            lines.append(f'{m}() => {display(raw)}')
        except Exception as e:
            lines.append(f'{m}() => {type(e).__name__}: {e}')
    for m in (
        'SetBlockHatchPattern', 'SetBlockHatchDistance',
        'SetBlockHatchColour', 'SetBlockHatchWidth', 'SetBlockHatchStyle',
    ):
        if names and m not in names:
            lines.append(f'{m}: NOT EXPOSED')
            continue
        try:
            lines.append(f'{m}{signature_hint(getattr(symbol, m))}')
        except Exception as e:
            lines.append(f'{m}: signature lookup failed: {e}')
    for m in ('GetType', 'GetSymbolType', 'GetName', 'IsBlock'):
        if m in names:
            try:
                lines.append(f'{m}() => {display(getattr(symbol, m)())}')
            except Exception as e:
                lines.append(f'{m}() => {type(e).__name__}: {e}')
    return lines


def invoke_format_setter(obj, name, setter, value):
    """Don't guess hatch flags or angles. Use exactly the source's values."""
    method = getattr(obj, setter)
    if name == 'Hatch pattern' and isinstance(value, (tuple, list)):
        return method(*value)
    return method(value)


def is_colour(name):
    """Use property name, not numeric type, to interpret an Automatic color."""
    return 'colour' in name.lower() or 'color' in name.lower()


def format_property(name, value):
    """Display -1 as a valid default/sentinel, not a failed selection."""
    if isinstance(value, (int, float)) and value == AUTO_COLOUR:
        if is_colour(name):
            return 'Automatic (-1)'
        return 'Default / inherited (-1)'
    return display(value)


def source_property_copyable(name, value):
    """Accept -1 defaults, including hatch targets/sources; reject other negative sizes."""
    # A -1 hatch pattern has not been confirmed to be a real selectable
    # pattern; it may indicate that no hatch exists or the symbol is
    # unsupported. Don't use it to deactivate another symbol's hatch.
    if name == 'Hatch pattern' and value == -1:
        return False
    # Other -1 default/Automatic values are allowed and verified after write.
    if isinstance(value, (int, float)) and value == -1:
        return True
    # Negative widths/heights other than -1 are not safe to apply.
    if isinstance(value, (int, float)) and value < 0 and any(
        word in name.lower() for word in ('width', 'height')
    ):
        return False
    return True


def same_value(a, b):
    if isinstance(a, (tuple, list)) or isinstance(b, (tuple, list)):
        return (isinstance(a, (tuple, list)) and isinstance(b, (tuple, list))
                and len(a) == len(b)
                and all(same_value(x, y) for x, y in zip(a, b)))
    if isinstance(a, (float, int)) and isinstance(b, (float, int)):
        return math.isclose(float(a), float(b), rel_tol=1e-5, abs_tol=1e-7)
    return a == b


def read_related_ids(obj, name, text_type=None, max_ids=800):
    """Read E3's COM out-array methods safely; works with e3series wrapper.

    Known E3 methods such as Symbol.GetPinIds and Pin.GetTextIds normally
    return (count, ids), NOT a plain ID. Never use the count as an ID.
    A text_type of None requests all objects (where supported).
    """
    names = get_com_names(obj)
    if names and name not in names:
        return [], f'{name} is not exposed'
    try:
        method = getattr(obj, name)
    except Exception as exc:
        return [], f'{name}: {type(exc).__name__}: {exc}'
    arguments = [()] if text_type is None else [
        (text_type,), (None, text_type), ([], text_type),
    ]
    errors = []
    for args in arguments:
        try:
            raw = method(*args)
            _count, ids = count_and_ids(raw)
            if ids:
                return list(dict.fromkeys(ids))[:max_ids], f'{name}{args}: {len(ids)} IDs'
            # Some wrapper variants expose the array without a leading count.
            if isinstance(raw, (tuple, list)) and raw and all(is_int_id(i) for i in raw):
                # If the method has one return scalar, it's probably a COUNT.
                if len(raw) > 1:
                    return list(dict.fromkeys(raw))[:max_ids], f'{name}{args}: raw ID array'
        except Exception as exc:
            errors.append(f'{type(exc).__name__}: {exc}')
    for flags, seed, label in (
        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None, 'BYREF VARIANT'),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, [], 'BYREF I4 ARRAY'),
    ):
        try:
            arr = win32.VARIANT(flags, seed)
            raw = method(arr) if text_type is None else method(arr, text_type)
            count, embedded = count_and_ids(raw)
            ids = embedded or collect_ids(arr.value)
            if ids:
                return list(dict.fromkeys(ids))[:max_ids], f'{name} {label}: {len(ids)} IDs'
        except Exception as exc:
            errors.append(f'{label}: {type(exc).__name__}: {exc}')
    return [], f'{name}: no object IDs returned' + (f' ({errors[0]})' if errors else '')


def safe_get_scalar(obj, method_names):
    names = get_com_names(obj)
    for method in method_names:
        if names and method not in names:
            continue
        try:
            value = getattr(obj, method)()
        except Exception:
            continue
        if isinstance(value, (str, int, float)) and value != MISSING_COM_PARAM and value is not None:
            return value
    return None


def equal_text_format(a, b):
    """Source pin text is uniform if its appearance matches (ignore text IDs)."""
    fields = ('Font name', 'Text height', 'Font style', 'Text alignment', 'Text ratio / mode')
    return all((name not in a.values and name not in b.values) or
               (name in a.values and name in b.values and same_value(a.values[name][2], b.values[name][2]))
               for name in fields)

class E3Connection:
    def __init__(self):
        self.backend = 'pywin32'
        self.app = None
        self.job = None
        self.details = []
        self.type_config = dict(KNOWN_TYPE_CONFIG)
        try:
            import e3series
            app = e3series.Application()
            job = app.CreateJobObject()
            if job.GetId():
                self.backend = 'official e3series wrapper'
                self.app, self.job = app, job
                return
            self.details.append('Official wrapper connected, but returned no active project ID.')
        except ImportError:
            self.details.append('Optional e3series wrapper is not installed.')
        except Exception as exc:
            self.details.append(f'Official wrapper did not connect: {exc}')
        self.app = win32.Dispatch('CT.Application')
        self.job = self.app.CreateJobObject()

    def preferred_text_types(self):
        return PREFERRED_TEXT_TYPES

    def type_name(self, typ):
        return text_type_description(typ, self.type_config)

    def job_id(self):
        return self.job.GetId()

    def read_selections(self):
        selections = {}
        diagnostics = {}
        for cat, selection_method, creator, attrs in CATEGORIES:
            try:
                ids, attempted = get_selected(
                    self.job, selection_method,
                    is_official_wrapper=(self.backend == 'official e3series wrapper'),
                )
                selections[cat] = ids
                diagnostics[cat] = attempted
            except Exception as exc:
                selections[cat] = []
                diagnostics[cat] = [f'{type(exc).__name__}: {exc}']
        try:
            result = self.job.GetSelectedBlockCount()
            diagnostics['block count'] = [repr(result)]
        except Exception:
            pass
        return selections, diagnostics

    def object_for(self, category, object_id):
        entry = next((c for c in CATEGORIES if c[0] == category), None)
        if entry is None:
            raise ValueError(f'Unsupported category: {category}')
        obj = getattr(self.job, entry[2])()
        obj.SetId(int(object_id))
        return obj, entry[3]

    def read_properties(self, category, object_id):
        obj, formats = self.object_for(category, object_id)
        names = get_com_names(obj)
        values = {}
        notes = []
        for label, getter, setter in formats:
            # We need BOTH getter and setter for true match-properties.
            if names and (getter not in names or setter not in names):
                notes.append(f'{label}: {getter}/{setter} not both exposed')
                continue
            try:
                value = getattr(obj, getter)()
                if valid_property_for(label, value):
                    values[label] = (getter, setter, value)
                else:
                    notes.append(f'{label}: getter yielded unusable {display(value)}')
            except Exception as exc:
                notes.append(f'{label}: {getter} failed: {exc}')
        return values, notes

    def related_text_ids(self, symbol_id):
        """Find actual symbol-owned text, including optional project-specific types.

        The symbol's COM GetTextIds may require the explicit type filter, or a
        by-reference output array, depending on the wrapper/build. The result
        includes ONLY text IDs returned for the selected symbol. No sheet-wide
        text is matched based on proximity or rendered text content.
        """
        symbol, _ = self.object_for('symbol', symbol_id)
        names = get_com_names(symbol)
        notes = []
        all_ids = []
        typed_ids = defaultdict(list)
        self._tagged_types = {}

        preferred = self.preferred_text_types()
        if names and 'GetTextIds' not in names:
            notes.append('Symbol.GetTextIds is NOT EXPOSED on this E3 version.')
        else:
            fn = getattr(symbol, 'GetTextIds')
            for typ in (*preferred, None):
                desc = f'text type #{typ}' if typ is not None else 'all text'
                if typ is None:
                    attempts = [('no args', ())]
                else:
                    attempts = [
                        ('type only', (typ,)),
                        ('None output and type', (None, typ)),
                        ('empty array and type', ([], typ)),
                    ]
                found = []
                for strategy, args in attempts:
                    try:
                        raw = fn(*args)
                        count, ids = count_and_ids(raw)
                        notes.append(f'Symbol {symbol_id} GetTextIds {desc} '
                                     f'[{strategy}]: {display(raw)} -> count {count}, IDs {ids[:30]}')
                        found.extend(ids)
                        if ids:
                            break  # avoid extra COM probes once IDs were decoded
                    except Exception as exc:
                        notes.append(f'Symbol {symbol_id} GetTextIds {desc} '
                                     f'[{strategy}]: {type(exc).__name__}: {exc}')
                # The original probe succeeded for device designation via
                # a BYREF array; retain this tested fallback for all types.
                if not found:
                    for flags, seed, variant_desc in (
                        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None, 'BYREF VARIANT'),
                        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, [], 'BYREF LONG ARRAY'),
                    ):
                        try:
                            arg = win32.VARIANT(flags, seed)
                            raw = fn(arg) if typ is None else fn(arg, typ)
                            cnt, embedded = count_and_ids(raw)
                            ids = embedded or collect_ids(arg.value)
                            notes.append(f'Symbol {symbol_id} GetTextIds {desc} '
                                         f'[{variant_desc}]: count={cnt} IDs={ids[:30]}')
                            found.extend(ids)
                            if ids:
                                break
                        except Exception as exc:
                            notes.append(f'Symbol {symbol_id} GetTextIds {desc} '
                                         f'[{variant_desc}]: {type(exc).__name__}: {exc}')
                if typ is None:
                    all_ids.extend(found)
                else:
                    typed_ids[typ].extend(found)
                    if found:
                        notes.append(f'  {self.type_name(typ)} (#{typ}): {len(set(found))} attached text(s)')

        # Explicit text-type filters take priority over text.GetType() because
        # some E3 builds report a generic object kind through GetType().
        for typ in preferred:
            for tid in typed_ids[typ]:
                previous = self._tagged_types.get(tid)
                if previous is not None and previous != typ:
                    notes.append(f'WARNING: text ID {tid} appeared under two text types '
                                 f'({previous}, {typ}); retaining first type.')
                    continue
                self._tagged_types[tid] = typ
        dedup = list(dict.fromkeys(
            [id_ for typ in preferred for id_ in typed_ids[typ]] + all_ids
        ))
        if dedup:
            return dedup[:300], notes

        # Existing verified-parent-only fallback remains available.
        job_names = get_com_names(self.job)
        if 'GetTextIds' in job_names:
            try:
                project_ids, attempts = get_selected(
                    self.job, 'GetTextIds',
                    is_official_wrapper=(self.backend == 'official e3series wrapper'))
                notes.extend(f'Job.GetTextIds: {a}' for a in attempts)
                if len(project_ids) > 3000:
                    notes.append(f'Job text scan has {len(project_ids)} entries; safety cap 3000.')
                    return [], notes
                attached = []
                for tid in project_ids:
                    txt, _ = self.object_for('text', tid)
                    txt_names = get_com_names(txt)
                    for owner_method in ('GetSymbolId', 'GetOwnerSymbolId', 'GetParentSymbolId'):
                        if owner_method not in txt_names:
                            continue
                        try:
                            owner = getattr(txt, owner_method)()
                            if is_int_id(owner):
                                if int(owner) == int(symbol_id):
                                    attached.append(tid)
                                break
                        except Exception:
                            continue
                if attached:
                    notes.append(f'Found {len(attached)} texts via verified parent symbol IDs.')
                    return list(dict.fromkeys(attached)), notes
            except Exception as exc:
                notes.append(f'Job-scoped owner fallback failed: {exc}')

        notes.append('WARNING: NO ATTACHED TEXT IDs FOUND. Text size/format is '
                     'UNVERIFIED, not identical. Hatch/outline matching still works.')
        notes.append('Symbol text methods: ' + repr(sorted(
            n for n in names if 'text' in n.lower() or 'device' in n.lower())))
        return [], notes

    def block_connector_pin_ids(self, symbol_id, notes):
        """Discover the selected block's pins and attached connector pins.

        Every returned pin must be linked by an API ownership relationship:
        Symbol.GetPinIds, Device.GetPinIds (via symbol's device),
        Block.GetPinIds / Block.GetConnectorIds (via selected block), or
        a connector returned by Block.GetConnectorIds.
        Do NOT scan unrelated connector devices on the sheet.
        """
        symbol, _ = self.object_for('symbol', symbol_id)
        pin_ids = []
        owners = []
        linked_connector_ids = []
        direct_connector_symbols = []
        for method in ('GetPinIds', 'GetAllPinIds'):
            ids, how = read_related_ids(symbol, method)
            if ids:
                notes.append(f'  BLOCK SYMBOL {symbol_id}: {how}')
                pin_ids.extend(ids)
        # On some E3 versions a block symbol can report its own connector
        # devices/symbols without looking up the enclosing block object.
        for method in ('GetConnectorIds', 'GetBlockConnectorIds'):
            ids, how = read_related_ids(symbol, method)
            if ids:
                notes.append(f'  BLOCK SYMBOL {symbol_id}: {how}')
                linked_connector_ids.extend(ids)

        # From a symbol to its owning device. This is a much stronger link
        # than searching all devices on the drawing or by proximity.
        dev_id = safe_get_scalar(symbol, ('GetDeviceId', 'GetOwnerDeviceId'))
        if is_int_id(dev_id):
            try:
                dev = self.job.CreateDeviceObject()
                dev.SetId(int(dev_id))
                owners.append(('device', int(dev_id), dev))
            except Exception as exc:
                notes.append(f'  Owner device {dev_id} could not be opened: {exc}')

        # E3's GetSelectedBlockIds returns the block selected on the drawing.
        # Only use it when this symbol is the unique selected symbol, so a
        # selected GROUP of blocks cannot be crossed accidentally.
        block_ids, _ = get_selected(self.job, 'GetSelectedBlockIds',
                                   is_official_wrapper=self.backend == 'official e3series wrapper') \
            if 'GetSelectedBlockIds' in get_com_names(self.job) else ([], [])
        selected_symbol_ids, _ = get_selected(self.job, 'GetSelectedSymbolIds',
                                    is_official_wrapper=self.backend == 'official e3series wrapper') \
            if 'GetSelectedSymbolIds' in get_com_names(self.job) else ([], [])
        candidate_block = safe_get_scalar(symbol, ('GetBlockId', 'GetParentBlockId'))
        if not is_int_id(candidate_block) and len(block_ids) == len(selected_symbol_ids) == 1 \
                and selected_symbol_ids[0] == int(symbol_id):
            candidate_block = block_ids[0]
        if is_int_id(candidate_block):
            # Many E3 installations represent blocks as *devices*, rather
            # than exposing a separate CreateBlockObject COM class.
            if not any(oid == int(candidate_block) for _,oid,_ in owners):
                try:
                    block_device = self.job.CreateDeviceObject()
                    block_device.SetId(int(candidate_block))
                    owners.append(('block-device', int(candidate_block), block_device))
                    notes.append(f'  Block device {candidate_block} linked to symbol {symbol_id}')
                except Exception as exc:
                    notes.append(f'  Block-device {candidate_block}: {exc}')
            creator = getattr(self.job, 'CreateBlockObject', None)
            if callable(creator):
                try:
                    block = creator()
                    block.SetId(int(candidate_block))
                    owners.append(('block', int(candidate_block), block))
                    notes.append(f'  Selected block ID {candidate_block} linked to symbol {symbol_id}')
                except Exception as exc:
                    notes.append(f'  Block {candidate_block} could not be opened: {exc}')

        # Pins directly attached to the block or its owning device.
        for owner_kind, owner_id, owner in owners:
            for method in ('GetPinIds', 'GetAllPinIds'):
                ids, how = read_related_ids(owner, method)
                if ids:
                    notes.append(f'  {owner_kind} {owner_id}: {how}')
                    pin_ids.extend(ids)
            # These methods, when exposed, report connectors belonging to
            # THIS block. Do not use generic project-wide device discovery.
            if owner_kind in ('block', 'block-device', 'device'):
                for method in ('GetConnectorIds', 'GetBlockConnectorIds', 'GetSubDeviceIds'):
                    ids, how = read_related_ids(owner, method)
                    if ids:
                        notes.append(f'  {owner_kind} {owner_id}: {how}')
                        linked_connector_ids.extend(ids)
        for conn_id in dict.fromkeys(linked_connector_ids):
            try:
                conn = self.job.CreateDeviceObject()
                conn.SetId(int(conn_id))
                for method in ('GetPinIds', 'GetAllPinIds'):
                    ids, how = read_related_ids(conn, method)
                    if ids:
                        notes.append(f'  block connector {conn_id}: {how}')
                        pin_ids.extend(ids)
                child_sids, how = read_related_ids(conn, 'GetSymbolIds')
                for sid in child_sids:
                    # Only collect connector symbols directly attached to the
                    # selected block's devices, NEVER from the whole sheet.
                    if sid != symbol_id:
                        direct_connector_symbols.append((conn_id, sid))
                if child_sids:
                    notes.append(f'  block connector {conn_id}: {how}')
            except Exception as exc:
                notes.append(f'  Could not inspect block connector {conn_id}: {exc}')
        dedup = list(dict.fromkeys(pin_ids))[:500]
        notes.append(f'  PIN INVENTORY FOR SYMBOL {symbol_id}: {len(dedup)} linked pins')
        if not dedup:
            notes.append('  WARNING: no block/connector pin IDs were found; '
                         'pin-owned designation and local signal formatting cannot be compared yet.')
            for method in ('GetPinIds', 'GetAllPinIds', 'GetBlockId', 'GetDeviceId'):
                notes.append(f'  Symbol.{method}: {method in get_com_names(symbol)}')
        notes.append(f'  LINKED CONNECTOR SYMBOLS: {len(direct_connector_symbols)}')
        return dedup, list(dict.fromkeys(direct_connector_symbols))[:200]

    def pin_owned_text_ids(self, symbol_id, notes, existing_ids):
        """Read pin-local texts (#7/#12), not just the block symbol text.

        Some E3 projects store visible labels directly on a pin; others keep
        them on the pin's placed symbol. We query both confirmed relationships.
        No region-based fuzzy search and no changes to pin records.
        """
        pin_ids, connector_symbols = self.block_connector_pin_ids(symbol_id, notes)
        tagged = {}
        text_owner = {}
        owner_names = {}
        parent_symbol, _ = self.object_for('symbol', symbol_id)
        parent_sheet = safe_get_scalar(parent_symbol, ('GetSheetId',))
        for ordinal, pin_id in enumerate(pin_ids):
            try:
                pin = self.job.CreatePinObject()
                pin.SetId(int(pin_id))
            except Exception as exc:
                notes.append(f'  Pin {pin_id}: cannot open: {exc}')
                continue
            pin_name = safe_get_scalar(pin, ('GetName', 'GetPinName'))
            owner_names[pin_id] = pin_name
            # The on-pin label can be type 7 or 12. Only copy these from
            # corresponding text that actually exists on the destination.
            for typ in (SIGNAL_NAME_TYPE, DEVICE_DESIGNATION_TYPE):
                ids, how = read_related_ids(pin, 'GetTextIds', text_type=typ)
                if ids:
                    notes.append(f'  PIN {pin_id} {pin_name or ""}: {how} [type {typ}]')
                for tid in ids:
                    tagged[tid] = typ
                    text_owner[tid] = (pin_id, pin_name, ordinal)
            # If the E3 version cannot filter, try the all-text query and
            # inspect GetTextType on each returned text later.
            ids, _ = read_related_ids(pin, 'GetTextIds')
            for tid in ids:
                if tid not in text_owner:
                    text_owner[tid] = (pin_id, pin_name, ordinal)
            # Some pins have their labels stored on a distinct placed symbol.
            child_symbol_ids, how = read_related_ids(pin, 'GetSymbolIds')
            direct_child = safe_get_scalar(pin, ('GetSymbolId',))
            if is_int_id(direct_child):
                child_symbol_ids.append(int(direct_child))
            for sid in list(dict.fromkeys(child_symbol_ids))[:25]:
                if sid == symbol_id:
                    continue
                try:
                    child = self.job.CreateSymbolObject()
                    child.SetId(int(sid))
                    child_sheet = safe_get_scalar(child, ('GetSheetId',))
                    if parent_sheet is not None and child_sheet is not None \
                            and child_sheet != parent_sheet:
                        continue  # another view/sheet of the same connector
                    # Only accept a linked placed pin symbol. Avoid reading
                    # other symbols of the connector/device or all sheet text.
                    for typ in (SIGNAL_NAME_TYPE, DEVICE_DESIGNATION_TYPE):
                        tids, detail = read_related_ids(child, 'GetTextIds', text_type=typ)
                        for tid in tids:
                            tagged[tid] = typ
                            text_owner[tid] = (pin_id, pin_name, ordinal)
                        if tids:
                            notes.append(f'  PIN SYMBOL {sid} (pin {pin_id}): {detail} type {typ}')
                except Exception as exc:
                    notes.append(f'  Pin {pin_id} symbol {sid}: {exc}')
        # A connector pin-view label can belong to the connector symbol
        # rather than the Pin COM object. The connector must be reached from
        # this block's GetConnectorIds/related device; same sheet when known.
        for connector_id, sid in connector_symbols:
            try:
                child = self.job.CreateSymbolObject()
                child.SetId(int(sid))
                child_sheet = safe_get_scalar(child, ('GetSheetId',))
                if parent_sheet is not None and child_sheet is not None and child_sheet != parent_sheet:
                    continue
                for typ in (SIGNAL_NAME_TYPE, DEVICE_DESIGNATION_TYPE):
                    tids, how = read_related_ids(child, 'GetTextIds', text_type=typ)
                    for tid in tids:
                        if tid not in existing_ids:
                            text_owner[tid] = (connector_id, f'connector {connector_id}', None)
                            tagged[tid] = typ
                    if tids:
                        notes.append(f'  BLOCK CONNECTOR SYMBOL {sid}: {how}; type {typ}')
            except Exception as exc:
                notes.append(f'  Connector symbol {sid} (device {connector_id}): {exc}')
        notes.append(f'  PIN TEXT INVENTORY: {len(text_owner)} pin/connector-owned text IDs; '
                     f'{sum(t == 7 for t in tagged.values())} signal names (#7), '
                     f'{sum(t == 12 for t in tagged.values())} device designations (#12).')
        return text_owner, tagged

    def read_symbol_texts(self, symbol_id):
        """Snapshot text styles and E3 text-type keys, never contents/locations."""
        ids, notes = self.related_text_ids(symbol_id)
        pin_owners, pin_types = self.pin_owned_text_ids(symbol_id, notes, set(ids))
        # Prefer a verified PIN owner if the same text is discoverable via
        # both symbol and pin API paths (do not write the same text twice).
        ids = list(dict.fromkeys(list(ids) + list(pin_owners)))
        notes.append(f'TOTAL MATCHABLE TEXTS: {len(ids)} including {len(pin_owners)} pin-related.')
        entries = []
        parent_symbol, _ = self.object_for('symbol', symbol_id)
        parent_sheet = safe_get_scalar(parent_symbol, ('GetSheetId',))
        for tid in ids:
            try:
                txt, _ = self.object_for('text', tid)
                names = get_com_names(txt)
                # A device pin can exist in multiple views. When E3 exposes
                # sheet IDs for both sides, never format another sheet's pin
                # text simply because it belongs to the same connector device.
                txt_sheet = safe_get_scalar(txt, ('GetSheetId',))
                if (is_int_id(parent_sheet) and is_int_id(txt_sheet)
                        and int(parent_sheet) != int(txt_sheet)):
                    notes.append(f'Text {tid}: SKIPPED (different sheet {txt_sheet}; '
                                 f'block is on {parent_sheet}).')
                    continue
                text_type = None
                for method in ('GetTextType', 'GetType'):
                    if names and method not in names:
                        continue
                    try:
                        v = getattr(txt, method)()
                        if valid_property(v) and is_int_id(v):
                            text_type = v
                            break
                    except Exception:
                        pass
                # A symbol-scoped filtered lookup is stronger evidence than
                # the text object's generic GetType() method in some builds.
                tagged_type = pin_types.get(tid, getattr(self, '_tagged_types', {}).get(tid))
                if tagged_type is not None:
                    if text_type != tagged_type:
                        notes.append(f'Text {tid}: getter returned type {text_type!r}; '
                                     f'symbol filtered lookup identifies {self.type_name(tagged_type)}.')
                    text_type = tagged_type
                values, prop_notes = self.read_properties('text', tid)
                attribute_name = read_attribute_name(txt, names) if text_type == ATTRIBUTE_TYPE else None
                if attribute_name:
                    notes.append(f'Text {tid}: type-1000 attribute name is {attribute_name!r}.')
                elif text_type == ATTRIBUTE_TYPE:
                    notes.append(f'Text {tid}: generic Attribute #1000 has no readable attribute name; '
                                 'only unambiguous one-to-one alias matching is allowed.')
                if values:
                    pin_data = pin_owners.get(tid)
                    if pin_data:
                        entries.append(ChildText(tid, text_type, values, attribute_name,
                                                 owner_scope='pin', pin_id=pin_data[0],
                                                 pin_name=pin_data[1], pin_ordinal=pin_data[2]))
                        notes.append(f'PIN TEXT {tid}: owner pin {pin_data[0]}, '
                                     f'name={pin_data[1]!r}, ordinal={pin_data[2]}, '
                                     f'type={text_type}, height={values.get("Text height",(None,None,None))[2]!r}')
                    else:
                        entries.append(ChildText(tid, text_type, values, attribute_name))
                    if text_type in self.preferred_text_types():
                        notes.append(f'{self.type_name(text_type).upper()} TEXT ID {tid}: '
                                     f'GetHeight={values.get("Text height", (None,None,"UNAVAILABLE"))[2]!r}, '
                                     f'font={values.get("Font name", (None,None,"UNAVAILABLE"))[2]!r}, '
                                     f'style={values.get("Font style", (None,None,"UNAVAILABLE"))[2]!r}')
                else:
                    notes.append(f'Text {tid}: no editable format properties.')
                notes.extend(f'Text {tid}: {x}' for x in prop_notes)
            except Exception as exc:
                notes.append(f'Text {tid}: could not inspect: {exc}')
        return entries, notes

    def hatch_diagnostics(self, object_id):
        obj, _ = self.object_for('symbol', object_id)
        return read_hatch_diagnostics(obj)

    def close(self):
        """Release E3 COM references as soon as this button action finishes."""
        self.job = None
        self.app = None
        gc.collect()


def release_e3_after_action(callback):
    """Never keep an E3 automation session alive between GUI button presses."""
    @wraps(callback)
    def wrapped(self, *args, **kwargs):
        try:
            return callback(self, *args, **kwargs)
        finally:
            if self.conn is not None:
                try:
                    self.conn.close()
                finally:
                    self.conn = None
    return wrapped

    def palette_changed(self):
        # Prevent using a preview created with a different hatch override.
        self.reset_preview()
        try:
            index = palette_override(self.palette_selection.get())
            if index is None:
                summary = 'Copy source hatch color'
            else:
                summary = f'{PALETTE_BY_INDEX[str(index)]} (E3 color index {index})'
            self.status.set(f'Hatch color: {summary}. Preview targets to apply this choice.')
        except ValueError as exc:
            self.status.set(str(exc))

    @staticmethod
    def read_type_config():
        # No editable text fields and no Learn Type action. All E3 IDs are
        # automatically available from the supplied text-type table.
        return dict(KNOWN_TYPE_CONFIG)

    def return_to_e3(self):
        # No grab_set(), window disable, or always-on-top state is ever used.
        # Minimize so the E3 drawing area can receive clicks again.
        self.root.iconify()

    def on_close(self):
        if self.conn is not None:
            self.conn.close()
            self.conn = None
        self.root.destroy()

    def reset_preview(self):
        self.preview = None
        self.apply_button.configure(state='disabled')

    def connect(self):
        try:
            # Defense in depth: no old COM job should survive across actions.
            if self.conn is not None:
                self.conn.close()
                self.conn = None
            self.conn = E3Connection()
            if not self.conn.job_id():
                raise RuntimeError('E3 is connected, but no active project is open.')
            return self.conn
        except Exception as exc:
            self.write(f'CONNECTION ERROR: {exc}\n{traceback.format_exc()}')
            self.status.set('Connection error - see copyable log.')
            messagebox.showerror('Connection error', f'Could not connect to an open E3 project:\n{exc}')
            return None

    def selection_report(self, conn, verbose=False):
        selected, diagnostics = conn.read_selections()
        self.write(f'Project ID: {conn.job_id()}, COM backend: {conn.backend}')
        for name, ids in selected.items():
            self.write(f'  {name}: {len(ids)} valid IDs: {ids[:20]}')
            if verbose:
                for attempt in diagnostics.get(name, []):
                    self.write('    ' + attempt)
        if verbose:
            for extra in conn.details:
                self.write(extra)
            if 'block count' in diagnostics:
                self.write('GetSelectedBlockCount: ' + str(diagnostics['block count']))
        if not any(selected.values()):

            self.write('Try installing the OFFICIAL E3.series Python package for your E3 version: py -m pip install e3series')
            self.write('Click Copy Log and share these diagnostics, rather than guessing IDs.')

    @staticmethod

    def find_targets(self, conn):
        if not self.source:
            raise ValueError('Copy a source first.')
        if conn.job_id() != self.source.job_id:
            raise RuntimeError('The active E3 project changed. Copy Source again in the current project.')
        selected = self.selection_report(conn)
        ids = [i for i in selected.get(self.source.category, []) if i != self.source.object_id]
        # Skip the source if it remains among selected objects.
        if not ids:
            raise ValueError(f'Select at least one OTHER {self.source.category} as a target.')
        if len(ids) > 100:
            raise ValueError('More than 100 targets were selected. Reduce selection for a safer test.')
        return list(dict.fromkeys(ids))

    def find_matching_child_source(self, target, source_by_type, dest_by_type, source_by_role, dest_by_role):
        """Prefer identical E3 type, allow Equipment ID <-> Additional Part ID.

        Never map unrelated Saber parameters or pin labels by list order.
        """
        ttype = target.text_type
        exact_source = source_by_type[ttype]
        exact_target = dest_by_type[ttype]
        if ttype == ATTRIBUTE_TYPE:
            # Many E3 text nodes can all have type 1000 (generic Attribute).
            # Pair identical READABLE attribute names, never by list position.
            dst_name = normalize_attribute_name(target.attribute_name)
            if dst_name:
                named_source = [s for s in exact_source
                                if normalize_attribute_name(s.attribute_name) == dst_name]
                named_target = [t for t in exact_target
                                if normalize_attribute_name(t.attribute_name) == dst_name]
                if len(named_source) == len(named_target) == 1:
                    return named_source[0], 'same named attribute'
                if named_source or named_target:
                    return None, 'ambiguous repeated attribute name'
            # If the attribute name is known but NOT equipment ID, it should
            # never be aliased to an unrelated standard Additional Part field.
            if dst_name and not equipment_attribute_name(dst_name):
                return None, 'type-1000 attribute name differs from source'
        if ttype is not None and len(exact_source) == len(exact_target) == 1:
            if ttype == ATTRIBUTE_TYPE:
                src_name = normalize_attribute_name(exact_source[0].attribute_name)
                dst_name = normalize_attribute_name(target.attribute_name)
                if src_name and dst_name and src_name != dst_name:
                    return None, 'different named type-1000 attributes'
                if src_name and not equipment_attribute_name(src_name):
                    return None, 'source #1000 is a different named attribute'
                if dst_name and not equipment_attribute_name(dst_name):
                    return None, 'target #1000 is a different named attribute'
            return exact_source[0], 'exact type'
        role = text_role(ttype, self.source.type_config, target.attribute_name)
        if role == 'Equipment / Additional Part ID':
            group_sources = source_by_role[role]
            group_targets = dest_by_role[role]
            if len(group_sources) == len(group_targets) == 1:
                return group_sources[0], 'equivalent Equipment ID / Additional Part ID'
        return None, 'unmatched or ambiguous'

    def match_pin_text_source(self, target, src_pin_texts, dst_pin_texts):
        """Match by pin + type; if source style uniform, broadcast that style.

        Top pin labels often have different pin names/IDs across blocks. The
        user wants their *formatting*, not signal/device IDs. If all source
        pin labels of this type have the same format, there is one clear style
        template for each corresponding target pin text of the same type.
        If the source has varied formatting, use exact pin-name equality, then
        the E3 symbol pin order only when both pin inventories line up.
        """
        typ = target.text_type
        source = [e for e in src_pin_texts if e.text_type == typ]
        dest = [e for e in dst_pin_texts if e.text_type == typ]
        if not source:
            return None, 'source has no pin-owned text of this type'
        # Strongest identity: pin names are unique on both source & target.
        name = (target.pin_name or '').strip().lower() if isinstance(target.pin_name, str) else ''
        if name:
            named_src = [e for e in source if isinstance(e.pin_name,str) and e.pin_name.strip().lower() == name]
            named_dst = [e for e in dest if isinstance(e.pin_name,str) and e.pin_name.strip().lower() == name]
            if len(named_src) == len(named_dst) == 1:
                return named_src[0], f'pin name {target.pin_name!r}'
        # For a consistent source pin-text style, apply the source font to
        # ALL destination pin labels of the matching type, regardless of pin
        # number/name/count. This is the primary fix for top pins.
        if all(equal_text_format(source[0], e) for e in source[1:]):
            return source[0], f'uniform source pin style ({len(source)} type-{typ} texts)'
        # The pin order is meaningful in E3's symbol definition. Only use it
        # if BOTH sides have the same number of *distinct* labeled pins,
        # and there is exactly one text of this type on each matching pin.
        src_orders = [e.pin_ordinal for e in source]
        dst_orders = [e.pin_ordinal for e in dest]
        if (len(source) == len(dest) and len(set(src_orders)) == len(source)
                and len(set(dst_orders)) == len(dest)
                and None not in src_orders and None not in dst_orders):
            src_sorted = sorted(source, key=lambda e:e.pin_ordinal)
            dst_sorted = sorted(dest, key=lambda e:e.pin_ordinal)
            key = next((i for i, e in enumerate(dst_sorted) if e.object_id == target.object_id), None)
            if key is not None:
                return src_sorted[key], 'same position in ordered pin list'
        return None, 'source pin styles differ and there is no unambiguous pin match'

    def plan_symbol_texts(self, conn, symbol_id, black):
        """Pair unique exact E3 text types or Equipment/Part ID aliases; force black."""
        target_texts, notes = conn.read_symbol_texts(symbol_id)
        for note in notes:
            self.write(f'  Symbol {symbol_id} text discovery: {note}')
        # Block symbol text and connector/pin text are DIFFERENT owners.
        # A block-level designation must not masquerade as the font on a
        # top connector pin. Keep the two scopes separate.
        src_pin_texts = [e for e in self.source.child_texts if e.owner_scope == 'pin']
        dst_pin_texts = [e for e in target_texts if e.owner_scope == 'pin']
        src_symbol_texts = [e for e in self.source.child_texts if e.owner_scope != 'pin']
        dst_symbol_texts = [e for e in target_texts if e.owner_scope != 'pin']
        self.write(f'  PIN FONT INVENTORY symbol {symbol_id}: '
                   f'source {len(src_pin_texts)} pin labels, target {len(dst_pin_texts)} pin labels.')
        for typ in (SIGNAL_NAME_TYPE, DEVICE_DESIGNATION_TYPE):
            a = [e for e in src_pin_texts if e.text_type == typ]
            b = [e for e in dst_pin_texts if e.text_type == typ]
            self.write(f'  TOP/CONNECTOR PIN TEXT TYPE {typ}: source {len(a)}, target {len(b)}; '
                       'match only if both contain the type.')
        src_by_type = defaultdict(list)
        dst_by_type = defaultdict(list)
        for e in src_symbol_texts:
            src_by_type[e.text_type].append(e)
        for e in dst_symbol_texts:
            dst_by_type[e.text_type].append(e)
        src_by_role = defaultdict(list)
        dst_by_role = defaultdict(list)
        for entry in src_symbol_texts:
            src_by_role[text_role(entry.text_type, self.source.type_config, entry.attribute_name)].append(entry)
        for entry in dst_symbol_texts:
            dst_by_role[text_role(entry.text_type, self.source.type_config, entry.attribute_name)].append(entry)
        planned = []
        for target in target_texts:
            # A type may occur several times, often with different positions;
            # never pair by list order or rendered text content (designations).
            if target.owner_scope == 'pin':
                source, why = self.match_pin_text_source(target, src_pin_texts, dst_pin_texts)
            else:
                source, why = self.find_matching_child_source(
                    target, src_by_type, dst_by_type, src_by_role, dst_by_role)
                if target.text_type is None and len(src_symbol_texts) == len(dst_symbol_texts) == 1:
                    source, why = src_symbol_texts[0], 'only source/target symbol text'
            if source is None:
                self.write(f'  Text ID {target.object_id}: {conn.type_name(target.text_type)} '
                           f'has no unique compatible source ({why}); only forcing black if needed.')
            changes = []
            for name, (getter, setter, current) in target.values.items():
                if name == 'Text colour':
                    desired = black  # ALWAYS black, even if source is Automatic or colored.
                elif source and name in source.values:
                    desired = source.values[name][2]
                else:
                    continue
                if not source_property_copyable(name, desired):
                    self.write(f'  Text ID {target.object_id}, {name}: skipped unsafe source value {desired!r}')
                    continue
                if not same_value(current, desired):
                    changes.append((name, getter, setter, current, desired))
            if changes:
                desc = conn.type_name(target.text_type)
                owner = (f'PIN {target.pin_name or target.pin_id}' if target.owner_scope == 'pin' else 'BLOCK')
                self.write(f'  Symbol {symbol_id}, {owner}, text ID {target.object_id} ({desc}): {len(changes)} changes'
                           + (f'; matched source text {source.object_id} ({why})' if source else '; black only'))
                for name, _, _, before, after in changes:
                    self.write(f'    {name}: {format_property(name, before)} -> {format_property(name, after)}'
                               + (' (forced black index)' if name == 'Text colour' else ''))
                planned.append((symbol_id, target.object_id, changes))
            elif source:
                self.write(f'  Symbol {symbol_id}, text ID {target.object_id}: source font already matched; colour black.')
        for typ in conn.preferred_text_types():
            src_typed = [t for t in self.source.child_texts if t.text_type == typ]
            dst_typed = [t for t in target_texts if t.text_type == typ]
            label = conn.type_name(typ)
            self.write(f'  {label.upper()} CHECK symbol {symbol_id}: '
                       f'source={len(src_typed)} text(s), target={len(dst_typed)} text(s).')
            if len(src_typed) == len(dst_typed) == 1:
                if 'Text height' not in src_typed[0].values or 'Text height' not in dst_typed[0].values:
                    self.write(f'  WARNING: {label} exists, but text height getter/setter unavailable.')
            elif typ == DEVICE_DESIGNATION_TYPE:
                self.write(f'  WARNING: {label} size/format NOT VERIFIED '
                           f'(source {len(src_typed)}, target {len(dst_typed)}). '
                           'This does NOT mean the fonts match.')
            elif src_typed or dst_typed:
                self.write(f'  NOTE: {label}: unmatched/multiple nodes; preserving text, '
                           'only unique type pairs receive font styles. Text color still forced black.')
            else:
                self.write(f'  NOTE: {label}: not present in either symbol; nothing to copy.')
        for group, type_ids in KNOWN_TYPE_CONFIG.items():
            src_count = sum(t.text_type in type_ids for t in self.source.child_texts)
            dst_count = sum(t.text_type in type_ids for t in target_texts)
            self.write(f'  {group.upper()} CHECK: source={src_count}, target={dst_count} '
                       '(only existing compatible nodes are formatted).')
        if not target_texts:
            self.write(f'  WARNING: Symbol {symbol_id}: no attached texts were found; '
                       'designation and other text formatting were NOT compared.')
        return planned

    @release_e3_after_action
    def preview_target(self):
        self.reset_preview()
        self.write('\n' + '=' * 65 + '\nPREVIEW TARGETS (read-only)')
        conn = self.connect()
        if not conn:
            return
        try:
            ids = self.find_targets(conn)
        except Exception as exc:
            self.write('PREVIEW BLOCKED: ' + str(exc))
            self.status.set(str(exc))
            return
        black = BLACK_TEXT_COLOUR
        palette_choice = self.palette_selection.get()
        try:
            hatch_override = palette_override(palette_choice)
        except ValueError as exc:
            self.write('PREVIEW BLOCKED: ' + str(exc))
            self.status.set(str(exc))
            return
        if self.source.category == 'symbol':
            self.write('HATCH COLOR MODE: ' + (
                f'{PALETTE_BY_INDEX[str(hatch_override)]} (E3 #{hatch_override})'
                if hatch_override is not None else 'Copy source hatch color'))
        self.write(f'TEXT COLOR MODE: Always black (E3 index {BLACK_TEXT_COLOUR})')
        plan = []
        text_plan = []
        skipped_count = 0
        for oid in ids:
            try:
                values, notes = conn.read_properties(self.source.category, oid)
                if self.source.category == 'symbol':
                    self.write(f'ID {oid} TARGET HATCH DIAGNOSTICS (read-only):')
                    for line in conn.hatch_diagnostics(oid):
                        self.write('  ' + line)
                changes = []

                # -1 is not an indication of an invalid target. For hatch
                # fields, E3 may use -1 for Automatic/inherited/unconfigured.
                # Preview the difference and let E3's setter plus readback
                # decide whether a concrete style can actually be applied.

                for name, (getter, setter, source_val) in self.source.values.items():
                    if name not in values:
                        continue
                    # Source hatch pattern/style/distance are always copied.
                    # The chosen palette index only overrides hatch COLOUR.
                    if name == 'Hatch colour' and hatch_override is not None:
                        source_val = hatch_override
                    # For directly selected TEXT, enforce black even when
                    # its source/target text colors are equal but non-black.
                    if name == 'Text colour' and self.source.category == 'text':
                        source_val = BLACK_TEXT_COLOUR
                    if not source_property_copyable(name, source_val):
                        if not same_value(values[name][2], source_val):
                            skipped_count += 1
                        self.write(
                            f'ID {oid}, {name}: SKIPPED source is '
                            f'{format_property(name, source_val)}; no safe setter value known.'
                        )
                        continue
                    current_val = values[name][2]
                    if not same_value(current_val, source_val):
                        changes.append((name, getter, setter, current_val, source_val))
                if changes:
                    plan.append((oid, changes))
                    self.write(f'ID {oid}: {len(changes)} format changes:')
                    for name, getter, setter, before, after in changes:
                        self.write(f'  {name}: {format_property(name, before)} -> {format_property(name, after)}')
                else:
                    self.write(f'ID {oid}: no supported formatting differences.')
                for note in notes:
                    self.write(f'  Skipped: {note}')
                if self.source.category == 'symbol':
                    text_plan.extend(self.plan_symbol_texts(conn, oid, black))
            except Exception as exc:
                self.write(f'ID {oid}: could not inspect; will SKIP ({exc})')
        if not plan and not text_plan:
            if self.source.category == 'symbol':
                self.status.set('No VERIFIED format differences. If designation text was not found, '
                                'its font size was NOT compared. See the diagnostic warnings.')
                self.write('PREVIEW SUMMARY: No AVAILABLE changes; designation must be verified '
                           'separately unless source AND target type-12 text were read.')
            else:
                self.status.set(f'No changes to apply. {skipped_count} unmatched properties skipped (see log).')
                self.write(f'PREVIEW SUMMARY: 0 changes available; {skipped_count} changes skipped.')
            return
        self.preview = (conn.job_id(), self.source.category, self.source.object_id,
                        ids, plan, skipped_count, text_plan, black, palette_choice)
        self.apply_button.configure(state='normal')
        changes_count = sum(len(p[1]) for p in plan) + sum(len(c) for _, _, c in text_plan)
        self.status.set(
            f'Preview: {changes_count} changes on {len(ids)} target(s); '
            f'{skipped_count} unmatched properties skipped. Apply requires confirmation.'
        )
        self.write(f'PREVIEW SUMMARY: {changes_count} planned changes, {skipped_count} skipped.')

    @release_e3_after_action
    def apply(self):
        if not self.preview or not self.source:
            return
        project_id, cat, source_id, selected_ids, plan, skipped_count, text_plan, black, palette_choice = self.preview
        self.reset_preview()
        conn = self.connect()
        if not conn:
            return
        if conn.job_id() != project_id or self.source.object_id != source_id:
            self.write('APPLY BLOCKED: active project or copied source changed. Preview again.')
            return
        try:
            live_ids = self.find_targets(conn)
        except Exception as exc:
            self.write('APPLY BLOCKED: ' + str(exc))
            return
        if set(live_ids) != set(selected_ids):
            self.write('APPLY BLOCKED: selected target IDs differ from preview. Preview again.')
            return
        if self.palette_selection.get() != palette_choice:
            self.write('APPLY BLOCKED: Palette choice changed since Preview. Preview again.')
            self.status.set('Hatch palette selection changed; Preview Targets again.')
            return
        n = sum(len(changes) for _, changes in plan) + sum(len(changes) for _, _, changes in text_plan)
        ok = messagebox.askyesno(
            'Confirm formatting change',
            f'Apply {n} formatting changes to {len(selected_ids)} selected {cat} target(s)?\n\n'
            'Text contents, object positions and device designations will NOT be changed.\n'
            'Matching compatible text types (including uniquely paired Equipment/Part ID aliases) uses source styles; ALL text black.\n'
            'Test this beta in a project copy first. Continue?',
            icon='warning',
        )
        if not ok:
            self.write('Apply cancelled; no changes were made.')
            self.status.set('Cancelled. Nothing changed.')
            return
        self.write('\n' + '=' * 65 + '\nAPPLY RESULTS')
        successful, failed = 0, 0
        for oid, changes in plan:
            try:
                obj, _ = conn.object_for(cat, oid)
            except Exception as exc:
                self.write(f'ID {oid}: cannot open target ({exc})')
                failed += len(changes)
                continue
            hatch_methods_logged = False
            hatch_pattern_applied = False
            for name, getter, setter, before, after in changes:
                try:
                    # Refuse to overwrite a property if it changed since preview.
                    live = getattr(obj, getter)()
                    if not valid_property_for(name, live):
                        failed += 1
                        self.write(f'ID {oid}, {name}: SKIPPED - target getter produced invalid data {display(live)}')
                        continue
                    if same_value(live, after):
                        successful += 1
                        self.write(f'ID {oid}, {name}: ALREADY MATCHED after earlier changes: {format_property(name, live)}')
                        continue
                    if not same_value(live, before):
                        # Changing the hatch pattern can update its dependent
                        # color/width/style/distance values automatically.
                        if hatch_pattern_applied and name.startswith('Hatch ') and name != 'Hatch pattern':
                            self.write(f'ID {oid}, {name}: value updated as hatch pattern was enabled: '
                                       f'{format_property(name, before)} -> {format_property(name, live)}; '
                                       'continuing to match source')
                        else:
                            failed += 1
                            self.write(f'ID {oid}, {name}: SKIPPED - property changed since preview')
                            continue
                    setter_result = invoke_format_setter(obj, name, setter, after)
                    actual = getattr(obj, getter)()
                    if valid_property_for(name, actual) and same_value(actual, after):
                        successful += 1
                        if name == 'Hatch pattern':
                            hatch_pattern_applied = True
                        self.write(f'ID {oid}, {name}: VERIFIED {format_property(name, before)} -> {format_property(name, actual)}')
                    else:
                        failed += 1
                        self.write(
                            f'ID {oid}, {name}: NOT APPLIED / UNVERIFIED. '
                            f'Requested {format_property(name, after)}, '
                            f'getter still reports {format_property(name, actual)}; '
                            f'setter returned {display(setter_result)}. '
                            'A value of -1 alone is NOT treated as an error; the setter did not persist. '
                            'For hatch fields, check source/target pattern details in the log.'
                        )
                        if name.startswith('Hatch ') and not hatch_methods_logged:
                            candidates = sorted(n for n in get_com_names(obj)
                                                if 'hatch' in n.lower() or 'pattern' in n.lower())
                            self.write(f'ID {oid}: Available hatch/pattern API methods: {candidates}')
                            for line in conn.hatch_diagnostics(oid):
                                self.write('  ' + line)
                            hatch_methods_logged = True
                except Exception as exc:
                    failed += 1
                    self.write(f'ID {oid}, {name}: ERROR: {exc}')
                    if name.startswith('Hatch ') and not hatch_methods_logged:
                        candidates = sorted(n for n in get_com_names(obj)
                                            if 'hatch' in n.lower() or 'pattern' in n.lower())
                        self.write(f'ID {oid}: Available hatch/pattern API methods: {candidates}')
                        for line in conn.hatch_diagnostics(oid):
                            self.write('  ' + line)
                        hatch_methods_logged = True
        # Apply attached symbol AND pin text formats after hatch/outline, and verify
        # each setter using the same live readback used for symbol properties.
        if text_plan:
            self.write('ATTACHED BLOCK + CONNECTOR/PIN TEXT RESULTS (values unchanged)')
        safe_children = {}
        for symbol_id, text_id, changes in text_plan:
            try:
                if symbol_id not in safe_children:
                    # IMPORTANT: recheck PIN-linked as well as symbol-linked text.
                    # Previously this check rejected correctly discovered top-pin
                    # texts because they were not on the block symbol itself.
                    current_texts, text_notes = conn.read_symbol_texts(symbol_id)
                    safe_children[symbol_id] = {e.object_id for e in current_texts}
                if text_id not in safe_children[symbol_id]:
                    self.write(f'Text {text_id}: SKIPPED - text no longer attached to block/pins of symbol {symbol_id}')
                    failed += len(changes)
                    continue
                txt, _ = conn.object_for('text', text_id)
                for name, getter, setter, before, after in changes:
                    try:
                        live = getattr(txt, getter)()
                        if not valid_property_for(name, live):
                            self.write(f'Text {text_id}, {name}: invalid live value {display(live)}')
                            failed += 1
                            continue
                        if same_value(live, after):
                            successful += 1
                            self.write(f'Text {text_id}, {name}: ALREADY MATCHED {format_property(name, live)}')
                            continue
                        if not same_value(live, before):
                            failed += 1
                            self.write(f'Text {text_id}, {name}: SKIPPED - changed since preview')
                            continue
                        result = invoke_format_setter(txt, name, setter, after)
                        after_live = getattr(txt, getter)()
                        if valid_property_for(name, after_live) and same_value(after_live, after):
                            successful += 1
                            self.write(f'Text {text_id}, {name}: VERIFIED {format_property(name, before)} '
                                       f'-> {format_property(name, after_live)}')
                        else:
                            failed += 1
                            self.write(f'Text {text_id}, {name}: UNVERIFIED; requested {after!r}, '
                                       f'getter {after_live!r}, setter returned {result!r}')
                    except Exception as exc:
                        failed += 1
                        self.write(f'Text {text_id}, {name}: ERROR: {exc}')
            except Exception as exc:
                failed += len(changes)
                self.write(f'Text {text_id}: ERROR opening attached text: {exc}')
        if failed and cat == 'symbol':
            self.write('DIAGNOSIS: The API did not confirm every requested write. '
                       'The target might not support an active hatch, or the setter may '
                       'require additional arguments for this E3 version. '
                       'See read-only pattern diagnostics from Copy/Preview above. '
                       'No guessed hatch pattern values were written.')
        self.write(
            f'FINISHED: {successful} VERIFIED; {skipped_count} SKIPPED at preview; '
            f'{failed} WRITE ISSUES. -1 values were not skipped automatically.'
        )
        self.status.set(
            f'Finished: {successful} verified, {skipped_count} skipped, {failed} write issues. '
            'See log for individual property results.'
        )
        messagebox.showinfo(
            'Match Properties complete',
            f'{successful} verified formatting changes.\n'
            f'{skipped_count} unmatched properties skipped.\n'
            f'{failed} write issues.\nSee the copyable log for details.'
        )

    def copy_log(self):
        self.root.clipboard_clear()
        self.root.clipboard_append(self.output.get('1.0', 'end-1c'))
        self.root.update()
        self.status.set('Diagnostics copied to clipboard.')


def launch_detached_gui():
    """Exit E3's add-on invocation immediately, leaving a separate GUI alive.

    E3 can wait for the launched add-on command to finish. Running the Tk
    mainloop in that synchronous command can block E3's drawing/UI selection.
    A detached process avoids that lifecycle problem.
    """
    file_path = os.path.abspath(__file__)
    interpreter = sys.executable
    # pythonw.exe gives a GUI without a lingering command-prompt window.
    # Keep the same Python installation (and installed pywin32 packages).
    candidate = os.path.join(os.path.dirname(interpreter), 'pythonw.exe')
    if os.path.isfile(candidate):
        interpreter = candidate

    flags = 0
    if os.name == 'nt':
        flags = (getattr(subprocess, 'DETACHED_PROCESS', 0)
                 | getattr(subprocess, 'CREATE_NEW_PROCESS_GROUP', 0))

    try:
        subprocess.Popen(
            [interpreter, file_path, '--gui-child'],
            cwd=os.path.dirname(file_path),
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            close_fds=True,
            creationflags=flags,
        )
    except Exception as exc:
        # Do NOT enter Tk.mainloop in the parent as that can lock the host.
        # Show one short error and return control to E3.
        root = tk.Tk()
        root.withdraw()
        messagebox.showerror(
            'Could not launch Match Properties',
            f'The separate Match Properties window could not start.\n\n{exc}\n\n'
            'E3 was not kept blocked by a running GUI.',
            parent=root,
        )
        root.destroy()


def main():
    # Parent process is intentionally very short-lived if started as an E3 add-on.
    if '--gui-child' not in sys.argv:
        launch_detached_gui()
        return

    pythoncom.CoInitialize()
    try:
        root = tk.Tk()
        MatchPropertiesApp(root)
        root.mainloop()
    finally:
        pythoncom.CoUninitialize()


if __name__ == '__main__':
    main()
