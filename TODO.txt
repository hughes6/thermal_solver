(

    -type d -printf '%f\n' |
  awk '/^[0-9]+([.][0-9]+)?([eE][-+]?[0-9]+)?$/' |
  sort -g | tail -1
)

bash ./prepare_regions_low_memory.sh "$TARGET_CASE"

  "$SEED_CASE" "$TARGET_CASE" 6 \
  --allow-unqualified "$CHECKPOINT"

THERMAL_SOLVER_OPENFOAM_ENV_READY=1 OPENFOAM_LAUNCHER=env \
bash ./run_parallel.sh 6 --unqualified-thermal 1 \
  2>&1 | tee -a unqualified-thermal.stdout.log

cd '/mnt/c/OpenFOAM/thermal_sim_SOLAR_ATB/SOLAR_ATB_20PCT_EXPLORATORY'
