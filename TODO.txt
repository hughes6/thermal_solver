"""E3 symbol-view probe — PHASE 2 (read-only, separate process).

Run AFTER e3_probe_1_place.py has EXITED and you have manually changed the
symbol in E3. This revision does NOT require the saved connector device ID
to be present in GetAllDeviceIds(): E3 may reassign a device ID after edits.

Works with the JSON file generated by the ORIGINAL Phase 1 script. You do
not need to rerun Phase 1. No objects are created, updated, or deleted.
"""
from pathlib import Path
import json
import traceback
import win32com.client as win32

STATE_FILE = Path(__file__).with_name("e3_symbol_probe_state.json")

GETTERS = (
    "GetName", "GetSymbolTypeName", "GetSymbolType", "GetSymbolTypeId",
    "GetCharacteristic", "GetValidCharacteristics", "GetVersion",
    "GetView", "GetViewId", "GetViewNumber", "GetSymbolView",
    "GetComponentName", "GetDeviceId", "GetSheetId",
)


def extract_ids(value):
    """Handle the (count, (IDs...)) format used by Zuken COM."""
    if isinstance(value, (tuple, list)):
        if len(value) >= 2 and isinstance(value[1], (tuple, list)):
            return [int(x) for x in value[1]
                    if isinstance(x, int) and not isinstance(x, bool) and x > 0]
        if all(isinstance(x, int) and not isinstance(x, bool) for x in value):
            return [int(x) for x in value if x > 0]
    return []


def get_ids(obj, method):
    if not hasattr(obj, method):
        return []
    fn = getattr(obj, method)
    for args in ((None,), ()):  # Different E3 releases expose different signatures.
        try:
            return extract_ids(fn(*args))
        except Exception:
            continue
    return []


def safe_get(obj, method):
    try:
        return repr(getattr(obj, method)()) if hasattr(obj, method) else "<not exposed>"
    except Exception as exc:
        return f"<unavailable: {type(exc).__name__}: {exc}>"


def get_snapshot(prj, symbol_id):
    """Read a symbol without modifying it. Return None if ID is invalid."""
    sym = prj.CreateSymbolObject()
    try:
        res = sym.SetId(int(symbol_id))
    except Exception:
        return None
    # An E3 SetId can have differing success return codes; don't rely solely
    # on its numeric status. Verify actual readable symbol data instead.
    getters = {method: safe_get(sym, method) for method in GETTERS}
    good = [v for v in getters.values()
            if not v.startswith("<not exposed>") and not v.startswith("<unavailable:")
            and v not in ("None", "''", "0", "-1")]
    try:
        text_ids = get_ids(sym, "GetTextIds")
    except Exception:
        text_ids = []
    if not good and not text_ids:
        return None
    txt = prj.CreateTextObject()
    texts = []
    for tid in text_ids:
        try:
            txt.SetId(tid)
            item = {"id": tid}
            for method, field in (("GetType", "type"), ("GetText", "value"),
                                  ("GetVisibility", "visible")):
                try:
                    item[field] = getattr(txt, method)()
                except Exception as exc:
                    item[field] = f"<unavailable: {type(exc).__name__}: {exc}>"
            texts.append(item)
        except Exception as exc:
            texts.append({"id": tid, "error": str(exc)})
    return {"symbol_id": int(symbol_id), "getters": getters, "texts": texts}


def show_diff(before, after):
    print("  COM getter differences:")
    changed = 0
    for key, old_val in before.get("getters", {}).items():
        new_val = after.get("getters", {}).get(key)
        if old_val != new_val:
            changed += 1
            print(f"    {key}: {old_val}  ->  {new_val}")
    if not changed:
        print("    (No changes in exposed zero-argument getters.)")
    bt = before.get("texts", [])
    at = after.get("texts", [])
    print(f"  Texts before ({len(bt)}): {bt}")
    print(f"  Texts after  ({len(at)}): {at}")
    if bt == at:
        print("    (Text list unchanged.)")
    print("  NOTE: GetName() is the symbol DESIGNATION, not the library Symbol Name.")
    print("  An unchanged COM snapshot does not prove the library symbol stayed the same.")


def discover_symbol_ids(prj, sheet_id, saved_device_id, expected_device_name):
    """Try several read-only routes if the original symbol ID has changed.

    No guessed mutating E3 methods are invoked. E3 installations expose
    different enumeration methods; nonexistent ones are safely ignored.
    """
    sources = {}
    sources["Job.GetAllSymbolIds"] = get_ids(prj, "GetAllSymbolIds")
    sources["Job.GetSymbolIds"] = get_ids(prj, "GetSymbolIds")

    sht = prj.CreateSheetObject()
    try:
        sht.SetId(int(sheet_id))
        sources["Sheet.GetSymbolIds"] = get_ids(sht, "GetSymbolIds")
        sources["Sheet.GetAllSymbolIds"] = get_ids(sht, "GetAllSymbolIds")
    except Exception as exc:
        print(f"  Could not scan saved sheet {sheet_id}: {exc}")

    device_ids = get_ids(prj, "GetAllDeviceIds")
    print(f"  Current GetAllDeviceIds: {len(device_ids)} entries")
    if int(saved_device_id) not in device_ids:
        print(f"  NOTE: saved connector device ID {saved_device_id} is no longer enumerated.")
        print("        This is NOT treated as a failure.")
    dev = prj.CreateDeviceObject()
    device_candidates = set(device_ids)
    device_candidates.add(int(saved_device_id))
    matched_devices = []
    for did in sorted(device_candidates):
        try:
            dev.SetId(did)
            dev_name = dev.GetName()
        except Exception:
            continue
        if str(dev_name).strip() == str(expected_device_name).strip():
            matched_devices.append(did)
    if matched_devices:
        print(f"  Found device(s) by designation: {matched_devices}")
    else:
        print(f"  No device named {expected_device_name!r} found (not fatal).")

    # Even when the device is missing from GetAllDeviceIds, SetId(saved ID)
    # may remain valid. Also scan any device matched by its designation.
    for did in set(matched_devices + [int(saved_device_id)]):
        try:
            dev.SetId(did)
            sources[f"Device {did}.GetSymbolIds"] = get_ids(dev, "GetSymbolIds")
        except Exception:
            pass

    all_ids = set()
    for name, ids in sources.items():
        if ids:
            print(f"  {name}: {len(ids)} symbol IDs")
        all_ids.update(ids)
    print(f"  Total unique enumerated symbols: {len(all_ids)}")
    return all_ids


def _repr_to_value(raw):
    # Only for evaluating IDs printed by a previously defined COM getter.
    try:
        return int(raw)
    except (ValueError, TypeError):
        return None


def find_replacement(prj, candidate_ids, before, sheet_id, label):
    """Find a replacement symbol, but NEVER silently guess an ambiguous match."""
    old = before.get("getters", {})
    old_name = old.get("GetName")
    old_component = old.get("GetComponentName")
    old_view = old.get("GetViewNumber")
    old_type = old.get("GetSymbolType")
    matches = []
    for sid in candidate_ids:
        snap = get_snapshot(prj, sid)
        if snap is None:
            continue
        g = snap["getters"]
        current_sheet = _repr_to_value(g.get("GetSheetId"))
        if current_sheet is not None and current_sheet > 0 and current_sheet != int(sheet_id):
            continue
        score = 0
        signals = []
        for k, oldval, weight in (
            ("GetName", old_name, 3),
            ("GetComponentName", old_component, 3),
            ("GetViewNumber", old_view, 1),
            ("GetSymbolType", old_type, 1),
        ):
            if oldval and not str(oldval).startswith("<") and oldval not in ("None", "''") and g.get(k) == oldval:
                score += weight
                signals.append(k)
        if score > 0:
            matches.append((score, sid, signals, snap))
    matches.sort(key=lambda x: (-x[0], x[1]))
    if not matches:
        print(f"  Could not locate {label} using available E3 enumeration/getters.")
        return None
    for score, sid, signals, _ in matches[:8]:
        print(f"  Possible {label}: symbol {sid} score={score} matched={signals}")
    if len(matches) == 1 or (matches[0][0] >= 6 and matches[0][0] > matches[1][0]):
        print(f"  Unique strongest replacement candidate: {matches[0][1]}")
        return matches[0][3]
    print(f"  {label}: multiple plausible symbols. Refusing to compare the wrong one.")
    return None


def inspect():
    if not STATE_FILE.is_file():
        raise RuntimeError(f"Missing {STATE_FILE}. Put this script next to the Phase 1 JSON.")
    state = json.loads(STATE_FILE.read_text(encoding="utf-8"))
    print(f"BASELINE: {STATE_FILE}")
    print(f"Created: {state['created_at']}")
    print(f"Connector: {state['connector_name']} [{state['component_code']}]")
    print(f"Expected device designation: {state['test_device_designation']}")
    print(f"Saved top symbol ID: {state['top']['symbol_id']}")
    print("\nConnecting to E3 in READ-ONLY mode...")
    app = win32.GetActiveObject("CT.Application")
    prj = app.CreateJobObject()
    sheet_ids = get_ids(prj, "GetAllSheetIds")
    if sheet_ids and int(state["sheet_id"]) not in sheet_ids:
        print("WARNING: saved sheet ID not enumerated. It may have been renumbered by E3.")
    print("NOTE: The old device ID is diagnostic only; its absence will NOT stop inspection.")

    candidates = None
    def get_candidates():
        nonlocal candidates
        if candidates is None:
            candidates = discover_symbol_ids(prj, state["sheet_id"],
                                             state["connector_id"],
                                             state["test_device_designation"])
        return candidates

    def inspect_one(title, entry):
        sid = int(entry["symbol_id"])
        print(f"\n========== {title} ==========", flush=True)
        print(f"Saved symbol ID {sid} / requested library symbol: {entry.get('requested_symbol')}")
        after = get_snapshot(prj, sid)
        if after is None:
            print(f"  Saved symbol {sid} cannot be read. Attempting to locate replacement...")
            after = find_replacement(prj, get_candidates(), entry["before"],
                                     state["sheet_id"], title)
        else:
            print(f"  Reading symbol {sid} directly; no device-ID lookup required.")
        if after is None:
            print("  SKIPPED: symbol no longer resolvable. No incorrect comparison shown.")
            return False
        if after["symbol_id"] != sid:
            print(f"  Symbol ID changed: {sid} -> {after['symbol_id']}")
        show_diff(entry["before"], after)
        return True

    ok = inspect_one("TOP LEVEL", state["top"])
    for entry in state.get("secondary", []):
        inspect_one(f"SECONDARY PIN {entry.get('pin_name')}", entry)

    print("\n========== READ-ONLY COM METHOD INVENTORY ==========")
    for title, obj in (("Symbol", prj.CreateSymbolObject()), ("Job", prj)):
        relevant = ("swap", "replace", "exchange", "change", "symbol", "view", "character")
        names = sorted(name for name in dir(obj) if not name.startswith("_")
                       and any(w in name.lower() for w in relevant))
        print(f"  {title}: {names}")
    print("\nINSPECTION COMPLETE" if ok else "\nINSPECTION PARTIAL: top-level symbol unresolved")
    print("This process exits immediately. No E3 objects were modified.", flush=True)


if __name__ == "__main__":
    try:
        inspect()
    except Exception:
        print("\nINSPECTION FAILED:\n" + traceback.format_exc(), flush=True)
        raise SystemExit(1)
