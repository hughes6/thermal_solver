"""E3.series block connector positioning A/B test driven by your Excel pin sheet.

Run on a COPY of your E3 project while E3 is open:
    python test_e3_block_connector_edge.py

As in main.py, a file-picker opens to select the .xlsm workbook. The active
worksheet supplies the connector names/types from columns B/C, beginning at
row 7 (connector count in C5). Choose one connector to test; press Enter to
use the first connector.

Creates TWO new CABBLOCK symbols and two NEW connector devices on E3 sheet 1:
    INSET: connector symbol origin 0.3 inside the right block edge
    EDGE:  connector symbol origin exactly on the right block edge

It does not delete or move pre-existing items. Run in a sacrificial project
copy and discard the test changes when finished.
"""

from datetime import datetime
from pathlib import Path
import tkinter as tk
from tkinter import filedialog

from openpyxl import load_workbook
import win32com.client as win32

# ---------------------- TEST CONFIGURATION ----------------------
E3_SHEET_NAME = "1"                # E3 sheet to place the test symbols on
BLOCK_SYMBOL = "CABBLOCK"
BLOCK_VERSION = 1
CONNECTOR_SYMBOL = "MilBlockSymbolReceptacleInt¶FULL"
CONNECTOR_VERSION = 1

# Choose an EMPTY area of E3 sheet 1. Coordinates use main.py units.
X_A, Y_A = 3.0, 4.0
X_B, Y_B = 11.0, 4.0
BLOCK_WIDTH, BLOCK_HEIGHT = 3.0, 4.0

RIGHT_SIDE_INSET = 0.3
PIN_Y_FROM_BOTTOM = 1.5
ROTATION = 0                      # Right edge, as in main.py
# ----------------------------------------------------------------


def pick_excel_sheet():
    """Replicates main.py's Tk Excel workbook picker (not an E3 sheet picker)."""
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()
    try:
        selected = filedialog.askopenfilename(
            parent=root,
            title="Select the block maker Excel sheet",
            filetypes=[
                ("Macro-enabled Excel files", "*.xlsm"),
                ("Excel files", "*.xlsx"),
                ("All files", "*.*"),
            ],
        )
    finally:
        root.destroy()
    if not selected:
        raise SystemExit("No Excel workbook selected; test cancelled.")
    return Path(selected)


def read_connectors_from_excel(path):
    """Match the main.py workbook layout: C5 count, B/C rows 7 onward."""
    wb = load_workbook(path, data_only=True, read_only=True)
    try:
        ws = wb.active
        raw_count = ws["C5"].value
        try:
            number_of_connectors = int(raw_count)
        except (ValueError, TypeError):
            raise ValueError(
                f"{path.name}: active worksheet {ws.title!r} has invalid C5 "
                f"(number of connectors): {raw_count!r}"
            )

        if number_of_connectors < 1:
            raise ValueError("The selected sheet does not contain any connectors (C5).")

        connectors = []
        # main.py uses min_row=7 and max_row=7+NUMBER_OF_CONNECTORS,
        # then filters entries with no connector type.
        for row_num in range(7, 8 + number_of_connectors):
            connector_name = ws.cell(row=row_num, column=2).value
            connector_type = ws.cell(row=row_num, column=3).value
            if connector_name is None or connector_type is None:
                continue
            name = str(connector_name).strip()
            component = str(connector_type).strip()
            if name and component:
                connectors.append((name, component, row_num))

        if not connectors:
            raise ValueError(
                "No connector names/types found in the active sheet's "
                "columns B/C starting at row 7."
            )
        return ws.title, connectors
    finally:
        wb.close()


def select_connector(connectors):
    print("\nConnectors found in the workbook:")
    for index, (name, component, excel_row) in enumerate(connectors, 1):
        print(f"  {index:>2}. {name}  /  {component}  (row {excel_row})")
    raw = input("\nWhich connector to test? [Enter = 1]: ").strip()
    if not raw:
        return connectors[0]
    try:
        index = int(raw)
    except ValueError:
        raise ValueError(f"Invalid connector selection: {raw!r}")
    if not 1 <= index <= len(connectors):
        raise ValueError(f"Choose a number between 1 and {len(connectors)}")
    return connectors[index - 1]


def ids(result):
    if isinstance(result, (tuple, list)):
        if len(result) > 1 and isinstance(result[1], (tuple, list)):
            return [int(v) for v in result[1] if v is not None and v != 0]
        if all(isinstance(v, int) for v in result):
            return [int(v) for v in result if v != 0]
    return []


def first_id(result):
    return result[0] if isinstance(result, (tuple, list)) else result


def say(app, msg):
    print(msg, flush=True)
    try:
        app.PutMessage(msg)
    except Exception:
        pass


def get_sheet_id(prj):
    sht = prj.CreateSheetObject()
    for sheet_id in ids(prj.GetAllSheetIds(None)):
        sht.SetId(sheet_id)
        if str(sht.GetName()) == E3_SHEET_NAME:
            return sheet_id
    raise RuntimeError(f"Could not find E3 sheet named {E3_SHEET_NAME!r}.")


def new_block(prj, sheet_id, x, y):
    symbol = prj.CreateSymbolObject()
    if not symbol.Load(BLOCK_SYMBOL, BLOCK_VERSION):
        raise RuntimeError(f"Cannot load block symbol {BLOCK_SYMBOL!r}.")
    result = symbol.PlaceBlock(sheet_id, x, y, BLOCK_WIDTH, BLOCK_HEIGHT)
    block_id = first_id(result)
    if not isinstance(block_id, int) or block_id <= 0:
        raise RuntimeError(f"PlaceBlock failed at ({x}, {y}): {result!r}")
    return block_id


def new_connector(prj, component, device_name):
    dev = prj.CreateDeviceObject()
    result = dev.Create(device_name, "", "", component, CONNECTOR_VERSION, 0)
    if not isinstance(result, int) or result <= 0:
        raise RuntimeError(f"dev.Create failed for {device_name}: {result!r}")

    # Resolve by exact name, as in the original main.py.
    device_id = None
    for did in ids(prj.GetAllDeviceIds(None)):
        dev.SetId(did)
        if str(dev.GetName()) == device_name:
            device_id = did
            break
    if device_id is None:
        raise RuntimeError(f"Cannot find new connector {device_name} in Device Tree")

    dev.SetId(device_id)
    pin_ids = ids(dev.GetPinIds(None))
    if not pin_ids:
        raise RuntimeError(f"Connector {device_name} has no pins")
    return device_id, pin_ids


def make_case(app, prj, sheet_id, component, label, x, y, inset, stamp):
    block_id = new_block(prj, sheet_id, x, y)
    say(app, f"{label}: placed CABBLOCK ID {block_id} at ({x}, {y})")

    device_name = f"BM_{label}_{stamp}"
    device_id, pin_ids = new_connector(prj, component, device_name)
    say(app, f"{label}: created device {device_name}, ID {device_id}, {len(pin_ids)} pins")

    # Everything except the horizontal insertion coordinate is identical.
    pin_x = x + BLOCK_WIDTH - inset
    pin_y = y + PIN_Y_FROM_BOTTOM
    sym = prj.CreateSymbolObject()
    result = sym.PlacePins(
        pin_ids,
        CONNECTOR_SYMBOL,
        CONNECTOR_VERSION,
        sheet_id,
        pin_x,
        pin_y,
        ROTATION,
        1.0,
    )
    symbol_id = first_id(result)
    say(app, f"{label}: PlacePins result={result!r}; x={pin_x}, y={pin_y}, flip={ROTATION}")
    if not isinstance(symbol_id, int) or symbol_id <= 0:
        say(app, f"{label}: PlacePins FAILED (test objects might still exist)")
    else:
        say(app, f"{label}: placed connector symbol ID {symbol_id}")
    return symbol_id


def main():
    excel_path = pick_excel_sheet()
    excel_tab, connectors = read_connectors_from_excel(excel_path)
    name, component, excel_row = select_connector(connectors)
    print(f"\nSelected workbook: {excel_path}")
    print(f"Active Excel sheet: {excel_tab}")
    print(f"Connector: {name}; type: {component}; Excel row: {excel_row}")

    app = win32.GetActiveObject("CT.Application")
    prj = app.CreateJobObject()
    sheet_id = get_sheet_id(prj)

    stamp = datetime.now().strftime("%H%M%S")
    say(app, f"A/B placement test on E3 sheet {E3_SHEET_NAME} (ID {sheet_id})")
    say(app, f"Selected {name} / component {component} from {excel_path.name}")
    say(app, "IMPORTANT: run only in an unsaved PROJECT COPY")

    lookup = app.GetDatabaseTableSelectedComponents(component, CONNECTOR_VERSION)
    if isinstance(lookup, (tuple, list)) and lookup and lookup[0] != 0:
        raise RuntimeError(f"Component {component!r} not in database: {lookup!r}")

    results = {}
    for label, x, y, inset in (
        ("INSET", X_A, Y_A, RIGHT_SIDE_INSET),
        ("EDGE", X_B, Y_B, 0.0),
    ):
        try:
            results[label] = make_case(
                app, prj, sheet_id, component, label, x, y, inset, stamp
            )
        except Exception as exc:
            results[label] = f"ERROR: {exc!r}"
            say(app, f"{label}: {results[label]}")

    say(app, "====================================")
    say(app, f"RESULTS: {results}")
    say(app, "Try mating the SAME compatible connector to INSET and EDGE.")
    say(app, "If only EDGE works, the original 0.3 inset is the likely cause.")
    say(app, "If both fail, investigate block association/PlacePins behavior.")
    say(app, "Discard the test changes when finished.")


if __name__ == "__main__":
    main()
