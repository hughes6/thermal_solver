def excel_column_letter(column_number):
    result = ""
    number = int(column_number)

    while number > 0:
        number, remainder = divmod(number - 1, 26)
        result = chr(65 + remainder) + result

    return result


def cell_address(row, column):
    return f"{excel_column_letter(column)}{row}"


def safe_text(value):
    if value is None:
        return ""

    if isinstance(value, bool):
        return "TRUE" if value else "FALSE"

    if isinstance(value, float):
        if math.isnan(value):
            return "NaN"
        if value.is_integer():
            return str(int(value))

    return str(value)


def report_text(value):
    text = safe_text(value)

    if text.startswith("=") or text.startswith("+") or text.startswith("-") or text.startswith("@"):
        return "'" + text

    return text


def normalize_blank(value):
    if value is None:
        return ""

    return value


def values_equal(left, right):
    left = normalize_blank(left)
    right = normalize_blank(right)

    if left == right:
        return True

    if isinstance(left, (int, float)) and isinstance(right, (int, float)):
        try:
            return abs(float(left) - float(right)) <= VALUE_TOLERANCE
        except Exception:
            return False

    return False


def formula_text(value):
    if isinstance(value, str) and value.startswith("="):
        return value

    return ""


def formulas_equal(left, right):
    return formula_text(left) == formula_text(right)


def matrix_from_com(value, row_count, column_count):
    if row_count <= 0 or column_count <= 0:
        return []

    if row_count == 1 and column_count == 1:
        return [[value]]

    if row_count == 1:
        if isinstance(value, tuple):
            return [list(value)]
        return [[value]]

    if column_count == 1:
        if isinstance(value, tuple):
            result = []

            for item in value:
                if isinstance(item, tuple):
                    result.append([item[0] if item else None])
                else:
                    result.append([item])

            return result

        return [[value]]

    if not isinstance(value, tuple):
        return [[value for _ in range(column_count)] for _ in range(row_count)]

    return [list(row) if isinstance(row, tuple) else [row] for row in value]


def get_matrix_value(matrix, row, column):
    if row < 1 or column < 1:
        return None

    row_index = row - 1
    column_index = column - 1

    if row_index >= len(matrix):
        return None

    if column_index >= len(matrix[row_index]):
        return None

    return matrix[row_index][column_index]

def meaningful_bounds(sheet):
    used = sheet.UsedRange

    first_row = int(used.Row)
    first_column = int(used.Column)
    last_row = int(used.Row + used.Rows.Count - 1)
    last_column = int(used.Column + used.Columns.Count - 1)

    if last_row < 1:
        last_row = 1

    if last_column < 1:
        last_column = 1

    return first_row, first_column, last_row, last_column


def read_sheet_snapshot(sheet):
    first_row, first_column, last_row, last_column = meaningful_bounds(sheet)

    row_count = last_row
    column_count = last_column

    data_range = sheet.Range(sheet.Cells(1, 1), sheet.Cells(row_count, column_count))

    values_raw = data_range.Value2

    try:
        formulas_raw = data_range.Formula2
    except Exception:
        formulas_raw = data_range.Formula

    values = matrix_from_com(values_raw, row_count, column_count)
    formulas = matrix_from_com(formulas_raw, row_count, column_count)

    return {
        "name": str(sheet.Name),
        "rows": row_count,
        "cols": column_count,
        "values": values,
        "formulas": formulas,
        "used_first_row": first_row,
        "used_first_col": first_column,
        "used_last_row": last_row,
        "used_last_col": last_column,
    }


def read_workbook_snapshot(workbook):
    sheets = {}
    order = []

    for sheet in workbook.Worksheets:
        snapshot = read_sheet_snapshot(sheet)
        sheets[snapshot["name"]] = snapshot
        order.append(snapshot["name"])

    return {
        "path": os.path.abspath(workbook.FullName),
        "sheets": sheets,
        "order": order,
    }

def selection_connector_count(sheet_snapshot):
    value = get_matrix_value(sheet_snapshot["values"], 5, 3)

    try:
        return max(0, int(value))
    except Exception:
        return 0


def value_from_either(old_sheet, new_sheet, row, column):
    new_value = get_matrix_value(new_sheet["values"], row, column) if new_sheet is not None else None

    if new_value not in (None, ""):
        return new_value

    old_value = get_matrix_value(old_sheet["values"], row, column) if old_sheet is not None else None

    return old_value


def describe_selection_connector_table(old_sheet, new_sheet, row, column):
    if row < SELECTION_CONNECTOR_FIRST_ROW:
        return None

    if column not in SELECTION_CONNECTOR_COLUMNS:
        return None

    old_count = selection_connector_count(old_sheet) if old_sheet is not None else 0
    new_count = selection_connector_count(new_sheet) if new_sheet is not None else 0
    connector_count = max(old_count, new_count)

    if connector_count <= 0:
        return None

    last_row = SELECTION_CONNECTOR_FIRST_ROW + connector_count - 1

    if row > last_row:
        return None

    block_number = row - SELECTION_CONNECTOR_FIRST_ROW + 1
    connector = value_from_either(old_sheet, new_sheet, row, 2)
    field = SELECTION_CONNECTOR_COLUMNS[column]

    return {
        "kind": "Connector Selection",
        "block": block_number,
        "connector": safe_text(connector),
        "pin": "",
        "field": field,
        "location": f"Connector Selection / Block {block_number} / {safe_text(connector)} / {field}",
    }


def block_number_for_column(column):
    if column < BLOCK_FIRST_COLUMN:
        return None

    return ((column - BLOCK_FIRST_COLUMN) // BLOCK_WIDTH) + 1


def block_start_column(block_number):
    return BLOCK_FIRST_COLUMN + ((block_number - 1) * BLOCK_WIDTH)


def describe_block_cell(old_sheet, new_sheet, row, column):
    block_number = block_number_for_column(column)

    if block_number is None:
        return None

    start_column = block_start_column(block_number)
    relative_column = column - start_column

    if relative_column < 0 or relative_column >= BLOCK_WIDTH:
        return None

    connector_column = start_column + BLOCK_CENTER_OFFSET
    connector = value_from_either(old_sheet, new_sheet, BLOCK_CONNECTOR_ROW, connector_column)
    connector_text = safe_text(connector)

    pin = ""
    field = "Block Cell"

    if row == BLOCK_PIN_COUNT_ROW:
        field = "Number of Pins"
    elif row == BLOCK_CONNECTOR_ROW:
        field = "Connector Name"
    elif row == BLOCK_HEADER_ROW:
        field = "Header"
    elif row >= BLOCK_DATA_FIRST_ROW:
        if relative_column == BLOCK_PIN_OFFSET:
            field = "Pin"
        elif relative_column == BLOCK_SIGNAL_OFFSET:
            field = "Local Signal"
        elif relative_column == BLOCK_DIRECTION_OFFSET:
            field = "IN/OUT"
        elif relative_column == 0:
            field = "Left Helper"
        elif relative_column == BLOCK_WIDTH - 1:
            field = "Right Helper"

        pin_value = value_from_either(old_sheet, new_sheet, row, start_column + BLOCK_PIN_OFFSET)

        if pin_value in (None, ""):
            pin = row - BLOCK_DATA_FIRST_ROW + 1
        else:
            pin = pin_value

    location_parts = [f"Block {block_number}"]

    if connector_text:
        location_parts.append(connector_text)

    if pin != "":
        location_parts.append(f"Pin {safe_text(pin)}")

    location_parts.append(field)

    return {
        "kind": "Block",
        "block": block_number,
        "connector": connector_text,
        "pin": safe_text(pin),
        "field": field,
        "location": " / ".join(location_parts),
    }


def generic_row_label(old_sheet, new_sheet, row):
    if row < 1:
        return ""

    label = value_from_either(old_sheet, new_sheet, row, 1)

    return safe_text(label)


def generic_column_label(old_sheet, new_sheet, column):
    if column < 1:
        return ""

    label = value_from_either(old_sheet, new_sheet, 1, column)

    return safe_text(label)


def describe_cell(sheet_name, old_sheet, new_sheet, row, column):
    if sheet_name.lower() == "selection":
        selection_info = describe_selection_connector_table(old_sheet, new_sheet, row, column)

        if selection_info is not None:
            return selection_info

        block_info = describe_block_cell(old_sheet, new_sheet, row, column)

        if block_info is not None:
            return block_info

    block_info = describe_block_cell(old_sheet, new_sheet, row, column)

    if block_info is not None:
        header_value = value_from_either(old_sheet, new_sheet, BLOCK_HEADER_ROW, block_start_column(block_info["block"]) + BLOCK_PIN_OFFSET)
        signal_header = value_from_either(old_sheet, new_sheet, BLOCK_HEADER_ROW, block_start_column(block_info["block"]) + BLOCK_SIGNAL_OFFSET)

        if safe_text(header_value).lower() == "pins" or "signal" in safe_text(signal_header).lower():
            return block_info

    row_label = generic_row_label(old_sheet, new_sheet, row)
    column_label = generic_column_label(old_sheet, new_sheet, column)
    address = cell_address(row, column)

    location = f"{sheet_name} / {address}"

    if row_label:
        location += f" / Row: {row_label}"

    if column_label:
        location += f" / Column: {column_label}"

    return {
        "kind": "Generic",
        "block": "",
        "connector": "",
        "pin": "",
        "field": column_label,
        "location": location,
    }

def classify_diff(old_value, new_value, old_formula, new_formula):
    value_changed = not values_equal(old_value, new_value)
    formula_changed = not formulas_equal(old_formula, new_formula)

    old_formula_text = formula_text(old_formula)
    new_formula_text = formula_text(new_formula)

    if not value_changed and not formula_changed:
        return None

    if old_value in (None, "") and new_value not in (None, "") and not old_formula_text and not new_formula_text:
        return "Added Value"

    if new_value in (None, "") and old_value not in (None, "") and not old_formula_text and not new_formula_text:
        return "Removed Value"

    if formula_changed and value_changed:
        return "Formula + Value"

    if formula_changed:
        return "Formula Only"

    if old_formula_text and old_formula_text == new_formula_text and value_changed:
        return "Same Formula / Different Result"

    return "Value"


def compare_matching_sheet(sheet_name, old_sheet, new_sheet):
    max_rows = max(old_sheet["rows"], new_sheet["rows"])
    max_cols = max(old_sheet["cols"], new_sheet["cols"])

    diffs = []

    for row in range(1, max_rows + 1):
        for column in range(1, max_cols + 1):
            old_value = get_matrix_value(old_sheet["values"], row, column)
            new_value = get_matrix_value(new_sheet["values"], row, column)
            old_formula_raw = get_matrix_value(old_sheet["formulas"], row, column)
            new_formula_raw = get_matrix_value(new_sheet["formulas"], row, column)
            diff_type = classify_diff(old_value, new_value, old_formula_raw, new_formula_raw)

            if diff_type is None:
                continue

            location_info = describe_cell(sheet_name, old_sheet, new_sheet, row, column)

            diffs.append({
                "sheet": sheet_name,
                "row": row,
                "column": column,
                "cell": cell_address(row, column),
                "kind": location_info["kind"],
                "block": location_info["block"],
                "connector": location_info["connector"],
                "pin": location_info["pin"],
                "field": location_info["field"],
                "location": location_info["location"],
                "diff_type": diff_type,
                "old_value": old_value,
                "new_value": new_value,
                "old_formula": formula_text(old_formula_raw),
                "new_formula": formula_text(new_formula_raw),
            })

    return diffs


def compare_workbooks(old_snapshot, new_snapshot):
    old_sheet_names = set(old_snapshot["sheets"].keys())
    new_sheet_names = set(new_snapshot["sheets"].keys())

    added_sheets = sorted(new_sheet_names - old_sheet_names)
    removed_sheets = sorted(old_sheet_names - new_sheet_names)
    common_sheets = [name for name in old_snapshot["order"] if name in new_sheet_names]

    diffs = []

    for sheet_name in common_sheets:
        print("COMPARING SHEET:", sheet_name)
        old_sheet = old_snapshot["sheets"][sheet_name]
        new_sheet = new_snapshot["sheets"][sheet_name]
        diffs.extend(compare_matching_sheet(sheet_name, old_sheet, new_sheet))

    return {
        "diffs": diffs,
        "added_sheets": added_sheets,
        "removed_sheets": removed_sheets,
    }

def make_header_cells(sheet, headers):
    fill = PatternFill(fill_type="solid", fgColor="D9EAF7")
    side = Side(style="thin", color="000000")
    border = Border(left=side, right=side, top=side, bottom=side)
    cells = []

    for value in headers:
        cell = WriteOnlyCell(sheet, value=value)
        cell.font = Font(bold=True)
        cell.fill = fill
        cell.border = border
        cell.alignment = Alignment(vertical="center")
        cells.append(cell)

    return cells


def set_write_only_widths(sheet, widths):
    for column_number, width in enumerate(widths, start=1):
        sheet.column_dimensions[get_column_letter(column_number)].width = width


def build_fast_report_aggregates(old_snapshot, new_snapshot, comparison):
    all_names = []

    for name in old_snapshot["order"]:
        if name not in all_names:
            all_names.append(name)

    for name in new_snapshot["order"]:
        if name not in all_names:
            all_names.append(name)

    added_set = set(comparison["added_sheets"])
    removed_set = set(comparison["removed_sheets"])

    sheet_counts = {}

    for sheet_name in all_names:
        if sheet_name in added_set:
            status = "Added Sheet"
        elif sheet_name in removed_set:
            status = "Removed Sheet"
        else:
            status = "Compared"

        sheet_counts[sheet_name] = {"status": status, "total": 0, "block": 0, "generic": 0}

    block_groups = {}
    total_block = 0
    total_generic = 0

    for diff in comparison["diffs"]:
        sheet_name = diff["sheet"]

        if sheet_name not in sheet_counts:
            sheet_counts[sheet_name] = {"status": "Compared", "total": 0, "block": 0, "generic": 0}
            all_names.append(sheet_name)

        sheet_counts[sheet_name]["total"] += 1

        if diff["kind"] in ("Block", "Connector Selection"):
            total_block += 1
            sheet_counts[sheet_name]["block"] += 1
            key = (diff["sheet"], diff["block"], diff["connector"])

            if key not in block_groups:
                block_groups[key] = {"sheet": diff["sheet"], "block": diff["block"], "connector": diff["connector"], "count": 0, "pins": set(), "fields": set(), "types": set()}

            group = block_groups[key]
            group["count"] += 1

            if diff["pin"] != "":
                group["pins"].add(str(diff["pin"]))

            if diff["field"]:
                group["fields"].add(str(diff["field"]))

            group["types"].add(str(diff["diff_type"]))
        else:
            total_generic += 1
            sheet_counts[sheet_name]["generic"] += 1

    return all_names, sheet_counts, block_groups, total_block, total_generic


def pin_sort_value(value):
    try:
        return int(value)
    except Exception:
        return 999999


def create_diff_report(excel, report_path, old_snapshot, new_snapshot, comparison):
    report_start = time.perf_counter()
    report_path = os.path.abspath(report_path)
    report_folder = os.path.dirname(report_path)
    os.makedirs(report_folder, exist_ok=True)

    if os.path.isfile(report_path):
        try:
            os.remove(report_path)
        except PermissionError:
            raise Exception(f"Close the existing diff report before running the checker: {report_path}")

    total_diffs = len(comparison["diffs"])

    print("REPORT PHASE 1/6 - AGGREGATING", total_diffs, "DIFFS")
    aggregate_start = time.perf_counter()
    all_names, sheet_counts, block_groups, total_block, total_generic = build_fast_report_aggregates(old_snapshot, new_snapshot, comparison)
    print("REPORT AGGREGATION TIME:", round(time.perf_counter() - aggregate_start, 2), "SECONDS")
    print("BLOCK-MAKER DIFFS:", total_block)
    print("OTHER DIFFS:", total_generic)

    block_rows_to_write = min(total_block, MAX_REPORT_ROWS_PER_SHEET)
    generic_rows_to_write = min(total_generic, MAX_REPORT_ROWS_PER_SHEET)

    print("BLOCK DIFF ROWS TO WRITE:", block_rows_to_write)
    print("OTHER DIFF ROWS TO WRITE:", generic_rows_to_write)

    if total_block > MAX_REPORT_ROWS_PER_SHEET:
        print("WARNING: BLOCK DIFF SHEET WILL BE TRUNCATED TO", MAX_REPORT_ROWS_PER_SHEET, "ROWS")

    if total_generic > MAX_REPORT_ROWS_PER_SHEET:
        print("WARNING: OTHER DIFF SHEET WILL BE TRUNCATED TO", MAX_REPORT_ROWS_PER_SHEET, "ROWS")

    workbook = Workbook(write_only=True)

    print("REPORT PHASE 2/6 - BUILDING SUMMARY")
    summary_sheet = workbook.create_sheet(title="Summary")
    summary_sheet.column_dimensions["A"].width = 24
    summary_sheet.column_dimensions["B"].width = 90
    summary_sheet.append(make_header_cells(summary_sheet, ["EXCEL DIFF REPORT", ""]))
    summary_sheet.append(["Reference Workbook", old_snapshot["path"]])
    summary_sheet.append(["Comparison Workbook", new_snapshot["path"]])
    summary_sheet.append(["Total Cell Diffs", total_diffs])
    summary_sheet.append(["Block-Maker Diffs", total_block])
    summary_sheet.append(["Generic Diffs", total_generic])
    summary_sheet.append(["Block Diff Rows Written", block_rows_to_write])
    summary_sheet.append(["Other Diff Rows Written", generic_rows_to_write])
    summary_sheet.append(["Added Sheets", ", ".join(comparison["added_sheets"])])
    summary_sheet.append(["Removed Sheets", ", ".join(comparison["removed_sheets"])])
    summary_sheet.append(make_header_cells(summary_sheet, ["Sheet", "Status", "Total Diffs", "Block-Maker Diffs", "Generic Diffs"]))

    for sheet_name in all_names:
        counts = sheet_counts[sheet_name]
        summary_sheet.append([sheet_name, counts["status"], counts["total"], counts["block"], counts["generic"]])

    print("REPORT PHASE 3/6 - BUILDING BLOCK SUMMARY")
    block_summary_sheet = workbook.create_sheet(title="Block Summary")
    set_write_only_widths(block_summary_sheet, [18, 10, 20, 12, 35, 35, 30])
    block_summary_sheet.freeze_panes = "A2"
    block_summary_sheet.append(make_header_cells(block_summary_sheet, ["Sheet", "Block #", "Connector", "Diff Count", "Pins Affected", "Fields Affected", "Diff Types"]))

    sorted_group_keys = sorted(block_groups, key=lambda item: (item[0], int(item[1]) if str(item[1]).isdigit() else 999999, item[2]))

    for key in sorted_group_keys:
        group = block_groups[key]
        pins = sorted(group["pins"], key=pin_sort_value)
        fields = sorted(group["fields"])
        types = sorted(group["types"])
        block_summary_sheet.append([group["sheet"], group["block"], group["connector"], group["count"], ", ".join(pins), ", ".join(fields), ", ".join(types)])

    print("REPORT PHASE 4/6 - STREAMING BLOCK DIFFS")
    block_sheet = workbook.create_sheet(title="Block Diffs")
    set_write_only_widths(block_sheet, [16, 9, 20, 9, 22, 28, 65, 10, 35, 35, 45, 45])
    block_sheet.freeze_panes = "A2"
    block_sheet.append(make_header_cells(block_sheet, ["Sheet", "Block #", "Connector", "Pin", "Field", "Diff Type", "Location", "Cell", "Old Value", "New Value", "Old Formula", "New Formula"]))

    block_written = 0

    for diff in comparison["diffs"]:
        if diff["kind"] not in ("Block", "Connector Selection"):
            continue

        if block_written >= MAX_REPORT_ROWS_PER_SHEET:
            break

        block_sheet.append([diff["sheet"], diff["block"], diff["connector"], diff["pin"], diff["field"], diff["diff_type"], diff["location"], diff["cell"], report_text(diff["old_value"]), report_text(diff["new_value"]), report_text(diff["old_formula"]), report_text(diff["new_formula"])])
        block_written += 1

        if block_written % 10000 == 0:
            print("BLOCK DIFF ROWS STREAMED:", block_written)

    print("BLOCK DIFF ROWS STREAMED:", block_written)

    print("REPORT PHASE 5/6 - STREAMING OTHER DIFFS")
    generic_sheet = workbook.create_sheet(title="Other Diffs")
    set_write_only_widths(generic_sheet, [18, 10, 9, 9, 28, 28, 70, 35, 35, 45, 45])
    generic_sheet.freeze_panes = "A2"
    generic_sheet.append(make_header_cells(generic_sheet, ["Sheet", "Cell", "Row", "Column", "Field/Header", "Diff Type", "Location", "Old Value", "New Value", "Old Formula", "New Formula"]))

    generic_written = 0

    for diff in comparison["diffs"]:
        if diff["kind"] != "Generic":
            continue

        if generic_written >= MAX_REPORT_ROWS_PER_SHEET:
            break

        generic_sheet.append([diff["sheet"], diff["cell"], diff["row"], diff["column"], diff["field"], diff["diff_type"], diff["location"], report_text(diff["old_value"]), report_text(diff["new_value"]), report_text(diff["old_formula"]), report_text(diff["new_formula"])])
        generic_written += 1

        if generic_written % 10000 == 0:
            print("OTHER DIFF ROWS STREAMED:", generic_written)

    print("OTHER DIFF ROWS STREAMED:", generic_written)

    sheet_change_sheet = workbook.create_sheet(title="Sheet Changes")
    set_write_only_widths(sheet_change_sheet, [16, 30])
    sheet_change_sheet.freeze_panes = "A2"
    sheet_change_sheet.append(make_header_cells(sheet_change_sheet, ["Change", "Sheet"]))

    for name in comparison["added_sheets"]:
        sheet_change_sheet.append(["Added", name])

    for name in comparison["removed_sheets"]:
        sheet_change_sheet.append(["Removed", name])

    fd, temp_report_path = tempfile.mkstemp(prefix="block_maker_diff_", suffix=".xlsx", dir=report_folder)
    os.close(fd)

    try:
        if os.path.exists(temp_report_path):
            os.remove(temp_report_path)

        print("REPORT PHASE 6/6 - SAVING XLSX")
        print("WRITING DIFF REPORT TO:", temp_report_path)
        save_start = time.perf_counter()
        workbook.save(temp_report_path)
        print("OPENPYXL SAVE TIME:", round(time.perf_counter() - save_start, 2), "SECONDS")

        if not os.path.isfile(temp_report_path):
            raise Exception(f"Openpyxl did not create the temporary diff report at {temp_report_path}")

        if os.path.getsize(temp_report_path) <= 0:
            raise Exception(f"Openpyxl created an empty diff report at {temp_report_path}")

        print("TEMP DIFF REPORT SIZE:", os.path.getsize(temp_report_path), "BYTES")
        print("MOVING DIFF REPORT TO:", report_path)
        os.replace(temp_report_path, report_path)

        if not os.path.isfile(report_path):
            raise Exception(f"Diff report was not visible after move to {report_path}")

        print("DIFF REPORT SAVED:", report_path)
        print("DIFF REPORT SIZE:", os.path.getsize(report_path), "BYTES")
        print("TOTAL REPORT CREATION TIME:", round(time.perf_counter() - report_start, 2), "SECONDS")

    finally:
        try:
            workbook.close()
        except Exception:
            pass

        if os.path.exists(temp_report_path):
            try:
                os.remove(temp_report_path)
            except Exception:
                pass

def print_terminal_summary(old_path, new_path, report_path, comparison):
    block_diffs = [diff for diff in comparison["diffs"] if diff["kind"] in ("Block", "Connector Selection")]
    generic_diffs = [diff for diff in comparison["diffs"] if diff["kind"] == "Generic"]

    print("")
    print("============================================================")
    print("EXCEL DIFF COMPLETE")
    print("============================================================")
    print("REFERENCE:", old_path)
    print("COMPARISON:", new_path)
    print("REPORT:", report_path)
    print("TOTAL DIFFS:", len(comparison["diffs"]))
    print("BLOCK-MAKER DIFFS:", len(block_diffs))
    print("OTHER DIFFS:", len(generic_diffs))
    print("ADDED SHEETS:", comparison["added_sheets"])
    print("REMOVED SHEETS:", comparison["removed_sheets"])
    print("============================================================")

    if block_diffs:
        print("")
        print("BLOCK-MAKER DIFF PREVIEW")
        print("============================================================")

        for diff in block_diffs[:40]:
            print(diff["location"], "|", diff["diff_type"], "|", repr(diff["old_value"]), "->", repr(diff["new_value"]))

        if len(block_diffs) > 40:
            print("... MORE BLOCK DIFFS IN REPORT:", len(block_diffs) - 40)

if __name__ == "__main__":
    start_time = time.perf_counter()

    reference_path = ask_excel_path("Select REFERENCE / OLD Block Maker Workbook")

    if reference_path is None:
        raise Exception("No reference workbook selected")

    comparison_path = ask_excel_path("Select COMPARISON / NEW Block Maker Workbook")

    if comparison_path is None:
        raise Exception("No comparison workbook selected")

    report_path = ask_report_path(os.path.dirname(comparison_path))

    if report_path is None:
        raise Exception("No report path selected")

    excel = win32.DispatchEx("Excel.Application")
    excel.Visible = False
    excel.DisplayAlerts = False
    excel.ScreenUpdating = False
    excel.EnableEvents = False
    excel.AskToUpdateLinks = False

    try:
        excel.AutomationSecurity = 3
    except Exception:
        pass

    reference_book = None
    comparison_book = None

    try:
        print("OPENING REFERENCE WORKBOOK")
        reference_book = excel.Workbooks.Open(reference_path, UpdateLinks=0, ReadOnly=True, AddToMru=False)

        print("READING REFERENCE WORKBOOK")
        reference_snapshot = read_workbook_snapshot(reference_book)

        reference_book.Close(SaveChanges=False)
        reference_book = None

        print("OPENING COMPARISON WORKBOOK")
        comparison_book = excel.Workbooks.Open(comparison_path, UpdateLinks=0, ReadOnly=True, AddToMru=False)

        print("READING COMPARISON WORKBOOK")
        comparison_snapshot = read_workbook_snapshot(comparison_book)

        comparison_book.Close(SaveChanges=False)
        comparison_book = None

        print("COMPARING WORKBOOKS")
        comparison = compare_workbooks(reference_snapshot, comparison_snapshot)

        print("CREATING DIFF REPORT")
        create_diff_report(excel, report_path, reference_snapshot, comparison_snapshot, comparison)

    finally:
        if reference_book is not None:
            reference_book.Close(SaveChanges=False)

        if comparison_book is not None:
            comparison_book.Close(SaveChanges=False)

        try:
            excel.EnableEvents = True
            excel.ScreenUpdating = True
        except Exception:
            pass

        excel.Quit()

    print_terminal_summary(reference_path, comparison_path, report_path, comparison)

    elapsed = time.perf_counter() - start_time

    print("TOTAL TIME:", round(elapsed, 2), "SECONDS")
