import tkinter as tk
from tkinter import scrolledtext
from tkinter import ttk
import csv
import io


current_count = 0
detected_table = []
detected_column_count = 0


# =========================================
# MAIN NUMBER FORMATTER
# =========================================

def convert():
    global current_count

    input_text = input_box.get("1.0", tk.END)

    values = input_text.split()
    values = [x for x in values if x.isdigit()]

    output_text = "\n".join(values)

    output_box.delete("1.0", tk.END)
    output_box.insert(tk.END, output_text)

    current_count = len(values)

    count_label.config(text=f"Items: {current_count}")


def copy_output():
    output_text = output_box.get("1.0", tk.END).strip()

    root.clipboard_clear()
    root.clipboard_append(output_text)
    root.update()


def clear_all():
    global current_count

    input_box.delete("1.0", tk.END)
    output_box.delete("1.0", tk.END)

    current_count = 0

    count_label.config(text="Items: 0")


# =========================================
# CONNECTOR NUMBER GENERATOR
# =========================================

def copy_connector():
    connector_number = connector_entry.get().strip()
    copy_count_text = copy_count_entry.get().strip()

    if connector_number == "":
        connector_status.config(text="Enter a connector number.")
        return

    if copy_count_text == "":
        copy_count = current_count
    else:
        try:
            copy_count = int(copy_count_text)
        except ValueError:
            connector_status.config(text="Copy count must be a number.")
            return

    if copy_count <= 0:
        connector_status.config(text="No items to copy.")
        return

    connector_output = "\n".join([connector_number] * copy_count)

    root.clipboard_clear()
    root.clipboard_append(connector_output)
    root.update()

    connector_status.config(text=f"Copied {copy_count} copies of connector {connector_number}")


# =========================================
# TABLE COLUMN EXTRACTOR
# =========================================

def get_table_input():
    text = table_input_box.get("1.0", "end-1c")

    if text == "":
        return ""

    text = text.replace("\r\n", "\n")
    text = text.replace("\r", "\n")

    return text


def get_manual_column_count():
    value = manual_column_count_entry.get().strip()

    if value == "":
        return None

    try:
        column_count = int(value)
    except ValueError:
        return None

    if column_count <= 0:
        return None

    return column_count


def parse_table_data():
    text = get_table_input()
    manual_columns = get_manual_column_count()

    if text == "":
        return [], 0, "NONE"

    rows = []

    if "\t" in text:
        delimiter_name = "TAB"

        for line in text.split("\n"):
            rows.append(line.split("\t"))

    elif "," in text:
        delimiter_name = "COMMA"

        reader = csv.reader(io.StringIO(text))

        for row in reader:
            rows.append(row)

    else:
        delimiter_name = "LINES"

        raw_lines = text.split("\n")

        if manual_columns is not None and manual_columns > 1:
            delimiter_name = "WHITESPACE / MANUAL"

            tokens = []

            for line in raw_lines:
                tokens.extend(line.split())

            for index in range(0, len(tokens), manual_columns):
                row = tokens[index:index + manual_columns]

                while len(row) < manual_columns:
                    row.append("")

                rows.append(row)

        else:
            for line in raw_lines:
                rows.append([line])

    if manual_columns is not None:
        column_count = manual_columns
    else:
        column_count = 0

        for row in rows:
            if len(row) > column_count:
                column_count = len(row)

    normalized_rows = []

    for row in rows:
        new_row = list(row)

        while len(new_row) < column_count:
            new_row.append("")

        if len(new_row) > column_count:
            new_row = new_row[:column_count]

        normalized_rows.append(new_row)

    return normalized_rows, column_count, delimiter_name


def detect_table():
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table_data()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        table_status.config(text="No table data detected.")
        column_selector["values"] = []
        return

    column_choices = []

    first_row = rows[0]

    for index in range(column_count):
        header = ""

        if index < len(first_row):
            header = first_row[index].strip()

        if header == "":
            column_choices.append(f"{index + 1}")
        else:
            column_choices.append(f"{index + 1}: {header}")

    column_selector["values"] = column_choices

    if column_count > 0:
        column_selector.current(0)

    table_status.config(text=f"Detected {len(rows)} rows x {column_count} columns | Delimiter: {delimiter_name}")


def get_selected_column_number():
    selected = column_selector.get().strip()

    if selected == "":
        return None

    first_piece = selected.split(":")[0].strip()

    try:
        column_number = int(first_piece)
    except ValueError:
        return None

    return column_number


def extract_table_column():
    global detected_table
    global detected_column_count

    rows, column_count, delimiter_name = parse_table_data()

    detected_table = rows
    detected_column_count = column_count

    if len(rows) == 0:
        table_status.config(text="No table data detected.")
        return

    selected_column = get_selected_column_number()

    if selected_column is None:
        table_status.config(text="Select a column to extract.")
        return

    if selected_column < 1 or selected_column > column_count:
        table_status.config(text=f"Column must be between 1 and {column_count}.")
        return

    start_row = 0

    if skip_header_var.get():
        start_row = 1

    output_values = []

    for row_index in range(start_row, len(rows)):
        row = rows[row_index]

        if selected_column - 1 < len(row):
            value = row[selected_column - 1]
        else:
            value = ""

        output_values.append(value)

    output_text = "\n".join(output_values)

    table_output_box.delete("1.0", tk.END)
    table_output_box.insert(tk.END, output_text)

    blank_count = 0

    for value in output_values:
        if value == "":
            blank_count += 1

    table_status.config(text=f"Extracted column {selected_column} | Rows: {len(output_values)} | Blank cells preserved: {blank_count}")


def copy_table_output():
    output_text = table_output_box.get("1.0", "end-1c")

    root.clipboard_clear()
    root.clipboard_append(output_text)
    root.update()

    line_count = 0

    if output_text != "":
        line_count = len(output_text.split("\n"))

    table_status.config(text=f"Copied {line_count} rows to clipboard. Blank rows preserved.")


def clear_table():
    global detected_table
    global detected_column_count

    table_input_box.delete("1.0", tk.END)
    table_output_box.delete("1.0", tk.END)

    manual_column_count_entry.delete(0, tk.END)

    column_selector.set("")
    column_selector["values"] = []

    detected_table = []
    detected_column_count = 0

    table_status.config(text="")
