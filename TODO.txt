"""Zuken E3.series - Match Properties (v4: hatch pattern first + detailed diagnostics)

Select one TEXT or BLOCK SYMBOL on a sheet and click Copy Source.
The separate window STAYS OPEN after copying. Click back into E3 to select
target(s), return to this window, Preview Targets, then Apply.

IMPORTANT: This file first launches a DETACHED GUI process and then exits.
This avoids keeping the E3 Add-on launcher waiting in an active script, which
can cause E3's drawing window to appear locked. COM objects are also released
after every click, instead of being retained for the lifetime of the GUI.

This script does NOT change names, IDs, text content, object locations,
connections, or component assignments. The only write calls are paired
format setters, and only after the user approves the preview.

Requires Windows with E3.series running and pywin32 installed:
    py -m pip install pywin32

For best COM output-array handling, optionally install Zuken's official
E3.series Python wrapper, matching your E3.series version:
    py -m pip install e3series

Automatic color (-1) is preserved for color properties. Other -1 format
values are treated as potentially meaningful defaults/inheritance rather
than automatically skipped. A target reporting -1 for hatch width/style WILL
be offered as a target. An E3 setter that does not stick is logged accurately.
The hatch pattern is copied before hatch colour/width/style/distance when
its getter/setter values can be read safely. E3 support varies by symbol.

Not executable/testable here: actual COM behavior depends on the user's
installed version. Test in a disposable copy of a project first.
"""

import gc
import inspect
import math
import os
import subprocess
import sys
import traceback
from functools import wraps
from dataclasses import dataclass
import tkinter as tk
from tkinter import ttk, messagebox

try:
    import pythoncom
    import win32com.client as win32
except ImportError:
    raise SystemExit('Missing pywin32. Run: py -m pip install pywin32')

MISSING_COM_PARAM = -2147352572  # 0x80020004 = DISP_E_PARAMNOTFOUND
AUTO_COLOUR = -1  # E3's automatic/default color on tested block outline.

# Do not include content/position/orientation or geometry by default.
# E3 text width may mean glyph ratio, not measured text width; no change
# to text width is attempted until that distinction is established.
TEXT_FORMAT = (
    ('Font name', 'GetFontName', 'SetFontName'),
    ('Text height', 'GetHeight', 'SetHeight'),
    ('Text colour', 'GetColour', 'SetColour'),
    ('Font style', 'GetStyle', 'SetStyle'),
    ('Text alignment', 'GetAlignment', 'SetAlignment'),
)
SYMBOL_FORMAT = (
    ('Outline colour', 'GetBlockOutlineColour', 'SetBlockOutlineColour'),
    ('Outline width', 'GetBlockOutlineWidth', 'SetBlockOutlineWidth'),
    ('Outline style', 'GetBlockOutlineStyle', 'SetBlockOutlineStyle'),
    # A hatch must have a pattern before line appearance settings may be
    # editable. Copy the pattern FIRST; never invent a hatch pattern ID.
    ('Hatch pattern', 'GetBlockHatchPattern', 'SetBlockHatchPattern'),
    ('Hatch distance', 'GetBlockHatchDistance', 'SetBlockHatchDistance'),
    ('Hatch colour', 'GetBlockHatchColour', 'SetBlockHatchColour'),
    ('Hatch width', 'GetBlockHatchWidth', 'SetBlockHatchWidth'),
    ('Hatch style', 'GetBlockHatchStyle', 'SetBlockHatchStyle'),
)
CATEGORIES = (
    ('text', 'GetSelectedTextIds', 'CreateTextObject', TEXT_FORMAT),
    ('symbol', 'GetSelectedSymbolIds', 'CreateSymbolObject', SYMBOL_FORMAT),
    ('graphic', 'GetSelectedGraphIds', 'CreateGraphObject', ()),
)


def is_int_id(value):
    return isinstance(value, int) and not isinstance(value, bool) and value > 0


def display(value, maxlen=175):
    s = repr(value)
    return s if len(s) <= maxlen else s[:maxlen] + ' ...'


def collect_ids(value):
    """Turn E3's variable COM output shapes into a list of positive IDs."""
    if isinstance(value, (list, tuple)):
        items = []
        for val in value:
            items.extend(collect_ids(val))
        return items
    return [int(value)] if is_int_id(value) else []


def count_and_ids(result):
    """Recognize (count, ids) without mistaking a count for an ID."""
    if isinstance(result, (tuple, list)) and result:
        if isinstance(result[0], int) and not isinstance(result[0], bool):
            return int(result[0]), collect_ids(result[1:])
    # Sometimes pywin32 returns a count alone, but it is NOT an object ID.
    return (result if isinstance(result, int) else None), []


def get_selected(job, method_name, is_official_wrapper=False):
    """Read current selection without making changes to the drawing.

    Strategies: default COM call, then explicit by-reference variants.
    A value of -2147352572 is NOT treated as a selected object ID.
    """
    fn = getattr(job, method_name)
    attempts = []
    best_count = 0
    try:
        raw = fn()
        count, ids = count_and_ids(raw)
        attempts.append(f'no args: {display(raw)}')
        best_count = max(0, count or 0)
        if ids:
            return ids[:best_count] if best_count else ids, attempts
        if best_count == 0:
            return [], attempts
    except Exception as exc:
        attempts.append(f'no args: {type(exc).__name__}: {exc}')

    # The official e3series wrapper should already return IDs as an output
    # tuple; COM VARIANT objects are for direct pywin32 only.
    if is_official_wrapper:
        return [], attempts

    # E3's VBScript signatures often use a byref output array. Different
    # versions/type-library registrations need different Python VARIANTs.
    for type_flags, seed, desc in (
        (pythoncom.VT_BYREF | pythoncom.VT_VARIANT, None, 'BYREF VARIANT'),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_I4, [], 'BYREF LONG ARRAY'),
        (pythoncom.VT_BYREF | pythoncom.VT_ARRAY | pythoncom.VT_VARIANT, [], 'BYREF VARIANT ARRAY'),
    ):
        try:
            arg = win32.VARIANT(type_flags, seed)
            raw = fn(arg)
            count, embedded = count_and_ids(raw)
            out_ids = collect_ids(arg.value)
            # COM output can also be placed in the returned tuple.
            ids = embedded or out_ids
            best_count = max(best_count, max(0, count or 0))
            attempts.append(f'{desc}: return={display(raw)}, out={display(arg.value)}')
            if ids:
                return ids[:best_count] if best_count else ids, attempts
        except Exception as exc:
            attempts.append(f'{desc}: {type(exc).__name__}: {exc}')
    return [], attempts


def get_com_names(obj):
    names = set()
    try:
        info = obj._oleobj_.GetTypeInfo()
        attr = info.GetTypeAttr()
        for i in range(attr.cFuncs):
            info_fn = info.GetFuncDesc(i)
            name_set = info.GetNames(info_fn.memid)
            if name_set:
                names.add(str(name_set[0]))
    except Exception:
        pass
    try:
        names.update(n for n in dir(obj) if not n.startswith('_'))
    except Exception:
        pass
    return names


def valid_property(value):
    if value is None or value == '':
        return False
    if isinstance(value, int) and value == MISSING_COM_PARAM:
        return False
    if isinstance(value, (tuple, list)):
        # Most scalar getter functions should not return a multi-value tuple.
        # An output-marker tuple is not a format value to copy.
        if not value or MISSING_COM_PARAM in value:
            return False
    return isinstance(value, (str, bool, int, float)) and (not isinstance(value, float) or math.isfinite(value))


def valid_property_for(name, value):
    """Pattern getters can return (pattern, angle1, angle2) on some versions.

    E3 has several COM signatures; do not treat output placeholders as data.
    """
    if name == 'Hatch pattern' and isinstance(value, (tuple, list)):
        return len(value) == 3 and all(valid_property(item) for item in value)
    return valid_property(value)


def signature_hint(method):
    try:
        return str(inspect.signature(method))
    except (ValueError, TypeError, Exception):
        return '(signature not available through COM)'


def read_hatch_diagnostics(symbol):
    """Only read existing values/methods; no E3 writes."""
    lines = []
    names = get_com_names(symbol)
    for m in (
        'GetBlockHatchPattern', 'GetBlockHatchDistance',
        'GetBlockHatchColour', 'GetBlockHatchWidth', 'GetBlockHatchStyle',
    ):
        if names and m not in names:
            lines.append(f'{m}: NOT EXPOSED')
            continue
        try:
            raw = getattr(symbol, m)()
            lines.append(f'{m}() => {display(raw)}')
        except Exception as e:
            lines.append(f'{m}() => {type(e).__name__}: {e}')
    for m in (
        'SetBlockHatchPattern', 'SetBlockHatchDistance',
        'SetBlockHatchColour', 'SetBlockHatchWidth', 'SetBlockHatchStyle',
    ):
        if names and m not in names:
            lines.append(f'{m}: NOT EXPOSED')
            continue
        try:
            lines.append(f'{m}{signature_hint(getattr(symbol, m))}')
        except Exception as e:
            lines.append(f'{m}: signature lookup failed: {e}')
    for m in ('GetType', 'GetSymbolType', 'GetName', 'IsBlock'):
        if m in names:
            try:
                lines.append(f'{m}() => {display(getattr(symbol, m)())}')
            except Exception as e:
                lines.append(f'{m}() => {type(e).__name__}: {e}')
    return lines


def invoke_format_setter(obj, name, setter, value):
    """Don't guess hatch flags or angles. Use exactly the source's values."""
    method = getattr(obj, setter)
    if name == 'Hatch pattern' and isinstance(value, (tuple, list)):
        return method(*value)
    return method(value)


def is_colour(name):
    """Use property name, not numeric type, to interpret an Automatic color."""
    return 'colour' in name.lower() or 'color' in name.lower()


def format_property(name, value):
    """Display -1 as a valid default/sentinel, not a failed selection."""
    if isinstance(value, (int, float)) and value == AUTO_COLOUR:
        if is_colour(name):
            return 'Automatic (-1)'
        return 'Default / inherited (-1)'
    return display(value)


def source_property_copyable(name, value):
    """Accept -1 defaults, including hatch targets/sources; reject other negative sizes."""
    # A -1 hatch pattern has not been confirmed to be a real selectable
    # pattern; it may indicate that no hatch exists or the symbol is
    # unsupported. Don't use it to deactivate another symbol's hatch.
    if name == 'Hatch pattern' and value == -1:
        return False
    # Other -1 default/Automatic values are allowed and verified after write.
    if isinstance(value, (int, float)) and value == -1:
        return True
    # Negative widths/heights other than -1 are not safe to apply.
    if isinstance(value, (int, float)) and value < 0 and any(
        word in name.lower() for word in ('width', 'height')
    ):
        return False
    return True


def same_value(a, b):
    if isinstance(a, (tuple, list)) or isinstance(b, (tuple, list)):
        return (isinstance(a, (tuple, list)) and isinstance(b, (tuple, list))
                and len(a) == len(b)
                and all(same_value(x, y) for x, y in zip(a, b)))
    if isinstance(a, (float, int)) and isinstance(b, (float, int)):
        return math.isclose(float(a), float(b), rel_tol=1e-5, abs_tol=1e-7)
    return a == b


@dataclass
class Source:
    job_id: object
    category: str
    object_id: int
    values: dict


class E3Connection:
    def __init__(self):
        self.backend = 'pywin32'
        self.app = None
        self.job = None
        self.details = []
        try:
            import e3series
            app = e3series.Application()
            job = app.CreateJobObject()
            if job.GetId():
                self.backend = 'official e3series wrapper'
                self.app, self.job = app, job
                return
            self.details.append('Official wrapper connected, but returned no active project ID.')
        except ImportError:
            self.details.append('Optional e3series wrapper is not installed.')
        except Exception as exc:
            self.details.append(f'Official wrapper did not connect: {exc}')
        self.app = win32.Dispatch('CT.Application')
        self.job = self.app.CreateJobObject()

    def job_id(self):
        return self.job.GetId()

    def read_selections(self):
        selections = {}
        diagnostics = {}
        for cat, selection_method, creator, attrs in CATEGORIES:
            try:
                ids, attempted = get_selected(
                    self.job, selection_method,
                    is_official_wrapper=(self.backend == 'official e3series wrapper'),
                )
                selections[cat] = ids
                diagnostics[cat] = attempted
            except Exception as exc:
                selections[cat] = []
                diagnostics[cat] = [f'{type(exc).__name__}: {exc}']
        try:
            result = self.job.GetSelectedBlockCount()
            diagnostics['block count'] = [repr(result)]
        except Exception:
            pass
        return selections, diagnostics

    def object_for(self, category, object_id):
        entry = next((c for c in CATEGORIES if c[0] == category), None)
        if entry is None:
            raise ValueError(f'Unsupported category: {category}')
        obj = getattr(self.job, entry[2])()
        obj.SetId(int(object_id))
        return obj, entry[3]

    def read_properties(self, category, object_id):
        obj, formats = self.object_for(category, object_id)
        names = get_com_names(obj)
        values = {}
        notes = []
        for label, getter, setter in formats:
            # We need BOTH getter and setter for true match-properties.
            if names and (getter not in names or setter not in names):
                notes.append(f'{label}: {getter}/{setter} not both exposed')
                continue
            try:
                value = getattr(obj, getter)()
                if valid_property_for(label, value):
                    values[label] = (getter, setter, value)
                else:
                    notes.append(f'{label}: getter yielded unusable {display(value)}')
            except Exception as exc:
                notes.append(f'{label}: {getter} failed: {exc}')
        return values, notes

    def hatch_diagnostics(self, object_id):
        obj, _ = self.object_for('symbol', object_id)
        return read_hatch_diagnostics(obj)

    def close(self):
        """Release E3 COM references as soon as this button action finishes."""
        self.job = None
        self.app = None
        gc.collect()


def release_e3_after_action(callback):
    """Never keep an E3 automation session alive between GUI button presses."""
    @wraps(callback)
    def wrapped(self, *args, **kwargs):
        try:
            return callback(self, *args, **kwargs)
        finally:
            if self.conn is not None:
                try:
                    self.conn.close()
                finally:
                    self.conn = None
    return wrapped


class MatchPropertiesApp:
    def __init__(self, root):
        self.root = root
        self.source = None
        self.preview = None
        self.conn = None
        self.root.title('Zuken E3 - Match Properties')
        self.root.geometry('900x660')
        self.root.minsize(670, 440)
        # Do not keep this window on top: E3 must be clickable while GUI runs.
        self.root.attributes('-topmost', False)
        self.root.protocol('WM_DELETE_WINDOW', self.on_close)

        panel = ttk.Frame(root, padding=14)
        panel.pack(fill='both', expand=True)
        ttk.Label(panel, text='Zuken Match Properties', font=('Segoe UI', 16, 'bold')).pack(anchor='w')
        ttk.Label(
            panel, text='Select source in E3 -> Copy Source -> select target(s) in E3 -> Preview -> Apply.',
        ).pack(anchor='w', pady=(5, 2))
        ttk.Label(
            panel, text='Text-to-text / block-to-block. Copies hatch PATTERN first; -1 is not automatically skipped.',
        ).pack(anchor='w', pady=(0, 12))

        row = ttk.Frame(panel)
        row.pack(fill='x')
        ttk.Button(row, text='1. Copy Source', command=self.copy_source).pack(side='left')
        ttk.Button(row, text='2. Preview Targets', command=self.preview_target).pack(side='left', padx=9)
        self.apply_button = ttk.Button(row, text='3. Apply Match', command=self.apply, state='disabled')
        self.apply_button.pack(side='left')
        ttk.Button(row, text='Scan / Diagnostics', command=self.scan).pack(side='left', padx=9)
        ttk.Button(row, text='Back to E3', command=self.return_to_e3).pack(side='left')
        ttk.Button(row, text='Copy Log', command=self.copy_log).pack(side='right')

        ttk.Label(
            panel,
            text='This window stays visible after copying. Click directly into E3 to select targets; Back to E3 minimizes only when clicked.',
        ).pack(anchor='w', pady=(9, 0))

        self.status = tk.StringVar(value='Ready. Nothing will be changed without a preview and confirmation.')
        ttk.Label(panel, textvariable=self.status, wraplength=830).pack(fill='x', pady=(10, 8))

        frame = ttk.Frame(panel)
        frame.pack(fill='both', expand=True)
        self.output = tk.Text(frame, wrap='none', font=('Consolas', 10), height=14)
        sy = ttk.Scrollbar(frame, orient='vertical', command=self.output.yview)
        sx = ttk.Scrollbar(frame, orient='horizontal', command=self.output.xview)
        self.output.configure(yscrollcommand=sy.set, xscrollcommand=sx.set)
        frame.columnconfigure(0, weight=1)
        frame.rowconfigure(0, weight=1)
        self.output.grid(row=0, column=0, sticky='nsew')
        sy.grid(row=0, column=1, sticky='ns')
        sx.grid(row=1, column=0, sticky='ew')

        ttk.Button(panel, text='Close', command=self.on_close).pack(anchor='e', pady=(9, 0))
        self.write('No source saved. Choose one text object or block symbol in E3, then click Copy Source.')

    def write(self, msg):
        self.output.insert('end', msg + '\n')
        self.output.see('end')

    def return_to_e3(self):
        # No grab_set(), window disable, or always-on-top state is ever used.
        # Minimize so the E3 drawing area can receive clicks again.
        self.root.iconify()

    def on_close(self):
        if self.conn is not None:
            self.conn.close()
            self.conn = None
        self.root.destroy()

    def reset_preview(self):
        self.preview = None
        self.apply_button.configure(state='disabled')

    def connect(self):
        try:
            # Defense in depth: no old COM job should survive across actions.
            if self.conn is not None:
                self.conn.close()
                self.conn = None
            self.conn = E3Connection()
            if not self.conn.job_id():
                raise RuntimeError('E3 is connected, but no active project is open.')
            return self.conn
        except Exception as exc:
            self.write(f'CONNECTION ERROR: {exc}\n{traceback.format_exc()}')
            self.status.set('Connection error - see copyable log.')
            messagebox.showerror('Connection error', f'Could not connect to an open E3 project:\n{exc}')
            return None

    def selection_report(self, conn, verbose=False):
        selected, diagnostics = conn.read_selections()
        self.write(f'Project ID: {conn.job_id()}, COM backend: {conn.backend}')
        for name, ids in selected.items():
            self.write(f'  {name}: {len(ids)} valid IDs: {ids[:20]}')
            if verbose:
                for attempt in diagnostics.get(name, []):
                    self.write('    ' + attempt)
        if verbose:
            for extra in conn.details:
                self.write(extra)
            if 'block count' in diagnostics:
                self.write('GetSelectedBlockCount: ' + str(diagnostics['block count']))
        if not any(selected.values()):
            self.write('No valid object IDs read. If selection counts are nonzero, the COM out-array is not decoded yet.')
            self.write('Try installing the OFFICIAL E3.series Python package for your E3 version: py -m pip install e3series')
            self.write('Click Copy Log and share these diagnostics, rather than guessing IDs.')
        return selected

    @release_e3_after_action
    def scan(self):
        self.write('\n' + '=' * 65 + '\nDIAGNOSTIC SCAN (read-only)')
        conn = self.connect()
        if not conn:
            return
        self.selection_report(conn, verbose=True)
        self.status.set('Scan complete. Nothing changed in E3.')

    @staticmethod
    def preferred_category(selected):
        # Selecting a block yields both symbol and block counts. Prefer the
        # symbol object because it exposes block outline/hatch properties.
        for cat in ('text', 'symbol', 'graphic'):
            if selected.get(cat):
                return cat
        return None

    @release_e3_after_action
    def copy_source(self):
        self.reset_preview()
        self.write('\n' + '=' * 65 + '\nCOPY SOURCE (read-only)')
        conn = self.connect()
        if not conn:
            return
        selected = self.selection_report(conn)
        cat = self.preferred_category(selected)
        if not cat or cat == 'graphic':
            self.source = None
            self.status.set('Select a text object or block symbol, then Copy Source.')
            self.write('Graphic selection is detected, but generic graphic property matching is not yet enabled.')
            return
        ids = selected.get(cat, [])
        if len(ids) != 1:
            self.source = None
            self.status.set('Copy Source requires exactly one selected object of one type.')
            self.write(f'Expected 1 {cat} ID, got {len(ids)}. Select exactly one source.')
            return
        try:
            values, notes = conn.read_properties(cat, ids[0])
        except Exception as exc:
            self.write(f'Could not read source formatting: {exc}\n{traceback.format_exc()}')
            self.status.set('Source read failed; see log.')
            return
        for note in notes:
            self.write('  Skipped: ' + note)
        if not values:
            self.source = None
            self.write('No format properties were validated as safe getter/setter pairs.')
            self.status.set('Source has no readable/matchable formatting. See log.')
            return
        self.source = Source(conn.job_id(), cat, ids[0], values)
        self.write(f'Saved {cat.upper()} source ID {ids[0]} with {len(values)} formatting properties:')
        for name, (getter, setter, value) in values.items():
            self.write(f'  {name}: {format_property(name, value)}')
        self.status.set(
            f'Source copied: {cat} ID {ids[0]}. Click into E3 and select your targets, then Preview.'
        )
        if cat == 'symbol':
            self.write('SOURCE HATCH DIAGNOSTICS (read-only):')
            for line in conn.hatch_diagnostics(ids[0]):
                self.write('  ' + line)
        self.write('Source copied. Window stays OPEN. E3 is available for selection once this action ends.')

    def find_targets(self, conn):
        if not self.source:
            raise ValueError('Copy a source first.')
        if conn.job_id() != self.source.job_id:
            raise RuntimeError('The active E3 project changed. Copy Source again in the current project.')
        selected = self.selection_report(conn)
        ids = [i for i in selected.get(self.source.category, []) if i != self.source.object_id]
        # Skip the source if it remains among selected objects.
        if not ids:
            raise ValueError(f'Select at least one OTHER {self.source.category} as a target.')
        if len(ids) > 100:
            raise ValueError('More than 100 targets were selected. Reduce selection for a safer test.')
        return list(dict.fromkeys(ids))

    @release_e3_after_action
    def preview_target(self):
        self.reset_preview()
        self.write('\n' + '=' * 65 + '\nPREVIEW TARGETS (read-only)')
        conn = self.connect()
        if not conn:
            return
        try:
            ids = self.find_targets(conn)
        except Exception as exc:
            self.write('PREVIEW BLOCKED: ' + str(exc))
            self.status.set(str(exc))
            return
        plan = []
        skipped_count = 0
        for oid in ids:
            try:
                values, notes = conn.read_properties(self.source.category, oid)
                if self.source.category == 'symbol':
                    self.write(f'ID {oid} TARGET HATCH DIAGNOSTICS (read-only):')
                    for line in conn.hatch_diagnostics(oid):
                        self.write('  ' + line)
                changes = []

                # -1 is not an indication of an invalid target. For hatch
                # fields, E3 may use -1 for Automatic/inherited/unconfigured.
                # Preview the difference and let E3's setter plus readback
                # decide whether a concrete style can actually be applied.

                for name, (getter, setter, source_val) in self.source.values.items():
                    if name not in values:
                        continue
                    if not source_property_copyable(name, source_val):
                        if not same_value(values[name][2], source_val):
                            skipped_count += 1
                        self.write(
                            f'ID {oid}, {name}: SKIPPED source is '
                            f'{format_property(name, source_val)}; no safe setter value known.'
                        )
                        continue
                    current_val = values[name][2]
                    if not same_value(current_val, source_val):
                        changes.append((name, getter, setter, current_val, source_val))
                if changes:
                    plan.append((oid, changes))
                    self.write(f'ID {oid}: {len(changes)} format changes:')
                    for name, getter, setter, before, after in changes:
                        self.write(f'  {name}: {format_property(name, before)} -> {format_property(name, after)}')
                else:
                    self.write(f'ID {oid}: no supported formatting differences.')
                for note in notes:
                    self.write(f'  Skipped: {note}')
            except Exception as exc:
                self.write(f'ID {oid}: could not inspect; will SKIP ({exc})')
        if not plan:
            self.status.set(f'No changes to apply. {skipped_count} unmatched properties skipped (see log).')
            self.write(f'PREVIEW SUMMARY: 0 changes available; {skipped_count} changes skipped by safeguards.')
            return
        self.preview = (conn.job_id(), self.source.category, self.source.object_id, ids, plan, skipped_count)
        self.apply_button.configure(state='normal')
        changes_count = sum(len(p[1]) for p in plan)
        self.status.set(
            f'Preview: {changes_count} changes on {len(plan)} target(s); '
            f'{skipped_count} unmatched properties skipped. Apply requires confirmation.'
        )
        self.write(f'PREVIEW SUMMARY: {changes_count} planned changes, {skipped_count} skipped.')

    @release_e3_after_action
    def apply(self):
        if not self.preview or not self.source:
            return
        project_id, cat, source_id, selected_ids, plan, skipped_count = self.preview
        self.reset_preview()
        conn = self.connect()
        if not conn:
            return
        if conn.job_id() != project_id or self.source.object_id != source_id:
            self.write('APPLY BLOCKED: active project or copied source changed. Preview again.')
            return
        try:
            live_ids = self.find_targets(conn)
        except Exception as exc:
            self.write('APPLY BLOCKED: ' + str(exc))
            return
        if set(live_ids) != set(selected_ids):
            self.write('APPLY BLOCKED: selected target IDs differ from preview. Preview again.')
            return
        n = sum(len(changes) for _, changes in plan)
        ok = messagebox.askyesno(
            'Confirm formatting change',
            f'Apply {n} formatting changes to {len(plan)} {cat} target(s)?\n\n'
            'Text content, object positions and device properties will not be changed.\n'
            'Test this beta in a project copy first. Continue?',
            icon='warning',
        )
        if not ok:
            self.write('Apply cancelled; no changes were made.')
            self.status.set('Cancelled. Nothing changed.')
            return
        self.write('\n' + '=' * 65 + '\nAPPLY RESULTS')
        successful, failed = 0, 0
        for oid, changes in plan:
            try:
                obj, _ = conn.object_for(cat, oid)
            except Exception as exc:
                self.write(f'ID {oid}: cannot open target ({exc})')
                failed += len(changes)
                continue
            hatch_methods_logged = False
            hatch_pattern_applied = False
            for name, getter, setter, before, after in changes:
                try:
                    # Refuse to overwrite a property if it changed since preview.
                    live = getattr(obj, getter)()
                    if not valid_property_for(name, live):
                        failed += 1
                        self.write(f'ID {oid}, {name}: SKIPPED - target getter produced invalid data {display(live)}')
                        continue
                    if same_value(live, after):
                        successful += 1
                        self.write(f'ID {oid}, {name}: ALREADY MATCHED after earlier changes: {format_property(name, live)}')
                        continue
                    if not same_value(live, before):
                        # Changing the hatch pattern can update its dependent
                        # color/width/style/distance values automatically.
                        if hatch_pattern_applied and name.startswith('Hatch ') and name != 'Hatch pattern':
                            self.write(f'ID {oid}, {name}: value updated as hatch pattern was enabled: '
                                       f'{format_property(name, before)} -> {format_property(name, live)}; '
                                       'continuing to match source')
                        else:
                            failed += 1
                            self.write(f'ID {oid}, {name}: SKIPPED - property changed since preview')
                            continue
                    setter_result = invoke_format_setter(obj, name, setter, after)
                    actual = getattr(obj, getter)()
                    if valid_property_for(name, actual) and same_value(actual, after):
                        successful += 1
                        if name == 'Hatch pattern':
                            hatch_pattern_applied = True
                        self.write(f'ID {oid}, {name}: VERIFIED {format_property(name, before)} -> {format_property(name, actual)}')
                    else:
                        failed += 1
                        self.write(
                            f'ID {oid}, {name}: NOT APPLIED / UNVERIFIED. '
                            f'Requested {format_property(name, after)}, '
                            f'getter still reports {format_property(name, actual)}; '
                            f'setter returned {display(setter_result)}. '
                            'A value of -1 alone is NOT treated as an error; the setter did not persist. '
                            'For hatch fields, check source/target pattern details in the log.'
                        )
                        if name.startswith('Hatch ') and not hatch_methods_logged:
                            candidates = sorted(n for n in get_com_names(obj)
                                                if 'hatch' in n.lower() or 'pattern' in n.lower())
                            self.write(f'ID {oid}: Available hatch/pattern API methods: {candidates}')
                            for line in conn.hatch_diagnostics(oid):
                                self.write('  ' + line)
                            hatch_methods_logged = True
                except Exception as exc:
                    failed += 1
                    self.write(f'ID {oid}, {name}: ERROR: {exc}')
                    if name.startswith('Hatch ') and not hatch_methods_logged:
                        candidates = sorted(n for n in get_com_names(obj)
                                            if 'hatch' in n.lower() or 'pattern' in n.lower())
                        self.write(f'ID {oid}: Available hatch/pattern API methods: {candidates}')
                        for line in conn.hatch_diagnostics(oid):
                            self.write('  ' + line)
                        hatch_methods_logged = True
        if failed and cat == 'symbol':
            self.write('DIAGNOSIS: The API did not confirm every requested write. '
                       'The target might not support an active hatch, or the setter may '
                       'require additional arguments for this E3 version. '
                       'See read-only pattern diagnostics from Copy/Preview above. '
                       'No guessed hatch pattern values were written.')
        self.write(
            f'FINISHED: {successful} VERIFIED; {skipped_count} SKIPPED at preview; '
            f'{failed} WRITE ISSUES. -1 values were not skipped automatically.'
        )
        self.status.set(
            f'Finished: {successful} verified, {skipped_count} skipped, {failed} write issues. '
            'See log for individual property results.'
        )
        messagebox.showinfo(
            'Match Properties complete',
            f'{successful} verified formatting changes.\n'
            f'{skipped_count} unmatched properties skipped.\n'
            f'{failed} write issues.\nSee the copyable log for details.'
        )

    def copy_log(self):
        self.root.clipboard_clear()
        self.root.clipboard_append(self.output.get('1.0', 'end-1c'))
        self.root.update()
        self.status.set('Diagnostics copied to clipboard.')


def launch_detached_gui():
    """Exit E3's add-on invocation immediately, leaving a separate GUI alive.

    E3 can wait for the launched add-on command to finish. Running the Tk
    mainloop in that synchronous command can block E3's drawing/UI selection.
    A detached process avoids that lifecycle problem.
    """
    file_path = os.path.abspath(__file__)
    interpreter = sys.executable
    # pythonw.exe gives a GUI without a lingering command-prompt window.
    # Keep the same Python installation (and installed pywin32 packages).
    candidate = os.path.join(os.path.dirname(interpreter), 'pythonw.exe')
    if os.path.isfile(candidate):
        interpreter = candidate

    flags = 0
    if os.name == 'nt':
        flags = (getattr(subprocess, 'DETACHED_PROCESS', 0)
                 | getattr(subprocess, 'CREATE_NEW_PROCESS_GROUP', 0))

    try:
        subprocess.Popen(
            [interpreter, file_path, '--gui-child'],
            cwd=os.path.dirname(file_path),
            stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            close_fds=True,
            creationflags=flags,
        )
    except Exception as exc:
        # Do NOT enter Tk.mainloop in the parent as that can lock the host.
        # Show one short error and return control to E3.
        root = tk.Tk()
        root.withdraw()
        messagebox.showerror(
            'Could not launch Match Properties',
            f'The separate Match Properties window could not start.\n\n{exc}\n\n'
            'E3 was not kept blocked by a running GUI.',
            parent=root,
        )
        root.destroy()


def main():
    # Parent process is intentionally very short-lived if started as an E3 add-on.
    if '--gui-child' not in sys.argv:
        launch_detached_gui()
        return

    pythoncom.CoInitialize()
    try:
        root = tk.Tk()
        MatchPropertiesApp(root)
        root.mainloop()
    finally:
        pythoncom.CoUninitialize()


if __name__ == '__main__':
    main()
