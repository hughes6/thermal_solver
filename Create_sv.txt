Private Sub Worksheet_Change(ByVal Target As Range)

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim src As Range
    Dim dest As Range
    Dim cfSrc As Range
    Dim cfDest As Range
    Dim fc As FormatCondition
    Dim f As String
    Dim newAnchor As String

    If Target.Address <> "$C$4" Then Exit Sub

    On Error GoTo SafeExit
    Application.EnableEvents = False

    n = CLng(Me.Range("C4").Value)

    Set src = Me.Range("H5:L10")

    'Full area containing conditional formatting
    Set cfSrc = Me.Range("I8:K307")

    'Clear old generated blocks
    Me.Range("M5:ZZ10").Clear

    'Remove old generated conditional formatting
    Me.Range("M8:ZZ307").FormatConditions.Delete

    For i = 1 To n - 1

        Set dest = src.Offset(0, i * 5)

        'Copy main block
        src.Copy Destination:=dest

        'Copy column widths
        For j = 1 To 5
            dest.Columns(j).ColumnWidth = src.Columns(j).ColumnWidth
        Next j

        'Destination CF area
        Set cfDest = cfSrc.Offset(0, i * 5)

        'Remove the partial CF copied with H5:L10
        cfDest.FormatConditions.Delete

        'Copy FULL formatting/rules through row 307
        cfSrc.Copy
        cfDest.PasteSpecial Paste:=xlPasteFormats

        'Left column of this block:
        'M for block 2, R for block 3, etc.
        newAnchor = ColLetter(dest.Column)

        'Translate $H to this block's left column
        For Each fc In cfDest.FormatConditions

            If fc.Type = xlExpression Then

                f = fc.Formula1
                f = Replace(f, "$H", "$" & newAnchor)

                fc.Modify Type:=xlExpression, Formula1:=f

            End If

        Next fc

    Next i

SafeExit:
    Application.CutCopyMode = False
    Application.EnableEvents = True

End Sub


Private Function ColLetter(ByVal colNum As Long) As String

    ColLetter = Split(Me.Cells(1, colNum).Address, "$")(1)

End Function
