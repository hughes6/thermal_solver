import time
import tkinter as tk

from utils import runn


def wait_for_equipment_identification():
    root = tk.Tk()
    root.title("Equipment Identification")
    root.attributes("-topmost", True)
    root.resizable(False, False)

    window_width = 620
    window_height = 260

    screen_width = root.winfo_screenwidth()
    screen_height = root.winfo_screenheight()

    x_position = int((screen_width - window_width) / 2)
    y_position = int((screen_height - window_height) / 2)

    root.geometry(f"{window_width}x{window_height}+{x_position}+{y_position}")

    message = "Turn ON all Equipment Identifications in E3.\n\nOnce every Equipment Identification is visible,\ncome back here and click DONE.\n\nThe formatter will then run again automatically."

    label = tk.Label(root, text=message, font=("Arial", 12), justify="center")
    label.pack(pady=32)

    done_button = tk.Button(root, text="DONE - RUN FORMATTER AGAIN", width=34, height=2, command=root.destroy)
    done_button.pack(pady=8)

    root.lift()
    root.focus_force()
    root.mainloop()

wait_for_equipment_identification()
