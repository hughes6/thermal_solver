def format_connector_text(connector_view_id, conn_name, top_pin_symbol_id, top_pin_id):
    dev.SetId(connector_view_id)

    #=========================================
    # FORMAT SPLIT PIN SYMBOL TEXT
    #=========================================
    symbolinfo = dev.GetSymbolIds(None)
    symbolids = [x for x in symbolinfo[1] if x is not None]

    for symbol_id in symbolids:
        sym.SetId(symbol_id)

        textinfo = sym.GetTextIds(None)
        textids = [x for x in textinfo[1] if x is not None]

        for text_id in textids:
            text.SetId(text_id)
            text_type = text.GetType()

            # COMPONENT CODE
            if text_type == 5:
                text.SetVisibility(0)

            # DEVICE DESIGNATION
            elif text_type == 12:
                if symbol_id == top_pin_symbol_id:
                    text.SetVisibility(1)
                    text.SetColour(0)
                    text.SetStyle(1)
                    text.SetHeight(0.2)
                else:
                    text.SetVisibility(0)

    #=========================================
    # FORMAT PIN TEXT
    #=========================================
    pininfo = dev.GetPinIds(None)
    pinids = [x for x in pininfo[1] if x is not None]

    for pin_id in pinids:
        pin.SetId(pin_id)

        pintextinfo = pin.GetTextIds(None)
        pintextids = [x for x in pintextinfo[1] if x is not None]

        for pin_text_id in pintextids:
            text.SetId(pin_text_id)
            text_type = text.GetType()

            # LOCAL SIGNAL NAME
            if text_type == 617:
                text.SetVisibility(1)
                text.SetColour(0)
                text.SetAlignment(3)
                text.SetStyle(1)
                text.SetHeight(0.2)

            # COMPONENT CODE
            elif text_type == 5:
                text.SetVisibility(0)

            # DEVICE DESIGNATION
            elif text_type == 12:
                if pin_id == top_pin_id:
                    text.SetVisibility(1)
                    text.SetColour(0)
                    text.SetStyle(1)
                    text.SetHeight(0.2)
                else:
                    text.SetVisibility(0)

            # EVERYTHING ELSE ON THE PIN
            else:
                text.SetVisibility(0)
