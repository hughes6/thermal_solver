def format_main_block_view(device_view_id,x,y,width,height):
    x_center=x+(width/2.0); function_spacing=0.5 if DEVICE_FUNCTION is not None else 0.0
    dev.SetId(device_view_id)

    # ATTRIBUTE TEXTS
    attrinfo=dev.GetAttributeIds(None)
    for att_id in [a for a in attrinfo[1] if a is not None]:
        attr.SetId(att_id)
        textinfo=attr.GetTextIds(None)
        for text_id in [t for t in textinfo[1] if t is not None]:
            text.SetId(text_id); txt=text.GetText()
            if txt==ADDITIONAL_PART_ID:
                text.SetColour(0); text.SetMode(1); text.SetAlignment(2); text.SetStyle(1); text.SetHeight(0.2)
                text.SetSchemaLocation(x_center,y+height-1.1-function_spacing)
            elif DEVICE_FUNCTION is not None and txt==DEVICE_FUNCTION:
                text.SetColour(0); text.SetMode(1); text.SetAlignment(2); text.SetStyle(1); text.SetHeight(0.2)
                text.SetSchemaLocation(x_center,y+height-1.5-function_spacing)

    # SYMBOL TEXTS
    syminfo=dev.GetSymbolIds(None)
    for sym_id in [s for s in syminfo[1] if s is not None]:
        sym.SetId(sym_id); sym.SetBlockHatchColour(58)
        textinfo=sym.GetTextIds(None)
        for text_id in [t for t in textinfo[1] if t is not None]:
            text.SetId(text_id); txt=text.GetText()
            if txt=="BxRxUx":
                text.SetColour(0); text.SetMode(1); text.SetAlignment(2); text.SetStyle(1); text.SetHeight(0.2)
                text.SetSchemaLocation(x_center,y+height-0.3-function_spacing)
            elif txt==DEVICE_DESIGNATION:
                text.SetColour(0); text.SetMode(1); text.SetAlignment(2); text.SetStyle(1); text.SetHeight(0.2)
                text.SetSchemaLocation(x_center,y+height-0.7-function_spacing)
