Private Sub Worksheet_Change(ByVal Target As Range)

    Const BLOCK_WIDTH As Long = 5
    Const FIRST_COL As Long = 8          'H
    Const CENTER_COL As Long = 10        'J
    Const MAX_BLOCKS As Long = 40
    Const LAST_ROW As Long = 307

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim maxCreated As Long

    Dim srcHeader As Range
    Dim srcFull As Range
    Dim destHeader As Range
    Dim destFull As Range
    Dim oldCell As Range

    If Target.Address <> "$C$4" Then Exit Sub

    On Error GoTo ErrHandler

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    n = CLng(Me.Range("C4").Value)

    If n < 1 Then n = 1
    If n > MAX_BLOCKS Then n = MAX_BLOCKS

    Set srcHeader = Me.Range("H5:L10")
    Set srcFull = Me.Range("H5:L" & LAST_ROW)


    '==================================================
    'REMOVE DUPLICATE NAMES CREATED BY OLD MACRO
    '
    'Old macro put names in:
    'H6, M6, R6, W6...
    '
    'Correct cells are:
    'J6, O6, T6, Y6...
    '==================================================

    For i = 0 To MAX_BLOCKS - 1

        Set oldCell = Me.Cells(6, FIRST_COL + i * BLOCK_WIDTH)

        If oldCell.Formula = "=$B$" & (7 + i) Then
            oldCell.ClearContents
        End If

    Next i


    '==================================================
    'FIND HOW MANY BLOCKS ALREADY EXIST
    '==================================================

    maxCreated = 1

    For i = 1 To MAX_BLOCKS - 1

        Set destHeader = Me.Range( _
            Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
            Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

        If Application.WorksheetFunction.CountA(destHeader) > 0 Then
            maxCreated = i + 1
        Else
            Exit For
        End If

    Next i


    '==================================================
    'CREATE NEW BLOCKS ONLY
    '==================================================

    If n > maxCreated Then

        For i = maxCreated To n - 1

            Set destHeader = Me.Range( _
                Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
                Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

            Set destFull = Me.Range( _
                Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
                Me.Cells(LAST_ROW, FIRST_COL + i * BLOCK_WIDTH + 4))


            'Copy template
            srcHeader.Copy Destination:=destHeader


            'Clear copied Number of Pins value
            'J5, O5, T5, Y5...
            Me.Cells(5, CENTER_COL + i * BLOCK_WIDTH).MergeArea.ClearContents


            'Copy widths
            For j = 0 To BLOCK_WIDTH - 1

                Me.Columns(FIRST_COL + i * BLOCK_WIDTH + j).ColumnWidth = _
                    Me.Columns(FIRST_COL + j).ColumnWidth

            Next j


            'Copy formatting/rules/validation
            CopyBlockRules _
                srcFull, _
                destFull, _
                FIRST_COL, _
                FIRST_COL + i * BLOCK_WIDTH

        Next i

        maxCreated = n

    End If


    '==================================================
    'CONNECTOR NAMES
    '
    'J6 = B7
    'O6 = B8
    'T6 = B9
    'Y6 = B10
    '
    'J5/O5/T5/Y5 remain NUMBER OF PINS
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Cells(6, CENTER_COL + i * BLOCK_WIDTH).MergeArea.Cells(1, 1).Formula = _
            "=$B$" & (7 + i)

    Next i


    '==================================================
    'REFRESH RULES ON EXISTING BLOCKS
    'WITHOUT CHANGING THEIR DATA
    '==================================================

    For i = 1 To maxCreated - 1

        Set destFull = Me.Range( _
            Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
            Me.Cells(LAST_ROW, FIRST_COL + i * BLOCK_WIDTH + 4))

        CopyBlockRules _
            srcFull, _
            destFull, _
            FIRST_COL, _
            FIRST_COL + i * BLOCK_WIDTH

    Next i


    '==================================================
    'SHOW / HIDE BLOCKS
    '
    'Hidden blocks keep all entered data.
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Range( _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH), _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH + 4) _
        ).EntireColumn.Hidden = (i >= n)

    Next i


SafeExit:

    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub


ErrHandler:

    MsgBox "Macro error " & Err.Number & ":" & vbCrLf & Err.Description
    Resume SafeExit

End Sub


Private Sub CopyBlockRules( _
    ByVal srcFull As Range, _
    ByVal destFull As Range, _
    ByVal srcStartCol As Long, _
    ByVal destStartCol As Long)

    Dim fc As FormatCondition
    Dim f As String
    Dim i As Long
    Dim srcCol As String
    Dim destCol As String


    'Delete destination conditional formatting only
    destFull.FormatConditions.Delete


    'Copy formatting + conditional formatting
    srcFull.Copy
    destFull.PasteSpecial Paste:=xlPasteFormats


    'Copy dropdowns / validation
    srcFull.Copy

    On Error Resume Next
    destFull.PasteSpecial Paste:=xlPasteValidation
    On Error GoTo 0

    Application.CutCopyMode = False


    '==================================================
    'TRANSLATE CONDITIONAL FORMATTING
    '
    '$H8  -> $M8 -> $R8 -> $W8
    '$J$5 -> $O$5 -> $T$5 -> $Y$5
    '==================================================

    For Each fc In destFull.FormatConditions

        If fc.Type = xlExpression Then

            f = fc.Formula1

            For i = 0 To 4

                srcCol = ColLetter(srcStartCol + i)
                destCol = ColLetter(destStartCol + i)

                f = Replace( _
                    f, _
                    "$" & srcCol, _
                    "$" & destCol)

            Next i

            If f <> fc.Formula1 Then

                On Error Resume Next
                fc.Modify Type:=xlExpression, Formula1:=f
                On Error GoTo 0

            End If

        End If

    Next fc

End Sub


Private Function ColLetter(ByVal colNum As Long) As String

    ColLetter = Split(Me.Cells(1, colNum).Address, "$")(1)

End Function
