def safe_string(value):

    try:
        return str(value)
    except Exception:
        return repr(value)


def is_missing_value(value):

    if value is None:
        return True

    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value > 0:
            cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    if result is None:
        return []

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                count = int(items[0])
            except Exception:
                count = None

            values = clean_id_list(items[1])

            if count is not None and len(values) > count:
                values = values[-count:]

            return values

        return clean_id_list(items)

    return clean_id_list(result)


def call_id_array(method):

    try:
        result = method(None)

        values = parse_count_array_result(result)

        if values:
            return values

    except Exception:
        pass

    try:
        result = method()

        values = parse_count_array_result(result)

        if values:
            return values

    except Exception:
        pass

    return []


def call_noarg(obj, method_name):

    if not hasattr(obj, method_name):
        return None

    try:
        return getattr(obj, method_name)()
    except Exception:
        return None


def extract_numeric_values(result):

    values = []

    if not isinstance(result, (tuple, list)):
        return values

    for value in result:

        if is_missing_value(value):
            continue

        try:
            values.append(float(value))
        except Exception:
            pass

    return values

def get_device_ids(prj):

    return call_id_array(prj.GetAllDeviceIds)


def get_pin_ids(dev):

    return call_id_array(dev.GetPinIds)


def get_view_ids(pin):

    return call_id_array(pin.GetViewIds)


def get_node_ids(pin):

    return call_id_array(pin.GetNodeIds)


def get_bound_ids(sym):

    return call_id_array(sym.GetBoundIds)


def get_text_ids(sym):

    return call_id_array(sym.GetTextIds)


def get_pin_name(pin):

    try:
        return str(pin.GetName())
    except Exception:
        return ""


def find_connector(prj, dev):

    section(f"FIND CONNECTOR {CONNECTOR_NAME}")

    for device_id in get_device_ids(prj):

        dev.SetId(device_id)

        try:
            name = str(dev.GetName())
        except Exception:
            name = ""

        log(f"DEVICE {device_id}: {name}")

        if name == CONNECTOR_NAME:
            log(f"FOUND CONNECTOR ID: {device_id}")
            return device_id

    raise RuntimeError(f"Could not find connector {CONNECTOR_NAME}.")


def find_logical_pin(dev, pin, connector_device_id, pin_name):

    dev.SetId(connector_device_id)

    for logical_pin_id in get_pin_ids(dev):

        pin.SetId(logical_pin_id)

        if get_pin_name(pin) == str(pin_name):

            return {
                "pin_name": str(pin_name),
                "logical_pin_id": logical_pin_id,
                "view_ids": get_view_ids(pin),
            }

    return None

def get_pin_view_schema_location(pin, pin_view_id):

    pin.SetId(pin_view_id)

    attempts = [
        lambda: pin.GetSchemaLocation(None, None, None, None, None),
        lambda: pin.GetSchemaLocation(None, None, None),
        lambda: pin.GetSchemaLocation(),
    ]

    for attempt in attempts:

        try:
            result = attempt()
        except Exception:
            continue

        values = extract_numeric_values(result)

        if len(values) >= 3:
            return values

    return None


def get_symbol_schema_location(sym):

    attempts = [
        lambda: sym.GetSchemaLocation(None, None, None, None, None),
        lambda: sym.GetSchemaLocation(None, None, None),
        lambda: sym.GetSchemaLocation(),
    ]

    for attempt in attempts:

        try:
            result = attempt()
        except Exception:
            continue

        values = extract_numeric_values(result)

        if len(values) >= 2:
            return values

    return None


def resolve_live_symbol_from_pin_view(prj, pin_view_id):

    sym = prj.CreateSymbolObject()

    try:
        live_symbol_id = sym.SetId(pin_view_id)
    except Exception:
        live_symbol_id = 0

    if live_symbol_id is None:
        live_symbol_id = 0

    try:
        live_symbol_id = int(live_symbol_id)
    except Exception:
        live_symbol_id = 0

    if live_symbol_id <= 0:
        return None

    resolved_symbol_id = call_noarg(sym, "GetId")

    return {
        "symbol_object": sym,
        "live_symbol_id": live_symbol_id,
        "resolved_symbol_id": resolved_symbol_id,
    }


def inspect_live_symbol_by_id(prj, symbol_id):

    sym = prj.CreateSymbolObject()

    set_result = sym.SetId(symbol_id)

    if set_result is None:
        set_result = 0

    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0

    if set_result <= 0:
        return None

    return {
        "symbol_id": symbol_id,
        "resolved_id": call_noarg(sym, "GetId"),
        "symbol_type_name": call_noarg(sym, "GetSymbolTypeName"),
        "characteristic": call_noarg(sym, "GetCharacteristic"),
        "rotation": call_noarg(sym, "GetRotation"),
        "scaling": call_noarg(sym, "GetScaling"),
        "sheet_id": call_noarg(sym, "GetSheetId"),
        "schema_location": get_symbol_schema_location(sym),
        "bound_ids": get_bound_ids(sym),
        "text_ids": get_text_ids(sym),
    }

def snapshot_connector(prj, dev, pin, connector_device_id):

    snapshot = []

    dev.SetId(connector_device_id)

    for logical_pin_id in get_pin_ids(dev):

        pin.SetId(logical_pin_id)

        pin_name = get_pin_name(pin)
        view_ids = get_view_ids(pin)

        record = {
            "pin_name": pin_name,
            "logical_pin_id": logical_pin_id,
            "view_ids": view_ids,
            "views": [],
        }

        for pin_view_id in view_ids:

            location = get_pin_view_schema_location(pin, pin_view_id)
            node_ids = get_node_ids(pin)
            live = resolve_live_symbol_from_pin_view(prj, pin_view_id)

            view_record = {
                "pin_view_id": pin_view_id,
                "location": location,
                "node_ids": node_ids,
                "live_symbol_id": None,
                "symbol_type_name": None,
                "characteristic": None,
                "rotation": None,
                "scaling": None,
                "bound_ids": [],
            }

            if live is not None:

                live_symbol_id = live["live_symbol_id"]
                sym = live["symbol_object"]

                view_record["live_symbol_id"] = live_symbol_id
                view_record["symbol_type_name"] = call_noarg(sym, "GetSymbolTypeName")
                view_record["characteristic"] = call_noarg(sym, "GetCharacteristic")
                view_record["rotation"] = call_noarg(sym, "GetRotation")
                view_record["scaling"] = call_noarg(sym, "GetScaling")
                view_record["bound_ids"] = get_bound_ids(sym)

            record["views"].append(view_record)

        snapshot.append(record)

    return snapshot


def print_snapshot(snapshot, title):

    section(title)

    for record in snapshot:

        log(f"PIN {record['pin_name']} | LOGICAL PIN ID {record['logical_pin_id']} | VIEW IDS {record['view_ids']}")

        for view in record["views"]:

            log(f"    VIEW {view['pin_view_id']} | LIVE SYMBOL {view['live_symbol_id']} | LOCATION {view['location']} | NODE IDS {view['node_ids']} | TYPE {view['symbol_type_name']} | CHARACTERISTIC {view['characteristic']} | BOUNDS {view['bound_ids']}")


def get_top_sorted_live_pin_records(snapshot):

    candidates = []

    for record in snapshot:

        if len(record["views"]) != 1:
            continue

        view = record["views"][0]

        location = view["location"]

        if location is None or len(location) < 3:
            continue

        if view["live_symbol_id"] is None:
            continue

        candidates.append({
            "pin_name": record["pin_name"],
            "logical_pin_id": record["logical_pin_id"],
            "pin_view_id": view["pin_view_id"],
            "live_symbol_id": view["live_symbol_id"],
            "sheet_id": int(location[0]),
            "x": float(location[1]),
            "y": float(location[2]),
            "grid": float(location[3]) if len(location) >= 4 else None,
            "symbol_type_name": view["symbol_type_name"],
            "characteristic": view["characteristic"],
            "rotation": view["rotation"],
            "scaling": view["scaling"],
        })

    candidates.sort(key=lambda item: item["y"], reverse=True)

    return candidates

def delete_live_symbol(prj, live_symbol_id):

    sym = prj.CreateSymbolObject()

    set_result = sym.SetId(live_symbol_id)

    if set_result is None:
        set_result = 0

    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0

    if set_result <= 0:
        raise RuntimeError(f"Could not resolve live symbol ID {live_symbol_id}.")

    delete_result = sym.Delete()

    return delete_result


def verify_ghost(prj, pin_view_id):

    sym = prj.CreateSymbolObject()

    set_result = sym.SetId(pin_view_id)

    if set_result is None:
        set_result = 0

    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0

    return set_result == 0


def place_ghost_pin_view(prj, pin_view_id, sheet_id, x, y, rotation, scaling):

    placement = prj.CreateSymbolObject()

    new_symbol_id = placement.PlacePinView(pin_view_id, sheet_id, DETACHED_SYMBOL_NAME, DETACHED_SYMBOL_VERSION, x, y, rotation, scaling, MAINTAIN_TEXT_SIZE)

    if new_symbol_id is None:
        new_symbol_id = 0

    try:
        new_symbol_id = int(new_symbol_id)
    except Exception:
        new_symbol_id = 0

    if new_symbol_id <= 0:
        raise RuntimeError(f"PlacePinView failed for stable pin-view ID {pin_view_id}.")

    return new_symbol_id


def move_pin_by_delete_recreate(prj, pin_view_id, current_live_symbol_id, sheet_id, x, y, rotation, scaling):

    delete_result = delete_live_symbol(prj, current_live_symbol_id)

    time.sleep(0.15)

    ghost_ok = verify_ghost(prj, pin_view_id)

    if not ghost_ok:
        raise RuntimeError(f"Pin-view ID {pin_view_id} did not become ghost/unplaced after deleting symbol {current_live_symbol_id}.")

    new_symbol_id = place_ghost_pin_view(prj, pin_view_id, sheet_id, x, y, rotation, scaling)

    time.sleep(0.15)

    new_metadata = inspect_live_symbol_by_id(prj, new_symbol_id)

    if new_metadata is None:
        raise RuntimeError(f"Returned symbol ID {new_symbol_id} could not be resolved.")

    return {
        "delete_result": delete_result,
        "new_symbol_id": new_symbol_id,
        "new_metadata": new_metadata,
    }

# ============================================================

def peel_pin_to_staging(prj, dev, pin, connector_device_id, pin_name, stage_x, stage_y):

    section(f"PEEL PIN {pin_name} TO STAGING")

    pin_record = find_logical_pin(dev, pin, connector_device_id, pin_name)

    if pin_record is None:
        raise RuntimeError(f"Could not find pin {pin_name}.")

    if len(pin_record["view_ids"]) != 1:
        raise RuntimeError(f"Pin {pin_name} must have exactly one stable view.")

    pin_view_id = pin_record["view_ids"][0]

    location = get_pin_view_schema_location(pin, pin_view_id)

    if location is None or len(location) < 3:
        raise RuntimeError(f"Pin {pin_name} has no live pin-view location.")

    live = resolve_live_symbol_from_pin_view(prj, pin_view_id)

    if live is None:
        raise RuntimeError(f"Could not resolve current live symbol for pin-view ID {pin_view_id}.")

    current_live_symbol_id = live["live_symbol_id"]
    current_sym = live["symbol_object"]

    characteristic = call_noarg(current_sym, "GetCharacteristic")
    symbol_type_name = call_noarg(current_sym, "GetSymbolTypeName")
    rotation = call_noarg(current_sym, "GetRotation")
    scaling = call_noarg(current_sym, "GetScaling")

    if characteristic not in ("END_TOP", "END_BOTTOM", "SPLIT_BOTTOM"):
        raise RuntimeError(f"Pin {pin_name} is not currently an endpoint. Characteristic: {characteristic}")

    if not isinstance(rotation, str) or rotation == "":
        rotation = "y"

    try:
        scaling = float(scaling)
    except Exception:
        scaling = 1.0

    sheet_id = int(location[0])

    log(f"PIN NAME: {pin_name}")
    log(f"LOGICAL PIN ID: {pin_record['logical_pin_id']}")
    log(f"STABLE PIN-VIEW ID: {pin_view_id}")
    log(f"CURRENT LIVE SYMBOL ID: {current_live_symbol_id}")
    log(f"CURRENT TYPE: {symbol_type_name}")
    log(f"CURRENT CHARACTERISTIC: {characteristic}")
    log(f"CURRENT LOCATION: {location}")
    log(f"STAGING LOCATION: ({stage_x}, {stage_y})")

    move_result = move_pin_by_delete_recreate(prj, pin_view_id, current_live_symbol_id, sheet_id, stage_x, stage_y, rotation, scaling)

    new_symbol_id = move_result["new_symbol_id"]
    new_metadata = move_result["new_metadata"]

    log(f"NEW STAGED SYMBOL ID: {new_symbol_id}")
    log(f"NEW TYPE: {new_metadata['symbol_type_name']}")
    log(f"NEW CHARACTERISTIC: {new_metadata['characteristic']}")
    log(f"NEW SYMBOL LOCATION: {new_metadata['schema_location']}")
    log(f"NEW BOUNDS: {new_metadata['bound_ids']}")

    return {
        "pin_name": pin_name,
        "logical_pin_id": pin_record["logical_pin_id"],
        "pin_view_id": pin_view_id,
        "original_live_symbol_id": current_live_symbol_id,
        "current_live_symbol_id": new_symbol_id,
        "sheet_id": sheet_id,
        "rotation": rotation,
        "scaling": scaling,
        "original_location": location,
        "stage_x": stage_x,
        "stage_y": stage_y,
        "stage_symbol_id": new_symbol_id,
        "stage_metadata": new_metadata,
    }

def apply_text_tweaks(prj, live_symbol_id):

    section(f"TEXT TWEAKS FOR SYMBOL {live_symbol_id}")

    sym = prj.CreateSymbolObject()
    text = prj.CreateTextObject()

    set_result = sym.SetId(live_symbol_id)

    if set_result is None:
        set_result = 0

    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0

    if set_result <= 0:
        raise RuntimeError(f"Could not resolve live symbol {live_symbol_id} for text formatting.")

    tweaks = []

    try:
        hatch_result = sym.SetBlockHatchColour(58)
    except Exception as exc:
        hatch_result = f"ERROR: {exc}"

    log(f"SetBlockHatchColour(58): {hatch_result}")

    for text_id in get_text_ids(sym):

        text.SetId(text_id)

        try:
            text_type = text.GetType()
        except Exception:
            continue

        try:
            text_value = str(text.GetText())
        except Exception:
            text_value = ""

        actions = []

        if text_type == 5:

            try:
                result = text.SetVisibility(0)
                actions.append(f"SetVisibility(0)={result}")
            except Exception as exc:
                actions.append(f"SetVisibility ERROR={exc}")

        elif text_type == 12:

            try:
                result = text.SetVisibility(0)
                actions.append(f"SetVisibility(0)={result}")
            except Exception as exc:
                actions.append(f"SetVisibility ERROR={exc}")

        elif text_type == 617:

            try:
                actions.append(f"SetVisibility(1)={text.SetVisibility(1)}")
            except Exception as exc:
                actions.append(f"SetVisibility ERROR={exc}")

            try:
                actions.append(f"SetColour(0)={text.SetColour(0)}")
            except Exception as exc:
                actions.append(f"SetColour ERROR={exc}")

            try:
                actions.append(f"SetAlignment(3)={text.SetAlignment(3)}")
            except Exception as exc:
                actions.append(f"SetAlignment ERROR={exc}")

            try:
                actions.append(f"SetStyle(1)={text.SetStyle(1)}")
            except Exception as exc:
                actions.append(f"SetStyle ERROR={exc}")

            try:
                actions.append(f"SetHeight(0.2)={text.SetHeight(0.2)}")
            except Exception as exc:
                actions.append(f"SetHeight ERROR={exc}")

        elif text_type == 203:

            try:
                actions.append(f"SetVisibility(1)={text.SetVisibility(1)}")
            except Exception as exc:
                actions.append(f"SetVisibility ERROR={exc}")

            try:
                actions.append(f"SetStyle(1)={text.SetStyle(1)}")
            except Exception as exc:
                actions.append(f"SetStyle ERROR={exc}")

        elif text_type == 213:

            actions.append("LEFT ALONE")

        if actions:

            log(f"TEXT {text_id} | TYPE {text_type} | VALUE {repr(text_value)}")
            log(f"    {' | '.join(actions)}")

            tweaks.append({
                "text_id": text_id,
                "text_type": text_type,
                "text_value": text_value,
                "actions": actions,
            })

    return tweaks

def build_final_plan(peeled_records, center_x, top_y):

    if len(peeled_records) != 3:
        raise RuntimeError("This proof-of-process script expects exactly three peeled pins.")

    first = peeled_records[0]
    second = peeled_records[1]
    third = peeled_records[2]

    left_x = center_x + LEFT_X_OFFSET
    right_x = center_x + RIGHT_X_OFFSET
    row_0_y = top_y + FINAL_Y_OFFSET
    row_1_y = row_0_y - FINAL_Y_SPACING

    plan = [
        {
            "pin_name": third["pin_name"],
            "side": "LEFT",
            "row": 0,
            "x": left_x,
            "y": row_0_y,
        },
        {
            "pin_name": first["pin_name"],
            "side": "RIGHT",
            "row": 0,
            "x": right_x,
            "y": row_0_y,
        },
        {
            "pin_name": second["pin_name"],
            "side": "LEFT",
            "row": 1,
            "x": left_x,
            "y": row_1_y,
        },
    ]

    return plan


def find_peeled_record(peeled_records, pin_name):

    for record in peeled_records:

        if record["pin_name"] == pin_name:
            return record

    raise RuntimeError(f"Could not find peeled record for pin {pin_name}.")


# ============================================================
# PLACE STAGED PINS INTO FINAL OUT-OF-ORDER POSITIONS
# ============================================================

def move_staged_pin_to_final(prj, peeled_record, final_item):

    section(f"FINAL PLACE PIN {peeled_record['pin_name']} -> {final_item['side']} ROW {final_item['row']}")

    pin_view_id = peeled_record["pin_view_id"]
    current_live_symbol_id = peeled_record["current_live_symbol_id"]
    sheet_id = peeled_record["sheet_id"]
    rotation = peeled_record["rotation"]
    scaling = peeled_record["scaling"]

    x = final_item["x"]
    y = final_item["y"]

    log(f"STABLE PIN-VIEW ID: {pin_view_id}")
    log(f"CURRENT STAGED SYMBOL ID: {current_live_symbol_id}")
    log(f"FINAL X: {x}")
    log(f"FINAL Y: {y}")

    move_result = move_pin_by_delete_recreate(prj, pin_view_id, current_live_symbol_id, sheet_id, x, y, rotation, scaling)

    new_symbol_id = move_result["new_symbol_id"]
    metadata = move_result["new_metadata"]

    log(f"FINAL SYMBOL ID: {new_symbol_id}")
    log(f"FINAL TYPE: {metadata['symbol_type_name']}")
    log(f"FINAL CHARACTERISTIC: {metadata['characteristic']}")
    log(f"FINAL LOCATION: {metadata['schema_location']}")
    log(f"FINAL BOUNDS: {metadata['bound_ids']}")

    tweaks = apply_text_tweaks(prj, new_symbol_id)

    peeled_record["current_live_symbol_id"] = new_symbol_id
    peeled_record["final_symbol_id"] = new_symbol_id
    peeled_record["final_side"] = final_item["side"]
    peeled_record["final_row"] = final_item["row"]
    peeled_record["final_x"] = x
    peeled_record["final_y"] = y
    peeled_record["final_metadata"] = metadata
    peeled_record["text_tweaks"] = tweaks

    return peeled_record

def main():

    RUN_RECORD["started"] = datetime.now().isoformat(timespec="milliseconds")

    section("E3 PEEL FIRST THREE -> STAGE CENTER -> REORDER LEFT/RIGHT")

    log("PROCESS:")
    log("1) Record every connector pin and original location.")
    log("2) Identify the first three pins from the top.")
    log("3) Peel them one at a time using the proven stable pin-view-ID workflow.")
    log("4) Place each peeled pin in a center staging strip.")
    log("5) Re-place those three out of order into left/right test positions.")
    log("6) Apply split text tweaks to the final live symbols.")
    log()
    log(f"DRY_RUN = {DRY_RUN}")

    app, prj, dev, pin = connect_to_e3()

    connector_device_id = find_connector(prj, dev)

    verify_detached_symbol_load(prj)

    initial_snapshot = snapshot_connector(prj, dev, pin, connector_device_id)

    RUN_RECORD["initial_snapshot"] = initial_snapshot

    print_snapshot(initial_snapshot, "INITIAL CONNECTOR SNAPSHOT")

    sorted_pins = get_top_sorted_live_pin_records(initial_snapshot)

    if len(sorted_pins) < PEEL_COUNT:
        raise RuntimeError(f"Need at least {PEEL_COUNT} live connector pins.")

    peel_targets = sorted_pins[:PEEL_COUNT]

    section("FIRST THREE PINS FROM TOP")

    for index, target in enumerate(peel_targets, start=1):
        log(f"{index}: PIN {target['pin_name']} | VIEW {target['pin_view_id']} | LOCATION ({target['x']}, {target['y']}) | TYPE {target['symbol_type_name']} | CHARACTERISTIC {target['characteristic']}")

    center_x = peel_targets[0]["x"]
    original_top_y = peel_targets[0]["y"]
    stage_y_start = original_top_y + STAGING_Y_OFFSET

    section("COMPUTED TEST COORDINATES")

    log(f"CENTER / STAGING X: {center_x}")
    log(f"STAGING Y START: {stage_y_start}")
    log(f"STAGING Y SPACING: {STAGING_Y_SPACING}")
    log(f"LEFT FINAL X: {center_x + LEFT_X_OFFSET}")
    log(f"RIGHT FINAL X: {center_x + RIGHT_X_OFFSET}")
    log(f"FINAL TOP Y: {original_top_y + FINAL_Y_OFFSET}")
    log(f"FINAL Y SPACING: {FINAL_Y_SPACING}")

    if DRY_RUN:

        section("DRY RUN PLAN")

        for index, target in enumerate(peel_targets):

            stage_y = stage_y_start - (index * STAGING_Y_SPACING)

            log(f"PEEL PIN {target['pin_name']} -> STAGE ({center_x}, {stage_y})")

        fake_records = []

        for target in peel_targets:

            fake_records.append({
                "pin_name": target["pin_name"],
            })

        final_plan = build_final_plan(fake_records, center_x, original_top_y)

        log()
        log("OUT-OF-ORDER FINAL PLAN:")

        for item in final_plan:
            log(f"PIN {item['pin_name']} -> {item['side']} ROW {item['row']} -> ({item['x']}, {item['y']})")

        log()
        log("No E3 objects were changed.")
        log("Set DRY_RUN = False to run the proof-of-process.")

        return

    if REQUIRE_RUN_CONFIRMATION:

        confirmation = input("Type RUN THREE PIN REORDER to execute the mutation test: ").strip()

        if confirmation != "RUN THREE PIN REORDER":
            log("RUN CANCELLED")
            return

    peeled_records = []

    for index, target in enumerate(peel_targets):

        stage_y = stage_y_start - (index * STAGING_Y_SPACING)

        peeled = peel_pin_to_staging(prj, dev, pin, connector_device_id, target["pin_name"], center_x, stage_y)

        peeled_records.append(peeled)

        RUN_RECORD["peeled"] = peeled_records

        time.sleep(0.2)

    staged_snapshot = snapshot_connector(prj, dev, pin, connector_device_id)

    RUN_RECORD["staged_snapshot"] = staged_snapshot

    print_snapshot(staged_snapshot, "CONNECTOR AFTER FIRST THREE ARE STAGED")

    final_plan = build_final_plan(peeled_records, center_x, original_top_y)

    RUN_RECORD["final_plan"] = final_plan

    section("OUT-OF-ORDER FINAL PLAN")

    for item in final_plan:
        log(f"PIN {item['pin_name']} -> {item['side']} ROW {item['row']} -> ({item['x']}, {item['y']})")

    for final_item in final_plan:

        peeled_record = find_peeled_record(peeled_records, final_item["pin_name"])

        move_staged_pin_to_final(prj, peeled_record, final_item)

        time.sleep(0.2)

    final_snapshot = snapshot_connector(prj, dev, pin, connector_device_id)

    RUN_RECORD["peeled"] = peeled_records
    RUN_RECORD["final_snapshot"] = final_snapshot

    print_snapshot(final_snapshot, "FINAL CONNECTOR SNAPSHOT")

    section("FINAL SUMMARY")

    log("PEELED / FINAL PINS:")

    for record in peeled_records:

        log(f"PIN {record['pin_name']} | STABLE VIEW {record['pin_view_id']} | FINAL SYMBOL {record.get('final_symbol_id')} | SIDE {record.get('final_side')} | ROW {record.get('final_row')} | TARGET ({record.get('final_x')}, {record.get('final_y')})")

    log()
    log("OUT-OF-ORDER TEST COMPLETE.")
    log("The JSON output contains the original connector locations, staging records, final plan, final symbol IDs, and text-tweak results.")
