print("\n" + "=" * 80)
print("CONNECTORS + NUMBER OF PINS")
print("=" * 80)

cur.execute("""
SELECT
    c.ENTRY,
    c.VERSION,
    c.Description,
    c.Class,
    COUNT(cp.component_id) AS NumberOfPins
FROM dbo.ComponentData c
LEFT JOIN dbo.component_pins cp
    ON cp.component_id = c.ID
WHERE c.ENTRYTYP = 24
GROUP BY
    c.ID,
    c.ENTRY,
    c.VERSION,
    c.Description,
    c.Class
ORDER BY
    c.ENTRY,
    c.VERSION
""")

t = 0

for entry, version, description, component_class, number_of_pins in cur.fetchall():

    component_class = str(component_class)

    if "SIMS" in component_class or "SCAU" in component_class or "MAU" in component_class or "Coaxial" in component_class:

        print(
            entry,
            "| VERSION:",
            version,
            "| PINS:",
            number_of_pins,
            "|",
            component_class,
            "|",
            description
        )

        t += 1

print()
print("TOTAL CONNECTORS:", t)
