def is_missing_value(value):

    if value is None:
        return True

    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value > 0:
            cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    if result is None:
        return []

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                count = int(items[0])
            except Exception:
                count = None

            values = clean_id_list(items[1])

            if count is not None and len(values) > count:
                values = values[-count:]

            return values

        return clean_id_list(items)

    return clean_id_list(result)


def call_id_array(method):

    try:
        result = method(None)
        values = parse_count_array_result(result)

        if values:
            return values

    except Exception:
        pass

    try:
        result = method()
        values = parse_count_array_result(result)

        if values:
            return values

    except Exception:
        pass

    return []


def call_noarg(obj, method_name):

    if not hasattr(obj, method_name):
        return None

    try:
        return getattr(obj, method_name)()
    except Exception:
        return None


def normalize_pin_value(value):

    if value is None:
        return None

    if isinstance(value, bool):
        return str(value)

    if isinstance(value, int):
        return str(value)

    if isinstance(value, float):

        if math.isfinite(value) and value.is_integer():
            return str(int(value))

        return str(value).strip()

    return str(value).strip()


def median(values):

    if not values:
        return None

    ordered = sorted(values)
    middle = len(ordered) // 2

    if len(ordered) % 2 == 0:
        return (ordered[middle - 1] + ordered[middle]) / 2.0

    return ordered[middle]

def get_symbol_schema_location(sym):

    try:
        result = sym.GetSchemaLocation(None, None, None, None, None)
    except Exception:
        return None

    if not isinstance(result, (tuple, list)):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None

    return {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "raw": result,
    }


def get_symbol_placed_area(sym):

    try:
        result = sym.GetPlacedArea(None, None, None, None)
    except Exception:
        return None

    if not isinstance(result, (tuple, list)):
        return None

    if len(result) < 5:
        return None

    try:
        success = int(result[0])
        left = float(result[1])
        bottom = float(result[2])
        right = float(result[3])
        top = float(result[4])
    except Exception:
        return None

    if success <= 0:
        return None

    return {
        "left": left,
        "bottom": bottom,
        "right": right,
        "top": top,
        "width": right - left,
        "height": top - bottom,
        "center_x": (left + right) / 2.0,
        "center_y": (bottom + top) / 2.0,
        "raw": result,
    }


def get_pin_schema_location(pin, pin_view_id):

    pin.SetId(pin_view_id)

    try:
        result = pin.GetSchemaLocation(None, None, None, None, None)
    except Exception:
        return None

    if not isinstance(result, (tuple, list)):
        return None

    numeric_values = []

    for value in result:

        if is_missing_value(value):
            continue

        try:
            numeric_values.append(float(value))
        except Exception:
            pass

    if len(numeric_values) < 3:
        return None

    return numeric_values

def snapshot_connector(prj, dev, pin, connector_device_id):

    snapshot = []

    dev.SetId(connector_device_id)

    for logical_pin_id in get_pin_ids(dev):

        pin.SetId(logical_pin_id)

        pin_name = normalize_pin_value(pin.GetName())
        pin_view_ids = get_view_ids(pin)

        record = {
            "pin_name": pin_name,
            "logical_pin_id": logical_pin_id,
            "view_ids": pin_view_ids,
            "views": [],
        }

        for pin_view_id in pin_view_ids:

            location = get_pin_schema_location(pin, pin_view_id)
            node_ids = get_node_ids(pin)
            live = resolve_live_symbol_from_pin_view(prj, pin_view_id)

            view_record = {
                "pin_view_id": pin_view_id,
                "location": location,
                "node_ids": node_ids,
                "live_symbol_id": None,
                "symbol_type_name": None,
                "characteristic": None,
                "rotation": None,
                "scaling": None,
                "bound_ids": [],
            }

            if live is not None:

                live_sym = live["symbol_object"]

                view_record["live_symbol_id"] = live["live_symbol_id"]
                view_record["symbol_type_name"] = call_noarg(live_sym, "GetSymbolTypeName")
                view_record["characteristic"] = call_noarg(live_sym, "GetCharacteristic")
                view_record["rotation"] = call_noarg(live_sym, "GetRotation")
                view_record["scaling"] = call_noarg(live_sym, "GetScaling")
                view_record["bound_ids"] = get_bound_ids(live_sym)

            record["views"].append(view_record)

        snapshot.append(record)

    return snapshot


def get_sorted_live_pin_records(snapshot):

    records = []

    for pin_record in snapshot:

        if len(pin_record["views"]) != 1:
            raise RuntimeError(f"Pin {pin_record['pin_name']} does not have exactly one pin view.")

        view = pin_record["views"][0]

        if view["live_symbol_id"] is None:
            raise RuntimeError(f"Pin {pin_record['pin_name']} does not have a live symbol.")

        location = view["location"]

        if location is None or len(location) < 3:
            raise RuntimeError(f"Pin {pin_record['pin_name']} does not have a valid schema location.")

        records.append({
            "pin_name": pin_record["pin_name"],
            "logical_pin_id": pin_record["logical_pin_id"],
            "pin_view_id": view["pin_view_id"],
            "live_symbol_id": view["live_symbol_id"],
            "sheet_id": int(location[0]),
            "x": float(location[1]),
            "y": float(location[2]),
            "symbol_type_name": view["symbol_type_name"],
            "characteristic": view["characteristic"],
            "rotation": view["rotation"],
            "scaling": view["scaling"],
        })

    records.sort(key=lambda item: item["y"], reverse=True)

    return records


def get_connector_block_geometry(prj, dev, connector_device_id, main_device_id):

    connector_view_ids = get_device_view_ids(dev, connector_device_id)

    if not connector_view_ids:
        raise RuntimeError("Connector has no device view IDs.")

    connector_view_numbers = []

    for connector_view_id in connector_view_ids:

        view_number = get_view_number(dev, connector_view_id)

        if view_number is not None and view_number not in connector_view_numbers:
            connector_view_numbers.append(view_number)

    if len(connector_view_numbers) != 1:
        raise RuntimeError(f"Expected one connector view number, found {connector_view_numbers}.")

    connector_view_number = connector_view_numbers[0]

    main_view_ids = get_device_view_ids(dev, main_device_id)

    matching_main_view_id = None

    for main_view_id in main_view_ids:

        main_view_number = get_view_number(dev, main_view_id)

        if main_view_number == connector_view_number:
            matching_main_view_id = main_view_id
            break

    if matching_main_view_id is None:
        raise RuntimeError(f"Could not find main-device view number {connector_view_number}.")

    main_view_symbol_ids = get_device_symbol_ids(dev, matching_main_view_id)

    block_symbol_id = None
    block_symbol_type = None
    block_location = None
    block_area = None

    for symbol_id in main_view_symbol_ids:

        local_sym = prj.CreateSymbolObject()

        try:
            set_result = local_sym.SetId(symbol_id)
        except Exception:
            continue

        if not set_result:
            continue

        try:
            symbol_type_name = str(local_sym.GetSymbolTypeName())
        except Exception:
            symbol_type_name = ""

        if "CABBLOCK" not in symbol_type_name.upper():
            continue

        block_symbol_id = symbol_id
        block_symbol_type = symbol_type_name
        block_location = get_symbol_schema_location(local_sym)
        block_area = get_symbol_placed_area(local_sym)
        break

    if block_symbol_id is None:
        raise RuntimeError(f"Main-device view {connector_view_number} has no CABBLOCK symbol.")

    if block_area is None:
        raise RuntimeError(f"CABBLOCK symbol {block_symbol_id} did not return a valid placed area.")

    return {
        "connector_view_number": connector_view_number,
        "connector_view_ids": connector_view_ids,
        "main_view_id": matching_main_view_id,
        "block_symbol_id": block_symbol_id,
        "block_symbol_type": block_symbol_type,
        "block_schema_location": block_location,
        "block_left": block_area["left"],
        "block_right": block_area["right"],
        "block_bottom": block_area["bottom"],
        "block_top": block_area["top"],
        "block_width": block_area["width"],
        "block_height": block_area["height"],
        "block_center_x": block_area["center_x"],
        "block_center_y": block_area["center_y"],
        "sheet_id": block_location["sheet_id"] if block_location is not None else None,
    }


def derive_pin_geometry(block_geometry, sorted_pin_records, connector_excel_record):

    y_differences = []

    for index in range(len(sorted_pin_records) - 1):

        difference = abs(sorted_pin_records[index]["y"] - sorted_pin_records[index + 1]["y"])

        if difference > 0.000001:
            y_differences.append(difference)

    pin_pitch = median(y_differences)

    if pin_pitch is None:
        pin_pitch = 0.25

    original_top_y = sorted_pin_records[0]["y"]

    left_x = block_geometry["block_left"] + PIN_EDGE_INSET
    right_x = block_geometry["block_right"] - PIN_EDGE_INSET

    left_count = len(connector_excel_record["left pins"])
    right_count = len(connector_excel_record["right pins"])
    max_side_count = max(left_count, right_count)

    if max_side_count > 0:
        required_bottom_y = original_top_y - ((max_side_count - 1) * pin_pitch)
    else:
        required_bottom_y = original_top_y

    minimum_allowed_y = block_geometry["block_bottom"] + BLOCK_BOTTOM_CLEARANCE

    if required_bottom_y < minimum_allowed_y:
        raise RuntimeError(f"Block is not tall enough for the requested pin order. Required bottom Y is {required_bottom_y}, allowed bottom Y is {minimum_allowed_y}. Automatic block resizing is not implemented yet.")

    if original_top_y > block_geometry["block_top"]:
        raise RuntimeError(f"Existing top pin Y {original_top_y} is above block top {block_geometry['block_top']}.")

    return {
        "pin_pitch": pin_pitch,
        "original_top_y": original_top_y,
        "final_top_y": original_top_y,
        "left_x": left_x,
        "right_x": right_x,
        "staging_center_x": block_geometry["block_center_x"],
        "staging_center_y": block_geometry["block_center_y"],
        "block_left": block_geometry["block_left"],
        "block_right": block_geometry["block_right"],
        "block_bottom": block_geometry["block_bottom"],
        "block_top": block_geometry["block_top"],
        "sheet_id": block_geometry["sheet_id"],
    }


def build_staging_map(pin_records, pin_geometry):

    staging_map = {}

    if not pin_records:
        return staging_map

    row_count = math.ceil(len(pin_records) / 2)

    if row_count <= 1:
        first_row_y = pin_geometry["staging_center_y"]
    else:
        total_height = (row_count - 1) * STAGING_ROW_SPACING
        first_row_y = pin_geometry["staging_center_y"] + (total_height / 2.0)

    minimum_stage_y = first_row_y - ((row_count - 1) * STAGING_ROW_SPACING)
    maximum_stage_y = first_row_y

    allowed_left_x = pin_geometry["block_left"] + STAGING_EDGE_CLEARANCE
    allowed_right_x = pin_geometry["block_right"] - STAGING_EDGE_CLEARANCE
    allowed_bottom_y = pin_geometry["block_bottom"] + STAGING_TOP_BOTTOM_CLEARANCE
    allowed_top_y = pin_geometry["block_top"] - STAGING_TOP_BOTTOM_CLEARANCE

    if minimum_stage_y < allowed_bottom_y or maximum_stage_y > allowed_top_y:
        raise RuntimeError(f"Temporary two-lane staging does not fit vertically inside the block. Staging Y range is {minimum_stage_y} to {maximum_stage_y}. Allowed block staging range is {allowed_bottom_y} to {allowed_top_y}.")

    used_x_values = set()

    for index, pin_record in enumerate(pin_records):

        lane_index = index % 2
        row_index = index // 2

        if lane_index == 0:
            x = pin_geometry["staging_center_x"] - STAGING_LANE_OFFSET - (row_index * STAGING_X_DRIFT_PER_ROW)
            lane_name = "CENTER LEFT"
        else:
            x = pin_geometry["staging_center_x"] + STAGING_LANE_OFFSET + (row_index * STAGING_X_DRIFT_PER_ROW)
            lane_name = "CENTER RIGHT"

        y = first_row_y - (row_index * STAGING_ROW_SPACING)

        x = round(x, 6)
        y = round(y, 6)

        if x < allowed_left_x or x > allowed_right_x:
            raise RuntimeError(f"Temporary two-lane staging is not wide enough for pin {pin_record['pin_name']}. Calculated X is {x}. Allowed block staging X range is {allowed_left_x} to {allowed_right_x}.")

        if x in used_x_values:
            raise RuntimeError(f"Temporary staging generated duplicate X coordinate {x}. No two staged pins may share the exact same X coordinate.")

        used_x_values.add(x)

        staging_map[pin_record["pin_name"]] = {
            "x": x,
            "y": y,
            "index": index,
            "lane": lane_name,
            "row": row_index,
        }

    return staging_map


def validate_excel_against_e3(connector_excel_record, sorted_pin_records):

    excel_pins = connector_excel_record["left pins"] + connector_excel_record["right pins"]
    e3_pins = [record["pin_name"] for record in sorted_pin_records]

    if len(e3_pins) != connector_excel_record["number of pins"]:
        raise RuntimeError(f"{connector_excel_record['connector name']}: E3 has {len(e3_pins)} pins but Excel expects {connector_excel_record['number of pins']}.")

    if set(excel_pins) != set(e3_pins):

        missing_from_excel = sorted(set(e3_pins) - set(excel_pins))
        missing_from_e3 = sorted(set(excel_pins) - set(e3_pins))

        raise RuntimeError(f"{connector_excel_record['connector name']}: Excel/E3 mismatch. Missing from Excel: {missing_from_excel}. Missing from E3: {missing_from_e3}.")

def move_pin_by_delete_recreate(prj, pin_view_id, current_live_symbol_id, sheet_id, x, y, rotation, scaling):

    delete_live_symbol(prj, current_live_symbol_id)

    time.sleep(0.15)

    if not verify_ghost(prj, pin_view_id):
        raise RuntimeError(f"Pin-view ID {pin_view_id} did not become a ghost after delete.")

    new_symbol_id = place_ghost_pin_view(prj, pin_view_id, sheet_id, x, y, rotation, scaling)

    time.sleep(0.15)

    metadata = inspect_live_symbol_by_id(prj, new_symbol_id)

    if metadata is None:
        raise RuntimeError(f"Returned symbol ID {new_symbol_id} could not be resolved.")

    return new_symbol_id, metadata


def find_logical_pin(dev, pin, connector_device_id, pin_name):

    target_name = normalize_pin_value(pin_name)

    dev.SetId(connector_device_id)

    for logical_pin_id in get_pin_ids(dev):

        pin.SetId(logical_pin_id)

            }

    return None


def peel_all_pins_to_staging(prj, dev, pin, connector_device_id, sorted_pin_records, staging_map, pin_geometry):

    staged_records = []

    allowed_characteristics = {
        "END_TOP",
        "END_BOTTOM",
        "SPLIT_BOTTOM",
        "END_SPLIT",
    }

    for index, original_record in enumerate(sorted_pin_records):

        pin_name = original_record["pin_name"]

        logical_pin = find_logical_pin(dev, pin, connector_device_id, pin_name)

        if logical_pin is None:
            raise RuntimeError(f"Could not find logical pin {pin_name}.")

        if len(logical_pin["view_ids"]) != 1:
            raise RuntimeError(f"Pin {pin_name} does not have exactly one stable pin-view ID.")

        pin_view_id = logical_pin["view_ids"][0]
        live = resolve_live_symbol_from_pin_view(prj, pin_view_id)

        if live is None:
            raise RuntimeError(f"Pin {pin_name} has no live symbol when its peel turn arrived.")

        live_symbol_id = live["live_symbol_id"]
        live_sym = live["symbol_object"]

        characteristic = call_noarg(live_sym, "GetCharacteristic")
        rotation = call_noarg(live_sym, "GetRotation")
        scaling = call_noarg(live_sym, "GetScaling")

        if characteristic not in allowed_characteristics:
            raise RuntimeError(f"Pin {pin_name} is not peelable. Current characteristic is {characteristic}.")

        if not isinstance(rotation, str) or rotation == "":
            rotation = "y"

        try:
            scaling = float(scaling)
        except Exception:
            scaling = 1.0

        stage = staging_map[pin_name]

        log(f"PEEL {index + 1}/{len(sorted_pin_records)} | PIN {pin_name} | {characteristic} -> ({stage['x']}, {stage['y']})")

        new_symbol_id, metadata = move_pin_by_delete_recreate(prj, pin_view_id, live_symbol_id, pin_geometry["sheet_id"], stage["x"], stage["y"], rotation, scaling)

        if metadata["symbol_type_name"] != DETACHED_SYMBOL_NAME:
            raise RuntimeError(f"Pin {pin_name} reattached during staging. Type is {metadata['symbol_type_name']}.")

        if metadata["characteristic"] != "END_SPLIT":
            raise RuntimeError(f"Pin {pin_name} reattached during staging. Characteristic is {metadata['characteristic']}.")

        if metadata["bound_ids"]:
            raise RuntimeError(f"Pin {pin_name} reattached during staging. Bounds are {metadata['bound_ids']}.")

        staged_records.append({
            "pin_name": pin_name,
            "logical_pin_id": original_record["logical_pin_id"],
            "pin_view_id": pin_view_id,
            "current_live_symbol_id": new_symbol_id,
            "rotation": rotation,
            "scaling": scaling,
            "sheet_id": pin_geometry["sheet_id"],
            "stage_x": stage["x"],
            "stage_y": stage["y"],
        })

    return staged_records


def build_final_plan(connector_excel_record, pin_geometry):

    final_plan = []

    for row_index, pin_name in enumerate(connector_excel_record["left pins"]):

        final_plan.append({
            "pin_name": pin_name,
            "side": "LEFT",
            "row": row_index,
            "x": pin_geometry["left_x"],
            "y": pin_geometry["final_top_y"] - (row_index * pin_geometry["pin_pitch"]),
        })

    for row_index, pin_name in enumerate(connector_excel_record["right pins"]):

        final_plan.append({
            "pin_name": pin_name,
            "side": "RIGHT",
            "row": row_index,
            "x": pin_geometry["right_x"],
            "y": pin_geometry["final_top_y"] - (row_index * pin_geometry["pin_pitch"]),
        })

    return final_plan


def find_staged_record(staged_records, pin_name):

    for record in staged_records:

        if record["pin_name"] == pin_name:
            return record

    raise RuntimeError(f"Could not find staged record for pin {pin_name}.")


def place_staged_pins(prj, staged_records, final_plan):

    for index, final_item in enumerate(final_plan):

        staged_record = find_staged_record(staged_records, final_item["pin_name"])

        log(f"PLACE {index + 1}/{len(final_plan)} | PIN {final_item['pin_name']} -> {final_item['side']} ROW {final_item['row']} -> ({final_item['x']}, {final_item['y']})")

        new_symbol_id, metadata = move_pin_by_delete_recreate(prj, staged_record["pin_view_id"], staged_record["current_live_symbol_id"], staged_record["sheet_id"], final_item["x"], final_item["y"], staged_record["rotation"], staged_record["scaling"])

        staged_record["current_live_symbol_id"] = new_symbol_id
        staged_record["final_returned_symbol_id"] = new_symbol_id
        staged_record["final_side"] = final_item["side"]
        staged_record["final_row"] = final_item["row"]
        staged_record["final_x"] = final_item["x"]
        staged_record["final_y"] = final_item["y"]
        staged_record["final_metadata_immediate"] = metadata

        time.sleep(0.20)


def collect_live_symbol_candidates(prj, sht, sheet_id):

    candidate_ids = set()

    if hasattr(prj, "GetSymbolIds"):

        try:
            for symbol_id in call_id_array(prj.GetSymbolIds):
                candidate_ids.add(symbol_id)
        except Exception:
            pass

    sht.SetId(sheet_id)

    if hasattr(sht, "GetSymbolIds"):

        try:
            for symbol_id in call_id_array(sht.GetSymbolIds):
                candidate_ids.add(symbol_id)
        except Exception:
            pass

    return sorted(candidate_ids)


def read_symbol_identity(prj, symbol_id):

    local_sym = prj.CreateSymbolObject()
    local_text = prj.CreateTextObject()

    try:
        set_result = local_sym.SetId(symbol_id)
    except Exception:
        return None

    if not set_result:
        return None

    pin_number = None
    device_text = None

    for text_id in get_text_ids(local_sym):

        try:
            local_text.SetId(text_id)
            text_type = local_text.GetType()
            text_value = str(local_text.GetText())
        except Exception:
            continue

        if text_type == 203:
            pin_number = normalize_pin_value(text_value)

        if text_type == 12:
            device_text = text_value

    return {
        "symbol_id": int(local_sym.GetId()),
        "pin_number": pin_number,
        "device_text": device_text,
        "location": get_symbol_schema_location(local_sym),
        "symbol_type_name": call_noarg(local_sym, "GetSymbolTypeName"),
        "characteristic": call_noarg(local_sym, "GetCharacteristic"),
    }


def coordinates_close(location, x, y):

    if location is None:
        return False

    try:
        current_x = float(location["x"])
        current_y = float(location["y"])
    except Exception:
        return False

    return abs(current_x - float(x)) <= SYMBOL_SCAN_COORDINATE_TOLERANCE and abs(current_y - float(y)) <= SYMBOL_SCAN_COORDINATE_TOLERANCE


def resolve_final_symbol(prj, candidate_ids, connector_name, staged_record):

    matches = []

    for candidate_id in candidate_ids:

        identity = read_symbol_identity(prj, candidate_id)

        if identity is None:
            continue

        if identity["pin_number"] != staged_record["pin_name"]:
            continue

        score = 100

        if coordinates_close(identity["location"], staged_record["final_x"], staged_record["final_y"]):
            score += 100

        if isinstance(identity["device_text"], str) and connector_name in identity["device_text"]:
            score += 30

        if identity["symbol_type_name"] in ("MilSymbolReceptacle¶END_TOP", "MilSymbolReceptacle¶END_SPLIT", "MilSymbolReceptacle¶SPLIT_BOTTOM", "MilSymbolReceptacle¶CONNECTOR_PIN"):
            score += 10

        matches.append({
            "score": score,
            "identity": identity,
        })

    matches.sort(key=lambda item: item["score"], reverse=True)

    if not matches:
        return None

    if matches[0]["score"] < 230:
        return None

    return matches[0]["identity"]


def apply_text_tweaks(prj, symbol_id, is_top_of_side):

    local_sym = prj.CreateSymbolObject()
    local_text = prj.CreateTextObject()

    set_result = local_sym.SetId(symbol_id)

    if not set_result:
        raise RuntimeError(f"Could not resolve symbol {symbol_id} for formatting.")

    local_sym.SetBlockHatchColour(58)

    for text_id in get_text_ids(local_sym):

        local_text.SetId(text_id)

        try:
            text_type = local_text.GetType()
        except Exception:
            continue

        if text_type == 5:
            local_text.SetVisibility(0)

        elif text_type == 12:

            if not is_top_of_side:
                local_text.SetVisibility(0)

        elif text_type == 617:
            local_text.SetVisibility(1)
            local_text.SetColour(0)
            local_text.SetAlignment(3)
            local_text.SetStyle(1)
            local_text.SetHeight(0.2)

        elif text_type == 203:
            local_text.SetVisibility(1)
            local_text.SetStyle(1)


def resolve_and_format_final_pins(prj, sht, connector_name, staged_records):

    if not staged_records:
        return

    candidate_ids = collect_live_symbol_candidates(prj, sht, staged_records[0]["sheet_id"])

    for staged_record in staged_records:

        identity = resolve_final_symbol(prj, candidate_ids, connector_name, staged_record)

        if identity is None:
            raise RuntimeError(f"Could not rediscover final symbol for {connector_name} pin {staged_record['pin_name']}.")

        is_top_of_side = staged_record["final_row"] == 0

        apply_text_tweaks(prj, identity["symbol_id"], is_top_of_side)

        staged_record["final_resolved_symbol_id"] = identity["symbol_id"]
