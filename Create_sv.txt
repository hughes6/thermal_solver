def make_gui(body):
    root = tk.Tk()
    root.title("Commit changes to database")

    choice = {"value": None}

    def pressed(value):
        choice["value"] = value
        root.quit()

    # Frame with scrollbar
    frame = tk.Frame(root)
    frame.pack(padx=10, pady=10)

    scrollbar = tk.Scrollbar(frame)
    scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

    text = tk.Text(frame, height=20, width=80, wrap="word",
                   yscrollcommand=scrollbar.set)
    text.pack(side=tk.LEFT, fill=tk.BOTH)

    scrollbar.config(command=text.yview)

    text.insert("1.0", body)
    text.config(state="disabled")

    tk.Button(root, text="COMMIT", width=50,
              command=lambda: pressed("YES")).pack(pady=5)
    tk.Button(root, text="ROLLBACK", width=50,
              command=lambda: pressed("NO")).pack(pady=5)

    root.mainloop()
    return choice["value"]

def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def generate_master_schematic_layout(PIN_SIGNAL_INFO):
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9
    BLOCK_X_SPACING = 1.0
    BLOCK_Y_SPACING = 1.0
    PIN_SPACING = 0.25
    PIN_START_OFFSET = 2.0
    # [ [ {name, type, number of pins}, [] ] ]
    # assuming that order is already correct
    # start first block at x_block + block_width + BLOCK_X_SPACING
    SPACING_DICT = []
    sheet = 1
    for i in range(len(PIN_SIGNAL_INFO)):
        connector_name = PIN_SIGNAL_INFO[i][0]['connector name']
        number_of_pins = PIN_SIGNAL_INFO[i][0]['number of pins']
        width = block_width
        height = PIN_START_OFFSET + (PIN_SPACING * number_of_pins) # + spacing for spares
        greatest_sheet = 1
        for dict in SPACING_DICT:
            if dict['sheet'] > greatest_sheet:
                greatest_sheet = dict['sheet']
        greatest_x = 0
        greatest_y = 0
        for dict in SPACING_DICT:
            if dict['x start'] > greatest_x:
                greatest_x = dict['x start']
            if dict['y start'] > greatest_y:
                greatest_y = dict['y start']

        if greatest_sheet + BLOCK_X_SPACING + width > RIGHT_OF_PAGE:
            sheet = greatest_sheet + 1
            x_start = x_block
            y_start = TOP_OF_PAGE - 1.0 - height
        else:
            sheet = greatest_sheet
            if greatest_y + BLOCK_Y_SPACING + height > BOTTOM_OF_PAGE:
                sheet += 1
                x_start = x_block
                y_start = TOP_OF_PAGE - 1.0 - height
            else:
                x_start = greatest_x
                y_start = greatest_y - BLOCK_Y_SPACING - height
        SPACING_DICT.append({'connector name' : connector_name, 'sheet' : sheet, 'y start' : y_start, 'x start' : x_start, 'height' : height, 'width' : width})

    return SPACING_DICT
