Private Sub Worksheet_Change(ByVal Target As Range)

    Const BLOCK_WIDTH As Long = 5
    Const FIRST_COL As Long = 8          'H
    Const MAX_BLOCKS As Long = 40
    Const LAST_ROW As Long = 307

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim maxCreated As Long

    Dim srcHeader As Range
    Dim srcFull As Range
    Dim destHeader As Range
    Dim destFull As Range

    Dim pinCell As Range
    Dim nameCell As Range
    Dim oldNameFormula As String

    'Run ONLY when C4 changes
    If Target.Address <> "$C$4" Then Exit Sub

    On Error GoTo ErrHandler

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    n = CLng(Me.Range("C4").Value)

    If n < 1 Then n = 1
    If n > MAX_BLOCKS Then n = MAX_BLOCKS

    'Original/template block
    Set srcHeader = Me.Range("H5:L10")
    Set srcFull = Me.Range("H5:L" & LAST_ROW)


    '==================================================
    'FIND HOW MANY BLOCKS ALREADY EXIST
    '==================================================

    maxCreated = 1

    For i = 1 To MAX_BLOCKS - 1

        Set destHeader = Me.Range( _
            Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
            Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

        If Application.WorksheetFunction.CountA(destHeader) > 0 Then
            maxCreated = i + 1
        Else
            Exit For
        End If

    Next i


    '==================================================
    'REMOVE OLD BAD NAME FORMULAS FROM NUMBER-OF-PINS
    '
    'Old macro accidentally made:
    'J5 = B7
    'O5 = B8
    'T5 = B9
    '
    'This ONLY clears those exact old formulas.
    'A real number entered by the user is preserved.
    '==================================================

    For i = 0 To maxCreated - 1

        Set pinCell = Me.Cells(5, 10 + i * BLOCK_WIDTH)

        oldNameFormula = "=$B$" & (7 + i)

        If pinCell.Formula = oldNameFormula Then
            pinCell.MergeArea.ClearContents
        End If

    Next i


    '==================================================
    'CREATE ONLY BLOCKS THAT HAVE NEVER EXISTED
    '==================================================

    If n > maxCreated Then

        For i = maxCreated To n - 1

            Set destHeader = Me.Range( _
                Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
                Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

            Set destFull = Me.Range( _
                Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
                Me.Cells(LAST_ROW, FIRST_COL + i * BLOCK_WIDTH + 4))


            '------------------------------------------
            'COPY TEMPLATE BLOCK
            '------------------------------------------

            srcHeader.Copy Destination:=destHeader


            '------------------------------------------
            'CLEAR COPIED NUMBER OF PINS
            '
            'J5 -> O5 -> T5 -> Y5...
            '
            'Each new block starts with its own
            'blank Number of Pins value.
            '------------------------------------------

            Set pinCell = Me.Cells(5, 10 + i * BLOCK_WIDTH)

            pinCell.MergeArea.ClearContents


            '------------------------------------------
            'COPY COLUMN WIDTHS
            '------------------------------------------

            For j = 0 To BLOCK_WIDTH - 1

                Me.Columns(FIRST_COL + i * BLOCK_WIDTH + j).ColumnWidth = _
                    Me.Columns(FIRST_COL + j).ColumnWidth

            Next j


            '------------------------------------------
            'COPY FORMATTING, CONDITIONAL FORMATTING,
            'AND DATA VALIDATION
            '------------------------------------------

            CopyBlockRules _
                srcFull, _
                destFull, _
                FIRST_COL, _
                FIRST_COL + i * BLOCK_WIDTH

        Next i

        maxCreated = n

    End If


    '==================================================
    'CONNECTOR NAMES
    '
    'Block 1: H6 = B7
    'Block 2: M6 = B8
    'Block 3: R6 = B9
    'Block 4: W6 = B10
    '
    'NUMBER OF PINS CELLS ARE NOT TOUCHED.
    '==================================================

    For i = 0 To maxCreated - 1

        Set nameCell = Me.Cells(6, FIRST_COL + i * BLOCK_WIDTH)

        nameCell.MergeArea.Cells(1, 1).Formula = _
            "=$B$" & (7 + i)

    Next i


    '==================================================
    'REFRESH RULES FOR EXISTING BLOCKS
    '
    'This changes FORMATTING/RULES only.
    'It does NOT overwrite user-entered data.
    '==================================================

    For i = 1 To maxCreated - 1

        Set destFull = Me.Range( _
            Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
            Me.Cells(LAST_ROW, FIRST_COL + i * BLOCK_WIDTH + 4))

        CopyBlockRules _
            srcFull, _
            destFull, _
            FIRST_COL, _
            FIRST_COL + i * BLOCK_WIDTH

    Next i


    '==================================================
    'SHOW/HIDE BLOCKS BASED ON C4
    '
    'Nothing gets deleted.
    '
    'Example:
    'C4 = 4
    'then C4 = 2
    'then C4 = 4
    '
    'Blocks 3-4 retain their old data.
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Range( _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH), _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH + 4) _
        ).EntireColumn.Hidden = (i >= n)

    Next i


SafeExit:

    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub


ErrHandler:

    MsgBox "Macro error " & Err.Number & ":" & vbCrLf & _
           Err.Description

    Resume SafeExit

End Sub



Private Sub CopyBlockRules( _
    ByVal srcFull As Range, _
    ByVal destFull As Range, _
    ByVal srcStartCol As Long, _
    ByVal destStartCol As Long)

    Dim fc As FormatCondition
    Dim f As String

    Dim i As Long
    Dim srcCol As String
    Dim destCol As String


    '==================================================
    'DELETE EXISTING CONDITIONAL FORMATTING
    'IN DESTINATION BLOCK
    '==================================================

    destFull.FormatConditions.Delete


    '==================================================
    'COPY FORMATTING + CONDITIONAL FORMATTING
    '
    'Does NOT overwrite values/formulas.
    '==================================================

    srcFull.Copy

    destFull.PasteSpecial Paste:=xlPasteFormats


    '==================================================
    'COPY DATA VALIDATION / DROPDOWNS
    '==================================================

    srcFull.Copy

    On Error Resume Next
    destFull.PasteSpecial Paste:=xlPasteValidation
    On Error GoTo 0

    Application.CutCopyMode = False


    '==================================================
    'TRANSLATE CONDITIONAL FORMATTING REFERENCES
    '
    'Block 1 H:L
    'Block 2 M:Q
    'Block 3 R:V
    'Block 4 W:AA
    '
    'Examples:
    '
    '$H8  -> $M8 -> $R8 -> $W8
    '
    '$J$5 -> $O$5 -> $T$5 -> $Y$5
    '
    'So every block uses its OWN Number of Pins.
    '==================================================

    For Each fc In destFull.FormatConditions

        If fc.Type = xlExpression Then

            f = fc.Formula1

            For i = 0 To 4

                srcCol = ColLetter(srcStartCol + i)
                destCol = ColLetter(destStartCol + i)

                f = Replace( _
                    f, _
                    "$" & srcCol, _
                    "$" & destCol)

            Next i


            If f <> fc.Formula1 Then

                On Error Resume Next

                fc.Modify _
                    Type:=xlExpression, _
                    Formula1:=f

                On Error GoTo 0

            End If

        End If

    Next fc

End Sub



Private Function ColLetter(ByVal colNum As Long) As String

    ColLetter = Split( _
        Me.Cells(1, colNum).Address, _
        "$")(1)

End Function
