def launch_test_after_main():
    script_directory = os.path.dirname(os.path.abspath(__file__))
    test_path = os.path.join(script_directory, "test.py")
    python_path = sys.executable

    print("TEST PATH:", test_path)
    print("PYTHON PATH:", python_path)

    if not os.path.isfile(test_path):
        raise Exception("test.py was not found at: " + test_path)

    command = f'timeout /t 4 /nobreak >nul & "{python_path}" "{test_path}"'

    process = subprocess.Popen(["cmd.exe", "/c", command], cwd=script_directory, creationflags=subprocess.CREATE_NEW_CONSOLE)

    print("LAUNCHER PID:", process.pid)
