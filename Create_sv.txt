def create_gui():
    from tkinter import ttk
    import threading

    root = tk.Tk()
    root.title("CT Symbol Formatter")
    root.geometry("560x430")
    root.resizable(False, False)

    style = ttk.Style()

    try:
        style.theme_use("vista")
    except:
        pass

    style.configure(
        "Title.TLabel",
        font=("Segoe UI", 18, "bold")
    )

    style.configure(
        "Subtitle.TLabel",
        font=("Segoe UI", 10)
    )

    style.configure(
        "Step.TLabel",
        font=("Segoe UI", 10)
    )

    style.configure(
        "Status.TLabel",
        font=("Segoe UI", 10, "bold")
    )

    style.configure(
        "Run.TButton",
        font=("Segoe UI", 11, "bold"),
        padding=(20, 10)
    )

    style.configure(
        "Secondary.TButton",
        font=("Segoe UI", 10),
        padding=(15, 8)
    )

    main_frame = ttk.Frame(
        root,
        padding=25
    )

    main_frame.pack(
        fill="both",
        expand=True
    )

    title = ttk.Label(
        main_frame,
        text="CT Symbol Formatter",
        style="Title.TLabel"
    )

    title.pack(anchor="w")

    subtitle = ttk.Label(
        main_frame,
        text="Configure and format generated connector symbols.",
        style="Subtitle.TLabel"
    )

    subtitle.pack(
        anchor="w",
        pady=(2, 20)
    )

    instruction_frame = ttk.LabelFrame(
        main_frame,
        text="Before Running",
        padding=15
    )

    instruction_frame.pack(
        fill="x",
        pady=(0, 20)
    )

    instructions = (
        "1. Open Device Properties in CT.\n\n"
        "2. Enable:\n"
        "   Assembly  →  Additional Part ID\n\n"
        "3. Return here and click Run Formatter."
    )

    instruction_label = ttk.Label(
        instruction_frame,
        text=instructions,
        justify="left",
        style="Step.TLabel"
    )

    instruction_label.pack(anchor="w")

    status_frame = ttk.LabelFrame(
        main_frame,
        text="Status",
        padding=12
    )

    status_frame.pack(
        fill="x",
        pady=(0, 20)
    )

    status_label = ttk.Label(
        status_frame,
        text="Ready",
        style="Status.TLabel"
    )

    status_label.pack(anchor="w")


    # ========================================================
    # OVERRIDE MESSAGE BOXES
    # ========================================================

    def gui_showinfo(title, message):
        root.after(
            0,
            lambda: status_label.config(text=message)
        )

    def gui_showerror(title, message):
        root.after(
            0,
            lambda: status_label.config(text=message)
        )

    messagebox.showinfo = gui_showinfo
    messagebox.showerror = gui_showerror


    # ========================================================
    # FINISH CALLBACK
    # ========================================================

    def finished():
        status_label.config(
            text="Finished successfully."
        )

        run_button.config(
            state="normal"
        )

        run_again_button.config(
            state="normal"
        )

        root.config(
            cursor=""
        )


    def failed(error):
        status_label.config(
            text=f"Error: {error}"
        )

        run_button.config(
            state="normal"
        )

        run_again_button.config(
            state="normal"
        )

        root.config(
            cursor=""
        )


    # ========================================================
    # BACKGROUND WORKER
    # ========================================================

    def worker():
        try:
            run_ct_code()

            root.after(
                0,
                finished
            )

        except Exception as e:
            root.after(
                0,
                lambda e=e: failed(e)
            )


    # ========================================================
    # RUN BUTTON
    # ========================================================

    def run_formatter():
        status_label.config(
            text="Running..."
        )

        run_button.config(
            state="disabled"
        )

        run_again_button.config(
            state="disabled"
        )

        root.config(
            cursor="watch"
        )

        thread = threading.Thread(
            target=worker,
            daemon=True
        )

        thread.start()


    # ========================================================
    # CLOSE
    # ========================================================

    def close_app():
        root.destroy()


    # ========================================================
    # BUTTONS
    # ========================================================

    button_frame = ttk.Frame(
        main_frame
    )

    button_frame.pack(
        fill="x"
    )

    run_button = ttk.Button(
        button_frame,
        text="Run Formatter",
        style="Run.TButton",
        command=run_formatter
    )

    run_button.pack(
        fill="x"
    )

    secondary_frame = ttk.Frame(
        main_frame
    )

    secondary_frame.pack(
        pady=(12, 0)
    )

    run_again_button = ttk.Button(
        secondary_frame,
        text="Run Again",
        style="Secondary.TButton",
        command=run_formatter
    )

    run_again_button.pack(
        side="left",
        padx=(0, 8)
    )

    close_button = ttk.Button(
        secondary_frame,
        text="Close",
        style="Secondary.TButton",
        command=close_app
    )

    close_button.pack(
        side="left"
    )

    root.protocol(
        "WM_DELETE_WINDOW",
        close_app
    )

    root.mainloop()
