name = "Cable Shield"
ver = 1

# ============================================================
# LOAD
# ============================================================

res1 = sym.Load(name, ver)

print("LOAD RESULT:", res1)

if res1 == 0:
    raise Exception(
        f"Could not load symbol {name}"
    )


# ============================================================
# PLACE
# ============================================================

x = 24.0
y = 28.0
flip = 0

placed_symbol_id = sym.Place(
    sheet_1_id,
    x,
    y,
    flip,
    1.0,
    0
)

print("PLACED SYMBOL ID:", placed_symbol_id)

if placed_symbol_id == 0:
    raise Exception(
        "Symbol placement failed"
    )


# ============================================================
# SET CURRENT SYMBOL TO THE PLACED INSTANCE
# ============================================================

set_result = sym.SetId(
    placed_symbol_id
)

print("SET ID RESULT:", set_result)
print("SYMBOL NAME:", sym.GetName())


# ============================================================
# CHECK ORIGINAL LENGTH
# ============================================================

original_length = sym.GetDisplayLength()

print(
    "ORIGINAL DISPLAY LENGTH:",
    original_length
)


# ============================================================
# CHANGE LENGTH
# ============================================================

new_length = 10.0

length_result = sym.SetDisplayLength(
    new_length
)

print(
    "SET DISPLAY LENGTH RESULT:",
    length_result
)


# ============================================================
# READ IT BACK IMMEDIATELY
# ============================================================

after_length = sym.GetDisplayLength()

print(
    "DISPLAY LENGTH AFTER SET:",
    after_length
)
