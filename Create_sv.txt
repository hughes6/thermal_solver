def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

BLOCK_OFFSET = 4
# output_path = ask_save_path()
output_path = 'C:/Users/conner.hughes/Downloads/Block_Pin_Ordering_Template.xlsm'
wb = load_workbook(output_path, data_only=True)
ws = wb.active

NUMBER_OF_CONNECTORS = ws["C5"].value

CONNECTOR_MODEL_CHECK = ws["B2"].value
if CONNECTOR_MODEL_CHECK != "Pass":
    raise Exception("Connector model check is failing, fix it")
DUPLICATE_PIN_CHECK = ws["C2"].value
if DUPLICATE_PIN_CHECK != "Pass":
    raise Exception("Duplicate pin check is failing, fix it")

CONNECTOR_PIN_INFO = []
for i in range(NUMBER_OF_CONNECTORS):
    left_pin_col = 9 + (i * BLOCK_OFFSET)
    right_pin_col = left_pin_col + 1

    number_of_pins = ws.cell(row=5, column=right_pin_col).value
    connector_name = ws.cell(row=6, column=right_pin_col).value

    if number_of_pins is None:
        number_of_pins = 0
    number_of_pins = int(number_of_pins)
    temp = [{'connector name': connector_name, 'number of pins': number_of_pins, 'left pins': [], 'right pins': []}]
    end_row = 8 + number_of_pins - 1
    left = []
    right = []
    for r in range(8, end_row + 1):
        left_pin = ws.cell(row=r, column=left_pin_col).value
        right_pin = ws.cell(row=r, column=right_pin_col).value
        if left_pin is not None:
            left.append(left_pin)
        if right_pin is not None:
            right.append(right_pin)
    temp[0]['left pins'].extend(left)
    temp[0]['right pins'].extend(right) 
    CONNECTOR_PIN_INFO.append(temp)

wb.close()
