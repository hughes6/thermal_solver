print("\n" + "=" * 80)
print("CONNECTORS + NUMBER OF PINS")
print("=" * 80)

cur.execute("""
SELECT
    c.ENTRY,
    c.VERSION,
    c.Description,
    c.Class,
    COUNT(DISTINCT p.PHYSPIN) AS PinCountFromPinTable,
    COUNT(DISTINCT pp.PinNumber) AS PinCountFromPhysPin
FROM dbo.ComponentData c
LEFT JOIN dbo.Pin p
    ON p.ENTRY = c.ENTRY
LEFT JOIN dbo.PhysPin pp
    ON pp.Entry = p.PHYSPIN
WHERE c.ENTRYTYP = 24
GROUP BY
    c.ENTRY,
    c.VERSION,
    c.Description,
    c.Class
ORDER BY
    c.ENTRY,
    c.VERSION
""")

for entry, version, description, component_class, pin_count, phys_pin_count in cur.fetchall():

    component_class = str(component_class)

    if "SIMS" in component_class or "SCAU" in component_class or "MAU" in component_class or "Coaxial" in component_class:
        print(
            entry,
            "|",
            version,
            "| PIN TABLE:",
            pin_count,
            "| PHYSPIN:",
            phys_pin_count,
            "|",
            component_class,
            "|",
            description
        )
