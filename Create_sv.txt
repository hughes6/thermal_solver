Option Explicit

Private Const CONNECTOR_COUNT_CELL As String = "C5"

Private Const CONNECTOR_NAME_FIRST_ROW As Long = 7

Private Const TEMPLATE_FIRST_COLUMN As Long = 8

' Block layout:
' H = left helper / sequence bar
' I = left pins
' J = right pins
' K = right helper / sequence bar
'
' Repeated with no blank column:
' H:K, L:O, P:S, T:W, ...
Private Const BLOCK_WIDTH As Long = 4
Private Const BLOCK_STEP As Long = 4

Private Const PIN_COUNT_ROW As Long = 5
Private Const PIN_COUNT_COLUMN_OFFSET As Long = 2

Private Const BLOCK_TOP_ROW As Long = 5
Private Const HEADER_BOTTOM_ROW As Long = 7

Private Const DATA_FIRST_ROW As Long = 8
Private Const RULES_LAST_ROW As Long = 307

Private Const MAX_BLOCKS As Long = 40


Private Sub Worksheet_Change(ByVal Target As Range)

    If Intersect(Target, Me.Range(CONNECTOR_COUNT_CELL)) Is Nothing Then Exit Sub

    On Error GoTo CleanFail

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    Dim connectorCount As Long

    If IsNumeric(Me.Range(CONNECTOR_COUNT_CELL).Value) Then
        connectorCount = CLng(Me.Range(CONNECTOR_COUNT_CELL).Value)
    Else
        connectorCount = 0
    End If

    If connectorCount < 0 Then connectorCount = 0
    If connectorCount > MAX_BLOCKS Then connectorCount = MAX_BLOCKS

    RefreshAllActiveBlocks connectorCount
    ShowOnlyActiveBlocks connectorCount

CleanExit:

    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub

CleanFail:

    MsgBox "Block update failed." & vbCrLf & vbCrLf & _
           "Error " & Err.Number & vbCrLf & _
           Err.Description, vbCritical, "Selection Sheet Macro Error"

    Resume CleanExit

End Sub


Private Sub RefreshAllActiveBlocks(ByVal connectorCount As Long)

    Dim blockNumber As Long

    For blockNumber = 1 To connectorCount

        If blockNumber > 1 Then
            RefreshBlockStructure blockNumber
        End If

        UpdateConnectorName blockNumber
        UpdatePinCount blockNumber

        If blockNumber > 1 Then
            CopyEdgeHelpers blockNumber
        End If

    Next blockNumber

End Sub


Private Sub RefreshBlockStructure(ByVal blockNumber As Long)

    Dim destinationStartColumn As Long
    Dim columnOffset As Long

    Dim sourceHeader As Range
    Dim destinationHeader As Range

    Dim sourceData As Range
    Dim destinationData As Range

    destinationStartColumn = GetBlockStartColumn(blockNumber)

    Set sourceHeader = Me.Range(Me.Cells(BLOCK_TOP_ROW, TEMPLATE_FIRST_COLUMN), Me.Cells(HEADER_BOTTOM_ROW, TEMPLATE_FIRST_COLUMN + BLOCK_WIDTH - 1))
    Set destinationHeader = Me.Range(Me.Cells(BLOCK_TOP_ROW, destinationStartColumn), Me.Cells(HEADER_BOTTOM_ROW, destinationStartColumn + BLOCK_WIDTH - 1))

    Set sourceData = Me.Range(Me.Cells(DATA_FIRST_ROW, TEMPLATE_FIRST_COLUMN), Me.Cells(RULES_LAST_ROW, TEMPLATE_FIRST_COLUMN + BLOCK_WIDTH - 1))
    Set destinationData = Me.Range(Me.Cells(DATA_FIRST_ROW, destinationStartColumn), Me.Cells(RULES_LAST_ROW, destinationStartColumn + BLOCK_WIDTH - 1))

    ' Refresh the header from the template.
    sourceHeader.Copy
    destinationHeader.PasteSpecial Paste:=xlPasteAll

    ' Remove stale conditional formatting from old layouts.
    destinationData.FormatConditions.Delete

    ' Copy formatting and conditional-format rules from the template.
    ' This does not overwrite the user's Left Pins / Right Pins values.
    sourceData.Copy
    destinationData.PasteSpecial Paste:=xlPasteFormats

    ' Copy validation separately.
    sourceData.Copy
    destinationData.PasteSpecial Paste:=xlPasteValidation

    ' Any conditional-format rule copied from Block 1 that points to J5
    ' must point to this block's own Number of Pins cell instead.
    FixConditionalFormattingPinCountReferences blockNumber

    ' Match template column widths.
    For columnOffset = 0 To BLOCK_WIDTH - 1
        Me.Columns(destinationStartColumn + columnOffset).ColumnWidth = Me.Columns(TEMPLATE_FIRST_COLUMN + columnOffset).ColumnWidth
    Next columnOffset

    Application.CutCopyMode = False

End Sub


Private Sub CopyEdgeHelpers(ByVal blockNumber As Long)

    Dim destinationStartColumn As Long

    Dim sourceLeftEdge As Range
    Dim destinationLeftEdge As Range

    Dim sourceRightEdge As Range
    Dim destinationRightEdge As Range

    destinationStartColumn = GetBlockStartColumn(blockNumber)

    ' ============================================================
    ' LEFT EDGE
    '
    ' Copy the ENTIRE helper/bar column from H to the equivalent
    ' left edge of the repeated block.
    '
    ' Using Excel Copy/Destination makes Excel translate any
    ' formulas exactly as if the user copied the cells manually.
    ' Constants such as "|" are copied too.
    ' ============================================================

    Set sourceLeftEdge = Me.Range(Me.Cells(DATA_FIRST_ROW, TEMPLATE_FIRST_COLUMN), Me.Cells(RULES_LAST_ROW, TEMPLATE_FIRST_COLUMN))
    Set destinationLeftEdge = Me.Range(Me.Cells(DATA_FIRST_ROW, destinationStartColumn), Me.Cells(RULES_LAST_ROW, destinationStartColumn))

    sourceLeftEdge.Copy Destination:=destinationLeftEdge

    ' Keep the left bracket/bar tight against the pin table.
    destinationLeftEdge.HorizontalAlignment = xlRight

    ' ============================================================
    ' RIGHT EDGE
    '
    ' Copy the ENTIRE helper/bar column from K to the equivalent
    ' right edge of the repeated block.
    '
    ' This copies all bars/brackets down the column.
    ' ============================================================

    Set sourceRightEdge = Me.Range(Me.Cells(DATA_FIRST_ROW, TEMPLATE_FIRST_COLUMN + BLOCK_WIDTH - 1), Me.Cells(RULES_LAST_ROW, TEMPLATE_FIRST_COLUMN + BLOCK_WIDTH - 1))
    Set destinationRightEdge = Me.Range(Me.Cells(DATA_FIRST_ROW, destinationStartColumn + BLOCK_WIDTH - 1), Me.Cells(RULES_LAST_ROW, destinationStartColumn + BLOCK_WIDTH - 1))

    sourceRightEdge.Copy Destination:=destinationRightEdge

    ' Keep the right bracket/bar tight against the pin table.
    destinationRightEdge.HorizontalAlignment = xlLeft

    ' Translate J5 to this block's own pin-count cell.
    FixEdgePinCountReferences blockNumber

    Application.CutCopyMode = False

End Sub


Private Sub FixEdgePinCountReferences(ByVal blockNumber As Long)

    Dim destinationStartColumn As Long

    Dim leftEdge As Range
    Dim rightEdge As Range
    Dim currentCell As Range

    Dim currentFormula As String
    Dim translatedFormula As String

    destinationStartColumn = GetBlockStartColumn(blockNumber)

    Set leftEdge = Me.Range(Me.Cells(DATA_FIRST_ROW, destinationStartColumn), Me.Cells(RULES_LAST_ROW, destinationStartColumn))
    Set rightEdge = Me.Range(Me.Cells(DATA_FIRST_ROW, destinationStartColumn + BLOCK_WIDTH - 1), Me.Cells(RULES_LAST_ROW, destinationStartColumn + BLOCK_WIDTH - 1))

    For Each currentCell In leftEdge.Cells

        If currentCell.HasFormula Then

            currentFormula = currentCell.Formula2
            translatedFormula = TranslatePinCountReference(currentFormula, destinationStartColumn)

            If translatedFormula <> currentFormula Then
                currentCell.Formula2 = translatedFormula
            End If

        End If

    Next currentCell

    For Each currentCell In rightEdge.Cells

        If currentCell.HasFormula Then

            currentFormula = currentCell.Formula2
            translatedFormula = TranslatePinCountReference(currentFormula, destinationStartColumn)

            If translatedFormula <> currentFormula Then
                currentCell.Formula2 = translatedFormula
            End If

        End If

    Next currentCell

End Sub


Private Sub FixConditionalFormattingPinCountReferences(ByVal blockNumber As Long)

    Dim destinationStartColumn As Long
    Dim destinationBlock As Range

    Dim rule As FormatCondition

    Dim oldFormula1 As String
    Dim oldFormula2 As String

    Dim newFormula1 As String
    Dim newFormula2 As String

    destinationStartColumn = GetBlockStartColumn(blockNumber)

    Set destinationBlock = Me.Range(Me.Cells(DATA_FIRST_ROW, destinationStartColumn), Me.Cells(RULES_LAST_ROW, destinationStartColumn + BLOCK_WIDTH - 1))

    For Each rule In destinationBlock.FormatConditions

        oldFormula1 = ""
        oldFormula2 = ""
        newFormula1 = ""
        newFormula2 = ""

        On Error Resume Next
        oldFormula1 = rule.Formula1
        oldFormula2 = rule.Formula2
        On Error GoTo 0

        newFormula1 = TranslatePinCountReference(oldFormula1, destinationStartColumn)
        newFormula2 = TranslatePinCountReference(oldFormula2, destinationStartColumn)

        If rule.Type = xlExpression Then

            If newFormula1 <> oldFormula1 Then
                rule.Modify Type:=xlExpression, Formula1:=newFormula1
            End If

        ElseIf rule.Type = xlCellValue Then

            If newFormula1 <> oldFormula1 Or newFormula2 <> oldFormula2 Then
                rule.Modify Type:=xlCellValue, Operator:=rule.Operator, Formula1:=newFormula1, Formula2:=newFormula2
            End If

        End If

    Next rule

End Sub


Private Function TranslatePinCountReference(ByVal sourceFormula As String, ByVal destinationStartColumn As Long) As String

    Dim translated As String

    Dim templateColumn As String
    Dim destinationColumn As String

    translated = sourceFormula

    If Len(translated) = 0 Then
        TranslatePinCountReference = translated
        Exit Function
    End If

    templateColumn = ColumnLetter(TEMPLATE_FIRST_COLUMN + PIN_COUNT_COLUMN_OFFSET)
    destinationColumn = ColumnLetter(destinationStartColumn + PIN_COUNT_COLUMN_OFFSET)

    ' Most specific forms first.
    translated = Replace(translated, "$" & templateColumn & "$" & PIN_COUNT_ROW, "$" & destinationColumn & "$" & PIN_COUNT_ROW, 1, -1, vbTextCompare)
    translated = Replace(translated, templateColumn & "$" & PIN_COUNT_ROW, destinationColumn & "$" & PIN_COUNT_ROW, 1, -1, vbTextCompare)
    translated = Replace(translated, "$" & templateColumn & PIN_COUNT_ROW, "$" & destinationColumn & PIN_COUNT_ROW, 1, -1, vbTextCompare)
    translated = Replace(translated, templateColumn & PIN_COUNT_ROW, destinationColumn & PIN_COUNT_ROW, 1, -1, vbTextCompare)

    TranslatePinCountReference = translated

End Function


Private Function ColumnLetter(ByVal columnNumber As Long) As String

    ColumnLetter = Split(Me.Cells(1, columnNumber).Address(False, False), "1")(0)

End Function


Private Sub UpdateConnectorName(ByVal blockNumber As Long)

    Dim startColumn As Long
    Dim connectorNameCell As Range
    Dim connectorSourceRow As Long

    startColumn = GetBlockStartColumn(blockNumber)
    connectorSourceRow = CONNECTOR_NAME_FIRST_ROW + blockNumber - 1

    ' J6 for block 1
    ' N6 for block 2
    ' R6 for block 3
    ' V6 for block 4
    Set connectorNameCell = Me.Cells(6, startColumn + 2)

    connectorNameCell.MergeArea.Cells(1, 1).Formula = "=$B$" & connectorSourceRow

End Sub


Private Sub UpdatePinCount(ByVal blockNumber As Long)

    Dim startColumn As Long
    Dim pinCountCell As Range
    Dim connectorSourceRow As Long

    startColumn = GetBlockStartColumn(blockNumber)
    connectorSourceRow = CONNECTOR_NAME_FIRST_ROW + blockNumber - 1

    ' J5 for block 1 uses connector type C7
    ' N5 for block 2 uses connector type C8
    ' R5 for block 3 uses connector type C9
    ' V5 for block 4 uses connector type C10
    '
    ' Library:
    ' A = Connector Data / Connector Type
    ' E = Number of Pins

    Set pinCountCell = Me.Cells(PIN_COUNT_ROW, startColumn + PIN_COUNT_COLUMN_OFFSET)

    pinCountCell.MergeArea.Cells(1, 1).Formula = _
        "=IFERROR(VLOOKUP($C" & connectorSourceRow & ",Library!$A:$E,5,FALSE),"""")"

End Sub


Private Sub ShowOnlyActiveBlocks(ByVal connectorCount As Long)

    Dim blockNumber As Long
    Dim startColumn As Long

    For blockNumber = 1 To MAX_BLOCKS

        startColumn = GetBlockStartColumn(blockNumber)

        Me.Range(Me.Columns(startColumn), Me.Columns(startColumn + BLOCK_WIDTH - 1)).EntireColumn.Hidden = (blockNumber > connectorCount)

    Next blockNumber

End Sub


Private Function GetBlockStartColumn(ByVal blockNumber As Long) As Long

    GetBlockStartColumn = TEMPLATE_FIRST_COLUMN + ((blockNumber - 1) * BLOCK_STEP)

End Function


Public Sub RefreshPinOrderBlocks()

    On Error GoTo CleanFail

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    Dim connectorCount As Long

    If IsNumeric(Me.Range(CONNECTOR_COUNT_CELL).Value) Then
        connectorCount = CLng(Me.Range(CONNECTOR_COUNT_CELL).Value)
    Else
        connectorCount = 0
    End If

    If connectorCount < 0 Then connectorCount = 0
    If connectorCount > MAX_BLOCKS Then connectorCount = MAX_BLOCKS

    RefreshAllActiveBlocks connectorCount
    ShowOnlyActiveBlocks connectorCount

CleanExit:

    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub

CleanFail:

    MsgBox "Block refresh failed." & vbCrLf & vbCrLf & _
           "Error " & Err.Number & vbCrLf & _
           Err.Description, vbCritical, "Selection Sheet Macro Error"

    Resume CleanExit

End Sub
