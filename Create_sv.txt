def launch_test_after_main_exits():
    main_pid = os.getpid()
    script_directory = os.path.dirname(os.path.abspath(__file__))
    test_path = os.path.join(script_directory, "test.py")
    python_path = sys.executable

    def ps_escape(value):
        return value.replace("'", "''")

    python_path_ps = ps_escape(python_path)
    test_path_ps = ps_escape(test_path)
    script_directory_ps = ps_escape(script_directory)

    command = f"$p = Get-Process -Id {main_pid} -ErrorAction SilentlyContinue; if ($p) {{ Wait-Process -Id {main_pid} }}; Start-Sleep -Seconds 2; Start-Process -FilePath '{python_path_ps}' -ArgumentList @('{test_path_ps}') -WorkingDirectory '{script_directory_ps}' -Wait"

    subprocess.Popen(["powershell.exe", "-NoProfile", "-Command", command], creationflags=subprocess.CREATE_NEW_CONSOLE)
