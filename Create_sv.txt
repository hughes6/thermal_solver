def generate_schematic_pin_map(connector_info,newpindict,layout,pin_spacing=0.25,pin_start_offset=2.0,edge_offset=0.3):
    x=layout["x start"]+layout["width"]-edge_offset
    y_start=layout["y start"]+layout["height"]-pin_start_offset
    pin_map=[]
    for pos,pin_data in enumerate(connector_info[1:]):
        pin_name=str(pin_data[0])
        if pin_name not in newpindict: raise Exception(f"Pin {pin_name} not found in new view for {connector_info[0]['connector name']}")
        pin_map.append({"pin name":pin_name,"pin id":newpindict[pin_name],"x":x,"y":y_start-(pos*pin_spacing),"flip":"/1"})
    return pin_map
