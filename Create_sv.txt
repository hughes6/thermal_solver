def make_gui(body):
    root = tk.Tk()
    root.title("Commit changes to database")

    choice = {"value": None}

    def pressed(value):
        choice["value"] = value
        root.quit()

    # Frame with scrollbar
    frame = tk.Frame(root)
    frame.pack(padx=10, pady=10)

    scrollbar = tk.Scrollbar(frame)
    scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

    text = tk.Text(frame, height=20, width=80, wrap="word",
                   yscrollcommand=scrollbar.set)
    text.pack(side=tk.LEFT, fill=tk.BOTH)

    scrollbar.config(command=text.yview)

    text.insert("1.0", body)
    text.config(state="disabled")

    tk.Button(root, text="COMMIT", width=50,
              command=lambda: pressed("YES")).pack(pady=5)
    tk.Button(root, text="ROLLBACK", width=50,
              command=lambda: pressed("NO")).pack(pady=5)

    root.mainloop()
    return choice["value"]

def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def generate_master_schematic_layout(PIN_SIGNAL_INFO,x_block,main_block_width):
    SHEET_DICT = {1:sheet_1_id}

    # PAGE BOUNDS
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    # PAGE MARGINS
    TOP_MARGIN = 1.0
    BOTTOM_MARGIN = 1.0
    LEFT_MARGIN = 1.0
    RIGHT_MARGIN = 1.0

    # BLOCK SPACING
    BLOCK_X_SPACING = 1.0
    BLOCK_Y_SPACING = 1.0
    SCHEMATIC_BLOCK_WIDTH = 3.0
    PIN_SPACING = 0.25
    PIN_START_OFFSET = 2.25

    # USABLE PAGE LIMITS
    PAGE_TOP = TOP_OF_PAGE - TOP_MARGIN
    PAGE_BOTTOM = BOTTOM_OF_PAGE + BOTTOM_MARGIN
    PAGE_LEFT = LEFT_OF_PAGE + LEFT_MARGIN
    PAGE_RIGHT = RIGHT_OF_PAGE - RIGHT_MARGIN
    USABLE_PAGE_HEIGHT = PAGE_TOP - PAGE_BOTTOM

    # OUTPUT
    SPACING_DICT = []
    sheet = 1
    # On sheet 1, start to the RIGHT of the main block
    current_x = (x_block + main_block_width + BLOCK_X_SPACING)

    # Current Y is the TOP edge available for placement
    current_top_y = PAGE_TOP
    # Tracks widest block in current column
    current_column_width = 0.0

    # ================================================
    # BUILD LAYOUT
    # ================================================
    for i in range(len(PIN_SIGNAL_INFO)):
        connector_name = (PIN_SIGNAL_INFO[i][0]["connector name"])
        number_of_pins = (PIN_SIGNAL_INFO[i][0]["number of pins"])
        width = SCHEMATIC_BLOCK_WIDTH
        height = (PIN_START_OFFSET + (PIN_SPACING * number_of_pins))

        # --------------------------------------------
        # SAFETY CHECK:
        # Block cannot physically fit on ANY page
        # --------------------------------------------
        if height > USABLE_PAGE_HEIGHT:
            raise Exception(f"Schematic block for {connector_name} is too tall for a page. Height = {height}, usable page height = {USABLE_PAGE_HEIGHT}")

        # CALCULATE BOTTOM Y IF PLACED HERE
        candidate_y = current_top_y - height

        # DOES IT FIT VERTICALLY?
        if candidate_y < PAGE_BOTTOM:
            next_x = (current_x + current_column_width + BLOCK_X_SPACING)

            # DOES NEXT COLUMN FIT ON THIS PAGE?
            if next_x + width <= PAGE_RIGHT:
                current_x = next_x
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            else:
                # PAGE IS FULL -> NEW PAGE
                sheet += 1
                res = sht.Create(0, sheet, "DINA3", 0)
                if res == 0:
                    app.PutMessage("failed to create new sheet")
                    raise Exception("failed to create new sheet")
                SHEET_DICT[sheet] = res
                current_x = PAGE_LEFT
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            # Recalculate Y at new column/page
            candidate_y = current_top_y - height

        # CHECK HORIZONTAL FIT
        if current_x + width > PAGE_RIGHT:
            sheet += 1
            res = sht.Create(0, sheet, "DINA3", 0)
            if res == 0:
                app.PutMessage("failed to create new sheet")
                raise Exception("failed to create new sheet")
            SHEET_DICT[sheet] = res
            current_x = PAGE_LEFT
            current_top_y = PAGE_TOP
            current_column_width = 0.0
            candidate_y = current_top_y - height

        # SAVE PLACEMENT
        x_start = current_x
        y_start = candidate_y
        SPACING_DICT.append({"connector name": connector_name, "connector index": i, "sheet": sheet, "x start": x_start, "y start": y_start, "x end": x_start + width, "y end": y_start + height, "height": height, "width": width, "number of pins": number_of_pins})
        # UPDATE CURSOR FOR NEXT BLOCK
        current_top_y = (y_start - BLOCK_Y_SPACING)
        current_column_width = max(current_column_width, width)

    return SPACING_DICT, SHEET_DICT



def generate_schematic_pin_map(connector_info,newpindict,layout,pin_spacing=0.25,pin_start_offset=2.0,edge_offset=0.3):
    x=layout["x start"]+layout["width"]-0.3
    y_start=layout["y start"]+layout["height"]-pin_start_offset
    pin_map=[]
    for pos,pin_data in enumerate(connector_info[1:]):
        pin_name=str(pin_data[0])
        if pin_name not in newpindict: raise Exception(f"Pin {pin_name} not found in new view for {connector_info[0]['connector name']}")
        pin_map.append({"pin name":pin_name,"pin id":newpindict[pin_name],"x":x,"y":y_start-(pos*pin_spacing),"flip":"/1"})
    return pin_map

def format_main_block_view(device_view_id, x, y, width, height):
    print("FORMATTING DEVICE VIEW:", device_view_id)

    x_center = x + (width / 2.0)

    dev.SetId(device_view_id)
    syminfo = dev.GetSymbolIds(None)
    symids = [s for s in syminfo[1] if s is not None]

    for sym_id in symids:
        sym.SetId(sym_id)

        sym.SetBlockHatchColour(58)
        sym.SetBlockOutlineColour(0)

        #=========================================
        # NORMAL SYMBOL TEXT
        #=========================================
        textinfo = sym.GetTextIds(None)
        textids = [t for t in textinfo[1] if t is not None]

        for text_id in textids:
            text.SetId(text_id)
            txt = str(text.GetText()).strip()

            print("SYMBOL TEXT:", "VIEW:", device_view_id, "SYM:", sym_id, "TEXT ID:", text_id, "VALUE:", repr(txt))

            if txt == "BxRxUx":
                text.SetColour(0)
                text.SetMode(1)
                text.SetAlignment(2)
                text.SetStyle(1)
                text.SetHeight(0.2)
                text.SetSchemaLocation(x_center, y + height - 0.3)

            elif txt == str(DEVICE_DESIGNATION).strip() or str(DEVICE_DESIGNATION).strip() in txt:
                text.SetColour(0)
                text.SetMode(1)
                text.SetAlignment(2)
                text.SetStyle(1)
                text.SetHeight(0.2)
                text.SetSchemaLocation(x_center, y + height - 0.7)

                print("MOVED DEVICE DESIGNATION:", repr(txt), "TO:", x_center, y + height - 0.7)

            elif DEVICE_FUNCTION is not None and txt == str(DEVICE_FUNCTION).strip():
                text.SetVisibility(0)

        #=========================================
        # ATTRIBUTE TEXT
        #=========================================
        attrinfo = sym.GetAttributeIds(None)
        attrids = [a for a in attrinfo[1] if a is not None]

        for attr_id in attrids:
            attr.SetId(attr_id)
            attr_name = str(attr.GetName()).strip()

            print("ATTRIBUTE:", "VIEW:", device_view_id, "SYM:", sym_id, "ATTR ID:", attr_id, "NAME:", repr(attr_name))

            if attr_name == "Equipment Identifacation":
                attrtextinfo = attr.GetTextIds(None)
                attrtextids = [t for t in attrtextinfo[1] if t is not None]

                print("EQUIPMENT IDENTIFICATION TEXT IDS:", attrtextids)

                for text_id in attrtextids:
                    text.SetId(text_id)
                    txt = str(text.GetText()).strip()

                    print("ATTRIBUTE TEXT:", "ID:", text_id, "VALUE:", repr(txt))

                    text.SetVisibility(1)
                    text.SetColour(0)
                    text.SetMode(1)
                    text.SetAlignment(2)
                    text.SetStyle(1)
                    text.SetHeight(0.2)
                    text.SetSchemaLocation(x_center, y + height - 1.1)

                    print("MOVED EQUIPMENT IDENTIFICATION:", repr(txt), "TO:", x_center, y + height - 1.1)


def format_connector_text(connector_view_id, conn_name):
    print("FORMATTING CONNECTOR TEXT:", conn_name, "VIEW:", connector_view_id)

    dev.SetId(connector_view_id)

    #=========================================
    # SYMBOL TEXT
    #=========================================
    syminfo = dev.GetSymbolIds(None)
    symids = [x for x in syminfo[1] if x is not None]

    for sym_id in symids:
        sym.SetId(sym_id)

        textinfo = sym.GetTextIds(None)
        textids = [x for x in textinfo[1] if x is not None]

        for text_id in textids:
            text.SetId(text_id)

            txt = str(text.GetText()).strip()
            txt_type = text.GetType()

            print("CONNECTOR SYMBOL TEXT:", conn_name, "TEXT:", repr(txt), "TYPE:", txt_type, "ID:", text_id)

            # HIDE COMPONENT CODE
            if txt_type == 5:
                text.SetVisibility(0)

            # BOLD CONNECTOR DEVICE NAME
            if txt == str(conn_name).strip():
                text.SetVisibility(1)
                text.SetColour(0)
                text.SetMode(1)
                text.SetStyle(1)
                text.SetHeight(0.2)

                print("BOLDED CONNECTOR NAME:", conn_name, "TEXT ID:", text_id)

    #=========================================
    # PIN ATTRIBUTE TEXT
    #=========================================
    pininfo = dev.GetAllPinIds(None)
    pinids = [x for x in pininfo[1] if x is not None]

    for pin_id in pinids:
        pin.SetId(pin_id)

        attrinfo = pin.GetAttributeIds(None)
        attrids = [x for x in attrinfo[1] if x is not None]

        for attr_id in attrids:
            attr.SetId(attr_id)

            attr_name = str(attr.GetName()).strip()

            attrtextinfo = attr.GetTextIds(None)
            attrtextids = [x for x in attrtextinfo[1] if x is not None]

            for text_id in attrtextids:
                text.SetId(text_id)

                txt = str(text.GetText()).strip()

                print("PIN ATTRIBUTE TEXT:", conn_name, "PIN:", pin.GetName(), "ATTR:", repr(attr_name), "TEXT:", repr(txt), "ID:", text_id)

                if attr_name == "Local Signal Name":
                    text.SetVisibility(1)
                    text.SetColour(0)
                    text.SetMode(1)
                    text.SetStyle(0)
                    text.SetHeight(0.2)

                    print("SHOWING LOCAL SIGNAL:", conn_name, "PIN:", pin.GetName(), "SIGNAL:", repr(txt))

                else:
                    text.SetVisibility(0)
