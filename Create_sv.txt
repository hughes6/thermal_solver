cur.execute("""
SELECT
    COPY,
    SUBENTRY,
    SUBVERSION
FROM dbo.Assembly
WHERE LPNTR = ?
ORDER BY COPY
""", component_lpntr)

members = cur.fetchall()

for copy_no, part, version in members:
    print(
        f"COPY {copy_no} | "
        f"{part} | "
        f"Version {version}"
    )


cur.execute("""
SELECT *
FROM dbo.GatePinRelation
WHERE ENTRY = ?
ORDER BY GateNumber
""", component_lpntr)

columns = [col[0] for col in cur.description]

for row in cur.fetchall():
    print("\n" + "-" * 60)

    for name, value in zip(columns, row):
        print(f"{name} = {value}")
