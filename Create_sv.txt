connector_spares = SPARES.get(connector, {})

if connector_spares:
    symname = "MilSymbolReceptacle¶END_SPLIT"

    app.PutMessage(
        f"placing {len(connector_spares)} spare pins for connector {connector}"
    )

    for spare_pin_name in connector_spares.keys():

        # IMPORTANT:
        # Use the pin ID from the NEW DEVICE VIEW,
        # not the original connector pin ID.
        spare_pin_id = newpindict.get(str(spare_pin_name))

        if spare_pin_id is None:
            app.PutMessage(
                f"WARNING: Could not find spare pin "
                f"{spare_pin_name} in new view for {connector}"
            )
            continue

        res = sym.PlacePins(
            spare_pin_id,
            symname,
            ver,
            sheet_1_id,
            SPARE_X,
            SPARE_Y,
            0,
            1.0
        )

        app.PutMessage(
            f"placing spare pin: {spare_pin_name} "
            f"id: {spare_pin_id} "
            f"symbol: {symname} "
            f"y: {SPARE_Y} "
            f"x: {SPARE_X} "
            f"rot: 0 "
            f"conn: {connector}"
        )

        if res == 0:
            app.PutMessage(
                f"WARNING: Spare pin placement failed for "
                f"{connector} pin {spare_pin_name} "
                f"id {spare_pin_id}"
            )

        SPARE_Y -= SPARE_PIN_SPACING

        # Same behavior as your normal split-pin stack
        symname = "MilSymbolReceptacle¶SPLIT_BOTTOM"

        time.sleep(0.5)

        app.PutMessage(f"Result: {res}")

    SPARE_Y -= SPARE_CONN_GAP
