def make_gui(body):
    root = tk.Tk()
    root.title("Commit changes to database")

    choice = {"value": None}

    def pressed(value):
        choice["value"] = value
        root.quit()

    # Frame with scrollbar
    frame = tk.Frame(root)
    frame.pack(padx=10, pady=10)

    scrollbar = tk.Scrollbar(frame)
    scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

    text = tk.Text(frame, height=20, width=80, wrap="word",
                   yscrollcommand=scrollbar.set)
    text.pack(side=tk.LEFT, fill=tk.BOTH)

    scrollbar.config(command=text.yview)

    text.insert("1.0", body)
    text.config(state="disabled")

    tk.Button(root, text="COMMIT", width=50,
              command=lambda: pressed("YES")).pack(pady=5)
    tk.Button(root, text="ROLLBACK", width=50,
              command=lambda: pressed("NO")).pack(pady=5)

    root.mainloop()
    return choice["value"]

def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)
