import time
import traceback

import pythoncom
import win32com.client


# ============================================================
# CONFIG
# ============================================================

CONNECTOR_NAME = "SCP-J1"
TARGET_PIN_NAME = "1"

TARGET_X = 5.5
TARGET_Y = 30.25

DETACHED_SYMBOL_NAME = "MilSymbolReceptacle¶END_SPLIT"
DETACHED_SYMBOL_VERSION = "1"

ROTATION = "y"
SCALE = 1.0
MAINTAIN_TEXT_SIZE = 1

COM_MISSING = -2147352572


# ============================================================
# HELPERS
# ============================================================

def is_missing_value(value):

    if value is None:
        return True

    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value > 0:
            cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    if result is None:
        return []

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                count = int(items[0])
            except Exception:
                count = None

            values = clean_id_list(items[1])

            if count is not None and len(values) > count:
                values = values[-count:]

            return values

        return clean_id_list(items)

    return clean_id_list(result)


def call_id_array(method):

    try:
        result = method(None)

        values = parse_count_array_result(result)

        if values:
            return values

    except Exception:
        pass

    try:
        result = method()

        values = parse_count_array_result(result)

        if values:
            return values

    except Exception:
        pass

    return []


# ============================================================
# E3 DISCOVERY
# ============================================================

def find_connector(prj, dev, connector_name):

    for device_id in call_id_array(prj.GetAllDeviceIds):

        dev.SetId(device_id)

        try:
            name = str(dev.GetName())
        except Exception:
            name = ""

        if name == connector_name:
            return device_id

    raise RuntimeError(f"Could not find connector {connector_name}.")


def find_logical_pin(dev, pin, connector_device_id, pin_name):

    dev.SetId(connector_device_id)

    for logical_pin_id in call_id_array(dev.GetPinIds):

        pin.SetId(logical_pin_id)

        try:
            current_name = str(pin.GetName())
        except Exception:
            current_name = ""

        if current_name == str(pin_name):

            return {
                "logical_pin_id": logical_pin_id,
                "view_ids": call_id_array(pin.GetViewIds),
            }

    raise RuntimeError(f"Could not find pin {pin_name}.")


def get_pin_schema_location(pin, view_id):

    pin.SetId(view_id)

    attempts = [
        lambda: pin.GetSchemaLocation(None, None, None, None, None),
        lambda: pin.GetSchemaLocation(None, None, None),
        lambda: pin.GetSchemaLocation(),
    ]

    for attempt in attempts:

        try:
            result = attempt()
        except Exception:
            continue

        if not isinstance(result, (tuple, list)):
            continue

        numeric = []

        for value in result:

            if is_missing_value(value):
                continue

            try:
                numeric.append(float(value))
            except Exception:
                pass

        if len(numeric) >= 3:
            return numeric

    return None


# ============================================================
# PROVEN MOVE OPERATION
# ============================================================

def move_endpoint_pin(prj, dev, pin, connector_device_id, pin_name, target_x, target_y):

    pin_record = find_logical_pin(dev, pin, connector_device_id, pin_name)

    if len(pin_record["view_ids"]) != 1:
        raise RuntimeError(f"Pin {pin_name} must have exactly one view. Found {pin_record['view_ids']}.")

    pin_view_id = pin_record["view_ids"][0]

    location = get_pin_schema_location(pin, pin_view_id)

    if location is None:
        raise RuntimeError(f"Pin {pin_name} does not currently have a live placed view.")

    sheet_id = int(location[0])

    current_symbol = prj.CreateSymbolObject()

    current_symbol_id = current_symbol.SetId(pin_view_id)

    if current_symbol_id <= 0:
        raise RuntimeError(f"Could not resolve live symbol for pin-view ID {pin_view_id}.")

    characteristic = current_symbol.GetCharacteristic()

    if characteristic not in ("END_TOP", "END_BOTTOM", "SPLIT_BOTTOM"):
        raise RuntimeError(f"Pin {pin_name} is not currently an endpoint. Characteristic is {characteristic}.")

    rotation = current_symbol.GetRotation()
    scaling = current_symbol.GetScaling()

    if not isinstance(rotation, str) or rotation == "":
        rotation = ROTATION

    try:
        scaling = float(scaling)
    except Exception:
        scaling = SCALE

    print()
    print(f"PIN NAME: {pin_name}")
    print(f"LOGICAL PIN ID: {pin_record['logical_pin_id']}")
    print(f"STABLE PIN-VIEW ID: {pin_view_id}")
    print(f"CURRENT LIVE SYMBOL ID: {current_symbol_id}")
    print(f"CURRENT CHARACTERISTIC: {characteristic}")
    print(f"CURRENT LOCATION: {location}")

    delete_result = current_symbol.Delete()

    print(f"DELETE RETURN: {delete_result}")

    time.sleep(0.2)

    ghost_pin_record = find_logical_pin(dev, pin, connector_device_id, pin_name)

    if pin_view_id not in ghost_pin_record["view_ids"]:
        raise RuntimeError(f"Stable pin-view ID {pin_view_id} disappeared after delete.")

    ghost_check = prj.CreateSymbolObject()

    ghost_symbol_id = ghost_check.SetId(pin_view_id)

    print(f"GHOST sym.SetId({pin_view_id}) RETURN: {ghost_symbol_id}")

    if ghost_symbol_id != 0:
        raise RuntimeError(f"Pin-view ID {pin_view_id} still resolves to a live symbol after delete.")

    placement = prj.CreateSymbolObject()

    new_symbol_id = placement.PlacePinView(pin_view_id, sheet_id, DETACHED_SYMBOL_NAME, DETACHED_SYMBOL_VERSION, target_x, target_y, rotation, scaling, MAINTAIN_TEXT_SIZE)

    print(f"PlacePinView RETURN / NEW SYMBOL ID: {new_symbol_id}")

    if new_symbol_id <= 0:
        raise RuntimeError("PlacePinView failed.")

    time.sleep(0.2)

    placed_symbol = prj.CreateSymbolObject()

    placed_set_result = placed_symbol.SetId(new_symbol_id)

    print(f"NEW SYMBOL SetId RETURN: {placed_set_result}")

    if placed_set_result <= 0:
        raise RuntimeError(f"Could not resolve new symbol ID {new_symbol_id}.")

    try:
        new_type = placed_symbol.GetSymbolTypeName()
    except Exception:
        new_type = ""

    try:
        new_characteristic = placed_symbol.GetCharacteristic()
    except Exception:
        new_characteristic = ""

    print(f"NEW SYMBOL TYPE: {new_type}")
    print(f"NEW CHARACTERISTIC: {new_characteristic}")

    return {
        "logical_pin_id": pin_record["logical_pin_id"],
        "pin_view_id": pin_view_id,
        "old_symbol_id": current_symbol_id,
        "new_symbol_id": new_symbol_id,
        "sheet_id": sheet_id,
        "target_x": target_x,
        "target_y": target_y,
        "symbol_type": new_type,
        "characteristic": new_characteristic,
    }


# ============================================================
# MAIN
# ============================================================

def main():

    pythoncom.CoInitialize()

    app = win32com.client.Dispatch("CT.Application")

    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()

    connector_device_id = find_connector(prj, dev, CONNECTOR_NAME)

    result = move_endpoint_pin(prj, dev, pin, connector_device_id, TARGET_PIN_NAME, TARGET_X, TARGET_Y)

    print()
    print("=" * 100)
    print("MOVE COMPLETE")
    print("=" * 100)

    for key, value in result.items():
        print(f"{key}: {value}")


if __name__ == "__main__":

    try:
        main()

    except Exception as exc:
        print()
        print("ERROR")
        print(exc)
        print(traceback.format_exc())

    finally:

        try:
            pythoncom.CoUninitialize()
        except Exception:
            pass
