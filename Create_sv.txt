def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsx"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        return None

    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def generate_ordered_pin_ids(dict_to_order, dest):
    return {str(pin): dict_to_order[str(pin)] for pin in dest}

def generate_placement_locations(Pins, layout):
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    PIN_OFFSET_FROM_CONN = 3.0

    X_START = 1.0
    Y_START = 32.5
    main_conn_locs = [] # conn, x, y, flip

    for i, (row) in enumerate(layout):
        for j, (conn) in enumerate(row):
            if conn is not None:
                flip = i > 0
                main_conn_locs.append([conn, X_START + (i * 3.0), Y_START - (j * 1.0), flip])

    y_pin_start = 32.9
    for r in main_conn_locs:
        if y_pin_start > r[2]:
            y_pin_start = r[2]
    y_pin_start -= PIN_OFFSET_FROM_CONN

    print(Pins)
    print(layout)

    pin_locs = [] # conn, xstart, ystart, flip

    for dictionary in Pins:
        # connector number from "W1.Pn"
        conn_key = list(dictionary.keys())[0]
        conn = int(conn_key.split(".")[1][1:])   # P1 -> 1, P2 -> 2, etc.

        # number of pins = all keys except the "W1.Pn"
        count = len(dictionary) - 1

        top = True
        flip = 0
        xstart = 0
        ystart = 0

        # find this connector's placement data
        for r in main_conn_locs:
            if int(r[0]) == conn:
                xstart = r[1]
                ystart = r[2]
                flip = r[3]

        # determine if connector is below another connector
        for r in main_conn_locs:
            if r[3] == flip:       # same flip side (top or bottom)
                if ystart < r[2]:  # this connector is below another
                    top = False

        if top:
            # normal top placement
            ystart = y_pin_start
        else:
            # below another connector:
            # new rule:
            #  p_start = top_p_start - pins_above*0.25 - conns_above*1

            pins_above = 0
            conns_above = 0

            # iterate over all pin dictionaries ABOVE our current connector
            for r in Pins:
                r_conn_key = list(r.keys())[0]
                r_conn = int(r_conn_key.split(".")[1][1:])

                if r_conn == conn:
                    break  # stop when reaching our connector

                # check if the connector is above AND same flip
                for c in main_conn_locs:
                    if int(c[0]) == r_conn and c[3] == flip:
                        if c[2] > ystart:
                            pins_above += (len(r) - 1)
                            conns_above += 1

            ystart = y_pin_start - (pins_above * 0.25) - (conns_above * 0.5)

        pin_locs.append([conn, xstart, ystart, flip])

    return main_conn_locs, pin_locs, y_pin_start + 2

def generate_aligned_pin_y(pinouts, y_start, pin_spacing=0.25, connector_gap=0.5):
    pin_y = {}

    current_y = y_start

    previous_c1 = None
    previous_c2 = None

    for row_index, (c1, p1, c2, p2) in enumerate(pinouts):

        # Add a gap whenever we move into a new connector group
        # on EITHER side of the cable.
        if row_index > 0:
            connector_changed = (
                c1 != previous_c1
                or c2 != previous_c2
            )

            if connector_changed:
                current_y -= connector_gap

        # Both ends of this wire get EXACTLY the same Y.
        pin_y[(int(c1), str(p1))] = current_y
        pin_y[(int(c2), str(p2))] = current_y

        current_y -= pin_spacing

        previous_c1 = c1
        previous_c2 = c2

    return pin_y

def get_spare_pins(pins, pinorder):
    spares = {}
    for pindict in pins:
        connector = list(pindict.keys())[0]
        conn_num = int(connector.split(".P")[1])
        used_pins = {str(pin) for pin in pinorder[conn_num]}
        spares[connector] = {}
        for pin_name, pin_id in pindict.items():
            if pin_name == connector:
                continue
            if str(pin_name) not in used_pins:
                spares[connector][str(pin_name)] = pin_id
