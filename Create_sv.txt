def make_gui(body):
    root = tk.Tk()
    root.title("Commit changes to database")

    choice = {"value": None}

    def pressed(value):
        choice["value"] = value
        root.quit()

    # Frame with scrollbar
    frame = tk.Frame(root)
    frame.pack(padx=10, pady=10)

    scrollbar = tk.Scrollbar(frame)
    scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

    text = tk.Text(frame, height=20, width=80, wrap="word",
                   yscrollcommand=scrollbar.set)
    text.pack(side=tk.LEFT, fill=tk.BOTH)

    scrollbar.config(command=text.yview)

    text.insert("1.0", body)
    text.config(state="disabled")

    tk.Button(root, text="COMMIT", width=50,
              command=lambda: pressed("YES")).pack(pady=5)
    tk.Button(root, text="ROLLBACK", width=50,
              command=lambda: pressed("NO")).pack(pady=5)

    root.mainloop()
    return choice["value"]

def ask_save_path(default_full_path=""):
    import os
    import tkinter as tk
    from tkinter import filedialog

    default_folder = os.path.dirname(default_full_path)

    # Create Tk root
    root = tk.Tk()
    root.withdraw()
    root.attributes("-topmost", True)
    root.update()

    try:
        # Let user PICK an existing Excel file
        selected_path = filedialog.askopenfilename(
            initialdir=default_folder,
            filetypes=[("Excel Files", "*.xlsm"), ("All Files", "*.*")]
        )
    except Exception:
        return None

    # Cancel or X
    if not selected_path:
        root.destroy()
        return None
    root.destroy()
    # Return ONLY the file name (not full path)
    return os.path.abspath(selected_path)

def generate_master_schematic_layout(PIN_SIGNAL_INFO,x_block,main_block_width):
    SHEET_DICT = {1:sheet_1_id}

    # PAGE BOUNDS
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    # PAGE MARGINS
    TOP_MARGIN = 1.0
    BOTTOM_MARGIN = 1.0
    LEFT_MARGIN = 1.0
    RIGHT_MARGIN = 1.0

    # BLOCK SPACING
    BLOCK_X_SPACING = 1.0
    BLOCK_Y_SPACING = 1.0
    SCHEMATIC_BLOCK_WIDTH = 3.0
    PIN_SPACING = 0.25
    PIN_START_OFFSET = 2.25

    # USABLE PAGE LIMITS
    PAGE_TOP = TOP_OF_PAGE - TOP_MARGIN
    PAGE_BOTTOM = BOTTOM_OF_PAGE + BOTTOM_MARGIN
    PAGE_LEFT = LEFT_OF_PAGE + LEFT_MARGIN
    PAGE_RIGHT = RIGHT_OF_PAGE - RIGHT_MARGIN
    USABLE_PAGE_HEIGHT = PAGE_TOP - PAGE_BOTTOM

    # OUTPUT
    SPACING_DICT = []
    sheet = 1
    # On sheet 1, start to the RIGHT of the main block
    current_x = (x_block + main_block_width + BLOCK_X_SPACING)

    # Current Y is the TOP edge available for placement
    current_top_y = PAGE_TOP
    # Tracks widest block in current column
    current_column_width = 0.0

    # ================================================
    # BUILD LAYOUT
    # ================================================
    for i in range(len(PIN_SIGNAL_INFO)):
        connector_name = (PIN_SIGNAL_INFO[i][0]["connector name"])
        number_of_pins = (PIN_SIGNAL_INFO[i][0]["number of pins"])
        width = SCHEMATIC_BLOCK_WIDTH
        height = (PIN_START_OFFSET + (PIN_SPACING * number_of_pins))

        # --------------------------------------------
        # SAFETY CHECK:
        # Block cannot physically fit on ANY page
        # --------------------------------------------
        if height > USABLE_PAGE_HEIGHT:
            raise Exception(f"Schematic block for {connector_name} is too tall for a page. Height = {height}, usable page height = {USABLE_PAGE_HEIGHT}")

        # CALCULATE BOTTOM Y IF PLACED HERE
        candidate_y = current_top_y - height

        # DOES IT FIT VERTICALLY?
        if candidate_y < PAGE_BOTTOM:
            next_x = (current_x + current_column_width + BLOCK_X_SPACING)

            # DOES NEXT COLUMN FIT ON THIS PAGE?
            if next_x + width <= PAGE_RIGHT:
                current_x = next_x
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            else:
                # PAGE IS FULL -> NEW PAGE
                sheet += 1
                res = sht.Create(0, sheet, "DINA3", 0)
                if res == 0:
                    app.PutMessage("failed to create new sheet")
                    raise Exception("failed to create new sheet")
                SHEET_DICT[sheet] = res
                current_x = PAGE_LEFT
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            # Recalculate Y at new column/page
            candidate_y = current_top_y - height

        # CHECK HORIZONTAL FIT
        if current_x + width > PAGE_RIGHT:
            sheet += 1
            res = sht.Create(0, sheet, "DINA3", 0)
            if res == 0:
                app.PutMessage("failed to create new sheet")
                raise Exception("failed to create new sheet")
            SHEET_DICT[sheet] = res
            current_x = PAGE_LEFT
            current_top_y = PAGE_TOP
            current_column_width = 0.0
            candidate_y = current_top_y - height

        # SAVE PLACEMENT
        x_start = current_x
        y_start = candidate_y
        SPACING_DICT.append({"connector name": connector_name, "connector index": i, "sheet": sheet, "x start": x_start, "y start": y_start, "x end": x_start + width, "y end": y_start + height, "height": height, "width": width, "number of pins": number_of_pins})
        # UPDATE CURSOR FOR NEXT BLOCK
        current_top_y = (y_start - BLOCK_Y_SPACING)
        current_column_width = max(current_column_width, width)

    return SPACING_DICT, SHEET_DICT



def generate_schematic_pin_map(connector_info,newpindict,layout,pin_spacing=0.25,pin_start_offset=2.0,edge_offset=0.3):
    x=layout["x start"]+layout["width"]-0.3
    y_start=layout["y start"]+layout["height"]-pin_start_offset
    pin_map=[]
    for pos,pin_data in enumerate(connector_info[1:]):
        pin_name=str(pin_data[0])
        if pin_name not in newpindict: raise Exception(f"Pin {pin_name} not found in new view for {connector_info[0]['connector name']}")
        pin_map.append({"pin name":pin_name,"pin id":newpindict[pin_name],"x":x,"y":y_start-(pos*pin_spacing),"flip":"/1"})
    return pin_map

def format_final_schematic_connectors():
    def extract_ids(result):
        if result is None:
            return []

        if isinstance(result, (tuple, list)):
            if len(result) >= 2 and isinstance(result[1], (tuple, list)):
                return [x for x in result[1] if x is not None and x != 0]

            return [x for x in result if isinstance(x, int) and x != 0]

        if isinstance(result, int):
            if result != 0:
                return [result]

        return []

    def get_pin_view_ids(pin_id):
        pin.SetId(pin_id)

        try:
            result = pin.GetViewIds(None)
            return extract_ids(result)
        except Exception:
            pass

        try:
            result = pin.GetViewIds()
            return extract_ids(result)
        except Exception as e:
            print("GetViewIds ERROR:", pin_id, repr(e))
            return []

    modified_texts = []

    print("")
    print("============================================================")
    print("FINAL SCHEMATIC CONNECTOR FORMATTING")
    print("============================================================")

    device_result = prj.GetAllDeviceIds(None)
    device_ids = extract_ids(device_result)

    for device_id in device_ids:
        dev.SetId(device_id)

        try:
            device_name = str(dev.GetName())
        except Exception:
            continue

        try:
            symbol_result = dev.GetSymbolIds(None)
            logical_symbol_ids = extract_ids(symbol_result)
        except Exception:
            logical_symbol_ids = []

        is_connector = False

        for logical_symbol_id in logical_symbol_ids:
            sym.SetId(logical_symbol_id)

            try:
                text_result = sym.GetTextIds(None)
                text_ids = extract_ids(text_result)
            except Exception:
                text_ids = []

            for text_id in text_ids:
                text.SetId(text_id)

                try:
                    text_type = text.GetType()
                except Exception:
                    continue

                if text_type == 5:
                    is_connector = True
                    break

            if is_connector:
                break

        if not is_connector:
            continue

        print("")
        print("CONNECTOR:", device_name, "DEVICE ID:", device_id)

        try:
            pin_result = dev.GetPinIds(None)
            logical_pin_ids = extract_ids(pin_result)
        except Exception:
            logical_pin_ids = []

        for logical_pin_id in logical_pin_ids:
            pin.SetId(logical_pin_id)

            try:
                pin_name = str(pin.GetName())
            except Exception:
                pin_name = "UNKNOWN"

            pin_view_ids = get_pin_view_ids(logical_pin_id)

            for pin_view_id in pin_view_ids:
                sym.SetId(pin_view_id)

                try:
                    text_result = sym.GetTextIds(None)
                    symbol_text_ids = extract_ids(text_result)
                except Exception:
                    symbol_text_ids = []

                split_position = None

                for text_id in symbol_text_ids:
                    text.SetId(text_id)

                    try:
                        text_type = text.GetType()
                        text_value = str(text.GetText()).strip()
                    except Exception:
                        continue

                    if text_type == 213 and "/" in text_value and not text_value.startswith("*"):
                        try:
                            split_position = int(text_value.split("/")[0])
                        except Exception:
                            split_position = None

                        break

                if split_position is None:
                    continue

                print("PIN:", pin_name, "VIEW:", pin_view_id, "POSITION:", split_position)

                for text_id in symbol_text_ids:
                    text.SetId(text_id)

                    try:
                        text_type = text.GetType()
                        text_value = str(text.GetText()).strip()
                    except Exception:
                        continue

                    if text_type == 5:
                        result = text.SetVisibility(0)
                        print("    COMPONENT CODE OFF:", repr(text_value), "RESULT:", result)
                        modified_texts.append({"text_id": text_id, "desired": 0, "type": 5, "value": text_value})

                    elif text_type == 12:
                        if split_position != 1:
                            result = text.SetVisibility(0)
                            print("    DEVICE DESIGNATION OFF:", repr(text_value), "RESULT:", result)
                            modified_texts.append({"text_id": text_id, "desired": 0, "type": 12, "value": text_value})
                        else:
                            print("    TOP DEVICE DESIGNATION LEFT ALONE:", repr(text_value))

                    elif text_type == 617:
                        text.SetVisibility(1)
                        text.SetColour(0)
                        text.SetAlignment(3)
                        text.SetStyle(1)
                        text.SetHeight(0.2)
                        print("    LOCAL SIGNAL ON:", repr(text_value))
                        modified_texts.append({"text_id": text_id, "desired": 1, "type": 617, "value": text_value})

    print("")
    print("WAITING 5 SECONDS TO VERIFY VISIBILITY")
    time.sleep(5.0)

    passed = 0
    failed = 0

    for item in modified_texts:
        text.SetId(item["text_id"])

        try:
            actual_visibility = text.GetVisibility()
        except Exception as e:
            actual_visibility = "ERROR: " + repr(e)

        if actual_visibility == item["desired"]:
            passed += 1
        else:
            failed += 1
            print("VISIBILITY VERIFY FAIL:", "TYPE:", item["type"], "VALUE:", repr(item["value"]), "EXPECTED:", item["desired"], "ACTUAL:", actual_visibility)

    print("")
    print("============================================================")
    print("FINAL FORMAT VERIFY PASS:", passed)
    print("FINAL FORMAT VERIFY FAIL:", failed)
    print("============================================================")

def wait_for_process_to_exit(process_id):
    import ctypes
    synchronize = 0x00100000
    infinite = 0xFFFFFFFF

    kernel32 = ctypes.windll.kernel32
    process_handle = kernel32.OpenProcess(synchronize, False, process_id)

    if process_handle:
        kernel32.WaitForSingleObject(process_handle, infinite)
        kernel32.CloseHandle(process_handle)
