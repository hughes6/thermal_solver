import win32com.client as win32

e3 = win32.gencache.EnsureDispatch("CT.Application")

dev = e3.CreateDeviceObject()
pin = e3.CreatePinObject()

COMPONENT = "TEMPLATE_CABLE_ASSY"
VERSION = "1"


# ============================================================
# CREATE ASSEMBLY
# ============================================================

assembly_id = dev.Create(
    "W1",
    "",
    "",
    COMPONENT,
    VERSION,
    0
)

print("Assembly ID:", assembly_id)

dev.SetId(assembly_id)

print("Name:", dev.GetName())
print("Component:", dev.GetComponentName())
print("IsAssembly:", dev.IsAssembly())
print("DeviceCount:", dev.GetDeviceCount())


# ============================================================
# GET CHILD DEVICE IDS
# ============================================================

result = dev.GetDeviceIds(None)

print("\nGetDeviceIds FULL RESULT:")
print(result)
print(type(result))


# E3 / pywin32 should return:
#
# (count, (id1, id2, id3, ...))
#
# or something similar

if isinstance(result, tuple):
    count = result[0]
    ids = result[1]

    print("Count:", count)
    print("IDs:", ids)
else:
    raise RuntimeError("Unexpected GetDeviceIds return")


# ============================================================
# NORMALIZE IDS
# ============================================================

if ids is None:
    child_ids = []

elif isinstance(ids, int):
    child_ids = [ids]

else:
    child_ids = list(ids)


print("\nChild IDs:", child_ids)


# ============================================================
# INSPECT CHILD DEVICES
# ============================================================

for child_id in child_ids:

    print("\n" + "=" * 60)

    dev.SetId(child_id)

    print("ID:", child_id)
    print("Name:", dev.GetName())
    print("Component:", dev.GetComponentName())
    print("IsAssemblyPart:", dev.IsAssemblyPart())
    print("PinCount:", dev.GetPinCount())

    # --------------------------------------------
    # GET PINS
    # --------------------------------------------

    pin_result = dev.GetPinIds(None)

    print("GetPinIds result:", pin_result)

    if isinstance(pin_result, tuple):
        pin_count = pin_result[0]
        pin_ids_raw = pin_result[1]

        if pin_ids_raw is None:
            pin_ids = []
        elif isinstance(pin_ids_raw, int):
            pin_ids = [pin_ids_raw]
        else:
            pin_ids = list(pin_ids_raw)

        print("Pin count:", pin_count)
        print("Pin IDs:", pin_ids)

        for pin_id in pin_ids:
            pin.SetId(pin_id)

            print(
                "    Pin ID:",
                pin_id,
                "| Name:",
                pin.GetName()
            )
