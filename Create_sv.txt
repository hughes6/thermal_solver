def run_ct_code():
    try:
        app = win32.gencache.EnsureDispatch('CT.Application')
        prj = app.CreateJobObject()
        text = prj.CreateTextObject()
        dev = prj.CreateDeviceObject()
        sym = prj.CreateSymbolObject()

        # =====================================================
        # LOOP THROUGH EACH NEW DEVICE VIEW
        # =====================================================

        for new_dev_id in new_dev_ids:

            connector = NEW_DEV_CONNECTORS.get(new_dev_id)

            print("\n======================================")
            print("NEW VIEW ID:", new_dev_id)
            print("CONNECTOR:", connector)
            print("======================================")

            if connector is None:
                print(
                    "WARNING: no connector mapping for view",
                    new_dev_id
                )
                continue


            # =================================================
            # FORMAT THE INDIVIDUAL PIN SYMBOLS
            # FROM THE NEW DEVICE VIEW
            # =================================================

            dev.SetId(new_dev_id)

            symbol_info = dev.GetSymbolIds(None)

            symbol_ids = [
                x for x in symbol_info[1]
                if x is not None
            ]

            print("NEW VIEW SYMBOL IDS:", symbol_ids)


            for symbol_id in symbol_ids:

                sym.SetId(symbol_id)

                print(
                    "new view symbol:",
                    symbol_id,
                    sym.GetName()
                )

                texinfo = sym.GetTextIds(None)

                text_ids = [
                    x for x in texinfo[1]
                    if x is not None
                ]


                for t_id in text_ids:

                    text.SetId(t_id)

                    print(
                        "PIN SYMBOL:",
                        sym.GetName(),
                        "| text:",
                        text.GetText(),
                        "| type:",
                        text.GetType()
                    )


                    if (
                        text.GetType() == 5
                        or text.GetType() == 1000
                    ):
                        text.SetVisibility(0)


                    if text.GetType() == 12:
                        text.SetStyle(1)


            # =================================================
            # FORMAT THE ORIGINAL / BULK CONNECTOR SYMBOL
            # =================================================

            connector_id = CONNECTORS.get(connector)

            print(
                "CONNECTOR:",
                connector,
                "ID:",
                connector_id
            )


            if connector_id is None:

                print(
                    "WARNING: connector ID not found for",
                    connector
                )

                continue


            dev.SetId(connector_id)

            print(
                "Formatting bulk connector:",
                dev.GetName()
            )


            assydevsymbolinfo = dev.GetSymbolIds(None)

            assydevsymbolids = [
                x for x in assydevsymbolinfo[1]
                if x is not None
            ]


            print(
                "BULK SYMBOL IDS:",
                assydevsymbolids
            )


            for assydevsymbol_id in assydevsymbolids:

                sym.SetId(assydevsymbol_id)

                print(
                    "BULK SYMBOL:",
                    assydevsymbol_id,
                    "| name:",
                    sym.GetName()
                )


                assydevsymboltextinfo = sym.GetTextIds(None)

                assydevsymboltextids = [
                    x for x in assydevsymboltextinfo[1]
                    if x is not None
                ]


                for assydevsymboltext_id in assydevsymboltextids:

                    text.SetId(assydevsymboltext_id)

                    print(
                        "name:",
                        sym.GetName(),
                        "| text:",
                        text.GetText(),
                        "| type:",
                        text.GetType()
                    )


                    # -----------------------------------------
                    # TYPE 5
                    # -----------------------------------------

                    if text.GetType() == 5:
                        text.SetVisibility(0)


                    # -----------------------------------------
                    # TYPE 12
                    # -----------------------------------------

                    if text.GetType() == 12:
                        text.SetStyle(1)


                    # -----------------------------------------
                    # TYPE 1000 FOR J1
                    # -----------------------------------------

                    if (
                        text.GetType() == 1000
                        and sym.GetName() == "J1"
                    ):

                        app.PutMessage(
                            "Configuring text"
                        )

                        text.SetVisibility(1)
                        text.SetStyle(1)
                        text.SetHeight(0.2)
                        text.SetColour(0)
                        text.SetMode(1)
                        text.SetAlignment(2)


                        xloc = [0.0]
                        yloc = [0.0]
                        grid = [""]
                        column = [""]
                        row = [""]


                        result = text.GetSchemaLocation(
                            xloc,
                            yloc,
                            grid,
                            column,
                            row
                        )


                        new_x = result[1] + 1.5
                        new_y = result[2] + 0.25


                        text.SetSchemaLocation(
                            new_x,
                            new_y
                        )


                    # -----------------------------------------
                    # OTHER TYPE 1000 TEXT
                    # -----------------------------------------

                    elif text.GetType() == 1000:

                        text.SetVisibility(0)


        del dev
        del prj
        del app

        messagebox.showinfo(
            "Success",
            "CT code ran successfully!"
        )


    except Exception as e:

        messagebox.showerror(
            "Error",
            f"An error occurred:\n{e}"
        )
