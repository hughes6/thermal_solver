import inspect
import win32com.client as win32

MAIN_DEVICE_NAME = "TEST_NAME_100000"
CONNECTOR_NAME = "SCP-J1"

HATCH_COLOUR = 58
HATCH_STYLE = 0
HATCH_WIDTH = 0.25
HATCH_DISTANCE = 1.0

PATTERN_CANDIDATES = [1, 2, 3, 4, 5, 6, 7, 8]

COM_MISSING = -2147352572


def is_missing_value(value):
    if value is None:
        return True
    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):
    cleaned = []
    if values is None:
        return cleaned
    if not isinstance(values, (tuple, list)):
        values = [values]
    for value in values:
        if is_missing_value(value):
            continue
        try:
            value = int(value)
        except Exception:
            continue
        if value > 0:
            cleaned.append(value)
    return cleaned


def parse_count_array_result(result):
    if result is None:
        return []
    if isinstance(result, (tuple, list)):
        items = list(result)
        if len(items) >= 2 and isinstance(items[1], (tuple, list)):
            try:
                count = int(items[0])
            except Exception:
                count = None
            values = clean_id_list(items[1])
            if count is not None and len(values) > count:
                values = values[-count:]
            return values
        return clean_id_list(items)
    return clean_id_list(result)


def call_id_array(method):
    try:
        result = method(None)
        values = parse_count_array_result(result)
        if values:
            return values
    except Exception:
        pass
    try:
        result = method()
        values = parse_count_array_result(result)
        if values:
            return values
    except Exception:
        pass
    return []


def find_device_id(prj, dev, target_name):
    for device_id in call_id_array(prj.GetAllDeviceIds):
        dev.SetId(device_id)
        try:
            device_name = str(dev.GetName()).strip()
        except Exception:
            device_name = ""
        if device_name == target_name:
            return device_id
    return None


def get_device_view_ids(dev, device_id):
    dev.SetId(device_id)
    try:
        return parse_count_array_result(dev.GetViewIds(None))
    except Exception:
        return []


def get_device_symbol_ids(dev, device_id):
    dev.SetId(device_id)
    try:
        return parse_count_array_result(dev.GetSymbolIds(None, 0))
    except Exception:
        return []


def get_view_number(dev, view_id):
    dev.SetId(view_id)
    try:
        return int(dev.GetViewNumber())
    except Exception:
        return None


def find_connector_block_symbol(prj, dev, connector_device_id, main_device_id):
    connector_view_ids = get_device_view_ids(dev, connector_device_id)
    if not connector_view_ids:
        raise RuntimeError(f"{CONNECTOR_NAME} has no device views.")

    connector_view_numbers = []
    for connector_view_id in connector_view_ids:
        view_number = get_view_number(dev, connector_view_id)
        if view_number is not None and view_number not in connector_view_numbers:
            connector_view_numbers.append(view_number)

    if len(connector_view_numbers) != 1:
        raise RuntimeError(f"Expected one connector view number, found {connector_view_numbers}.")

    connector_view_number = connector_view_numbers[0]

    matching_main_view_id = None
    for main_view_id in get_device_view_ids(dev, main_device_id):
        if get_view_number(dev, main_view_id) == connector_view_number:
            matching_main_view_id = main_view_id
            break

    if matching_main_view_id is None:
        raise RuntimeError(f"Could not find main-device view number {connector_view_number}.")

    for symbol_id in get_device_symbol_ids(dev, matching_main_view_id):
        local_sym = prj.CreateSymbolObject()
        try:
            set_result = local_sym.SetId(symbol_id)
        except Exception:
            continue

        if not set_result:
            continue

        try:
            symbol_type_name = str(local_sym.GetSymbolTypeName())
        except Exception:
            symbol_type_name = ""

        if "CABBLOCK" in symbol_type_name.upper():
            return {
                "connector_view_number": connector_view_number,
                "main_view_id": matching_main_view_id,
                "block_symbol_id": symbol_id,
                "block_symbol_type": symbol_type_name,
            }

    raise RuntimeError(f"Could not find CABBLOCK for {CONNECTOR_NAME}.")


def print_method_info(obj, method_name):
    print()
    print(method_name)

    method = getattr(obj, method_name, None)

    if method is None:
        print("    NOT AVAILABLE")
        return

    try:
        print("    SIGNATURE:", inspect.signature(method))
    except Exception as error:
        print("    SIGNATURE ERROR:", error)

    try:
        print("    DOC:", method.__doc__)
    except Exception as error:
        print("    DOC ERROR:", error)


def read_hatch_values(sym):
    values = {}

    getters = [
        "GetBlockHatchPattern",
        "GetBlockHatchStyle",
        "GetBlockHatchWidth",
        "GetBlockHatchDistance",
        "GetBlockHatchColour",
    ]

    for method_name in getters:
        method = getattr(sym, method_name, None)

        if method is None:
            values[method_name] = "NOT AVAILABLE"
            continue

        try:
            values[method_name] = method()
        except Exception as error:
            values[method_name] = f"ERROR: {error}"

    return values


def print_hatch_values(title, values):
    print()
    print(title)

    for key, value in values.items():
        print("   ", key, "=", value)


def try_pattern(sym, pattern_value):
    method = sym.SetBlockHatchPattern

    try:
        result = method(pattern_value)
        return {
            "call": f"SetBlockHatchPattern({pattern_value})",
            "result": result,
            "error": None,
        }
    except Exception as one_arg_error:
        pass

    try:
        result = method(pattern_value, 45.0, 135.0)
        return {
            "call": f"SetBlockHatchPattern({pattern_value}, 45.0, 135.0)",
            "result": result,
            "error": None,
        }
    except Exception as three_arg_error:
        return {
            "call": f"SetBlockHatchPattern({pattern_value}) / 3-arg fallback",
            "result": None,
            "error": str(three_arg_error),
        }

    "GetBlockHatchStyle",

    "GetBlockHatchWidth",
    "SetBlockHatchWidth",
    "GetBlockHatchDistance",
    "SetBlockHatchDistance",
    "GetBlockHatchColour",
    "SetBlockHatchColour",
]:
    print_method_info(sym, method_name)

    attempt = try_pattern(sym, candidate)

    print()
    print("PATTERN CANDIDATE:", candidate)
    print("    CALL:", attempt["call"])
    print("    RESULT:", attempt["result"])
    print("    ERROR:", attempt["error"])

    try:
        current_pattern = sym.GetBlockHatchPattern()
    except Exception as error:
        current_pattern = f"ERROR: {error}"

    print("    GET AFTER SET:", current_pattern)

    if current_pattern not in (-1, None) and not isinstance(current_pattern, str):
        selected_pattern = current_pattern
        print("    ACCEPTED PATTERN:", current_pattern)
        break

if selected_pattern is None:
    print()
    print("NO PATTERN CANDIDATE CHANGED GetBlockHatchPattern AWAY FROM -1.")
    print("This strongly suggests the placed CABBLOCK is inheriting its dynamic-symbol hatch settings.")
