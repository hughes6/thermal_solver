COM_MISSING = -2147352572

def log(message=""):

def is_missing_value(value):

    if value is None:
        return True

    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value > 0:
            cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    if result is None:
        return []

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                count = int(items[0])
            except Exception:
                count = None

            values = clean_id_list(items[1])

            if count is not None and len(values) > count:
                values = values[-count:]

            return values

        return clean_id_list(items)

    return clean_id_list(result)


def normalize_name(value):

    if value is None:
        return ""

    return str(value).strip()


def find_device_id(prj, dev, target_name):

    raw = prj.GetAllDeviceIds(None)
    device_ids = parse_count_array_result(raw)

    for device_id in device_ids:

        dev.SetId(device_id)

        try:
            device_name = normalize_name(dev.GetName())
        except Exception:
            device_name = ""

        if device_name == target_name:
            return device_id

    return None


def get_device_view_ids(dev, device_id):

    dev.SetId(device_id)

    try:
        raw = dev.GetViewIds(None)
    except Exception:
        return []

    return parse_count_array_result(raw)


def get_device_symbol_ids(dev, device_id):

    dev.SetId(device_id)

    try:
        raw = dev.GetSymbolIds(None, 0)
    except Exception:
        return []

    return parse_count_array_result(raw)


def get_view_number(dev, view_id):

    dev.SetId(view_id)

    try:
        return int(dev.GetViewNumber())
    except Exception:
        return None


def get_schema_location(sym):

    try:
        result = sym.GetSchemaLocation(None, None, None, None, None)
    except Exception:
        return None

    if not isinstance(result, (tuple, list)):
        return None

    if len(result) < 3:
        return None

    try:
        sheet_id = int(result[0])
        x = float(result[1])
        y = float(result[2])
    except Exception:
        return None

    return {
        "sheet_id": sheet_id,
        "x": x,
        "y": y,
        "raw": result,
    }


def get_area(sym):

    try:
        result = sym.GetArea(None, None, None, None)
    except Exception:
        return None

    if not isinstance(result, (tuple, list)):
        return None

    if len(result) < 5:
        return None

    try:
        success = int(result[0])
        xmin = float(result[1])
        ymin = float(result[2])
        xmax = float(result[3])
        ymax = float(result[4])
    except Exception:
        return None

    if success <= 0:
        return None

    return {
        "xmin": xmin,
        "ymin": ymin,
        "xmax": xmax,
        "ymax": ymax,
        "width": xmax - xmin,
        "height": ymax - ymin,
        "raw": result,
    }


def get_placed_area(sym):

    try:
        result = sym.GetPlacedArea(None, None, None, None)
    except Exception:
        return None

    if not isinstance(result, (tuple, list)):
        return None

    if len(result) < 5:
        return None

    try:
        success = int(result[0])
        xmin = float(result[1])
        ymin = float(result[2])
        xmax = float(result[3])
        ymax = float(result[4])
    except Exception:
        return None

    if success <= 0:
        return None

    return {
        "left": xmin,
        "bottom": ymin,
        "right": xmax,
        "top": ymax,
        "width": xmax - xmin,
        "height": ymax - ymin,
        "center_x": (xmin + xmax) / 2.0,
        "center_y": (ymin + ymax) / 2.0,
        "raw": result,
    }


def get_symbol_info(prj, symbol_id):

    sym = prj.CreateSymbolObject()

    try:
        set_result = sym.SetId(symbol_id)
    except Exception:
        return None

    try:
        set_result = int(set_result)
    except Exception:
        set_result = 0

    if set_result <= 0:
        return None

    try:
        symbol_type_name = normalize_name(sym.GetSymbolTypeName())
    except Exception:
        symbol_type_name = ""

    return {
        "symbol_id": symbol_id,
        "name": normalize_name(sym.GetName()) if hasattr(sym, "GetName") else "",
        "code": sym.GetCode() if hasattr(sym, "GetCode") else None,
        "symbol_type_name": symbol_type_name,
        "schema_location": get_schema_location(sym),
        "area": get_area(sym),
        "placed_area": get_placed_area(sym),
    }

# ================================================

try:

    section("FIND MAIN DEVICE")

    main_device_id = find_device_id(prj, dev, MAIN_DEVICE_NAME)

    if main_device_id is None:
        raise RuntimeError(f"Could not find main device {MAIN_DEVICE_NAME}.")

    log(f"MAIN DEVICE NAME: {MAIN_DEVICE_NAME}")
    log(f"MAIN DEVICE ID: {main_device_id}")


    # ================================================
    # MAP MAIN DEVICE VIEWS
    # ================================================

    section("MAIN DEVICE VIEW MAP")

    main_view_ids = get_device_view_ids(dev, main_device_id)

    log(f"MAIN VIEW IDS: {main_view_ids}")

    main_view_map = {}

    for main_view_id in main_view_ids:

        view_number = get_view_number(dev, main_view_id)

        symbol_ids = get_device_symbol_ids(dev, main_view_id)

        log()
        log(f"MAIN VIEW ID: {main_view_id}")
        log(f"VIEW NUMBER: {view_number}")
        log(f"SYMBOL IDS: {symbol_ids}")

        symbol_records = []

        for symbol_id in symbol_ids:

            symbol_info = get_symbol_info(prj, symbol_id)

            if symbol_info is None:
                continue

            symbol_records.append(symbol_info)

            log(f"    SYMBOL {symbol_id}")
            log(f"        TYPE: {symbol_info['symbol_type_name']}")
            log(f"        SCHEMA LOCATION: {symbol_info['schema_location']}")
            log(f"        AREA: {symbol_info['area']}")
            log(f"        PLACED AREA: {symbol_info['placed_area']}")

        main_view_map[view_number] = {
            "view_id": main_view_id,
            "view_number": view_number,
            "symbols": symbol_records,
        }


    # ================================================
    # MAP EACH CONNECTOR TO ITS BLOCK VIEW
    # ================================================

    section("CONNECTOR -> BLOCK GEOMETRY")

    for connector_name in CONNECTOR_NAMES:

        connector_device_id = find_device_id(prj, dev, connector_name)

        if connector_device_id is None:
            log(f"{connector_name}: DEVICE NOT FOUND")
            continue

        connector_view_ids = get_device_view_ids(dev, connector_device_id)

        log()
        log("-" * 120)
        log(f"CONNECTOR: {connector_name}")
        log(f"DEVICE ID: {connector_device_id}")
        log(f"VIEW IDS: {connector_view_ids}")

        connector_view_numbers = []

        for connector_view_id in connector_view_ids:

            connector_view_number = get_view_number(dev, connector_view_id)

            connector_view_numbers.append(connector_view_number)

            log(f"    CONNECTOR VIEW ID {connector_view_id} -> VIEW NUMBER {connector_view_number}")

        unique_view_numbers = []

        for view_number in connector_view_numbers:

            if view_number is None:
                continue

            if view_number not in unique_view_numbers:
                unique_view_numbers.append(view_number)

        if not unique_view_numbers:
            log("    NO CONNECTOR VIEW NUMBER FOUND")
            continue

        if len(unique_view_numbers) > 1:
            log(f"    WARNING: MULTIPLE CONNECTOR VIEW NUMBERS: {unique_view_numbers}")

        block_view_number = unique_view_numbers[0]

        log(f"    MATCHED MAIN BLOCK VIEW NUMBER: {block_view_number}")

        if block_view_number not in main_view_map:
            log("    NO MATCHING MAIN DEVICE VIEW")
            continue

        main_view_record = main_view_map[block_view_number]

        cabblock_symbols = []

        for symbol_record in main_view_record["symbols"]:

            if "CABBLOCK" in symbol_record["symbol_type_name"].upper():
                cabblock_symbols.append(symbol_record)

        if not cabblock_symbols:
            log("    MATCHING MAIN VIEW HAS NO CABBLOCK SYMBOL")
            continue

        if len(cabblock_symbols) > 1:
            log(f"    WARNING: MATCHING VIEW HAS {len(cabblock_symbols)} CABBLOCK SYMBOLS")

        block = cabblock_symbols[0]

        log(f"    BLOCK SYMBOL ID: {block['symbol_id']}")
        log(f"    BLOCK TYPE: {block['symbol_type_name']}")
        log(f"    BLOCK SCHEMA LOCATION: {block['schema_location']}")
        log(f"    BLOCK LOCAL AREA: {block['area']}")
        log(f"    BLOCK PLACED AREA: {block['placed_area']}")

        placed_area = block["placed_area"]

        if placed_area is not None:

            log()
            log("    FINAL GEOMETRY VALUES:")
            log(f"        block_left = {placed_area['left']}")
            log(f"        block_right = {placed_area['right']}")
            log(f"        block_bottom = {placed_area['bottom']}")
            log(f"        block_top = {placed_area['top']}")
            log(f"        block_width = {placed_area['width']}")
            log(f"        block_height = {placed_area['height']}")
            log(f"        block_center_x = {placed_area['center_x']}")
            log(f"        block_center_y = {placed_area['center_y']}")


    # ================================================
    # TEST COMPLETE
    # ================================================

    section("TEST COMPLETE")

    log("If the connector view numbers match the main-device view numbers,")
    log("this is the mapping we can drop directly into the production pin orderer:")
    log()
    log("connector device")
    log("    -> connector GetViewIds()")
    log("    -> connector view GetViewNumber()")
    log("    -> main device view with same GetViewNumber()")
    log("    -> main view GetSymbolIds()")
    log("    -> CABBLOCK")
    log("    -> GetPlacedArea()")
