import pyodbc

CONNECTION_STRING = (
    "DRIVER={SQL Server};"
    "SERVER=sql-zuken-prd;"
    "DATABASE=Zuken_E3Library;"
    "Trusted_Connection=yes;"
)

conn = pyodbc.connect(CONNECTION_STRING, autocommit=True)
cur = conn.cursor()


# ============================================================
# WIRES
# ============================================================

print("\n" + "=" * 80)
print("WIRES")
print("=" * 80)

cur.execute("""
SELECT ENTRY, VERSION, Description, Class
FROM dbo.ComponentData
WHERE ENTRYTYP = 22
ORDER BY ENTRY, VERSION
""")

for entry, version, description, component_class in cur.fetchall():
    print(f"{entry} | Version {version} | {component_class} | {description}")


# ============================================================
# CONNECTORS
# ============================================================

print("\n" + "=" * 80)
print("CONNECTORS")
print("=" * 80)

cur.execute("""
SELECT ENTRY, VERSION, Description, Class
FROM dbo.ComponentData
WHERE ENTRYTYP = 24
ORDER BY ENTRY, VERSION
""")

for entry, version, description, component_class in cur.fetchall():
    print(f"{entry} | Version {version} | {component_class} | {description}")


cur.close()
conn.close()
