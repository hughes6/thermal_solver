Private Sub Worksheet_Change(ByVal Target As Range)

    Const BLOCK_WIDTH As Long = 5
    Const FIRST_COL As Long = 8          'H
    Const CENTER_COL As Long = 10        'J
    Const MAX_BLOCKS As Long = 40
    Const LAST_ROW As Long = 307

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim maxCreated As Long

    Dim srcHeader As Range
    Dim destHeader As Range

    If Target.Address <> "$C$4" Then Exit Sub

    On Error GoTo ErrHandler

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    n = CLng(Me.Range("C4").Value)

    If n < 1 Then n = 1
    If n > MAX_BLOCKS Then n = MAX_BLOCKS

    Set srcHeader = Me.Range("H5:L10")


    '==================================================
    'REMOVE OLD LEFT-SIDE CONNECTOR NAME FORMULAS
    'H6, M6, R6, W6...
    '==================================================

    For i = 0 To MAX_BLOCKS - 1

        With Me.Cells(6, FIRST_COL + i * BLOCK_WIDTH)

            If .Formula = "=$B$" & (7 + i) Then
                .ClearContents
            End If

        End With

    Next i


    '==================================================
    'FIND HOW MANY BLOCKS ALREADY EXIST
    '==================================================

    maxCreated = 1

    For i = 1 To MAX_BLOCKS - 1

        Set destHeader = Me.Range( _
            Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
            Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

        If Application.WorksheetFunction.CountA(destHeader) > 0 Then
            maxCreated = i + 1
        Else
            Exit For
        End If

    Next i


    '==================================================
    'CREATE ONLY BRAND-NEW BLOCKS
    '==================================================

    If n > maxCreated Then

        For i = maxCreated To n - 1

            Set destHeader = Me.Range( _
                Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
                Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

            'Copy visible block
            srcHeader.Copy Destination:=destHeader


            'Clear copied Number of Pins
            'J5, O5, T5, Y5...
            Me.Cells(5, CENTER_COL + i * BLOCK_WIDTH).MergeArea.ClearContents


            'Copy column widths
            For j = 0 To BLOCK_WIDTH - 1

                Me.Columns(FIRST_COL + i * BLOCK_WIDTH + j).ColumnWidth = _
                    Me.Columns(FIRST_COL + j).ColumnWidth

            Next j


            'Copy dropdowns / validation
            CopyValidation i


            'Create correctly translated CF rules
            CopyConditionalFormatting i

        Next i

        maxCreated = n

    End If


    '==================================================
    'CONNECTOR NAMES
    '
    'J6 = B7
    'O6 = B8
    'T6 = B9
    'Y6 = B10
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Cells(6, CENTER_COL + i * BLOCK_WIDTH) _
            .MergeArea.Cells(1, 1).Formula = _
            "=$B$" & (7 + i)

    Next i


    '==================================================
    'REFRESH CONDITIONAL FORMATTING FOR ALL
    'EXISTING GENERATED BLOCKS
    '==================================================

    For i = 1 To maxCreated - 1

        CopyConditionalFormatting i
        CopyValidation i

    Next i


    '==================================================
    'SHOW / HIDE BLOCKS
    '
    'NO DATA IS DELETED.
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Range( _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH), _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH + 4) _
        ).EntireColumn.Hidden = (i >= n)

    Next i


SafeExit:

    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub


ErrHandler:

    MsgBox "Macro error " & Err.Number & ":" & vbCrLf & _
           Err.Description

    Resume SafeExit

End Sub



Private Sub CopyConditionalFormatting(ByVal blockIndex As Long)

    Const BLOCK_WIDTH As Long = 5
    Const FIRST_COL As Long = 8
    Const LAST_ROW As Long = 307

    Dim sourceBlock As Range
    Dim destBlock As Range

    Dim fc As FormatCondition
    Dim newFC As FormatCondition

    Dim sourceApplies As Range
    Dim destApplies As Range

    Dim formulaNew As String

    Dim colOffset As Long

    colOffset = blockIndex * BLOCK_WIDTH

    Set sourceBlock = Me.Range("H5:L" & LAST_ROW)

    Set destBlock = Me.Range( _
        Me.Cells(5, FIRST_COL + colOffset), _
        Me.Cells(LAST_ROW, FIRST_COL + colOffset + 4))


    'Remove only rules belonging to this generated block
    destBlock.FormatConditions.Delete


    '==================================================
    'LOOP THROUGH EVERY RULE IN BLOCK 1
    '==================================================

    For Each fc In sourceBlock.FormatConditions

        Set sourceApplies = Application.Intersect( _
            fc.AppliesTo, sourceBlock)

        If Not sourceApplies Is Nothing Then

            'Move Applies To range right by block offset
            Set destApplies = sourceApplies.Offset(0, colOffset)


            If fc.Type = xlExpression Then

                'Translate formula exactly as Excel would
                'when copied to the new block
                formulaNew = Application.ConvertFormula( _
                    Formula:=fc.Formula1, _
                    FromReferenceStyle:=xlA1, _
                    ToReferenceStyle:=xlA1, _
                    ToAbsolute:=xlRelative, _
                    RelativeTo:=Me.Cells( _
                        sourceApplies.Row, _
                        sourceApplies.Column))

                formulaNew = Application.ConvertFormula( _
                    Formula:=formulaNew, _
                    FromReferenceStyle:=xlA1, _
                    ToReferenceStyle:=xlA1, _
                    ToAbsolute:=xlAbsolute, _
                    RelativeTo:=Me.Cells( _
                        destApplies.Row, _
                        destApplies.Column))


                Set newFC = destApplies.FormatConditions.Add( _
                    Type:=xlExpression, _
                    Formula1:=formulaNew)


                'Copy formatting from original rule
                CopyCFStyle fc, newFC

            End If

        End If

    Next fc

End Sub



Private Sub CopyCFStyle( _
    ByVal sourceFC As FormatCondition, _
    ByVal destFC As FormatCondition)

    On Error Resume Next

    'Fill
    destFC.Interior.Pattern = sourceFC.Interior.Pattern
    destFC.Interior.Color = sourceFC.Interior.Color

    'Font
    destFC.Font.Color = sourceFC.Font.Color
    destFC.Font.Bold = sourceFC.Font.Bold
    destFC.Font.Italic = sourceFC.Font.Italic
    destFC.Font.Strikethrough = sourceFC.Font.Strikethrough

    'Borders
    destFC.Borders(xlEdgeLeft).LineStyle = _
        sourceFC.Borders(xlEdgeLeft).LineStyle

    destFC.Borders(xlEdgeLeft).Weight = _
        sourceFC.Borders(xlEdgeLeft).Weight


    destFC.Borders(xlEdgeRight).LineStyle = _
        sourceFC.Borders(xlEdgeRight).LineStyle

    destFC.Borders(xlEdgeRight).Weight = _
        sourceFC.Borders(xlEdgeRight).Weight


    destFC.Borders(xlEdgeTop).LineStyle = _
        sourceFC.Borders(xlEdgeTop).LineStyle

    destFC.Borders(xlEdgeTop).Weight = _
        sourceFC.Borders(xlEdgeTop).Weight


    destFC.Borders(xlEdgeBottom).LineStyle = _
        sourceFC.Borders(xlEdgeBottom).LineStyle

    destFC.Borders(xlEdgeBottom).Weight = _
        sourceFC.Borders(xlEdgeBottom).Weight


    destFC.StopIfTrue = sourceFC.StopIfTrue

    On Error GoTo 0

End Sub



Private Sub CopyValidation(ByVal blockIndex As Long)

    Const BLOCK_WIDTH As Long = 5
    Const LAST_ROW As Long = 307

    Dim sourceRange As Range
    Dim destRange As Range

    Dim colOffset As Long

    colOffset = blockIndex * BLOCK_WIDTH

    Set sourceRange = Me.Range("H5:L" & LAST_ROW)
    Set destRange = sourceRange.Offset(0, colOffset)

    sourceRange.Copy

    On Error Resume Next

    destRange.PasteSpecial Paste:=xlPasteValidation

    On Error GoTo 0

    Application.CutCopyMode = False

End Sub
