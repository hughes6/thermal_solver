import win32com.client as win32
import pythoncom

e3 = win32.gencache.EnsureDispatch("CT.Application")

dev = e3.CreateDeviceObject()

COMPONENT = "2627453-500"
VERSION = "1"

assembly_id = dev.Create(
    "W1",
    "",
    "",
    COMPONENT,
    VERSION,
    0
)

print("Assembly ID:", assembly_id)

dev.SetId(assembly_id)

print("IsAssembly:", dev.IsAssembly())
print("DeviceCount:", dev.GetDeviceCount())


# ============================================================
# GET CHILD DEVICE IDS
# ============================================================

device_ids = win32.VARIANT(
    pythoncom.VT_BYREF | pythoncom.VT_VARIANT,
    None
)

count = dev.GetDeviceIds(device_ids)

print("GetDeviceIds count:", count)
print("device_ids.value:", device_ids.value)
print("type:", type(device_ids.value))


# ============================================================
# NORMALIZE RETURN
# ============================================================

raw_ids = device_ids.value

if raw_ids is None:
    raise RuntimeError(
        f"GetDeviceIds returned count={count}, but no IDs were populated"
    )

if isinstance(raw_ids, int):
    child_ids = [raw_ids]
else:
    child_ids = list(raw_ids)

print("Child IDs:", child_ids)


# ============================================================
# INSPECT CHILDREN
# ============================================================

for child_id in child_ids:
    print("\n----------------------------")
    print("Trying child:", child_id)

    if child_id is None:
        continue

    dev.SetId(int(child_id))

    print("ID:", child_id)
    print("Name:", dev.GetName())
    print("Component:", dev.GetComponentName())
    print("Pins:", dev.GetPinCount())
    print("IsAssemblyPart:", dev.IsAssemblyPart())
