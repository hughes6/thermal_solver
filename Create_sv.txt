import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

_, ids = app.GetJobIds()
jid = [x for x in ids if isinstance(x, int) and x > 0][-1]

job = app.CreateDbeJobObject()
job.SetId(jid)

print("JOB:", job.GetId(), job.GetName())

print("Active component:", job.GetActiveComponentId())

print("Components:", job.GetComponentIds())
print("Models:", job.GetModelIds())
print("Symbols:", job.GetSymbolIds())
