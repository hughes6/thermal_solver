import win32com.client as win32


e3 = win32.gencache.EnsureDispatch("CT.Application")

dev = e3.CreateDeviceObject()
pin = e3.CreatePinObject()

COMPONENT = "YOUR_CABLE_ASSEMBLY"
VERSION = "1"
DEVICE_NAME = "W1"


# ============================================================
# CREATE ASSEMBLY
# ============================================================

assembly_id = dev.Create(
    DEVICE_NAME,
    "",
    "",
    COMPONENT,
    VERSION,
    0
)

print("Assembly ID:", assembly_id)

if assembly_id <= 0:
    raise RuntimeError("Failed to create assembly")

dev.SetId(assembly_id)

print("IsAssembly:", dev.IsAssembly())
print("DeviceCount:", dev.GetDeviceCount())


# ============================================================
# GET CHILD DEVICE IDS
# ============================================================

count, raw_ids = dev.GetDeviceIds()

print("Count:", count)
print("Raw IDs:", raw_ids)

if raw_ids is None:
    raise RuntimeError("No child IDs returned")

# E3 arrays are 1-based:
# raw_ids[0] = None
child_ids = [
    child_id
    for child_id in raw_ids[1:]
    if child_id is not None
]

print("\nChild IDs:")
print(child_ids)


# ============================================================
# INSPECT CHILD DEVICES
# ============================================================

for child_id in child_ids:

    dev.SetId(child_id)

    print("\n" + "=" * 60)

    print("ID:", child_id)
    print("Name:", dev.GetName())
    print("Component:", dev.GetComponentName())
    print("IsAssemblyPart:", dev.IsAssemblyPart())
    print("PinCount:", dev.GetPinCount())

    # Only look at pins for actual connector/device children
    if dev.GetPinCount() <= 0:
        continue

    # ========================================================
    # GET PIN IDS
    # ========================================================

    pin_count, raw_pin_ids = dev.GetPinIds()

    print("Raw Pin Count:", pin_count)
    print("Raw Pin IDs:", raw_pin_ids)

    if raw_pin_ids is None:
        continue

    pin_ids = [
        pin_id
        for pin_id in raw_pin_ids[1:]
        if pin_id is not None
    ]

    print("Pin IDs:", pin_ids)

    for pin_id in pin_ids:

        pin.SetId(pin_id)

        print(
            "    Pin ID:", pin_id,
            "| Name:", pin.GetName()
        )
