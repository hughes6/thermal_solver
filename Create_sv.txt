    (
        wire_part,
        wire_version,
        connector1,
        pin1,
        connector2,
        pin2,
        wire_copy,
        no_wires,
        conductor_no
    ) = wire_data

    # Add physical wire/cable only once
    if wire_copy not in added_wire_copies:

        app.PutMessage(
            f"Adding wire COPY {wire_copy}: "
            f"{wire_part} ({no_wires} conductor(s))"
        )

        add_member(wire_copy, wire_part, wire_version)

        added_wire_copies.add(wire_copy)

    # Get connector part numbers
    connector1_part = CONNECTORS[connector1]
    connector2_part = CONNECTORS[connector2]

    # Convert physical pins to Zuken pin COPY numbers
    pin1_copy = get_pin_copy(
        connector1_part,
        1,
        pin1
    )

    pin2_copy = get_pin_copy(
        connector2_part,
        1,
        pin2
    )

    # Determine internal conductor node
    if no_wires == 1:
        node_no = 4
    else:
        node_no = 4 + (conductor_no - 1)

    app.PutMessage(
        f"Wire COPY {wire_copy}, "
        f"Conductor {conductor_no}: "
        f"{wire_part} | "
        f"Connector {connector1} Pin {pin1} -> "
        f"Connector {connector2} Pin {pin2} | "
        f"Node {node_no}"
    )

    connect_wire(
        wire_copy,
        node_no,
        connector1,
        pin1_copy,
        connector2,
        pin2_copy
    )
