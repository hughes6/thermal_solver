TEST_ID = "98BC1321-47D5-455C-9620-AF43FB05B2F1"

print("\n" + "=" * 100)
print("PIN ROWS USING COMPONENTDATA.ID")
print("=" * 100)

cur.execute("""
SELECT
    ENTRY,
    COPY,
    PHYSPIN,
    PINNAME
FROM dbo.Pin
WHERE ENTRY = ?
ORDER BY PINNAME
""", TEST_ID)

rows = cur.fetchall()

for row in rows:
    print(row)

print()
print("PIN ROW COUNT:", len(rows))





print("\n" + "=" * 100)
print("PHYSICAL PIN JOIN")
print("=" * 100)

cur.execute("""
SELECT
    p.PINNAME,
    p.PHYSPIN,
    pp.PinNumber
FROM dbo.Pin p
LEFT JOIN dbo.PhysPin pp
    ON pp.Entry = p.PHYSPIN
WHERE p.ENTRY = ?
ORDER BY pp.PinNumber
""", TEST_ID)

rows = cur.fetchall()

for row in rows:
    print(row)

print()
print("JOIN ROW COUNT:", len(rows))
