def is_missing_value(value):

    if value is None:
        return True

    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value > 0:
            cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    if result is None:
        return []

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                count = int(items[0])
            except Exception:
                count = None

            values = clean_id_list(items[1])

            if count is not None and len(values) > count:
                values = values[-count:]

            return values

        return clean_id_list(items)

    return clean_id_list(result)


def get_ids(method, *args):

    try:
        raw = method(*args)
    except Exception as exc:
        log(f"ERROR: {exc}")
        return []

    log(f"RAW: {raw}")

    ids = parse_count_array_result(raw)

    log(f"IDS: {ids}")

    return ids


def normalize_name(value):

    if value is None:
        return ""

    return str(value).strip()


def get_schema_location(sym):

    try:
        return sym.GetSchemaLocation(None, None, None, None, None)
    except Exception as exc:
        return f"ERROR: {exc}"


def get_area(sym):

    try:
        return sym.GetArea(None, None, None, None)
    except Exception as exc:
        return f"ERROR: {exc}"


def get_placed_area(sym):

    try:
        return sym.GetPlacedArea(None, None, None, None)
    except Exception as exc:
        return f"ERROR: {exc}"


def get_placed_polygon(sym):

    try:
        return sym.GetPlacedPolygon(None, None, None)
    except Exception as exc:
        return f"ERROR: {exc}"


def print_symbol_info(sym, symbol_id):

    section(f"SYMBOL {symbol_id}")

    try:
        set_result = sym.SetId(symbol_id)
    except Exception as exc:
        log(f"sym.SetId ERROR: {exc}")
        return

    log(f"sym.SetId({symbol_id}) RETURN: {set_result}")

    if not set_result:
        return

    for method_name in (
        "GetId",
        "GetName",
        "GetCode",
        "GetSymbolTypeName",
        "GetCharacteristic",
        "GetRotation",
        "GetScaling",
        "GetDynamicParentId",
        "IsDynamic",
        "IsDynamicFixed",
        "IsPinView",
    ):

        if not hasattr(sym, method_name):
            continue

        try:
            result = getattr(sym, method_name)()
            log(f"{method_name}(): {result}")
        except Exception as exc:
            log(f"{method_name} ERROR: {exc}")

    log(f"GetSchemaLocation: {get_schema_location(sym)}")
    log(f"GetArea: {get_area(sym)}")
    log(f"GetPlacedArea: {get_placed_area(sym)}")
    log(f"GetPlacedPolygon: {get_placed_polygon(sym)}")

    if hasattr(sym, "GetDynamicChildrenIds"):

        try:
            raw = sym.GetDynamicChildrenIds(None)
            ids = parse_count_array_result(raw)

            log(f"GetDynamicChildrenIds RAW: {raw}")
            log(f"GetDynamicChildrenIds IDS: {ids}")

        except Exception as exc:
            log(f"GetDynamicChildrenIds ERROR: {exc}")
