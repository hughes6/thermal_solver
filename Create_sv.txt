import tkinter as tk
from tkinter import scrolledtext


def convert():
    # Get text from input box
    input_text = input_box.get("1.0", tk.END)

    # Split on any whitespace: spaces, tabs, newlines, etc.
    values = input_text.split()
    values = [x for x in values if x.isdigit()]

    # Put each value on its own line
    output_text = "\n".join(values)

    # Clear old output
    output_box.delete("1.0", tk.END)

    # Insert converted output
    output_box.insert(tk.END, output_text)

    # Update count label
    count_label.config(text=f"Items: {len(values)}")


def copy_output():
    output_text = output_box.get("1.0", tk.END).strip()

    root.clipboard_clear()
    root.clipboard_append(output_text)
    root.update()


def clear_all():
    input_box.delete("1.0", tk.END)
    output_box.delete("1.0", tk.END)


# =========================================
# GUI
# =========================================

root = tk.Tk()
root.title("Excel Column Formatter")
root.geometry("600x560")


# Input label
tk.Label(
    root,
    text="Paste numbers here:",
    font=("Arial", 12, "bold")
).pack(pady=(15, 5))


# Small input window
input_box = scrolledtext.ScrolledText(
    root,
    width=65,
    height=6,
    font=("Consolas", 11)
)
input_box.pack(padx=15)


# Convert button
tk.Button(
    root,
    text="Convert",
    command=convert,
    width=20,
    height=2
).pack(pady=10)


# Output label
tk.Label(
    root,
    text="Excel-ready output:",
    font=("Arial", 12, "bold")
).pack(pady=(5, 5))


# Large output window
output_box = scrolledtext.ScrolledText(
    root,
    width=65,
    height=15,
    font=("Consolas", 11)
)
output_box.pack(padx=15)


# Count label (NEW)
count_label = tk.Label(root, text="Items: 0", font=("Arial", 12, "bold"), fg="blue")
count_label.pack(pady=(5, 10))


# Buttons
button_frame = tk.Frame(root)
button_frame.pack(pady=12)

tk.Button(
    button_frame,
    text="Copy to Clipboard",
    command=copy_output,
    width=20
).pack(side=tk.LEFT, padx=5)

tk.Button(
    button_frame,
    text="Clear",
    command=clear_all,
    width=12
).pack(side=tk.LEFT, padx=5)


root.mainloop()
