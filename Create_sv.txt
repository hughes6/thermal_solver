Option Explicit

Private Const BLOCK_WIDTH As Long = 5
Private Const FIRST_COL As Long = 8          'H
Private Const CENTER_COL As Long = 10        'J
Private Const MAX_BLOCKS As Long = 40
Private Const FIRST_DATA_ROW As Long = 8
Private Const LAST_ROW As Long = 307


Private Sub Worksheet_Change(ByVal Target As Range)

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim maxCreated As Long

    Dim srcHeader As Range
    Dim destHeader As Range
    Dim destInput As Range

    'Only run when C4 changes
    If Intersect(Target, Me.Range("C4")) Is Nothing Then Exit Sub

    On Error GoTo ErrHandler

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    '-----------------------------------------
    'GET NUMBER OF CONNECTORS / BLOCKS
    '-----------------------------------------
    If IsNumeric(Me.Range("C4").Value) Then
        n = CLng(Me.Range("C4").Value)
    Else
        n = 1
    End If

    If n < 1 Then n = 1
    If n > MAX_BLOCKS Then n = MAX_BLOCKS

    Set srcHeader = Me.Range("H5:L10")


    '==================================================
    'FIND HOW MANY BLOCKS HAVE ALREADY BEEN CREATED
    '
    'We use the connector-name cell:
    '
    'J6, O6, T6, Y6...
    '==================================================

    maxCreated = 1

    For i = 1 To MAX_BLOCKS - 1

        If Len(Me.Cells(6, CENTER_COL + i * BLOCK_WIDTH).Formula) > 0 _
           Or Len(Me.Cells(6, CENTER_COL + i * BLOCK_WIDTH).Value) > 0 Then

            maxCreated = i + 1

        Else

            Exit For

        End If

    Next i


    '==================================================
    'CREATE ONLY BLOCKS THAT HAVE NEVER EXISTED
    '
    'Existing blocks are NOT recopied.
    'This preserves entered data.
    '==================================================

    If n > maxCreated Then

        For i = maxCreated To n - 1

            Set destHeader = Me.Range( _
                Me.Cells(5, FIRST_COL + i * BLOCK_WIDTH), _
                Me.Cells(10, FIRST_COL + i * BLOCK_WIDTH + 4))

            '------------------------------------------
            'COPY ORIGINAL H5:L10
            '------------------------------------------
            srcHeader.Copy Destination:=destHeader


            '------------------------------------------
            'CLEAR COPIED NUMBER OF PINS
            '
            'J5, O5, T5, Y5...
            '------------------------------------------
            Me.Cells(5, CENTER_COL + i * BLOCK_WIDTH) _
                .MergeArea.ClearContents


            '------------------------------------------
            'CLEAR USER INPUT CELLS IN NEW BLOCK
            '
            'Block 1 = I:K
            'Block 2 = N:P
            'Block 3 = S:U
            'etc.
            '------------------------------------------
            Set destInput = Me.Range( _
                Me.Cells(FIRST_DATA_ROW, FIRST_COL + i * BLOCK_WIDTH + 1), _
                Me.Cells(LAST_ROW, FIRST_COL + i * BLOCK_WIDTH + 3))

            destInput.ClearContents


            '------------------------------------------
            'COPY COLUMN WIDTHS
            '------------------------------------------
            For j = 0 To BLOCK_WIDTH - 1

                Me.Columns(FIRST_COL + i * BLOCK_WIDTH + j).ColumnWidth = _
                    Me.Columns(FIRST_COL + j).ColumnWidth

            Next j


            '------------------------------------------
            'COPY FORMATTING / RULES / VALIDATION
            '------------------------------------------
            RefreshBlockFormatting i

        Next i

        maxCreated = n

    End If


    '==================================================
    'SET CONNECTOR NAMES
    '
    'Block 1:
    'J6 = B7
    '
    'Block 2:
    'O6 = B8
    '
    'Block 3:
    'T6 = B9
    '
    'etc.
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Cells(6, CENTER_COL + i * BLOCK_WIDTH) _
            .MergeArea.Cells(1, 1).Formula = _
            "=$B$" & (7 + i)

    Next i


    '==================================================
    'BUILD THE | HELPER SPILL FOR EVERY BLOCK
    '
    'This is the important off-by-one fix.
    '
    'Block 1:
    'H8 spills exactly J5 rows
    '
    'Block 2:
    'M8 spills exactly O5 rows
    '
    'Block 3:
    'R8 spills exactly T5 rows
    '
    'Examples:
    '
    'J5 = 4 -> H8:H11 contains |
    'O5 = 6 -> M8:M13 contains |
    '==================================================

    For i = 0 To maxCreated - 1

        SetupHelperSpill i

    Next i


    '==================================================
    'REFRESH CONDITIONAL FORMATTING FOR GENERATED
    'BLOCKS
    '
    'Does not overwrite entered values.
    '==================================================

    For i = 1 To maxCreated - 1

        RefreshBlockFormatting i

    Next i


    '==================================================
    'SHOW / HIDE BLOCKS BASED ON C4
    '
    'Blocks are hidden, NEVER deleted.
    '
    'So:
    '
    'C4 = 4
    'enter data
    '
    'C4 = 2
    'blocks 3-4 hide
    '
    'C4 = 4
    'blocks 3-4 return with their old data
    '==================================================

    For i = 0 To maxCreated - 1

        Me.Range( _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH), _
            Me.Columns(FIRST_COL + i * BLOCK_WIDTH + 4) _
        ).EntireColumn.Hidden = (i >= n)

    Next i


SafeExit:

    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True

    Exit Sub


ErrHandler:

    MsgBox "Macro error " & Err.Number & ":" & vbCrLf & _
           Err.Description

    Resume SafeExit

End Sub



Private Sub SetupHelperSpill(ByVal blockIndex As Long)

    Dim helperCol As Long
    Dim pinCol As Long

    Dim helperRange As Range
    Dim helperCell As Range

    Dim pinAddress As String
    Dim spillFormula As String

    '-----------------------------------------
    'BLOCK 1
    'helper = H
    'pin count = J
    '
    'BLOCK 2
    'helper = M
    'pin count = O
    '
    'BLOCK 3
    'helper = R
    'pin count = T
    '-----------------------------------------

    helperCol = FIRST_COL + blockIndex * BLOCK_WIDTH
    pinCol = CENTER_COL + blockIndex * BLOCK_WIDTH

    Set helperRange = Me.Range( _
        Me.Cells(FIRST_DATA_ROW, helperCol), _
        Me.Cells(LAST_ROW, helperCol))

    Set helperCell = Me.Cells(FIRST_DATA_ROW, helperCol)

    'Clear any old helper formulas / values first
    helperRange.ClearContents

    'Examples:
    'J$5
    'O$5
    'T$5
    pinAddress = Me.Cells(5, pinCol).Address( _
                    RowAbsolute:=True, _
                    ColumnAbsolute:=False)

    '-----------------------------------------
    'ONE FORMULA.
    '
    'If Number of Pins = 4:
    '
    '|   row 8
    '|   row 9
    '|   row 10
    '|   row 11
    '
    'Exactly four rows.
    '-----------------------------------------

    spillFormula = _
        "=IF(" & pinAddress & ">0," & _
        "IF(SEQUENCE(" & pinAddress & "),""|""),"""")"

    helperCell.Formula2 = spillFormula

End Sub



Private Sub RefreshBlockFormatting(ByVal blockIndex As Long)

    Dim srcRange As Range
    Dim destRange As Range

    Dim fc As FormatCondition
    Dim f As String

    Dim colOffset As Long

    colOffset = blockIndex * BLOCK_WIDTH

    Set srcRange = Me.Range("H5:L" & LAST_ROW)

    Set destRange = Me.Range( _
        Me.Cells(5, FIRST_COL + colOffset), _
        Me.Cells(LAST_ROW, FIRST_COL + colOffset + 4))


    '==================================================
    'REMOVE EXISTING CONDITIONAL FORMATTING
    'FROM THIS GENERATED BLOCK
    '==================================================

    destRange.FormatConditions.Delete


    '==================================================
    'COPY FORMATTING FROM ORIGINAL BLOCK
    '
    'Values are NOT copied.
    '==================================================

    srcRange.Copy

    destRange.PasteSpecial Paste:=xlPasteFormats

    Application.CutCopyMode = False


    '==================================================
    'TRANSLATE CONDITIONAL-FORMATTING FORMULAS
    '
    'Block 1:
    '$H8
    '$J$5
    '
    'Block 2:
    '$M8
    '$O$5
    '
    'Block 3:
    '$R8
    '$T$5
    '
    'etc.
    '==================================================

    For Each fc In destRange.FormatConditions

        If fc.Type = xlExpression Then

            f = TranslateBlockFormula( _
                    fc.Formula1, _
                    FIRST_COL, _
                    FIRST_COL + colOffset)

            If f <> fc.Formula1 Then

                On Error Resume Next

                fc.Modify _
                    Type:=xlExpression, _
                    Formula1:=f

                On Error GoTo 0

            End If

        End If

    Next fc


    '==================================================
    'COPY DATA VALIDATION / DROPDOWNS
    '==================================================

    srcRange.Copy

    On Error Resume Next

    destRange.PasteSpecial Paste:=xlPasteValidation

    On Error GoTo 0

    Application.CutCopyMode = False

End Sub



Private Function TranslateBlockFormula( _
    ByVal formulaText As String, _
    ByVal sourceStartCol As Long, _
    ByVal destStartCol As Long) As String

    Dim i As Long

    Dim f As String

    Dim sourceLetter As String
    Dim destLetter As String

    f = formulaText


    '==================================================
    'FIRST PASS
    '
    'Turn source block columns into temporary markers.
    '
    '$H -> marker 0
    '$I -> marker 1
    '$J -> marker 2
    '$K -> marker 3
    '$L -> marker 4
    '
    'This prevents replacements interfering with
    'each other.
    '==================================================

    For i = 0 To BLOCK_WIDTH - 1

        sourceLetter = ColLetter(sourceStartCol + i)

        f = Replace( _
                f, _
                "$" & sourceLetter, _
                "{BLOCKCOL" & CStr(i) & "}")

    Next i


    '==================================================
    'SECOND PASS
    '
    'Replace markers with destination columns.
    '==================================================

    For i = 0 To BLOCK_WIDTH - 1

        destLetter = ColLetter(destStartCol + i)

        f = Replace( _
                f, _
                "{BLOCKCOL" & CStr(i) & "}", _
                "$" & destLetter)

    Next i

    TranslateBlockFormula = f

End Function



Private Function ColLetter(ByVal colNum As Long) As String

    ColLetter = Split( _
        Me.Cells(1, colNum).Address, _
        "$")(1)

End Function
