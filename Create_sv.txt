# =========================
for data in SCHEMATIC_DATA:
    layout=data["layout"]

    # FORMAT THE EXACT BLOCK SYMBOL RETURNED BY PLACEBLOCK
    sym.SetId(data["block_symbol_id"])
    print("FORMATTING BLOCK:",data["name"],"SYMBOL ID:",data["block_symbol_id"],"SYMBOL NAME:",sym.GetName())

    res=sym.SetBlockHatchColour(58)
    if res==-1: app.PutMessage(f"Failed hatch color for {data['name']} block symbol {data['block_symbol_id']}")

    res=sym.SetBlockOutlineColour(0)
    if res==-1: app.PutMessage(f"Failed outline color for {data['name']} block symbol {data['block_symbol_id']}")

    # THEN DO YOUR EXISTING TEXT FORMATTING
    format_main_block_view(data["parent_block"],layout["x start"],layout["y start"],layout["width"],layout["height"])
