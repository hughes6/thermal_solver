Private Sub Worksheet_Change(ByVal Target As Range)

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim k As Long
    Dim c As Long

    Dim src As Range
    Dim dest As Range
    Dim sfc As FormatCondition
    Dim dfc As FormatCondition
    Dim f As String

    If Target.Address <> "$C$4" Then Exit Sub

    On Error GoTo SafeExit
    Application.EnableEvents = False

    n = CLng(Me.Range("C4").Value)
    Set src = Me.Range("H5:L10")

    Me.Range("M5:ZZ10").Clear

    For i = 1 To n - 1

        Set dest = src.Offset(0, i * 5)

        'Copy block
        src.Copy
        dest.PasteSpecial Paste:=xlPasteAll

        'Copy column widths
        For j = 1 To 5
            dest.Columns(j).ColumnWidth = src.Columns(j).ColumnWidth
        Next j

        'Translate conditional-formatting formulas
        For k = 1 To src.FormatConditions.Count

            If k <= dest.FormatConditions.Count Then

                Set sfc = src.FormatConditions(k)
                Set dfc = dest.FormatConditions(k)

                If sfc.Type = xlExpression Then

                    f = sfc.Formula1

                    For c = 1 To 5
                        f = Replace(f, _
                            "$" & ColLetter(src.Column + c - 1), _
                            "$" & ColLetter(dest.Column + c - 1))
                    Next c

                    dfc.Modify Type:=xlExpression, Formula1:=f

                End If

            End If

        Next k

    Next i

SafeExit:
    Application.CutCopyMode = False
    Application.EnableEvents = True

End Sub


Private Function ColLetter(ByVal colNum As Long) As String
    ColLetter = Split(Me.Cells(1, colNum).Address, "$")(1)
End Function
