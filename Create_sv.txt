Private Sub Worksheet_Change(ByVal Target As Range)

    Dim n As Long
    Dim i As Long
    Dim j As Long
    Dim src As Range
    Dim dest As Range

    If Target.Address <> "$C$4" Then Exit Sub

    Application.EnableEvents = False

    n = CLng(Me.Range("C4").Value)
    Set src = Me.Range("H5:L10")

    'Clear previous generated blocks
    Me.Range("M5:ZZ10").Clear

    For i = 1 To n - 1

        Set dest = src.Offset(0, i * 5)

        'Normal Excel copy - translates relative references
        src.Copy Destination:=dest

        'Match column widths
        For j = 1 To 5
            dest.Columns(j).ColumnWidth = src.Columns(j).ColumnWidth
        Next j

    Next i

    Application.EnableEvents = True

End Sub
