import inspect
import traceback
from datetime import datetime
from pathlib import Path

import pythoncom
import win32com.client


OUTPUT_DIRECTORY = Path(__file__).resolve().parent


LOG_LINES = []


def log(message=""):

    text = str(message)

    print(text)

    LOG_LINES.append(text)


def section(title):

    log()
    log("=" * 100)
    log(title)
    log("=" * 100)


def save_log():

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

    path = OUTPUT_DIRECTORY / f"e3_method_signature_audit_{timestamp}.txt"

    path.write_text("\n".join(LOG_LINES), encoding="utf-8")

    print()
    print(f"LOG WRITTEN: {path}")


def safe_string(value):

    try:
        return str(value)
    except Exception:
        return repr(value)


def connect_to_e3():

    section("CONNECTING TO E3")

    pythoncom.CoInitialize()

    app = win32com.client.Dispatch("CT.Application")

    prj = app.CreateJobObject()
    sym = prj.CreateSymbolObject()
    graph = prj.CreateGraphObject()
    pin = prj.CreatePinObject()

    log(f"APPLICATION: {app}")
    log(f"PROJECT: {prj}")
    log(f"SYMBOL OBJECT: {sym}")
    log(f"GRAPH OBJECT: {graph}")
    log(f"PIN OBJECT: {pin}")

    return app, prj, sym, graph, pin


def print_class_module_info(obj, object_name):

    section(f"{object_name} CLASS / MODULE INFO")

    cls = obj.__class__

    log(f"CLASS: {cls}")
    log(f"CLASS NAME: {getattr(cls, '__name__', None)}")
    log(f"MODULE NAME: {getattr(cls, '__module__', None)}")

    try:
        module = inspect.getmodule(cls)
    except Exception:
        module = None

    log(f"MODULE OBJECT: {module}")

    if module is not None:

        log(f"MODULE FILE: {getattr(module, '__file__', None)}")

    try:
        source_file = inspect.getsourcefile(cls)
    except Exception as exc:
        source_file = None
        log(f"inspect.getsourcefile ERROR: {exc}")

    log(f"SOURCE FILE: {source_file}")

    return source_file


def method_signature_details(obj, object_name, method_name):

    section(f"{object_name}.{method_name}")

    if not hasattr(obj, method_name):

        log("METHOD DOES NOT EXIST")

        return

    method = getattr(obj, method_name)

    log(f"METHOD OBJECT: {method}")

    try:
        signature = inspect.signature(method)
        log(f"inspect.signature: {signature}")
    except Exception as exc:
        log(f"inspect.signature ERROR: {exc}")

    try:
        fullargspec = inspect.getfullargspec(method)
        log(f"inspect.getfullargspec: {fullargspec}")
    except Exception as exc:
        log(f"inspect.getfullargspec ERROR: {exc}")

    log(f"__doc__: {safe_string(getattr(method, '__doc__', None))}")
    log(f"__defaults__: {safe_string(getattr(method, '__defaults__', None))}")
    log(f"__kwdefaults__: {safe_string(getattr(method, '__kwdefaults__', None))}")

    function_object = getattr(method, "__func__", method)

    code_object = getattr(function_object, "__code__", None)

    if code_object is not None:

        log(f"co_argcount: {code_object.co_argcount}")
        log(f"co_posonlyargcount: {getattr(code_object, 'co_posonlyargcount', None)}")
        log(f"co_kwonlyargcount: {code_object.co_kwonlyargcount}")
        log(f"co_varnames: {code_object.co_varnames}")
        log(f"co_names: {code_object.co_names}")
        log(f"co_consts: {code_object.co_consts}")

    try:
        source = inspect.getsource(method)
        log("PYTHON GENERATED SOURCE:")
        log(source)
    except Exception as exc:
        log(f"inspect.getsource ERROR: {exc}")

    try:
        source_file = inspect.getsourcefile(method)
        log(f"METHOD SOURCE FILE: {source_file}")
    except Exception as exc:
        log(f"METHOD SOURCE FILE ERROR: {exc}")


def extract_method_source_from_file(source_file, method_names):

    section("RAW GENERATED WRAPPER SOURCE EXTRACT")

    if source_file is None:

        log("NO SOURCE FILE AVAILABLE")

        return

    path = Path(source_file)

    if not path.exists():

        log(f"SOURCE FILE DOES NOT EXIST: {path}")

        return

    log(f"READING: {path}")

    try:
        lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    except Exception as exc:
        log(f"READ ERROR: {exc}")

        return

    for method_name in method_names:

        section(f"RAW SOURCE AROUND def {method_name}")

        target = f"def {method_name}("

        matches = []

        for index, line in enumerate(lines):

            if target in line:
                matches.append(index)

        if not matches:

            log("NO MATCH FOUND")

            continue

        for match_index in matches:

            start = max(0, match_index - 5)
            end = min(len(lines), match_index + 35)

            log(f"MATCH AT FILE LINE {match_index + 1}")

            for index in range(start, end):
                log(f"{index + 1:6d}: {lines[index]}")


def audit_object_methods(obj, object_name, method_names):

    source_file = print_class_module_info(obj, object_name)

    for method_name in method_names:
        method_signature_details(obj, object_name, method_name)

    extract_method_source_from_file(source_file, method_names)


def main():

    section("E3 METHOD SIGNATURE / GENERATED WRAPPER AUDIT")

    log("READ-ONLY")
    log("THIS SCRIPT DOES NOT MOVE, DELETE, PLACE, OR MODIFY ANYTHING.")

    app, prj, sym, graph, pin = connect_to_e3()

    symbol_methods = [
        "Place",
        "PlacePinView",
        "PlacePins",
        "PlaceInteractively",
        "GetBoundIds",
        "IsConnected",
        "AssignToConnector",
        "GetSchemaLocation",
    ]

    graph_methods = [
        "Delete",
        "Place",
        "GetLine",
        "CreateLine",
        "GetLineColour",
        "GetLineStyle",
        "GetLineWidth",
    ]

    pin_methods = [
        "PlaceNode",
        "GetSchemaLocation",
        "GetPhysicalPosition",
        "GetViewIds",
        "GetNodeIds",
        "GetConnectedNodeIds",
        "SetNodeId",
    ]

    audit_object_methods(sym, "e3Symbol", symbol_methods)
    audit_object_methods(graph, "e3Graph", graph_methods)
    audit_object_methods(pin, "e3Pin", pin_methods)


if __name__ == "__main__":

    try:
        main()

    except KeyboardInterrupt:
        log()
        log("STOPPED BY USER")

    except Exception as exc:
        log()
        log("FATAL ERROR")
        log(safe_string(exc))
        log(traceback.format_exc())

    finally:

        try:
            save_log()
        except Exception as exc:
            print(f"COULD NOT SAVE LOG: {exc}")

        try:
            pythoncom.CoUninitialize()
        except Exception:
            pass
