def format_schematic_connector_text():
    device_result = prj.GetAllDeviceIds(None)
    device_ids = [x for x in device_result[1] if x is not None]

    for device_id in device_ids:
        dev.SetId(device_id)

        try:
            device_name = str(dev.GetName())
        except Exception:
            continue

        try:
            symbol_result = dev.GetSymbolIds(None)
            logical_symbol_ids = [x for x in symbol_result[1] if x is not None]
        except Exception:
            logical_symbol_ids = []

        is_connector = False

        for logical_symbol_id in logical_symbol_ids:
            sym.SetId(logical_symbol_id)

            try:
                text_result = sym.GetTextIds(None)
                text_ids = [x for x in text_result[1] if x is not None]
            except Exception:
                text_ids = []

            for text_id in text_ids:
                text.SetId(text_id)

                try:
                    text_type = text.GetType()
                    text_value = str(text.GetText()).strip()
                except Exception:
                    continue

                if text_type == 5 or text_value.startswith("Generic_"):
                    is_connector = True
                    break

            if is_connector:
                break

        if not is_connector:
            continue

        print("FORMATTING CONNECTOR:", device_name)

        try:
            pin_result = dev.GetPinIds(None)
            logical_pin_ids = [x for x in pin_result[1] if x is not None]
        except Exception:
            logical_pin_ids = []

        for logical_pin_id in logical_pin_ids:
            pin.SetId(logical_pin_id)

            try:
                logical_pin_name = str(pin.GetName())
            except Exception:
                logical_pin_name = "UNKNOWN"

            try:
                view_result = pin.GetViewIds(None)
                pin_view_ids = [x for x in view_result[1] if x is not None]
            except Exception:
                try:
                    view_result = pin.GetViewIds()
                    pin_view_ids = [x for x in view_result[1] if x is not None]
                except Exception:
                    pin_view_ids = []

            for pin_view_id in pin_view_ids:
                sym.SetId(pin_view_id)

                try:
                    text_result = sym.GetTextIds(None)
                    symbol_text_ids = [x for x in text_result[1] if x is not None]
                except Exception:
                    symbol_text_ids = []

                split_position = None

                for text_id in symbol_text_ids:
                    text.SetId(text_id)

                    try:
                        text_type = text.GetType()
                        text_value = str(text.GetText()).strip()
                    except Exception:
                        continue

                    if text_type == 213 and "/" in text_value and not text_value.startswith("*"):
                        try:
                            split_position = int(text_value.split("/")[0])
                        except Exception:
                            split_position = None

                        break

                if split_position is None:
                    continue

                print("PIN:", logical_pin_name, "VIEW:", pin_view_id, "POSITION:", split_position)

                for text_id in symbol_text_ids:
                    text.SetId(text_id)

                    try:
                        text_type = text.GetType()
                        text_value = str(text.GetText()).strip()
                    except Exception:
                        continue

                    # ================================================
                    # COMPONENT CODE
                    # ALWAYS HIDE
                    # ================================================
                    if text_type == 5 or text_value.startswith("Generic_"):
                        result = text.SetVisibility(0)
                        print("    COMPONENT CODE OFF:", repr(text_value), "RESULT:", result)

                    # ================================================
                    # DEVICE DESIGNATION
                    #
                    # IMPORTANT:
                    # NEVER TURN TYPE 12 ON.
                    # TOP PIN IS LEFT COMPLETELY ALONE.
                    #
                    # ONLY TURN OFF POSITIONS 2, 3, 4...
                    # ================================================
                    elif text_type == 12:
                        if split_position != 1:
                            result = text.SetVisibility(0)
                            print("    DEVICE DESIGNATION OFF:", repr(text_value), "RESULT:", result)
                        else:
                            print("    TOP DEVICE DESIGNATION LEFT ALONE:", repr(text_value))

                    # ================================================
                    # LOCAL SIGNAL NAME
                    # SHOW AND FORMAT
                    # ================================================
                    elif text_type == 617:
                        text.SetVisibility(1)
                        text.SetColour(0)
                        text.SetAlignment(3)
                        text.SetStyle(1)
                        text.SetHeight(0.2)
                        print("    LOCAL SIGNAL ON:", repr(text_value))
