import pyodbc


# ============================================================
# CONFIG
# ============================================================

COMPONENT_NAME = "YOUR_TEST_COMPONENT"
COMPONENT_VERSION = "1"

CONNECTION_STRING = (
    "DRIVER={SQL Server};"
    "SERVER=sql-zuken-prd;"
    "DATABASE=Zuken_E3Library;"
    "Trusted_Connection=yes;"
)


# ============================================================
# CONNECTION
# ============================================================

conn = pyodbc.connect(CONNECTION_STRING, autocommit=True)
cur = conn.cursor()


# ============================================================
# GET COMPONENT ID + LPNTR
# ============================================================

cur.execute("""
SELECT *
FROM dbo.ComponentData
WHERE ENTRY = ?
AND VERSION = ?
""", COMPONENT_NAME, COMPONENT_VERSION)

component_row = cur.fetchone()

if not component_row:
    raise RuntimeError("Component not found.")

columns = [x[0] for x in cur.description]

component_data = dict(zip(columns, component_row))

component_id = component_data["ID"]
lpntr = component_data["LPNTR"]

print("COMPONENT ID:", component_id)
print("LPNTR:", lpntr)


# ============================================================
# FIND RELEVANT TABLES
# ============================================================

cur.execute("""
SELECT DISTINCT TABLE_NAME
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'dbo'
AND COLUMN_NAME IN (
    'LPNTR',
    'ENTRY',
    'ID'
)
ORDER BY TABLE_NAME
""")

tables = [row[0] for row in cur.fetchall()]

print("\nTABLES FOUND:", len(tables))


# ============================================================
# SNAPSHOT FUNCTION
# ============================================================

def snapshot():
    data = {}

    for table in tables:
        cur.execute("""
        SELECT COLUMN_NAME
        FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_SCHEMA = 'dbo'
        AND TABLE_NAME = ?
        ORDER BY ORDINAL_POSITION
        """, table)

        table_columns = [row[0] for row in cur.fetchall()]

        conditions = []
        params = []

        if "LPNTR" in table_columns:
            conditions.append("[LPNTR] = ?")
            params.append(lpntr)

        if "ENTRY" in table_columns:
            conditions.append("[ENTRY] IN (?, ?)")
            params.extend([COMPONENT_NAME, lpntr])

        if "ID" in table_columns:
            conditions.append("[ID] = ?")
            params.append(component_id)

        if not conditions:
            continue

        sql = f"""
        SELECT *
        FROM dbo.[{table}]
        WHERE {" OR ".join(conditions)}
        """

        try:
            cur.execute(sql, params)
            rows = {tuple(row) for row in cur.fetchall()}

            if rows:
                data[table] = rows

        except Exception as e:
            print("SKIPPED:", table, e)

    return data


# ============================================================
# BEFORE
# ============================================================

print("\nTAKING BEFORE SNAPSHOT...")

before = snapshot()

print("Tables containing component data:", len(before))

input("\nNow manually ADD EXACTLY ONE CONNECTOR in DBE, SAVE, then press ENTER...")


# ============================================================
# AFTER
# ============================================================

print("\nTAKING AFTER SNAPSHOT...")

after = snapshot()


# ============================================================
# DELTA
# ============================================================

print("\n" + "=" * 80)
print("DATABASE DELTA")
print("=" * 80)

all_tables = sorted(set(before) | set(after))

for table in all_tables:
    old = before.get(table, set())
    new = after.get(table, set())

    added = new - old
    removed = old - new

    if not added and not removed:
        continue

    print("\n" + "-" * 80)
    print(table)
    print("-" * 80)

    if added:
        print("ADDED:")
        for row in added:
            print(row)

    if removed:
        print("REMOVED:")
        for row in removed:
            print(row)


cur.close()
conn.close()
