def is_missing_value(value):

    if value is None:
        return True

    try:
        return int(value) == COM_MISSING
    except Exception:
        return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value > 0:
            cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    if result is None:
        return []

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                count = int(items[0])
            except Exception:
                count = None

            values = clean_id_list(items[1])

            if count is not None and len(values) > count:
                values = values[-count:]

            return values

        return clean_id_list(items)

    return clean_id_list(result)


def call_id_array(method):

    try:
        raw = method(None)

        log(f"RAW: {raw}")

        return parse_count_array_result(raw)

    except Exception as exc:
        log(f"method(None) ERROR: {exc}")

    try:
        raw = method()

        log(f"RAW: {raw}")

        return parse_count_array_result(raw)

    except Exception as exc:
        log(f"method() ERROR: {exc}")

    return []


def normalize_name(value):

    if value is None:
        return ""

    return str(value).strip()


def show_method_signature(obj, method_name):

    if not hasattr(obj, method_name):

        log(f"{method_name}: METHOD NOT PRESENT")

        return

    method = getattr(obj, method_name)

    try:
        signature = inspect.signature(method)
    except Exception as exc:
        signature = f"<signature error: {exc}>"

    try:
        doc = method.__doc__
    except Exception:
        doc = None

    log(f"{method_name}:")
    log(f"    signature: {signature}")
    log(f"    doc: {doc}")


def call_method_variants(obj, method_name, variants):

    section(f"{method_name} TEST")

    if not hasattr(obj, method_name):

        log("METHOD NOT PRESENT")

        return

    method = getattr(obj, method_name)

    show_method_signature(obj, method_name)

    for label, args in variants:

        try:
            result = method(*args)

            log(f"{label}")
            log(f"    RETURN: {result}")

        except Exception as exc:

            log(f"{label}")
            log(f"    ERROR: {exc}")
