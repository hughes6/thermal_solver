def connect_wire(wire_copy, node_no, end1_copy, pin1, end2_copy, pin2):
    print(
        "Wire:", wire_copy,
        "Node:", node_no,
        "End1:", end1_copy,
        "Pin1:", pin1,
        "End2:", end2_copy,
        "Pin2:", pin2
    )

    cur.execute("""
    INSERT INTO dbo.WIRTYP_CONNECT_INFOS
    (LPNTR, FLAGS, COPY, NODENO, COPYEND1, PINCOPYEND1, COPYEND2, PINCOPYEND2)
    VALUES (?, 0, ?, ?, ?, ?, ?, ?)
    """,
    LPNTR,
    wire_copy,
    node_no,
    end1_copy,
    pin1,
    end2_copy,
    pin2)




# ============================================================
# WIRES + PIN ASSIGNMENTS
# ============================================================

added_wire_copies = set()

for wire_data in WIRES:

    (
        wire_part,
        wire_version,
        pin1,
        pin2,
        wire_copy,
        no_wires,
        conductor_no
    ) = wire_data

    # --------------------------------------------------------
    # Add the physical wire/cable to the assembly only once
    # --------------------------------------------------------
    if wire_copy not in added_wire_copies:

        app.PutMessage(
            f"Adding wire COPY {wire_copy}: "
            f"{wire_part} ({no_wires} conductor(s))"
        )

        add_member(
            wire_copy,
            wire_part,
            wire_version
        )

        added_wire_copies.add(wire_copy)

    # --------------------------------------------------------
    # Resolve connector physical pins -> database pin COPYs
    # --------------------------------------------------------
    pin1_copy = get_pin_copy(
        CONNECTOR_P1,
        1,
        pin1
    )

    pin2_copy = get_pin_copy(
        CONNECTOR_P2,
        1,
        pin2
    )

    # --------------------------------------------------------
    # Determine the node for this conductor
    # --------------------------------------------------------

    if no_wires == 1:
        node_no = 4
    else:
        node_no = 4 + (conductor_no - 1)

    app.PutMessage(
        f"Wire COPY {wire_copy}, "
        f"Conductor {conductor_no}: "
        f"{wire_part} | "
        f"Connector 1 Pin {pin1} -> "
        f"Connector 2 Pin {pin2} | "
        f"Node {node_no}"
    )

    connect_wire(
        wire_copy,
        node_no,
        1,
        pin1_copy,
        2,
        pin2_copy
    )


cur.execute("""
SELECT
    COPY,
    NODENO,
    COPYEND1,
    PINCOPYEND1,
    COPYEND2,
    PINCOPYEND2
FROM dbo.WIRTYP_CONNECT_INFOS
WHERE LPNTR = ?
AND COPY = ?
ORDER BY NODENO
""", known_lpntr, known_twisted_pair_copy)

for row in cur.fetchall():
    print(tuple(row))
