cur.execute("""
SELECT
    c.ENTRY,
    c.VERSION,
    c.Description,
    COUNT(cb.COPY) AS NumberOfWires
FROM dbo.ComponentData c

LEFT JOIN dbo.Cable cb
    ON cb.ENTRY = c.LPNTR
    AND cb.BundleID = 0
    AND cb.NODENO > 0

WHERE c.ENTRYTYP = 22

GROUP BY
    c.ENTRY,
    c.VERSION,
    c.Description

ORDER BY
    c.ENTRY,
    c.VERSION
""")

for entry, version, description, number_of_wires in cur.fetchall():
    print(
        f"{entry} | "
        f"Wires: {number_of_wires} | "
        f"desc: {description}"
    )
