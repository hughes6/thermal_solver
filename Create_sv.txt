# TEST CREATED VIEW IDS DIRECTLY
# ================================================

print("")
print("============================================================")
print("CREATED VIEW ID TEST")
print("============================================================")

for data in SCHEMATIC_DATA:
    print("")
    print("CONNECTOR:", data["name"])
    print("PARENT BLOCK ID:", data["parent_block"])
    print("CONNECTOR VIEW ID:", data["connector_view"])

    test_ids = [
        ("PARENT_BLOCK", data["parent_block"]),
        ("CONNECTOR_VIEW", data["connector_view"])
    ]

    for label, test_id in test_ids:
        print("")
        print("----------------", label, test_id, "----------------")

        try:
            dev.SetId(test_id)
            print("DEV NAME:", repr(dev.GetName()))
        except Exception as e:
            print("DEV SET/NAME ERROR:", repr(e))
            continue

        try:
            symbolinfo = dev.GetSymbolIds(None)
            symbolids = [x for x in symbolinfo[1] if x is not None]
        except Exception as e:
            symbolids = []
            print("GetSymbolIds ERROR:", repr(e))

        print("SYMBOL IDS:", symbolids)

        for symbol_id in symbolids:
            sym.SetId(symbol_id)

            try:
                textinfo = sym.GetTextIds(None)
                textids = [x for x in textinfo[1] if x is not None]
            except Exception as e:
                textids = []

            print("    SYMBOL:", symbol_id)

            for text_id in textids:
                text.SetId(text_id)

                try:
                    text_value = text.GetText()
                except Exception:
                    text_value = "<ERROR>"

                try:
                    text_type = text.GetType()
                except Exception:
                    text_type = "<ERROR>"

                print("        TEXT:", text_id, "TYPE:", text_type, "VALUE:", repr(text_value))

        try:
            pininfo = dev.GetPinIds(None)
            pinids = [x for x in pininfo[1] if x is not None]
        except Exception as e:
            pinids = []
            print("GetPinIds ERROR:", repr(e))

        print("PIN IDS:", pinids)

        for pin_id in pinids:
            pin.SetId(pin_id)

            try:
                pin_name = pin.GetName()
            except Exception:
                pin_name = "<ERROR>"

            print("    PIN:", pin_id, "NAME:", repr(pin_name))

            try:
                pintextinfo = pin.GetTextIds(None)
                pintextids = [x for x in pintextinfo[1] if x is not None]
            except Exception as e:
                pintextids = []

            for pin_text_id in pintextids:
                text.SetId(pin_text_id)

                try:
                    text_value = text.GetText()
                except Exception:
                    text_value = "<ERROR>"

                try:
                    text_type = text.GetType()
                except Exception:
                    text_type = "<ERROR>"

                print("        PIN TEXT:", pin_text_id, "TYPE:", text_type, "VALUE:", repr(text_value))

print("")
print("============================================================")
print("END CREATED VIEW TEST")
print("============================================================")
