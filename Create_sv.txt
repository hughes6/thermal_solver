def safe_string(value):

    try:
        return str(value)
    except Exception:
        return repr(value)


def is_missing_value(value):

    if value is None:
        return True

    try:
        if int(value) == COM_MISSING:
            return True
    except Exception:
        pass

    return False


def clean_id_list(values):

    cleaned = []

    if values is None:
        return cleaned

    if not isinstance(values, (tuple, list)):
        values = [values]

    for value in values:

        if is_missing_value(value):
            continue

        try:
            value = int(value)
        except Exception:
            continue

        if value <= 0:
            continue

        cleaned.append(value)

    return cleaned


def parse_count_array_result(result):

    parsed = {
        "count": None,
        "raw_array": None,
        "values": [],
    }

    if result is None:
        return parsed

    if isinstance(result, (tuple, list)):

        items = list(result)

        if len(items) >= 2 and isinstance(items[1], (tuple, list)):

            try:
                parsed["count"] = int(items[0])
            except Exception:
                parsed["count"] = None

            parsed["raw_array"] = items[1]
            parsed["values"] = clean_id_list(items[1])

            if parsed["count"] is not None and len(parsed["values"]) > parsed["count"]:
                parsed["values"] = parsed["values"][-parsed["count"]:]

            return parsed

        parsed["raw_array"] = items
        parsed["values"] = clean_id_list(items)

        return parsed

    parsed["raw_array"] = result
    parsed["values"] = clean_id_list(result)

    return parsed


def call_id_array(method, method_name):

    attempts = []

    try:
        result = method(None)
        parsed = parse_count_array_result(result)

        attempts.append({
            "call": f"{method_name}(None)",
            "raw": safe_string(result),
            "count": parsed["count"],
            "values": parsed["values"],
        })

        if parsed["values"]:
            return parsed["values"], attempts

    except Exception as exc:
        attempts.append({
            "call": f"{method_name}(None)",
            "error": safe_string(exc),
        })

    try:
        result = method()
        parsed = parse_count_array_result(result)

        attempts.append({
            "call": f"{method_name}()",
            "raw": safe_string(result),
            "count": parsed["count"],
            "values": parsed["values"],
        })

        if parsed["values"]:
            return parsed["values"], attempts

    except Exception as exc:
        attempts.append({
            "call": f"{method_name}()",
            "error": safe_string(exc),
        })

    return [], attempts


def get_method_names(obj):

    names = set()

    try:
        for name in dir(obj):
            names.add(name)
    except Exception:
        pass

    try:
        type_info = obj._oleobj_.GetTypeInfo()
        type_attr = type_info.GetTypeAttr()
        function_count = type_attr[8]

        for index in range(function_count):

            try:
                func_desc = type_info.GetFuncDesc(index)
                member_id = func_desc[0]
                name = type_info.GetDocumentation(member_id)[0]

                if name:
                    names.add(name)

            except Exception:
                pass

    except Exception:
        pass

    return sorted(names)

def connect_to_e3():

    section("CONNECTING TO E3")

    pythoncom.CoInitialize()

    app = win32com.client.Dispatch("CT.Application")
    prj = app.CreateJobObject()
    dev = prj.CreateDeviceObject()
    pin = prj.CreatePinObject()
    sym = prj.CreateSymbolObject()

    log(f"APPLICATION: {app}")
    log(f"PROJECT / JOB: {prj}")

    return app, prj, dev, pin, sym

def get_all_device_ids(prj):

    ids, attempts = call_id_array(prj.GetAllDeviceIds, "GetAllDeviceIds")

    for attempt in attempts:

        if "error" in attempt:
            log(f"{attempt['call']} ERROR: {attempt['error']}")
        else:
            log(f"{attempt['call']} RAW: {attempt['raw']}")
            log(f"{attempt['call']} VALUES: {attempt['values']}")

    return ids


def get_device_pin_ids(dev):

    ids, attempts = call_id_array(dev.GetPinIds, "GetPinIds")

    return ids


def get_pin_view_ids(pin):

    ids, attempts = call_id_array(pin.GetViewIds, "GetViewIds")

    return ids


def get_symbol_bound_ids(sym):

    ids, attempts = call_id_array(sym.GetBoundIds, "GetBoundIds")

    return ids, attempts


def get_device_name(dev):

    try:
        return safe_string(dev.GetName())
    except Exception:
        return ""


def get_pin_name(pin):

    try:
        return safe_string(pin.GetName())
    except Exception:
        return ""


def find_connector_device_id(prj, dev):

    section(f"FIND CONNECTOR: {CONNECTOR_NAME}")

    device_ids = get_all_device_ids(prj)

    log(f"DEVICE IDS: {device_ids}")

    matches = []

    for device_id in device_ids:

        try:
            dev.SetId(device_id)
        except Exception:
            continue

        name = get_device_name(dev)

        log(f"DEVICE {device_id}: {name}")

        if name == CONNECTOR_NAME:
            matches.append(device_id)

    if len(matches) == 1:

        log(f"SELECTED DEVICE ID: {matches[0]}")

        return matches[0]

    if len(matches) > 1:
        raise RuntimeError(f"Multiple devices named {CONNECTOR_NAME}: {matches}")

    raise RuntimeError(f"Could not find connector {CONNECTOR_NAME}.")


def build_pin_table(dev, pin, sym, connector_device_id):

    dev.SetId(connector_device_id)

    pin_ids = get_device_pin_ids(dev)

    records = []

    for pin_id in pin_ids:

        pin.SetId(pin_id)

        pin_name = get_pin_name(pin)
        view_ids = get_pin_view_ids(pin)

        record = {
            "pin_id": pin_id,
            "pin_name": pin_name,
            "view_ids": view_ids,
            "views": [],
        }

        for view_id in view_ids:

            sym.SetId(view_id)

            bound_ids, bound_attempts = get_symbol_bound_ids(sym)

            record["views"].append({
                "view_id": view_id,
                "bound_ids": bound_ids,
                "bound_attempts": bound_attempts,
            })

        records.append(record)

    return records


def print_pin_table(records):

    section("CONNECTOR PIN / BOUND TABLE")

    for record in records:

        log(f"PIN {record['pin_name']} | PIN ID {record['pin_id']} | VIEW IDS {record['view_ids']}")

        for view in record["views"]:
            log(f"    VIEW {view['view_id']} | BOUND IDS {view['bound_ids']}")


def find_pin_record(records, pin_name):

    target = str(pin_name)

    for record in records:

        if record["pin_name"] == target:
            return record

    return None

def get_project_creator_names(prj):

    names = get_method_names(prj)

    creators = []

    for name in names:

        if not name.startswith("Create"):
            continue

        if not name.endswith("Object"):
            continue

        creators.append(name)

    return sorted(creators)


def instantiate_project_objects(prj):

    section("DISCOVERING PROJECT OBJECT FACTORIES")

    creator_names = get_project_creator_names(prj)

    log(f"FOUND {len(creator_names)} Create*Object METHODS")

    objects = []

    for creator_name in creator_names:

        log(f"TRY {creator_name}")

        try:
            creator = getattr(prj, creator_name)
            obj = creator()
        except Exception as exc:
            log(f"    CREATE FAILED: {exc}")
            continue

        if obj is None:
            log("    RETURNED None")
            continue

        has_set_id = hasattr(obj, "SetId")
        has_get_id = hasattr(obj, "GetId")

        log(f"    CREATED: {obj}")
        log(f"    HAS SetId: {has_set_id}")
        log(f"    HAS GetId: {has_get_id}")

        objects.append({
            "creator_name": creator_name,
            "object": obj,
            "has_set_id": has_set_id,
            "has_get_id": has_get_id,
        })

    add_event("project_object_factories", {
        "creators": [item["creator_name"] for item in objects],
    })

    return objects

def safe_getter_call(obj, getter_name):

    if not hasattr(obj, getter_name):
        return {
            "available": False,
        }

    getter = getattr(obj, getter_name)

    try:
        value = getter()

        return {
            "available": True,
            "success": True,
            "value": value,
        }

    except Exception as exc:

        return {
            "available": True,
            "success": False,
            "error": safe_string(exc),
        }


def object_identity_fingerprint(obj):

    fingerprint = {}

    for getter_name in SAFE_GETTERS:

        result = safe_getter_call(obj, getter_name)

        if not result.get("available"):
            continue

        fingerprint[getter_name] = result

    return fingerprint


def count_meaningful_fingerprint_values(fingerprint, bound_id):

    count = 0

    for getter_name, result in fingerprint.items():

        if not result.get("success"):
            continue

        value = result.get("value")

        if value is None:
            continue

        if is_missing_value(value):
            continue

        if getter_name == "GetId":

            try:
                if int(value) == int(bound_id):
                    count += 2
            except Exception:
                pass

            continue

        if isinstance(value, str):

            if value.strip() != "":
                count += 1

            continue

        if isinstance(value, (tuple, list)):

            if len(value) > 0:
                count += 1

            continue

        try:
            numeric_value = float(value)

            if numeric_value != 0:
                count += 1

        except Exception:
            count += 1

    return count

def get_relevant_methods(obj):

    names = get_method_names(obj)

    relevant = []

    for name in names:

        lower_name = name.lower()

        if any(keyword in lower_name for keyword in METHOD_KEYWORDS):
            relevant.append(name)

    return relevant

def classify_bound_id(bound_id, project_objects):

    section(f"CLASSIFY BOUND ID {bound_id}")

    matches = []

    for item in project_objects:

        creator_name = item["creator_name"]
        obj = item["object"]

        if not item["has_set_id"]:
            continue

        log()
        log(f"TRYING {creator_name}")

        try:
            set_result = obj.SetId(bound_id)
            log(f"    SetId({bound_id}) RETURN: {set_result}")
        except Exception as exc:
            log(f"    SetId FAILED: {exc}")
            continue

        returned_id = None

        if item["has_get_id"]:

            try:
                returned_id = obj.GetId()
                log(f"    GetId() RETURN: {returned_id}")
            except Exception as exc:
                log(f"    GetId FAILED: {exc}")

        fingerprint = object_identity_fingerprint(obj)
        meaningful_count = count_meaningful_fingerprint_values(fingerprint, bound_id)

        id_match = False

        try:
            id_match = int(returned_id) == int(bound_id)
        except Exception:
            id_match = False

        log(f"    ID MATCH: {id_match}")
        log(f"    MEANINGFUL FINGERPRINT SCORE: {meaningful_count}")

        for getter_name, result in fingerprint.items():

            if result.get("success"):
                log(f"    {getter_name}: {result.get('value')}")
            else:
                log(f"    {getter_name}: ERROR {result.get('error')}")

        relevant_methods = get_relevant_methods(obj)

        log("    RELEVANT METHODS:")

        if relevant_methods:

            for method_name in relevant_methods:
                log(f"        {method_name}")

        else:
            log("        NONE")

        if id_match or meaningful_count > 0:

            match = {
                "bound_id": bound_id,
                "creator_name": creator_name,
                "set_result": safe_string(set_result),
                "returned_id": returned_id,
                "id_match": id_match,
                "meaningful_score": meaningful_count,
                "fingerprint": fingerprint,
                "relevant_methods": relevant_methods,
            }

            matches.append(match)

    subsection(f"SUMMARY FOR BOUND ID {bound_id}")

    if not matches:

        log("NO PROJECT OBJECT INTERFACE PRODUCED A MEANINGFUL MATCH")

    else:

        ranked = sorted(matches, key=lambda item: (item["id_match"], item["meaningful_score"]), reverse=True)

        for match in ranked:

            log(f"{match['creator_name']} | ID MATCH {match['id_match']} | SCORE {match['meaningful_score']} | GetId {match['returned_id']}")

    add_event("bound_id_classification", {
        "bound_id": bound_id,
        "matches": matches,
    })

    return matches


def classify_pin_bounds(records, pin_name, project_objects):

    pin_record = find_pin_record(records, pin_name)

    if pin_record is None:
        raise RuntimeError(f"Pin {pin_name} not found.")

    section(f"BOUND CLASSIFICATION FOR PIN {pin_name}")

    log(f"PIN ID: {pin_record['pin_id']}")

    all_bound_ids = []

    for view in pin_record["views"]:

        log(f"VIEW {view['view_id']} -> BOUND IDS {view['bound_ids']}")

        for bound_id in view["bound_ids"]:

            if bound_id not in all_bound_ids:
                all_bound_ids.append(bound_id)

    if not all_bound_ids:

        log("THIS PIN CURRENTLY HAS NO BOUND IDS")

        return

    for bound_id in all_bound_ids:
        classify_bound_id(bound_id, project_objects)

# MANUAL BEFORE / AFTER TEST
# ============================================================

def compare_bound_tables(before, after):

    section("BOUND-ID DIFFERENCE REPORT")

    before_map = {record["pin_name"]: record for record in before}
    after_map = {record["pin_name"]: record for record in after}

    pin_names = sorted(set(before_map.keys()) | set(after_map.keys()))

    change_count = 0

    for pin_name in pin_names:

        before_record = before_map.get(pin_name)
        after_record = after_map.get(pin_name)

        if before_record is None or after_record is None:
            log(f"PIN {pin_name}: PIN RECORD ADDED OR REMOVED")
            change_count += 1
            continue

        before_bounds = []

        for view in before_record["views"]:
            before_bounds.extend(view["bound_ids"])

        after_bounds = []

        for view in after_record["views"]:
            after_bounds.extend(view["bound_ids"])

        before_bounds = sorted(set(before_bounds))
        after_bounds = sorted(set(after_bounds))

        if before_bounds != after_bounds:
            log(f"PIN {pin_name}: {before_bounds} -> {after_bounds}")
            change_count += 1

    if change_count == 0:
        log("NO BOUND-ID DIFFERENCES DETECTED")
    else:
        log(f"TOTAL PINS WITH BOUND-ID CHANGES: {change_count}")


def manual_move_compare(dev, pin, sym, connector_device_id, project_objects):

    before = build_pin_table(dev, pin, sym, connector_device_id)

    print_pin_table(before)

    pin_name = input(f"Pin to manually move [default {DEFAULT_PIN_NAME}]: ").strip()

    if pin_name == "":
        pin_name = DEFAULT_PIN_NAME

    before_record = find_pin_record(before, pin_name)

    if before_record is None:
        log(f"PIN {pin_name} NOT FOUND")
        return

    log()
    log(f"BEFORE MANUAL MOVE, PIN {pin_name} HAS:")

    for view in before_record["views"]:
        log(f"    VIEW {view['view_id']} -> {view['bound_ids']}")

    input("Manually move that pin in E3, then press ENTER here: ")

    after = build_pin_table(dev, pin, sym, connector_device_id)

    print_pin_table(after)

    compare_bound_tables(before, after)

    after_record = find_pin_record(after, pin_name)

    if after_record is not None:

        log()
        log(f"AFTER MANUAL MOVE, PIN {pin_name} HAS:")

        for view in after_record["views"]:
            log(f"    VIEW {view['view_id']} -> {view['bound_ids']}")

    answer = input("Classify the OLD bound IDs from before the move? [y/n]: ").strip().lower()

    if answer == "y":

        old_bound_ids = []

        for view in before_record["views"]:

            for bound_id in view["bound_ids"]:

                if bound_id not in old_bound_ids:
                    old_bound_ids.append(bound_id)

        for bound_id in old_bound_ids:
            classify_bound_id(bound_id, project_objects)

def save_results():

    AUDIT_DATA["finished"] = datetime.now().isoformat(timespec="milliseconds")

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

    txt_path = OUTPUT_DIRECTORY / f"e3_bound_classifier_{timestamp}.txt"
    json_path = OUTPUT_DIRECTORY / f"e3_bound_classifier_{timestamp}.json"

    txt_path.write_text("\n".join(LOG_LINES), encoding="utf-8")
    json_path.write_text(json.dumps(AUDIT_DATA, indent=2, default=str), encoding="utf-8")

    print()
    print("=" * 100)
    print("AUDIT FILES WRITTEN")
    print("=" * 100)
    print(txt_path)
    print(json_path)

def main():

    AUDIT_DATA["started"] = datetime.now().isoformat(timespec="milliseconds")

    section("E3 BOUND-ID CLASSIFIER")

    log("READ-ONLY AUDIT")
    log("THIS SCRIPT DOES NOT CALL Delete, Remove, Disconnect, Break, or any other destructive method.")
    log(f"CONNECTOR_NAME = {CONNECTOR_NAME}")

    app, prj, dev, pin, sym = connect_to_e3()

    connector_device_id = find_connector_device_id(prj, dev)

    project_objects = instantiate_project_objects(prj)

    while True:

        records = build_pin_table(dev, pin, sym, connector_device_id)

        print_menu()

        choice = input("Select test: ").strip().lower()

        if choice == "1":

            print_pin_table(records)

        elif choice == "2":

            print_pin_table(records)

            pin_name = input(f"Pin to classify [default {DEFAULT_PIN_NAME}]: ").strip()

            if pin_name == "":
                pin_name = DEFAULT_PIN_NAME

            classify_pin_bounds(records, pin_name, project_objects)

        elif choice == "3":

            print_pin_table(records)

            for record in records:

                has_bounds = any(view["bound_ids"] for view in record["views"])

                if has_bounds:
                    classify_pin_bounds(records, record["pin_name"], project_objects)

        elif choice == "4":

            manual_move_compare(dev, pin, sym, connector_device_id, project_objects)

        elif choice == "5":

            project_objects = instantiate_project_objects(prj)

        elif choice == "q":

            break

        else:

            log("INVALID SELECTION")


if __name__ == "__main__":

    try:
        main()

    except KeyboardInterrupt:
        log()
        log("STOPPED BY USER")

    except Exception as exc:
        log()
        log("FATAL ERROR")
        log(safe_string(exc))
        log(traceback.format_exc())

    finally:

        try:
            save_results()
        except Exception as exc:
            print(f"COULD NOT SAVE RESULTS: {exc}")

        try:
            pythoncom.CoUninitialize()
        except Exception:
            pass
