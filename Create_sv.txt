def safe_string(value):

    try:
        return str(value)
    except Exception:
        return repr(value)


def normalize_ids(value):

    if value is None:
        return []

    if isinstance(value, bool):
        return [int(value)]

    if isinstance(value, int):
        return [value]

    if isinstance(value, float):
        return [int(value)]

    if isinstance(value, str):

        if value.strip() == "":
            return []

        try:
            return [int(value)]
        except Exception:
            return [value]

    try:
        return list(value)
    except Exception:
        return [value]


def safe_float(value):

    try:
        return float(value)
    except Exception:
        return None


def values_close(first, second, tolerance=0.0001):

    if first is None:
        return False

    if second is None:
        return False

    try:
        return abs(float(first) - float(second)) <= tolerance
    except Exception:
        return False


def unique_preserve_order(values):

    output = []

    seen = set()

    for value in values:

        key = safe_string(value)

        if key in seen:
            continue

        seen.add(key)

        output.append(value)

    return output


def input_choice(prompt, allowed):

    allowed_lower = {item.lower() for item in allowed}

    while True:

        value = input(prompt).strip().lower()

        if value in allowed_lower:
            return value

        print(f"Allowed values: {', '.join(sorted(allowed_lower))}")

def get_com_method_info(obj, method_name):

    result = {
        "method": method_name,
        "found": False,
        "names": None,
        "func_desc": None,
        "documentation": None,
        "python_doc": None,
    }

    try:
        type_info = obj._oleobj_.GetTypeInfo()
        type_attr = type_info.GetTypeAttr()
        function_count = type_attr[8]
    except Exception as exc:
        result["type_info_error"] = safe_string(exc)
        return result

    for index in range(function_count):

        try:
            func_desc = type_info.GetFuncDesc(index)
            member_id = func_desc[0]
            documentation = type_info.GetDocumentation(member_id)
            name = documentation[0]
        except Exception:
            continue

        if safe_string(name).lower() != method_name.lower():
            continue

        result["found"] = True
        result["func_desc"] = safe_string(func_desc)
        result["documentation"] = safe_string(documentation)

        try:
            names = type_info.GetNames(member_id)
            result["names"] = [safe_string(item) for item in names]
        except Exception as exc:
            result["names_error"] = safe_string(exc)

        break

    try:
        method = getattr(obj, method_name)
        result["python_doc"] = safe_string(getattr(method, "__doc__", None))
    except Exception as exc:
        result["python_doc_error"] = safe_string(exc)

    return result


def print_method_info(obj, object_name, method_name):

    info = get_com_method_info(obj, method_name)

    subsection(f"{object_name}.{method_name}")

    log(f"FOUND: {info.get('found')}")
    log(f"NAMES: {info.get('names')}")
    log(f"FUNC DESC: {info.get('func_desc')}")
    log(f"DOCUMENTATION: {info.get('documentation')}")
    log(f"PYTHON DOC: {info.get('python_doc')}")

    if "type_info_error" in info:
        log(f"TYPE INFO ERROR: {info['type_info_error']}")

    if "names_error" in info:
        log(f"NAMES ERROR: {info['names_error']}")

    if "python_doc_error" in info:
        log(f"PYTHON DOC ERROR: {info['python_doc_error']}")

    add_event("method_info", {
        "object": object_name,
        "method": method_name,
        "info": info,
    })

    return info

def try_calls(call_specs):

    results = []

    for description, function in call_specs:

        record = {
            "description": description,
            "success": False,
            "value": None,
            "error": None,
        }

        try:
            value = function()
            record["success"] = True
            record["value"] = value
        except Exception as exc:
            record["error"] = safe_string(exc)

        results.append(record)

    return results


def first_successful_value(results):

    for record in results:

        if record.get("success"):
            return record.get("value")

    return None


def print_call_results(label, results):

    log(f"{label}:")

    for record in results:

        if record.get("success"):
            log(f"    SUCCESS {record['description']} -> {record.get('value')}")
        else:
            log(f"    FAIL    {record['description']} -> {record.get('error')}")

def find_connector_device_id(prj, dev):

    section(f"FIND CONNECTOR: {CONNECTOR_NAME}")

    device_ids = get_all_device_ids(prj)

    log(f"TOTAL DEVICE IDS: {len(device_ids)}")

    exact_matches = []
    partial_matches = []

    for device_id in device_ids:

        try:
            dev.SetId(device_id)
        except Exception:
            continue

        name = get_device_name(dev)

        if name == CONNECTOR_NAME:
            exact_matches.append((device_id, name))

        if CONNECTOR_NAME.lower() in name.lower():
            partial_matches.append((device_id, name))

    if exact_matches:

        selected = exact_matches[0]

        log(f"EXACT MATCH: {selected}")

        return selected[0]

    if partial_matches:

        log("PARTIAL MATCHES:")

        for device_id, name in partial_matches:
            log(f"    {device_id} -> {name}")

        selected = partial_matches[0]

        log(f"USING FIRST PARTIAL MATCH: {selected}")

        return selected[0]

    log("NO CONNECTOR MATCH FOUND")

    return None

def extract_four_numeric_values(value):

    if value is None:
        return None

    if not isinstance(value, (tuple, list)):
        return None

    numeric = []

    for item in value:

        converted = safe_float(item)

        if converted is not None:
            numeric.append(converted)

    if len(numeric) < 4:
        return None

    return numeric[-4:]


def get_symbol_area(sym):

    calls = [
        ("GetArea()", lambda: sym.GetArea()),
        ("GetArea(None, None, None, None)", lambda: sym.GetArea(None, None, None, None)),
        ("GetPlacedArea()", lambda: sym.GetPlacedArea()),
        ("GetPlacedArea(None, None, None, None)", lambda: sym.GetPlacedArea(None, None, None, None)),
    ]

    results = try_calls(calls)

    for record in results:

        if not record.get("success"):
            continue

        area = extract_four_numeric_values(record.get("value"))

        if area is not None:
            return {
                "area": area,
                "source": record.get("description"),
                "calls": results,
            }

    return {
        "area": None,
        "source": None,
        "calls": results,
    }


def area_center(area):

    if area is None:
        return None, None

    if len(area) < 4:
        return None, None

    x1 = safe_float(area[0])
    y1 = safe_float(area[1])
    x2 = safe_float(area[2])
    y2 = safe_float(area[3])

    if x1 is None:
        return None, None

    if y1 is None:
        return None, None

    if x2 is None:
        return None, None

    if y2 is None:
        return None, None

    x = (x1 + x2) / 2.0
    y = (y1 + y2) / 2.0

    return x, y


def get_symbol_relation_value(sym, method_name):

    if not hasattr(sym, method_name):

        return {
            "method": method_name,
            "value": None,
            "calls": [],
            "missing": True,
        }

    method = getattr(sym, method_name)

    calls = [
        (f"{method_name}()", lambda: method()),
        (f"{method_name}(None)", lambda: method(None)),
        (f"{method_name}(None, None)", lambda: method(None, None)),
    ]

    results = try_calls(calls)

    value = first_successful_value(results)

    return {
        "method": method_name,
        "value": value,
        "calls": results,
        "missing": False,
    }

def build_snapshot(prj, dev, pin, sym, connector_device_id):

    snapshot = {
        "time": datetime.now().isoformat(timespec="milliseconds"),
        "connector_device_id": connector_device_id,
        "connector_name": "",
        "pins": [],
    }

    dev.SetId(connector_device_id)

    snapshot["connector_name"] = get_device_name(dev)

    pin_ids = get_pin_ids(dev)

    for logical_pin_id in pin_ids:

        pin_record = {
            "logical_pin_id": logical_pin_id,
            "pin_name": "",
            "logical_pin_is_view": None,
            "logical_pin_is_pin_view": None,
            "logical_physical_position": None,
            "logical_schema_location": None,
            "logical_symbol_name": None,
            "logical_template_symbol": None,
            "view_ids": [],
            "views": [],
        }

        try:
            pin.SetId(logical_pin_id)
        except Exception as exc:
            pin_record["logical_set_id_error"] = safe_string(exc)
            snapshot["pins"].append(pin_record)
            continue

        pin_record["pin_name"] = get_pin_name(pin)
        pin_record["logical_pin_is_view"] = pin_is_view(pin)
        pin_record["logical_pin_is_pin_view"] = pin_is_pin_view(pin)
        pin_record["logical_physical_position"] = get_pin_physical_position(pin)
        pin_record["logical_schema_location"] = get_pin_schema_location(pin)
        pin_record["logical_symbol_name"] = get_pin_symbol_name(pin)
        pin_record["logical_template_symbol"] = get_pin_template_symbol_id(pin)

        view_ids = unique_preserve_order(get_pin_view_ids(pin))

        pin_record["view_ids"] = view_ids

        for view_id in view_ids:

            view_record = {
                "view_id": view_id,
                "pin_object": {},
                "symbol_object": {},
            }

            try:
                pin.SetId(view_id)

                view_record["pin_object"]["set_id_ok"] = True
                view_record["pin_object"]["name"] = get_pin_name(pin)
                view_record["pin_object"]["is_view"] = pin_is_view(pin)
                view_record["pin_object"]["is_pin_view"] = pin_is_pin_view(pin)
                view_record["pin_object"]["physical_position"] = get_pin_physical_position(pin)
                view_record["pin_object"]["schema_location"] = get_pin_schema_location(pin)
                view_record["pin_object"]["symbol_name"] = get_pin_symbol_name(pin)
                view_record["pin_object"]["template_symbol"] = get_pin_template_symbol_id(pin)

            except Exception as exc:
                view_record["pin_object"]["set_id_ok"] = False
                view_record["pin_object"]["set_id_error"] = safe_string(exc)

            try:
                sym.SetId(view_id)

                view_record["symbol_object"]["set_id_ok"] = True
                view_record["symbol_object"]["name"] = get_symbol_name(sym)
                view_record["symbol_object"]["is_pin_view"] = symbol_is_pin_view(sym)
                view_record["symbol_object"]["sheet_id"] = get_symbol_sheet_id(sym, prj)

                area_info = get_symbol_area(sym)

                view_record["symbol_object"]["area"] = area_info

                center_x, center_y = area_center(area_info.get("area"))

                view_record["symbol_object"]["center_x"] = center_x
                view_record["symbol_object"]["center_y"] = center_y
                view_record["symbol_object"]["relationships"] = audit_symbol_relationships(sym)

            except Exception as exc:
                view_record["symbol_object"]["set_id_ok"] = False
                view_record["symbol_object"]["set_id_error"] = safe_string(exc)

            pin_record["views"].append(view_record)

        snapshot["pins"].append(pin_record)

    return snapshot

def relationship_compact(relations):

    output = {}

    for method_name, record in relations.items():

        value = record.get("value")

        output[method_name] = value

    return output


def print_snapshot(snapshot, title):

    section(title)

    log(f"CONNECTOR NAME: {snapshot.get('connector_name')}")
    log(f"CONNECTOR DEVICE ID: {snapshot.get('connector_device_id')}")
    log(f"LOGICAL PIN COUNT: {len(snapshot.get('pins', []))}")

    positioned = []

    for pin_record in snapshot.get("pins", []):

        pin_name = pin_record.get("pin_name")
        logical_pin_id = pin_record.get("logical_pin_id")

        log()
        log(f"PIN {pin_name}")
        log(f"    LOGICAL ID: {logical_pin_id}")
        log(f"    LOGICAL IsView: {pin_record.get('logical_pin_is_view')}")
        log(f"    LOGICAL IsPinView: {pin_record.get('logical_pin_is_pin_view')}")
        log(f"    VIEW IDS: {pin_record.get('view_ids')}")

        views = pin_record.get("views", [])

        for view_record in views:

            view_id = view_record.get("view_id")
            pin_info = view_record.get("pin_object", {})
            sym_info = view_record.get("symbol_object", {})

            log(f"    VIEW {view_id}")
            log(f"        PIN OBJECT SetId OK: {pin_info.get('set_id_ok')}")
            log(f"        PIN OBJECT name: {pin_info.get('name')}")
            log(f"        PIN OBJECT IsView: {pin_info.get('is_view')}")
            log(f"        PIN OBJECT IsPinView: {pin_info.get('is_pin_view')}")
            log(f"        PIN OBJECT physical position: {pin_info.get('physical_position', {}).get('value')}")
            log(f"        PIN OBJECT schema location: {pin_info.get('schema_location', {}).get('value')}")
            log(f"        PIN OBJECT symbol name: {pin_info.get('symbol_name')}")
            log(f"        SYMBOL OBJECT SetId OK: {sym_info.get('set_id_ok')}")
            log(f"        SYMBOL OBJECT name: {sym_info.get('name')}")
            log(f"        SYMBOL OBJECT IsPinView: {sym_info.get('is_pin_view')}")
            log(f"        SYMBOL OBJECT sheet ID: {sym_info.get('sheet_id')}")
            log(f"        SYMBOL OBJECT area: {sym_info.get('area', {}).get('area')}")
            log(f"        SYMBOL OBJECT center: ({sym_info.get('center_x')}, {sym_info.get('center_y')})")
            log(f"        RELATIONSHIPS: {relationship_compact(sym_info.get('relationships', {}))}")

            if sym_info.get("center_y") is not None:

                positioned.append({
                    "pin_name": pin_name,
                    "logical_pin_id": logical_pin_id,
                    "view_id": view_id,
                    "center_x": sym_info.get("center_x"),
                    "center_y": sym_info.get("center_y"),
                })

    if positioned:

        log()
        log("POSITION ORDER LOW Y -> HIGH Y")

        ordered_low_high = sorted(positioned, key=lambda item: item["center_y"])

        for item in ordered_low_high:
            log(f"    PIN {item['pin_name']} | VIEW {item['view_id']} | X {item['center_x']} | Y {item['center_y']}")

        log()
        log("POSITION ORDER HIGH Y -> LOW Y")

        ordered_high_low = sorted(positioned, key=lambda item: item["center_y"], reverse=True)

        for item in ordered_high_low:
            log(f"    PIN {item['pin_name']} | VIEW {item['view_id']} | X {item['center_x']} | Y {item['center_y']}")

# SNAPSHOT COMPARISON
# ============================================================

def snapshot_by_pin_name(snapshot):

    output = {}

    for pin_record in snapshot.get("pins", []):

        pin_name = safe_string(pin_record.get("pin_name"))

        output[pin_name] = pin_record

    return output


def view_by_id(pin_record):

    output = {}

    for view_record in pin_record.get("views", []):
        output[safe_string(view_record.get("view_id"))] = view_record

    return output


def compare_relationships(before_relations, after_relations):

    changes = []

    method_names = sorted(set(before_relations.keys()) | set(after_relations.keys()))

    for method_name in method_names:

        before_record = before_relations.get(method_name, {})
        after_record = after_relations.get(method_name, {})

        before_value = before_record.get("value")
        after_value = after_record.get("value")

        if safe_string(before_value) != safe_string(after_value):

            changes.append({
                "method": method_name,
                "before": before_value,
                "after": after_value,
            })

    return changes


def compare_snapshots(before, after, title):

    section(title)

    before_map = snapshot_by_pin_name(before)
    after_map = snapshot_by_pin_name(after)

    all_names = sorted(set(before_map.keys()) | set(after_map.keys()))

    change_count = 0

    report = []

    for pin_name in all_names:

        before_pin = before_map.get(pin_name)
        after_pin = after_map.get(pin_name)

        if before_pin is None:
            change_count += 1
            log(f"PIN {pin_name}: ADDED")
            report.append({
                "pin": pin_name,
                "change": "added",
            })
            continue

        if after_pin is None:
            change_count += 1
            log(f"PIN {pin_name}: REMOVED")
            report.append({
                "pin": pin_name,
                "change": "removed",
            })
            continue

        before_ids = before_pin.get("view_ids", [])
        after_ids = after_pin.get("view_ids", [])

        if safe_string(before_ids) != safe_string(after_ids):
            change_count += 1
            log(f"PIN {pin_name}: VIEW IDS CHANGED {before_ids} -> {after_ids}")

        before_views = view_by_id(before_pin)
        after_views = view_by_id(after_pin)

        all_view_ids = sorted(set(before_views.keys()) | set(after_views.keys()))

        for view_id in all_view_ids:

            before_view = before_views.get(view_id)
            after_view = after_views.get(view_id)

            if before_view is None:
                change_count += 1
                log(f"PIN {pin_name} VIEW {view_id}: VIEW ADDED")
                continue

            if after_view is None:
                change_count += 1
                log(f"PIN {pin_name} VIEW {view_id}: VIEW REMOVED")
                continue

            before_sym = before_view.get("symbol_object", {})
            after_sym = after_view.get("symbol_object", {})

            before_x = before_sym.get("center_x")
            before_y = before_sym.get("center_y")
            after_x = after_sym.get("center_x")
            after_y = after_sym.get("center_y")

            if not values_close(before_x, after_x) or not values_close(before_y, after_y):
                change_count += 1
                log(f"PIN {pin_name} VIEW {view_id}: POSITION ({before_x}, {before_y}) -> ({after_x}, {after_y})")

            before_rel = before_sym.get("relationships", {})
            after_rel = after_sym.get("relationships", {})

            relation_changes = compare_relationships(before_rel, after_rel)

            for relation_change in relation_changes:
                change_count += 1
                log(f"PIN {pin_name} VIEW {view_id}: {relation_change['method']} {relation_change['before']} -> {relation_change['after']}")

            report.append({
                "pin": pin_name,
                "view_id": view_id,
                "before_position": [before_x, before_y],
                "after_position": [after_x, after_y],
                "relationship_changes": relation_changes,
            })

    if change_count == 0:
        log("NO DIFFERENCES DETECTED")

    else:
        log()
        log(f"TOTAL DIFFERENCES DETECTED: {change_count}")

    add_event("snapshot_compare", {
        "title": title,
        "change_count": change_count,
        "report": report,
    })

    return change_count


# ============================================================
# PIN SELECTION
# ============================================================

def find_pin_record(snapshot, pin_name):

    target = safe_string(pin_name)

    for pin_record in snapshot.get("pins", []):

        if safe_string(pin_record.get("pin_name")) == target:
            return pin_record

    return None


def get_positioned_pin_candidates(snapshot):

    candidates = []

    for pin_record in snapshot.get("pins", []):

        pin_name = safe_string(pin_record.get("pin_name"))

        for view_record in pin_record.get("views", []):

            sym_info = view_record.get("symbol_object", {})

            center_y = sym_info.get("center_y")

            if center_y is None:
                continue

            candidates.append({
                "pin_name": pin_name,
                "logical_pin_id": pin_record.get("logical_pin_id"),
                "view_id": view_record.get("view_id"),
                "sheet_id": sym_info.get("sheet_id"),
                "center_x": sym_info.get("center_x"),
                "center_y": center_y,
            })

    return candidates


def get_top_candidate(snapshot):

    candidates = get_positioned_pin_candidates(snapshot)

    if not candidates:
        return None

    if TOP_IS_HIGH_Y:
        return max(candidates, key=lambda item: item["center_y"])

    return min(candidates, key=lambda item: item["center_y"])


def get_bottom_candidate(snapshot):

    candidates = get_positioned_pin_candidates(snapshot)

    if not candidates:
        return None

    if TOP_IS_HIGH_Y:
        return min(candidates, key=lambda item: item["center_y"])

    return max(candidates, key=lambda item: item["center_y"])


# ============================================================
# PLACE TARGET HELPERS
# ============================================================

def generate_y_targets():

    targets = []

    current = Y_MAX

    while current >= Y_MIN - 0.000001:

        targets.append(round(current, 6))

        current = current - Y_STEP

    return targets


def validate_place_target(x, y):

    if not values_close(x, FIXED_X, tolerance=0.000001):
        raise RuntimeError(f"X must be {FIXED_X}. Received {x}.")

    if y < Y_MIN:
        raise RuntimeError(f"Y {y} is below minimum {Y_MIN}.")

    if y > Y_MAX:
        raise RuntimeError(f"Y {y} is above maximum {Y_MAX}.")


def choose_different_y(current_y, preferred_index=0):

    targets = generate_y_targets()

    if not targets:
        raise RuntimeError("No Y targets generated.")

    for offset in range(len(targets)):

        index = (preferred_index + offset) % len(targets)

        candidate = targets[index]

        if current_y is None:
            return candidate

        if not values_close(candidate, current_y, tolerance=0.02):
            return candidate

    return targets[0]

# ============================================================

def observe_after_operation(prj, dev, pin, sym, connector_device_id, before_snapshot, label):

    observations = []

    start = time.perf_counter()

    for delay in OBSERVATION_DELAYS:

        target_time = start + delay
        remaining = target_time - time.perf_counter()

        if remaining > 0:
            time.sleep(remaining)

        snapshot = build_snapshot(prj, dev, pin, sym, connector_device_id)

        observations.append({
            "delay": delay,
            "snapshot": snapshot,
        })

        compare_snapshots(before_snapshot, snapshot, f"{label} AT +{delay:.3f} SEC")

    return observations

def run_symbol_place_test(prj, dev, pin, sym, connector_device_id, pin_name=None):

    section("CONTROLLED e3Symbol.Place TEST")

    before = build_snapshot(prj, dev, pin, sym, connector_device_id)

    print_snapshot(before, "PRE-PLACE SNAPSHOT")

    if pin_name is None:

        top = get_top_candidate(before)

        if top is None:
            log("NO POSITIONED PIN CANDIDATE FOUND")
            return

        pin_name = top["pin_name"]

        log(f"DEFAULTING TO CURRENT TOP PIN: {pin_name}")

    pin_record = find_pin_record(before, pin_name)

    if pin_record is None:
        log(f"PIN {pin_name} NOT FOUND")
        return

    if not pin_record.get("views"):
        log(f"PIN {pin_name} HAS NO VIEW")
        return

    view_record = pin_record["views"][0]
    sym_info = view_record.get("symbol_object", {})

    view_id = view_record.get("view_id")
    sheet_id = sym_info.get("sheet_id")
    current_y = sym_info.get("center_y")

    log(f"VIEW ID: {view_id}")
    log(f"SHEET ID: {sheet_id}")
    log(f"CURRENT CENTER: ({sym_info.get('center_x')}, {current_y})")

    if choice == "n":
        log("PLACE TEST CANCELLED")
        return

    validate_place_target(FIXED_X, target_y)

    try:
        sym.SetId(view_id)
    except Exception as exc:
        log(f"sym.SetId FAILED: {exc}")
        return

    started = time.perf_counter()

    try:
        result = sym.Place(sheet_id, FIXED_X, target_y, ROTATION, SCALE, MAINTAIN_TEXT_SIZE)
        elapsed = time.perf_counter() - started

        log(f"Place RETURN: {result}")
        log(f"Place ELAPSED: {elapsed:.6f} sec")

    except Exception as exc:
        elapsed = time.perf_counter() - started

        log(f"Place EXCEPTION AFTER {elapsed:.6f} sec: {exc}")
        log(traceback.format_exc())

        add_event("symbol_place_exception", {
            "pin_name": pin_name,
            "view_id": view_id,
            "target_x": FIXED_X,
            "target_y": target_y,
            "error": safe_string(exc),
        })

        return

    observations = observe_after_operation(prj, dev, pin, sym, connector_device_id, before, f"e3Symbol.Place PIN {pin_name}")

    add_event("symbol_place_test", {
        "pin_name": pin_name,
        "view_id": view_id,
        "target_x": FIXED_X,
        "target_y": target_y,
        "result": safe_string(result),
        "observation_count": len(observations),
    })

        log()
        result = get_symbol_relation_value(sym, method_name)

        print_call_results(method_name, result.get("calls", []))

def save_results():

    AUDIT_DATA["finished"] = datetime.now().isoformat(timespec="milliseconds")

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

    txt_path = OUTPUT_DIRECTORY / f"e3_pin_relationship_audit_{timestamp}.txt"
    json_path = OUTPUT_DIRECTORY / f"e3_pin_relationship_audit_{timestamp}.json"

    txt_path.write_text("\n".join(LOG_LINES), encoding="utf-8")
    json_path.write_text(json.dumps(AUDIT_DATA, indent=2, default=str), encoding="utf-8")

    print()
    print("=" * 100)
    print("AUDIT FILES WRITTEN")
    print("=" * 100)
    print(txt_path)
    print(json_path)

# ============================================================
# MAIN
# ============================================================

def main():

    AUDIT_DATA["started"] = datetime.now().isoformat(timespec="milliseconds")

    section("E3 PIN VIEW / SPLIT CONNECTOR BEHAVIOR AUDIT")

    log(f"CONNECTOR_NAME = {CONNECTOR_NAME}")
    log(f"FIXED_X = {FIXED_X}")
    log(f"Y_MIN = {Y_MIN}")
    log(f"Y_MAX = {Y_MAX}")
    log(f"Y_STEP = {Y_STEP}")
    log(f"TOP_IS_HIGH_Y = {TOP_IS_HIGH_Y}")

    app, prj, dev, pin, sym = connect_to_e3()

    connector_device_id = find_connector_device_id(prj, dev)

    if connector_device_id is None:
        raise RuntimeError(f"Could not find connector {CONNECTOR_NAME}.")

    audit_key_method_signatures(pin, sym)

    baseline = build_snapshot(prj, dev, pin, sym, connector_device_id)

    print_snapshot(baseline, "INITIAL BASELINE SNAPSHOT")

    add_event("baseline", baseline)

    while True:

        print_menu()

        choice = input("Select test: ").strip().lower()

        if choice == "1":

            snapshot = build_snapshot(prj, dev, pin, sym, connector_device_id)

            print_snapshot(snapshot, "FULL CONNECTOR SNAPSHOT")

            add_event("manual_snapshot", snapshot)

        elif choice == "2":

            run_manual_change_compare(prj, dev, pin, sym, connector_device_id)

        elif choice == "3":

            run_manual_top_then_script(prj, dev, pin, sym, connector_device_id)

        elif choice == "4":

            run_manual_bottom_then_script(prj, dev, pin, sym, connector_device_id)

        elif choice == "5":

            snapshot = build_snapshot(prj, dev, pin, sym, connector_device_id)

            top = get_top_candidate(snapshot)

            default_name = top["pin_name"] if top else ""

            pin_name = input(f"Pin name to test [default {default_name}]: ").strip()

            if pin_name == "":
                pin_name = default_name

            run_symbol_place_test(prj, dev, pin, sym, connector_device_id, pin_name)

        elif choice == "6":

            run_place_pin_view_signature_audit(sym)

        elif choice == "7":

            run_place_pins_signature_audit(sym)

        elif choice == "8":

            run_experimental_place_pin_view_test(prj, dev, pin, sym, connector_device_id)

        elif choice == "9":

            run_repeated_script_move_test(prj, dev, pin, sym, connector_device_id)

        elif choice == "10":

            run_raw_relationship_call_audit(prj, dev, pin, sym, connector_device_id)

        elif choice == "11":

            audit_key_method_signatures(pin, sym)

        elif choice == "q":

            log("USER REQUESTED QUIT")

            break

        else:

            log("INVALID MENU SELECTION")


if __name__ == "__main__":

    try:
        main()

    except KeyboardInterrupt:
        log()
        log("AUDIT STOPPED BY USER")

    except Exception as exc:
        log()
        log("FATAL ERROR")
        log(safe_string(exc))
        log(traceback.format_exc())

    finally:

        try:
            save_results()
        except Exception as exc:
            print(f"COULD NOT SAVE AUDIT FILES: {exc}")

        try:
            pythoncom.CoUninitialize()
        except Exception:
            pass
