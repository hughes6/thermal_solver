def generate_master_schematic_layout(
    PIN_SIGNAL_INFO,
    x_block,
    main_block_width
):
    # ================================================
    # PAGE BOUNDS
    # ================================================
    TOP_OF_PAGE = 32.9
    BOTTOM_OF_PAGE = 0.0
    LEFT_OF_PAGE = 0.0
    RIGHT_OF_PAGE = 42.9

    # ================================================
    # PAGE MARGINS
    # ================================================
    TOP_MARGIN = 1.0
    BOTTOM_MARGIN = 1.0
    LEFT_MARGIN = 1.0
    RIGHT_MARGIN = 1.0

    # ================================================
    # BLOCK SPACING
    # ================================================
    BLOCK_X_SPACING = 1.0
    BLOCK_Y_SPACING = 1.0

    # ================================================
    # SCHEMATIC BLOCK SIZE
    # ================================================
    SCHEMATIC_BLOCK_WIDTH = 3.0

    PIN_SPACING = 0.25
    PIN_START_OFFSET = 2.0

    # ================================================
    # USABLE PAGE LIMITS
    # ================================================
    PAGE_TOP = TOP_OF_PAGE - TOP_MARGIN
    PAGE_BOTTOM = BOTTOM_OF_PAGE + BOTTOM_MARGIN
    PAGE_LEFT = LEFT_OF_PAGE + LEFT_MARGIN
    PAGE_RIGHT = RIGHT_OF_PAGE - RIGHT_MARGIN

    USABLE_PAGE_HEIGHT = PAGE_TOP - PAGE_BOTTOM

    # ================================================
    # OUTPUT
    # ================================================
    SPACING_DICT = []

    # Start on sheet 1
    sheet = 1

    # On sheet 1, start to the RIGHT of the main block
    current_x = (
        x_block
        + main_block_width
        + BLOCK_X_SPACING
    )

    # Current Y is the TOP edge available for placement
    current_top_y = PAGE_TOP

    # Tracks widest block in current column
    current_column_width = 0.0

    # ================================================
    # BUILD LAYOUT
    # ================================================
    for i in range(len(PIN_SIGNAL_INFO)):

        connector_name = (
            PIN_SIGNAL_INFO[i][0]["connector name"]
        )

        number_of_pins = (
            PIN_SIGNAL_INFO[i][0]["number of pins"]
        )

        width = SCHEMATIC_BLOCK_WIDTH

        height = (
            PIN_START_OFFSET
            + (PIN_SPACING * number_of_pins)
        )

        # --------------------------------------------
        # SAFETY CHECK:
        # Block cannot physically fit on ANY page
        # --------------------------------------------
        if height > USABLE_PAGE_HEIGHT:

            raise Exception(
                f"Schematic block for {connector_name} "
                f"is too tall for a page. "
                f"Height = {height}, "
                f"usable page height = {USABLE_PAGE_HEIGHT}"
            )

        # --------------------------------------------
        # CALCULATE BOTTOM Y IF PLACED HERE
        # --------------------------------------------
        candidate_y = current_top_y - height

        # ============================================
        # DOES IT FIT VERTICALLY?
        # ============================================
        if candidate_y < PAGE_BOTTOM:

            # No vertical room.
            # Try next column.
            next_x = (
                current_x
                + current_column_width
                + BLOCK_X_SPACING
            )

            # ========================================
            # DOES NEXT COLUMN FIT ON THIS PAGE?
            # ========================================
            if next_x + width <= PAGE_RIGHT:

                current_x = next_x
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            else:

                # ====================================
                # PAGE IS FULL -> NEW PAGE
                # ====================================
                sheet += 1

                current_x = PAGE_LEFT
                current_top_y = PAGE_TOP
                current_column_width = 0.0

            # Recalculate Y at new column/page
            candidate_y = current_top_y - height

        # ============================================
        # CHECK HORIZONTAL FIT
        # ============================================
        if current_x + width > PAGE_RIGHT:

            # Start a new sheet
            sheet += 1

            current_x = PAGE_LEFT
            current_top_y = PAGE_TOP
            current_column_width = 0.0

            candidate_y = current_top_y - height

        # ============================================
        # SAVE PLACEMENT
        # ============================================
        x_start = current_x
        y_start = candidate_y

        SPACING_DICT.append({
            "connector name": connector_name,
            "connector index": i,

            "sheet": sheet,

            "x start": x_start,
            "y start": y_start,

            "x end": x_start + width,
            "y end": y_start + height,

            "height": height,
            "width": width,

            "number of pins": number_of_pins
        })

        # ============================================
        # UPDATE CURSOR FOR NEXT BLOCK
        # ============================================
        current_top_y = (
            y_start
            - BLOCK_Y_SPACING
        )

        current_column_width = max(
            current_column_width,
            width
        )

    return SPACING_DICT
