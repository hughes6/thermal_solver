import win32com.client as win32


# ================================================
# CONNECT TO ACTIVE E3 PROJECT
# ================================================
app = win32.gencache.EnsureDispatch("CT.Application")
prj = app.CreateJobObject()
dev = prj.CreateDeviceObject()
pin = prj.CreatePinObject()
sym = prj.CreateSymbolObject()
text = prj.CreateTextObject()


# ================================================
# HELPERS
# ================================================
def extract_ids(result):
    if result is None:
        return []

    if isinstance(result, (tuple, list)):
        if len(result) >= 2:
            values = result[1]

            if isinstance(values, (tuple, list)):
                return [x for x in values if x is not None]

        return [x for x in result if isinstance(x, int) and x != 0]

    if isinstance(result, int):
        if result == 0:
            return []

        return [result]

    return []


def get_pin_view_ids(pin_id):
    pin.SetId(pin_id)

    try:
        result = pin.GetViewIds(None)
        return extract_ids(result)
    except Exception:
        pass

    try:
        result = pin.GetViewIds()
        return extract_ids(result)
    except Exception as e:
        print("GetViewIds ERROR FOR PIN", pin_id, ":", repr(e))
        return []


def print_text_object(text_id, prefix):
    text.SetId(text_id)

    try:
        text_value = text.GetText()
    except Exception:
        text_value = "<ERROR>"

    try:
        text_type = text.GetType()
    except Exception:
        text_type = "<ERROR>"

    print(prefix, "TEXT ID:", text_id, "TYPE:", text_type, "VALUE:", repr(text_value))


# ================================================
# GET LOGICAL DEVICES
# ================================================
device_result = prj.GetAllDeviceIds(None)
device_ids = extract_ids(device_result)

print("")
print("============================================================")
print("LOGICAL DEVICES")
print("============================================================")
print("DEVICE IDS:", device_ids)


# ================================================
# SAVE EVERY PIN VIEW WE DISCOVER
# ================================================
ALL_PIN_VIEW_IDS = set()
PIN_VIEW_OWNER = {}


# ================================================
# WALK LOGICAL DEVICES
# ================================================
for device_id in device_ids:
    dev.SetId(device_id)

    try:
        device_name = dev.GetName()
    except Exception:
        device_name = "<ERROR>"

    print("")
    print("############################################################")
    print("DEVICE ID:", device_id)
    print("DEVICE NAME:", repr(device_name))
    print("############################################################")

    # ================================================
    # DEVICE SYMBOLS
    # ================================================
    try:
        symbol_result = dev.GetSymbolIds(None)
        symbol_ids = extract_ids(symbol_result)
    except Exception as e:
        symbol_ids = []
        print("GetSymbolIds ERROR:", repr(e))

    print("DEVICE SYMBOL IDS:", symbol_ids)

    for symbol_id in symbol_ids:
        sym.SetId(symbol_id)

        try:
            symbol_name = sym.GetName()
        except Exception:
            symbol_name = "<ERROR>"

        print("")
        print("    SYMBOL ID:", symbol_id)
        print("    SYMBOL NAME:", repr(symbol_name))

        try:
            text_result = sym.GetTextIds(None)
            text_ids = extract_ids(text_result)
        except Exception as e:
            text_ids = []
            print("    SYMBOL GetTextIds ERROR:", repr(e))

        for text_id in text_ids:
            print_text_object(text_id, "        SYMBOL")

    # ================================================
    # LOGICAL DEVICE PINS
    # ================================================
    try:
        pin_result = dev.GetPinIds(None)
        pin_ids = extract_ids(pin_result)
    except Exception as e:
        pin_ids = []
        print("GetPinIds ERROR:", repr(e))

    print("")
    print("LOGICAL PIN IDS:", pin_ids)

    for pin_id in pin_ids:
        pin.SetId(pin_id)

        try:
            pin_name = pin.GetName()
        except Exception:
            pin_name = "<ERROR>"

        print("")
        print("    --------------------------------------------------------")
        print("    LOGICAL PIN ID:", pin_id)
        print("    LOGICAL PIN NAME:", repr(pin_name))

        # ================================================
        # NORMAL PIN TEXT
        # ================================================
        try:
            pin_text_result = pin.GetTextIds(None)
            pin_text_ids = extract_ids(pin_text_result)
        except Exception as e:
            pin_text_ids = []
            print("    GetTextIds ERROR:", repr(e))

        print("    LOGICAL PIN TEXT IDS:", pin_text_ids)

        for text_id in pin_text_ids:
            print_text_object(text_id, "        LOGICAL PIN")

        # ================================================
        # THIS IS THE IMPORTANT PART
        # FIND ALL VIEWS OF THIS PIN
        # ================================================
        pin_view_ids = get_pin_view_ids(pin_id)

        print("    PIN VIEW IDS:", pin_view_ids)

        for pin_view_id in pin_view_ids:
            ALL_PIN_VIEW_IDS.add(pin_view_id)

            if pin_view_id not in PIN_VIEW_OWNER:
                PIN_VIEW_OWNER[pin_view_id] = []

            PIN_VIEW_OWNER[pin_view_id].append({
                "device_id": device_id,
                "device_name": device_name,
                "pin_id": pin_id,
                "pin_name": pin_name
            })


# ================================================
# NOW INSPECT EVERY PIN VIEW DIRECTLY
# ================================================
print("")
print("")
print("============================================================")
print("PIN VIEW OBJECTS")
print("============================================================")
print("ALL UNIQUE PIN VIEW IDS:", sorted(ALL_PIN_VIEW_IDS))


for pin_view_id in sorted(ALL_PIN_VIEW_IDS):
    print("")
    print("############################################################")
    print("PIN VIEW ID:", pin_view_id)
    print("OWNERS:", PIN_VIEW_OWNER.get(pin_view_id))
    print("############################################################")

    # ================================================
    # TRY THE ID AS A PIN OBJECT
    # ================================================
    print("")
    print("TRYING AS PIN OBJECT")

    try:
        pin.SetId(pin_view_id)

        try:
            pin_name = pin.GetName()
        except Exception:
            pin_name = "<ERROR>"

        print("PIN VIEW NAME:", repr(pin_name))

        try:
            pin_text_result = pin.GetTextIds(None)
            pin_text_ids = extract_ids(pin_text_result)
        except Exception as e:
            pin_text_ids = []
            print("PIN VIEW GetTextIds ERROR:", repr(e))

        print("PIN VIEW TEXT IDS:", pin_text_ids)

        for text_id in pin_text_ids:
            print_text_object(text_id, "    PIN VIEW")

    except Exception as e:
        print("FAILED AS PIN OBJECT:", repr(e))

    # ================================================
    # ALSO TRY THE SAME ID AS A DEVICE VIEW
    # ================================================
    print("")
    print("TRYING AS DEVICE OBJECT")

    try:
        dev.SetId(pin_view_id)

        try:
            view_name = dev.GetName()
        except Exception:
            view_name = "<ERROR>"

        print("DEVICE NAME:", repr(view_name))

        try:
            is_view = dev.IsView()
        except Exception as e:
            is_view = "<ERROR>"

        print("IS VIEW:", is_view)

        try:
            original_id = dev.GetOriginalId()
        except Exception:
            original_id = "<ERROR>"

        print("ORIGINAL DEVICE ID:", original_id)

        try:
            symbol_result = dev.GetSymbolIds(None)
            symbol_ids = extract_ids(symbol_result)
        except Exception as e:
            symbol_ids = []
            print("DEVICE VIEW GetSymbolIds ERROR:", repr(e))

        print("DEVICE VIEW SYMBOL IDS:", symbol_ids)

        for symbol_id in symbol_ids:
            sym.SetId(symbol_id)

            try:
                symbol_name = sym.GetName()
            except Exception:
                symbol_name = "<ERROR>"

            print("")
            print("    DEVICE VIEW SYMBOL:", symbol_id)
            print("    SYMBOL NAME:", repr(symbol_name))

            try:
                text_result = sym.GetTextIds(None)
                text_ids = extract_ids(text_result)
            except Exception as e:
                text_ids = []
                print("    SYMBOL GetTextIds ERROR:", repr(e))

            for text_id in text_ids:
                print_text_object(text_id, "        DEVICE VIEW SYMBOL")

        try:
            pin_result = dev.GetPinIds(None)
            view_pin_ids = extract_ids(pin_result)
        except Exception as e:
            view_pin_ids = []
            print("DEVICE VIEW GetPinIds ERROR:", repr(e))

        print("DEVICE VIEW PIN IDS:", view_pin_ids)

        for view_pin_id in view_pin_ids:
            pin.SetId(view_pin_id)

            try:
                view_pin_name = pin.GetName()
            except Exception:
                view_pin_name = "<ERROR>"

            print("")
            print("    DEVICE VIEW PIN:", view_pin_id)
            print("    PIN NAME:", repr(view_pin_name))

            try:
                pin_text_result = pin.GetTextIds(None)
                pin_text_ids = extract_ids(pin_text_result)
            except Exception as e:
                pin_text_ids = []
                print("    PIN GetTextIds ERROR:", repr(e))

            for text_id in pin_text_ids:
                print_text_object(text_id, "        DEVICE VIEW PIN")

    except Exception as e:
        print("FAILED AS DEVICE OBJECT:", repr(e))


print("")
print("============================================================")
print("SEARCH COMPLETE")
print("LOOK FOR TYPE 617 IN THE OUTPUT")
print("============================================================")
