name = "Cable Shield"
ver = 1

# ============================================================
# LOAD SYMBOL - DO NOT PLACE YET
# ============================================================

res = sym.Load(name, ver)

print("LOAD:", res)

if res == 0:
    raise Exception("Could not load Cable Shield")


# ============================================================
# GET PINS FROM LOADED SYMBOL
# ============================================================

pin_info = sym.GetPinIds(None)

print("PIN INFO:", pin_info)

pin_ids = [
    x for x in pin_info[1]
    if x is not None
]

print("PIN IDS:", pin_ids)


# ============================================================
# INSPECT PIN LOCATIONS BEFORE PLACEMENT
# ============================================================

for pin_id in pin_ids:

    pin.SetId(pin_id)

    xloc = [0.0]
    yloc = [0.0]
    grid = [""]
    column = [""]
    row = [""]

    result = pin.GetSchemaLocation(
        xloc,
        yloc,
        grid,
        column,
        row
    )

    print(
        "PIN:",
        pin_id,
        "NAME:",
        pin.GetName(),
        "X:",
        result[1],
        "Y:",
        result[2]
    )
