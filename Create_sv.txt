cur.execute("""
SELECT
    ENTRY,
    VERSION,
    Description,
    LPNTR
FROM dbo.SymbolData
WHERE ENTRY LIKE ?
ORDER BY ENTRY, VERSION
""", "%MilSymbol%")

for row in cur.fetchall():
    print(tuple(row))

cur.execute("""
SELECT
    ENTRY,
    VERSION,
    Description,
    LPNTR
FROM dbo.SymbolData
WHERE Description LIKE ?
ORDER BY ENTRY, VERSION
""", "%Recept%")

for row in cur.fetchall():
    print(tuple(row))

cur.execute("""
SELECT
    ENTRY,
    VERSION,
    Description
FROM dbo.SymbolData
WHERE ENTRY LIKE '%MIL%'
   OR ENTRY LIKE '%REC%'
   OR Description LIKE '%MIL%'
   OR Description LIKE '%REC%'
ORDER BY ENTRY
""")

for row in cur.fetchall():
    print(tuple(row))
