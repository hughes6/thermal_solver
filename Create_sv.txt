def format_connector_text(connector_view_id, conn_name):
    print("FORMATTING CONNECTOR:", conn_name, "VIEW:", connector_view_id)

    dev.SetId(connector_view_id)

    #=========================================
    # FORMAT CONNECTOR DEVICE TEXT
    # TYPE 12 = CONNECTOR DEVICE DESIGNATION
    #=========================================
    symbolinfo = dev.GetSymbolIds(None)
    symbolids = [x for x in symbolinfo[1] if x is not None]

    for symbol_id in symbolids:
        sym.SetId(symbol_id)

        textinfo = sym.GetTextIds(None)
        textids = [x for x in textinfo[1] if x is not None]

        for text_id in textids:
            text.SetId(text_id)
            text_type = text.GetType()

            if text_type == 12:
                text.SetVisibility(1)
                text.SetColour(0)
                text.SetStyle(1)
                text.SetHeight(0.2)
                print("BOLDED CONNECTOR NAME:", text.GetText(), "TYPE:", text_type)

    #=========================================
    # FORMAT PIN TEXT
    # TYPE 617 = LOCAL SIGNAL NAME
    #=========================================
    pininfo = dev.GetPinIds(None)
    pinids = [x for x in pininfo[1] if x is not None]

    for pin_id in pinids:
        pin.SetId(pin_id)

        pintextinfo = pin.GetTextIds(None)
        pintextids = [x for x in pintextinfo[1] if x is not None]

        for pin_text_id in pintextids:
            text.SetId(pin_text_id)
            text_type = text.GetType()

            if text_type == 617:
                text.SetVisibility(1)
                text.SetColour(0)
                text.SetAlignment(3)
                text.SetStyle(1)
                text.SetHeight(0.2)
                print("LOCAL SIGNAL:", conn_name, "PIN:", pin.GetName(), "TEXT:", text.GetText())

            else:
                text.SetVisibility(0)
