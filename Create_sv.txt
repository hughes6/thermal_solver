import tkinter as tk
from tkinter import scrolledtext


def convert():
    # Get text from input box
    input_text = input_box.get("1.0", tk.END)

    # Split on any whitespace: spaces, tabs, newlines, etc.
    values = input_text.split()
    values = [x for x in values if x.isdigit()]

    # Put each value on its own line
    output_text = "\n".join(values)

    # Clear old output
    output_box.delete("1.0", tk.END)

    # Insert converted output
    output_box.insert(tk.END, output_text)

    # Update count label
    count_label.config(text=f"Items: {len(values)}")

    # Store count for connector copy default
    item_count_var.set(len(values))


def copy_output():
    output_text = output_box.get("1.0", tk.END).strip()

    root.clipboard_clear()
    root.clipboard_append(output_text)
    root.update()


def clear_all():
    input_box.delete("1.0", tk.END)
    output_box.delete("1.0", tk.END)

    count_label.config(text="Items: 0")
    item_count_var.set(0)


def copy_connector():
    connector_number = connector_entry.get().strip()
    copies_text = copies_entry.get().strip()

    if connector_number == "":
        connector_status.config(text="Enter a connector number")
        return

    # If copies box is blank, use number of items from main formatter
    if copies_text == "":
        copies = item_count_var.get()
    else:
        try:
            copies = int(copies_text)
        except ValueError:
            connector_status.config(text="Copies must be a number")
            return

    if copies <= 0:
        connector_status.config(text="No items to copy")
        return

    # Make Excel-ready vertical output
    connector_output = "\n".join(
        [connector_number] * copies
    )

    root.clipboard_clear()
    root.clipboard_append(connector_output)
    root.update()

    connector_status.config(
        text=f"Copied {copies} copies of connector {connector_number}"
    )


# =========================================
# GUI
# =========================================

root = tk.Tk()
root.title("Excel Column Formatter")
root.geometry("600x700")


# Tracks number of items in normal output
item_count_var = tk.IntVar(value=0)


# =========================================
# MAIN FORMATTER
# =========================================

tk.Label(
    root,
    text="Paste numbers here:",
    font=("Arial", 12, "bold")

    height=6,

).pack(pady=10)

    text="Excel-ready output:",
    font=("Arial", 12, "bold")
).pack(pady=(5, 5))

    root,
    width=65,
    height=15,
    font=("Consolas", 11)
)

count_label = tk.Label(
    root,
    text="Items: 0",
    font=("Arial", 12, "bold"),
    fg="blue"
)
count_label.pack(pady=(5, 10))


button_frame = tk.Frame(root)
button_frame.pack(pady=12)


tk.Button(
    button_frame,
    text="Copy to Clipboard",
    command=copy_output,
    width=20
).pack(side=tk.LEFT, padx=5)


tk.Button(
    button_frame,
    text="Clear",
    command=clear_all,
    width=12
).pack(side=tk.LEFT, padx=5)


# =========================================
# CONNECTOR COPY SECTION

connector_frame = tk.LabelFrame(

    row=0,

    pady=5
)


# Optional number of copies
tk.Label(
    connector_frame,
    text="Copies (optional):"
).grid(
    row=0,
    column=2,
    padx=5,
    pady=5,
    sticky="e"
)

    connector_frame,

# Copy button
