""", component_id)

for element_id, copy_no in elements:

    print("ELEMENT ID:", element_id)

    cur.execute("""
    SELECT *
    FROM dbo.Cable
    WHERE ParentID = ?
       OR ENTRY = ?
    """, element_id, element_id)

    rows = cur.fetchall()

    print("Cable matches:", len(rows))

    if rows:
        columns = [col[0] for col in cur.description]

        for row in rows:

                print(f"{name} = {value}")
