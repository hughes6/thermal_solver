table = "WIRTYPE_CONNECT_INFOS"

cur.execute("""
SELECT COLUMN_NAME
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA='dbo'
  AND TABLE_NAME=?
ORDER BY ORDINAL_POSITION
""", table)

cols = [r[0] for r in cur.fetchall()]

print("COLUMNS:")
for c in cols:
    print(repr(c))

print("\nFIRST 30 ROWS:")
cur.execute(f"SELECT TOP (30) * FROM dbo.[{table}]")

for row in cur.fetchall():
    print(row)
