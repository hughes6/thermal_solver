import win32com.client as win32
from win32com.client import VARIANT
import pythoncom
import inspect


app = win32.gencache.EnsureDispatch(
    "CT.DBEApplication"
)

cid = app.GetActiveComponentId()

comp = app.CreateDbeComponentObject()
comp.SetId(cid)


print("=" * 90)
print("BOUND COMPONENT")
print("=" * 90)

print("ID:      ", comp.GetId())
print("Name:    ", comp.GetName())
print("Version: ", comp.GetVersion())
print("Type:    ", comp.GetComponentType())
print("Subtype: ", comp.GetSubType())


# ============================================================
# 1. LOOK AT GENERATED PYWIN32 METHOD SOURCE
# ============================================================

print()
print("=" * 90)
print("GENERATED WRAPPER SOURCE")
print("=" * 90)


for name, method in [
    ("GetAttributeIds", comp.GetAttributeIds),
    ("AddAttributeValue", comp.AddAttributeValue),
    ("GetComponentIds", app.GetComponentIds),
    ("NewComponent", app.NewComponent),
]:

    print()
    print("-" * 90)
    print(name)
    print("-" * 90)

    try:
        print(inspect.getsource(method))
    except Exception as e:
        print("getsource failed:", repr(e))

    try:
        print("module:", method.__module__)
    except Exception:
        pass


# ============================================================
# 2. TRY GetAttributeIds WITH EXPLICIT VARIANTS
# ============================================================

print()
print("=" * 90)
print("ATTRIBUTE ARRAY TESTS")
print("=" * 90)


tests = []


# Variant containing an empty Python list
try:
    v = VARIANT(
        pythoncom.VT_VARIANT | pythoncom.VT_BYREF,
        None
    )

    tests.append(
        (
            "BYREF VARIANT",
            lambda: comp.GetAttributeIds(
                v,
                0,
                ""
            ),
            v
        )
    )
except Exception as e:
    print("Could not make BYREF VARIANT:", repr(e))


# SAFEARRAY of LONG by reference
try:
    v_int_array = VARIANT(
        pythoncom.VT_ARRAY
        | pythoncom.VT_I4
        | pythoncom.VT_BYREF,
        []
    )

    tests.append(
        (
            "BYREF LONG ARRAY",
            lambda: comp.GetAttributeIds(
                v_int_array,
                0,
                ""
            ),
            v_int_array
        )
    )
except Exception as e:
    print("Could not make LONG ARRAY:", repr(e))


# SAFEARRAY of VARIANT by reference
try:
    v_variant_array = VARIANT(
        pythoncom.VT_ARRAY
        | pythoncom.VT_VARIANT
        | pythoncom.VT_BYREF,
        []
    )

    tests.append(
        (
            "BYREF VARIANT ARRAY",
            lambda: comp.GetAttributeIds(
                v_variant_array,
                0,
                ""
            ),
            v_variant_array
        )
    )
except Exception as e:
    print("Could not make VARIANT ARRAY:", repr(e))


for label, function, holder in tests:

    print()
    print("---", label, "---")

    try:
        result = function()

        print("RETURN:", repr(result))
        print("HOLDER VALUE:", repr(holder.value))

    except Exception as e:
        print("FAILED:", repr(e))


# ============================================================
