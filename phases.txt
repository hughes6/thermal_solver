import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

# ------------------------------------------------------------
# GET THE OPEN DBE JOB
# ------------------------------------------------------------

result = app.GetJobIds()
print("RAW JOB IDS:", result)

# From your current result:
# (1, (None, 118))
job_id = result[1][1]

job = app.CreateDbeJobObject()
job.SetId(job_id)

print("\nBOUND JOB")
print("ID:   ", job.GetId())
print("Name: ", job.GetName())
print("Type: ", job.GetType())
print("Path: ", job.GetPath())


# ------------------------------------------------------------
# SEE WHAT COMPONENT THE JOB THINKS IS ACTIVE
# ------------------------------------------------------------

print("\nACTIVE COMPONENT")

try:
    cid = job.GetActiveComponentId()
    print("job.GetActiveComponentId():", cid)
except Exception as e:
    print("GetActiveComponentId failed:", repr(e))
    cid = None


# ------------------------------------------------------------
# CREATE COMPONENT OBJECT THROUGH THE JOB
# ------------------------------------------------------------

print("\nCOMPONENT CREATED THROUGH JOB")

comp = job.CreateDbeComponentObject()

print("Raw ID:      ", comp.GetId())
print("Raw Name:    ", repr(comp.GetName()))
print("Raw Version: ", repr(comp.GetVersion()))
print("Raw Type:    ", comp.GetComponentType())
print("Raw Subtype: ", comp.GetSubType())


# If job actually gives us an ID, bind explicitly
if isinstance(cid, int) and cid > 0:

    print("\nBinding component to job active ID:", cid)

    comp.SetId(cid)

    print("ID:      ", comp.GetId())
    print("Name:    ", repr(comp.GetName()))
    print("Version: ", repr(comp.GetVersion()))
    print("Type:    ", comp.GetComponentType())
    print("Subtype: ", comp.GetSubType())


# ------------------------------------------------------------
# JOB COMPONENT IDS
# ------------------------------------------------------------

print("\nJOB COMPONENT IDS")

try:
    r = job.GetComponentIds()
    print("GetComponentIds():", repr(r))
except Exception as e:
    print("GetComponentIds failed:", repr(e))


# ------------------------------------------------------------
# OUTPUT/RESULT WINDOW
# ------------------------------------------------------------

print("\nJOB OUTPUT")

for i in range(20):

    try:
        r = job.GetResultText(i, "")
        if r not in (None, "", -1, 0):
            print("RESULT", i, repr(r))
    except:
        pass

    try:
        r = job.GetOutbarText(i, "")
        if r not in (None, "", -1, 0):
            print("OUTBAR", i, repr(r))
    except:
        pass
