import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

module = __import__(
    app.__class__.__module__,
    fromlist=["*"]
)

for name, value in vars(module).items():
    text = name.lower()

    if any(x in text for x in [
        "dbe",
        "cable",
        "wire",
        "component",
        "connector"
    ]):
        if isinstance(value, (int, str)):
            print(name, "=", value)
