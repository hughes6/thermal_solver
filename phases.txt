import os

root = app.GetInstallationPath()

terms = [
    b"WINRTYP_CONNECT_INFOS",
    b"PINCOPYEND1",
    b"COPYEND1",
    b"GatePinRelation",
]

for base, dirs, files in os.walk(root):

    for file in files:

        path = os.path.join(base, file)

        try:
            with open(path, "rb") as f:
                data = f.read()

            for term in terms:
                if term.lower() in data.lower():
                    print(term.decode(), "->", path)

        except:
            pass
