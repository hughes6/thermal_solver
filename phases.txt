    "ComponentAttribute",
    "ComponentData",
    "GatePinRelation",
]

cur = conn.cursor()

for table in tables:
    print("\n" + "=" * 100)
    print(table)
    print("=" * 100)

    # --------------------------------------------------
    # GET COLUMNS
    # --------------------------------------------------

    cur.execute("""
        SELECT
            COLUMN_NAME,
            DATA_TYPE
        FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_SCHEMA = 'dbo'
          AND TABLE_NAME = ?
        ORDER BY ORDINAL_POSITION
    """, table)

    columns = cur.fetchall()

    print("\nCOLUMNS:")

    for col, dtype in columns:
        print(f"{col:40} {dtype}")

    # --------------------------------------------------
    # FIND IDENTITY COLUMN
    # --------------------------------------------------

    cur.execute("""
        SELECT c.name
        FROM sys.columns c
        JOIN sys.tables t
            ON c.object_id = t.object_id
        JOIN sys.schemas s
            ON t.schema_id = s.schema_id
        WHERE s.name = 'dbo'
          AND t.name = ?
          AND c.is_identity = 1
    """, table)

    row = cur.fetchone()

    identity_col = row[0] if row else None

    # --------------------------------------------------
    # FIND PRIMARY KEY IF NO IDENTITY
    # --------------------------------------------------

    if identity_col is None:

        cur.execute("""
            SELECT TOP 1 c.name
            FROM sys.indexes i
            JOIN sys.index_columns ic
                ON i.object_id = ic.object_id
               AND i.index_id = ic.index_id
            JOIN sys.columns c
                ON ic.object_id = c.object_id
               AND ic.column_id = c.column_id
            JOIN sys.tables t
                ON i.object_id = t.object_id
            JOIN sys.schemas s
                ON t.schema_id = s.schema_id
            WHERE s.name = 'dbo'
              AND t.name = ?
              AND i.is_primary_key = 1
            ORDER BY ic.key_ordinal
        """, table)

        row = cur.fetchone()

        if row:
            identity_col = row[0]

    print("\nORDER COLUMN:", identity_col)

    # --------------------------------------------------
    # PRINT LATEST ROWS
    # --------------------------------------------------

    if identity_col:

        sql = f"""
            SELECT TOP (15) *
            FROM dbo.[{table}]
            ORDER BY [{identity_col}] DESC
        """

    else:

        sql = f"""
            SELECT TOP (15) *
            FROM dbo.[{table}]
        """

    cur.execute(sql)

    print("\nROWS:")

    for row in cur.fetchall():
        print(row)
