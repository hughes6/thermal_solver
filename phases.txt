# ============================================================
# SAFE CALL HELPER
# ============================================================

def call(label, func, *args):

    try:
        result = func(*args)

        print(
            f"[OK] {label}:",
            repr(result)
        )

        return result

    except Exception as exc:

        print(
            f"[FAIL] {label}:",
            repr(exc)
        )

        return None


# ============================================================
# GET ACTIVE COMPONENT
# ============================================================

def get_active_component():

    print("\n--- ACTIVE COMPONENT ---")

    active_id = call(
        "GetActiveComponentId",
        app.GetActiveComponentId
    )

    comp = app.CreateDbeComponentObject()

    values = {}

    values["id"] = call(
        "component.GetId",
        comp.GetId
    )

    values["name"] = call(
        "component.GetName",
        comp.GetName
    )

    values["version"] = call(
        "component.GetVersion",
        comp.GetVersion
    )

    values["type"] = call(
        "component.GetComponentType",
        comp.GetComponentType
    )

    values["subtype"] = call(
        "component.GetSubType",
        comp.GetSubType
    )

    values["model"] = call(
        "component.GetModelName(0)",
        comp.GetModelName,
        0
    )

    print("\nSummary:")

    for key, value in values.items():
        print(
            f"    {key:10}: {value}"
        )

    return comp, values

# ============================================================
# COMPONENT ATTRIBUTES
# ============================================================

def get_attribute_ids(comp, end, attribute_name=""):

    print()
    print(
        f"--- ATTRIBUTES: END={end} ---"
    )

    # COM output parameters often do not need to be supplied
    # to pywin32. Try the most likely signatures.

    attempts = [

        (
            "GetAttributeIds(end, name)",
            (end, attribute_name)
        ),

        (
            "GetAttributeIds(None, end, name)",
            (None, end, attribute_name)
        ),

        (
            "GetAttributeIds([], end, name)",
            ([], end, attribute_name)
        ),

    ]

    for label, args in attempts:

        try:

            result = comp.GetAttributeIds(
                *args
            )

            print(
                "[SUCCESS]",
                label,
                "=>",
                repr(result)
            )

            return result

        except Exception as exc:

            print(
                "[FAILED]",
                label,
                "=>",
                repr(exc)
            )

    return None


# ============================================================
# NORMALIZE ID RESULT
# ============================================================

def normalize_ids(result):

    if result is None:
        return []

    # Common E3 return:
    #
    #   (count, (id1,id2,id3,...))

    if (
        isinstance(result, tuple)
        and len(result) == 2
        and isinstance(result[1], (tuple, list))
    ):
        return list(result[1])

    if isinstance(
        result,
        (tuple, list)
    ):
        return list(result)

    return [result]


# ============================================================
# READ ATTRIBUTE
# ============================================================

def inspect_attribute(attribute_id):

    attr = app.CreateDbeAttributeObject()

    try:
        attr.SetId(attribute_id)
    except Exception as exc:

        print(
            f"Cannot SetId({attribute_id}):",
            repr(exc)
        )

        return None

    data = {
        "id": attribute_id
    }

    getters = {

        "name":
            attr.GetName,

        "internal_name":
            attr.GetInternalName,

        "value":
            attr.GetValue,

        "internal_value":
            attr.GetInternalValue,

        "formatted_value":
            attr.GetFormattedValue,
    }

    for key, getter in getters.items():

        try:
            data[key] = getter()

        except Exception as exc:
            data[key] = (
                "<ERROR "
                + repr(exc)
                + ">"
            )

    print(json.dumps(
        data,
        indent=4,
        default=str
    ))

    return data


# ============================================================
# INSPECT CABLE ENDS
# ============================================================

def inspect_cable_ends(comp):

    print()
    print("=" * 80)
    print("CABLE END INVESTIGATION")
    print("=" * 80)

    # "end" is very interesting because the DBE component
    # API explicitly includes it in:
    #
    # GetAttributeIds(ids, end, attnam)
    # AddAttributeValue(name,value,end)
    #
    # Probe plausible values READ-ONLY.

    for end in [
        -1,
        0,
        1,
        2
    ]:

        result = get_attribute_ids(
            comp,
            end,
            ""
        )

        ids = normalize_ids(result)

        print(
            f"\nEND {end}: "
            f"{len(ids)} attribute IDs"
        )

        for attribute_id in ids:

            inspect_attribute(
                attribute_id
            )
