import win32com.client as win32

for progid in [
    "CT.Application",
    "CT.E3Application",
]:
    try:
        e3 = win32.gencache.EnsureDispatch(progid)
        print("WORKED:", progid)
        print("Name:", e3.GetName())
        print("Version:", e3.GetVersion())

        for name in dir(e3):
            if any(x in name.lower() for x in [
                "device",
                "pin",
                "connection",
                "component"
            ]):
                print(name)

        break

    except Exception as e:
        print("FAILED:", progid, repr(e))
