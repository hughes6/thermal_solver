lpntr = "LC-14840D29-FF00-4442-B1BE-7C93C4894335"

tables = [
    "Cable",
    "COMPONENT_PINS",
    "Pin",
    "WIRTYPE_CONNECT_INFOS",
]

for table in tables:

    print("\n" + "=" * 90)
    print(table)
    print("=" * 90)

    cur.execute("""
        SELECT COLUMN_NAME
        FROM INFORMATION_SCHEMA.COLUMNS
        WHERE TABLE_SCHEMA='dbo'
          AND TABLE_NAME=?
        ORDER BY ORDINAL_POSITION
    """, table)

    cols = [r[0] for r in cur.fetchall()]
    print("COLUMNS:", cols)

    # Only directly filter tables that actually contain LPNTR
    if "LPNTR" in cols:

        cur.execute(
            f"SELECT * FROM dbo.[{table}] WHERE LPNTR=?",
            lpntr
        )

    elif "ENTRY" in cols:

        cur.execute(
            f"SELECT TOP 100 * FROM dbo.[{table}] WHERE ENTRY=?",
            lpntr
        )

    else:
        print("No direct LPNTR/ENTRY key - skipping rows for now.")
        continue

    for row in cur.fetchall():
        print(row)
