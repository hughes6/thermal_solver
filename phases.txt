import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

ti = app._oleobj_.GetTypeInfo()
tlb, _ = ti.GetContainingTypeLib()

for i in range(tlb.GetTypeInfoCount()):
    name = str(tlb.GetDocumentation(i)[0])

    if any(x in name.lower() for x in [
        "dbe",
        "component",
        "wire",
        "cable",
        "connector"
    ]):
        print(i, name)
