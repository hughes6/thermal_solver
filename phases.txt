CABLE = "2627451-500_26S HD TO 104S HD"

cur = conn.cursor()

# Get cable record + internal pointer
cur.execute("""
SELECT ID, ENTRY, VERSION, ENTRYTYP, LPNTR
FROM dbo.ComponentData
WHERE ENTRY = ?
""", CABLE)

rows = cur.fetchall()

print("COMPONENT DATA")
for r in rows:
    print(r)

if not rows:
    raise RuntimeError("Cable not found")

lpntr = rows[0].LPNTR

print("\nLPNTR =", lpntr)


# What parts are inside the cable?
print("\nASSEMBLY MEMBERS")

cur.execute("""
SELECT COPY, FLAGS, NAME, SUBENTRY, SUBVERSION
FROM dbo.Assembly
WHERE LPNTR = ?
ORDER BY COPY
""", lpntr)

for r in cur.fetchall():
    print(r)


# Pin relationships for this exact assembly
print("\nGATE / PIN RELATIONS")

cur.execute("""
SELECT
    GateNumber,
    PinNumber,
    Representation,
    NodeNumber
FROM dbo.GatePinRelation
WHERE ENTRY = ?
ORDER BY GateNumber, PinNumber
""", lpntr)

for r in cur.fetchall():
    print(r)
