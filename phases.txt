import pythoncom
import win32com.client as win32

CLSID = pythoncom.MakeIID(
    "{C1040041-F9E8-4502-803B-2B61FBFDEEF6}"
)

obj = pythoncom.CoCreateInstance(
    CLSID,
    None,
    pythoncom.CLSCTX_LOCAL_SERVER |
    pythoncom.CLSCTX_INPROC_SERVER,
    pythoncom.IID_IDispatch
)

bridge = win32.Dispatch(obj)

print("BRIDGE CREATED:", bridge)

for name in [
    "BringToForeground",
    "New",
    "Edit",
    "NewWithPreconditions",
    "EditWithPreconditions"
]:
    print(name, hasattr(bridge, name))





import inspect

for name in [
    "New",
    "Edit",
    "NewWithPreconditions",
    "EditWithPreconditions"
]:
    m = getattr(bridge, name)

    print("\n", name)
    print("SIG:", inspect.signature(m))

    try:
        print(inspect.getsource(m))
    except Exception as e:
        print("source error:", e)
