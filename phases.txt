import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

ti = app._oleobj_.GetTypeInfo()
typelib, _ = ti.GetContainingTypeLib()

for i in range(typelib.GetTypeInfoCount()):
    name = typelib.GetDocumentation(i)[0]

    if name != "IDbeBridge":
        continue

    info = typelib.GetTypeInfo(i)
    attr = info.GetTypeAttr()

    print("IDbeBridge")

    for j in range(attr[6]):
        try:
            fd = info.GetFuncDesc(j)
            print(info.GetNames(fd[0]))
        except:
            pass
