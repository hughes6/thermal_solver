import time
import pyodbc
import win32com.client as win32

COMPONENT_NAME = "PYTHON_APP_TEST"
COMPONENT_VERSION = "1"

CONNECTION_STRING = (
    "DRIVER={SQL Server};"
    "SERVER=sql-zuken-prd;"
    "DATABASE=Zuken_E3Library;"
    "Trusted_Connection=yes;"
)

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

print("Creating through DBE APPLICATION...")

result = app.NewComponent(COMPONENT_NAME, COMPONENT_VERSION, "", "", 0)
print("app.NewComponent():", result)

input("Finish the wizard, then press ENTER...")

time.sleep(1)

cid = app.GetActiveComponentId()
print("Active ID:", cid)

if cid > 0:
    comp = app.CreateDbeComponentObject()
    comp.SetId(cid)

    print("Name:", comp.GetName())
    print("Version:", comp.GetVersion())
    print("Type:", comp.GetComponentType())
    print("Save:", comp.Save())

time.sleep(2)

conn = pyodbc.connect(CONNECTION_STRING, autocommit=True)
cur = conn.cursor()

cur.execute("""
SELECT ID, ENTRY, VERSION, LPNTR
FROM dbo.ComponentData
WHERE ENTRY = ?
AND VERSION = ?
""", COMPONENT_NAME, COMPONENT_VERSION)

row = cur.fetchone()

print("\nDATABASE ROW:")
print(tuple(row) if row else None)

cur.close()
conn.close()
