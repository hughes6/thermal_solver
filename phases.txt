def set_attribute(name, value):
    cur.execute("""
    SELECT COUNT(*)
    FROM dbo.ComponentAttribute
    WHERE ID = ?
    AND AttributeName = ?
    """, component_db_id, name)

    if cur.fetchone()[0]:
        cur.execute("""
        UPDATE dbo.ComponentAttribute
        SET AttributeValue = ?
        WHERE ID = ?
        AND AttributeName = ?
        """, value, component_db_id, name)
    else:
        cur.execute("""
        INSERT INTO dbo.ComponentAttribute
        (ID, ENTRY, VERSION, BundleID, NODENO, Flags, AttributeName, Copy, AttributeValue)
        VALUES (?, ?, ?, 0, 0, 0, ?, 1, ?)
        """, component_db_id, COMPONENT_NAME, COMPONENT_VERSION, name, value)
