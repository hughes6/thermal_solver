import winreg

for i in range(typelib.GetTypeInfoCount()):
    name = typelib.GetDocumentation(i)[0]

    if name != "e3DbeBridge":
        continue

    info = typelib.GetTypeInfo(i)
    clsid = str(info.GetTypeAttr()[0])

    print("CLSID:", clsid)

    try:
        key = winreg.OpenKey(
            winreg.HKEY_CLASSES_ROOT,
            rf"CLSID\{clsid}\ProgID"
        )

        progid, _ = winreg.QueryValueEx(key, None)

        print("PROGID:", progid)

    except Exception as e:
        print("No ProgID:", e)
