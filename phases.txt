import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

# --------------------------------------------------
# Bind to current DBE job
# --------------------------------------------------

_, raw_ids = app.GetJobIds()

job_ids = [
    x for x in raw_ids
    if isinstance(x, int) and x > 0
]

if not job_ids:
    raise RuntimeError("No open DBE job found.")

jid = job_ids[-1]

job = app.CreateDbeJobObject()
job.SetId(jid)

print("JOB")
print("ID:", job.GetId())
print("Name:", job.GetName())
print("Type:", job.GetType())
print("Path:", job.GetPath())


# --------------------------------------------------
# Create component INSIDE this job
# --------------------------------------------------

print("\nCREATING TEST COMPONENT")

result = job.NewComponent(
    "PYTHON_TEST_DELETE_ME",
    "1",
    "",     # baseName
    "",     # baseVersion
    0       # flags
)

print("NewComponent returned:", repr(result))


# --------------------------------------------------
# Check what DBE created
# --------------------------------------------------

print("\nAFTER NewComponent")

active_id = job.GetActiveComponentId()

print("Active component ID:", active_id)
print("Component IDs:", repr(job.GetComponentIds()))


# --------------------------------------------------
# Bind to it if possible
# --------------------------------------------------

if isinstance(active_id, int) and active_id > 0:

    comp = job.CreateDbeComponentObject()
    comp.SetId(active_id)

    print("\nCOMPONENT")

    print("ID:", comp.GetId())
    print("Name:", repr(comp.GetName()))
    print("Version:", repr(comp.GetVersion()))
    print("Type:", comp.GetComponentType())
    print("Subtype:", comp.GetSubType())

else:

    print("\nNo positive component ID yet.")
