import win32com.client as win32
import inspect
import sys


# ============================================================
# CONNECT
# ============================================================

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

print("=" * 90)
print("ZUKEN DBE ACTIVE COMPONENT BINDING TEST")
print("=" * 90)

print("Application:", app.GetName())
print("Version:", app.GetVersion())
print("Database:", app.GetActualDatabase())


# ============================================================
# HELPERS
# ============================================================

def safe(label, func, *args):
    try:
        result = func(*args)
        print(f"[OK]   {label}: {result!r}")
        return result
    except Exception as e:
        print(f"[FAIL] {label}: {e!r}")
        return None


def dump_component(comp, title="COMPONENT"):
    print()
    print("-" * 90)
    print(title)
    print("-" * 90)

    safe("GetId", comp.GetId)
    safe("GetName", comp.GetName)
    safe("GetVersion", comp.GetVersion)
    safe("GetComponentType", comp.GetComponentType)
    safe("GetSubType", comp.GetSubType)
    safe("GetModelName(0)", comp.GetModelName, 0)


# ============================================================
# 1. GET ACTIVE COMPONENT ID
# ============================================================

print()
print("=" * 90)
print("STEP 1 - ACTIVE ID")
print("=" * 90)

active_id = safe(
    "GetActiveComponentId",
    app.GetActiveComponentId
)

print("Python type:", type(active_id))


if active_id in (None, 0, -1):
    print()
    print("No valid component appears to be active.")
    print("Open/select the test cable manually in DBE and run again.")
    sys.exit(1)


# ============================================================
# 2. CREATE EMPTY DBE COMPONENT OBJECT
# ============================================================

print()
print("=" * 90)
print("STEP 2 - RAW DBE COMPONENT OBJECT")
print("=" * 90)

comp = app.CreateDbeComponentObject()

dump_component(
    comp,
    "BEFORE SetId"
)


# ============================================================
# 3. TRY BINDING OBJECT TO ACTIVE COMPONENT
# ============================================================

print()
print("=" * 90)
print("STEP 3 - SetId(active_id)")
print("=" * 90)

set_result = safe(
    f"SetId({active_id})",
    comp.SetId,
    active_id
)

dump_component(
    comp,
    "AFTER SetId"
)


# ============================================================
# 4. VERIFY AGAINST EXPECTED ACTIVE ID
# ============================================================

bound_id = safe(
    "Bound GetId",
    comp.GetId
)

if bound_id == active_id:
    print()
    print("*** SUCCESS ***")
    print("DbeComponentObject is now bound to the active component.")
else:
    print()
    print("*** OBJECT STILL NOT BOUND ***")


# ============================================================
# 5. GET ALL COMPONENT IDS
# ============================================================

print()
print("=" * 90)
print("STEP 5 - GET COMPONENT IDS")
print("=" * 90)

try:
    ids = app.GetComponentIds()
    print("GetComponentIds():", repr(ids))
except Exception as e:
    print("GetComponentIds() failed:", repr(e))

    # Show generated wrapper documentation
    try:
        print()
        print("GetComponentIds doc:")
        print(app.GetComponentIds.__doc__)
    except Exception:
        pass


# ============================================================
# 6. SHOW GENERATED WRAPPER DOCS
# ============================================================

print()
print("=" * 90)
print("STEP 6 - COM METHOD DOCUMENTATION")
print("=" * 90)

methods = [
    ("Component.SetId", comp.SetId),
    ("Component.GetAttributeIds", comp.GetAttributeIds),
    ("Component.AddAttributeValue", comp.AddAttributeValue),
    ("Application.GetComponentIds", app.GetComponentIds),
    ("Application.GetComponentList", app.GetComponentList),
    ("Application.EditComponent", app.EditComponent),
    ("Application.SelectComponentFromTable", app.SelectComponentFromTable),
]

for name, method in methods:
    print()
    print(name)
    print("-" * len(name))

    try:
        print("DOC:", method.__doc__)
    except Exception as e:
        print("No doc:", repr(e))

    try:
        print("SIGNATURE:", inspect.signature(method))
    except Exception as e:
        print("No Python signature:", repr(e))


# ============================================================
# STOP IF COMPONENT DID NOT BIND
# ============================================================

if bound_id != active_id:
    print()
    print("=" * 90)
    print("STOPPING BEFORE ATTRIBUTE TEST")
    print("=" * 90)

    print(
        "We have not successfully attached the component object "
        "to the real active cable, so attribute results would be meaningless."
    )

    sys.exit(0)


# ============================================================
# 7. NOW THAT COMPONENT IS REAL, INSPECT ATTRIBUTES
# ============================================================

print()
print("=" * 90)
print("STEP 7 - ATTRIBUTE TEST")
print("=" * 90)


def try_attributes(end):
    print()
    print(f"--- END {end} ---")

    attempts = [
        ("GetAttributeIds()", ()),
        ("GetAttributeIds(None,end,'')", (None, end, "")),
        ("GetAttributeIds((),end,'')", ((), end, "")),
    ]

    for label, args in attempts:
        try:
            result = comp.GetAttributeIds(*args)

            print(
                "[SUCCESS]",
                label,
                "=>",
                repr(result)
            )

            return result

        except Exception as e:
            print(
                "[FAIL]",
                label,
                "=>",
                repr(e)
            )

    return None


for end in [-1, 0, 1, 2]:
    try_attributes(end)


# ============================================================
# 8. BASIC CABLE CLASSIFICATION
# ============================================================

print()
print("=" * 90)
print("STEP 8 - COMPONENT CLASSIFICATION")
print("=" * 90)

component_type = safe(
    "Component Type",
    comp.GetComponentType
)

subtype = safe(
    "Subtype",
    comp.GetSubType
)

name = safe(
    "Name",
    comp.GetName
)

version = safe(
    "Version",
    comp.GetVersion
)

print()
print("FINAL:")
print("Name          :", name)
print("Version       :", version)
print("Component Type:", component_type)
print("Subtype       :", subtype)

print()
print("=" * 90)
print("DONE")
print("=" * 90)
