cur.execute("""
SELECT
    TABLE_NAME,
    STRING_AGG(COLUMN_NAME, ', ') AS columns
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'dbo'
GROUP BY TABLE_NAME
HAVING
       MAX(CASE WHEN COLUMN_NAME LIKE '%Pin%'  THEN 1 ELSE 0 END) = 1
    OR MAX(CASE WHEN COLUMN_NAME LIKE '%Node%' THEN 1 ELSE 0 END) = 1
    OR MAX(CASE WHEN COLUMN_NAME LIKE '%Gate%' THEN 1 ELSE 0 END) = 1
    OR MAX(CASE WHEN COLUMN_NAME LIKE '%Conn%' THEN 1 ELSE 0 END) = 1
ORDER BY TABLE_NAME
""")

for row in cur.fetchall():
    print(row)
