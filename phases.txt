import win32com.client as win32

app = win32.gencache.EnsureDispatch("CT.DBEApplication")

# ------------------------------------------------------------
# BIND TO ACTIVE COMPONENT
# ------------------------------------------------------------

cid = app.GetActiveComponentId()

comp = app.CreateDbeComponentObject()
comp.SetId(cid)

print("ID:", comp.GetId())
print("Name:", comp.GetName())
print("Version:", comp.GetVersion())
print("Type:", comp.GetComponentType())
print("Subtype:", comp.GetSubType())

if comp.GetComponentType() != 7:
    raise RuntimeError("Active DBE component is not a cable.")


# ------------------------------------------------------------
# GET ATTRIBUTE IDS
# ------------------------------------------------------------

result = comp.GetAttributeIds()

print("\nFULL RESULT:")
print(repr(result))
print("RESULT TYPE:")
print(type(result))


# ------------------------------------------------------------
# NORMALIZE RETURN
# ------------------------------------------------------------

attribute_ids = []
count = None

if isinstance(result, tuple):

    # Common case:
    # (10, (None, 15375, 15379, ...))
    if len(result) == 2:

        count = result[0]
        raw_ids = result[1]

        if isinstance(raw_ids, (tuple, list)):

            attribute_ids = [
                x for x in raw_ids
                if isinstance(x, int) and x > 0
            ]

        elif isinstance(raw_ids, int):

            # Could be a single returned ID
            if raw_ids > 0:
                attribute_ids = [raw_ids]

        else:

            print(
                "Unexpected raw_ids type:",
                type(raw_ids),
                repr(raw_ids)
            )

    # Sometimes pywin32 may flatten things:
    # (10, None, 15375, 15379, ...)
    elif len(result) > 2:

        count = result[0]

        attribute_ids = [
            x for x in result[1:]
            if isinstance(x, int) and x > 0
        ]

else:

    print(
        "Unexpected GetAttributeIds return:",
        type(result),
        repr(result)
    )


print("\nCOUNT:", count)
print("ATTRIBUTE IDS:", attribute_ids)


# ------------------------------------------------------------
# READ ATTRIBUTES
# ------------------------------------------------------------

print("\n" + "=" * 80)
print("CABLE ATTRIBUTES")
print("=" * 80)

for aid in attribute_ids:

    attr = app.CreateDbeAttributeObject()

    try:
        attr.SetId(aid)
    except Exception as e:
        print(
            "\nCould not bind attribute",
            aid,
            repr(e)
        )
        continue

    print("\nATTRIBUTE ID:", aid)

    getters = [
        ("Name", attr.GetName),
        ("Internal Name", attr.GetInternalName),
        ("Value", attr.GetValue),
        ("Internal Value", attr.GetInternalValue),
        ("Formatted Value", attr.GetFormattedValue),
    ]

    for label, getter in getters:

        try:
            value = getter()
            print(f"  {label:20}: {value!r}")

        except Exception as e:
            print(
                f"  {label:20}: ERROR {e!r}"
            )
