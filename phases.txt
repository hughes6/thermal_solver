def header(text):

    print()

def safe_call(label, func, *args):
    try:
        result = func(*args)

        print(
            f"[OK]   {label}: "
            f"{result!r}"
        )

        return result

    except Exception as exc:

        print(
            f"[FAIL] {label}: "
            f"{exc!r}"
        )

        return None


def usable_ids(raw_result):
    """
    Handle E3 / pywin32 results such as:

        (1, (None, 118))

    or

        (10, (None, 15375, 15379, ...))

    Returns:

        [118]

    or

        [15375, 15379, ...]
    """

    if raw_result is None:
        return []

    # Expected pattern:
    #
    # (count, sequence)

    if (
        isinstance(raw_result, tuple)
        and len(raw_result) == 2
    ):

        raw_values = raw_result[1]

        if isinstance(
            raw_values,
            (tuple, list)
        ):

            return [
                value
                for value in raw_values
                if (
                    isinstance(value, int)
                    and value > 0
                )
            ]

    # Flattened fallback

    if isinstance(
        raw_result,
        (tuple, list)
    ):

        return [
            value
            for value in raw_result
            if (
                isinstance(value, int)
                and value > 0
            )
        ]

    if (
        isinstance(raw_result, int)
        and raw_result > 0
    ):
        return [raw_result]

    return []

def get_active_job():

    header(
        "LOCATING ACTIVE DBE JOB"
    )

    job_count = safe_call(
        "GetJobCount",
        app.GetJobCount
    )

    raw_ids = safe_call(
        "GetJobIds",
        app.GetJobIds
    )

    ids = usable_ids(
        raw_ids
    )

    print(
        "Usable job IDs:",
        ids
    )

    if not ids:

        raise RuntimeError(
            "No DBE job is currently open."
        )

    # Usually only one job exists.
    #
    # If multiple jobs exist, choose the last one for now.

    job_id = ids[-1]

    job = (
        app.CreateDbeJobObject()
    )

    job.SetId(
        job_id
    )

    print()
    print("BOUND JOB")

    safe_call(
        "Job ID",
        job.GetId
    )

    safe_call(
        "Job Name",
        job.GetName
    )

    safe_call(
        "Job Type",
        job.GetType
    )

    safe_call(
        "Job Path",
        job.GetPath
    )

    return job

def bind_active_component(job):

    header(
        "BINDING NEWLY CREATED COMPONENT"
    )

    cid = safe_call(
        "job.GetActiveComponentId",
        job.GetActiveComponentId
    )

    if (
        not isinstance(cid, int)
        or cid <= 0
    ):

        print()

        print(
            "No positive active component ID."
        )

        print(
            "Checking job component IDs..."
        )

        raw = safe_call(
            "job.GetComponentIds",
            job.GetComponentIds
        )

        ids = usable_ids(
            raw
        )

        print(
            "Usable IDs:",
            ids
        )

        if ids:
            cid = ids[-1]

    if (
        not isinstance(cid, int)
        or cid <= 0
    ):

        raise RuntimeError(
            "Could not find the newly created component."
        )

    comp = (
        job.CreateDbeComponentObject()
    )

    comp.SetId(
        cid
    )

    print()
    print(
        "COMPONENT BOUND SUCCESSFULLY"
    )

    return comp

def read_component_attributes(
    comp
):

    header(
        "READING COMPONENT ATTRIBUTES"
    )

    raw_result = safe_call(
        "GetAttributeIds",
        comp.GetAttributeIds
    )

    print(
        "Raw result:",
        repr(raw_result)
    )

    attribute_ids = (
        usable_ids(
            raw_result
        )
    )

    print(
        "Usable attribute IDs:",
        attribute_ids
    )

    attributes = []

    for attribute_id in attribute_ids:

        subheader(
            f"ATTRIBUTE {attribute_id}"
        )

        attr = (
            job.CreateDbeAttributeObject()
        )

        try:

            attr.SetId(
                attribute_id
            )

        except Exception as exc:

            print(
                "Could not bind attribute:",
                repr(exc)
            )

            continue

        data = {
            "id": attribute_id
        }

        getters = [
            (
                "name",
                attr.GetName
            ),
            (
                "internal_name",
                attr.GetInternalName
            ),
            (
                "value",
                attr.GetValue
            ),
            (
                "internal_value",
                attr.GetInternalValue
            ),
            (
                "formatted_value",
                attr.GetFormattedValue
            ),
        ]

        for key, getter in getters:

            try:
                data[key] = getter()

            except Exception as exc:
                data[key] = (
                    f"<ERROR {exc!r}>"
                )

        pprint(
            data
        )

        attributes.append(
            data
        )

    return attributes

def inspect_job_output(
    job,
    limit=20
):

    header(
        "JOB OUTPUT / RESULT WINDOW"
    )

    for index in range(limit):

        for label, method in [
            (
                "OUTBAR",
                job.GetOutbarText
            ),
            (
                "RESULT",
                job.GetResultText
            ),
        ]:

            try:

                result = method(
                    index,
                    ""
                )

                if result not in (
                    None,
                    "",
                    0,
                    -1
                ):

                    print(
                        label,
                        index,
                        repr(result)
                    )

            except Exception:
                pass


inspect_job_output(
    job
)
